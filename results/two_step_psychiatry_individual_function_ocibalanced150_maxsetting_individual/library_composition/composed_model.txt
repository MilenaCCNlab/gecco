def cognitive_model(action_1, state, action_2, reward, model_parameters):
    """
    Composed cognitive model: backbone + Direct Model-Free Update, Reward-Dependent Stickiness Stage 1, Value Decay and Init to Center
    
    Parameters:
    learning_rate: [0, 1] - TD learning rate.
    beta: [0, 10] - Softmax inverse temperature.
    p_win_1: [-5, 5] - module parameter
    p_lose_1: [-5, 5] - module parameter
    decay_rate_center: [0, 1] - module parameter
    center: [0, 1] - module parameter
    """
    learning_rate, beta, p_win_1, p_lose_1, decay_rate_center, center = model_parameters
    n_trials = len(action_1)
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    log_loss = 0.0
    eps = 1e-10
    reward_dependent_stickiness_stage1_last_a1 = -1
    reward_dependent_stickiness_stage1_last_r = 0.0
    q_stage1_mf = np.full(2, center)
    q_stage2_mf = np.full((2, 2), center)

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s_idx = int(state[trial])
        a2 = int(action_2[trial])
        r = float(reward[trial])

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        stage1_values = q_stage1_mb
        logits_1 = (beta) * stage1_values
        if reward_dependent_stickiness_stage1_last_a1 != -1:
            logits_1[reward_dependent_stickiness_stage1_last_a1] += beta * (p_win_1 if reward_dependent_stickiness_stage1_last_r > 0 else p_lose_1)

        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        if a1 != -1:
            log_loss -= np.log(probs_1[a1] + eps)

        if s_idx != -1 and a2 != -1:
            stage2_values = q_stage2_mf[s_idx]
            logits_2 = (beta) * stage2_values
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            log_loss -= np.log(probs_2[a2] + eps)

        if a1 != -1 and s_idx != -1 and a2 != -1:
            delta_stage1 = q_stage2_mf[s_idx, a2] - q_stage1_mf[a1]
            delta_stage1 = r - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1
            delta_stage2 = r - q_stage2_mf[s_idx, a2]
            q_stage2_mf[s_idx, a2] += learning_rate * delta_stage2
        if a1 != -1:
            reward_dependent_stickiness_stage1_last_a1 = a1
            reward_dependent_stickiness_stage1_last_r = r
        if a1 != -1:
            q_stage1_mf[1 - a1] += decay_rate_center * (center - q_stage1_mf[1 - a1])
        if s_idx != -1 and a2 != -1:
            for value_decay_and_init_center_s in range(2):
                for value_decay_and_init_center_a in range(2):
                    if not (value_decay_and_init_center_s == s_idx and value_decay_and_init_center_a == a2):
                        q_stage2_mf[value_decay_and_init_center_s, value_decay_and_init_center_a] += decay_rate_center * (center - q_stage2_mf[value_decay_and_init_center_s, value_decay_and_init_center_a])

    return log_loss
