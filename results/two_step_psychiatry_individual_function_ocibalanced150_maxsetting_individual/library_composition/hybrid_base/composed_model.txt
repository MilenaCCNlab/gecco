def cognitive_model(action_1, state, action_2, reward, model_parameters):
    """
    Composed cognitive model: backbone + Eligibility Trace, MB/MF Mixture Weight, Separate Learning Rates, Separate Stage 2 Inverse Temperature, Stage 1 Choice Stickiness, Value Decay to 0.5
    
    Parameters:
    learning_rate: [0, 1] - TD learning rate.
    beta: [0, 10] - Softmax inverse temperature.
    lam: [0, 1] - module parameter
    w: [0, 1] - module parameter
    alpha_1: [0, 1] - module parameter
    alpha_2: [0, 1] - module parameter
    beta_2: [0, 10] - module parameter
    stickiness_1: [-5, 5] - module parameter
    decay_rate_half: [0, 1] - module parameter
    """
    learning_rate, beta, lam, w, alpha_1, alpha_2, beta_2, stickiness_1, decay_rate_half = model_parameters
    n_trials = len(action_1)
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    log_loss = 0.0
    eps = 1e-10
    stage1_stickiness_last_a1 = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s_idx = int(state[trial])
        a2 = int(action_2[trial])
        r = float(reward[trial])

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        stage1_values = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = (beta) * stage1_values
        if stage1_stickiness_last_a1 != -1:
            logits_1[stage1_stickiness_last_a1] += beta * stickiness_1

        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        if a1 != -1:
            log_loss -= np.log(probs_1[a1] + eps)

        if s_idx != -1 and a2 != -1:
            stage2_values = q_stage2_mf[s_idx]
            logits_2 = (beta_2) * stage2_values
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            log_loss -= np.log(probs_2[a2] + eps)

        if a1 != -1 and s_idx != -1 and a2 != -1:
            delta_stage1 = q_stage2_mf[s_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += alpha_1 * delta_stage1
            delta_stage2 = r - q_stage2_mf[s_idx, a2]
            q_stage2_mf[s_idx, a2] += alpha_2 * delta_stage2
            q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        if a1 != -1:
            stage1_stickiness_last_a1 = a1
        if a1 != -1:
            q_stage1_mf[1 - a1] += decay_rate_half * (0.5 - q_stage1_mf[1 - a1])
        if s_idx != -1 and a2 != -1:
            for value_decay_to_half_s in range(2):
                for value_decay_to_half_a in range(2):
                    if not (value_decay_to_half_s == s_idx and value_decay_to_half_a == a2):
                        q_stage2_mf[value_decay_to_half_s, value_decay_to_half_a] += decay_rate_half * (0.5 - q_stage2_mf[value_decay_to_half_s, value_decay_to_half_a])

    return log_loss
