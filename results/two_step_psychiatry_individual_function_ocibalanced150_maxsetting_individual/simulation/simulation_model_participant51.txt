import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Model 2: Loss Aversion with Unchosen Value Decay.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta1, beta2, w, lam, p1, decay, loss_aversion]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha, beta1, beta2, w, lam, p1, decay, loss_aversion = parameters

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Determine reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State transition
        state_idx = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward realization
        r_raw = int(np.random.random() < reward_probs[state_idx][a2])
        r = 1.0 if r_raw == 1 else -loss_aversion

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lam * delta_stage2

        # Unchosen value decay towards 0.0
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] += decay * (0.0 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - a2
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.0 - q_stage2_mf[state_idx, unchosen_2])
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay * (0.0 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.0 - q_stage2_mf[unvisited_state, 1])

        # Update last action for perseveration
        last_action_1 = a1

        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = state_idx
        stage2_choice[trial] = a2
        reward[trial] = r_raw

    return stage1_choice, state2, stage2_choice, reward
