import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using the Hybrid RL Model with a forgetting mechanism 
    where unchosen Q-values decay towards a neutral value (0.5).

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta1, beta2, w, lam, pers1, pers2, forget_rate]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta1, beta2, w, lam, pers1, pers2, forget_rate = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = -1

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for t in range(n_trials):
        # Current trial's reward probabilities
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]

        # --- Stage 1 ---
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers1_bonus[last_action_1] = pers1
            
        q_stage1_total = beta1 * q_stage1_net + pers1_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # --- Stage 2 ---
        pers2_bonus = np.zeros(2)
        if last_action_2 != -1:
            pers2_bonus[last_action_2] = pers2
            
        q_stage2_total = beta2 * q_stage2_mf[s2] + pers2_bonus
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])

        # --- Learning (Update Q-values) ---
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2

        # --- Forgetting ---
        for a in range(2):
            if a != a1:
                q_stage1_mf[a] += forget_rate * (0.5 - q_stage1_mf[a])
                
        for s in range(2):
            for a in range(2):
                if s != s2 or a != a2:
                    q_stage2_mf[s, a] += forget_rate * (0.5 - q_stage2_mf[s, a])
        
        # Update last actions
        last_action_1 = a1
        last_action_2 = a2

        # Record data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

    return stage1_choice, state2, stage2_choice, reward
