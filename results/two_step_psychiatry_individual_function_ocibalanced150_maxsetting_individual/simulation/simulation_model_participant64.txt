import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using the MB/MF hybrid model with choice stickiness 
    (perseveration) for stage 1 and a decay (forgetting) parameter for unchosen action values.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta, w, lam, p, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha, beta, w, lam, p, decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]
        
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        act1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        state_idx = np.random.choice([0, 1], p=transition_matrix[act1])
        
        # Stage 2 choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        act2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[state_idx][act2])
        
        # Learning and update
        delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
        q_stage1_mf[act1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, act2]
        q_stage2_mf[state_idx, act2] += alpha * delta_stage2

        q_stage1_mf[act1] += alpha * lam * delta_stage2

        # Decay for unchosen action in stage 1
        unchosen_act1 = 1 - act1
        q_stage1_mf[unchosen_act1] *= (1 - decay)

        # Decay for unchosen actions in stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == act2):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action_1 = act1
        
        # Store data
        stage1_choice[trial] = act1
        state2[trial] = state_idx
        stage2_choice[trial] = act2
        reward[trial] = r
        
    return stage1_choice, state2, stage2_choice, reward
