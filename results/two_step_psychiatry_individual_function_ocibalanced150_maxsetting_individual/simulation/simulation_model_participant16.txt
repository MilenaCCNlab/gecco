import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using Model 3: Subjective Penalty for Zero Reward.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, decay, beta_1, beta_2, w, lam, pers, penalty_zero]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha, decay, beta_1, beta_2, w, lam, pers, penalty_zero = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action_1 = -1

    for t in range(n_trials):
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]
        
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + pers * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2
        
        # Stage 2 Choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # Stage 2 MF update with Subjective Penalty for Zero Reward
        subj_reward = 1.0 if r == 1 else -penalty_zero
        delta_stage2 = subj_reward - q_stage2_mf[state_idx, a2]
        
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Eligibility trace update for Stage 1 MF
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Decay for unchosen options
        unchosen_action_1 = 1 - a1
        q_stage1_mf[unchosen_action_1] *= decay
        
        unchosen_action_2 = 1 - a2
        q_stage2_mf[state_idx, unchosen_action_2] *= decay
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= decay
        q_stage2_mf[other_state, 1] *= decay
        
        # Update last action
        last_action_1 = a1
        
        # Store trial data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

    return stage1_choice, state2, stage2_choice, reward
