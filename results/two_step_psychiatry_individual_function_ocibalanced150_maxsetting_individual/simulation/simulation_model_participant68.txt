import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Model 2: Separate learning rates for Stage 1 
    and Stage 2 with Q-value decay and perseveration.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha1, alpha2, beta, w, lam, pers, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha1, alpha2, beta, w, lam, pers, decay = parameters
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]

        # --- Stage 1 Choice ---
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # --- Transition to Stage 2 ---
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2

        # --- Stage 2 Choice ---
        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)

        # --- Reward ---
        r = int(np.random.random() < reward_probs[s2][a2])

        # --- Learning & Updating Q-values ---
        # 1. Decay Q-values
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf

        # 2. Stage 1 TD error & update
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        # 3. Stage 2 TD error & update
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2

        # 4. Eligibility trace update for Stage 1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        # Update last actions
        last_action1 = a1
        last_action2[state_idx] = a2

        # Store data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
