import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using a model with an exploration bonus (UCB-like) 
    for Stage 1 choices, decay, and perseveration.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, decay, beta_mb, beta_mf, beta_2, lambd, pers_1, explore_1]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, decay, beta_mb, beta_mf, beta_2, lambd, pers_1, explore_1 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    time_since_chosen_1 = np.zeros(2)

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for t in range(n_trials):
        # Drift reward probabilities for the current trial
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]

        # Stage 1 model-based Q-values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Perseveration bonus
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers_1
            
        # Stage 1 choice probabilities
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus_1 + explore_1 * np.sqrt(time_since_chosen_1)
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2 state
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 choice probabilities
        logits_2 = beta_2 * q_stage2_mf[s2]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Observe reward
        r = int(np.random.random() < reward_probs[s2][a2])

        # Update Q-values
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += alpha * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[a1] += alpha * lambd * delta_stage2

        # Decay unchosen options towards 0.5
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - a2
        q_stage2_mf[s2, unchosen_2] += decay * (0.5 - q_stage2_mf[s2, unchosen_2])
        
        unvisited_state = 1 - s2
        q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
        
        # Update trackers
        last_a1 = a1
        time_since_chosen_1 += 1
        time_since_chosen_1[a1] = 0

        # Store trial data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

    return stage1_choice, state2, stage2_choice, reward
