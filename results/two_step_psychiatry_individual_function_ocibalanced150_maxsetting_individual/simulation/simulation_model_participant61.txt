import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using Model 3: Two-step model with separate 
    perseveration parameters for stage 1 and stage 2, and unchosen Q-value decay.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta_1, beta_2, w, lam, p_1, p_2, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2, decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    for t in range(n_trials):
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]
        
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            persev_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p_1 * persev_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2
        
        # Stage 2
        persev_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            persev_2[last_action_2[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2 * persev_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Unchosen Q-value decay
        for a in range(2):
            if a != a1:
                q_stage1_mf[a] *= (1 - decay)
                
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != a2:
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        # Store choices for perseveration
        last_action_1 = a1
        last_action_2[state_idx] = a2
        
        # Record data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r
        
    return stage1_choice, state2, stage2_choice, reward
