import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the hybrid model-based/model-free 2-step model
    with reward-dependent stickiness, separate inverse temperatures, and Q-value decay.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta_1, beta_2, w, lambda_, stick_rew, stick_unrew, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta_1, beta_2, w, lambda_, stick_rew, stick_unrew, decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action_1 = -1
    last_reward = 0

    for t in range(n_trials):
        # Drift reward probs
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]

        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            if last_reward == 1:
                q_stage1_net[last_action_1] += stick_rew / (beta_1 + 1e-10)
            else:
                q_stage1_net[last_action_1] += stick_unrew / (beta_1 + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta_1 * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State transition
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2

        # Stage 2
        max_q2 = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta_2 * (q_stage2_mf[state_idx] - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[state_idx][a2])

        # Learning & Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        # Log choices and rewards
        last_action_1 = a1
        last_reward = r
        
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

    return stage1_choice, state2, stage2_choice, reward
