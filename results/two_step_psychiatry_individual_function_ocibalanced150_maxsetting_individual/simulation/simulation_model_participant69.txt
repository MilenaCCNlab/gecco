import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the hybrid model-based/model-free 2-step model
    with eligibility traces, stage 1 and stage 2 perseveration, a single inverse temperature, 
    and dynamic learning of the transition matrix.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta, w, lam, p_1, p_2, alpha_t]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta, w, lam, p_1, p_2, alpha_t = parameters
    
    # The true transition matrix dictating the environment's actual transitions
    true_transition_matrix = np.array([[0.7, 0.3],
                                       [0.3, 0.7]])
    
    # The agent's learned transition matrix, initialized at 0.5
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p_1 * pers_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State transition based on true transition probabilities
        s2 = np.random.choice([0, 1], p=true_transition_matrix[a1])
        state_idx = s2
        
        # Stage 2 choice
        pers_2 = np.zeros(2)
        if last_action_2 != -1:
            pers_2[last_action_2] = 1

        logits_2 = beta * q_stage2_mf[state_idx] + p_2 * pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward generation
        r = int(np.random.random() < reward_probs[state_idx][a2])
        
        # Learning / Updating Q-values
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2

        # Updating the agent's transition matrix belief
        transition_matrix[a1, state_idx] += alpha_t * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_t * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        # Record keeping
        last_action_1 = a1
        last_action_2 = a2
        
        stage1_choice[trial] = a1
        state2[trial] = state_idx
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
