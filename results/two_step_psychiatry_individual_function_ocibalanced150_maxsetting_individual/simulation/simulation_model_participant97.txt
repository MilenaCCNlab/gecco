import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Cognitive Model 2: Q-value Decay 
    and Shared Learning Rate Choice Kernel.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, decay, beta1, beta2, w, lambda_, ck_weight1, ck_weight2]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    learning_rate, decay, beta1, beta2, w, lambda_, ck_weight1, ck_weight2 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))
    
    ck_lr = learning_rate
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    for t in range(n_trials):
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]
        
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf) + ck_weight1 * ck_1
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State transition
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2 choice
        V2 = beta2 * q_stage2_mf[s2] + ck_weight2 * ck_2[s2]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Store data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r
        
        # Q-value updates
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lambda_ * delta_stage2
        
        # Choice kernel updates
        ck_1 = (1 - ck_lr) * ck_1
        ck_1[a1] += ck_lr
        
        ck_2[s2] = (1 - ck_lr) * ck_2[s2]
        ck_2[s2, a2] += ck_lr
        
        other_state = 1 - s2
        ck_2[other_state] = (1 - ck_lr) * ck_2[other_state]
        
        # Decay for unchosen Q-values
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1]
        
        unchosen_2 = 1 - a2
        q_stage2_mf[s2, unchosen_2] = (1 - decay) * q_stage2_mf[s2, unchosen_2]
        q_stage2_mf[other_state, 0] = (1 - decay) * q_stage2_mf[other_state, 0]
        q_stage2_mf[other_state, 1] = (1 - decay) * q_stage2_mf[other_state, 1]

    return stage1_choice, state2, stage2_choice, reward
