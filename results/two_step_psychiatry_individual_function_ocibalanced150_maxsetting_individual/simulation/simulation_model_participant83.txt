import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using Model 3: Asymmetric learning rates for positive 
    and negative prediction errors at both stages, with a shared inverse temperature 
    and stage 1 perseveration.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha1_pos, alpha1_neg, alpha2_pos, alpha2_neg, beta, w, lambda_, pi1]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha1_pos, alpha1_neg, alpha2_pos, alpha2_neg, beta, w, lambda_, pi1 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Current reward probabilities for this trial
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * net_q1
        if prev_action_1 != -1:
            logits_1[prev_action_1] += pi1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2

        # Stage 2 choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)
  
        # Reward generation
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Prediction errors and learning
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha1_st1 = alpha1_pos if delta_stage1 > 0 else alpha1_neg
        q_stage1_mf[a1] += alpha1_st1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha2 = alpha2_pos if delta_stage2 > 0 else alpha2_neg
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        alpha1_st2 = alpha1_pos if delta_stage2 > 0 else alpha1_neg
        q_stage1_mf[a1] += alpha1_st2 * lambda_ * delta_stage2
        
        # Store variables for next iteration and output
        prev_action_1 = a1
        
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
