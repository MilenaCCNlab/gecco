import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using Model 2: Reward-Dependent Perseveration + Value Decay.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta1, beta2, w, lam, pers_win, pers_lose, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    # Unpack parameters
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    # Initialize Q-values
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Initialize output arrays
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    # Track previous trial's action and reward for perseveration
    prev_a1 = -1
    prev_r = -1
    
    for t in range(n_trials):
        # Current trial's reward probabilities
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]
        
        # Stage 1: Compute model-based and model-free Q-values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Calculate perseveration based on previous trial's reward
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_r == 1:
                perseveration_1[prev_a1] = pers_win
            else:
                perseveration_1[prev_a1] = pers_lose
                
        # Stage 1 Choice
        logits_1 = beta1 * q_stage1_net + perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Determine Stage 2 State
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2 Choice
        logits_2 = beta2 * q_stage2_mf[s2]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Determine Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Decay unchosen options towards 0.5
        q_stage1_mf[1 - a1] += decay * (0.5 - q_stage1_mf[1 - a1])
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
                    
        # Update chosen options (Learning)
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += alpha * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update previous action and reward for next trial
        prev_a1 = a1
        prev_r = r
        
        # Store trial results
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r
        
    return stage1_choice, state2, stage2_choice, reward
