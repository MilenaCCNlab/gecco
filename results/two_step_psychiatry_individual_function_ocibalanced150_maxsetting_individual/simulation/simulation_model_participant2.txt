import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the Model 3: Daw et al. Hybrid Model 
    with both Global and State-Specific Stage 2 Perseveration.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta_1, beta_2, w, lam, pers_1, pers_2_global, pers_2_state]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2_global, pers_2_state = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2_global = -1
    last_a2_in_state = np.full(2, -1)
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]
        
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if prev_a1 != -1:
            pers_bonus_1[prev_a1] = pers_1
            
        logits_1 = beta_1 * q_stage1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition
        state_idx = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2
        pers_bonus_2 = np.zeros(2)
        if prev_a2_global != -1:
            pers_bonus_2[prev_a2_global] += pers_2_global
        if last_a2_in_state[state_idx] != -1:
            pers_bonus_2[last_a2_in_state[state_idx]] += pers_2_state
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[state_idx][a2])
        
        # Learning / Q-value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update perseveration variables
        prev_a1 = a1
        prev_a2_global = a2
        last_a2_in_state[state_idx] = a2
        
        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = state_idx
        stage2_choice[trial] = a2
        reward[trial] = r
        
    return stage1_choice, state2, stage2_choice, reward
