import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using a Hybrid Model-Free/Model-Based RL model
    with separate perseveration and decay rates for stage 1 and stage 2.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta, w, lam, pers_1, pers_2, decay_1, decay_2]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha, beta, w, lam, pers_1, pers_2, decay_1, decay_2 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta * q_1_total - np.max(beta * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        state_idx = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2 Choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2
            
        q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
        exp_q2 = np.exp(beta * q_2_total - np.max(beta * q_2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.rand() < reward_probs[state_idx][a2])
        
        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Decay Updates
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay_1 * (0.5 - q_stage1_mf[unchosen_a1])
        
        unchosen_a2 = 1 - a2
        q_stage2_mf[state_idx, unchosen_a2] += decay_2 * (0.5 - q_stage2_mf[state_idx, unchosen_a2])

        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay_2 * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay_2 * (0.5 - q_stage2_mf[unvisited_state, 1])

        # Track perseveration variables
        last_action_2[state_idx] = a2
        last_action_1 = a1
        
        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = state_idx
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
