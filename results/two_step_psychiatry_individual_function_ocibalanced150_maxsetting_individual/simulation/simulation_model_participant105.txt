import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using the Model 1: Model-based and Model-free mixture 
    with eligibility trace.
    
    Parameters:
    n_trials (int): Number of trials to simulate.
    parameters (list): [alpha, beta, w, lam]
        alpha: learning rate [0,1]
        beta: inverse temperature for softmax choice [0,10]
        w: weight of model-based vs model-free control [0,1]
        lam: eligibility trace for updating stage 1 action values from stage 2 rewards [0,1]
    drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities for each state-action pair.
    
    Returns:
    stage1_choice (np.ndarray): First-stage choices (0 or 1).
    state2 (np.ndarray): Second-stage states (0 or 1).
    stage2_choice (np.ndarray): Second-stage choices (0 or 1).
    reward (np.ndarray): Rewards (0 or 1).
    """
    alpha, beta, w, lam = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_max = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - q1_max))
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2
        
        # Stage 2 Choice
        q2_max = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - q2_max))
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward Realization
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]
        r = int(np.random.random() < reward_probs[state_idx][a2])
        
        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Store Data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
