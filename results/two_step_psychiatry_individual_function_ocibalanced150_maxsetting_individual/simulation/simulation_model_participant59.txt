import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the hybrid model-based/model-free 2-step model
    with eligibility traces, reward-dependent Stage 1 perseveration, and Stage 1 bias.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta_1, beta_2, w, lam, p_win, p_lose, bias]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta_1, beta_2, w, lam, p_win, p_lose, bias = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # Trial-wise reward probabilities for state-action pairs
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1: Compute model-based and model-free Q-values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply reward-dependent perseveration (Win-stay/Lose-shift)
        if last_action_1 != -1:
            if last_reward == 1:
                q_stage1_net[last_action_1] += p_win
            elif last_reward == 0:
                q_stage1_net[last_action_1] += p_lose
                
        # Apply Stage 1 bias towards action 0
        q_stage1_net[0] += bias
            
        # Stage 1 choice probabilities
        q1_scaled = beta_1 * q_stage1_net
        q1_scaled -= np.max(q1_scaled) # for numerical stability
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # Sample Stage 1 choice
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Determine Stage 2 state based on transition probabilities
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 choice probabilities
        q_stage2_net = np.copy(q_stage2_mf[s2])
        q2_scaled = beta_2 * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        # Sample Stage 2 choice
        a2 = np.random.choice([0, 1], p=probs_2)

        # Generate reward
        r = int(np.random.random() < reward_probs[s2][a2])

        # Value Updates
        # 1. Update Stage 1 MF Q-values with Stage 2 state arrival
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # 2. Update Stage 2 and Stage 1 MF Q-values with reward (eligibility trace)
        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Store for next trial's perseveration
        last_action_1 = a1
        last_reward = r

        # Save data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
