import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Cognitive Model 2: Daw et al. model with asymmetric learning rates, 
    memory decay, and outcome-dependent stickiness (win-stay vs lose-stay) for Stage 1. 
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha_pos, alpha_neg, decay_rate, beta_1, beta_2, w, stickiness_win, stickiness_lose]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha_pos, alpha_neg, decay_rate, beta_1, beta_2, w, stickiness_win, stickiness_lose = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1
    last_reward = 0

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Current trial's reward probabilities
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 Model-Based and Model-Free Integration
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Outcome-dependent stickiness
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                stickiness_bonus[last_action_1] = stickiness_win
            else:
                stickiness_bonus[last_action_1] = stickiness_lose
            
        # Stage 1 Choice
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)

        # Transition to State 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 Choice
        q2_total = beta_2 * q_stage2_mf[s2]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward Generation
        r = int(np.random.rand() < reward_probs[s2][a2])

        # Value Decay
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        # Stage 1 Update
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += lr1 * delta_stage1

        # Stage 2 Update
        delta_stage2 = r - q_stage2_mf[s2, a2]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s2, a2] += lr2 * delta_stage2
        
        # Update last actions and rewards for next trial
        last_action_1 = a1
        last_reward = r
        
        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
