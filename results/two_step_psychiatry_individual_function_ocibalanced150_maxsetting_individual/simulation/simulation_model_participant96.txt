import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the model with streak-based perseveration,
    separate learning rates for positive and negative prediction errors, and 
    unchosen option decay.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha_pos, alpha_neg, beta, w, pers_1_streak, pers_2, decay_rate]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha_pos, alpha_neg, beta, w, pers_1_streak, pers_2, decay_rate = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    streak_1 = 0
    last_action_2 = -np.ones(2, dtype=int)

    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and streak_1 > 0:
            pers_bonus_1[last_action_1] = pers_1_streak * np.log(streak_1 + 1)
            
        q1_total = beta * q_stage1_net + pers_bonus_1
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2 choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[s2] != -1:
            pers_bonus_2[last_action_2[s2]] = pers_2
            
        q2_total = beta * q_stage2_mf[s2] + pers_bonus_2
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Learning updates
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[a1] += alpha_1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[s2, a2]
        alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[s2, a2] += alpha_2 * delta_stage2

        # Decay for unchosen options
        unchosen_action_2 = 1 - a2
        q_stage2_mf[s2, unchosen_action_2] = (1 - decay_rate) * q_stage2_mf[s2, unchosen_action_2] + decay_rate * 0.5
        
        other_state = 1 - s2
        q_stage2_mf[other_state, 0] = (1 - decay_rate) * q_stage2_mf[other_state, 0] + decay_rate * 0.5
        q_stage2_mf[other_state, 1] = (1 - decay_rate) * q_stage2_mf[other_state, 1] + decay_rate * 0.5

        # Update perseveration trackers
        if a1 == last_action_1:
            streak_1 += 1
        else:
            streak_1 = 1
            
        last_action_1 = a1
        last_action_2[s2] = a2
        
        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
