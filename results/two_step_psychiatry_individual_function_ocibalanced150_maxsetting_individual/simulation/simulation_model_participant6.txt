import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using the Hybrid Model with Single Temperature 
    and Reward-Modulated Stickiness for Both Stages.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta, w, p_1_win, p_1_lose, p_2_win, p_2_lose, lam]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha, beta, w, p_1_win, p_1_lose, p_2_win, p_2_lose, lam = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Determine reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_1[last_action_1] = p_1_win
            elif last_reward == 0:
                stickiness_1[last_action_1] = p_1_lose
            
        logits_1 = beta * q_stage1_net + stickiness_1
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)

        # Transition to Stage 2
        s = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 choice
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            if last_reward == 1:
                stickiness_2[last_action_2] = p_2_win
            elif last_reward == 0:
                stickiness_2[last_action_2] = p_2_lose
            
        logits_2 = beta * q_stage2_mf[s] + stickiness_2
        logits_2 -= np.max(logits_2) # for numerical stability
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)

        # Observe Reward
        r = int(np.random.random() < reward_probs[s][a2])

        # Learning (Updating Values)
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Keep track of previous events for stickiness
        last_action_1 = a1
        last_action_2 = a2
        last_reward = r
        
        # Store data
        stage1_choice[trial] = a1
        state2[trial] = s
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
