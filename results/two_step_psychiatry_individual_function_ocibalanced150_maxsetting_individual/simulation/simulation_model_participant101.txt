import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the Q-learning for Stage 1 Model-Free values 
    with Independent Forgetting model.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, decay1, decay2, beta1, beta2, w_mb, lam, p_stick]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    learning_rate, decay1, decay2, beta1, beta2, w_mb, lam, p_stick = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    last_action_1 = -1

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w_mb * q_stage1_mb + (1 - w_mb) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p_stick
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)

        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[s2])
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward realization
        r = int(np.random.rand() < reward_probs[s2][a2])

        # Prediction errors
        delta_stage1 = np.max(q_stage2_mf[s2]) - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s2, a2]

        # Stage 1 Updates & Forgetting
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay1 * (0.5 - q_stage1_mf[unchosen_a1])
        q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2

        # Stage 2 Updates & Forgetting
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] += decay2 * (0.5 - q_stage2_mf[s, a])

        q_stage2_mf[s2, a2] += learning_rate * delta_stage2
        
        last_action_1 = a1

        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
