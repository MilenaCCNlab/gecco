import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Model 1: Dual Temperature Model.
    Separates the inverse temperature for model-free and model-based systems at stage 1.
    Includes biases for action 0 at both stages, perseveration at stage 1, and eligibility trace.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, beta_mf, beta_mb, beta2, p, lambd, bias1, bias2]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    
    learning_rate, beta_mf, beta_mb, beta2, p, lambd, bias1, bias2 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]

        # Compute Model-Based Q-values for Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Perseveration component
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        action_bias1 = np.array([bias1, 0.0])
            
        # Stage 1 choice probabilities
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * pers + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # Make Stage 1 choice
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2 state
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        action_bias2 = np.array([bias2, 0.0])
        
        # Stage 2 choice probabilities
        logits_2 = beta2 * q_stage2_mf[s2] + action_bias2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        # Make Stage 2 choice
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Observe reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Update Q-values
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lambd * delta_stage2
        
        last_action_1 = a1
        
        # Store data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
