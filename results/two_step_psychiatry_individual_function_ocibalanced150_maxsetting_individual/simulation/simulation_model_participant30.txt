import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using Model 2: Q-learning with outcome-dependent 
    perseveration at both stages.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, beta, w, p_stage1_rew, p_stage1_unrew, 
                            p_stage2_rew, p_stage2_unrew, lam]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    learning_rate, beta, w, p_stage1_rew, p_stage1_unrew, p_stage2_rew, p_stage2_unrew, lam = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_a1 = -1
    last_a2 = -1
    last_s = -1
    last_r = -1

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        val_1 = beta * q_net
        if last_a1 != -1 and last_r != -1:
            if last_r == 1:
                val_1[last_a1] += p_stage1_rew
            else:
                val_1[last_a1] += p_stage1_unrew
        
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)

        # Transition to Stage 2
        s = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 choice
        val_2 = beta * q_stage2_mf[s]
        if last_s == s and last_a2 != -1 and last_r != -1:
            if last_r == 1:
                val_2[last_a2] += p_stage2_rew
            else:
                val_2[last_a2] += p_stage2_unrew
                
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward
        r = int(np.random.random() < reward_probs[s][a2])

        # Update Q-values
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        # Update history
        last_a1 = a1
        last_s = s
        last_a2 = a2
        last_r = r

        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
