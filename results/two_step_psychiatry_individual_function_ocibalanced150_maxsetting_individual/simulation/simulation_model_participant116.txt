import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the Exploration Bonus + Value Forgetting model.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, beta, w, lam, st, expl_bonus, decay, q0]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    
    learning_rate, beta, w, lam, st, expl_bonus, decay, q0 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2) + q0
    q_stage2_mf = np.zeros((2, 2)) + q0
    
    trials_since_1 = np.zeros(2)
    trials_since_2 = np.zeros((2, 2))
    
    last_action = -1
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]
        
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_total = q_net + expl_bonus * np.sqrt(trials_since_1)
        
        # Softmax for Stage 1
        exp_q1 = np.exp(beta * q1_total + st * stickiness - np.max(beta * q1_total + st * stickiness))
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2
        
        # Stage 2 choice
        q2_total = q_stage2_mf[state_idx] + expl_bonus * np.sqrt(trials_since_2[state_idx])
        
        # Softmax for Stage 2
        exp_q2 = np.exp(beta * q2_total - np.max(beta * q2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Q-value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Decay unchosen actions
        unchosen_action_1 = 1 - a1
        q_stage1_mf[unchosen_action_1] = q_stage1_mf[unchosen_action_1] * (1 - decay) + q0 * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + q0 * decay
                    
        # Update exploration counters
        trials_since_1 += 1
        trials_since_1[a1] = 0
        
        trials_since_2 += 1
        trials_since_2[state_idx, a2] = 0
                    
        last_action = a1
        
        # Store data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r
        
    return stage1_choice, state2, stage2_choice, reward
