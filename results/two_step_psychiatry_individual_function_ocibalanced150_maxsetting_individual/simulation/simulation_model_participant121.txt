import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the Daw et al. (2011) model where 
    unchosen actions decay towards a specific center value.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, beta, w, lambda_, p, decay, decay_center]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    learning_rate, beta, w, lambda_, p, decay, decay_center = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.full(2, decay_center, dtype=float)
    q_stage2_mf = np.full((2, 2), decay_center, dtype=float)
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State transition
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 choice
        exp_q2 = np.exp(beta * q_stage2_mf[s2] - np.max(beta * q_stage2_mf[s2]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)
  
        # Reward generation
        r = int(np.random.random() < reward_probs[s2][a2])

        # Update Q-values
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += learning_rate * delta_stage2
        
        q_stage1_mf[a1] += learning_rate * lambda_ * delta_stage2

        # Decay for unchosen actions
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + decay_center * decay

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = q_stage1_mf[unchosen_a1] * (1 - decay) + decay_center * decay
        
        last_action_1 = a1

        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
