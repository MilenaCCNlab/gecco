import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Model 3: Reward-dependent stickiness 
    (Win-Stay / Lose-Shift) for stage 1, with unchosen Q-values decaying to a center.
    Uses a single inverse temperature for both stages.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta, w, lam, st_rew, st_unrew, decay, center]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta, w, lam, st_rew, st_unrew, decay, center = parameters
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.full(2, float(center))
    q_stage2_mf = np.full((2, 2), float(center))
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action_1 = -1
    last_reward = -1
    
    for t in range(n_trials):
        # Drift reward probabilities for the current trial
        reward_probs = [[drift1[t], drift2[t]], 
                        [drift3[t], drift4[t]]]
        
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_bonus[last_action_1] = st_rew
            else:
                stickiness_bonus[last_action_1] = st_unrew
                
        q1_val = beta * q_net + stickiness_bonus
        q1_val -= np.max(q1_val) # Prevent overflow
        exp_q1 = np.exp(q1_val)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2 Choice
        q2_val = beta * q_stage2_mf[s2]
        q2_val -= np.max(q2_val) # Prevent overflow
        exp_q2 = np.exp(q2_val)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward Generation
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Prediction Errors
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s2, a2]
        
        # Value Updates
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s2, a2] += alpha * delta_stage2
        
        # Unchosen Decay
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] = q_stage1_mf[unchosen_1] * (1 - decay) + center * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + center * decay
                    
        # Update last action and reward for next trial's stickiness
        last_action_1 = a1
        last_reward = r
        
        # Store trial data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r
        
    return stage1_choice, state2, stage2_choice, reward
