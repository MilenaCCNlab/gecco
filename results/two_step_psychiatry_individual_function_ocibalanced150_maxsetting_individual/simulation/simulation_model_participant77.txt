import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Model 3: Stage-specific Reward Perseveration.
    Includes separate perseveration bonuses for stage 1 and stage 2 choices,
    applied only when the previous trial was rewarded. Stage 2 perseveration is state-dependent.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win_1, pers_win_2]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win_1, pers_win_2 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = 0

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Trial-wise reward probabilities
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # --- Stage 1 ---
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward > 0:
            pers_bonus_1[last_action_1] = 1.0
                
        q_net_total = beta_1 * q_net + pers_win_1 * pers_bonus_1
        q_net_total = q_net_total - np.max(q_net_total) # avoid overflow
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # --- Transition ---
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # --- Stage 2 ---
        pers_bonus_2 = np.zeros(2)
        if last_reward > 0 and s2 == last_state and last_action_2 != -1:
            pers_bonus_2[last_action_2] = 1.0

        q_stage2_total = beta_2 * q_stage2_mf[s2] + pers_win_2 * pers_bonus_2
        q_stage2_total = q_stage2_total - np.max(q_stage2_total) # avoid overflow
        exp_q2 = np.exp(q_stage2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)

        # --- Reward ---
        r = int(np.random.random() < reward_probs[s2][a2])

        # --- Learning / Value Updates ---
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha_1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += alpha_2 * delta_stage2

        q_stage1_mf[a1] += alpha_1 * lam * delta_stage2
        
        # --- Update track variables ---
        last_reward = r
        last_action_2 = a2
        last_state = s2
        last_action_1 = a1
        
        # --- Store data ---
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
