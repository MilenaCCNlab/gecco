import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the two-step model with separate MB and MF inverse 
    temperatures at stage 1, and a decay mechanism for unchosen options towards 0.5.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, decay, beta_mb, beta_mf, beta2, lam, pers1, pers2]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, decay, beta_mb, beta_mf, beta2, lam, pers1, pers2 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_stage2 = np.array([-1, -1])

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[stage1_choice[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # Ensure probabilities sum exactly to 1 for np.random.choice
        probs_1 /= np.sum(probs_1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2

        # Stage 2 Choice
        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        probs_2 /= np.sum(probs_2)
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Update last action for the visited state
        last_action_stage2[state_idx] = a2

        # Reward Generation
        r = int(np.random.rand() < reward_probs[s2][a2])

        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lam * delta_stage2

        # Decay unchosen options towards neutral value (0.5)
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
