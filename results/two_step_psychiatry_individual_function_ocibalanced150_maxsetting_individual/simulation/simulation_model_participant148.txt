import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Model 1: 
    Asymmetric learning rates for positive/negative RPEs, separate stickiness, and unchosen decay.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha_pos, alpha_neg, beta, w, lam, rho1, rho2, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha_pos, alpha_neg, beta, w, lam, rho1, rho2, decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for t in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]

        # --- Stage 1 ---
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if t > 0:
            stickiness_1[stage1_choice[t - 1]] = 1
            
        logits_1 = beta * q_net + rho1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # --- Stage 2 ---
        stickiness_2 = np.zeros(2)
        if t > 0 and state2[t - 1] == s2:
            stickiness_2[stage2_choice[t - 1]] = 1
            
        logits_2 = beta * q_stage2_mf[s2] + rho2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward outcome
        r = int(np.random.random() < reward_probs[s2][a2])

        # --- Learning Updates ---
        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        # Stage 2 MF update
        delta_stage2 = r - q_stage2_mf[s2, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s2, a2] += alpha_s2 * delta_stage2

        # Stage 1 MF eligibility trace update from Stage 2
        q_stage1_mf[a1] += alpha_s2 * lam * delta_stage2

        # Unchosen decay
        q_stage1_mf[1 - a1] *= (1 - decay)
        q_stage2_mf[s2, 1 - a2] *= (1 - decay)
        q_stage2_mf[1 - s2, :] *= (1 - decay)

        # Record trial data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

    return stage1_choice, state2, stage2_choice, reward
