import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Model 3: Hybrid model with choice stickiness 
    and forgetting of unchosen Q-values.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, beta, w, lam, pi_1, pi_2, unchosen_decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    learning_rate, beta, w, lam, pi_1, pi_2, unchosen_decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Drift reward probs
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]
        
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + pi_1 * stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s

        # Stage 2 choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = 1.0
            
        logits_2 = beta * q_stage2_mf[state_idx] + pi_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward generation
        r = int(np.random.rand() < reward_probs[s][a2])
        
        # Record keeping
        stage1_choice[trial] = a1
        state2[trial] = s
        stage2_choice[trial] = a2
        reward[trial] = r

        # Learning & Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        # Decay of unchosen options back towards 0.5
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += unchosen_decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        unchosen_a2 = 1 - a2
        q_stage2_mf[state_idx, unchosen_a2] += unchosen_decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])
        
        unvisited_s = 1 - state_idx
        q_stage2_mf[unvisited_s, 0] += unchosen_decay * (0.5 - q_stage2_mf[unvisited_s, 0])
        q_stage2_mf[unvisited_s, 1] += unchosen_decay * (0.5 - q_stage2_mf[unvisited_s, 1])

        # Track last choices for perseveration/stickiness
        last_action_1 = a1
        last_action_2[state_idx] = a2

    return stage1_choice, state2, stage2_choice, reward
