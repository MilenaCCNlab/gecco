import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the hybrid model-based/model-free 2-step model
    with a decay parameter for unchosen action values (forgetting) and stickiness.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta, w, stickiness, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta, w, stickiness, decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action = -1

    for t in range(n_trials):
        # Trial-wise reward probabilities for states and actions
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus = np.zeros(2)
        if last_action != -1:
            stick_bonus[last_action] = stickiness
            
        q_vals_1 = beta * q_net + stick_bonus
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State transition
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2 choice
        q_vals_2 = beta * q_stage2_mf[s2]
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Store data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

        # Learning & Decay Updates
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * delta_stage2
        
        last_action = a1

    return stage1_choice, state2, stage2_choice, reward
