import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using the specified cognitive model.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha_pos, alpha_neg, beta, w, lam, stickiness_1, stickiness_2, decay_rate]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    
    alpha_pos, alpha_neg, beta, w, lam, stickiness_1, stickiness_2, decay_rate = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action_1 = -1
    last_action_2 = -1
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]
        
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + stickiness_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State transition
        s = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2 choice
        rep_2 = np.zeros(2)
        if last_action_2 != -1:
            rep_2[last_action_2] = 1.0
            
        logits_2 = beta * q_stage2_mf[s] + stickiness_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s][a2])
        
        # Learning and updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_stage1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha_stage2 * delta_stage2
        q_stage1_mf[a1] += alpha_stage2 * lam * delta_stage2
        
        # Decay unchosen Q-values towards 0.5
        q_stage1_mf[1 - a1] += decay_rate * (0.5 - q_stage1_mf[1 - a1])
        q_stage2_mf[s, 1 - a2] += decay_rate * (0.5 - q_stage2_mf[s, 1 - a2])
        q_stage2_mf[1 - s, 0] += decay_rate * (0.5 - q_stage2_mf[1 - s, 0])
        q_stage2_mf[1 - s, 1] += decay_rate * (0.5 - q_stage2_mf[1 - s, 1])
        
        # Save actions for stickiness in the next trial
        last_action_1 = a1
        last_action_2 = a2
        
        # Record data
        stage1_choice[trial] = a1
        state2[trial] = s
        stage2_choice[trial] = a2
        reward[trial] = r
        
    return stage1_choice, state2, stage2_choice, reward
