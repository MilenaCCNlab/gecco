import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using a model incorporating perseveration 
    at both stage 1 and stage 2 (state-specific), along with decay for 
    unchosen actions, eligibility trace, and model-based weighting.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta, w, lam, pers1, pers2, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta, w, lam, pers1, pers2, decay = parameters
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Stage 1 Net Values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stage 1 Perseveration
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers1
            
        # Stage 1 Choice
        exp_q1 = np.exp(beta * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State Transition
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 Perseveration
        pers_bonus_2 = np.zeros(2)
        if last_action_2[s2] != -1:
            pers_bonus_2[last_action_2[s2]] = pers2

        # Stage 2 Choice
        exp_q2 = np.exp(beta * q_stage2_mf[s2] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward Observation
        r = int(np.random.random() < reward_probs[s2][a2])

        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

        # Update last actions
        last_action_1 = a1
        last_action_2[s2] = a2

        # Stage 1 Model-Free Update
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # Stage 2 Update
        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += alpha * delta_stage2

        # Eligibility Trace Update for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2

        # Decay for unchosen action in the visited Stage 2 state
        unchosen_action_2 = 1 - a2
        q_stage2_mf[s2, unchosen_action_2] += decay * (0.5 - q_stage2_mf[s2, unchosen_action_2])

        # Decay for both actions in the unvisited Stage 2 state
        unvisited_state = 1 - s2
        q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])

    return stage1_choice, state2, stage2_choice, reward
