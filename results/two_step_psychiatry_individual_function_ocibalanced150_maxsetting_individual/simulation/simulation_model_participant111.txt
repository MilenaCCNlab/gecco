import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the two-step model with 
    positive/negative learning rates, eligibility trace, 
    and win-stay stickiness for both stage 1 and stage 2 choices.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha_pos, alpha_neg, beta1, beta2, lam, w, p_win1, p_win2]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p_win1, p_win2 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward_1 = 0
    
    last_action_2 = np.array([-1, -1])
    last_reward_2 = np.array([0, 0])

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if last_action_1 != -1 and last_reward_1 > 0:
            rep1[last_action_1] = 1
            
        q1 = beta1 * q_net + p_win1 * rep1
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2

        # Stage 2
        rep2 = np.zeros(2)
        if last_action_2[state_idx] != -1 and last_reward_2[state_idx] > 0:
            rep2[last_action_2[state_idx]] = 1
            
        q2 = beta2 * q_stage2_mf[state_idx] + p_win2 * rep2
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward
        reward_probs = [[drift1[trial], drift2[trial]], 
                        [drift3[trial], drift4[trial]]]
        r = int(np.random.random() < reward_probs[s2][a2])

        # Prediction Errors & Learning
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        alpha1_current = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha2_current = alpha_pos if delta_stage2 > 0 else alpha_neg

        q_stage1_mf[a1] += alpha1_current * delta_stage1 + alpha1_current * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha2_current * delta_stage2
        
        # Update history
        last_action_1 = a1
        last_reward_1 = r
        
        last_action_2[state_idx] = a2
        last_reward_2[state_idx] = r

        # Record data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
