import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the Hybrid Model-Based / Model-Free Q-learning 
    with transition-dependent stage 1 stickiness.
    
    Parameters:
    n_trials (int): Number of trials to simulate.
    parameters (list): [alpha_pos, alpha_neg, beta, w, lam, pi1_comm, pi1_rare, pi2]
    drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
    
    Returns:
    stage1_choice (np.ndarray): First-stage choices (0 or 1).
    state2 (np.ndarray): Second-stage states (0 or 1).
    stage2_choice (np.ndarray): Second-stage choices (0 or 1).
    reward (np.ndarray): Rewards (0 or 1).
    """
    
    alpha_pos, alpha_neg, beta, w, lam, pi1_comm, pi1_rare, pi2 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action_1 = -1
    last_action_2 = -1
    last_trans_common = True
    
    for t in range(n_trials):
        # Reward probabilities for current trial
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]
        
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        pi1 = pi1_comm if last_trans_common else pi1_rare
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Determine if transition was common or rare
        if (a1 == 0 and s2 == 0) or (a1 == 1 and s2 == 1):
            last_trans_common = True
        else:
            last_trans_common = False
            
        # Stage 2 choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0
            
        exp_q2 = np.exp(beta * q_stage2_mf[s2] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Learning / Updates
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += lr_stage1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s2, a2]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s2, a2] += lr_stage2 * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[a1] += lr_stage1 * lam * delta_stage2
        
        # Store actions for next trial's stickiness
        last_action_1 = a1
        last_action_2 = a2
        
        # Save trial data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r
        
    return stage1_choice, state2, stage2_choice, reward
