import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the Daw model with learned transition matrix 
    and unchosen option decay.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta_1, beta_2, w, lam, rho, decay, alpha_t]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta_1, beta_2, w, lam, rho, decay, alpha_t = parameters
    
    # Environment's true transition matrix
    true_transition_matrix = np.array([[0.7, 0.3], 
                                       [0.3, 0.7]])
    
    # Agent's belief about the transition matrix
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Stage 1 Net Values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        # Choice Stickiness
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = rho
            
        # Stage 1 Choice
        q1_vals = beta_1 * q_stage1_net + stickiness
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State Transition
        s2 = np.random.choice([0, 1], p=true_transition_matrix[a1])
        state_idx = s2
        
        # Stage 2 Choice
        q2_vals = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.rand() < reward_probs[s2][a2])
        
        # Transition Matrix Update
        transition_matrix[a1, state_idx] += alpha_t * (1 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_t * (0 - transition_matrix[a1, 1 - state_idx])

        # Stage 1 Value Update (Model-Free)
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # Stage 2 Value Update
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility Trace Update for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2

        # Decay for Unchosen Options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)

        last_action = a1
        
        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
