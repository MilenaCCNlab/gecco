import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards using the Hybrid Model with Stage 1 Bias, Stickiness, and Decay.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, beta, w, stickiness_1, stickiness_2, decay_rate, bias_1]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.
        
    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, decay_rate, bias_1 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.full((2, 2), 0.5)
    last_action_1 = -1
    last_action_2 = np.full(2, -1)
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    for trial in range(n_trials):
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]
        
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        bias_vector_1 = np.array([bias_1, 0.0])
            
        logits_1 = beta * q_net + stickiness_1 * stick_vector_1 + bias_vector_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        stick_vector_2 = np.zeros(2)
        if last_action_2[s2] != -1:
            stick_vector_2[last_action_2[s2]] = 1.0
            
        logits_2 = beta * q_stage2_mf[s2] + stickiness_2 * stick_vector_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        r = int(np.random.random() < reward_probs[s2][a2])
        
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += learning_rate * delta_stage2
        
        unchosen_action_1 = 1 - a1
        q_stage1_mf[unchosen_action_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_action_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
                    
        last_action_1 = a1
        last_action_2[s2] = a2
        
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r
        
    return stage1_choice, state2, stage2_choice, reward
