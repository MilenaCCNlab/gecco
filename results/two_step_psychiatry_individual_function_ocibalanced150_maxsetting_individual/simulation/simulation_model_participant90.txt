import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the hybrid model-based/model-free 2-step model
    with asymmetric learning rates, separate betas, choice stickiness, and unchosen action decay.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [lr_pos, lr_neg, decay_rate, beta1, beta2, w, stickiness]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    lr_pos, lr_neg, decay_rate, beta1, beta2, w, stickiness = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for t in range(n_trials):
        # Drift reward probabilities for the current trial
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]

        # --- Stage 1 ---
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply stickiness to the previously chosen action
        if t > 0:
            q_stage1_net[stage1_choice[t-1]] += stickiness

        # Stage 1 choice
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)

        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # --- Stage 2 ---
        exp_q2 = np.exp(beta2 * q_stage2_mf[s2])
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward determination
        r = int(np.random.random() < reward_probs[s2][a2])

        # --- Learning Updates ---
        # Stage 1 update
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        lr1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[a1] += lr1 * delta_stage1

        # Stage 1 decay for unchosen action
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        # Stage 2 update
        delta_stage2 = r - q_stage2_mf[s2, a2]
        lr2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[s2, a2] += lr2 * delta_stage2

        # Stage 2 decay for unchosen actions
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

        # Store trial data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

    return stage1_choice, state2, stage2_choice, reward
