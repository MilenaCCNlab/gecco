import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using Model 1: Asymmetric learning rates for 
    positive vs negative prediction errors, combined with transition-dependent 
    perseveration and unchosen decay.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha_pos, alpha_neg, beta1, beta2, w, pi_common, pi_rare, decay]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha_pos, alpha_neg, beta1, beta2, w, pi_common, pi_rare, decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_common = -1

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for t in range(n_trials):
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        if last_trans_common == 1:
            pi = pi_common
        elif last_trans_common == 0:
            pi = pi_rare
        else:
            pi = 0.0
            
        logits1 = beta1 * (q_stage1_net + pi * rep)
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)

        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2

        # Stage 2 choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])

        # Learning Stage 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        # Learning Stage 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - a2
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1 - decay)
        q_stage2_mf[other_state, 1] *= (1 - decay)
        
        # Update history
        last_action = a1
        if (a1 == 0 and state_idx == 0) or (a1 == 1 and state_idx == 1):
            last_trans_common = 1
        else:
            last_trans_common = 0

        # Store data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

    return stage1_choice, state2, stage2_choice, reward
