import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the hybrid MB/MF Q-learning model
    with choice stickiness and value decay.
    """
    alpha, beta, w, stickiness, decay = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Retrieve reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]
        
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec = np.zeros(2)
        if last_action != -1:
            stick_vec[last_action] = 1.0
            
        logits_1 = beta * q_net + stickiness * stick_vec
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        state_idx = s2
        
        # Stage 2 Choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward generation
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Stage 1 Update
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] *= (1 - decay) # Decay unchosen
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # Stage 2 Update and Stage 1 Eligibility Trace Update
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay) # Decay unchosen
                    
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * delta_stage2 # Implicit lam=1
        
        # Update last action
        last_action = a1
        
        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
