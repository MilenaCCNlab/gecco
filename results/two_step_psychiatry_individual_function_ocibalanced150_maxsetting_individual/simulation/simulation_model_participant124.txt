import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using a model with separate learning rates for stage 1 and stage 2,
    a shared decay rate for unchosen Q-values, and separate model-based and model-free inverse temperatures.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha_1, alpha_2, decay, beta_mb, beta_mf, beta_2, lam, p_1]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha_1, alpha_2, decay, beta_mb, beta_mf, beta_2, lam, p_1 = parameters
    
    transition_matrix = np.array([[0.7, 0.3], 
                                  [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    for t in range(n_trials):
        # Drift reward probs
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]
        
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])
        
        # Stage 2
        logits_2 = beta_2 * q_stage2_mf[s2]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = np.random.choice([0, 1], p=probs_2)
        
        # Reward
        r = int(np.random.random() < reward_probs[s2][a2])
        
        # Decay for unchosen Q-values
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        # Updates
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha_1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s2, a2]
        q_stage2_mf[s2, a2] += alpha_2 * delta_stage2
        
        q_stage1_mf[a1] += alpha_1 * lam * delta_stage2
        
        last_action_1 = a1
        
        # Save trial data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r
        
    return stage1_choice, state2, stage2_choice, reward
