import numpy as np

def simulate_model(n_trials, parameters, drift1, drift2, drift3, drift4):
    """
    Simulates choices and rewards based on Model 1: Single learning rate with decay 
    for unchosen options, and separate perseveration for win/loss.
    
    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, decay, beta_1, beta_2, w, lam, pers_win, pers_lose]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """
    learning_rate, decay, beta_1, beta_2, w, lam, pers_win, pers_lose = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action_1 = -1
    last_reward = -1

    for t in range(n_trials):
        # Drift reward probabilities
        reward_probs = [[drift1[t], drift2[t]], [drift3[t], drift4[t]]]

        # Stage 1 values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        # Perseveration bonus based on last trial outcome
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers_bonus[last_action_1] = pers_win
            else:
                pers_bonus[last_action_1] = pers_lose
            
        # Stage 1 choice
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2 state
        s2 = np.random.choice([0, 1], p=transition_matrix[a1])

        # Stage 2 choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[s2])
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)

        # Receive reward
        r = int(np.random.random() < reward_probs[s2][a2])

        # Calculate prediction errors (before decay applied)
        delta_stage1 = q_stage2_mf[s2, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s2, a2]

        # Decay all Q-values
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)

        # Update chosen Q-values
        q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[s2, a2] += learning_rate * delta_stage2
        
        # Store for next trial's perseveration calculation
        last_action_1 = a1
        last_reward = r
        
        # Log trial data
        stage1_choice[t] = a1
        state2[t] = s2
        stage2_choice[t] = a2
        reward[t] = r

    return stage1_choice, state2, stage2_choice, reward
