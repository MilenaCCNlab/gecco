import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the hybrid model-based/model-free 2-step model
    with eligibility traces and perseveration.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [alpha, beta, w, lambd, p]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    alpha, beta, w, lambd, p = parameters
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * (q_net - np.max(q_net))) # subtracted max to prevent overflow
        probs_1 = exp_q1 / np.sum(exp_q1)
        action_1 = np.random.choice([0, 1], p=probs_1)
        
        # Transition to Stage 2
        state_idx = np.random.choice([0, 1], p=transition_matrix[action_1])

        # Stage 2 choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        action_2 = np.random.choice([0, 1], p=probs_2)
  
        # Observe Reward
        r = int(np.random.random() < reward_probs[state_idx][action_2])

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2] - q_stage1_mf[action_1]
        q_stage1_mf[action_1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, action_2]
        q_stage2_mf[state_idx, action_2] += alpha * delta_stage2
        q_stage1_mf[action_1] += alpha * lambd * delta_stage2
        
        # Store choices and state for next trial
        last_action_1 = action_1
        
        stage1_choice[trial] = action_1
        state2[trial] = state_idx
        stage2_choice[trial] = action_2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
