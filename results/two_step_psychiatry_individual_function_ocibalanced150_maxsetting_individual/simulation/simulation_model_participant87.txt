import numpy as np

def simulate_model(
    n_trials,
    parameters,
    drift1,
    drift2,
    drift3,
    drift4):
    """
    Simulates choices and rewards using the cognitive model with a single temperature, 
    loss aversion for 0-coin rewards, choice perseveration, and transition learning.

    Parameters:
        n_trials (int): Number of trials to simulate.
        parameters (list): [learning_rate, decay_unchosen, lr_trans, beta, lam, w, p_pers, loss_av]
        drift1, drift2, drift3, drift4 (np.ndarray): Trial-wise reward probabilities.

    Returns:
        stage1_choice (np.ndarray): First-stage choices (0 or 1).
        state2 (np.ndarray): Second-stage states (0 or 1).
        stage2_choice (np.ndarray): Second-stage choices (0 or 1).
        reward (np.ndarray): Rewards (0 or 1).
    """

    learning_rate, decay_unchosen, lr_trans, beta, lam, w, p_pers, loss_av = parameters
    
    # Environment's true transition matrix
    true_transition_matrix = np.array([[0.7, 0.3], 
                                       [0.3, 0.7]])
    
    # Agent's estimated transition matrix and Q-values
    transition_matrix = np.array([[0.5, 0.5], 
                                  [0.5, 0.5]])
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    stage1_choice = np.zeros(n_trials, dtype=int)
    state2 = np.zeros(n_trials, dtype=int)
    stage2_choice = np.zeros(n_trials, dtype=int)
    reward = np.zeros(n_trials, dtype=int)
    
    last_action = -1

    for trial in range(n_trials):
        # Reward probabilities for the current trial
        reward_probs = [[drift1[trial], drift2[trial]], [drift3[trial], drift4[trial]]]

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent_1 = beta * q_stage1_net
        if last_action != -1:
            exponent_1[last_action] += p_pers
            
        max_exp_1 = np.max(exponent_1)
        exp_q1 = np.exp(exponent_1 - max_exp_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        a1 = np.random.choice([0, 1], p=probs_1)
        
        # State transition 
        s2 = np.random.choice([0, 1], p=true_transition_matrix[a1])
        state_idx = s2

        # Stage 2 Choice
        exponent_2 = beta * q_stage2_mf[state_idx]
        max_exp_2 = np.max(exponent_2)
        exp_q2 = np.exp(exponent_2 - max_exp_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        a2 = np.random.choice([0, 1], p=probs_2)

        # Reward realization
        r = int(np.random.rand() < reward_probs[s2][a2])

        # Agent updates
        # 1. Update transition matrix based on observed transition
        transition_matrix[a1, s2] += lr_trans * (1.0 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += lr_trans * (0.0 - transition_matrix[a1, 1 - s2])

        # 2. Decay unchosen options
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] *= (1 - decay_unchosen)
        
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay_unchosen)

        # 3. Model-free Q-value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        eff_reward = 1.0 if r == 1 else -loss_av
        
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        # 4. Eligibility trace update for stage 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Update last action for perseveration
        last_action = a1
        
        # Store trial data
        stage1_choice[trial] = a1
        state2[trial] = s2
        stage2_choice[trial] = a2
        reward[trial] = r

    return stage1_choice, state2, stage2_choice, reward
