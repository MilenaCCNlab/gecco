Here are 3 new cognitive models proposing novel mechanisms to explain the participant's behavior, based on the provided data and avoiding previously tried parameter combinations.

### Model 1: Counterfactual Learning at Stage 2
Participants often assume that the reward probabilities of the two aliens on a planet are anti-correlated. If one gives a reward, the other probably wouldn't have. This model includes a `cf_lr` (counterfactual learning rate) parameter to update the value of the unchosen alien at Stage 2 in the opposite direction of the received reward.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Counterfactual Learning at Stage 2.
    Updates the unchosen action at stage 2 assuming its outcome is opposite to the chosen action's outcome.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-values
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Weighting of model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers: [-5,5] - Choice perseveration (stickiness) for stage 1
    decay: [0,1] - Decay rate for unchosen actions towards 0.5
    cf_lr: [0,1] - Counterfactual learning rate for the unchosen stage 2 action
    """
    import numpy as np
    learning_rate, beta, w, lam, pers, decay, cf_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Counterfactual update for unchosen action at Stage 2
        unchosen_2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_2]
        q_stage2_mf[state_idx, unchosen_2] += cf_lr * delta_cf

        # Decay for Stage 1 unchosen
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        # Decay for the completely unvisited state at Stage 2
        other_state = 1 - state_idx
        for a in range(2):
            q_stage2_mf[other_state, a] += decay * (0.5 - q_stage2_mf[other_state, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Action Sequence Stickiness
Rather than just perseverating on Stage 1 or Stage 2 actions independently, participants might develop a habit of repeating a specific action sequence (e.g., Spaceship 0 -> Alien 1). This model adds a sequence perseveration bonus at Stage 2, which is only applied if the participant has already chosen to repeat their Stage 1 action from the previous trial.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Action Sequence Stickiness.
    Adds a perseveration bonus to Stage 2 actions if the same Stage 1 and Stage 2 action sequence was chosen in the previous trial.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-values
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Weighting of model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers: [-5,5] - Choice perseveration (stickiness) for stage 1
    decay: [0,1] - Decay rate for unchosen actions towards 0.5
    seq_pers: [-5,5] - Sequence perseveration bonus applied at stage 2
    """
    import numpy as np
    learning_rate, beta, w, lam, pers, decay, seq_pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_1 == action_1[trial] and last_action_2 != -1:
            # Sequence stickiness: bias towards repeating the exact 2-step sequence
            logits_2[last_action_2] += seq_pers

        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Decay
The rate at which participants forget (or regress to the mean) the value of unchosen options may depend on whether the current trial resulted in a reward. If they win, they might focus entirely on the chosen option and let alternatives decay faster. If they lose, they might actively retain alternative values to prepare for a switch.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Outcome-Dependent Decay.
    The decay of unchosen Q-values differs depending on whether the current trial was rewarded or not.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-values
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Weighting of model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers: [-5,5] - Choice perseveration (stickiness) for stage 1
    decay_win: [0,1] - Decay rate for unchosen actions when the trial is rewarded
    decay_lose: [0,1] - Decay rate for unchosen actions when the trial is unrewarded
    """
    import numpy as np
    learning_rate, beta, w, lam, pers, decay_win, decay_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Choose the decay rate based on the outcome of the current trial
        current_decay = decay_win if reward[trial] == 1 else decay_lose

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += current_decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += current_decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
