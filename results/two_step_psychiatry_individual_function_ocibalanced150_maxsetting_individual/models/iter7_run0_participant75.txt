def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Reward-Dependent Learning Rates + Win/Lose Perseveration
    This model posits that the learning rate differs based on whether the current 
    trial's outcome is a reward or not, capturing asymmetric updating based on valence.
    It also includes reward-dependent perseveration at stage 1.
    
    Parameters:
    alpha_rew: [0, 1] - Learning rate for rewarded trials (reward = 1).
    alpha_unrew: [0, 1] - Learning rate for unrewarded trials (reward = 0).
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Stage 1 perseveration after a win.
    pers_lose: [-5, 5] - Stage 1 perseveration after a loss.
    """
    alpha_rew, alpha_unrew, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_r = -1
    
    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            continue
            
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if prev_a1 != -1:
            if prev_r == 1:
                pers[prev_a1] = pers_win
            else:
                pers[prev_a1] = pers_lose
                
        logits_1 = beta1 * q_stage1_net + pers
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        alpha = alpha_rew if r == 1 else alpha_unrew
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-Dependent Model-Based Weight + Win/Lose Perseveration
    The degree of model-based planning (w) fluctuates depending on whether the 
    previous trial was rewarded, capturing changes in cognitive control after wins vs losses.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w_win: [0, 1] - Model-based weight after a win.
    w_lose: [0, 1] - Model-based weight after a loss.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Stage 1 perseveration after a win.
    pers_lose: [-5, 5] - Stage 1 perseveration after a loss.
    """
    alpha, beta1, beta2, w_win, w_lose, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

            continue

            w = 0.5

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: State Perseveration + Win/Lose Perseveration
    Includes a tendency to choose the stage 1 action that commonly leads to the 
    planet visited on the previous trial, capturing a form of "state perseveration" 
    regardless of how the state was reached.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Stage 1 action perseveration after a win.
    pers_lose: [-5, 5] - Stage 1 action perseveration after a loss.
    pers_state: [-5, 5] - Perseveration towards the action that commonly leads to the previous state.
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, pers_state = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_r = -1
    prev_state = -1
    
    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            prev_state = -1
            continue
            
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if prev_a1 != -1:
            if prev_r == 1:
                pers[prev_a1] += pers_win
            else:
                pers[prev_a1] += pers_lose
                
        if prev_state != -1:
            # Action 0 commonly leads to State 0, Action 1 commonly leads to State 1
            pers[prev_state] += pers_state
                
        logits_1 = beta1 * q_stage1_net + pers
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r
        prev_state = state_idx
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
