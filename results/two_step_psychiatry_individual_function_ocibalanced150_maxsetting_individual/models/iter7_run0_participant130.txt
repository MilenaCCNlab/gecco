Here are 3 new cognitive models based on your requirements and the patterns in the participant's data. 

The participant data shows long streaks of choosing the same spaceship and alien, indicating strong choice perseveration (stickiness). To capture this, the models introduce new mechanisms for how choice stickiness and transitions are learned and applied.

### Model 1: Choice Trace Stickiness
Instead of just remembering the single last choice made, this model assumes the participant maintains a fading memory (exponential moving average) of their past choices, which drives their strong block-like perseveration.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Hybrid MB/MF RL with choice trace (exponential moving average) for stickiness.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updating.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for stage 1 updating.
    rho_1: [0,10] - Stickiness weight for stage 1 choice trace.
    rho_2: [0,10] - Stickiness weight for stage 2 choice trace.
    tau: [0,1] - Update rate (decay) for the choice traces.
    """
    alpha, beta, w, lam, rho_1, rho_2, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Choice traces
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        v1 = beta * q_net + rho_1 * trace_1
        exp_q1 = np.exp(v1 - np.max(v1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        v2 = beta * q_stage2_mf[state_idx, :] + rho_2 * trace_2[state_idx, :]
        exp_q2 = np.exp(v2 - np.max(v2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Choice trace updating
        trace_1 *= (1 - tau)
        trace_1[action_1[trial]] += tau
        
        trace_2 *= (1 - tau)
        trace_2[state_idx, action_2[trial]] += tau

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition Matrix Learning
Instead of assuming a static understanding of the task structure, this model allows the participant to dynamically update their internal transition matrix based on experienced transitions, which can explain shifts in model-based choices.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Hybrid MB/MF RL with dynamic transition matrix learning.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updating.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for stage 1 updating.
    rho_1: [0,10] - Choice stickiness parameter for stage 1 actions.
    rho_2: [0,10] - Choice stickiness parameter for stage 2 actions.
    alpha_trans: [0,1] - Learning rate for the transition matrix.
    """
    alpha, beta, w, lam, rho_1, rho_2, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1
    last_choice_2 = [-1, -1]

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_choice_1 != -1:
            stickiness_1[last_choice_1] = 1.0
            
        v1 = beta * q_net + rho_1 * stickiness_1
        exp_q1 = np.exp(v1 - np.max(v1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_choice_2[state_idx] != -1:
            stickiness_2[last_choice_2[state_idx]] = 1.0

        v2 = beta * q_stage2_mf[state_idx, :] + rho_2 * stickiness_2
        exp_q2 = np.exp(v2 - np.max(v2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Transition matrix updating (increases prob of observed transition, decreases the other)
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] -= alpha_trans * transition_matrix[action_1[trial], 1 - state_idx]
        
        last_choice_1 = action_1[trial]
        last_choice_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Stickiness
This model parses stickiness based on recent successes versus failures. Participants may exhibit stronger perseveration if they just won (win-stay) compared to if they just lost (lose-shift), and this mechanism is separately parameterized for both stages.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Hybrid MB/MF RL with outcome-dependent (win vs loss) stickiness at both stages.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updating.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for stage 1 updating.
    rho_1_win: [0,10] - Stage 1 choice stickiness after a globally rewarded trial.
    rho_1_lose: [0,10] - Stage 1 choice stickiness after a globally unrewarded trial.
    rho_2_win: [0,10] - Stage 2 choice stickiness after a rewarded trial in that specific state.
    rho_2_lose: [0,10] - Stage 2 choice stickiness after an unrewarded trial in that specific state.
    """
    alpha, beta, w, lam, rho_1_win, rho_1_lose, rho_2_win, rho_2_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1
    last_choice_2 = [-1, -1]
    last_reward = -1
    last_reward_state = [-1, -1]

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_choice_1 != -1:
            stickiness_1[last_choice_1] = 1.0
            
        rho_1_current = 0
        if last_reward == 1:
            rho_1_current = rho_1_win
        elif last_reward == 0:
            rho_1_current = rho_1_lose
            
        v1 = beta * q_net + rho_1_current * stickiness_1
        exp_q1 = np.exp(v1 - np.max(v1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_choice_2[state_idx] != -1:
            stickiness_2[last_choice_2[state_idx]] = 1.0

        rho_2_current = 0
        if last_reward_state[state_idx] == 1:
            rho_2_current = rho_2_win
        elif last_reward_state[state_idx] == 0:
            rho_2_current = rho_2_lose

        v2 = beta * q_stage2_mf[state_idx, :] + rho_2_current * stickiness_2
        exp_q2 = np.exp(v2 - np.max(v2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_choice_1 = action_1[trial]
        last_choice_2[state_idx] = action_2[trial]
        last_reward = reward[trial]
        last_reward_state[state_idx] = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
