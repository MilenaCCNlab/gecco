def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition matrix learning along with stage 1 perseveration and bias.
    Instead of assuming a fixed transition matrix, this model learns the transition 
    probabilities based on experienced state transitions.
    
    Bounds:
    alpha: [0, 1] - Learning rate for action values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    p_1: [-5, 5] - Perseveration parameter for previous stage 1 choice.
    bias: [-5, 5] - Baseline preference (bias) for action 0 at stage 1.
    alpha_trans: [0, 1] - Learning rate for the transition matrix.
    """
    alpha, beta_1, beta_2, w, lam, p_1, bias, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to uniform (0.5) to allow learning from scratch
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += p_1
        q_stage1_net[0] += bias
            
        q1_scaled = beta_1 * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0
            
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
            
        q2_scaled = beta_2 * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        if a2 != -1:
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and a2 != -1:
            # Update the learned transition matrix
            transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
            transition_matrix[a1, 1 - state_idx] = 1.0 - transition_matrix[a1, state_idx]
            
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += alpha * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2
            
            q_stage1_mf[a1] += alpha * lam * delta_stage2
            
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-dependent model-based control with stage 1 perseveration and bias.
    This model assumes that the weight placed on model-based control (w) fluctuates 
    depending on whether the previous trial was rewarded (w_win) or unrewarded (w_lose).
    
    Bounds:
    alpha: [0, 1] - Learning rate for chosen updates.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w_win: [0, 1] - Weight on model-based control if previous trial was rewarded.
    w_lose: [0, 1] - Weight on model-based control if previous trial was unrewarded.
    lam: [0, 1] - Eligibility trace decay parameter.
    p_1: [-5, 5] - Perseveration parameter for previous stage 1 choice.
    bias: [-5, 5] - Baseline preference (bias) for action 0 at stage 1.
    """
    alpha, beta_1, beta_2, w_win, w_lose, lam, p_1, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 1 # Default to win for the first trial

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine model-based weight based on previous reward
        w = w_win if last_reward > 0 else w_lose
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += p_1
        q_stage1_net[0] += bias
            
        q1_scaled = beta_1 * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0
            
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
            
        q2_scaled = beta_2 * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        if a2 != -1:
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += alpha * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2
            
            q_stage1_mf[a1] += alpha * lam * delta_stage2
            
            last_action_1 = a1
            last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dual bias model (stage 1 and stage 2 biases) with stage 1 perseveration.
    This model captures the participant's robust preference for choice 0 at stage 1 
    and extends it to capture potential systematic preference for alien 0 at stage 2.
    
    Bounds:
    alpha: [0, 1] - Learning rate for chosen updates.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    p_1: [-5, 5] - Perseveration parameter for previous stage 1 choice.
    bias_1: [-5, 5] - Baseline preference (bias) for action 0 at stage 1.
    bias_2: [-5, 5] - Baseline preference (bias) for action 0 at stage 2.
    """

    n_trials = len(action_1)

    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        q_stage2_net[0] += bias_2
            
        q2_scaled = beta_2 * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
