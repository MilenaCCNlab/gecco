Here are 3 new cognitive models that build upon the best model found so far by exploring different parameterizations involving the decay mechanism, which proved effective.

### Model 1: Decay with Separate Inverse Temperatures
This model extends the best model by allowing the choice stochasticity (inverse temperature) to differ between the first and second stages. Participants often show different exploration/exploitation balances at different decision stages.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) Model with decay and separate inverse temperatures for each stage.
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    decay: [0, 1] - Rate at which unchosen Q-values decay towards 0.
    beta_1: [0, 10] - Inverse temperature controlling choice stochasticity for stage 1.
    beta_2: [0, 10] - Inverse temperature controlling choice stochasticity for stage 2.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    stickiness: [-5, 5] - Tendency to repeat the previous first-stage choice.
    """
    learning_rate, decay, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay for unchosen actions and unvisited states
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1.0 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1.0 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, :] *= (1.0 - decay)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Decay with Asymmetric Learning Rates
This model introduces asymmetric learning rates for positive versus negative prediction errors, alongside decay. This accounts for participants who may weigh unexpected positive outcomes (rewards) differently from unexpected negative outcomes (reward omissions).

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) Model with decay and asymmetric learning rates.
    
    Parameters/Bounds:
    lr_pos: [0, 1] - Learning rate for positive prediction errors.
    lr_neg: [0, 1] - Learning rate for negative prediction errors.
    decay: [0, 1] - Rate at which unchosen Q-values decay towards 0.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    stickiness: [-5, 5] - Tendency to repeat the previous first-stage choice.
    """
    lr_pos, lr_neg, decay, beta, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += lr_stage2 * lam * delta_stage2

        # Decay for unchosen actions and unvisited states
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1.0 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1.0 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, :] *= (1.0 - decay)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Decay with Stage-Specific Stickiness (Perseveration)
This model adds a second stickiness parameter, `stickiness_2`, for the second-stage choices (state-specific perseveration), which allows capturing habitual choosing of specific aliens completely independent of the value-based learning.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) Model with decay and stage-specific stickiness.
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    decay: [0, 1] - Rate at which unchosen Q-values decay towards 0.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    stickiness_1: [-5, 5] - Tendency to repeat the previous first-stage choice.
    stickiness_2: [-5, 5] - Tendency to repeat the previous second-stage choice in the same state.
    """
    learning_rate, decay, beta, w, lam, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx, :]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay for unchosen actions and unvisited states
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1.0 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1.0 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, :] *= (1.0 - decay)
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
