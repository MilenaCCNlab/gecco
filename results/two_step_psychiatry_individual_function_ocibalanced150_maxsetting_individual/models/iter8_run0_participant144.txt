Here are three new cognitive models that explore alternative mechanisms for the two-step decision task.

### Model 1: Transition Matrix Learning Model
This model tests the hypothesis that participants do not rely on a fixed transition matrix (e.g., 70%/30%) but instead continuously update their internal model of the transition probabilities based on their experience. When a transition occurs, they update the probability of that transition using a dedicated learning rate.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Transition Matrix Learning.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature for choice probabilities.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    eligibility: [0, 1] - Eligibility trace (lambda) for updating Stage 1 values from Stage 2 prediction errors.
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    lr_trans: [0, 1] - Learning rate for updating the transition matrix probabilities.
    """
    learning_rate, beta, w, eligibility, stickiness_1, stickiness_2, lr_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to the commonly instructed 0.7/0.3 probabilities
    trans_prob = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = trans_prob @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q_stage1_net.copy()
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        # Robust softmax
        q1_scaled = beta * q1_eff
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        q2_scaled = beta * q2_eff
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += eligibility * learning_rate * delta_stage2
        
        # Transition matrix updating based on experienced transition
        trans_prob[action_1[trial], state_idx] += lr_trans * (1.0 - trans_prob[action_1[trial], state_idx])
        trans_prob[action_1[trial], 1 - state_idx] += lr_trans * (0.0 - trans_prob[action_1[trial], 1 - state_idx])
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: UCB-like Exploration Bonus Model
This model builds on the idea that participants might become increasingly curious or uncertain about options they haven't chosen in a while. It implements an Upper Confidence Bound (UCB) style exploration bonus that scales with the square root of the time since the option was last chosen, applying it to both Stage 1 and Stage 2 decisions.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with UCB-like Exploration Bonus for Unchosen Options.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature for choice probabilities.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    eligibility: [0, 1] - Eligibility trace (lambda) for updating Stage 1 values from Stage 2 prediction errors.
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    exploration_bonus: [0, 5] - Bonus added to Q-values based on sqrt(time) since last chosen.
    """
    learning_rate, beta, w, eligibility, stickiness_1, stickiness_2, exploration_bonus = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    # Track time since last chosen (starts at 1)
    time_since_1 = np.ones(2)
    time_since_2 = np.ones((2, 2))
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Incorporate exploration bonus into the model-based calculation
        q2_ucb = q_stage2_mf + exploration_bonus * np.sqrt(time_since_2)
        max_q_stage2 = np.max(q2_ucb, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q_stage1_net + exploration_bonus * np.sqrt(time_since_1)
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        q1_scaled = beta * q1_eff
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_eff = q2_ucb[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        q2_scaled = beta * q2_eff
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += eligibility * learning_rate * delta_stage2
        
        # Update time since chosen
        time_since_1 += 1
        time_since_1[action_1[trial]] = 1
        
        time_since_2 += 1
        time_since_2[state_idx, action_2[trial]] = 1
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Differentiated Decay Model
This model extends the best-performing model by separating the decay rates for Stage 1 and Stage 2 unchosen options. Given the hierarchical structure of the task, participants might forget or regress unchosen spaceships (Stage 1) at a different rate than unchosen aliens (Stage 2).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Separate Unchosen Option Decay for Stage 1 and Stage 2.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    decay_rate_1: [0, 1] - Decay rate towards 0.5 for Stage 1 unchosen options.
    decay_rate_2: [0, 1] - Decay rate towards 0.5 for Stage 2 unchosen options.
    beta: [0, 10] - Inverse temperature for choice probabilities.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    eligibility: [0, 1] - Eligibility trace (lambda) for updating Stage 1 values from Stage 2 prediction errors.
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    learning_rate, decay_rate_1, decay_rate_2, beta, w, eligibility, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q_stage1_net.copy()
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        q1_scaled = beta * q1_eff
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        q2_scaled = beta * q2_eff
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen options at specific rates
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate_1 * (0.5 - q_stage1_mf[unchosen_a1])
        
        unchosen_a2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_a2] += decay_rate_2 * (0.5 - q_stage2_mf[state_idx, unchosen_a2])
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay_rate_2 * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay_rate_2 * (0.5 - q_stage2_mf[unvisited_state, 1])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += eligibility * learning_rate * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
