import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Incorporates a baseline reward shift.
    This model assumes that participants evaluate rewards relative to a subjective baseline (or reference point).
    A positive baseline reward means they are pessimistic (expecting more, so actual rewards feel smaller), 
    while a negative baseline reward means they are optimistic.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Model-based weight (1 = pure model-based, 0 = pure model-free).
    eligibility: [0, 1] - Eligibility trace for updating stage 1 values from stage 2 rewards.
    stickiness: [-5, 5] - Choice stickiness (perseverance) for stage 1.
    forgetting_rate: [0, 1] - Decay rate for unchosen actions towards 0.5.
    baseline_reward: [-1, 1] - Subjective baseline subtracted from the actual reward.
    """
    learning_rate, beta, w, eligibility, stickiness, forgetting_rate, baseline_reward = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        # Shift the reward by the baseline
        r = reward[trial] - baseline_reward

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        if a2 != -1:
            logits_2 = beta * q_stage2_mf[s]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            # Action value updating
            delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = r - q_stage2_mf[s, a2]
            q_stage2_mf[s, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += eligibility * learning_rate * delta_stage2

            # Forgetting for unchosen stage 2 actions
            for state_idx in range(2):
                for action_idx in range(2):
                    if not (state_idx == s and action_idx == a2):
                        q_stage2_mf[state_idx, action_idx] = (1 - forgetting_rate) * q_stage2_mf[state_idx, action_idx] + forgetting_rate * 0.5
        else:
            p_choice_2[trial] = 1.0
            # Forgetting for all stage 2 actions if no action was taken
            q_stage2_mf = (1 - forgetting_rate) * q_stage2_mf + forgetting_rate * 0.5

        # Forgetting for unchosen stage 1 action
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1 - forgetting_rate) * q_stage1_mf[unchosen_a1] + forgetting_rate * 0.5

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric eligibility traces for positive and negative prediction errors.
    This model proposes that the degree to which stage 2 outcomes reinforce stage 1 choices 
    depends on whether the outcome was better or worse than expected.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Model-based weight (1 = pure model-based, 0 = pure model-free).
    eligibility_pos: [0, 1] - Eligibility trace for positive stage 2 prediction errors.
    eligibility_neg: [0, 1] - Eligibility trace for negative stage 2 prediction errors.
    stickiness: [-5, 5] - Choice stickiness (perseverance) for stage 1.
    forgetting_rate: [0, 1] - Decay rate for unchosen actions towards 0.5.
    """
    learning_rate, beta, w, eligibility_pos, eligibility_neg, stickiness, forgetting_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)

            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            # Action value updating
            delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = r - q_stage2_mf[s, a2]
            q_stage2_mf[s, a2] += learning_rate * delta_stage2

            # Asymmetric eligibility trace update
            if delta_stage2 > 0:

                q_stage1_mf[a1] += eligibility_neg * learning_rate * delta_stage2

            # Forgetting for unchosen stage 2 actions
            for state_idx in range(2):
                for action_idx in range(2):
                    if not (state_idx == s and action_idx == a2):

            p_choice_2[trial] = 1.0
            q_stage2_mf = (1 - forgetting_rate) * q_stage2_mf + forgetting_rate * 0.5

        # Forgetting for unchosen stage 1 action
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1 - forgetting_rate) * q_stage1_mf[unchosen_a1] + forgetting_rate * 0.5

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Incorporates stage 2 stickiness.
    This model adds a state-specific tendency to repeat the same action at stage 2,
    capturing habitual or perseverative behavior on the planets independently of stage 1 choices.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Model-based weight (1 = pure model-based, 0 = pure model-free).
    eligibility: [0, 1] - Eligibility trace for updating stage 1 values from stage 2 rewards.
    stickiness_1: [-5, 5] - Choice stickiness (perseverance) for stage 1.
    forgetting_rate: [0, 1] - Decay rate for unchosen actions towards 0.5.
    stickiness_2: [-5, 5] - Choice stickiness (perseverance) for stage 2 actions (state-specific).
    """
    learning_rate, beta, w, eligibility, stickiness_1, forgetting_rate, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])  # Tracks the last action taken in each of the two stage 2 states

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        if a2 != -1:
            logits_2 = beta * q_stage2_mf[s]
            if last_action_2[s] != -1:
                logits_2[last_action_2[s]] += stickiness_2
                
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            # Action value updating
            delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]

            delta_stage2 = r - q_stage2_mf[s, a2]
            q_stage2_mf[s, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += eligibility * learning_rate * delta_stage2

            # Forgetting for unchosen stage 2 actions
            for state_idx in range(2):
                for action_idx in range(2):

                        q_stage2_mf[state_idx, action_idx] = (1 - forgetting_rate) * q_stage2_mf[state_idx, action_idx] + forgetting_rate * 0.5
            
            # Update last action for the current state
            last_action_2[s] = a2
        else:
            p_choice_2[trial] = 1.0
            q_stage2_mf = (1 - forgetting_rate) * q_stage2_mf + forgetting_rate * 0.5

        # Forgetting for unchosen stage 1 action
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1 - forgetting_rate) * q_stage1_mf[unchosen_a1] + forgetting_rate * 0.5

    eps = 1e-10
