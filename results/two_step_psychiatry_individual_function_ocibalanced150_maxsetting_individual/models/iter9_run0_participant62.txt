import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF Q-learning with outcome-dependent stickiness.
    This model separates the stickiness (perseveration) based on whether the 
    previous trial was rewarded or not, capturing "win-stay" vs "lose-stay" 
    tendencies. Also includes decay for unchosen options.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - weight for model-based vs model-free
    stick_rewarded: [0, 10] - choice stickiness after a rewarded trial (reward > 0)
    stick_unrewarded: [0, 10] - choice stickiness after an unrewarded trial (reward <= 0)
    decay: [0, 1] - decay rate for unchosen options
    """
    alpha, beta, w, stick_rewarded, stick_unrewarded, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vec_1[last_action_1] = 1.0
            
        current_stickiness = stick_rewarded if last_reward > 0 else stick_unrewarded
        
        logits_1 = beta * q_net + current_stickiness * stick_vec_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stick_vec_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vec_2[last_action_2] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + current_stickiness * stick_vec_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay) 
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay) 
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * delta_stage2 
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF Q-learning with loss aversion.
    Rewards of -1 are scaled by a loss_aversion parameter to capture 
    differential sensitivity to negative outcomes. Includes choice stickiness 
    and decay for unchosen options.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - weight for model-based vs model-free
    stickiness: [0, 10] - choice stickiness parameter
    decay: [0, 1] - decay rate for unchosen options
    loss_aversion: [0, 10] - multiplier for negative rewards (-1)
    """
    alpha, beta, w, stickiness, decay, loss_aversion = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vec_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + stickiness * stick_vec_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stick_vec_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vec_2[last_action_2] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness * stick_vec_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay) 
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Apply loss aversion to negative rewards
        eff_reward = reward[trial]
        if eff_reward < 0:
            eff_reward = eff_reward * loss_aversion
            
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay) 
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * delta_stage2 
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF Q-learning with state-specific stickiness in Stage 2.
    This model uses a stickiness parameter for Stage 1, and a state-specific 
    stickiness for Stage 2, meaning the participant is likely to repeat the

    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - weight for model-based vs model-free

    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:

        q_stage1_mf[action_1[trial]] += alpha * delta_stage2 
        
        last_action_1 = action_1[trial]

    eps = 1e-10
