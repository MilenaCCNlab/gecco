Here are 3 new cognitive models that explore alternative mechanisms and parameter combinations to explain the participant's behavior. 

### Model 1: Daw et al. with Stage 1 and Stage 2 Stickiness
This model explores the idea that perseveration (stickiness) might occur not just at the first stage (spaceship choice) but also at the second stage (alien choice). Given the participant's sequences of repeated alien choices, adding a `stickiness_2` parameter captures motor or stimulus perseveration in Stage 2.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 1: Daw et al. model with stickiness for both Stage 1 and Stage 2 choices.
    This model captures perseveration at both stages of the decision-making process.
    
    Parameters and Bounds:
    learning_rate: [0, 1] - Learning rate for updating MF values.
    beta_1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness_1: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    stickiness_2: [-5, 5] - Bonus added to the previously chosen Stage 2 action.
    lambda_: [0, 1] - Eligibility trace scaling Stage 2 PE updates to Stage 1.
    """
    learning_rate, beta_1, beta_2, w, stickiness_1, stickiness_2, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus_1[last_action_1] = stickiness_1
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus_1
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 Policy
        state_idx = state[trial]
        
        stickiness_bonus_2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness_bonus_2[last_action_2] = stickiness_2
            
        q2_total = beta_2 * q_stage2_mf[state_idx] + stickiness_bonus_2
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Learning Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Daw et al. with Outcome-Dependent Stickiness (Win-Stay / Lose-Shift)
Instead of a flat perseveration bonus, this model assumes the participant's tendency to stick with the same spaceship depends on whether they were rewarded on the previous trial. This captures win-stay/lose-shift heuristics interacting with model-based and model-free learning.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 2: Daw et al. model with outcome-dependent stickiness (Win-Stay/Lose-Shift).
    The stickiness bonus for Stage 1 depends on whether the previous trial was rewarded.
    
    Parameters and Bounds:
    learning_rate: [0, 1] - Learning rate for updating MF values.
    beta_1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness_win: [-5, 5] - Bonus to the previously chosen Stage 1 action if previously rewarded.
    stickiness_lose: [-5, 5] - Bonus to the previously chosen Stage 1 action if previously unrewarded.
    lambda_: [0, 1] - Eligibility trace scaling Stage 2 PE updates to Stage 1.
    """
    learning_rate, beta_1, beta_2, w, stickiness_win, stickiness_lose, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_bonus[last_action_1] = stickiness_win
            elif last_reward == 0:
                stickiness_bonus[last_action_1] = stickiness_lose
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 Policy
        state_idx = state[trial]
        
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Learning Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates combined with Memory Decay
This model combines the best features of asymmetric learning (where humans often update expectations differently for better-than-expected vs. worse-than-expected outcomes) and memory decay. This combination (8 parameters) can capture nuanced value drifting when outcomes are volatile.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 3: Full Daw et al. model with asymmetric learning rates for positive and negative 
    prediction errors, combined with memory decay (forgetting) for all actions.
    
    Parameters and Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    decay_rate: [0, 1] - Rate at which Q-values decay towards 0.5.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    lambda_: [0, 1] - Eligibility trace.
    """
    alpha_pos, alpha_neg, decay_rate, beta_1, beta_2, w, stickiness, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = stickiness
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 Policy
        state_idx = state[trial]
        
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay before update
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        # Learning Updates with Asymmetric LRs
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
