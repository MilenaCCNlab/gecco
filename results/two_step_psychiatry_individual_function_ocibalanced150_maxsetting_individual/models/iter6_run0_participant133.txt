def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Daw (2011) model with eligibility trace, separate inverse temperatures for stage 1 and stage 2,
    model-based weight, and stage 1 stickiness.
    
    Parameters/Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stickiness: [-5, 5] - bonus for repeating the previous stage 1 action
    lambda_trace: [0, 1] - eligibility trace
    """
    alpha, beta_1, beta_2, w, stickiness, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        q_vals_1 = beta_1 * q_net
        if last_action != -1:
            q_vals_1[last_action] += stickiness
            
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_vals_2 = beta_2 * q_stage2_mf[state_idx]
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_trace * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate inverse temperatures and stickiness parameters for both stages, 
    plus an eligibility trace and model-based weight.
    
    Parameters/Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stick_1: [-5, 5] - stage 1 stickiness
    stick_2: [-5, 5] - stage 2 stickiness
    lambda_trace: [0, 1] - eligibility trace
    """
    alpha, beta_1, beta_2, w, stick_1, stick_2, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        q_vals_1 = beta_1 * q_net
        if last_action_1 != -1:
            q_vals_1[last_action_1] += stick_1

        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_vals_2 = beta_2 * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            q_vals_2[last_action_2[state_idx]] += stick_2
            
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lambda_trace * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates for positive and negative prediction errors,
    separate inverse temperatures for each stage, and separate stickiness for both stages.
    
    Parameters/Bounds:
    alpha_pos: [0, 1] - learning rate for positive RPE
    alpha_neg: [0, 1] - learning rate for negative RPE
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stick_1: [-5, 5] - stage 1 stickiness
    stick_2: [-5, 5] - stage 2 stickiness
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, stick_1, stick_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        q_vals_1 = beta_1 * q_net
        if last_action_1 != -1:
            q_vals_1[last_action_1] += stick_1
            
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_vals_2 = beta_2 * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            q_vals_2[last_action_2[state_idx]] += stick_2
            
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
        
        # update stage 1 MF value with stage 2 RPE (implicit lambda=1)
        q_stage1_mf[action_1[trial]] += alpha_stage2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
