def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with Q-value decay for unchosen options and perseveration at both decision stages.
    This model assumes memory decay for unchosen actions and that participants might have a tendency 
    to repeat their previous choices at both the first (spaceship) and second (alien) stages.
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers_1: [-5,5]
    pers_2: [-5,5]
    decay: [0,1]
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + pers_1 * pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2 != -1:
            pers_bonus_2[last_action_2] = 1.0
            
        logits_2 = beta_2 * q_stage2_mf[s] + pers_2 * pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2

        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Decay for unchosen options
        for act1 in range(2):
            if act1 != a1:
                q_stage1_mf[act1] *= (1 - decay)
                
        for st2 in range(2):
            for act2 in range(2):
                if not (st2 == s and act2 == a2):
                    q_stage2_mf[st2, act2] *= (1 - decay)
        
        last_action_1 = a1
        last_action_2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with transition matrix learning and Q-value decay for unchosen options.
    This model assumes participants learn the transition probabilities dynamically rather 
    than having fixed knowledge of them, while also experiencing memory decay for unchosen actions.
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers: [-5,5]
    alpha_trans: [0,1]
    decay: [0,1]
    """
    alpha, beta_1, beta_2, w, lam, pers, alpha_trans, decay = model_parameters
    n_trials = len(action_1)
  
    # Initialize learned transition matrix (starts uniform)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + pers * pers_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Update transition matrix
        transition_matrix[a1, s] += alpha_trans * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_trans * (0.0 - transition_matrix[a1, 1 - s])

        # Decay for unchosen options
        for act1 in range(2):
            if act1 != a1:
                q_stage1_mf[act1] *= (1 - decay)
                
        for st2 in range(2):
            for act2 in range(2):
                if not (st2 == s and act2 == a2):
                    q_stage2_mf[st2, act2] *= (1 - decay)
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with distinct learning rates for positive and negative prediction errors,
    combined with Q-value decay for unchosen options. Captures asymmetric learning from 
    better-than-expected versus worse-than-expected outcomes alongside memory decay.
    
    Bounds:
    alpha_pos: [0,1]
    alpha_neg: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers: [-5,5]
    decay: [0,1]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]

        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg

        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[a1] += alpha_1 * delta_stage1 + alpha_2 * lam * delta_stage2

        q_stage2_mf[s, a2] += alpha_2 * delta_stage2

        # Decay for unchosen options
        for act1 in range(2):

                q_stage1_mf[act1] *= (1 - decay)
                
        for st2 in range(2):
            for act2 in range(2):
                if not (st2 == s and act2 == a2):

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
