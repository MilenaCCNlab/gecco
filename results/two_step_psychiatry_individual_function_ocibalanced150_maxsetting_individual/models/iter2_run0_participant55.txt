Here are 3 new cognitive models that build upon the baseline and best-performing model mechanisms. They explore different facets of learning and decision-making, such as eligibility traces (Daw et al., 2011), asymmetric learning from positive/negative prediction errors, and memory decay for unchosen options.

### Model 1: Comprehensive Daw et al. (2011) Model
This model introduces an eligibility trace (`lambda_`) that allows the reward prediction error from the second stage to directly update the model-free value of the first-stage action. It also features separate inverse temperatures (`beta1`, `beta2`) for stage 1 and stage 2 to capture differences in choice stochasticity between the two stages.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Comprehensive two-step model (Daw et al., 2011) with eligibility trace and separate inverse temperatures.
    
    Bounds:
    learning_rate: [0, 1]
    w: [0, 1]
    lambda_: [0, 1]
    stickiness: [-5, 5]
    beta1: [0, 10]
    beta2: [0, 10]
    """
    learning_rate, w, lambda_, stickiness, beta1, beta2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_net[last_action] += stickiness
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta1 * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # policy for the second choice
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1 based on stage 2 reward
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates Model
In many learning tasks, individuals update their expectations differently depending on whether the outcome was better or worse than expected. This model implements two distinct learning rates (`alpha_pos` and `alpha_neg`) for positive and negative reward prediction errors.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with asymmetric learning rates for positive and negative prediction errors.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    w: [0, 1]
    stickiness: [-5, 5]
    beta: [0, 10]
    """
    alpha_pos, alpha_neg, w, stickiness, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_net[last_action] += stickiness
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # policy for the second choice
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        if delta_stage1 > 0:
            q_stage1_mf[action_1[trial]] += alpha_pos * delta_stage1
        else:
            q_stage1_mf[action_1[trial]] += alpha_neg * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_pos * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_neg * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Unchosen Q-Value Decay Model
This model accounts for the possibility that the participant slowly forgets the values of the aliens and spaceships they haven't chosen recently. We introduce a `decay_rate` parameter that progressively pulls the Q-values of unchosen/unvisited states and actions back to 0.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with decay of unchosen Q-values to model forgetting over time.
    
    Bounds:
    learning_rate: [0, 1]
    decay_rate: [0, 1]
    w: [0, 1]
    stickiness: [-5, 5]
    beta: [0, 10]
    """
    learning_rate, decay_rate, w, stickiness, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_net[last_action] += stickiness
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # policy for the second choice
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Decay unchosen stage 1 action
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - decay_rate)

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Decay unchosen stage 2 actions
        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] *= (1 - decay_rate)
        
        # Decay actions in the unvisited state
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay_rate)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay_rate)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
