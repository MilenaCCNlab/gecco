Here are 3 new cognitive models that explore alternative mechanisms not yet tested, focusing on asymmetric learning, outcome-dependent stickiness (win-stay/lose-shift), and combinations with forgetting (decay).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates with Forgetting (Decay).
    Uses different learning rates for positive and negative prediction errors,
    combined with Q-value decay (forgetting) over time.
    
    Parameters:
    alpha_pos: Learning rate for positive prediction errors [0, 1]
    alpha_neg: Learning rate for negative prediction errors [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Forgetting rate for all Q-values per trial [0, 1]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 Choice
        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating for Stage 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # Action Value Updating for Stage 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility Trace Update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2

        # Q-value Decay (Forgetting)
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Win-Stay / Lose-Shift Stickiness.
    Differentiates stage 1 choice stickiness depending on whether the previous
    trial was rewarded (st_win) or unrewarded (st_lose).
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    st_win: Choice stickiness for stage 1 after a rewarded trial [-10, 10]
    st_lose: Choice stickiness for stage 1 after an unrewarded trial [-10, 10]
    """
    alpha, beta1, beta2, w, lam, st_win, st_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_r = 0

    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net
        if prev_a1 != -1:
            if prev_r > 0:
                logit1[prev_a1] += st_win
            else:
                logit1[prev_a1] += st_lose
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 Choice
        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Track previous actions and rewards
        prev_a1 = action_1[trial]
        prev_r = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
