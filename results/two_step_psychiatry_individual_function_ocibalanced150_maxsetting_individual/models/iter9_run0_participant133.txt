def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates (alpha_pos, alpha_neg) for positive and negative RPEs,
    separate stickiness parameters for stage 1 and stage 2 choices, and a value decay parameter.
    This captures potential differences in how participants learn from gains versus omissions,
    and allows for perseveration at both the spaceship and alien selection stages.
    
    Parameters/Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta: [0, 10] - inverse temperature for both stages
    w: [0, 1] - model-based weight
    stick_1: [-5, 5] - bonus for repeating the previous stage 1 action
    stick_2: [-5, 5] - bonus for repeating the previous stage 2 action
    decay: [0, 1] - decay rate for all action values towards 0.5
    """
    alpha_pos, alpha_neg, beta, w, stick_1, stick_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_bonus_1[last_action_1] = stick_1
            
        q_vals_1 = beta * q_net + stick_bonus_1
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stick_bonus_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_bonus_2[last_action_2] = stick_2
            
        q_vals_2 = beta * q_stage2_mf[state_idx] + stick_bonus_2
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay before updating
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace for stage 1 based on stage 2 reward
        q_stage1_mf[action_1[trial]] += alpha_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model where the model-based weight (w) depends dynamically on whether the previous trial 
    was rewarded or not. This tests the hypothesis that participants might rely more heavily 
    on model-based planning after a loss (to correct behavior) or after a win (reinforcement).
    
    Parameters/Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w_win: [0, 1] - model-based weight after a rewarded trial
    w_lose: [0, 1] - model-based weight after an unrewarded trial
    stickiness: [-5, 5] - bonus for repeating the previous stage 1 action
    decay: [0, 1] - decay rate for all action values towards 0.5
    """
    alpha, beta, w_win, w_lose, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    last_reward = 1 # Default to win for the first trial

    for trial in range(n_trials):

        # Determine w based on last reward
        w = w_win if last_reward == 1 else w_lose

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus = np.zeros(2)
        if last_action != -1:
            stick_bonus[last_action] = stickiness
            
        q_vals_1 = beta * q_net + stick_bonus
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_vals_2 = beta * q_stage2_mf[state_idx]
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay before updating
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace for stage 1 based on stage 2 reward
        q_stage1_mf[action_1[trial]] += alpha * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with completely separate learning rates, inverse temperatures, and decay parameters 
    for stage 1 and stage 2. This isolates mechanisms for the abstract spaceship choice versus 
    the concrete alien choice, allowing for distinct levels of forgetting and exploration.
    
    Parameters/Bounds:
    alpha_1: [0, 1] - learning rate for stage 1
    alpha_2: [0, 1] - learning rate for stage 2
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stickiness: [-5, 5] - bonus for repeating the previous stage 1 action
    decay_1: [0, 1] - decay rate for stage 1 action values towards 0.5
    decay_2: [0, 1] - decay rate for stage 2 action values towards 0.5
    """
    alpha_1, alpha_2, beta_1, beta_2, w, stickiness, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
