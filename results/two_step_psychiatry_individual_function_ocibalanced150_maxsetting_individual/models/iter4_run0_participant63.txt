def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Asymmetric learning rates for positive/negative PEs with Decay and Stickiness.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta: [0, 10] - Inverse temperature
    w: [0, 1] - Model-based weight
    p: [0, 10] - Choice stickiness for stage 1
    decay: [0, 1] - Decay rate for unchosen options towards 0.5
    """
    alpha_pos, alpha_neg, beta, w, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = p
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)
        
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2
        q_stage1_mf[a1] += alpha_s2 * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
