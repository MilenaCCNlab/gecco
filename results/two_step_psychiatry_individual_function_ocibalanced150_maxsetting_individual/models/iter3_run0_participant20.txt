Here are 3 new cognitive models that explore different mechanisms to explain the participant's decision-making behavior.

### Model 1: Outcome-Dependent Perseveration (Win-Stay / Lose-Shift)
This model extends the baseline by adding an outcome-dependent perseveration mechanism at Stage 1. Instead of a general tendency to repeat the previous choice, the participant's likelihood of repeating a choice depends on whether that choice was rewarded (`stay_win`) or unrewarded (`stay_lose`) on the previous trial.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with outcome-dependent perseveration at Stage 1 (Win-stay, Lose-shift).
    
    Bounds:
    alpha: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stay_win: [-5, 5]
    stay_lose: [-5, 5]
    """
    alpha, beta1, beta2, w, lam, stay_win, stay_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1
    prev_r = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if prev_a1 != -1:
            if prev_r > 0:
                logits_1[prev_a1] += stay_win
            else:
                logits_1[prev_a1] += stay_lose
                
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition Matrix Learning + Value Decay
Instead of assuming the participant knows the fixed 0.7/0.3 transition probabilities, this model assumes the participant learns the transition matrix over time via a separate learning rate (`alpha_tr`). It also implements value decay to 0.5 for unchosen options to account for forgetting.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with a learned transition matrix and value decay for unchosen actions.
    
    Bounds:
    alpha: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    alpha_tr: [0, 1]
    decay: [0, 1]
    """
    alpha, beta1, beta2, w, lam, alpha_tr, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Update transition matrix based on observed state
        transition_matrix[a1, s] = (1.0 - alpha_tr) * transition_matrix[a1, s] + alpha_tr * 1.0
        transition_matrix[a1, 1 - s] = (1.0 - alpha_tr) * transition_matrix[a1, 1 - s] + alpha_tr * 0.0

        # Decay unchosen options towards neutral expectation
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay) * q_stage2_mf[state_idx, action_idx] + decay * 0.5

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Exponentially Decaying Choice Trace (Habitual Stickiness)
This model replaces the simple previous-trial perseveration parameter with a cumulative, continuous choice trace. The trace decays over trials by `pers_decay`, tracking longer-term choice habits rather than just single-trial perseveration.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with an exponentially decaying choice trace for Stage 1 perseveration.
    
    Bounds:
    alpha: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pers: [-5, 5]
    pers_decay: [0, 1]
    """
    alpha, beta1, beta2, w, lam, pers, pers_decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    choice_trace = np.zeros(2)
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        # Add the perseveration/habit trace directly to the logits
        logits_1 = beta1 * q_net + pers * choice_trace
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Decay the trace and increment it for the chosen action
        choice_trace = pers_decay * choice_trace
        choice_trace[a1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
