def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Direct Model-Free Update with Asymmetric Learning, Decay, and Stage 1 & 2 Stickiness.
    Uses separate learning rates for positive and negative prediction errors (reward vs. no reward),
    along with decay for unchosen options, and stickiness for both choice stages.
    
    Bounds:
    alpha_pos: [0,1] (Learning rate for positive PE)
    alpha_neg: [0,1] (Learning rate for negative PE)
    decay: [0,1] (Decay rate for unchosen options)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    pi_1: [-5,5] (Choice stickiness for stage 1)
    pi_2: [-5,5] (Choice stickiness for stage 2)
    """
    alpha_pos, alpha_neg, decay, beta_1, beta_2, w, pi_1, pi_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep_1 = np.zeros(2)
        if trial > 0:
            rep_1[action_1[trial-1]] = 1
            
        logits_1 = beta_1 * q_net + pi_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        rep_2 = np.zeros(2)
        if trial > 0:
            rep_2[action_2[trial-1]] = 1
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + pi_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        pe_1 = reward[trial] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if pe_1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * pe_1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Direct Model-Free Update with Asymmetric Learning, Decay, and Outcome-Dependent Stickiness.
    Uses separate learning rates for positive and negative prediction errors,
    decay for unchosen options, and choice stickiness that depends on the previous trial's reward (win-stay/lose-shift).
    
    Bounds:
    alpha_pos: [0,1] (Learning rate for positive PE)
    alpha_neg: [0,1] (Learning rate for negative PE)
    decay: [0,1] (Decay rate for unchosen options)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    pi_win: [-5,5] (Stickiness after a rewarded trial)
    pi_lose: [-5,5] (Stickiness after an unrewarded trial)
    """
    alpha_pos, alpha_neg, decay, beta_1, beta_2, w, pi_win, pi_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if trial > 0:
            if reward[trial-1] > 0:
                rep[action_1[trial-1]] = pi_win
            else:
                rep[action_1[trial-1]] = pi_lose
            
        logits_1 = beta_1 * q_net + rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        pe_1 = reward[trial] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if pe_1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * pe_1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition Learning with Asymmetric Reward Learning and Eligibility Trace.
    Learns the transition matrix dynamically using alpha_tr, updates reward values using 
    standard TD with eligibility trace and asymmetric learning rates for positive/negative PEs.
    
    Bounds:
    alpha_pos: [0,1] (Learning rate for positive reward PE)
    alpha_neg: [0,1] (Learning rate for negative reward PE)
    alpha_tr: [0,1] (Learning rate for transition matrix)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    lam: [0,1] (Eligibility trace)
    pi: [-5,5] (Choice stickiness for stage 1)
    """
    alpha_pos, alpha_neg, alpha_tr, beta_1, beta_2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix based on instructions
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if trial > 0:
            rep[action_1[trial-1]] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2

        # Transition matrix update based on experienced transition
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0 - transition_matrix[action_1[trial], 1 - state_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
