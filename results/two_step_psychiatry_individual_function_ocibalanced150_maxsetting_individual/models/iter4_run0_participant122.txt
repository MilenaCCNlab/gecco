Here are three new cognitive models that propose different mechanisms for how participants make decisions in the two-step task. Each model avoids the previously tried parameter combinations and explores novel psychological mechanisms.

### Model 1: Reward-dependent Model-Based Weight
This model tests the hypothesis that a participant's reliance on model-based planning fluctuates depending on recent success. Specifically, the weight placed on the model-based system ($w$) changes dynamically based on whether the previous trial was rewarded or unrewarded.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid model with reward-dependent model-based weight (w).
    The weight on the model-based system depends on whether the previous 
    trial was rewarded or unrewarded, reflecting changes in strategy after success/failure.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w_rew: [0,1] - weight on model-based strategy following a reward
    w_unrew: [0,1] - weight on model-based strategy following no reward
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter linking stage 2 reward to stage 1
    """
    import numpy as np
    learning_rate, beta, w_rew, w_unrew, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        current_w = w_rew if last_reward > 0 else w_unrew
        q_stage1_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        if last_action != -1:
            q_stage1_net[last_action] += p
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        # Updates
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Extended Habit Trace (Autocorrelation)
This model tests whether participants exhibit longer-term choice autocorrelation instead of just a simple lag-1 perseveration. It replaces the single-trial perseveration parameter with a decaying "habit vector" that captures a longer history of past choices.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with an extended habit trace (autocorrelation).
    Instead of just repeating the immediate last action, this model maintains 
    a habit vector for Stage 1 actions that updates and decays over time, 
    capturing longer-term choice autocorrelation.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    lam: [0,1] - eligibility trace parameter linking stage 2 reward to stage 1
    alpha_habit: [0,1] - update rate (decay) for the habit trace
    beta_habit: [-10,10] - weight of the habit trace on Stage 1 choices
    """
    import numpy as np
    learning_rate, beta, w, lam, alpha_habit, beta_habit = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    habit_1 = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf + beta_habit * habit_1
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # policy for the second choice
        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
            
        # Updates
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
        # Update habit trace
        habit_1 = (1 - alpha_habit) * habit_1
        habit_1[action_1[trial]] += alpha_habit
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-dependent Eligibility Traces
This model hypothesizes that the backward updating of stage 1 values via the eligibility trace is asymmetric. A participant might be highly effective at assigning credit to stage 1 when they receive a reward, but fail to properly assign credit (or do so at a different rate) when they do not.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with reward-dependent eligibility traces.
    The eligibility trace (lambda) that updates Stage 1 based on Stage 2 outcomes
    differs depending on whether the outcome was rewarded or unrewarded, allowing 
    asymmetric credit assignment for positive vs negative outcomes.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam_rew: [0,1] - eligibility trace parameter when reward is positive
    lam_unrew: [0,1] - eligibility trace parameter when reward is zero or negative
    """
    import numpy as np
    learning_rate, beta, w, p, lam_rew, lam_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += p
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # policy for the second choice
        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
            
        # Updates
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
            current_lam = lam_rew if reward[trial] > 0 else lam_unrew
            q_stage1_mf[action_1[trial]] += learning_rate * current_lam * delta_stage2
            
        last_action = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
