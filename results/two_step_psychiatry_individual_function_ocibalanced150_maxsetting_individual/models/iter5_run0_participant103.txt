Here are 3 new cognitive models that build upon the successful mechanisms identified in previous iterations, while introducing novel psychological features. 

### Model 1: Asymmetric Learning Rates with Forgetting and Stickiness
This model uses the successful base of Q-value decay and stage 1 choice perseveration (stickiness), but introduces asymmetric learning rates for positive and negative reward prediction errors (RPEs). Participants often update their beliefs differently depending on whether outcomes are better or worse than expected.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates with Forgetting and Stickiness.
    Combines positive/negative RPE asymmetric learning rates with Q-value decay 
    and choice perseveration at stage 1.
    
    Parameters:
    alpha_pos: Learning rate for positive RPEs [0, 1]
    alpha_neg: Learning rate for negative RPEs [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Forgetting rate for all Q-values per trial [0, 1]
    st: Choice stickiness for stage 1 [-10, 10]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, decay, st = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net
        if prev_a1 != -1:
            logit1[prev_a1] += st
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Stage 1 Update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # Stage 2 Update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility Trace Update
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2

        # Decay all Q-values
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
        prev_a1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unchosen Option Decay with Stage 1 Stickiness
Instead of decaying all Q-values equally (which simulates general memory leak), this model assumes that memory decay primarily affects the options that were *not* chosen on the current trial. The chosen options are rehearsed/updated and thus protected from decay on that specific trial.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Unchosen Option Decay with Stage 1 Stickiness.
    Applies forgetting (decay) only to the Q-values of actions that were NOT chosen 
    on the current trial, alongside standard stage 1 stickiness.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    unchosen_decay: Forgetting rate applied only to unchosen Q-values [0, 1]
    st: Choice stickiness for stage 1 [-10, 10]
    """
    alpha, beta1, beta2, w, lam, unchosen_decay, st = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net
        if prev_a1 != -1:
            logit1[prev_a1] += st
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Unchosen option decay
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1 - unchosen_decay)
                
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - unchosen_decay)
        
        prev_a1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Choice Trace Stickiness
In previous models, choice stickiness only considered the *immediately preceding trial*. This model introduces a persistent **Choice Trace** that tracks the participant's history of choices via an exponential moving average. This allows a strong habit to build up if a participant repeatedly chooses the same spaceship across many trials.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Choice Trace Stickiness.
    Maintains an exponentially updating choice trace for stage 1 actions,
    allowing stickiness to build up over multiple consecutive identical choices
    rather than just relying on the single previous trial.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Forgetting rate for all Q-values per trial [0, 1]
    st: Stickiness weight applied to the choice trace [-10, 10]
    alpha_c: Update rate for the choice trace [0, 1]
    """
    alpha, beta1, beta2, w, lam, decay, st, alpha_c = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        # Integrate the choice trace into the stage 1 policy
        logit1 = beta1 * q_net + st * choice_trace
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Standard Q-value decay
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
        # Update choice trace for the next trial
        for a in range(2):
            if a == action_1[trial]:
                choice_trace[a] = choice_trace[a] + alpha_c * (1.0 - choice_trace[a])
            else:
                choice_trace[a] = choice_trace[a] * (1.0 - alpha_c)
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
