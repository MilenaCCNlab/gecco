Here are 3 new cognitive models based on the participant's data. 

### Rationale for the Models
A careful analysis of the participant's choices reveals a striking pattern in Stage 2: the participant often repeatedly chooses the exact same alien index (e.g., Alien 0) across consecutive trials, *even when the planet (state) changes*. Standard models treat Alien 0 on Planet 0 and Alien 0 on Planet 1 as completely independent, which fails to capture this strong generalization or "spillover" effect across states. 

To account for this, the proposed models introduce mechanisms to capture this cross-state generalization:
- **Model 1** replaces the standard state-specific Stage 2 perseveration with a **general Stage 2 perseveration** parameter (`p_2_gen`), meaning the participant tends to repeat the last Stage 2 action index regardless of which planet they are on. It also retains the successful Q-value decay mechanism from previous top models.
- **Model 2** tests whether both types of perseveration coexist by including **both state-specific and general Stage 2 perseveration parameters** (`p_2_spec` and `p_2_gen`), allowing them to add up or compete.
- **Model 3** approaches the same phenomenon through a value-learning mechanism rather than choice stickiness. It introduces a **Q-value generalization parameter** (`gen`). When the participant receives a reward from an alien, they also update the value of the alien with the same index on the *other* planet.

import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-step model with stage 1 perseveration, general stage 2 perseveration 
    (action index regardless of state), and unchosen Q-value decay.
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    p_1: [-5,5]
    p_2_gen: [-5,5] (general stage 2 perseveration across states)
    decay: [0,1]
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2_gen, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_any = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            persev_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p_1 * persev_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        persev_2 = np.zeros(2)
        if last_action_2_any != -1:
            persev_2[last_action_2_any] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2_gen * persev_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay unchosen options
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1 - decay)
                
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != action_2[trial]:
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action_1 = action_1[trial]
        last_action_2_any = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-step model with stage 1 perseveration, and BOTH state-specific 
    and general stage 2 perseveration.
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    p_1: [-5,5]
    p_2_spec: [-5,5] (state-specific stage 2 perseveration)
    p_2_gen: [-5,5] (general stage 2 perseveration across states)
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2_spec, p_2_gen = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_spec = np.full(2, -1)
    last_action_2_any = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            persev_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p_1 * persev_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        persev_2_spec = np.zeros(2)
        if last_action_2_spec[state_idx] != -1:
            persev_2_spec[last_action_2_spec[state_idx]] = 1.0
            
        persev_2_gen = np.zeros(2)
        if last_action_2_any != -1:
            persev_2_gen[last_action_2_any] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2_spec * persev_2_spec + p_2_gen * persev_2_gen)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2_spec[state_idx] = action_2[trial]
        last_action_2_any = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-step model with Q-value generalization across states in stage 2.
    When a stage 2 action is rewarded/punished, the value of the same action index 
    in the other state is also updated based on the generalization parameter.
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    p_1: [-5,5]
    p_2: [-5,5]
    gen: [0,1] (generalization rate to the other state)
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2, gen = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            persev_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p_1 * persev_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        persev_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            persev_2[last_action_2[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2 * persev_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Generalization to the other state's corresponding alien
        other_state = 1 - state_idx
        delta_stage2_other = reward[trial] - q_stage2_mf[other_state, action_2[trial]]
        q_stage2_mf[other_state, action_2[trial]] += gen * alpha * delta_stage2_other
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
