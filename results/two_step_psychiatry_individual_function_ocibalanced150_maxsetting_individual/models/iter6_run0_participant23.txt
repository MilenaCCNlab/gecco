Here are 3 new cognitive models that build upon your best model so far (which features forgetting/decay of unchosen actions). These models introduce well-documented decision-making mechanisms such as choice perseveration (stickiness), asymmetric learning rates for positive vs. negative prediction errors, and stage-specific learning and temperature parameters.

### Model 1: Hybrid Model with Decay and Two-Stage Perseveration (Stickiness)
This model adds choice perseveration (stickiness) for both the first and second stage choices. Participants often show a tendency to repeat their previous actions independently of the reward outcomes.
```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Hybrid Model with decay, and separate stickiness for stage 1 and stage 2.
    Bounds:
    alpha: [0,1]
    beta: [0,10]
    w: [0,1]
    lmbda: [0,1]
    decay: [0,1]
    st1: [-5,5]
    st2: [-5,5]
    """
    alpha, beta, w, lmbda, decay, st1, st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_1 = np.zeros(2)
        if last_a1 != -1:
            stick_1[last_a1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st1 * stick_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick_2 = np.zeros(2)
        if last_a2 != -1:
            stick_2[last_a2] = 1.0
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + st2 * stick_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # action value updating
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lmbda * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_a1 = action_1[trial]
        last_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid Model with Decay, Asymmetric Learning, and Stage 1 Stickiness
This model adds separate learning rates for positive and negative prediction errors (a common finding in reinforcement learning highlighting distinct learning mechanisms for better-than-expected vs. worse-than-expected outcomes) along with choice stickiness at stage 1.
```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Hybrid Model with decay, separate learning rates for positive/negative PEs, and stage 1 stickiness.
    Bounds:
    alpha_pos: [0,1]
    alpha_neg: [0,1]
    beta: [0,10]
    w: [0,1]
    lmbda: [0,1]
    decay: [0,1]
    st: [-5,5]
    """
    alpha_pos, alpha_neg, beta, w, lmbda, decay, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_1 = np.zeros(2)
        if last_a1 != -1:
            stick_1[last_a1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st * stick_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg

        # action value updating
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1 + alpha_1 * lmbda * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid Model with Decay, Stage-Specific Parameters, and Stage 1 Stickiness
This model parses out learning rates and decision temperatures for stage 1 and stage 2 separately (since the cognitive load and certainty level usually differ across stages), along with choice stickiness at stage 1.

    Bounds:

    alpha2: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    w: [0,1]

    st: [-5,5]
