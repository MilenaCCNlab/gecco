def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with a single learning rate, unchosen option decay, shared inverse temperature,
    model-based weight, eligibility trace, stage 1 and stage 2 perseveration, and stage 1 bias.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    decay: [0, 1] - Decay rate for unchosen actions (towards 0.5).
    beta: [0, 10] - Shared inverse temperature.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p_1: [0, 5] - Perseveration for stage 1.
    p_2: [0, 5] - Perseveration for stage 2.
    bias_1: [-5, 5] - Bias for choosing spaceship 0 in stage 1.
    """
    alpha, decay, beta, w, lam, p_1, p_2, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_a1 != -1:
            pers_1[last_a1] = 1
            
        bias = np.array([bias_1, 0.0])
            
        logits_1 = beta * q_stage1_net + p_1 * pers_1 + bias
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        pers_2 = np.zeros(2)
        if last_a2 != -1:
            pers_2[last_a2] = 1

        logits_2 = beta * q_stage2_mf[state_idx] + p_2 * pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_a1 = a1
        last_a2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with single learning rate, unchosen option decay, shared inverse temperature,
    model-based weight, eligibility trace, outcome-dependent stage 1 perseveration, and stage 1 bias.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    decay: [0, 1] - Decay rate for unchosen actions (towards 0.5).
    beta: [0, 10] - Shared inverse temperature.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p_win: [-5, 5] - Perseveration for stage 1 if previous trial was rewarded.
    p_lose: [-5, 5] - Perseveration for stage 1 if previous trial was unrewarded.
    bias_1: [-5, 5] - Bias for choosing spaceship 0 in stage 1.
    """
    alpha, decay, beta, w, lam, p_win, p_lose, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_r = -1

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy

        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_a1 != -1:
            if last_r == 1:
                pers_1[last_a1] = p_win
            else:
                pers_1[last_a1] = p_lose
            
        bias = np.array([bias_1, 0.0])
            
        logits_1 = beta * q_stage1_net + pers_1 + bias
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_a1 = a1
        last_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
