Here are 3 new cognitive models based on the participant data and avoiding the previously explored parameter configurations.

### Model 1: Stage-Specific Asymmetric Learning Rates
This model explores the possibility that participants update their expectations differently for positive versus negative prediction errors, and that this asymmetry may differ between the first stage (spaceship choice) and the second stage (alien choice). 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate positive and negative learning rates for Stage 1 and Stage 2.
    
    Parameters:
    alpha1_pos: [0, 1] - Stage 1 learning rate for positive prediction errors.
    alpha1_neg: [0, 1] - Stage 1 learning rate for negative prediction errors.
    alpha2_pos: [0, 1] - Stage 2 learning rate for positive prediction errors.
    alpha2_neg: [0, 1] - Stage 2 learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness for stage 1.
    """
    alpha1_pos, alpha1_neg, alpha2_pos, alpha2_neg, beta, w, lam, pi1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # update stage 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha1_pos if delta_stage1 > 0 else alpha1_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # update stage 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha2_pos if delta_stage2 > 0 else alpha2_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # eligibility trace
        q_stage1_mf[action_1[trial]] += lr1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Stickiness & Stage-Specific Temperatures
Participants frequently exhibit simple win-stay/lose-shift heuristics. This model introduces two separate stickiness parameters for Stage 1 depending on whether the previous trial was rewarded (`pi1_win`) or unrewarded (`pi1_lose`). It also isolates choice stochasticity at the two stages using `beta1` and `beta2`.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with reward-dependent Stage 1 stickiness and separate betas.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Mixing weight for model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pi1_win: [-5, 5] - Stage 1 stickiness after a rewarded trial.
    pi1_lose: [-5, 5] - Stage 1 stickiness after an unrewarded trial.
    pi2: [-5, 5] - Stage 2 stickiness.
    """
    alpha, beta1, beta2, w, lam, pi1_win, pi1_lose, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        current_pi1 = 0
        if last_reward == 1:
            current_pi1 = pi1_win
        elif last_reward == 0:
            current_pi1 = pi1_lose
            
        exp_q1 = np.exp(beta1 * q_stage1_net + current_pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Learned Transition Probabilities
Building off of the best model logic, this model relaxes the assumption that participants hold perfect knowledge of the static 0.7/0.3 transition probabilities. Instead, it assumes participants learn the transition contingencies experientially over time via a transition learning rate (`eta`), starting from a uniform prior (0.5/0.5).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with learned transition probabilities.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Stage 1 choice stickiness.
    pi2: [-5, 5] - Stage 2 choice stickiness.
    eta: [0, 1] - Learning rate for transition probabilities.
    """
    alpha_pos, alpha_neg, beta, w, lam, pi1, pi2, eta = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to 0.5 (uniform prior)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]

        # Update transition probabilities based on observation
        transition_matrix[action_1[trial], state_idx] += eta * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += eta * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
