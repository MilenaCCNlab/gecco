def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Free and Model-Based model with separate stage inverse temperatures.
    
    This model computes a weighted combination of model-based and model-free values 
    for the first-stage choice. It assumes separate inverse temperatures (betas) 
    for the first and second stages, but no choice stickiness or eligibility traces.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updates.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully model-based).
    """
    alpha, beta1, beta2, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        q1_vals -= np.max(q1_vals)  # scale to prevent overflow
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model-Free and Model-Based model with choice stickiness.
    
    Builds upon Model 1 by adding a perseveration parameter (pi) to capture the 
    participant's tendency to repeat the previous first-stage choice regardless of reward.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updates.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi: [-5, 5] - Choice stickiness (perseveration) parameter.
    """
    alpha, beta1, beta2, w, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2

        last_action = action_1[trial]

    eps = 1e-10

    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model-Free and Model-Based model with choice stickiness and eligibility trace.
    
    The full standard model for the two-step task (Daw et al., 2011). It includes 
    an eligibility trace parameter (lam) that allows the second-stage reward prediction error 
    to directly update the first-stage model-free action value.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updates.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi: [-5, 5] - Choice stickiness (perseveration) parameter.
    lam: [0, 1] - Eligibility trace parameter (lambda).
    """
    alpha, beta1, beta2, w, pi, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            q1_vals[last_action] += pi
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
