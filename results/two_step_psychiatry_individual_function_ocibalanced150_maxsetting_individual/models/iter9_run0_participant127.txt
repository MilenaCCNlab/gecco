def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Win/Lose Perseveration with Decay.
    Differentiates stage 1 stickiness based on whether the previous trial was rewarded (p_win) or not (p_lose).
    This allows for capturing asymmetric perseveration depending on the outcome.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p_win: stickiness for stage 1 if previous reward > 0 [0, 10]
    p_lose: stickiness for stage 1 if previous reward == 0 [0, 10]
    decay: forgetting rate for unchosen actions [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, p_win, p_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            if last_reward > 0:
                stickiness[last_action] = p_win
            else:
                stickiness[last_action] = p_lose
            
        q1_vals = beta_1 * q_stage1_net + stickiness
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta_2 * q_stage2_mf[state_idx, :]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen actions
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1.0 - decay)

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1.0 - decay)
                    
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates with Penalty and Decay.
    Uses separate learning rates for positive and negative prediction errors,
    a unified temperature parameter, and a penalty for zero rewards.
    
    Parameters:
    alpha_pos: learning rate for positive PE [0, 1]
    alpha_neg: learning rate for negative PE [0, 1]
    beta: inverse temperature for both stages [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p: stickiness parameter for stage 1 [0, 10]
    decay: forgetting rate for unchosen actions [0, 1]
    penalty: perceived negative value of 0 reward [0, 1]
    """
    alpha_pos, alpha_neg, beta, w, lam, p, decay, penalty = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta * q_stage1_net + p * stickiness
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta * q_stage2_mf[state_idx, :]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1

        # effective reward with penalty
        eff_reward = reward[trial] if reward[trial] > 0 else -penalty

        # action value updating for choice 2
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2

        # decay for unchosen actions
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1.0 - decay)

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1.0 - decay)
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Habit Trace with Decay.
    Incorporates an exponentially updated habit trace for stage 1 actions,
    representing a longer-term perseveration beyond just the last action.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p: stickiness parameter for stage 1 [0, 10]
    decay: forgetting rate for unchosen actions [0, 1]
    w_habit: weight of the habit trace for stage 1 [0, 10]
    """
    alpha, beta_1, beta_2, w, lam, p, decay, w_habit = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    habit = np.zeros(2)
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta_1 * q_stage1_net + p * stickiness + w_habit * habit
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta_2 * q_stage2_mf[state_idx, :]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update habit trace using an exponential moving average
        habit[action_1[trial]] += alpha * (1.0 - habit[action_1[trial]])
        habit[1 - action_1[trial]] += alpha * (0.0 - habit[1 - action_1[trial]])

        # decay for unchosen actions
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1.0 - decay)

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1.0 - decay)
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
