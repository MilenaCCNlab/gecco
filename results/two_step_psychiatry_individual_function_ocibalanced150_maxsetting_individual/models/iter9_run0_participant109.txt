def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates with Stage 1 and Stage 2 Perseveration.
    
    This model splits the learning rate into positive and negative prediction errors,
    allowing participants to learn differently from better-than-expected vs worse-than-expected
    outcomes. It also includes separate perseveration parameters for both choice stages,
    capturing stickiness to both spaceships and aliens.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pers1: [-5, 5] - Perseveration bonus for Stage 1.
    pers2: [-5, 5] - Perseveration bonus for Stage 2.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_a1 = -1
    last_a2 = -1
    
    for trial in range(n_trials):
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_a1 != -1:
            q_stage1_net[last_a1] += pers1
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        # Stage 2
        state_idx = state[trial]
        q_stage2_net = q_stage2_mf[state_idx].copy()
        if last_a2 != -1:
            q_stage2_net[last_a2] += pers2
            
        q2_centered = q_stage2_net - np.max(q_stage2_net)
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage1_mf[a1] += alpha_s2 * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2
        
        last_a1 = a1
        last_a2 = a2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Action Trace Decay Model.
    
    Instead of only perseverating on the immediately preceding trial, this model keeps 
    an exponentially decaying trace of past actions. This allows a build-up of habit 
    for frequently chosen actions (e.g., spaceship 0 for this participant), rather than
    relying strictly on the last choice.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1.
    alpha2: [0, 1] - Learning rate for Stage 2.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pers: [-5, 5] - Weight of the action trace on Stage 1 choices.
    action_trace_decay: [0, 1] - Decay rate of the action trace (0 = only last trial, 1 = no decay).
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, action_trace_decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    action_trace = np.zeros(2)
    
    for trial in range(n_trials):
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q_stage1_net += pers * action_trace
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        # Stage 2
        state_idx = state[trial]
        q_stage2_net = q_stage2_mf[state_idx].copy()
            
        q2_centered = q_stage2_net - np.max(q_stage2_net)
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        # Update action trace
        action_trace = action_trace * action_trace_decay
        action_trace[a1] += 1.0
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Outcome-Dependent Model-Based Weighting.
    
    This model posits that participants might shift their balance between model-based 
    and model-free control depending on whether the previous trial was rewarded or not.
    A win might reinforce a model-free habit, while a loss might trigger model-based planning.
    
    Parameters:
    alpha: [0, 1] - Learning rate for both stages.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w_win: [0, 1] - Model-based weight following a rewarded trial.
    w_lose: [0, 1] - Model-based weight following an unrewarded trial.
    lam: [0, 1] - Eligibility trace.
    pers1: [-5, 5] - Perseveration bonus for Stage 1.
    pers2: [-5, 5] - Perseveration bonus for Stage 2.
    """
    alpha, beta1, beta2, w_win, w_lose, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_a1 = -1
    last_a2 = -1
    last_reward = -1
    
    for trial in range(n_trials):
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine w based on last reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = (w_win + w_lose) / 2.0 # Trial 1 default
            
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_a1 != -1:
            q_stage1_net[last_a1] += pers1
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        # Stage 2
        state_idx = state[trial]
        q_stage2_net = q_stage2_mf[state_idx].copy()
        
        if last_a2 != -1:
            q_stage2_net[last_a2] += pers2
            
        q2_centered = q_stage2_net - np.max(q_stage2_net)
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        last_a1 = a1
        last_a2 = a2
        last_reward = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
