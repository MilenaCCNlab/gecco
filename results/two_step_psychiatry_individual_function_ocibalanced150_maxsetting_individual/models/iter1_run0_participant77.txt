Here are 3 new cognitive models that explore alternative mechanisms for learning and decision-making in the two-step task. 

### Model 1: Stage-Specific Learning Rates
This model tests the hypothesis that participants might update their values at different speeds for the two stages. For example, they might quickly adapt to the changing reward probabilities of the aliens (Stage 2) but more slowly adjust the expected values of the spaceships (Stage 1).
```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. 2011 Two-Step Task Model with Stage-Specific Learning Rates.
    Includes separate learning rates for stage 1 and stage 2, a mixture of model-based 
    and model-free control, an eligibility trace, and choice perseveration.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers: choice perseveration bonus [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + pers * pers_bonus)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # Stage 1 PE uses alpha_1
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            # Stage 2 PE uses alpha_2
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            # Eligibility trace update of stage 1 uses alpha_1
            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates (Positive vs Negative Prediction Errors)
This model posits that the participant updates values differently depending on whether the outcome was better or worse than expected. This is a common phenomenon in reinforcement learning, often linked to differential striatal dopamine functions.
```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. 2011 Two-Step Task Model with Asymmetric Learning Rates.
    Includes separate learning rates for positive and negative prediction errors, 
    a mixture of model-based and model-free control, an eligibility trace, 
    and choice perseveration.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers: choice perseveration bonus [-2,2]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + pers * pers_bonus)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage 1 & Stage 2 Choice Perseveration
Participants often exhibit "stickiness" in their actions not only at the first stage (spaceships) but also at the second stage (aliens). This model includes distinct perseveration terms for both stages, tracking the last choice made at each planet for Stage 2.
```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. 2011 Two-Step Task Model with Stage 1 and Stage 2 Perseveration.
    Includes a mixture of model-based and model-free control, an eligibility trace, 
    and choice perseveration for both stage 1 and stage 2 decisions.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_1: choice perseveration bonus for stage 1 [-2,2]
    pers_2: choice perseveration bonus for stage 2 [-2,2]
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # Last action taken in each state/planet

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + pers_1 * pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = 1.0

            exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + pers_2 * pers_bonus_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
            
            last_action_2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
