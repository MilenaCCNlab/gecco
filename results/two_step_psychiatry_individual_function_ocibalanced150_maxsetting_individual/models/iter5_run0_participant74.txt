Here are 3 new cognitive models that build upon the most successful mechanisms identified so far. The key insight from your best model is that participants treat unrewarded trials not just as a lack of reward, but as an active penalty (`omega`). These 3 models systematically combine this `omega` penalty with other well-established reinforcement learning mechanisms (asymmetric learning rates, outcome-dependent perseveration, and multi-stage perseveration) that have not yet been evaluated in this specific configuration. 

### Model 1: Asymmetric Learning Rates + Penalty
This model tests whether participants learn differently from positive versus negative reward prediction errors (RPEs), while still applying the penalty for unrewarded trials.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with Asymmetric Learning Rates and Penalty for Unrewarded Trials.
    
    Parameters:
    alpha_pos: Learning rate for positive RPE [0, 1]
    alpha_neg: Learning rate for negative RPE [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight for model-based vs model-free influence [0, 1]
    pi: Perseveration bonus for repeating the previous stage 1 choice [-5, 5]
    lam: Eligibility trace parameter [0, 1]
    omega: Penalty for unrewarded trials (0 reward becomes -omega) [0, 5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, pi, lam, omega = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        eff_reward = reward[trial] if reward[trial] > 0 else -omega
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Outcome-Dependent Perseveration + Penalty
This model tests whether the tendency to perseverate (repeat the previous action) depends on whether the previous trial resulted in a reward or a non-reward (Win-Stay / Lose-Shift bias), interacting with the `omega` penalty.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with Penalty for Unrewarded Trials and Outcome-Dependent Perseveration.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight for model-based vs model-free influence [0, 1]
    pi_win: Perseveration bonus after a rewarded trial [-5, 5]
    pi_loss: Perseveration bonus after an unrewarded trial [-5, 5]
    lam: Eligibility trace parameter [0, 1]
    omega: Penalty for unrewarded trials (0 reward becomes -omega) [0, 5]
    """
    alpha, beta1, beta2, w, pi_win, pi_loss, lam, omega = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        current_pi = pi_win if last_reward > 0 else pi_loss
        logits_1 = beta1 * q_stage1_net + current_pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        eff_reward = reward[trial] if reward[trial] > 0 else -omega
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage 1 & Stage 2 Perseveration + Penalty
This model isolates whether the choice inertia (perseveration) occurs solely at the broader goal-oriented stage (Spaceship choice), or if it also extends down to the specific state-dependent stage (Alien choice).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with Penalty for Unrewarded Trials and Stage 1 & 2 Perseveration.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight for model-based vs model-free influence [0, 1]
    pi1: Perseveration bonus for repeating the previous stage 1 choice [-5, 5]
    pi2: Perseveration bonus for repeating the previous stage 2 choice in the same state [-5, 5]
    lam: Eligibility trace parameter [0, 1]
    omega: Penalty for unrewarded trials (0 reward becomes -omega) [0, 5]
    """
    alpha, beta1, beta2, w, pi1, pi2, lam, omega = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        eff_reward = reward[trial] if reward[trial] > 0 else -omega
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
