def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based and Model-Free model with stage 1 and stage 2 choice stickiness.
    This model adds a perseveration parameter for the second-stage choice (alien), 
    allowing us to capture stickiness at both stages of the decision-making process.
    
    Parameters:
    learning_rate: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    rep1: stage 1 choice stickiness [-5,5]
    rep2: stage 2 choice stickiness [-5,5]
    """
    learning_rate, beta, w, lam, rep1, rep2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1:
            last_action1 = -1
            last_action2 = np.array([-1, -1])
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness1 = np.zeros(2)
        if last_action1 != -1:
            stickiness1[last_action1] = 1.0
            
        q1_vals = beta * (q_stage1_net + rep1 * stickiness1)
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            stickiness2[last_action2[state_idx]] = 1.0

        q2_vals = beta * (q_stage2_mf[state_idx] + rep2 * stickiness2)
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model-Based and Model-Free model with Q-value decay and stage 1 & 2 choice stickiness.
    This model combines memory decay for unchosen options with perseveration for both stages, 
    accounting for both forgetting and choice repetition.
    
    Parameters:
    learning_rate: learning rate [0,1]
    decay_rate: decay rate for unchosen options [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    rep1: stage 1 choice stickiness [-5,5]
    rep2: stage 2 choice stickiness [-5,5]
    """
    learning_rate, decay_rate, beta, w, lam, rep1, rep2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1:
            last_action1 = -1
            last_action2 = np.array([-1, -1])
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        q1_vals -= np.max(q1_vals)

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            stickiness2[last_action2[state_idx]] = 1.0

        q2_vals = beta * (q_stage2_mf[state_idx] + rep2 * stickiness2)
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Q-value decay for unchosen options
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - decay_rate)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_rate)

        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model-Based and Model-Free model with Q-value decay and reward-dependent stickiness.
    This model separates the stage 1 choice stickiness into rep_win (after a reward) and rep_loss (after no reward),
    which helps to capture distinct win-stay and lose-shift behavioral patterns.
    
    Parameters:
    learning_rate: learning rate [0,1]
    decay_rate: decay rate for unchosen options [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    rep_win: choice stickiness after a rewarded trial [-5,5]
    rep_loss: choice stickiness after an unrewarded trial [-5,5]
    """
    learning_rate, decay_rate, beta, w, lam, rep_win, rep_loss = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            last_action = -1
            last_reward = 0
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        current_rep = rep_win if last_reward == 1 else rep_loss
            
        q1_vals = beta * (q_stage1_net + current_rep * stickiness)
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Q-value decay for unchosen options
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - decay_rate)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_rate)

        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
