### Model 1: UCB-like Exploration Bonus
This model introduces an exploration mechanism for Stage 1 choices. It tracks how many trials have passed since each Stage 1 action was last chosen and adds a bonus proportional to this count (`c_exp`). This helps capture periods where the participant may switch their spaceship choice simply out of a desire to explore options that have been neglected.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: UCB-like Exploration Bonus
    
    This model includes an exploration bonus based on the number of trials since 
    an action was last chosen at Stage 1. This captures a tendency to explore 
    options that haven't been selected recently.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1.
    alpha2: [0, 1] - Learning rate for Stage 2.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pers: [-5, 5] - Perseveration on the previous Stage 1 choice.
    c_exp: [0, 1] - Exploration bonus multiplier for unchosen Stage 1 actions.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, c_exp = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    unchosen_count = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        # Add exploration bonus based on how many trials it's been since chosen
        q_stage1_net += c_exp * unchosen_count
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2

        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        last_action = a1
        unchosen_count += 1
        unchosen_count[a1] = 0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Stage 2 Learning Rates
This model targets the second stage by allowing separate learning rates for positive and negative reward prediction errors (RPEs). Participants often weigh the receipt of a reward (coin) differently than an omission (0 coins). By isolating this asymmetry strictly to Stage 2, we can see if learning bias happens primarily at the immediate reward level.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Stage 2 Learning Rates
    
    This model allows for different learning rates for positive and negative 
    reward prediction errors at Stage 2, reflecting potential differences in 
    how participants learn from rewards vs. omissions of rewards. Stage 1 
    uses a single learning rate.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1.
    alpha2_pos: [0, 1] - Learning rate for positive RPEs at Stage 2.
    alpha2_neg: [0, 1] - Learning rate for negative RPEs at Stage 2.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pers: [-5, 5] - Perseveration on the previous Stage 1 choice.
    """
    alpha1, alpha2_pos, alpha2_neg, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers

        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2

        # Asymmetric Stage 2 update
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, a2] += alpha2_pos * delta_stage2
        else:
            q_stage2_mf[state_idx, a2] += alpha2_neg * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Perseveration
This model investigates if the participant's tendency to repeat a Stage 1 choice is modulated by the type of transition they experienced. A common transition (e.g., Spaceship 0 -> Planet 0) might reinforce sticking to the same spaceship more (or less) than a rare transition (e.g., Spaceship 0 -> Planet 1).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-Dependent Perseveration
    
    This model implements perseveration that depends on the type of transition 
    experienced in the previous trial. Participants might be more or less likely 
    to repeat a Stage 1 choice if it led to a common vs. a rare transition.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1.
    alpha2: [0, 1] - Learning rate for Stage 2.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pers_common: [-5, 5] - Perseveration bonus if previous transition was common.
    pers_rare: [-5, 5] - Perseveration bonus if previous transition was rare.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers_common, pers_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    last_transition_common = False

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            if last_transition_common:
                q_stage1_net[last_action] += pers_common
            else:
                q_stage1_net[last_action] += pers_rare
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2

        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        last_action = a1
        # Determine if transition was common
        # 0 -> 0 and 1 -> 1 are common transitions (70% probability)
        last_transition_common = (a1 == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
