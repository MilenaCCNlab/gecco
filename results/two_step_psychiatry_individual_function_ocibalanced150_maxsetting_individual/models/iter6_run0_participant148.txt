Here are 3 new cognitive models that explore alternative parameter configurations and mechanisms not yet tested. 

### Model 1: Classic Daw et al. (2011) Model
This model implements the classic 7-parameter configuration from the original Daw et al. (2011) paper. It includes separate learning rates and separate inverse temperatures (betas) for Stage 1 and Stage 2, acknowledging that learning and exploration/exploitation trade-offs may differ systematically across the two stages.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Classic Daw et al. (2011) model with separate learning rates and inverse temperatures for Stage 1 and Stage 2.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for Stage 1.
    alpha2: [0, 1] - Learning rate for Stage 2.
    beta1: [0, 10] - Inverse temperature for Stage 1 softmax.
    beta2: [0, 10] - Inverse temperature for Stage 2 softmax.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for Stage 1 MF updates from Stage 2 rewards.
    rho: [-5, 5] - Choice stickiness for Stage 1.
    """
    alpha1, alpha2, beta1, beta2, w, lam, rho = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if trial > 0:
            stickiness[action_1[trial - 1]] = 1
            
        logits_1 = beta1 * q_net + rho * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates + Stage-Specific Stickiness
This model proposes that participants learn differently from positive vs. negative prediction errors (e.g., they might update values faster when receiving unexpected rewards than when omitting expected rewards) while also maintaining separate tendencies to repeat actions at Stage 1 and Stage 2.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for positive and negative prediction errors, combined with separate choice stickiness for Stage 1 and Stage 2.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rho1: [-5, 5] - Choice stickiness for Stage 1.
    rho2: [-5, 5] - Choice stickiness for Stage 2 (applied if the same state is visited).
    """
    alpha_pos, alpha_neg, beta, w, lam, rho1, rho2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if trial > 0:
            stickiness_1[action_1[trial - 1]] = 1
            
        logits_1 = beta * q_net + rho1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 choice
        stickiness_2 = np.zeros(2)
        if trial > 0 and state[trial - 1] == state_idx:
            stickiness_2[action_2[trial - 1]] = 1
            
        logits_2 = beta * q_stage2_mf[state_idx] + rho2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace uses alpha1 because it's updating the Stage 1 value
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Win-Stay / Lose-Shift Stage 1 Stickiness + Stage 2 Stickiness
This model refines choice perseveration by allowing the participant's Stage 1 stickiness to depend directly on whether the previous trial was rewarded (win-stay) or unrewarded (lose-shift). It also includes a separate Stage 2 stickiness parameter.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separates Stage 1 choice stickiness based on whether the previous trial was rewarded (win-stay) or not (lose-shift), and includes Stage 2 stickiness.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rho_win: [-5, 5] - Choice stickiness for Stage 1 if previous trial was rewarded.
    rho_lose: [-5, 5] - Choice stickiness for Stage 1 if previous trial was unrewarded.
    rho2: [-5, 5] - Choice stickiness for Stage 2.
    """
    alpha, beta, w, lam, rho_win, rho_lose, rho2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        rho1 = 0
        if trial > 0:
            stickiness_1[action_1[trial - 1]] = 1
            if reward[trial - 1] > 0:
                rho1 = rho_win
            else:
                rho1 = rho_lose
            
        logits_1 = beta * q_net + rho1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 choice
        stickiness_2 = np.zeros(2)
        if trial > 0 and state[trial - 1] == state_idx:
            stickiness_2[action_2[trial - 1]] = 1
            
        logits_2 = beta * q_stage2_mf[state_idx] + rho2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
