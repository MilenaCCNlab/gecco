def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Counterfactual Updating in Stage 2.
    
    This model posits that in addition to updating the chosen alien's value,
    participants also update the unchosen alien on the same planet using a 
    counterfactual reward (1 - reward) and a separate counterfactual learning rate.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for Stage 1 updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates (chosen action).
    alpha_cf: [0, 1] - Learning rate for Stage 2 updates (unchosen action).
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    """
    alpha_1, alpha_2, alpha_cf, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Counterfactual update for unchosen action in stage 2
        unchosen_a2 = 1 - action_2[trial]
        delta_cf = (1 - reward[trial]) - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += alpha_cf * delta_cf

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Reward-Dependent Model-Based Weight.
    
    This model assumes that reliance on the model-based system fluctuates 
    depending on whether the previous trial was rewarded or unrewarded.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for Stage 1 updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w_rew: [0, 1] - Model-based weight following a rewarded trial.
    w_unrew: [0, 1] - Model-based weight following an unrewarded trial.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    """
    alpha_1, alpha_2, beta, w_rew, w_unrew, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        if last_reward == 1:
            w = w_rew
        elif last_reward == 0:
            w = w_unrew
        else:
            w = (w_rew + w_unrew) / 2.0
            
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        last_choice = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Transition-Dependent Stage 1 Learning Rates.
    
    This model implements different learning rates for Stage 1 depending on whether 
    the experienced transition was common or rare.
    
    Bounds:
    alpha_1_common: [0, 1] - Learning rate for Stage 1 updates following a common transition.
    alpha_1_rare: [0, 1] - Learning rate for Stage 1 updates following a rare transition.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    """
    alpha_1_common, alpha_1_rare, alpha_2, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        is_common = (action_1[trial] == state_idx)
        alpha_1 = alpha_1_common if is_common else alpha_1_rare

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
