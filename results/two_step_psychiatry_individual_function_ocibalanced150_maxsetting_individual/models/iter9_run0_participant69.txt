def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free reinforcement learning model with eligibility trace,
    stage 1 and stage 2 choice perseveration, a single inverse temperature (beta), 
    dynamic learning of the transition matrix, and a decay parameter for Q-values (forgetting).
    
    Parameters:
    alpha: learning rate for updating Q-values [0, 1]
    beta: softmax inverse temperature for both stage 1 and stage 2 choices [0, 10]
    w: mixing weight for model-based vs model-free control at stage 1 [0, 1]
    lam: eligibility trace parameter [0, 1]
    p_1: choice perseveration parameter for stage 1 [-5, 5]
    p_2: choice perseveration parameter for stage 2 [-5, 5]
    alpha_t: learning rate for updating transition probabilities [0, 1]
    decay: forgetting rate that decays Q-values towards zero at the end of each trial [0, 1]
    """
    alpha, beta, w, lam, p_1, p_2, alpha_t, decay = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p_1 * pers_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        pers_2 = np.zeros(2)
        if last_action_2 != -1:
            pers_2[last_action_2] = 1

        logits_2 = beta * q_stage2_mf[state_idx] + p_2 * pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_t * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_t * (0.0 - transition_matrix[a1, 1 - s])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        
        # Apply decay to Q-values
        q_stage1_mf *= (1.0 - decay)
        q_stage2_mf *= (1.0 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free reinforcement learning model with eligibility trace,
    dynamic transition learning, stage 2 perseveration, and outcome-dependent stage 1 
    perseveration (win-stay/lose-stay).
    
    Parameters:
    alpha: learning rate for updating Q-values [0, 1]
    beta: softmax inverse temperature for both stage 1 and stage 2 choices [0, 10]
    w: mixing weight for model-based vs model-free control at stage 1 [0, 1]
    lam: eligibility trace parameter [0, 1]
    p_win: stage 1 perseveration parameter if previous trial was rewarded [-5, 5]
    p_lose: stage 1 perseveration parameter if previous trial was unrewarded [-5, 5]
    p_2: choice perseveration parameter for stage 2 [-5, 5]
    alpha_t: learning rate for updating transition probabilities [0, 1]
    """
    alpha, beta, w, lam, p_win, p_lose, p_2, alpha_t = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers_1[last_action_1] = p_win
            elif last_reward == 0:
                pers_1[last_action_1] = p_lose
            
        logits_1 = beta * q_stage1_net + pers_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        pers_2 = np.zeros(2)
        if last_action_2 != -1:
            pers_2[last_action_2] = p_2

        logits_2 = beta * q_stage2_mf[state_idx] + pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_t * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_t * (0.0 - transition_matrix[a1, 1 - s])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free reinforcement learning model with eligibility trace,
    dynamic transition learning, stage 1 and stage 2 choice perseveration, and an 
    intrinsic bias parameter for selecting action 0 at stage 1.
    
    Parameters:
    alpha: learning rate for updating Q-values [0, 1]
    beta: softmax inverse temperature for both stage 1 and stage 2 choices [0, 10]
    w: mixing weight for model-based vs model-free control at stage 1 [0, 1]
    lam: eligibility trace parameter [0, 1]
    p_1: choice perseveration parameter for stage 1 [-5, 5]
    p_2: choice perseveration parameter for stage 2 [-5, 5]
    alpha_t: learning rate for updating transition probabilities [0, 1]
    bias: intrinsic preference (added to logit) for action 0 at stage 1 [-5, 5]
    """
    alpha, beta, w, lam, p_1, p_2, alpha_t, bias = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p_1 * pers_1
        logits_1[0] += bias
        
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        pers_2 = np.zeros(2)
        if last_action_2 != -1:
            pers_2[last_action_2] = 1

        logits_2 = beta * q_stage2_mf[state_idx] + p_2 * pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_t * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_t * (0.0 - transition_matrix[a1, 1 - s])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
