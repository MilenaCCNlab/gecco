import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Learned Transition Matrix
    Instead of assuming a fixed transition matrix, the participant updates their 
    estimate of the transition probabilities based on experience.
    
    Bounds:
    alpha: [0, 1] - Learning rate for action values.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseveration parameter for stage 1.
    alpha_trans: [0, 1] - Learning rate for transition matrix.
    """
    alpha, beta_1, beta_2, w, lam, p, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize learned transition matrix uniformly
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        # Handle anomaly in data where action_2 might be -1
        a2_idx = 1 if a2 == -1 else a2
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            rep = np.zeros(2)
            rep[last_action_1] = 1
            q_stage1_net += p * rep
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[s]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2_idx]

        # Update transition matrix probabilities based on observed state
        transition_matrix[a1, s] += alpha_trans * (1 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_trans * (0 - transition_matrix[a1, 1 - s])

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2_idx] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2_idx]
        q_stage2_mf[s, a2_idx] += alpha * delta_stage2

        # Eligibility trace for stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Success-driven Model-Based Control
    The weight placed on the model-based system (w) dynamically depends on whether 
    the previous trial was rewarded (w_win) or unrewarded (w_lose).
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w_win: [0, 1] - Model-based weight after a win.
    w_lose: [0, 1] - Model-based weight after a loss.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseveration parameter for stage 1.
    """
    alpha, beta_1, beta_2, w_win, w_lose, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        a2_idx = 1 if a2 == -1 else a2
        r = reward[trial]

        # Determine w based on last reward
        if trial == 0:
            w = 0.5
        else:
            w = w_win if last_reward > 0 else w_lose

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            rep = np.zeros(2)
            rep[last_action_1] = 1
            q_stage1_net += p * rep
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[s]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2_idx]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2_idx] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2_idx]
        q_stage2_mf[s, a2_idx] += alpha * delta_stage2

        # Eligibility trace for stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate Learning Rates + Decay
    Features independent learning rates for Stage 1 and Stage 2, along with 
    value decay (forgetting) for unchosen options.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for Stage 1.
    alpha_2: [0, 1] - Learning rate for Stage 2.
    decay: [0, 1] - Decay rate for unchosen action values.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseveration parameter for stage 1.
    """
    alpha_1, alpha_2, decay, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        a2_idx = 1 if a2 == -1 else a2
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            rep = np.zeros(2)
            rep[last_action_1] = 1
            q_stage1_net += p * rep
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[s]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2_idx]

        # Decay unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        for s_idx in range(2):
            for act_idx in range(2):
                if not (s_idx == s and act_idx == a2_idx):
                    q_stage2_mf[s_idx, act_idx] *= (1 - decay)

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2_idx] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha_1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2_idx]
        q_stage2_mf[s, a2_idx] += alpha_2 * delta_stage2

        # Eligibility trace for stage 1 uses stage 1 learning rate
        q_stage1_mf[a1] += alpha_1 * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
