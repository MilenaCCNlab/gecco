def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Separate inverse temperatures for model-based, model-free, and stage 2.
    Instead of a single beta and a weighting parameter w, this model estimates 
    separate inverse temperatures for the model-based and model-free systems at Stage 1.
    
    Bounds:
    learning_rate: [0, 1]
    beta_mb: [0, 10]
    beta_mf: [0, 10]
    beta2: [0, 10]
    """
    learning_rate, beta_mb, beta_mf, beta2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Logits combine MB and MF values with independent temperatures
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        chosen_1 = action_1[trial]
        chosen_2 = action_2[trial]

        # ACTION VALUE UPDATING
        delta_stage1 = q_stage2_mf[state_idx, chosen_2] - q_stage1_mf[chosen_1]
        q_stage1_mf[chosen_1] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, chosen_2]
        q_stage2_mf[state_idx, chosen_2] += learning_rate * delta_stage2
        
        # Stage 1 eligibility trace update
        q_stage1_mf[chosen_1] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Separate learning rates for Stage 1 and Stage 2.
    This model allows for different speeds of learning at the two stages, 
    reflecting potential differences in credit assignment over temporal delays.
    
    Bounds:
    alpha1: [0, 1]
    alpha2: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    """
    alpha1, alpha2, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        chosen_1 = action_1[trial]
        chosen_2 = action_2[trial]

        # ACTION VALUE UPDATING with separate learning rates
        delta_stage1 = q_stage2_mf[state_idx, chosen_2] - q_stage1_mf[chosen_1]
        q_stage1_mf[chosen_1] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, chosen_2]
        q_stage2_mf[state_idx, chosen_2] += alpha2 * delta_stage2
        
        # Stage 1 eligibility trace update uses alpha1
        q_stage1_mf[chosen_1] += alpha1 * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Choice perseveration (stickiness) for both Stage 1 and Stage 2.
    Includes bias parameters that increase the probability of repeating the 
    previous choice at Stage 1, and repeating the previous choice in a specific 
    state at Stage 2.
    
    Bounds:
    learning_rate: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    pi1: [-5, 5]
    pi2: [-5, 5]
    """
    learning_rate, beta, w, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # Last action taken in each state

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net
        if last_action_1 != -1:
            logits_1[last_action_1] += pi1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += pi2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        chosen_1 = action_1[trial]
        chosen_2 = action_2[trial]

        # ACTION VALUE UPDATING
        delta_stage1 = q_stage2_mf[state_idx, chosen_2] - q_stage1_mf[chosen_1]
        q_stage1_mf[chosen_1] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, chosen_2]
        q_stage2_mf[state_idx, chosen_2] += learning_rate * delta_stage2
        
        # Stage 1 eligibility trace update
        q_stage1_mf[chosen_1] += learning_rate * delta_stage2
        
        # Update last actions for perseveration
        last_action_1 = chosen_1
        last_action_2[state_idx] = chosen_2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
