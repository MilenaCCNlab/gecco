def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw Model with Transition Learning and Separate Learning Rates
    This model extends the best model by separating the learning rates for Stage 1 
    and Stage 2 Q-value updates, while retaining dynamic transition probability learning.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-values.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-values.
    beta1: [0, 10] - Inverse temperature for Stage 1 choice.
    beta2: [0, 10] - Inverse temperature for Stage 2 choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration parameter for Stage 1 choice.
    alpha_trans: [0, 1] - Learning rate for updating transition probabilities.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, alpha_trans = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            perseveration_1[prev_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pers * perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Transition matrix updating
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2

        # Eligibility trace for Stage 1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2

        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw Model with Transition Learning and Stage 2 Perseveration
    This model builds on the best model by adding a perseveration parameter for Stage 2,
    capturing the tendency to repeat the same second-stage action in a given state.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for Stage 1 choice.
    beta2: [0, 10] - Inverse temperature for Stage 2 choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers1: [-5, 5] - Perseveration parameter for Stage 1 choice.
    pers2: [-5, 5] - Perseveration parameter for Stage 2 choice.
    alpha_trans: [0, 1] - Learning rate for updating transition probabilities.
    """
    alpha, beta1, beta2, w, lam, pers1, pers2, alpha_trans = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1
    prev_a2 = np.full(2, -1) # Store previous action for each state

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if prev_a1 != -1:

        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        perseveration_2 = np.zeros(2)
        if prev_a2[state_idx] != -1:
            perseveration_2[prev_a2[state_idx]] = 1.0
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + pers2 * perseveration_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility trace for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2

        prev_a1 = a1
        prev_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw Model with Transition Learning and Reward-Dependent Perseveration
    This model refines the best model by splitting the perseveration parameter into two:
    one for when the previous trial was rewarded (pers_win) and one for when it was not (pers_lose).
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for Stage 1 choice.
    beta2: [0, 10] - Inverse temperature for Stage 2 choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Perseveration parameter for Stage 1 choice after a win (reward > 0).
    pers_lose: [-5, 5] - Perseveration parameter for Stage 1 choice after a loss (reward == 0).
    alpha_trans: [0, 1] - Learning rate for updating transition probabilities.
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, alpha_trans = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            continue

        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            perseveration_1[prev_a1] = 1.0
            
        if prev_r > 0:
            pers = pers_win
        elif prev_r == 0:
            pers = pers_lose
        else:
            pers = 0.0
            
        logits_1 = beta1 * q_stage1_net + pers * perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Transition matrix updating
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility trace for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2

        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
