def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates with win-perseveration and decay.
    Positive and negative prediction errors are updated with different learning rates.
    Unchosen actions decay towards zero.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-2,2]
    decay: decay rate for unchosen actions [0,1]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers_win, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1 and last_reward > 0:
            pers_bonus[last_action] = 1.0
                
        q_net_total = beta_1 * q_net + pers_win * pers_bonus
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            alpha_1_eff = alpha_pos if delta_stage1 > 0 else alpha_neg
            q_stage1_mf[action_1[trial]] += alpha_1_eff * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            alpha_2_eff = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2_eff * delta_stage2

            alpha_trace_eff = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage1_mf[action_1[trial]] += alpha_trace_eff * lam * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] *= (1 - decay)
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
            
            other_state = 1 - state_idx
            q_stage2_mf[other_state, 0] *= (1 - decay)
            q_stage2_mf[other_state, 1] *= (1 - decay)
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Single learning rate with outcome-dependent perseveration and decay.
    Uses a single learning rate for both stages, but separates perseveration 
    into win-stay (pers_win) and lose-shift/stay (pers_lose). Unchosen actions decay.
    
    Parameters:
    alpha: learning rate for both stages [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-2,2]
    pers_lose: perseveration bonus if previous trial was unrewarded [-2,2]
    decay: decay rate for unchosen actions [0,1]
    """
    alpha, beta_1, beta_2, w, lam, pers_win, pers_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1:
            if last_reward > 0:
                pers_bonus[last_action] = pers_win
            else:
                pers_bonus[last_action] = pers_lose
                
        q_net_total = beta_1 * q_net + pers_bonus
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] *= (1 - decay)
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
            
            other_state = 1 - state_idx
            q_stage2_mf[other_state, 0] *= (1 - decay)
            q_stage2_mf[other_state, 1] *= (1 - decay)
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Win-perseveration with global forgetting.
    Instead of only decaying unchosen options, all Q-values passively decay 
    (forgetting) towards zero at each trial, applied after the learning update.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-2,2]
    forget_rate: global decay rate for all Q-values [0,1]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win, forget_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1 and last_reward > 0:
            pers_bonus[last_action] = 1.0
                
        q_net_total = beta_1 * q_net + pers_win * pers_bonus
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

            # Global forgetting applied to all Q-values
            q_stage1_mf *= (1 - forget_rate)
            q_stage2_mf *= (1 - forget_rate)
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
