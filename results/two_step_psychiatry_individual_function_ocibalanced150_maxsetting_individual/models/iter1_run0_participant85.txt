def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative prediction errors.
    
    This model separates the learning rate into two components: one for positive 
    prediction errors (when the outcome is better than expected) and one for negative 
    prediction errors (when the outcome is worse than expected). It also includes
    a perseveration parameter for the first stage.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi: [-5, 5] - Choice stickiness (perseveration) parameter.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            q1_vals[last_action] += pi
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Decay for unchosen options.
    
    This model assumes that the values of unchosen options are not static but decay 
    towards a neutral value (0.5) over time. This captures a form of forgetting where 
    participants become less certain of the value of options they haven't selected recently.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi: [-5, 5] - Choice stickiness (perseveration) parameter.
    decay: [0, 1] - Decay rate for unchosen options towards 0.5.
    """
    alpha, beta1, beta2, w, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize at 0.5 for decay mechanism to be symmetric
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            q1_vals[last_action] += pi
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] += decay * (0.5 - q_stage2_mf[other_state, 0])
        q_stage2_mf[other_state, 1] += decay * (0.5 - q_stage2_mf[other_state, 1])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: First and second stage choice stickiness.
    
    This model adds a stickiness parameter for the second stage to capture state-specific 
    perseveration on aliens. It accounts for the tendency to repeat the same second-stage 
    action given the same second-stage state, independent of reward history.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi1: [-5, 5] - First-stage choice stickiness.
    pi2: [-5, 5] - Second-stage choice stickiness.
    """
    alpha, beta1, beta2, w, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1]) # Track last action for each state

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action1 != -1:
            q1_vals[last_action1] += pi1
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        if last_action2[state_idx] != -1:
            q2_vals[last_action2[state_idx]] += pi2
            
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
