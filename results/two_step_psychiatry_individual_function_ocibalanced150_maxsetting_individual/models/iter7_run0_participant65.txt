def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF mixture with separate betas for each stage, choice perseveration (pi), 
    and transition learning (lr_trans). Instead of assuming fixed transition probabilities, 
    the model learns them dynamically from observations.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for model-free Q-values.
    beta1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting parameter for model-based vs model-free influence.
    pi: [-5,5] - Perseveration parameter for stage 1 choice.
    lr_trans: [0,1] - Learning rate for updating the transition matrix.
    """
    learning_rate, beta1, beta2, w, pi, lr_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix uniformly, to be learned
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        a1 = action_1[trial]
        a2 = action_2[trial]

        # Update transition matrix based on observed state
        transition_matrix[a1, state_idx] += lr_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += lr_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF mixture with eligibility trace (lam) and outcome-dependent 
    perseveration (pi_win, pi_lose). This allows the tendency to repeat a choice to 
    vary depending on whether the previous trial was rewarded or not.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for model-based vs model-free influence.
    pi_win: [-5,5] - Perseveration parameter after a rewarded trial.
    pi_lose: [-5,5] - Perseveration parameter after an unrewarded trial.
    lam: [0,1] - Eligibility trace parameter connecting stage 2 reward to stage 1.
    """
    learning_rate, beta, w, pi_win, pi_lose, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        pi = pi_win if last_reward > 0 else pi_lose
        
        logits_1 = beta * q_stage1_net + pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        a1 = action_1[trial]
        a2 = action_2[trial]
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Eligibility trace for stage 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF mixture with fully independent learning rates and betas for each stage,
    along with eligibility trace (lam) and choice perseveration (pi). This allows for 
    different learning and exploration dynamics at the transition vs. reward stages.
    
    Bounds:
    lr1: [0,1] - Learning rate for stage 1 Q-values.
    lr2: [0,1] - Learning rate for stage 2 Q-values.
    beta1: [0,10] - Inverse temperature for stage 1 softmax.
    beta2: [0,10] - Inverse temperature for stage 2 softmax.
    w: [0,1] - Weighting parameter for model-based vs model-free influence.
    pi: [-5,5] - Perseveration parameter for stage 1 choice.
    lam: [0,1] - Eligibility trace parameter.
    """
    lr1, lr2, beta1, beta2, w, pi, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        probs_1 = exp_q1 / np.sum(exp_q1)

        # policy for the second choice

    return log_loss
