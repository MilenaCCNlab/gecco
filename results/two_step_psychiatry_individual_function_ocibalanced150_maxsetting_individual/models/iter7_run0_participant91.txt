Here are three new cognitive models, each building upon the best-performing mechanisms so far (Hybrid MB/MF, eligibility traces, and choice stickiness) while exploring novel dimensions of decision-making:

### Model 1: Distinct Stage Temperatures
This model separates the decision noise (inverse temperature) into two parameters for Stage 1 and Stage 2. This accounts for the possibility that participants exhibit different levels of exploration or stochasticity when choosing spaceships versus when choosing aliens.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF with eligibility traces and distinct stage 1 and 2 temperatures.
    Separate beta parameters for stage 1 and stage 2 allow for different levels of exploration/noise.
    
    Parameters:
    learning_rate: [0,1] - learning rate for Q-values
    beta_1: [0,10] - inverse temperature for stage 1 softmax
    beta_2: [0,10] - inverse temperature for stage 2 softmax
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    """
    learning_rate, beta_1, beta_2, w, stickiness_1, stickiness_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Continuous Choice Traces
Instead of only boosting the probability of the immediate last action, this model maintains a decaying "choice trace" for both stages. This captures longer-term perseveration (autocorrelation in choices) where participants might favor actions they have chosen frequently in the recent past.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF with eligibility traces and continuous choice traces.
    Instead of just repeating the immediate last action, this model maintains an exponentially 
    decaying trace of past choices to model longer-term perseveration (autocorrelation).
    
    Parameters:
    learning_rate: [0,1] - learning rate for Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice trace weight for stage 1
    stickiness_2: [-5,5] - choice trace weight for stage 2
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    trace_decay: [0,1] - decay rate for the choice traces (1 = immediate decay, 0 = no decay)
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, lam, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net + stickiness_1 * choice_trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_2 * choice_trace_2[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Update choice traces
        choice_trace_1 = (1 - trace_decay) * choice_trace_1
        choice_trace_1[action_1[trial]] += 1
        
        choice_trace_2[state_idx] = (1 - trace_decay) * choice_trace_2[state_idx]
        choice_trace_2[state_idx, action_2[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates
This model introduces valence-dependent learning rates (one for positive prediction errors, one for negative). This tests the hypothesis that the participant might update their internal representations more aggressively when they receive an unexpected reward (positive PE) compared to an unexpected omission (negative PE), or vice versa.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF with eligibility traces and asymmetric learning rates.
    Uses separate learning rates for positive vs negative prediction errors to capture 
    differential sensitivity to better-than-expected vs worse-than-expected outcomes.
    
    Parameters:
    learning_rate_pos: [0,1] - learning rate for positive prediction errors (RPE > 0)
    learning_rate_neg: [0,1] - learning rate for negative prediction errors (RPE <= 0)
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    """
    learning_rate_pos, learning_rate_neg, beta, w, stickiness_1, stickiness_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = learning_rate_pos if delta_stage1 > 0 else learning_rate_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = learning_rate_pos if delta_stage2 > 0 else learning_rate_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += lr_stage2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
