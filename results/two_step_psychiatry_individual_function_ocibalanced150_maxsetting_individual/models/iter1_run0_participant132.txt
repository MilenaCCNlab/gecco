Here are 3 new cognitive models that explore alternative mechanisms and parameter combinations to explain the participant's behavior in the two-step decision-making task.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model tests the hypothesis that participants might update their values at different speeds for the two stages. They might quickly adapt to the changing probabilities of aliens giving gold (Stage 2) but slowly update their preferences for the spaceships (Stage 1).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate learning rates for stage 1 and stage 2.
    
    Parameters:
    learning_rate1: [0,1] - learning rate for stage 1 value updates
    learning_rate2: [0,1] - learning rate for stage 2 value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi: [-5,5] - choice perseveration (stickiness) for stage 1
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    """
    learning_rate1, learning_rate2, beta1, beta2, w, pi, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
        logits_1 = beta1 * q_net + pi * pers
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            logits_2 = beta2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate2 * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate1 * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Dual-Stage Choice Perseveration
This model evaluates if participants exhibit stickiness not just in their initial spaceship choice (Stage 1), but also in their alien choice (Stage 2). 

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with choice perseveration at both stages.
    
    Parameters:
    learning_rate: [0,1] - learning rate for value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi_1: [-5,5] - choice perseveration (stickiness) for stage 1
    pi_2: [-5,5] - choice perseveration (stickiness) for stage 2
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    """
    learning_rate, beta1, beta2, w, pi_1, pi_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        logits_1 = beta1 * q_net + pi_1 * pers_1
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            pers_2 = np.zeros(2)
            if last_action_2 != -1:
                pers_2[last_action_2] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi_2 * pers_2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
            last_action_2 = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Value Decay for Unchosen Options
This model assumes that participants' estimates for the options they don't choose gradually decay towards a neutral expectation (0.5). This introduces a form of forgetting that could accurately capture how participants behave when they haven't sampled a particular state-action pair in a long time.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with value decay for unchosen options.
    
    Parameters:
    learning_rate: [0,1] - learning rate for value updates
    decay: [0,1] - decay rate for unchosen Q-values towards 0.5 (neutral expectation)
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi: [-5,5] - choice perseveration (stickiness) for stage 1
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    """
    learning_rate, decay, beta1, beta2, w, pi, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 (neutral) due to decay mechanics
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
        logits_1 = beta1 * q_net + pi * pers
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            logits_2 = beta2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # action value updating for chosen options
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
            # value decay for unchosen options
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == action_2[trial]):
                        q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        else:
            p_choice_2[trial] = 1.0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
