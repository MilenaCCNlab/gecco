def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Adds stage 1 and stage 2 choice perseveration (stickiness_1 and stickiness_2) 
    along with eligibility trace (lambda_) to the standard MB/MF model.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature for both stages.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness_1: [-5, 5] - Tendency to repeat the previous stage-1 choice.
    stickiness_2: [-5, 5] - Tendency to repeat the previous stage-2 choice.
    """
    alpha, beta, w, lambda_, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_1 = np.zeros(2)
        if trial > 0:
            stick_1[action_1[trial-1]] = 1.0
            
        exp_q1 = np.exp(beta * q_net_1 + stickiness_1 * stick_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick_2 = np.zeros(2)
        if trial > 0:
            stick_2[action_2[trial-1]] = 1.0
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + stickiness_2 * stick_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates for positive and negative prediction errors,
    separate betas for each stage, eligibility trace, and stage-specific stickiness.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness_1: [-5, 5] - Tendency to repeat the previous stage-1 choice.
    stickiness_2: [-5, 5] - Tendency to repeat the previous stage-2 choice.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])

    p_choice_2 = np.zeros(n_trials)

    q_stage2_mf = np.zeros((2, 2))
