import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with decay and outcome-dependent model-based weighting.
    The weight placed on the model-based system depends on whether the previous trial was rewarded (w_win) or not (w_lose).
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Forgetting rate for all actions (values revert toward 0.5)
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w_win: [0, 1] - Model-based weight if previous trial was rewarded
    w_lose: [0, 1] - Model-based weight if previous trial was unrewarded
    lam: [0, 1] - Eligibility trace parameter
    pers: [-5, 5] - Perseveration parameter
    """
    alpha, decay, beta1, beta2, w_win, w_lose, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = 0.5 

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits1 = beta1 * q_net + pers * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with decay and asymmetric eligibility traces.
    The eligibility trace lam depends on whether the second-stage prediction error is positive (lam_pos) or negative (lam_neg).
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Forgetting rate for all actions (values revert toward 0.5)
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam_pos: [0, 1] - Eligibility trace for positive prediction errors
    lam_neg: [0, 1] - Eligibility trace for negative prediction errors
    pers: [-5, 5] - Perseveration parameter
    """
    alpha, decay, beta1, beta2, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits1 = beta1 * q_net + pers * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        lam = lam_pos if delta_stage2 > 0 else lam_neg

        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with decay and transition-dependent perseveration.
    Perseveration on the first-stage action depends on whether the previous trial was a common transition (pers_common) or rare transition (pers_rare).
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Forgetting rate for all actions (values revert toward 0.5)
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers_common: [-5, 5] - Perseveration after a common transition
    pers_rare: [-5, 5] - Perseveration after a rare transition
    """
    alpha, decay, beta1, beta2, w, lam, pers_common, pers_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_transition_common = -1 

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        if last_transition_common == 1:
            pers = pers_common
        elif last_transition_common == 0:
            pers = pers_rare
        else:
            pers = 0.0

        logits1 = beta1 * q_net + pers * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        if action_1[trial] == state_idx:
            last_transition_common = 1
        else:
            last_transition_common = 0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
