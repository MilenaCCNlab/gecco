Here are 3 new cognitive models that build upon the best-performing mechanisms (stickiness, decay) while exploring alternative parameter configurations and new mechanisms like habit traces and stage-specific parameters.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and temperatures for each stage, combined with w, lam, stickiness, and decay.
    This model investigates if the participant's learning and choice stochasticity differ between the first and second stage.
    
    Bounds:
    alpha_1: [0, 1] - learning rate for stage 1
    alpha_2: [0, 1] - learning rate for stage 2
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - weight for model-based vs model-free
    lam: [0, 1] - eligibility trace
    stickiness: [0, 10] - choice stickiness parameter for stage 1
    decay: [0, 1] - decay rate for unchosen options
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec = np.zeros(2)
        if last_action != -1:
            stick_vec[last_action] = 1.0
            
        logits_1 = beta_1 * q_net + stickiness * stick_vec
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF Q-learning with choice stickiness, value decay, and a slow-updating habit trace.
    This model adds a long-term "habit" trace that learns to prefer frequently chosen actions, capturing the participant's persistent bias toward action 0.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - weight for model-based vs model-free
    stickiness: [0, 10] - 1-trial choice stickiness parameter
    decay: [0, 1] - decay rate for unchosen options
    habit_lr: [0, 1] - learning rate for the long-term habit trace
    habit_weight: [0, 10] - weight for the habit trace influence on choice
    """
    alpha, beta, w, stickiness, decay, habit_lr, habit_weight = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    habit = np.ones(2) * 0.5  # Initialize uniform habit trace
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec = np.zeros(2)
        if last_action != -1:
            stick_vec[last_action] = 1.0
            
        logits_1 = beta * q_net + stickiness * stick_vec + habit_weight * habit
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * delta_stage2 # Implicit lam=1
        
        # Update long-term habit trace
        habit[action_1[trial]] += habit_lr * (1.0 - habit[action_1[trial]])
        habit[unchosen_1] += habit_lr * (0.0 - habit[unchosen_1])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF Q-learning with separate stickiness for stage 1 and stage 2, plus decay and eligibility trace.
    This model tests if the participant is exhibiting generalized motor perseveration (stickiness) at both stages independently.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - weight for model-based vs model-free
    stickiness_1: [0, 10] - choice stickiness parameter for stage 1
    stickiness_2: [0, 10] - choice stickiness parameter for stage 2 (motor perseveration)
    decay: [0, 1] - decay rate for unchosen options
    lam: [0, 1] - eligibility trace
    """
    alpha, beta, w, stickiness_1, stickiness_2, decay, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vec_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + stickiness_1 * stick_vec_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stick_vec_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vec_2[last_action_2] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_2 * stick_vec_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
