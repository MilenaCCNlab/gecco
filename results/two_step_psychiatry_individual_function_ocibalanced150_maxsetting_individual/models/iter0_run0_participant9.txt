Here are 3 new cognitive models as Python functions, designed to capture different psychological mechanisms that might explain the participant's choices in the two-step task. 

### Model 1: Full Mixed Model-Free and Model-Based Learning (Daw et al., 2011)
This model assumes the participant uses a mixture of model-free (habitual) and model-based (goal-directed) learning. It also includes an eligibility trace (to update stage 1 values based on stage 2 rewards) and a choice stickiness parameter (to capture the tendency to repeat the previous first-stage choice regardless of reward).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Mixed Model-Free and Model-Based learning (Daw et al. 2011).
    Includes model-free and model-based updates, eligibility trace, and stickiness.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices (exploration vs exploitation).
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0, 1] - Eligibility trace (how much stage 2 reward updates stage 1 MF value).
    rep: [-5, 5] - Choice stickiness/perseveration for stage 1 (positive = tendency to repeat).
    """
    alpha, beta_1, beta_2, w, lam, rep = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        q1_logits = beta_1 * q_stage1_net + rep * stickiness
        q1_logits -= np.max(q1_logits) # for numerical stability
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates (Positive vs. Negative Prediction Errors)
People often update their beliefs asymmetrically based on whether an outcome was better than expected (positive RPE) or worse than expected (negative RPE). This model allows the learning rate to vary based on the sign of the prediction error.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive and negative RPEs.
    Extends the mixed MF/MB model by allowing different learning rates
    depending on whether the prediction error is positive or negative.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (reward > expected).
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (reward < expected).
    beta: [0, 10] - Inverse temperature for both stages.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    """
    alpha_pos, alpha_neg, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_logits = beta * q_stage1_net
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_logits = beta * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Forgetting / Value Decay Q-Learning
Because the reward probabilities of the aliens drift slowly over time, a participant may naturally "forget" the value of unchosen options, causing those values to decay towards a neutral baseline (0.5).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Forgetting Q-learning Model.
    Values of unchosen actions decay toward a neutral starting value (0.5), 
    capturing the tendency to forget or become uncertain about unvisited states.
    
    Parameters:
    alpha: [0, 1] - Learning rate for chosen actions.
    decay: [0, 1] - Decay rate for unchosen actions (0 = no decay, 1 = instant reset to 0.5).
    beta: [0, 10] - Inverse temperature for choices.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    """
    alpha, decay, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values to 0.5 prior
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_logits = beta * q_stage1_net
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_logits = beta * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Decay all unchosen options towards 0.5
        for i in range(2):
            if i != action_1[trial]:
                q_stage1_mf[i] += decay * (0.5 - q_stage1_mf[i])
            for j in range(2):
                if not (i == state_idx and j == action_2[trial]):
                    q_stage2_mf[i, j] += decay * (0.5 - q_stage2_mf[i, j])
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
