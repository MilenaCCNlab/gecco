def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Eligibility trace combined with decay of unchosen options.
    Bounds:
    learning_rate: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    decay: [0, 1]
    """
    learning_rate, beta, w, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta * q_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        chosen_1 = action_1[trial]
        unchosen_1 = 1 - chosen_1
        chosen_2 = action_2[trial]
        unchosen_2 = 1 - chosen_2

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, chosen_2] - q_stage1_mf[chosen_1]
        q_stage1_mf[chosen_1] += learning_rate * delta_stage1
        q_stage1_mf[unchosen_1] *= (1 - decay)

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, chosen_2]
        q_stage2_mf[state_idx, chosen_2] += learning_rate * delta_stage2
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)

        # decay unvisited state's Q-values
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1 - decay)
        q_stage2_mf[other_state, 1] *= (1 - decay)

        # eligibility trace update for stage 1
        q_stage1_mf[chosen_1] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Separate learning rates and inverse temperatures for stage 1 and stage 2.
    Bounds:
    alpha1: [0, 1]
    alpha2: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    """
    alpha1, alpha2, beta1, beta2, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        chosen_1 = action_1[trial]
        chosen_2 = action_2[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, chosen_2] - q_stage1_mf[chosen_1]
        q_stage1_mf[chosen_1] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, chosen_2]
        q_stage2_mf[state_idx, chosen_2] += alpha2 * delta_stage2
        
        # eligibility trace update for stage 1 (lambda = 1)
        q_stage1_mf[chosen_1] += alpha1 * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Separate choice perseveration (stickiness) for stage 1 and stage 2.
    Bounds:
    learning_rate: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    stick1: [-5, 5]
    stick2: [-5, 5]
    """
    learning_rate, beta, w, stick1, stick2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # Tracks the last action taken in each stage 2 state

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_bonus_1[last_action_1] = stick1
            
        logits_1 = beta * q_net + stick_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stick_bonus_2[last_action_2[state_idx]] = stick2

        logits_2 = beta * q_stage2_mf[state_idx] + stick_bonus_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        chosen_1 = action_1[trial]
        chosen_2 = action_2[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, chosen_2] - q_stage1_mf[chosen_1]
        q_stage1_mf[chosen_1] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, chosen_2]
        q_stage2_mf[state_idx, chosen_2] += learning_rate * delta_stage2
        
        # eligibility trace update for stage 1 (lambda = 1)
        q_stage1_mf[chosen_1] += learning_rate * delta_stage2
        
        # Update last actions for stickiness calculation
        last_action_1 = chosen_1
        last_action_2[state_idx] = chosen_2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
