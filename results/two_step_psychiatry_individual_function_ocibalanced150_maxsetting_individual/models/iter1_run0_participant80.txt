Here are 3 new cognitive models that build upon the hybrid Model-Based (MB) / Model-Free (MF) architecture, each exploring a different psychological mechanism that could explain the participant's choice sequences.

### Model 1: Stage-Specific Inverse Temperatures
Participants often exhibit different levels of exploration or decision noise at different stages of a task. For instance, they might be more exploratory when choosing a spaceship (Stage 1) but more exploitative when choosing an alien (Stage 2). This model introduces separate `beta` parameters for each stage.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with separate inverse temperatures for Stage 1 and Stage 2, and choice perseveration.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values.
    beta_1: [0,10] - Inverse temperature for Stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for Stage 2 softmax action selection.
    w: [0,1] - Weighting of model-based vs model-free values.
    lambda_: [0,1] - Eligibility trace parameter.
    stickiness: [0,10] - Bonus added to the logit of the previously chosen stage 1 action.
    """
    alpha, beta_1, beta_2, w, lambda_, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action != -1:
            q_stage1_net[last_action] += stickiness / (beta_1 + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta_1 * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        max_q2 = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta_2 * (q_stage2_mf[state_idx] - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Dual-Stage Choice Perseveration
The baseline model only considers perseveration ("stickiness") at Stage 1. However, participants often exhibit state-dependent motor chunking or habituation at Stage 2 as well (e.g., repeating the same alien choice if they land on the same planet). This model adds a `stickiness_2` parameter.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with choice perseveration (stickiness) at both Stage 1 and Stage 2.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting of model-based vs model-free values.
    lambda_: [0,1] - Eligibility trace parameter.
    stickiness_1: [0,10] - Bonus added to the previously chosen stage 1 action.
    stickiness_2: [0,10] - Bonus added to the previously chosen stage 2 action in the current state.
    """
    alpha, beta, w, lambda_, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness_1 / (beta + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q_stage2_net[last_action_2[state_idx]] += stickiness_2 / (beta + 1e-10)

        max_q2 = np.max(q_stage2_net)
        exp_q2 = np.exp(beta * (q_stage2_net - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Unchosen Action Decay (Forgetting)
Human memory is imperfect. As participants learn the value of chosen actions, the values of unchosen alternatives often regress toward a neutral baseline (0.5 for binary probabilities) due to forgetting. This model implements a `decay` parameter to account for unchosen Q-value drift.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Q-value decay (forgetting) for unchosen actions.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting of model-based vs model-free values.
    lambda_: [0,1] - Eligibility trace parameter.
    stickiness: [0,10] - Bonus added to the logit of the previously chosen stage 1 action.
    decay: [0,1] - Decay rate of unchosen Q-values towards 0.5.
    """
    alpha, beta, w, lambda_, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize at 0.5 to naturally pair with the decay
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action != -1:
            q_stage1_net[last_action] += stickiness / (beta + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        max_q2 = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Decay unchosen actions towards baseline (0.5)
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
