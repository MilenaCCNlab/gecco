Here are three new cognitive models that explore different mechanisms of learning and choice behavior, building upon the insights from the best model while introducing novel parameter combinations and dynamics.

### Model 1: Asymmetric Learning Rates for Positive and Negative Prediction Errors
This model hypothesizes that the participant learns differently from better-than-expected outcomes compared to worse-than-expected outcomes. It features separate learning rates for positive and negative reward prediction errors (RPEs), alongside separate inverse temperatures and perseveration bonuses for each stage.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with asymmetric learning rates for positive and negative RPEs.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers_1: perseveration bonus for stage 1 [-5, 5]
    pers_2: perseveration bonus for stage 2 [-5, 5]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if a2 != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers_2
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
            alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
            
            # ACTION VALUE UPDATING FOR CHOICE 1
            q_stage1_mf[a1] += alpha_1 * delta_stage1 + alpha_2 * lam * delta_stage2
            
            # ACTION VALUE UPDATING FOR CHOICE 2
            q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
            
            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning Rates with Decay
In this model, learning rates are divided by task stage rather than outcome valence. This assumes that participants might update their values at the first step (spaceship choice) differently than the second step (alien choice). It also includes a global perseveration parameter and decay for unchosen actions.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with separate learning rates for each stage and decay for unchosen options.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2: learning rate for stage 2 [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers: perseveration bonus for both stages [-5, 5]
    decay: forgetting rate for unchosen options [0, 1]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if a2 != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            # ACTION VALUE UPDATING FOR CHOICE 1
            q_stage1_mf[a1] += alpha_1 * delta_stage1 + alpha_1 * lam * delta_stage2
            
            # ACTION VALUE UPDATING FOR CHOICE 2
            q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
            
            # Decay unchosen options
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] *= (1 - decay)
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] *= (1 - decay)

            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] *= (1 - decay)
            q_stage2_mf[unvisited_state, 1] *= (1 - decay)
            
            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Neutral-Point Decay 
While the best model so far decayed unchosen options towards zero, this model decays unchosen values towards a neutral expectation of 0.5. Given that rewards are strictly 0 or 1, defaulting unchosen actions to 0.5 ensures that prolonged ignorance of an option slowly resets it to a neutral expected value rather than complete aversion, which may better capture dynamic exploration.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with decay for unchosen options towards 0.5 (center).
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers_1: perseveration bonus for stage 1 [-5, 5]
    pers_2: perseveration bonus for stage 2 [-5, 5]
    decay: forgetting rate for unchosen options towards 0.5 [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values at 0.5
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if a2 != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers_2
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            # ACTION VALUE UPDATING FOR CHOICE 1
            q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
            
            # ACTION VALUE UPDATING FOR CHOICE 2
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2
            
            # Decay unchosen options towards 0.5
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])

            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
