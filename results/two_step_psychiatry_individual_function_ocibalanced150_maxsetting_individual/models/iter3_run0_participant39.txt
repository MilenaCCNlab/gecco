Here are 3 new cognitive models based on the two-step task. Each model introduces a different well-established psychological mechanism to better explain the participant's choices.

### Model 1: Hybrid Model with Perseveration and Q-value Decay (Forgetting)
This model introduces a `decay` parameter. In human reinforcement learning, participants often forget the value of options they haven't chosen recently. Here, the values of unchosen actions decay towards zero, which can explain why participants might suddenly explore an option they haven't tried in a while.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Choice Perseveration and Q-value Decay.
    Unchosen Q-values decay towards zero, simulating forgetting of unchosen options over time.
    
    Parameters:
    learning_rate: learning rate for updating Q-values [0,1]
    beta: inverse temperature for softmax choice [0,10]
    w: weight for model-based vs model-free control [0,1]
    pers: perseveration bonus for repeating the previous stage 1 choice [-5, 5]
    decay: forgetting rate for unchosen Q-values [0,1]
    """
    learning_rate, beta, w, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_term = np.zeros(2)
        if last_action_1 != -1:
            pers_term[last_action_1] = pers
            
        q1_vals = beta * q_stage1_net + pers_term
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_vals = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update Chosen Actions
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Decay Unchosen Actions
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        unchosen_a2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_a2] *= (1 - decay)
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1 - decay)
        q_stage2_mf[other_state, 1] *= (1 - decay)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning Rates and Betas with Eligibility Trace
This model provides more granularity by separating both the learning rates (`alpha1`, `alpha2`) and inverse temperatures (`beta1`, `beta2`) for Stage 1 and Stage 2. It also includes an eligibility trace (`lam`) which dictates how much the Stage 2 reward directly updates the Stage 1 action value.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Stage-Specific Learning Rates and Temperatures, Eligibility Trace, and Perseveration.
    
    Parameters:
    alpha1: learning rate for Stage 1 [0,1]
    alpha2: learning rate for Stage 2 [0,1]
    beta1: inverse temperature for Stage 1 [0,10]
    beta2: inverse temperature for Stage 2 [0,10]
    w: model-based weight [0,1]
    pers: stage 1 choice perseveration [-5, 5]
    lam: eligibility trace for stage 2 reward directly updating stage 1 [0,1]
    """
    alpha1, alpha2, beta1, beta2, w, pers, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_term = np.zeros(2)
        if last_action_1 != -1:
            pers_term[last_action_1] = pers
            
        q1_vals = beta1 * q_stage1_net + pers_term
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Stage 1 Update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Stage 2 Update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates (Positive vs. Negative Prediction Errors)
Humans often exhibit asymmetric updating for better-than-expected versus worse-than-expected outcomes. This model assigns different learning rates depending on the sign of the prediction error (`alpha_pos` for positive, `alpha_neg` for negative). This is combined with stage-specific choice temperatures.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Learning Rates (Pos/Neg PE), Stage-Specific Temperatures, and Perseveration.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    pers: stage 1 choice perseveration [-5, 5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_term = np.zeros(2)
        if last_action_1 != -1:
            pers_term[last_action_1] = pers
            
        q1_vals = beta1 * q_stage1_net + pers_term
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Stage 1 Update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Stage 2 Update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
