def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for stage 1 and stage 2, with stickiness at both stages.
    
    This model extends the hybrid model by allowing different learning rates for the 
    first-stage and second-stage Q-value updates, which can capture differences in 
    learning speed between the choice of spaceship and the choice of alien.
    
    Bounds:
    learning_rate_1: [0, 1] - Learning rate for stage 1 Q-value updates
    learning_rate_2: [0, 1] - Learning rate for stage 2 Q-value updates
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter
    pi_1: [-5, 5] - Stickiness parameter for repeating the previous stage 1 choice
    pi_2: [-5, 5] - Stickiness parameter for repeating the previous stage 2 choice in the given state
    """
    learning_rate_1, learning_rate_2, beta, w, lam, pi_1, pi_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + pi_1 * stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = s

        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = 1.0
            
        logits_2 = beta * q_stage2_mf[state_idx] + pi_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate_1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate_2 * delta_stage2
        q_stage1_mf[a1] += learning_rate_1 * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with choice stickiness based on an exponentially decaying choice trace.
    
    Instead of only tracking the immediately preceding choice, this model maintains a 
    continuous "choice trace" that updates over time. This captures a longer-term 
    habitual tendency to choose options that have been frequently selected in the past.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter
    pi_1: [-5, 5] - Weight of the choice trace at stage 1
    pi_2: [-5, 5] - Weight of the choice trace at stage 2
    alpha_trace: [0, 1] - Update rate for the choice trace (higher means faster updating)
    """
    learning_rate, beta, w, lam, pi_1, pi_2, alpha_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net + pi_1 * choice_trace_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = s

        logits_2 = beta * q_stage2_mf[state_idx] + pi_2 * choice_trace_2[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Update choice traces
        choice_trace_1 *= (1 - alpha_trace)
        choice_trace_1[a1] += alpha_trace
        
        choice_trace_2[state_idx] *= (1 - alpha_trace)
        choice_trace_2[state_idx, a2] += alpha_trace

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with choice stickiness and forgetting of unchosen Q-values.
    
    This model assumes that the values of unchosen options are not perfectly retained 
    but instead decay back towards a neutral baseline expectation (0.5) over time. 
    This captures a form of memory decay or uncertainty about unchosen options.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter
    pi_1: [-5, 5] - Stickiness parameter for repeating the previous stage 1 choice
    pi_2: [-5, 5] - Stickiness parameter for repeating the previous stage 2 choice
    unchosen_decay: [0, 1] - Rate at which unchosen Q-values decay towards 0.5
    """
    learning_rate, beta, w, lam, pi_1, pi_2, unchosen_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    
    # Initialize Q-values to 0.5 (neutral expectation)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + pi_1 * stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = s

        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = 1.0
            
        logits_2 = beta * q_stage2_mf[state_idx] + pi_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        # Updates for chosen actions
        q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Decay for unchosen actions towards 0.5
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += unchosen_decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        unchosen_a2 = 1 - a2
        q_stage2_mf[state_idx, unchosen_a2] += unchosen_decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])
        
        unvisited_s = 1 - state_idx
        q_stage2_mf[unvisited_s, 0] += unchosen_decay * (0.5 - q_stage2_mf[unvisited_s, 0])
        q_stage2_mf[unvisited_s, 1] += unchosen_decay * (0.5 - q_stage2_mf[unvisited_s, 1])

        last_action_1 = a1
        last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
