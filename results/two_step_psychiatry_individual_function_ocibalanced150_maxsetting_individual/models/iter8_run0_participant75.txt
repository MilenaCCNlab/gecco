def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice Kernel for Stage 1 and Stage 2
    This model uses a choice kernel (exponentially decaying trace of past choices) 
    for both stage 1 and stage 2 to capture complex perseveration or alternation patterns,
    independent of the reward received.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    alpha_ck: [0, 1] - Learning rate for the choice kernel.
    pers_s1: [-5, 5] - Perseveration weight for stage 1 choice kernel.
    pers_s2: [-5, 5] - Perseveration weight for stage 2 choice kernel.
    """
    alpha, beta1, beta2, w, lam, alpha_ck, pers_s1, pers_s2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_stage1_net + pers_s1 * ck_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx] + pers_s2 * ck_2[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # update choice kernels
        ck_1 = (1 - alpha_ck) * ck_1
        ck_1[a1] += alpha_ck
        
        ck_2[state_idx] = (1 - alpha_ck) * ck_2[state_idx]
        ck_2[state_idx, a2] += alpha_ck

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Eligibility Trace + Reward-Dependent Perseveration
    This model allows the eligibility trace (lambda) to differ depending on whether 
    the stage 2 prediction error is positive or negative, reflecting asymmetric 
    credit assignment for better-than-expected vs worse-than-expected outcomes.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam_pos: [0, 1] - Eligibility trace for positive stage 2 PE.
    lam_neg: [0, 1] - Eligibility trace for negative stage 2 PE.
    pers_win: [-5, 5] - Perseveration parameter for stage 1 after a win.
    pers_lose: [-5, 5] - Perseveration parameter for stage 1 after a loss.
    """
    alpha, beta1, beta2, w, lam_pos, lam_neg, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_r == 1:
                perseveration_1[prev_a1] = pers_win
            else:
                perseveration_1[prev_a1] = pers_lose
                
        logits_1 = beta1 * q_stage1_net + perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # asymmetric eligibility trace update
        if delta_stage2 > 0:
            q_stage1_mf[a1] += alpha * lam_pos * delta_stage2
        else:
            q_stage1_mf[a1] += alpha * lam_neg * delta_stage2
            
        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-Dependent Perseveration
    This model assumes that perseveration on the stage 1 choice depends not only 
    on the reward but also on whether the transition to stage 2 was common or rare.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win_common: [-5, 5] - Perseveration after a win on a common transition.
    pers_lose_common: [-5, 5] - Perseveration after a loss on a common transition.
    pers_rare: [-5, 5] - Perseveration after any rare transition.
    """
    alpha, beta1, beta2, w, lam, pers_win_common, pers_lose_common, pers_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_r = -1
    prev_trans_common = False

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            prev_trans_common = False
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_trans_common:
                if prev_r == 1:
                    perseveration_1[prev_a1] = pers_win_common
                else:
                    perseveration_1[prev_a1] = pers_lose_common
            else:
                perseveration_1[prev_a1] = pers_rare
                
        logits_1 = beta1 * q_stage1_net + perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r
        prev_trans_common = (a1 == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
