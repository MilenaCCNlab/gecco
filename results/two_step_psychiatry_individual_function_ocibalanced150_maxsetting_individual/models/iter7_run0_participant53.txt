Here are 3 new cognitive models, expressed as Python functions, that build upon the most successful model so far (Hybrid Model-Based / Model-Free with Decay and Perseveration) by introducing novel, theoretically grounded mechanisms to explain the participant's decision-making behavior.

### Model 1: Asymmetric Valuation of Losses (Punishment)
This model introduces a `punishment` parameter. Standard models assume unrewarded trials yield a reward of 0. However, participants often perceive omissions of expected rewards as active punishments. This model maps unrewarded trials to a negative value (`-punishment`), scaling the prediction error depending on how strongly the participant dislikes not receiving a coin.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based / Model-Free learning with Decay, Perseveration, and Punishment.
    Instead of unrewarded trials having a reward of 0, they have a reward of -punishment.
    This captures asymmetric valuation of losses (omissions) vs gains.
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    decay: [0, 1] - decay rate for unchosen options
    p: [-10, 10] - general perseveration bonus for stage 1
    punishment: [0, 1] - negative reward value for unrewarded trials
    """
    alpha, beta_1, beta_2, w, lam, decay, p, punishment = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    # Values initialized to 0 since rewards are 1 and -punishment
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        
        # Effective reward mapping
        r = 1.0 if reward[trial] == 1 else -punishment

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net

        if trial > 0:
            logits_1[action_1[trial - 1]] += p
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        unchosen_a1 = 1 - a1
        # Decay targets 0, aligning with the initialization
        q_stage1_mf[unchosen_a1] += decay * (0.0 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay * (0.0 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Counterfactual Updating
This model introduces `alpha_cf`. Often in these tasks, participants assume that if one alien on a planet gives gold, the other alien on that same planet likely does not (anti-correlation). This model explicitly updates the value of the unchosen alien on the visited planet based on the inverse of the reward received.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model-Based / Model-Free learning with Decay, Perseveration, and Counterfactual Updating.
    Updates the unchosen alien on the visited planet based on the assumption that its outcome 
    is anti-correlated with the chosen alien.
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    decay: [0, 1] - decay rate for unchosen options
    p: [-10, 10] - general perseveration bonus
    alpha_cf: [0, 1] - counterfactual learning rate for unchosen stage 2 action
    """
    alpha, beta_1, beta_2, w, lam, decay, p, alpha_cf = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net

        if trial > 0:
            logits_1[action_1[trial - 1]] += p
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Counterfactual update for unchosen action at stage 2
        unchosen_a2 = 1 - a2
        cf_r = 1.0 - r
        q_stage2_mf[s, unchosen_a2] += alpha_cf * (cf_r - q_stage2_mf[s, unchosen_a2])

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s): # Decay only applies to unvisited states here
                    q_stage2_mf[s_idx, a_idx] += decay * (0.5 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Eligibility Traces
This model assumes that credit assignment from stage 2 back to stage 1 differs depending on whether the outcome was better or worse than expected. It splits the eligibility trace (`lam`) into `lam_pos` and `lam_neg`, capturing optimistic or pessimistic reinforcement tracking.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model-Based / Model-Free learning with Decay, Perseveration, and Asymmetric Eligibility Traces.
    The eligibility trace parameter (lam) differs depending on whether the stage 2 prediction error is positive or negative.
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam_pos: [0, 1] - eligibility trace for positive PE
    lam_neg: [0, 1] - eligibility trace for negative PE
    decay: [0, 1] - decay rate for unchosen options
    p: [-10, 10] - general perseveration bonus
    """
    alpha, beta_1, beta_2, w, lam_pos, lam_neg, decay, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net

        if trial > 0:
            logits_1[action_1[trial - 1]] += p
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        # Asymmetric eligibility trace based on PE valence
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay * (0.5 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
