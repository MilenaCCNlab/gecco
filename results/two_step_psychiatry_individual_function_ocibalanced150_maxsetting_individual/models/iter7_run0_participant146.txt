Here are 3 new cognitive models that introduce different mechanisms to explain participant behavior, specifically focusing on win-stay-lose-shift (WSLS) heuristics, motor perseveration, and continuous choice traces.

### Model 1: Q-learning with Win-Stay-Lose-Shift (WSLS) Bias
This model adds an explicit heuristic parameter, `wsls`, which acts as a bonus to repeat the previous stage 1 choice if it was rewarded, and a penalty if it was not.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-learning with stage-specific stickiness and a win-stay-lose-shift (WSLS) bias.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Model-based weight for stage 1
    lam: [0, 1] - Eligibility trace
    stickiness_1: [-5, 5] - Perseveration for stage 1
    stickiness_2: [-5, 5] - Perseveration for stage 2
    wsls: [-5, 5] - Win-stay-lose-shift bias for stage 1 (positive if rewarded, negative if not)
    """
    learning_rate, beta, w, lam, stickiness_1, stickiness_2, wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness_1
            if last_reward == 1:
                q_stage1_net[last_action_1] += wsls
            else:
                q_stage1_net[last_action_1] -= wsls
            
        q1_scaled = beta * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_state == state_idx and last_action_2 != -1:
            q_stage2_net[last_action_2] += stickiness_2
            
        q2_scaled = beta * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Q-learning with Motor Stickiness
This model introduces `motor_stickiness` to differentiate between state-specific perseveration (choosing the same alien) and motor perseveration (pressing the same button regardless of the planet).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Q-learning with decay, stage 1 stickiness, state-specific stage 2 stickiness, and motor stickiness.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    decay_rate: [0, 1] - Decay rate for unchosen actions towards 0.5
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    stickiness_1: [-5, 5] - Perseveration for stage 1
    stickiness_2: [-5, 5] - State-specific perseveration for stage 2
    motor_stickiness: [-5, 5] - General motor perseveration for stage 2 regardless of state
    """
    learning_rate, decay_rate, beta, w, lam, stickiness_1, stickiness_2, motor_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness_1
            
        q1_scaled = beta * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1:
            q_stage2_net[last_action_2] += motor_stickiness
            if last_state == state_idx:
                q_stage2_net[last_action_2] += stickiness_2
            
        q2_scaled = beta * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Q-learning with Decaying Choice Traces
Instead of only observing the immediate last trial, this model incorporates a continuous choice trace for perseveration that exponentially decays over trials using parameter `tau`.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Q-learning with decay, and a continuous choice trace for perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    decay_rate: [0, 1] - Decay rate for unchosen actions towards 0.5
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    stickiness_1: [-5, 5] - Perseveration weight for stage 1
    stickiness_2: [-5, 5] - Perseveration weight for stage 2
    tau: [0, 1] - Decay rate of the choice trace (1 = no decay, 0 = only last trial)
    """
    learning_rate, decay_rate, beta, w, lam, stickiness_1, stickiness_2, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf + stickiness_1 * choice_trace_1
            
        q1_scaled = beta * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_stage2_net = q_stage2_mf[state_idx] + stickiness_2 * choice_trace_2[state_idx]
            
        q2_scaled = beta * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
        
        # Update choice traces
        choice_trace_1 *= tau
        choice_trace_1[action_1[trial]] += 1
        
        choice_trace_2 *= tau
        choice_trace_2[state_idx, action_2[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
