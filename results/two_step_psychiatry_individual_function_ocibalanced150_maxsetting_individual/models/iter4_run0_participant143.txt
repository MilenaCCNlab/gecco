Here are 3 new cognitive models that explore alternative mechanisms for participant behavior in the two-step task. They avoid the parameter combinations you have already tested and introduce new theoretically grounded features: **Loss Penalty (asymmetric reward valuation)**, **Counterfactual Learning**, and **Transition Matrix Learning**.

### Model 1: Loss Penalty Model
Participants often do not treat a missed reward (0 coins) as a neutral outcome, but rather as a negative outcome (a loss). This model scales the negative impact of unrewarded trials by treating them as a negative prediction error driven by a `loss_penalty` parameter, rather than just 0.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with a loss penalty for unrewarded trials.
    Unrewarded trials (0 coins) are treated as a negative reward (-loss_penalty) rather than 0.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p: [0, 5] - Perseveration/stickiness for repeating the last stage 1 choice.
    loss_penalty: [0, 5] - Negative reward value assigned to 0-coin outcomes.
    """
    alpha, beta_1, beta_2, w, lam, p, loss_penalty = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        
        # Apply loss penalty for 0-coin outcomes
        r = 1.0 if reward[trial] > 0 else -loss_penalty

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p * pers
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Counterfactual Learning Model
When choosing an alien on a planet, the participant might assume the unchosen alien on that same planet would have given the opposite reward (i.e. if I got 1 coin, the other probably would have given 0). This model applies an inverse update to the unchosen Stage 2 action.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with counterfactual learning at Stage 2.
    The unchosen action at Stage 2 is updated assuming it would have yielded the opposite reward.
    
    Parameters:
    alpha: [0, 1] - Learning rate for chosen actions.
    alpha_cf: [0, 1] - Learning rate for the unchosen action at Stage 2.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p: [0, 5] - Perseveration/stickiness for repeating the last stage 1 choice.
    """
    alpha, alpha_cf, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Counterfactual update for unchosen action at stage 2
        unchosen_a2 = 1 - a2
        cf_r = 1.0 - r
        q_stage2_mf[state_idx, unchosen_a2] += alpha_cf * (cf_r - q_stage2_mf[state_idx, unchosen_a2])

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Dynamic Transition Matrix Model
Instead of assuming the participant possesses a perfectly fixed model of the transition matrix (0.7 / 0.3) from the beginning, this model allows the participant to actively learn the transition probability matrix based on the observed transitions over time.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with Transition Matrix Learning.
    The participant dynamically updates their internal model of the spaceship transitions based on observations.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p: [0, 5] - Perseveration/stickiness for repeating the last stage 1 choice.
    alpha_t: [0, 1] - Learning rate for the transition matrix probabilities.
    """
    alpha, beta_1, beta_2, w, lam, p, alpha_t = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix as uniform (0.5), participant builds MB control as they learn
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy (using dynamically learned transition matrix)
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Q-value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Transition matrix update (updating probability of state given action)
        transition_matrix[a1, state_idx] += alpha_t * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] = 1.0 - transition_matrix[a1, state_idx]

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
