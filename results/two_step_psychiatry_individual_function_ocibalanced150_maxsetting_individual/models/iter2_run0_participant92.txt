def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with transition matrix learning. The participant updates their internal 
    model of the environment's transition probabilities based on observed transitions, 
    allowing the model-based system to adapt to perceived changes in the environment.
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers: [-5,5]
    alpha_trans: [0,1]
    """
    alpha, beta_1, beta_2, w, lam, pers, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + pers * pers_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Update transition matrix for the chosen stage 1 action
        transition_matrix[a1, s] += alpha_trans * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_trans * (0.0 - transition_matrix[a1, 1 - s])
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model where the weight of model-based control depends on the previous 
    trial's reward (w_win and w_lose). This captures shifts in planning strategies 
    following success versus failure (e.g., relying more on habits after non-reward).
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w_win: [0,1]
    w_lose: [0,1]
    lam: [0,1]
    pers: [-5,5]
    """
    alpha, beta_1, beta_2, w_win, w_lose, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:

            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Select model-based weight based on previous trial's outcome
        w = w_win if last_reward > 0 else w_lose
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)

    eps = 1e-10
