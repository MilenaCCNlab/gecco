import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates, inverse temperatures, and stickiness for Stage 1 and Stage 2.
    
    Bounds:
    alpha_1: [0,1] - Learning rate for stage 1.
    alpha_2: [0,1] - Learning rate for stage 2.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weighting of model-based vs model-free values.
    lambda_: [0,1] - Eligibility trace.
    stick_1: [0,10] - Stickiness for stage 1.
    stick_2: [0,10] - Stickiness for stage 2.
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lambda_, stick_1, stick_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stick_1 / (beta_1 + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta_1 * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1:
            q_stage2_net[last_action_2] += stick_2 / (beta_2 + 1e-10)
            
        max_q2 = np.max(q_stage2_net)
        exp_q2 = np.exp(beta_2 * (q_stage2_net - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with learned transition matrix, Q-value decay, and stage 1 & 2 stickiness.
    
    Bounds:
    alpha: [0,1] - Learning rate for Q-values.
    beta: [0,10] - Inverse temperature for softmax.
    w: [0,1] - Model-based weight.
    lambda_: [0,1] - Eligibility trace.
    stick_1: [0,10] - Stickiness for stage 1.
    stick_2: [0,10] - Stickiness for stage 2.
    alpha_trans: [0,1] - Learning rate for transition matrix.
    decay: [0,1] - Decay rate for unchosen Q-values.
    """
    alpha, beta, w, lambda_, stick_1, stick_2, alpha_trans, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stick_1 / (beta + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1:
            q_stage2_net[last_action_2] += stick_2 / (beta + 1e-10)
            
        max_q2 = np.max(q_stage2_net)
        exp_q2 = np.exp(beta * (q_stage2_net - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Transition matrix update
        for s in range(2):
            if s == state_idx:
                transition_matrix[action_1[trial], s] += alpha_trans * (1 - transition_matrix[action_1[trial], s])
            else:
                transition_matrix[action_1[trial], s] += alpha_trans * (0 - transition_matrix[action_1[trial], s])
        
        # Decay unchosen
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with reward-dependent stickiness (Win-Stay/Lose-Shift) for stage 1, separate inverse temperatures, and Q-value decay.
    
    Bounds:
    alpha: [0,1] - Learning rate for Q-values.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight.
    lambda_: [0,1] - Eligibility trace.
    stick_rew: [-5,5] - Stickiness for stage 1 if previous trial was rewarded.
    stick_unrew: [-5,5] - Stickiness for stage 1 if previous trial was unrewarded.
    decay: [0,1] - Decay rate for unchosen Q-values.
    """
    alpha, beta_1, beta_2, w, lambda_, stick_rew, stick_unrew, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
