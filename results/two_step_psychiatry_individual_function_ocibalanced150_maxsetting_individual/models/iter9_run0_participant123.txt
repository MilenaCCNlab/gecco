Here are 3 new cognitive models that explore alternative mechanisms for decision-making in the two-step task, avoiding previously tried parameter combinations. Each is constrained to a maximum of 8 parameters.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model allows the agent to update values at different speeds for the first and second stages. This is useful if the participant adapts quickly to the changing alien reward probabilities (Stage 2) but maintains a more stable, slowly-updating evaluation of the spaceships (Stage 1).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for Stage 1 and Stage 2, separate MB and MF temperatures.
    This model allows the two stages to learn at different speeds, while retaining independent 
    scaling of MB and MF values, eligibility traces, and stage-specific stickiness.
    
    Bounds:
    learning_rate1: [0, 1] (learning rate for stage 1)
    learning_rate2: [0, 1] (learning rate for stage 2)
    beta_mb: [0, 10] (inverse temperature for model-based system)
    beta_mf: [0, 10] (inverse temperature for model-free system)
    beta2: [0, 10] (inverse temperature for stage 2)
    lam: [0, 1] (eligibility trace parameter)
    pi1: [-5, 5] (choice stickiness/perseveration for stage 1)
    pi2: [-5, 5] (choice stickiness/perseveration for stage 2)
    """
    learning_rate1, learning_rate2, beta_mb, beta_mf, beta2, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        if a2 == -1:
            p_choice_2[trial] = 1.0
            last_a1 = a1
            continue

        # Stage 2 policy
        stickiness2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            stickiness2[last_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx, :] + pi2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate2 * delta_stage2

        # Eligibility trace update to Stage 1 uses learning_rate1
        q_stage1_mf[a1] += learning_rate1 * lam * delta_stage2

        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Choice Trace (Exponential Perseveration)
Instead of a simple 1-trial stickiness parameter, this model tracks an exponentially decaying choice trace. This allows the model to capture longer-term perseverative tendencies (habits) that build up over consecutive identical choices.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Choice trace model with separate MB and MF temperatures.
    Instead of 1-trial stickiness, this model uses an exponentially updated choice trace 
    to capture longer-term perseveration in Stage 1 choices, along with Q-value decay.
    
    Bounds:
    learning_rate: [0, 1] (learning rate for updating values)
    decay: [0, 1] (decay rate for unchosen options)
    beta_mb: [0, 10] (inverse temperature for model-based system)
    beta_mf: [0, 10] (inverse temperature for model-free system)
    beta2: [0, 10] (inverse temperature for stage 2)
    lam: [0, 1] (eligibility trace parameter)
    pi_trace: [-5, 5] (weight of choice trace on Stage 1 choice)
    alpha_trace: [0, 1] (update rate for choice trace)
    """
    learning_rate, decay, beta_mb, beta_mf, beta2, lam, pi_trace, alpha_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi_trace * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Update choice trace
        choice_trace = (1 - alpha_trace) * choice_trace
        choice_trace[a1] += alpha_trace

        if a2 == -1:
            p_choice_2[trial] = 1.0
            continue

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        # Decay unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Eligibility Traces (Positive vs. Negative RPEs)
This model separates the eligibility trace parameter into `lam_pos` and `lam_neg`. This tests the hypothesis that positive second-stage prediction errors (unexpected rewards) backpropagate differently to first-stage values than negative prediction errors (unexpected omissions).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric eligibility traces for positive and negative RPEs.
    This model separates the eligibility trace parameter into lam_pos and lam_neg,
    allowing positive and negative second-stage prediction errors to back-propagate 
    differently to first-stage model-free values.
    
    Bounds:
    learning_rate: [0, 1] (learning rate for updating values)
    beta_mb: [0, 10] (inverse temperature for model-based system)
    beta_mf: [0, 10] (inverse temperature for model-free system)
    beta2: [0, 10] (inverse temperature for stage 2)
    lam_pos: [0, 1] (eligibility trace for positive RPEs)
    lam_neg: [0, 1] (eligibility trace for negative RPEs)
    pi1: [-5, 5] (choice stickiness/perseveration for stage 1)
    pi2: [-5, 5] (choice stickiness/perseveration for stage 2)
    """
    learning_rate, beta_mb, beta_mf, beta2, lam_pos, lam_neg, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        if a2 == -1:
            p_choice_2[trial] = 1.0
            last_a1 = a1
            continue

        # Stage 2 policy
        stickiness2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            stickiness2[last_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx, :] + pi2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        # Asymmetric eligibility trace application
        current_lam = lam_pos if delta_stage2 >= 0 else lam_neg
        q_stage1_mf[a1] += learning_rate * current_lam * delta_stage2

        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
