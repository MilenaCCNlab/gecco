def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Reward Learning + Transition Learning
    This model extends the Daw model by incorporating dynamic learning of the transition 
    probabilities and asymmetric learning rates for positive and negative prediction errors. 
    People often learn differently from better-than-expected vs worse-than-expected outcomes.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (delta > 0).
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (delta <= 0).
    beta1: [0, 10] - Inverse temperature for stage 1 choice.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration parameter for stage 1 choice.
    alpha_trans: [0, 1] - Learning rate for updating transition probabilities.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers, alpha_trans = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            perseveration_1[prev_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pers * perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Update transition probabilities
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha_2 * lam * delta_stage2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Value Decay + Transition Learning
    This model assumes that participants dynamically learn the transition probabilities,
    but also gradually forget the values of unchosen options over time, pulling them 
    back towards a neutral prior (0.5).
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration parameter for stage 1 choice.
    alpha_trans: [0, 1] - Learning rate for updating transition probabilities.
    decay: [0, 1] - Decay rate pulling unchosen Q-values toward 0.5.
    """
    alpha, beta1, beta2, w, lam, pers, alpha_trans, decay = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            perseveration_1[prev_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pers * perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Update transition probabilities
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Decay unchosen options toward 0.5
        q_stage1_mf[1 - a1] += decay * (0.5 - q_stage1_mf[1 - a1])
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate MB/MF Temperatures + Transition Learning
    Instead of using a single mixing weight 'w', this model assigns independent 
    inverse temperatures to the model-based and model-free Q-values at stage 1. 
    This allows for independent scaling of the two systems' contributions.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values (same for both stages).
    beta_mb: [0, 10] - Inverse temperature scaling the model-based Q-values at stage 1.
    beta_mf: [0, 10] - Inverse temperature scaling the model-free Q-values at stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration parameter for stage 1 choice.
    alpha_trans: [0, 1] - Learning rate for updating transition probabilities.
    """
    alpha, beta_mb, beta_mf, beta2, lam, pers, alpha_trans = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            perseveration_1[prev_a1] = 1.0
            
        # Independent scaling of MB and MF values
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers * perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Update transition probabilities
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
