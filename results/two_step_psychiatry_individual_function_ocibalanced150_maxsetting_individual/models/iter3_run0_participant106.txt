def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate stage 1 and stage 2 learning rates, separate temperatures,
    model-based weight, eligibility trace, stage 1 perseveration, and decay.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1 Q-values.
    alpha2: [0, 1] - Learning rate for stage 2 Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1 choice.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace for updating stage 1 based on stage 2 reward.
    rho: [0, 10] - Perseveration bonus for repeating the previous stage 1 choice.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.
    """
    alpha1, alpha2, beta1, beta2, w, lam, rho, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1
            
        logits1 = beta1 * q_stage1_net + rho * pers1
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
