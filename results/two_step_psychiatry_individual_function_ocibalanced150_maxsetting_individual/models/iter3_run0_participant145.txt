def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MF/MB mixture with decay and stage-specific perseveration.
    Adds a separate perseveration parameter for the second-stage choice (alien),
    allowing the model to capture independent stickiness at both decision stages,
    along with value decay for unchosen options.
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    decay: [0,1] - decay rate for unchosen Q-values
    rep1: [-5,5] - stage 1 perseveration
    rep2: [-5,5] - stage 2 perseveration
    """
    alpha, beta1, beta2, w, lam, decay, rep1, rep2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.full(2, -1)
    
    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_term1 = np.zeros(2)
        if last_action1 != -1:
            rep_term1[last_action1] = 1
            
        logits1 = beta1 * q_net + rep1 * rep_term1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        rep_term2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            rep_term2[last_action2[state_idx]] = 1
            
        logits2 = beta2 * q_stage2_mf[state_idx] + rep2 * rep_term2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MF/MB mixture with action and state perseveration.
    Participant behavior shows they often choose the spaceship that commonly leads 
    to the planet they just visited (e.g. switching to Sp1 after landing on Pl1 via Sp0).
    This is modeled with a state perseveration parameter `rep_state`, alongside standard `rep_action`.
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    decay: [0,1] - decay rate for unchosen Q-values
    rep_action: [-5,5] - perseveration on the previous stage 1 choice
    rep_state: [-5,5] - perseveration on the spaceship that commonly leads to previous planet
    """
    alpha, beta1, beta2, w, lam, decay, rep_action, rep_state = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_state = -1
    
    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_term = np.zeros(2)
        if last_action != -1:
            rep_term[last_action] += rep_action
            rep_term[last_state] += rep_state
            
        logits1 = beta1 * q_net + rep_term
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action = action_1[trial]
        last_state = state_idx
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MF/MB mixture with choice trace (running perseveration) and value decay.
    Instead of only perseverating on the immediately preceding choice, the model maintains a 
    running choice trace that builds up over repeated choices, explaining long blocks of 
    staying with one spaceship.
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    decay: [0,1] - decay rate for unchosen Q-values
    rep: [-5,5] - perseveration weight
    alpha_c: [0,1] - choice trace update rate
    """
    alpha, beta1, beta2, w, lam, decay, rep, alpha_c = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    choice_trace = np.zeros(2)
    
    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits1 = beta1 * q_net + rep * choice_trace
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        choice_trace[action_1[trial]] += alpha_c * (1 - choice_trace[action_1[trial]])
        choice_trace[unchosen_1] *= (1 - alpha_c)
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
