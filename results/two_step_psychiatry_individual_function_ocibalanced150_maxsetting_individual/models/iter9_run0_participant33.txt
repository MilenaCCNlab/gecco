def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Reward-Dependent Model-Based Weight Model.
    
    This model allows the weighting of the model-based system to vary depending on 
    whether the previous trial was rewarded or not. Participants might shift their 
    reliance on model-based planning after a win versus after a loss.
    
    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w_win: [0, 1] - Weighting of model-based vs model-free values after a win.
        w_lose: [0, 1] - Weighting of model-based vs model-free values after a loss.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1.
    """
    alpha, beta, w_win, w_lose, lambda_, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None
    last_reward = 0

    for trial in range(n_trials):
        w = w_win if last_reward == 1 else w_lose

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Stage 2 Anti-Correlated Update Model.
    
    This model assumes that participants might infer the value of the unchosen alien 
    on the visited planet based on the outcome of the chosen alien. Since reward 
    probabilities might be perceived as anti-correlated, if one alien yields a reward, 
    the other is assumed less likely to, and vice versa.
    
    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1.
        omega: [0, 1] - Weight for updating the unchosen stage 2 action inversely.
    """
    alpha, beta, w, lambda_, pi, omega = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Update unchosen action in stage 2 based on the anti-correlated assumption
        unchosen_a2 = 1 - action_2[trial]
        delta_unchosen = (1 - reward[trial]) - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += alpha * omega * delta_unchosen

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Subjective Transition Probability Model.
    
    Instead of assuming the participant relies on the exact objective transition 
    probabilities (0.7/0.3), this model fits a subjective transition probability 
    parameter, allowing for over- or under-estimation of the transition structure.
    
    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1.
        subj_trans: [0.5, 1.0] - Subjective probability of the common transition.
    """
    alpha, beta, w, lambda_, pi, subj_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[subj_trans, 1 - subj_trans], [1 - subj_trans, subj_trans]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
