import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) with subjective transition probability.
    Instead of assuming fixed 0.7/0.3 transitions, the model estimates a subjective
    probability for the common transition. This captures participants who might not 
    fully believe or learn the exact 70/30 transition structure.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi: [-5, 5] - stage 1 choice stickiness
    p_common: [0.5, 1] - subjective probability of common transition
    """
    alpha, beta_1, beta_2, w, lambda_, pi, p_common = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[p_common, 1 - p_common], [1 - p_common, p_common]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_a1 != -1:
            rep[last_a1] = 1.0
            
        q1_scaled = beta_1 * q_stage1_net + pi * rep
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) with asymmetric learning rates for positive and negative 
    prediction errors, and an initial Q-value parameter. This allows for differential 
    sensitivity to better-than-expected versus worse-than-expected outcomes, while 
    starting from a non-zero prior expectation.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi: [-5, 5] - stage 1 choice stickiness
    q0: [0, 1] - initial Q-value for all states and actions
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, pi, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, float(q0))
    q_stage2_mf = np.full((2, 2), float(q0))
    last_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_a1 != -1:
            rep[last_a1] = 1.0
            
        q1_scaled = beta_1 * q_stage1_net + pi * rep
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_stage1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_stage2 * delta_stage2
        
        q_stage1_mf[a1] += alpha_stage2 * lambda_ * delta_stage2

        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) with choice stickiness for both stage 1 and stage 2,
    and an initial Q-value parameter. This model captures habitual motor perseveration 
    or choice stickiness that might occur independently at both decision stages.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi_1: [-5, 5] - stage 1 choice stickiness
    pi_2: [-5, 5] - stage 2 choice stickiness

    """
    alpha, beta_1, beta_2, w, lambda_, pi_1, pi_2, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, float(q0))
    q_stage2_mf = np.full((2, 2), float(q0))
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:

        a1 = action_1[trial]

        rep1 = np.zeros(2)

        if last_a2 != -1:

        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        last_a1 = a1
        last_a2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
