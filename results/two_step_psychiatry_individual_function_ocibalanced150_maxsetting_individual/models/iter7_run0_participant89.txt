def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Asymmetric learning rates for positive and negative prediction errors, with central decay.
    Builds on the best model by differentiating how participants learn from better-than-expected versus worse-than-expected outcomes.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta: inverse temperature [0, 10]
    w: weight of model-based vs model-free control [0, 1]
    lam: eligibility trace parameter [0, 1]
    p: choice perseverance [-5, 5]
    decay: forgetting rate for unchosen options [0, 1]
    center: target expected value for decay [0, 1]
    """
    import numpy as np
    alpha_pos, alpha_neg, beta, w, lam, p, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        exp_q1 = np.exp(beta * q_stage1_net + p * rep)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        q_stage1_mf[1 - action_1[trial]] = (1 - decay) * q_stage1_mf[1 - action_1[trial]] + decay * center

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
        
        # Decay unchosen stage 2 actions
        q_stage2_mf[state_idx, 1 - action_2[trial]] = (1 - decay) * q_stage2_mf[state_idx, 1 - action_2[trial]] + decay * center
        q_stage2_mf[1 - state_idx, 0] = (1 - decay) * q_stage2_mf[1 - state_idx, 0] + decay * center
        q_stage2_mf[1 - state_idx, 1] = (1 - decay) * q_stage2_mf[1 - state_idx, 1] + decay * center

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Choice perseverance for both stage 1 and stage 2, with central decay.
    Assumes that participants may exhibit distinct motor or choice stickiness at both stages, evaluating state-specific perseverance for stage 2.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: weight of model-based vs model-free control [0, 1]
    lam: eligibility trace parameter [0, 1]
    p1: choice perseverance for stage 1 [-5, 5]
    p2: choice perseverance for stage 2 [-5, 5]
    decay: forgetting rate for unchosen options [0, 1]
    center: target expected value for decay [0, 1]
    """
    import numpy as np
    alpha, beta, w, lam, p1, p2, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)
    last_action1 = -1
    last_action2 = np.array([-1, -1])  # Store last action for each state

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if last_action1 != -1:
            rep1[last_action1] = 1
            
        exp_q1 = np.exp(beta * q_stage1_net + p1 * rep1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            rep2[last_action2[state_idx]] = 1

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + p2 * rep2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] = (1 - decay) * q_stage1_mf[1 - action_1[trial]] + decay * center

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay unchosen stage 2 actions
        q_stage2_mf[state_idx, 1 - action_2[trial]] = (1 - decay) * q_stage2_mf[state_idx, 1 - action_2[trial]] + decay * center
        q_stage2_mf[1 - state_idx, 0] = (1 - decay) * q_stage2_mf[1 - state_idx, 0] + decay * center
        q_stage2_mf[1 - state_idx, 1] = (1 - decay) * q_stage2_mf[1 - state_idx, 1] + decay * center

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
