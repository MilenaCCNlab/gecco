import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with short-term stickiness and long-term choice trace perseveration.
    This helps explain the participant's behavior of sticking to one spaceship 
    for long periods and occasionally switching to the other.
    
    Parameters/Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    stickiness: [-5, 5] - bonus for repeating the previous stage 1 action
    choice_trace_weight: [-5, 5] - weight of the long-term choice trace
    choice_trace_decay: [0, 1] - retention rate of the choice trace
    """
    alpha, beta, w, stickiness, choice_trace_weight, choice_trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus = np.zeros(2)
        if last_action != -1:
            stick_bonus[last_action] = stickiness
            
        trace_bonus = choice_trace_weight * choice_trace
        
        q_vals_1 = beta * q_net + stick_bonus + trace_bonus
        q_vals_1 -= np.max(q_vals_1) # Prevent overflow
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        q_vals_2 = beta * q_stage2_mf[state_idx]
        q_vals_2 -= np.max(q_vals_2) # Prevent overflow
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Update stage 1 with stage 2 prediction error (eligibility trace)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage2
        
        # Update choice trace
        choice_trace = choice_trace * choice_trace_decay
        choice_trace[action_1[trial]] += 1.0
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a decay parameter for unchosen action values (forgetting),
    separate inverse temperatures for stage 1 and stage 2,
    and standard hybrid model features.
    
    Parameters/Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stickiness: [-5, 5] - bonus for repeating the previous stage 1 action
    decay: [0, 1] - decay rate for all action values towards 0.5 before updating
    """
    alpha, beta_1, beta_2, w, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus = np.zeros(2)
        if last_action != -1:
            stick_bonus[last_action] = stickiness
            
        q_vals_1 = beta_1 * q_net + stick_bonus
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        q_vals_2 = beta_2 * q_stage2_mf[state_idx]
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay values towards 0.5 prior to updating
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Update stage 1 with stage 2 prediction error (eligibility trace)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model combining asymmetric learning rates for positive and negative prediction errors
    with value decay towards 0.5, and standard hybrid model features.
    
    Parameters/Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    stickiness: [-5, 5] - bonus for repeating the previous stage 1 action
    decay: [0, 1] - decay rate for all action values towards 0.5 before updating
    """
    alpha_pos, alpha_neg, beta, w, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus = np.zeros(2)
        if last_action != -1:
            stick_bonus[last_action] = stickiness
            
        q_vals_1 = beta * q_net + stick_bonus
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        q_vals_2 = beta * q_stage2_mf[state_idx]
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay values towards 0.5 prior to updating
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # Update stage 1 with stage 2 prediction error (eligibility trace)
        q_stage1_mf[action_1[trial]] += alpha_stage2 * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
