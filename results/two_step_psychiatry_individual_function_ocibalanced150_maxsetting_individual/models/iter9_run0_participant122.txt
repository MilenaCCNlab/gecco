def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid model with separate learning rates for positive and negative prediction errors,
    decay of unchosen actions, perseveration, and eligibility trace.
    
    Parameters:
    lr_pos: [0,1] - learning rate for positive prediction errors
    lr_neg: [0,1] - learning rate for negative prediction errors
    decay: [0,1] - decay rate pulling unchosen Q-values toward 0.5
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter (lambda) linking stage 2 reward to stage 1
    """
    lr_pos, lr_neg, decay, beta, w, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += p
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            lr1 = lr_pos if delta_stage1 > 0 else lr_neg
            q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            lr2 = lr_pos if delta_stage2 > 0 else lr_neg
            q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

            q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with reward-dependent perseveration and decay of unchosen actions.
    Perseveration parameter depends on whether the previous trial was rewarded or not.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    decay: [0,1] - decay rate pulling unchosen Q-values toward 0.5
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p_rew: [-5,5] - perseveration bonus if previous trial was rewarded
    p_unrew: [-5,5] - perseveration bonus if previous trial was unrewarded
    lam: [0,1] - eligibility trace parameter (lambda) linking stage 2 reward to stage 1
    """
    learning_rate, decay, beta, w, p_rew, p_unrew, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            if last_reward > 0:
                q_stage1_net[last_action] += p_rew
            else:
                q_stage1_net[last_action] += p_unrew
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
            
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with decay of unchosen actions and initial Q-value as a free parameter.
    Allows for modeling optimistic or pessimistic initialization, which also serves as the decay target.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    decay: [0,1] - decay rate pulling unchosen Q-values toward q_init
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter (lambda) linking stage 2 reward to stage 1
    q_init: [0,1] - initial Q-value and decay target for all states and actions
    """
    learning_rate, decay, beta, w, p, lam, q_init = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * q_init
    q_stage2_mf = np.ones((2, 2)) * q_init
    last_action = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += p
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (q_init - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (q_init - q_stage2_mf[s, a])
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
