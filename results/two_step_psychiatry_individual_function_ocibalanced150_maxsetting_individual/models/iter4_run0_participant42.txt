### Model 1: Dual Learning Rates & Dual Perseveration
This model tests whether participants learn and perseverate differently at the two stages of the task. It separates the learning rates for Stage 1 and Stage 2, and also introduces a separate choice stickiness parameter for Stage 2 (tracking the last action taken in each specific planet state).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual learning rates and dual perseveration.
    Separate learning rates (alpha1, alpha2) and inverse temperatures (beta1, beta2) for each stage.
    Also includes separate perseveration parameters for stage 1 (p1) and stage 2 (p2).
    
    Bounds:
    alpha1: [0, 1] - Learning rate for stage 1
    alpha2: [0, 1] - Learning rate for stage 2
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    p1: [-5, 5] - Choice perseveration for stage 1
    p2: [-5, 5] - Choice perseveration for stage 2
    """
    alpha1, alpha2, beta1, beta2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action1 = -1
    last_action2 = [-1, -1]

    for trial in range(n_trials):
        state_idx = state[trial]

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_stage1_net
        if last_action1 != -1:
            logits_1[last_action1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Stage 2 Policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action2[state_idx] != -1:
            logits_2[last_action2[state_idx]] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Asymmetric Learning Rates + Q-Value Decay
This model tests the hypothesis that participants learn differently from positive versus negative prediction errors (e.g., they might update values more aggressively when receiving a reward than when omitted a reward), while still incorporating the successful Q-value decay mechanism from your best model.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive/negative prediction errors, plus decay.
    Separates learning from positive vs negative RPEs at both stages. Includes Q-value decay.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive RPEs
    alpha_neg: [0, 1] - Learning rate for negative RPEs
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    p: [-5, 5] - Choice perseveration for stage 1
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        state_idx = state[trial]

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += p
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Stage 2 Policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1_eff = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1_eff * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2_eff = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_eff * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2_eff * lam * delta_stage2

        # Q-value Decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Reward-Dependent Perseveration + Q-Value Decay
This model evaluates if the participant's tendency to repeat the previous Stage 1 choice is modulated by whether the previous trial was rewarded (`p_rew`) or unrewarded (`p_unrew`), combining this dynamic stickiness with the previously successful Q-value decay.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent perseveration with Q-value decay.
    Perseveration parameter for stage 1 depends on whether the previous trial was rewarded (p_rew)
    or unrewarded (p_unrew). Also includes decay of unchosen Q-values.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    p_rew: [-5, 5] - Perseveration if previous trial was rewarded
    p_unrew: [-5, 5] - Perseveration if previous trial was unrewarded
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5
    """
    alpha, beta1, beta2, w, lam, p_rew, p_unrew, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        state_idx = state[trial]

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_stage1_net
        
        if last_action != -1:
            if last_reward == 1:
                logits_1[last_action] += p_rew
            elif last_reward == 0:
                logits_1[last_action] += p_unrew
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Stage 2 Policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Q-value Decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
