def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with Choice Traces (Perseveration over multiple past trials).
    Instead of only sticking to the immediately preceding action, this model maintains
    a choice trace that updates via a learning rate, capturing longer-term perseveration
    for both stage 1 and stage 2 choices.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p_1: [-5, 5] - Choice trace weight for stage 1.
    p_2: [-5, 5] - Choice trace weight for stage 2.
    lam: [0, 1] - Eligibility trace.
    alpha_c: [0, 1] - Learning rate for the choice trace.
    """
    alpha, beta_1, beta_2, w, p_1, p_2, lam, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Initialize choice traces
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + p_1 * trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s] + p_2 * trace_2[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update choice traces
        trace_1 = (1 - alpha_c) * trace_1
        trace_1[a1] += alpha_c
        
        trace_2 = (1 - alpha_c) * trace_2
        trace_2[s, a2] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with Reward-Dependent Model-Based Weight.
    The reliance on the model-based system at stage 1 might change depending on whether 
    the participant was rewarded or not on the previous trial, capturing dynamic 
    shifts in decision-making strategy based on recent success.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w_win: [0, 1] - Model-based weight following a rewarded trial.
    w_lose: [0, 1] - Model-based weight following an unrewarded trial.
    p_1: [-5, 5] - Choice stickiness for stage 1.
    p_2: [-5, 5] - Choice stickiness for stage 2.
    lam: [0, 1] - Eligibility trace.
    """
    alpha, beta_1, beta_2, w_win, w_lose, p_1, p_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # Determine model-based weight based on previous reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = 0.5 * (w_win + w_lose) # Default for the first trial

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + p_1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness_2[last_action_2] = 1.0
            
        logits_2 = beta_2 * q_stage2_mf[s] + p_2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with Valence-Dependent Eligibility Trace.
    The eligibility trace parameter (lambda) differs depending on whether the stage 2 
    reward prediction error is positive or negative, allowing for asymmetric updating
    of stage 1 values from stage 2 outcomes.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p_1: [-5, 5] - Choice stickiness for stage 1.
    p_2: [-5, 5] - Choice stickiness for stage 2.
    lam_pos: [0, 1] - Eligibility trace for positive prediction errors.
    lam_neg: [0, 1] - Eligibility trace for negative prediction errors.
    """
    alpha, beta_1, beta_2, w, p_1, p_2, lam_pos, lam_neg = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + p_1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness_2[last_action_2] = 1.0
            
        logits_2 = beta_2 * q_stage2_mf[s] + p_2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Valence-dependent lambda
        lam = lam_pos if delta_stage2 >= 0 else lam_neg
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
