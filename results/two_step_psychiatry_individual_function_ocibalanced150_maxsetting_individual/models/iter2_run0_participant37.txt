Here are 3 new cognitive models that explore alternative mechanisms for learning and decision-making in the two-step task. They introduce new features such as stage-specific choice stickiness, asymmetric learning rates for positive and negative prediction errors, and an exponentially decaying choice trace (perseverance).

### Model 1: Stage-Specific Choice Stickiness
This model introduces separate choice stickiness parameters for Stage 1 and Stage 2. It hypothesizes that participants might have different tendencies to repeat their previous choices depending on whether they are at the first stage (spaceship selection) or the second stage (alien selection, which is state-dependent).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL with eligibility trace and separate choice stickiness for Stage 1 and Stage 2.
    
    Bounds:
    alpha: [0,1] - Learning rate.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace parameter.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    st1: [-5,5] - Choice stickiness for stage 1.
    st2: [-5,5] - Choice stickiness for stage 2.
    """
    alpha, beta1, beta2, lam, w, st1, st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta1 * q_net
        if last_action_1 != -1:
            q1_total[last_action_1] += st1
            
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            q2_total[last_action_2[state_idx]] += st2
            
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Action value updating
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Eligibility Trace
This model assumes that participants learn differently from better-than-expected outcomes compared to worse-than-expected outcomes. It applies separate learning rates for positive and negative prediction errors while maintaining the eligibility trace mechanism and stage 1 stickiness.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL with separate learning rates for positive and negative prediction errors, eligibility trace, and stage 1 choice stickiness.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace parameter.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    st: [-5,5] - Choice stickiness for stage 1.
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta1 * q_net
        if last_action_1 != -1:
            q1_total[last_action_1] += st
            
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Action value updating
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update uses the learning rate of the first stage for consistency
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Exponentially Decaying Choice Trace (Perseverance)
Instead of relying strictly on the immediate previous choice (stickiness), this model assumes that participants build up a "habit" or "perseverance" trace over multiple trials. The trace accumulates for chosen options and decays exponentially over time.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL with eligibility trace and an exponentially decaying choice trace (perseverance) for stage 1.
    
    Bounds:
    alpha: [0,1] - Learning rate.
    beta1: [0,10] - Inverse temperature for stage 1 choice.
    beta2: [0,10] - Inverse temperature for stage 2 choice.
    lam: [0,1] - Eligibility trace parameter.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    st: [-5,5] - Choice stickiness weight.
    tau: [0,1] - Decay rate for the choice trace.
    """
    alpha, beta1, beta2, lam, w, st, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta1 * q_net + st * choice_trace
            
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Action value updating
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update choice trace
        choice_trace *= tau
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
