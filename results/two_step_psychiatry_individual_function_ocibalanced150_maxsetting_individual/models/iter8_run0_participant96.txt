def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for positive and negative prediction errors,
    stage-specific perseveration, unchosen option decay, and an epsilon lapse rate.
    The lapse rate accounts for random exploration or lapses in attention, which 
    can help explain occasional breaks in long perseverative streaks.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    pers_1: [-5, 5] - stage 1 perseveration
    pers_2: [-5, 5] - stage 2 perseveration
    decay_rate: [0, 1] - rate at which unchosen stage 2 values decay toward 0.5
    epsilon: [0, 1] - random lapse rate
    """
    alpha_pos, alpha_neg, beta, w, pers_1, pers_2, decay_rate, epsilon = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -np.ones(2, dtype=int)

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q1_total = beta * q_stage1_net + pers_bonus_1
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        probs_1 = (1 - epsilon) * probs_1 + epsilon / 2.0
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2
            
        q2_total = beta * q_stage2_mf[state_idx] + pers_bonus_2
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        probs_2 = (1 - epsilon) * probs_2 + epsilon / 2.0
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] = (1 - decay_rate) * q_stage2_mf[state_idx, unchosen_action_2] + decay_rate * 0.5
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] = (1 - decay_rate) * q_stage2_mf[other_state, 0] + decay_rate * 0.5
        q_stage2_mf[other_state, 1] = (1 - decay_rate) * q_stage2_mf[other_state, 1] + decay_rate * 0.5

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Positive/negative learning rates, unchosen decay, and Stage 2 perseveration.
    For Stage 1, perseveration is split into win-stay (pers_1_win) and lose-stay (pers_1_lose)
    depending on the reward received in the previous trial. This helps capture behavior where
    a participant's tendency to repeat choices is highly modulated by recent feedback.

    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors

    pers_1_lose: [-5, 5] - stage 1 perseveration if previous trial was unrewarded
