def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and betas for stages 1 and 2, with perseveration at both stages.
    Bounds:
    alpha_1: [0, 1] (learning rate for stage 1)
    alpha_2: [0, 1] (learning rate for stage 2)
    beta_1: [0, 10] (inverse temperature for stage 1)
    beta_2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (model-based weight)
    lambda_: [0, 1] (eligibility trace)
    p_1: [-5, 5] (perseveration at stage 1)
    p_2: [-5, 5] (perseveration at stage 2)
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lambda_, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + p_1 * pers_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue
            
        pers_2 = np.zeros(2)
        if last_action_2 != -1:
            pers_2[last_action_2] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2 * pers_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive and negative RPEs, win-stay/lose-shift perseveration at stage 1.
    Bounds:
    alpha_pos: [0, 1] (learning rate for positive prediction errors)
    alpha_neg: [0, 1] (learning rate for negative prediction errors)
    beta: [0, 10] (inverse temperature)
    w: [0, 1] (model-based weight)
    lambda_: [0, 1] (eligibility trace)
    p_win: [-5, 5] (perseveration after a reward)
    p_lose: [-5, 5] (perseveration after no reward)
    """
    alpha_pos, alpha_neg, beta, w, lambda_, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                pers_1[last_action_1] = p_win
            else:
                pers_1[last_action_1] = p_lose
                
        exp_q1 = np.exp(beta * q_stage1_net + pers_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_s2 * lambda_ * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate decay rates for stage 1 and stage 2 unchosen options.
    Bounds:
    alpha: [0, 1] (learning rate)
    beta: [0, 10] (inverse temperature)
    w: [0, 1] (model-based weight)
    lambda_: [0, 1] (eligibility trace)
    p: [-5, 5] (perseveration at stage 1)
    decay_1: [0, 1] (decay rate for unchosen option at stage 1)
    decay_2: [0, 1] (decay rate for unchosen options at stage 2)
    """
    alpha, beta, w, lambda_, p, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + p * pers_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # apply decay to unchosen stage 1 action
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay_1)
        
        # apply decay to unchosen stage 2 actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_2)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
