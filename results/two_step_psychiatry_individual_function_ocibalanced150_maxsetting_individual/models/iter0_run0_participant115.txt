def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free with Choice Perseveration (Stickiness).
    
    This model integrates model-based (forward planning) and model-free (caching) 
    learning to evaluate first-stage choices. It also includes a perseveration 
    parameter to capture the participant's tendency to repeat the previous choice 
    regardless of the reward outcome.
    
    Parameters:
    alpha: learning rate for updating expected values [0, 1]
    beta1: inverse temperature for stage 1 (exploit vs explore) [0, 10]
    beta2: inverse temperature for stage 2 (exploit vs explore) [0, 10]
    w: model-based weight (1 = fully model-based, 0 = fully model-free) [0, 1]
    lam: eligibility trace (how much stage 2 reward updates stage 1 value) [0, 1]
    p: stickiness (perseveration) for stage 1 choice [-5, 5]
    """
    alpha, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Handle missing data trials
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits_1 = beta1 * q_stage1_net + p * rep
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Learning Rates for Positive/Negative PEs.
    
    This model assesses whether the participant learns differently from 
    better-than-expected outcomes compared to worse-than-expected outcomes, 
    a common pattern in reward learning tasks.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta1: inverse temperature for stage 1 [0, 10]
    beta2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += lr1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += lr2 * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[a1] += lr2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Value Forgetting (Decay of Unchosen Options).
    
    This model assumes that the values of options not chosen on a given trial 
    gradually decay toward the initial default expectation (0.5), which captures 
    memory constraints and explains exploratory shifts after long periods of exploitation.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta1: inverse temperature for stage 1 [0, 10]
    beta2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    decay: decay rate for unchosen options towards 0.5 [0, 1]
    """
    alpha, beta1, beta2, w, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values to 0.5 (neutral expectation)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Decay unchosen option at stage 1
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        # Decay unchosen options at stage 2
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] += decay * (0.5 - q_stage2_mf[state_idx, action_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
