Here are 3 new cognitive models that extend the classic Daw et al. (2011) mixed model-free and model-based learning architecture. Each introduces a unique mechanism to capture additional nuances in participant behavior, avoiding parameter combinations you have already tested.

### Model 1: Reward-Dependent Stickiness (Win-Stay / Lose-Shift)
Instead of a single perseveration parameter for Stage 1 choices, this model separates stickiness based on whether the previous trial was rewarded (`rep_win`) or unrewarded (`rep_lose`). This accounts for the well-known "win-stay, lose-shift" heuristic often observed in sequential decision-making tasks.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Mixed MF/MB with reward-dependent stickiness.
    Separates choice stickiness based on whether the previous trial was rewarded 
    (win-stay) or unrewarded (lose-shift).
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0, 1] - Eligibility trace (how much stage 2 reward updates stage 1 MF value).
    rep_win: [-5, 5] - Stickiness for stage 1 if the previous trial was rewarded.
    rep_lose: [-5, 5] - Stickiness for stage 1 if the previous trial was unrewarded.
    """
    alpha, beta_1, beta_2, w, lam, rep_win, rep_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness[last_action_1] = rep_win
            elif last_reward == 0:
                stickiness[last_action_1] = rep_lose
                
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unchosen Option Decay (Forgetting)
This model introduces a `decay` parameter. The Q-values of actions that were *not* chosen on the current trial slowly decay towards an uninformative prior (0.5), modeling the psychological process of forgetting the value of options that haven't been sampled recently.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Mixed MF/MB with decay for unchosen options.
    Q-values of actions not selected on the current trial decay towards 0.5, 
    modeling gradual forgetting of unsampled options.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    rep: [-5, 5] - Choice stickiness for stage 1.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5.
    """
    alpha, beta_1, beta_2, w, lam, rep, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = rep
            
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Apply decay to unchosen Q-values towards 0.5
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] += decay * (0.5 - q_stage1_mf[unchosen_action_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
                    
        # Apply learning updates to chosen Q-values
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Learned Transition Matrix
Rather than assuming participants perfectly know the fixed 0.7/0.3 transition probabilities from the start, this model assumes they start with a uniform prior (0.5/0.5) and update their model of the transition matrix over time using a transition learning rate (`alpha_tr`). 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Mixed MF/MB with a dynamically learned transition matrix.
    Instead of assuming fixed 0.7/0.3 transition probabilities, the model learns the 
    transition structure over time starting from a uniform 0.5/0.5 prior.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    rep: [-5, 5] - Choice stickiness for stage 1.
    alpha_tr: [0, 1] - Learning rate for updating the transition matrix.
    """
    alpha, beta_1, beta_2, w, lam, rep, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize learned transition matrix uniformly
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = rep
            
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update dynamically learned transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
