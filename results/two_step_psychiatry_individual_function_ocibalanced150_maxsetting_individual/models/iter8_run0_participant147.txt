Here are three new cognitive models that introduce novel mechanisms to explain participant behavior, building upon the best features of previous models while avoiding exact repetitions of previously tried parameter combinations. 

### Model 1: Transition Learning Model
Participants might not fully trust or internalize the fixed transition probabilities (0.7/0.3) provided in the instructions. This model assumes the agent dynamically learns the transition matrix from observed state transitions while engaging in hybrid model-based and model-free control.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with Transition Matrix Learning.
    Instead of assuming fixed transition probabilities (0.7/0.3), the agent learns them over time.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based versus model-free values (1 = MB, 0 = MF).
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    lam: [0, 1] - Eligibility trace parameter.
    alpha_trans: [0, 1] - Learning rate for the transition matrix.
    """
    learning_rate, beta, w, p, lam, alpha_trans = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix to uniform probabilities
    T = np.array([[0.5, 0.5], [0.5, 0.5]]) 
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = T @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p * pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Transition matrix update
        T[action_1[trial], state_idx] += alpha_trans * (1 - T[action_1[trial], state_idx])
        T[action_1[trial], 1 - state_idx] += alpha_trans * (0 - T[action_1[trial], 1 - state_idx])
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition Learning with Q-Value Decay
This model combines dynamic transition learning with a decay mechanism for unchosen actions. This captures both shifting beliefs about the task structure and the natural decay/forgetting of unchosen action values, driving exploration.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with Transition Matrix Learning and Q-value Decay.
    Combines learning the transition matrix with decaying unchosen Q-values towards 0.5.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based versus model-free values (1 = MB, 0 = MF).
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    lam: [0, 1] - Eligibility trace parameter.
    alpha_trans: [0, 1] - Learning rate for the transition matrix.
    """
    learning_rate, decay, beta, w, p, lam, alpha_trans = model_parameters
    n_trials = len(action_1)
    
    T = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = T @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p * pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Transition matrix update
        T[action_1[trial], state_idx] += alpha_trans * (1 - T[action_1[trial], state_idx])
        T[action_1[trial], 1 - state_idx] += alpha_trans * (0 - T[action_1[trial], 1 - state_idx])
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])

        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])

        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Loss Aversion (Omega) with Decay
Participants (especially in psychiatric contexts) may treat unrewarded trials not just as an absence of reward, but as a punishment. This model introduces `omega` to scale the aversiveness of unrewarded trials. Q-values decay to `0.0` to reflect a neutral baseline.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with Loss Aversion (Omega) and Decay.
    Unrewarded trials are treated as negative rewards (-omega) to capture stronger 
    avoidance of unrewarded options. Q-values decay towards 0.0.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.0.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free (1 = MB, 0 = MF).
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    lam: [0, 1] - Eligibility trace parameter.
    omega: [0, 5] - Effective punishment for unrewarded trials (0 reward becomes -omega).
    """
    learning_rate, decay, beta, w, p, lam, omega = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    # Initialize Q-values to 0 since rewards are mapped to 1 and -omega
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p * pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Map reward to capture loss aversion
        eff_reward = 1.0 if reward[trial] == 1 else -omega
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Decay for unchosen actions towards neutral 0.0
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.0 - q_stage1_mf[unchosen_1])

        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.0 - q_stage2_mf[state_idx, unchosen_2])

        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay * (0.0 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.0 - q_stage2_mf[unvisited_state, 1])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
