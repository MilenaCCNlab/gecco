Here are 3 new cognitive models based on the observed participant behavior and the requested constraints.

### Model 1: Unchosen Option Decay (Forgetting)
Participants may forget the value of options they do not choose, pulling their Q-values towards zero. This model introduces a `decay` parameter that shrinks the unchosen Q-values on every trial, which can explain bursts of exploration when a frequently chosen option's value drops relative to the forgotten alternative.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Baseline with unchosen option decay (forgetting).
    
    Parameters:
    alpha: [0,1] - Learning rate.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight.
    pers1: [-5,5] - Perseveration bonus for choice 1.
    bias1: [-5,5] - Explicit bias for choice 0 in stage 1.
    decay: [0,1] - Decay rate for unchosen Q-values towards 0.
    """
    alpha, beta_1, beta_2, w, pers1, bias1, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_net_1
        
        if last_choice_1 != -1:
            logits_1[last_choice_1] += pers1
            
        logits_1[0] += bias1
        
        logits_1 -= np.max(logits_1)
        probs_1 = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            
            # Update chosen options
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            # Decay unchosen options
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] *= (1 - decay)
            
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == action_2[trial]):
                        q_stage2_mf[s, a] *= (1 - decay)
            
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning Rates & Stage 2 Perseveration
This model evaluates if learning rates vary depending on the stage of the decision (e.g. learning transitions vs. learning final rewards). It also assesses whether the participant exhibits stickiness (perseveration) in the second stage choice as well as the first stage.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for stage 1 and stage 2, perseveration for both stages, and choice 0 bias.
    
    Parameters:
    alpha_1: [0,1] - Learning rate for stage 1.
    alpha_2: [0,1] - Learning rate for stage 2.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight.
    pers1: [-5,5] - Perseveration bonus for stage 1.
    bias1: [-5,5] - Explicit bias for choice 0 in stage 1.
    pers2: [-5,5] - Perseveration bonus for stage 2.
    """
    alpha_1, alpha_2, beta_1, beta_2, w, pers1, bias1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1
    last_choice_2 = np.full(2, -1) 

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_net_1
        
        if last_choice_1 != -1:
            logits_1[last_choice_1] += pers1
            
        logits_1[0] += bias1
        
        logits_1 -= np.max(logits_1)
        probs_1 = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            
            if last_choice_2[state_idx] != -1:
                logits_2[last_choice_2[state_idx]] += pers2
                
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            last_choice_2[state_idx] = action_2[trial]
            
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Perseveration (Win-Stay / Lose-Shift)
The participant's choice stickiness may depend strictly on the outcome of the previous trial. This model splits the standard perseveration parameter into two context-dependent bonuses: one for when the last trial was rewarding, and one for when it was unrewarding.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Outcome-dependent perseveration (win-stay/lose-shift style) in stage 1, along with bias.
    
    Parameters:
    alpha: [0,1] - Learning rate.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight.
    pers_rew: [-5,5] - Perseveration bonus for stage 1 if previous trial was rewarded (R > 0).
    pers_unrew: [-5,5] - Perseveration bonus for stage 1 if previous trial was unrewarded (R <= 0).
    bias1: [-5,5] - Explicit bias for choice 0 in stage 1.
    """
    alpha, beta_1, beta_2, w, pers_rew, pers_unrew, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_net_1
        
        if last_choice_1 != -1:
            if last_reward > 0:
                logits_1[last_choice_1] += pers_rew
            else:
                logits_1[last_choice_1] += pers_unrew
            
        logits_1[0] += bias1
        
        logits_1 -= np.max(logits_1)
        probs_1 = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            last_reward = reward[trial]
        else:
            last_reward = 0
            
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
