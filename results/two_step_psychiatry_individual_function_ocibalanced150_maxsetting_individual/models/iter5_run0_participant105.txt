def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF mixture with separate stage 1 and stage 2 perseveration, and value decay.
    This model adds a value decay mechanism to both model-free Q-values to capture forgetting,
    and includes separate perseveration parameters for stage 1 and stage 2 choices.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature for softmax choice [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace for updating stage 1 action values from stage 2 rewards [0,1]
    pers1: stage 1 perseveration bonus [-5,5]
    pers2: stage 2 perseveration bonus [-5,5]
    decay: value decay parameter for unchosen and chosen options [0,1]
    """
    alpha, beta, w, lam, pers1, pers2, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_a1 != -1:
            q_stage1_net[last_a1] += pers1
            
        q1_max = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - q1_max))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_state == state_idx and last_a2 != -1:
            q_stage2_net[last_a2] += pers2
            
        q2_max = np.max(q_stage2_net)
        exp_q2 = np.exp(beta * (q_stage2_net - q2_max))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Apply decay to all Q-values before updating
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_a1 = action_1[trial]
        last_a2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates and betas for stage 1 and 2, with perseveration and decay.
    This model decouples the learning rate and exploration/exploitation trade-off (beta) 
    between the first and second stage choices to capture potential differences in decision 
    processes at each stage.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace [0,1]
    pers: stage 1 perseveration bonus [-5,5]
    decay: value decay parameter [0,1]
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_a1 != -1:
            q_stage1_net[last_a1] += pers
            
        q1_max = np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * (q_stage1_net - q1_max))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q2_max = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - q2_max))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Apply decay
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF mixture with reward-dependent perseveration at stage 1.
    This model breaks down perseveration based on the outcome of the previous trial,
    allowing for differential stickiness if the last trial was rewarded (win-stay)
    versus unrewarded (lose-shift).
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace [0,1]
    pers: baseline stage 1 perseveration bonus [-5,5]
    stick_succ: additional perseveration if last trial was rewarded [-5,5]
    stick_unsucc: additional perseveration if last trial was unrewarded [-5,5]
    """
    alpha, beta, w, lam, pers, stick_succ, stick_unsucc = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_a1 != -1:
            bonus = pers
            if last_reward == 1:
                bonus += stick_succ
            else:
                bonus += stick_unsucc
            q_stage1_net[last_a1] += bonus
            
        q1_max = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - q1_max))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q2_max = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - q2_max))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_a1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
