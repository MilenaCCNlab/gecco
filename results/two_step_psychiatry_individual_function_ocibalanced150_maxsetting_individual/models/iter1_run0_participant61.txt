Here are three novel cognitive models for the two-step decision-making task. They build upon the standard Daw et al. (2011) model but introduce new cognitive mechanisms to better capture the participant's trial-by-trial decision-making processes.

### Model 1: Stage 1 & Stage 2 Perseveration
This model extends the baseline by adding a choice perseveration (stickiness) parameter not only for the first-stage choice but also for the second-stage choice. It captures the tendency of participants to repeat the same action at the second stage (e.g., asking the same alien) when visiting the same planet again.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-step model with both Stage 1 and Stage 2 choice perseveration.
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p_1: [-5,5] (stage 1 choice perseveration / stickiness)
    p_2: [-5,5] (stage 2 choice perseveration / stickiness)
    """
    import numpy as np
    alpha, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        perseveration_1 = np.zeros(2)
        if last_action_1 != -1:
            perseveration_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p_1 * perseveration_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        perseveration_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            perseveration_2[last_action_2[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2 * perseveration_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unchosen Q-Value Decay
This model introduces a memory decay mechanism. Instead of only updating the chosen options, the Q-values of unchosen options naturally decay towards an initial/neutral value of 0.5 over time. This accounts for forgetting when an option is not sampled for an extended period.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-step model with unchosen Q-value decay.
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p: [-5,5] (choice perseveration / stickiness)
    decay: [0,1] (decay rate of unchosen options towards initial value 0.5)
    """
    import numpy as np
    alpha, beta_1, beta_2, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to neutral 0.5
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        perseveration = np.zeros(2)
        if last_action != -1:
            perseveration[last_action] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p * perseveration)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay unchosen Q-values
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
                
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != action_2[trial]:
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-dependent Perseveration (Win-Stay / Lose-Shift)
Instead of a static choice stickiness parameter, this model breaks perseveration into two parts based on the immediate outcome of the last trial. It allows for independent modeling of "Win-Stay" (repeating a choice after a reward) and "Lose-Shift" (avoiding a choice after a null reward) tendencies at the first stage.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-step model with outcome-dependent choice perseveration (win-stay / lose-shift).
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p_win: [-5,5] (stage 1 perseveration after a reward)
    p_lose: [-5,5] (stage 1 perseveration after no reward)
    """
    import numpy as np
    alpha, beta_1, beta_2, w, lam, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        perseveration = np.zeros(2)
        p_weight = 0.0
        if last_action != -1:
            perseveration[last_action] = 1.0
            if last_reward == 1:
                p_weight = p_win
            elif last_reward == 0:
                p_weight = p_lose
            
        exp_q1 = np.exp(beta_1 * q_net + p_weight * perseveration)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
