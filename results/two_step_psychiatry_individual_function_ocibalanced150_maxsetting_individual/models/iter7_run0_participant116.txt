def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Differential Eligibility Trace (Common vs Rare) + Value Forgetting
    Eligibility trace depends on whether the transition was common or rare, allowing 
    asymmetric credit assignment for expected vs unexpected transitions. Unchosen 
    actions decay towards a baseline parameter q0.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight
    lam_common: [0,1] - Eligibility trace parameter for common transitions
    lam_rare: [0,1] - Eligibility trace parameter for rare transitions
    st: [-5,5] - Choice stickiness
    decay: [0,1] - Forgetting rate for unchosen actions (0 = no decay, 1 = complete forgetting)
    q0: [0,1] - Baseline Q-value that unchosen actions decay towards
    """
    learning_rate, beta, w, lam_common, lam_rare, st, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + q0
    q_stage2_mf = np.zeros((2, 2)) + q0
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Determine transition type for eligibility trace
        if action_1[trial] == state_idx:
            lam = lam_common
        else:
            lam = lam_rare
            
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] = q_stage1_mf[unchosen_action_1] * (1 - decay) + q0 * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + q0 * decay
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Exploration Bonus + Value Forgetting
    Adds an exploration (novelty) bonus based on the square root of the number of trials 
    since an action was last chosen. Integrates with the decay-to-baseline mechanism.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness
    expl_bonus: [0,1] - Weight of the exploration bonus for unchosen actions
    decay: [0,1] - Forgetting rate for unchosen actions
    q0: [0,1] - Baseline Q-value that unchosen actions decay towards
    """
    learning_rate, beta, w, lam, st, expl_bonus, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + q0
    q_stage2_mf = np.zeros((2, 2)) + q0
    
    trials_since_1 = np.zeros(2)
    trials_since_2 = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_total = q_net + expl_bonus * np.sqrt(trials_since_1)
        exp_q1 = np.exp(beta * q1_total + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_total = q_stage2_mf[state_idx] + expl_bonus * np.sqrt(trials_since_2[state_idx])
        exp_q2 = np.exp(beta * q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] = q_stage1_mf[unchosen_action_1] * (1 - decay) + q0 * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + q0 * decay
                    
        # Update trials since chosen
        trials_since_1 += 1
        trials_since_1[action_1[trial]] = 0
        
        trials_since_2 += 1
        trials_since_2[state_idx, action_2[trial]] = 0
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):

    Model 3: Separate Stage Betas + Value Forgetting
    Uses different inverse temperatures (beta1, beta2) for stage 1 and stage 2 choices, 
    accounting for differing degrees of stochasticity/exploration across the two stages.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    w: [0,1] - Model-based weight
    lam: [0,1] - Eligibility trace parameter
