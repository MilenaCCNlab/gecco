import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice trace for perseveration.
    Instead of only the immediate past choice, perseveration is based on an exponentially decaying trace of past choices.
    This allows the model to capture longer-term stickiness to specific actions.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace. Bounds: [0,1]
    pers (float): Perseveration weight for the choice trace. Bounds: [-5,5]
    tau (float): Decay rate for the choice trace. Bounds: [0,1]
    decay (float): Decay rate for Q-values towards zero. Bounds: [0,1]
    """
    alpha1, alpha2, beta, w, lam, pers, tau, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace1 = np.zeros(2)
    choice_trace2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf + pers * choice_trace1
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_net2 = q_stage2_mf[state_idx] + pers * choice_trace2[state_idx]
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        # Update choice traces
        choice_trace1 = choice_trace1 * tau
        choice_trace1[action_1[trial]] += (1.0 - tau)
        
        choice_trace2[state_idx] = choice_trace2[state_idx] * tau
        choice_trace2[state_idx, action_2[trial]] += (1.0 - tau)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Q-values decay towards 0.5.
    Instead of decaying Q-values towards zero, they decay towards 0.5, representing a regression to the mean expected value 
    (since rewards are 0 and 1). This assumes unchosen options regress to a neutral prior rather than zero.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers (float): Perseveration for both stages. Bounds: [-5,5]
    decay (float): Decay rate for Q-values towards 0.5. Bounds: [0,1]
    """
    alpha1, alpha2, beta, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay towards 0.5
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition matrix learning.
    Instead of assuming fixed transition probabilities (0.7/0.3), the participant learns the state transition probabilities 
    dynamically over time starting from a completely uncertain prior (0.5/0.5).
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace. Bounds: [0,1]
    pers (float): Perseveration for both stages. Bounds: [-5,5]
    decay (float): Decay rate for Q-values. Bounds: [0,1]
    alpha_trans (float): Learning rate for transition probabilities. Bounds: [0,1]
    """
    alpha1, alpha2, beta, w, lam, pers, decay, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transitions as completely uncertain
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        # Update transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        other_state = 1 - state_idx
        transition_matrix[action_1[trial], other_state] += alpha_trans * (0.0 - transition_matrix[action_1[trial], other_state])
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
