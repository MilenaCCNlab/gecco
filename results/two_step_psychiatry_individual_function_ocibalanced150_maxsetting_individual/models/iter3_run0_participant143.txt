def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates and separate perseveration for stage 1 and stage 2.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1.
    alpha_2: [0, 1] - Learning rate for stage 2.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p1: [0, 5] - Perseveration for stage 1 choice.
    p2: [0, 5] - Perseveration for stage 2 choice.
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1
            
        logits_1 = beta_1 * q_net_1 + p1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_a2 != -1:
            pers2[last_a2] = 1
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + p2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += alpha_1 * delta_stage1 + alpha_1 * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
        
        last_a1 = a1
        last_a2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for stage 1 and 2, decay for unchosen options, and perseveration.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1.
    alpha_2: [0, 1] - Learning rate for stage 2.
    decay: [0, 1] - Decay rate for unchosen actions (decays towards 0.5).
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p: [0, 5] - Perseveration for stage 1 choice.
    """
    alpha_1, alpha_2, decay, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_a1 = -1

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1
            
        logits_1 = beta_1 * q_net_1 + p * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += alpha_1 * delta_stage1 + alpha_1 * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
        
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates and outcome-dependent perseveration (Win-Stay / Lose-Shift).
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p_win: [0, 5] - Perseveration for stage 1 choice after a rewarded trial.
    p_lose: [-5, 5] - Perseveration for stage 1 choice after an unrewarded trial.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_a1 = -1
    last_r = -1

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_a1 != -1:
            pers[last_a1] = 1
            
        p_weight = 0
        if last_r == 1:
            p_weight = p_win
        elif last_r == 0:
            p_weight = p_lose
            
        logits_1 = beta_1 * q_net_1 + p_weight * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage1_mf[a1] += lr1 * delta_stage1 + lr2 * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += lr2 * delta_stage2
        
        last_a1 = a1
        last_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
