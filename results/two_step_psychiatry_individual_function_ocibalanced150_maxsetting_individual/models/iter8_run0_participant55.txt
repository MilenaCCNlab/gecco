import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with model-based/model-free weighting, eligibility trace, 
    choice stickiness, and a decay parameter for unchosen Q-values.
    
    Parameters:
    model_parameters (tuple):
        learning_rate: [0, 1] - Learning rate for Q-value updates.
        decay: [0, 1] - Decay rate for unchosen Q-values towards 0.
        w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
        lambda_: [0, 1] - Eligibility trace parameter.
        stickiness: [-5, 5] - Bonus added to the previously chosen stage 1 action.
        beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    """
    learning_rate, decay, w, lambda_, stickiness, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_net[last_action] += stickiness
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        last_action = a1

        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        # Decay unchosen actions
        q_stage1_mf[1 - a1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        # Update chosen actions
        q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lambda_ * delta_stage2
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with model-based/model-free weighting, choice stickiness based on a choice trace, 
    and choice trace learning rate.
    
    Parameters:
    model_parameters (tuple):
        learning_rate: [0, 1] - Learning rate for Q-value updates.
        w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
        stickiness: [-5, 5] - Bonus applied to stage 1 actions based on choice trace.
        beta: [0, 10] - Inverse temperature controlling choice stochasticity.
        alpha_c: [0, 1] - Learning rate for the choice trace.
    """
    learning_rate, w, stickiness, beta, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf + stickiness * choice_trace
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # Update choice trace
        choice_trace = (1 - alpha_c) * choice_trace
        choice_trace[a1] += alpha_c

        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
