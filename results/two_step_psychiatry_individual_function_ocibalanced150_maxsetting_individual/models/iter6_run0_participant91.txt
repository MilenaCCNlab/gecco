def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF learning with Q-value decay and eligibility traces.
    Adds an eligibility trace (lam) to the best model, allowing stage 2 prediction 
    errors to update stage 1 model-free Q-values. This captures the extent to which 
    rewards directly reinforce stage 1 actions.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, decay_rate, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values before update
        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF learning with asymmetric learning rates and Q-value decay.
    Allows for different learning rates for positive and negative reward prediction errors,
    which can capture confirmation bias or risk sensitivity (e.g., learning more from wins than losses).
    
    Parameters:
    learning_rate_pos: [0,1] - learning rate for positive prediction errors
    learning_rate_neg: [0,1] - learning rate for negative prediction errors
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    """
    learning_rate_pos, learning_rate_neg, beta, w, stickiness_1, stickiness_2, decay_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values before update
        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = learning_rate_pos if delta_stage1 > 0 else learning_rate_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = learning_rate_pos if delta_stage2 > 0 else learning_rate_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF learning with continuous choice traces and Q-value decay.
    Instead of just a binary indicator for the last action, this model maintains an 
    exponentially decaying trace of past choices to capture longer-term habitual perseveration.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - weight on the stage 1 choice trace
    stickiness_2: [-5,5] - weight on the stage 2 choice trace
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    trace_decay: [0,1] - rate at which choice traces decay (0 = no decay/infinite memory, 1 = only last action matters)
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, decay_rate, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy() + stickiness_1 * choice_trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx].copy() + stickiness_2 * choice_trace_2[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values before update
        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Update choice traces
        choice_trace_1 *= (1 - trace_decay)
        choice_trace_1[action_1[trial]] += 1
        
        choice_trace_2[state_idx] *= (1 - trace_decay)
        choice_trace_2[state_idx, action_2[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
