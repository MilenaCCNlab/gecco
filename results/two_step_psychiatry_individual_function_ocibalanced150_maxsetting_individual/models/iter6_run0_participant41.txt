import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) with continuous choice traces for both stages.
    Instead of a binary stickiness to the immediate last action, this model
    maintains an exponentially decaying choice trace for each action, which
    can capture longer-term perseveration or alternation tendencies.
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    alpha_c: [0, 1] - Choice trace update rate (higher means faster forgetting of past choices).
    phi_1: [-5, 5] - Stage 1 choice trace weight (positive = perseveration, negative = alternation).
    phi_2: [-5, 5] - Stage 2 choice trace weight.
    """
    learning_rate, beta, w, lam, alpha_c, phi_1, phi_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))
    
    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net + phi_1 * trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx, :] + phi_2 * trace_2[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Choice trace updating
        trace_1 *= (1.0 - alpha_c)
        trace_1[action_1[trial]] += alpha_c
        
        trace_2 *= (1.0 - alpha_c)
        trace_2[state_idx, action_2[trial]] += alpha_c
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) with asymmetric learning rates for positive and negative 
    prediction errors, Q-value decay, and stage-specific stickiness. This model captures 
    optimistic/pessimistic learning biases alongside gradual forgetting of unchosen options.
    
    Parameters/Bounds:
    lr_pos: [0, 1] - Learning rate for positive prediction errors.
    lr_neg: [0, 1] - Learning rate for negative prediction errors.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    stickiness_1: [-5, 5] - Tendency to repeat the previous first-stage choice.
    stickiness_2: [-5, 5] - Tendency to repeat the previous second-stage choice in the same state.
    """
    lr_pos, lr_neg, decay, beta, w, lam, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx, :]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += lr_2 * lam * delta_stage2
        
        # Decay for unchosen/unvisited options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1.0 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1.0 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, :] *= (1.0 - decay)
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) with stage-specific learning rates, stage-specific betas,
    and stage-specific stickiness. This model dissociates the flexibility and stochasticity
    of learning and choice behavior between the first and second stages of the task.
    
    Parameters/Bounds:
    lr_1: [0, 1] - Learning rate for stage 1 prediction errors.
    lr_2: [0, 1] - Learning rate for stage 2 prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.

    q_stage2_mf = np.zeros((2, 2))
