def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition Learning
    In addition to Q-learning with decay and a subjective penalty for zero reward, 
    this model updates its internal transition matrix based on observed transitions.
    This allows the model to adapt if it perceives changes in the transition probabilities.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values
    decay: [0, 1] - Retention rate for unchosen Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    penalty_zero: [0, 2] - Penalty applied when 0 coins are received
    alpha_trans: [0, 1] - Learning rate for the transition matrix
    """
    alpha, decay, beta_1, beta_2, w, lam, penalty_zero, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        subj_reward = 1.0 if reward[trial] == 1 else -penalty_zero
        delta_stage2 = subj_reward - q_stage2_mf[state_idx, action_2[trial]]

        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= decay
        
        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] *= decay
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= decay
        q_stage2_mf[other_state, 1] *= decay
        
        # transition matrix update based on observed state
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] -= alpha_trans * transition_matrix[action_1[trial], 1 - state_idx]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate Stage Learning Rates
    This model uses distinct learning rates for stage 1 and stage 2 updates,
    combined with decay for unchosen actions and a subjective penalty for 0 reward.
    This allows for differential flexibility in learning spaceship vs. alien values.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for stage 1 Q-values
    alpha_2: [0, 1] - Learning rate for stage 2 Q-values
    decay: [0, 1] - Retention rate for unchosen Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    penalty_zero: [0, 2] - Penalty applied when 0 coins are received
    """
    alpha_1, alpha_2, decay, beta_1, beta_2, w, lam, penalty_zero = model_parameters
    n_trials = len(action_1)

    p_choice_2 = np.zeros(n_trials)

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: State Perseveration (Planet Stickiness)
    Incorporates a bias to return to the planet visited on the previous trial,
    in addition to action perseveration and standard Q-learning mechanisms.
    This accounts for 'stimulus stickiness' independent of the action taken.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Retention rate for unchosen Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers: [-5, 5] - Action perseveration for stage 1
    state_pers: [-5, 5] - State perseveration (bias to return to previous state)
    """
    alpha, decay, beta_1, beta_2, w, lam, pers, state_pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_state = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        state_bias = np.zeros(2)
        if last_state != -1:
            state_bias = transition_matrix[:, last_state]
            
        logits_1 = beta_1 * q_stage1_net + pers * rep_1 + state_pers * state_bias
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= decay
        
        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] *= decay
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= decay
        q_stage2_mf[other_state, 1] *= decay
        
        last_action_1 = action_1[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
