def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate Stage 1 Learning Rate, Asymmetric Stage 2 Learning Rates, and Eligibility Trace.
    
    This model separates the learning rate of stage 1 from stage 2, and allows for asymmetric 
    learning (positive vs negative prediction errors) in stage 2. It incorporates an eligibility 
    trace (lam) to update stage 1 model-free values based on stage 2 prediction errors.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for stage 1
    alpha_2_pos: [0, 1] - Learning rate for stage 2 positive prediction errors
    alpha_2_neg: [0, 1] - Learning rate for stage 2 negative prediction errors
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    p: [0, 10] - Perseveration (stickiness) for stage 1 choice
    lam: [0, 1] - Eligibility trace parameter
    """
    alpha_1, alpha_2_pos, alpha_2_neg, beta_1, beta_2, w, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        alpha_2 = alpha_2_pos if delta_stage2 > 0 else alpha_2_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition Matrix Learning with Asymmetric Reward Learning.
    
    This model assumes participants do not rigidly hold a 0.7/0.3 representation of the 
    transition probabilities, but instead update them dynamically from experience. It also 
    features asymmetric learning rates for positive and negative reward prediction errors.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive reward prediction errors
    alpha_neg: [0, 1] - Learning rate for negative reward prediction errors
    alpha_trans: [0, 1] - Learning rate for the transition matrix
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    p: [0, 10] - Perseveration for stage 1 choice
    """
    alpha_pos, alpha_neg, alpha_trans, beta_1, beta_2, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]]) # starts at instructed probabilities
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # update transition matrix
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_trans * (1 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0 - transition_matrix[a1, 1 - state_idx])

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Learning Rates with Win-Stay / Lose-Shift Stickiness.
    
    This model assumes participants have different stickiness (perseveration) 
    depending on whether the previous trial was rewarded (Win-Stay) or 
    unrewarded (Lose-Shift). It combines this with asymmetric learning rates.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    p_win: [0, 10] - Perseveration for stage 1 choice if previously rewarded
    p_lose: [-10, 10] - Perseveration for stage 1 choice if previously unrewarded
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        current_p = p_win if last_reward > 0 else p_lose
            
        logits_1 = beta_1 * q_net + current_p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
