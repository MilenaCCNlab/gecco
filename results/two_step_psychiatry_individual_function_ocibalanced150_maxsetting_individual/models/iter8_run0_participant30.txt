import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-learning with forgetting (decay) and outcome-dependent perseveration at Stage 1.
    Unchosen and chosen Q-values decay towards 0.5 on each trial, capturing a tendency to forget 
    values over time.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    p_stage1_rew: [-5, 5] - Stage 1 perseveration bonus after a rewarded trial
    p_stage1_unrew: [-5, 5] - Stage 1 perseveration bonus after an unrewarded trial
    p_stage2: [-5, 5] - Stage 2 perseveration bonus
    lam: [0, 1] - Eligibility trace connecting Stage 2 prediction error to Stage 1 Q-value
    decay: [0, 1] - Decay rate of Q-values towards initial value of 0.5
    """
    learning_rate, beta, w, p_stage1_rew, p_stage1_unrew, p_stage2, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)

    last_a1 = -1
    last_a2 = -1
    last_s = -1
    last_r = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Decay Q-values towards 0.5 before making a choice
        q_stage1_mf += decay * (0.5 - q_stage1_mf)
        q_stage2_mf += decay * (0.5 - q_stage2_mf)

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        val_1 = beta * q_net
        if last_a1 != -1 and last_r != -1:
            if last_r == 1:
                val_1[last_a1] += p_stage1_rew
            else:
                val_1[last_a1] += p_stage1_unrew
        
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        val_2 = beta * q_stage2_mf[s]
        if last_s == s and last_a2 != -1:
            val_2[last_a2] += p_stage2
                
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        last_a1 = a1
        last_s = s
        last_a2 = a2
        last_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Q-learning with asymmetric eligibility traces for positive and negative RPEs.

    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values

    lam_pos: [0, 1] - Eligibility trace for positive Stage 2 RPEs

        current_lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[a1] += learning_rate * current_lam * delta_stage2

        last_a1 = a1
        last_s = s
        last_a2 = a2
        last_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Q-learning with transition-dependent and outcome-dependent perseveration at Stage 1.
    Differentiates perseveration after rewarded trials based on whether the previous transition was common or rare.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    p_rew_com: [-5, 5] - Stage 1 perseveration if last trial was rewarded and common transition
    p_rew_rare: [-5, 5] - Stage 1 perseveration if last trial was rewarded and rare transition
    p_unrew: [-5, 5] - Stage 1 perseveration if last trial was unrewarded
    p_stage2: [-5, 5] - Stage 2 perseveration bonus
    lam: [0, 1] - Eligibility trace connecting Stage 2 prediction error to Stage 1 Q-value
    """
    learning_rate, beta, w, p_rew_com, p_rew_rare, p_unrew, p_stage2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)

    last_a1 = -1
    last_a2 = -1
    last_s = -1
    last_r = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        val_1 = beta * q_net
        if last_a1 != -1 and last_r != -1:
            if last_r == 1:
                # Differentiate based on whether the transition was common (a1 == s) or rare (a1 != s)
                if last_a1 == last_s:
                    val_1[last_a1] += p_rew_com
                else:
                    val_1[last_a1] += p_rew_rare
            else:
                val_1[last_a1] += p_unrew
        
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        val_2 = beta * q_stage2_mf[s]
        if last_s == s and last_a2 != -1:
            val_2[last_a2] += p_stage2
                
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        last_a1 = a1
        last_s = s
        last_a2 = a2
        last_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
