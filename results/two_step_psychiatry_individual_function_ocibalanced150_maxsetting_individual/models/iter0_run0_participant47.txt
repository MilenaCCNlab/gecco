Here are 3 cognitive models that explain the participant's behavior in the two-step decision-making task. 

The models build upon the standard hybrid Model-Based (MB) / Model-Free (MF) Q-learning framework (Daw et al., 2011), which is widely used to capture behavior in this task.

### Model 1: Hybrid MB/MF Q-learning with Eligibility Trace
This model assumes the participant uses a mixture of model-based and model-free reinforcement learning. The model-based system computes values using the transition probabilities, while the model-free system learns through experience. An eligibility trace parameter (`lam`) allows the second-stage reward to directly update the first-stage action value.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with eligibility trace.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB, 0 = pure MF).
    lam: [0, 1] - Eligibility trace parameter connecting stage 2 reward to stage 1 value.
    """
    learning_rate, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Hybrid MB/MF value for stage 1
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid MB/MF Q-learning with Choice Stickiness
Participants often exhibit "perseveration" or choice stickiness, tending to repeat their previous first-stage choice regardless of the outcome. This model introduces a stickiness parameter (`pi`) to account for this autocorrelation in choices.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with choice stickiness.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi: [-5, 5] - Choice stickiness (perseveration) for stage 1.
    """
    learning_rate, beta, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Determine stickiness vector
        stick_vector = np.zeros(2)
        if last_action_1 != -1:
            stick_vector[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi * stick_vector)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # Update last action
        last_action_1 = action_1[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid MB/MF Q-learning with Asymmetric Learning Rates
Individuals may update their beliefs differently depending on whether outcomes are better or worse than expected. This model splits the learning rate into two components: one for positive prediction errors and one for negative prediction errors. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with asymmetric learning rates.
    
    Parameters:
    learning_rate_pos: [0, 1] - Learning rate for positive prediction errors (better than expected).
    learning_rate_neg: [0, 1] - Learning rate for negative prediction errors (worse than expected).
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    """
    learning_rate_pos, learning_rate_neg, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        lr1 = learning_rate_pos if delta_stage1 > 0 else learning_rate_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        lr2 = learning_rate_pos if delta_stage2 > 0 else learning_rate_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
