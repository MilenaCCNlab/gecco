import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate Learning Rates + Stage 1 & 2 Stickiness
    
    This model uses different learning rates for stage 1 and stage 2 updates,
    along with choice stickiness at both stages.
    
    Bounds:
    alpha1: [0,1] - Learning rate for stage 1
    alpha2: [0,1] - Learning rate for stage 2
    beta: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace
    stick1: [-5,5] - Choice stickiness for stage 1
    stick2: [-5,5] - Choice stickiness for stage 2
    """
    alpha1, alpha2, beta, beta2, w, lam, stick1, stick2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stick1
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q2_net[last_action_2[state_idx]] += stick2
            
        q2_net = q2_net - np.max(q2_net)
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Decay + Asymmetric Eligibility Trace + Stage 1 Stickiness
    
    This model incorporates a decay parameter for unchosen actions and uses 
    separate eligibility traces depending on whether the second-stage prediction
    error is positive or negative, allowing for asymmetric credit assignment.
    
    Bounds:
    learning_rate: [0,1] - Learning rate
    decay: [0,1] - Decay rate for unchosen actions
    beta: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weighting of MB vs MF control
    lam_pos: [0,1] - Eligibility trace for positive PE
    lam_neg: [0,1] - Eligibility trace for negative PE
    stick1: [-5,5] - Choice stickiness for stage 1
    """
    learning_rate, decay, beta, beta2, w, lam_pos, lam_neg, stick1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stick1
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = np.copy(q_stage2_mf[state_idx])
            
        q2_net = q2_net - np.max(q2_net)
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
