def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free learning with choice stickiness at both stages.
    This model assumes participants use a mixture of MB and MF strategies, 
    and tend to perseverate on their previous choices not only at the first stage 
    (spaceship choice) but also at the second stage (alien choice).
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen action.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness1: [-5,5] - Perseveration parameter for stage 1.
    stickiness2: [-5,5] - Perseveration parameter for stage 2.
    """
    learning_rate, beta, w, stickiness1, stickiness2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MF and MB with stickiness for stage 1
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_mb[action_1[trial-1]] += stickiness1
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if trial > 0 and state[trial-1] == state_idx:
            q_stage2_net[action_2[trial-1]] += stickiness2
            
        exp_q2 = np.exp(beta * q_stage2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free learning with counterfactual updating, 
    eligibility traces for stage 1 updating, and stage 1 choice stickiness.
    This model adds an eligibility trace (lambda) allowing the stage 2 reward 
    prediction error to directly update the stage 1 action value, combining 
    it with counterfactual updates for the unchosen option.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen action.
    counterfactual_rate: [0,1] - Learning rate for the unchosen action in the chosen state.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness: [-5,5] - Perseveration parameter for stage 1.
    lambda_trace: [0,1] - Eligibility trace for stage 2 reward updating stage 1.
    """
    learning_rate, counterfactual_rate, beta, w, stickiness, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MF and MB with stickiness for stage 1
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_mb[action_1[trial-1]] += stickiness
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += lambda_trace * learning_rate * delta_stage2
        
        # Counterfactual update for unchosen alien
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        cf_delta = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += counterfactual_rate * cf_delta

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free learning with asymmetric learning rates 
    for positive and negative RPEs, counterfactual updating, and choice stickiness.
    This model separates the learning process based on whether the outcome was 
    better or worse than expected, which is common in reinforcement learning models 
    of human behavior to capture optimism/pessimism bias.
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors.
    lr_neg: [0,1] - Learning rate for negative prediction errors.
    counterfactual_rate: [0,1] - Learning rate for the unchosen action in the chosen state.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness: [-5,5] - Perseveration parameter for stage 1.
    """
    lr_pos, lr_neg, counterfactual_rate, beta, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MF and MB with stickiness for stage 1
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_mb[action_1[trial-1]] += stickiness
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        # Counterfactual update for unchosen alien
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        cf_delta = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += counterfactual_rate * cf_delta

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
