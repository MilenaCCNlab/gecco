Here are three novel cognitive models that incorporate different psychological mechanisms to explain the participant's behavior. The participant data shows periods of strong preference for one option, switching behavior after unrewarded outcomes or rare transitions, and potential fluctuations in learning. 

### Model 1: Reward-Dependent Perseveration (Win-Stay / Lose-Stay)
Instead of a single "stickiness" or perseveration parameter, this model differentiates between perseveration following a reward versus an unrewarded trial. This captures whether the participant is more likely to repeat their first-stage choice when they win (win-stay) compared to when they lose (lose-stay/shift).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Reward-dependent perseveration.
    Instead of a single perseveration parameter, this model has separate 
    perseveration parameters for when the previous trial was rewarded vs unrewarded.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta1: [0,10] - Inverse temperature for first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weighting of model-based vs. model-free values (1 = fully model-based).
    p_rew: [-5,5] - Perseveration parameter if the previous trial was rewarded.
    p_unrew: [-5,5] - Perseveration parameter if the previous trial was unrewarded.
    lambd: [0,1] - Eligibility trace parameter.
    """
    learning_rate, beta1, beta2, w, p_rew, p_unrew, lambd = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                pers[last_action_1] = p_rew
            else:
                pers[last_action_1] = p_unrew
                
        logits_1 = beta1 * q_stage1_net + pers
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambd * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Q-Value Decay (Forgetting)
This model posits that the participant slowly forgets the values of unchosen options. At the end of each trial, all Q-values decay towards 0.5 (the expected value of a random chance). This can explain why the participant might suddenly switch spaceships after a long streak of choosing the other one, as the unchosen option's value resets to a neutral baseline.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Q-value decay (forgetting).
    Incorporates a decay parameter that pulls all Q-values towards 0.5 at each trial,
    modeling forgetting of unchosen options over time.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta1: [0,10] - Inverse temperature for first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weighting of model-based vs. model-free values.
    p: [-5,5] - Perseveration parameter for first-stage choice.
    lambd: [0,1] - Eligibility trace parameter.
    decay: [0,1] - Rate at which Q-values decay towards 0.5.
    """
    learning_rate, beta1, beta2, w, p, lambd, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 as they will decay towards this neutral baseline
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
                
        logits_1 = beta1 * q_stage1_net + pers
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambd * delta_stage2
        
        # apply decay to all Q-values
        q_stage1_mf = (1 - decay) * q_stage1_mf + decay * 0.5
        q_stage2_mf = (1 - decay) * q_stage2_mf + decay * 0.5
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition Matrix Learning
Instead of assuming the participant perfectly knows and fixes the state transition probabilities at 0.7/0.3, this model assumes they actively learn the transition structure dynamically based on the transitions they experience. This can explain localized changes in behavior if they experience a cluster of rare transitions.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition learning.
    Instead of assuming fixed transition probabilities (0.7/0.3), this model learns
    the transition matrix dynamically based on experienced transitions.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta1: [0,10] - Inverse temperature for first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weighting of model-based vs. model-free values.
    p: [-5,5] - Perseveration parameter for first-stage choice.
    lambd: [0,1] - Eligibility trace parameter.
    lr_trans: [0,1] - Learning rate for updating the transition matrix.
    """
    learning_rate, beta1, beta2, w, p, lambd, lr_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to uniform (0.5) prior to learning
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
                
        logits_1 = beta1 * q_stage1_net + pers
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambd * delta_stage2
        
        # update transition matrix based on observed transitions
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += lr_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += lr_trans * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
