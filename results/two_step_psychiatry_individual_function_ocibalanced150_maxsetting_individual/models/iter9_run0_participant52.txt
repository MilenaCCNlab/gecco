import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with transition matrix learning and outcome-dependent stickiness.
    Instead of assuming fixed transition probabilities, the agent dynamically updates 
    its internal transition matrix based on observed state transitions.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-values
    alpha_tr: [0,1] - learning rate for transition probabilities
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - model-based weight
    lam: [0,1] - eligibility trace parameter
    st_win: [-5,5] - stickiness for stage 1 if previous trial was rewarded
    st_lose: [-5,5] - stickiness for stage 1 if previous trial was unrewarded
    """
    alpha, alpha_tr, beta1, beta2, w, lam, st_win, st_lose = model_parameters
    n_trials = len(action_1)
  
    # Initialize with instructed probabilities but allow them to update
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                stickiness_bonus[last_action_1] = st_win
            else:
                stickiness_bonus[last_action_1] = st_lose
            
        val_1 = q_net + stickiness_bonus
        exp_q1 = np.exp(beta1 * (val_1 - np.max(val_1)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        val_2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (val_2 - np.max(val_2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update transition matrix based on observed transition
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1 - transition_matrix[action_1[trial], state_idx]

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with transition-dependent eligibility trace (lam_common vs lam_rare).
    The degree to which stage 2 rewards update stage 1 actions depends on whether 
    the transition to stage 2 was expected (common) or unexpected (rare).
    
    Bounds:
    alpha: [0,1] - learning rate
    decay: [0,1] - decay rate for unchosen options
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - model-based weight
    lam_common: [0,1] - eligibility trace for common transitions
    lam_rare: [0,1] - eligibility trace for rare transitions
    st: [-5,5] - stickiness for stage 1
    """
    alpha, decay, beta1, beta2, w, lam_common, lam_rare, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = st
            
        val_1 = q_net + stickiness_bonus
        exp_q1 = np.exp(beta1 * (val_1 - np.max(val_1)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        val_2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (val_2 - np.max(val_2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Determine eligibility trace based on transition type
        if action_1[trial] == state_idx:
            lam = lam_common
        else:
            lam = lam_rare

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen options
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] += decay * (0.5 - q_stage1_mf[unchosen_action_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model where model-based weight (w) depends on whether the previous 
    transition was common or rare, capturing how surprise from a rare transition 
    might alter the balance between MB and MF systems on the subsequent trial.
    
    Bounds:
    alpha: [0,1] - learning rate
    decay: [0,1] - decay rate for unchosen options
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w_common: [0,1] - model-based weight if previous transition was common
    w_rare: [0,1] - model-based weight if previous transition was rare
    lam: [0,1] - eligibility trace parameter
    st: [-5,5] - stickiness for stage 1
    """
    alpha, decay, beta1, beta2, w_common, w_rare, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_transition_common = True # Default for the first trial

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Use w_common or w_rare based on the previous transition
        w = w_common if last_transition_common else w_rare
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = st

        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        val_2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (val_2 - np.max(val_2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen options
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] += decay * (0.5 - q_stage1_mf[unchosen_action_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_transition_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
