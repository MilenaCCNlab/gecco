def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual learning rates and dual temperatures with eligibility trace and stickiness.
    This model allows for different learning speeds and choice stochasticity between the first 
    and second stages, which is useful for capturing differences in how participants explore 
    or exploit at each step of the task.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1.
    alpha_2: [0, 1] - Learning rate for stage 2.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness: [-5, 5] - Tendency to repeat the previous stage-1 choice.
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lambda_, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + stickiness * stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # action value updating for choice 2 and trace update for choice 1
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Eligibility trace, stickiness, and separate decay rates for stage 1 and stage 2.
    Participants may forget or discount the value of unchosen options at different rates 
    depending on the stage (e.g., higher decay for the aliens compared to the spaceships).
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting between model-based and model-free control.
    lambda_: [0, 1] - Eligibility trace.
    stickiness: [-5, 5] - Stage-1 choice perseveration.
    decay_1: [0, 1] - Decay rate for unchosen actions in stage 1.
    decay_2: [0, 1] - Decay rate for unchosen actions in stage 2.
    """
    alpha, beta, w, lambda_, stickiness, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness * stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay_1)
        
        # action value updating for choice 2 and trace update for choice 1
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay_2)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay_2)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay_2)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates for positive/negative prediction errors, dual temperatures, 
    eligibility trace, and stickiness. This model captures the tendency to update beliefs differently 
    following better-than-expected versus worse-than-expected outcomes, while also letting stage 1 
    and stage 2 temperatures differ.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting between model-based and model-free control.
    lambda_: [0, 1] - Eligibility trace.
    stickiness: [-5, 5] - Stage-1 choice perseveration.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + stickiness * stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        # action value updating for choice 2 and trace update for choice 1
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
