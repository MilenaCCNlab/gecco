def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with Outcome-Dependent Eligibility Traces.
    
    This model extends the baseline by allowing the eligibility trace (lam) to differ
    depending on whether the trial was rewarded or unrewarded, capturing potential
    asymmetries in how credit is assigned to Stage 1 choices after wins vs. losses.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight for model-based vs model-free influence [0, 1]
    pi: Perseveration bonus for repeating the previous stage 1 choice [-5, 5]
    lam_pos: Eligibility trace for rewarded trials [0, 1]
    lam_neg: Eligibility trace for unrewarded trials [0, 1]
    omega: Penalty for unrewarded trials (0 reward becomes -omega) [0, 5]
    """
    alpha, beta1, beta2, w, pi, lam_pos, lam_neg, omega = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        eff_reward = reward[trial] if reward[trial] > 0 else -omega
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        current_lam = lam_pos if reward[trial] > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * current_lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with Perseveration Trace.
    
    This model replaces the single-trial choice perseveration with a decaying 
    perseveration trace. This captures longer-term tendencies (stickiness) to repeat 
    choices that were frequently made in the recent past, rather than just the immediate last trial.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight for model-based vs model-free influence [0, 1]
    pi: Perseveration weight for stage 1 choice trace [-5, 5]
    lam: Eligibility trace parameter [0, 1]
    omega: Penalty for unrewarded trials (0 reward becomes -omega) [0, 5]
    pers_decay: Decay rate for the perseveration trace [0, 1]
    """
    alpha, beta1, beta2, w, pi, lam, omega, pers_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    pers_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta1 * q_stage1_net + pi * pers_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        eff_reward = reward[trial] if reward[trial] > 0 else -omega
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update perseveration trace
        pers_trace *= pers_decay
        pers_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with Outcome-Dependent Stage 1 Temperature.
    
    This model tests the hypothesis that participants shift between exploratory and 
    exploitative behavior depending on the outcome of the previous trial (e.g., exploring 
    more after a loss). It employs two different inverse temperatures for Stage 1.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1_win: Inverse temperature for stage 1 following a rewarded trial [0, 10]
    beta1_loss: Inverse temperature for stage 1 following an unrewarded trial [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight for model-based vs model-free influence [0, 1]
    pi: Perseveration bonus for repeating the previous stage 1 choice [-5, 5]
    lam: Eligibility trace parameter [0, 1]
    omega: Penalty for unrewarded trials (0 reward becomes -omega) [0, 5]
    """
    alpha, beta1_win, beta1_loss, beta2, w, pi, lam, omega = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_reward > 0:
            current_beta1 = beta1_win
        elif last_reward == 0:
            current_beta1 = beta1_loss
        else:
            current_beta1 = (beta1_win + beta1_loss) / 2.0

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        logits_1 = current_beta1 * q_stage1_net + pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        eff_reward = reward[trial] if reward[trial] > 0 else -omega
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
