def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive/negative RPEs, separate stickiness, and unchosen decay.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive RPEs.
    alpha_neg: [0, 1] - Learning rate for negative RPEs.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for Stage 1 MF updates from Stage 2 rewards.
    rho1: [-5, 5] - Choice stickiness for Stage 1.
    rho2: [-5, 5] - Choice stickiness for Stage 2 (applied if the same state is visited).
    decay: [0, 1] - Decay rate for unchosen Q-values.
    """
    alpha_pos, alpha_neg, beta, w, lam, rho1, rho2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if trial > 0:
            stickiness_1[action_1[trial - 1]] = 1
            
        logits_1 = beta * q_net + rho1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        stickiness_2 = np.zeros(2)
        if trial > 0 and state[trial - 1] == state_idx:
            stickiness_2[action_2[trial - 1]] = 1
            
        logits_2 = beta * q_stage2_mf[state_idx] + rho2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2

        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, :] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
