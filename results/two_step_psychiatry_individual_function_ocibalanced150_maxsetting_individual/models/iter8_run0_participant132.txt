def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates (lr_pos, lr_neg) for positive and negative RPEs,
    value decay, separate MB and MF inverse temperatures, stage 1 perseveration, and eligibility trace.
    
    Bounds:
    lr_pos: [0,1]
    lr_neg: [0,1]
    decay: [0,1]
    beta_mb: [0,10]
    beta_mf: [0,10]
    beta2: [0,10]
    pi1: [-5,5]
    lam: [0,1]
    """
    lr_pos, lr_neg, decay, beta_mb, beta_mf, beta2, pi1, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            # policy for the second choice
            logits_2 = beta2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # action value updating for choice 1
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            lr1 = lr_pos if delta_stage1 > 0 else lr_neg
            q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

            # action value updating for choice 2
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            lr2 = lr_pos if delta_stage2 > 0 else lr_neg
            q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

            # eligibility trace update
            q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

            # value decay for unchosen actions
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == action_2[trial]):
                        q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with a dynamic choice trace for stage 1 (alpha_c, weight_c),
    value decay, separate MB and MF inverse temperatures, and eligibility trace.
    
    Bounds:
    learning_rate: [0,1]
    decay: [0,1]
    beta_mb: [0,10]
    beta_mf: [0,10]
    beta2: [0,10]
    alpha_c: [0,1]
    weight_c: [-5,5]
    lam: [0,1]
    """
    learning_rate, decay, beta_mb, beta_mf, beta2, alpha_c, weight_c, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + weight_c * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            # policy for the second choice
            logits_2 = beta2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # action value updating for choice 1
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

            # action value updating for choice 2
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            # eligibility trace update
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

            # value decay for unchosen actions
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == action_2[trial]):
                        q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        else:
            p_choice_2[trial] = 1.0
            
        # Update dynamic choice trace
        choice_trace = (1 - alpha_c) * choice_trace
        choice_trace[action_1[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
