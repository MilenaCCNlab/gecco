def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates, inverse temperatures, and stickiness for both stages with decay.
    This model captures the participant's repetitive behavior by applying stickiness at both stages of choice,
    while also allowing stage-specific softmax temperatures to account for different levels of noise in 
    decision-making at the spaceship vs. alien level.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stick_1: [0, 10] - choice stickiness parameter for stage 1
    stick_2: [0, 10] - choice stickiness parameter for stage 2
    decay: [0, 1] - decay rate for unchosen options
    """
    alpha, beta_1, beta_2, w, stick_1, stick_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vec_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + stick_1 * stick_vec_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick_vec_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vec_2[last_action_2] = 1.0
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + stick_2 * stick_vec_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay) # Decay unchosen
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay) # Decay unchosen
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * delta_stage2 # Implicit lam=1
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF Q-learning with choice stickiness, decay, eligibility trace, and win-stay behavior.
    This model introduces a reward-dependent stickiness (win-stay) parameter to model 
    increased probability of repeating a stage 1 choice specifically after a successful (rewarded) trial.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    stickiness: [0, 10] - general choice stickiness for stage 1
    decay: [0, 1] - decay rate for unchosen options
    lam: [0, 1] - eligibility trace parameter
    win_stay: [0, 10] - additional stickiness on stage 1 if previous trial was rewarded
    """
    alpha, beta, w, stickiness, decay, lam, win_stay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec = np.zeros(2)
        if last_action != -1:
            stick_vec[last_action] = stickiness
            if last_reward > 0:
                stick_vec[last_action] += win_stay
            
        logits_1 = beta * q_net + stick_vec
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates (positive/negative RPEs), separate inverse temperatures 
    and stickiness for each stage, and eligibility trace.
    This model evaluates whether the participant learns differently from better-than-expected 
    vs. worse-than-expected outcomes, while strictly modeling the dual-stage choice dynamics.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive RPE
    alpha_neg: [0, 1] - learning rate for negative RPE
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stick_1: [0, 10] - choice stickiness parameter for stage 1
    stick_2: [0, 10] - choice stickiness parameter for stage 2
    lam: [0, 1] - eligibility trace parameter
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, stick_1, stick_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vec_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vec_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + stick_1 * stick_vec_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick_vec_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vec_2[last_action_2] = 1.0
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + stick_2 * stick_vec_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
