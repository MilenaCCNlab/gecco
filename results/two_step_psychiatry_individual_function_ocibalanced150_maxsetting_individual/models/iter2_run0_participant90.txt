def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based/Model-Free learning with counterfactual updating 
    for the unchosen action in the chosen state, and choice perseveration (stickiness).
    This model assumes participants use a mixture of MB and MF strategies, 
    tend to repeat their previous first-stage choice, and update their beliefs about 
    the unchosen alien on the visited planet assuming anti-correlated rewards.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen action.
    counterfactual_rate: [0,1] - Learning rate for the unchosen action in the chosen state.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness: [-5,5] - Perseveration parameter for stage 1.
    """
    learning_rate, counterfactual_rate, beta, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MB and MF values, and add stickiness
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_mb[action_1[trial-1]] += stickiness
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Counterfactual update for the unchosen action in the chosen state
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        cf_delta = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += counterfactual_rate * cf_delta

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model-Based/Model-Free learning with an eligibility trace 
    and decay for unchosen options. This models a participant who updates their 
    stage 1 action directly from the stage 2 reward (TD(lambda)) and gradually 
    forgets the values of unvisited states and actions.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    decay_rate: [0,1] - Decay rate for unchosen Q-values.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    lambda_trace: [0,1] - Eligibility trace parameter for stage 1 update from stage 2 reward.
    """
    learning_rate, decay_rate, beta, w, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MB and MF values
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay_rate)
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Apply eligibility trace to Stage 1 MF value
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_trace * delta_stage2
        
        # Decay unchosen stage 2 actions
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay_rate)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay_rate)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay_rate)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model-Based/Model-Free learning with asymmetric learning rates.
    This model assumes participants learn differently from positive versus negative 
    prediction errors, while still balancing model-based and model-free control.
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors.
    lr_neg: [0,1] - Learning rate for negative prediction errors.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    """
    lr_pos, lr_neg, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MB and MF values
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1 (Asymmetric)
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        if delta_stage1 > 0:
            q_stage1_mf[action_1[trial]] += lr_pos * delta_stage1
        else:
            q_stage1_mf[action_1[trial]] += lr_neg * delta_stage1
            
        # Action value updating for Choice 2 (Asymmetric)
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += lr_pos * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += lr_neg * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
