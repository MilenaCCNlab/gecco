def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual-system model with forgetting (decay) and outcome-dependent stickiness.
    Stickiness at Stage 1 depends on whether the previous trial was rewarded (st_win) or unrewarded (st_loss).
    This captures the participant's tendency to switch spaceships primarily after receiving 0 coins.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st_win: [-5, 5] (choice stickiness after a reward)
    st_loss: [-5, 5] (choice stickiness after no reward)
    decay: [0, 1] (decay rate for unchosen Q-values)
    """
    alpha, beta_1, beta_2, w, lam, st_win, st_loss, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness[last_action_1] = st_win
            else:
                stickiness[last_action_1] = st_loss
            
        logits_1 = beta_1 * q_net + stickiness
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[1 - a1] = (1 - decay) * q_stage1_mf[1 - a1] + decay * 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
                    
        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dual-system model with separate forgetting (decay) rates for Stage 1 and Stage 2.
    Unchosen actions decay towards a neutral value of 0.5 over time, with different rates for each stage.
    This allows the model to capture different memory spans for spaceship values vs alien values.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    decay_1: [0, 1] (decay rate for unchosen Stage 1 Q-values)
    decay_2: [0, 1] (decay rate for unchosen Stage 2 Q-values)
    """
    alpha, beta_1, beta_2, w, lam, st, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + st * stickiness
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[1 - a1] = (1 - decay_1) * q_stage1_mf[1 - a1] + decay_1 * 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = (1 - decay_2) * q_stage2_mf[s, a] + decay_2 * 0.5
                    
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dual-system model with forgetting (decay) and transition matrix learning.
    Instead of assuming a fixed transition matrix, the agent learns the transition probabilities
    between stage 1 actions and stage 2 states using a transition learning rate (alpha_tr).
    
    Bounds:
    alpha: [0, 1] (Q-value learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    decay: [0, 1] (decay rate for unchosen Q-values)
    alpha_tr: [0, 1] (learning rate for the transition matrix)
    """
    alpha, beta_1, beta_2, w, lam, st, decay, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix as uniform
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + st * stickiness
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Update learned transition matrix based on observed state
        transition_matrix[a1, state_idx] += alpha_tr * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[a1, 1 - state_idx])

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[1 - a1] = (1 - decay) * q_stage1_mf[1 - a1] + decay * 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
                    
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
