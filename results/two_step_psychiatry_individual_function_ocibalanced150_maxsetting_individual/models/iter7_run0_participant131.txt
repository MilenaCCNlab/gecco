Here are 3 new cognitive models that introduce different mechanisms to explain the participant's behavior, such as differences between the two stages, dynamic learning of the task structure, and long-term choice autocorrelation.

### Model 1: Separate Learning Rates and Inverse Temperatures
This model extends the best model by allowing the participant to learn and explore at different rates for the spaceship stage (Stage 1) and the alien stage (Stage 2). Since spaceship transitions are mostly fixed but alien reward probabilities drift over time, participants often use different learning rates and exploration strategies (betas) for the two stages.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Hybrid MB/MF with separate learning rates and separate betas for Stage 1 and Stage 2.
    
    Parameters:
    lr_1: [0, 1] - Learning rate for updating Stage 1 Q-values.
    lr_2: [0, 1] - Learning rate for updating Stage 2 Q-values.
    w: [0, 1] - Weight given to model-based vs model-free values (1 = pure MB, 0 = pure MF).
    beta_1: [0, 10] - Inverse temperature for Stage 1 softmax action selection.
    beta_2: [0, 10] - Inverse temperature for Stage 2 softmax action selection.
    stickiness_1: [-5, 5] - Log-odds bonus for repeating the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Log-odds bonus for repeating the previous Stage 2 choice in the same state.
    """
    lr_1, lr_2, w, beta_1, beta_2, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 policy
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Dynamic Transition Learning
Instead of assuming the participant fully trusts the 70%/30% transition probabilities between spaceships and planets, this model assumes they actively learn and update their internal model of these transitions based on their experience.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Hybrid MB/MF with Transition Probability Learning.
    
    This model assumes participants update their internal model of the transition 
    probabilities between spaceships and planets dynamically, which alters their 
    model-based Q-values over time.
    
    Parameters:
    learning_rate_q: [0, 1] - Learning rate for updating Q-values.
    learning_rate_t: [0, 1] - Learning rate for updating transition probabilities.
    w: [0, 1] - Weight given to model-based vs model-free values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    stickiness_1: [-5, 5] - Log-odds bonus for repeating the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Log-odds bonus for repeating the previous Stage 2 choice.
    """
    learning_rate_q, learning_rate_t, w, beta, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 policy
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Q-value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate_q * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_q * delta_stage2
        
        # Transition matrix update
        transition_matrix[action_1[trial], state_idx] += learning_rate_t * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += learning_rate_t * (0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Autocorrelated Choice Traces
The participant data shows extremely long streaks of repeating the same spaceship (e.g., choosing spaceship 1 for 50+ trials in a row). Standard stickiness only provides a bonus for the *immediate* last action. This model uses an autocorrelated choice trace that builds up over time, capturing deeper perseveration habits.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Hybrid MB/MF with Autocorrelated Choice Traces.
    
    Instead of just repeating the immediate last action, this model maintains a 
    choice trace for each action that decays over time, capturing longer-term 
    perseveration (habitual streaks) which is prevalent in the participant's data.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    w: [0, 1] - Weight given to model-based vs model-free values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    alpha_c: [0, 1] - Update/decay rate for the choice traces.
    stickiness_1: [-5, 5] - Weight of the choice trace on Stage 1 logits.
    stickiness_2: [-5, 5] - Weight of the choice trace on Stage 2 logits.
    """
    learning_rate, w, beta, alpha_c, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net + stickiness_1 * choice_trace_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 policy
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_2 * choice_trace_2[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Update choice traces
        choice_trace_1 *= (1 - alpha_c)
        choice_trace_1[action_1[trial]] += alpha_c
        
        choice_trace_2[state_idx] *= (1 - alpha_c)
        choice_trace_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
