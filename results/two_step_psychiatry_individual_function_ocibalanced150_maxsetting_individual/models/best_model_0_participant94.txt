def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Shared beta, separate choice stickiness for stage 1 and stage 2, with unchosen option decay towards q0.
    This model adds a perseveration parameter for the second-stage choices (pi_2), acknowledging that 
    participants might exhibit stickiness in their alien choices as well as their spaceship choices.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - shared inverse temperature for both stages
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi_1: [-5, 5] - stage 1 choice stickiness
    pi_2: [-5, 5] - stage 2 choice stickiness
    decay: [0, 1] - decay rate for unchosen options
    q0: [0, 1] - baseline value to decay towards
    """
    alpha, beta, w, lambda_, pi_1, pi_2, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, float(q0))
    q_stage2_mf = np.full((2, 2), float(q0))
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if last_a1 != -1:
            rep1[last_a1] = 1.0
            
        q1_scaled = beta * q_stage1_net + pi_1 * rep1
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        rep2 = np.zeros(2)
        if last_a2 != -1:
            rep2[last_a2] = 1.0

        q2_scaled = beta * q_stage2_mf[state_idx] + pi_2 * rep2
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (q0 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (q0 - q_stage2_mf[s, a])
        
        last_a1 = a1
        last_a2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
