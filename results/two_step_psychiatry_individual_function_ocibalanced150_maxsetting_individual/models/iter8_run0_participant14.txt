def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and temperatures for Stage 1 and Stage 2, with eligibility trace and separate stickiness for both stages.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1.
    alpha_2: [0, 1] - Learning rate for stage 2.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness_1: [-5, 5] - Tendency to repeat the previous stage-1 choice.
    stickiness_2: [-5, 5] - Tendency to repeat the previous stage-2 choice.
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lambda_, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var_1 = np.zeros(2)
        if trial > 0:
            stick_var_1[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + stickiness_1 * stick_var_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick_var_2 = np.zeros(2)
        if trial > 0 and state[trial] == state[trial - 1]:
            stick_var_2[action_2[trial - 1]] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + stickiness_2 * stick_var_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Eligibility trace, unchosen value decay, and separate stage 1 perseveration 
    based on previous reward (win-stay, lose-stay), along with stage 2 perseveration.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    decay: [0, 1] - Rate at which unchosen action values decay toward zero.
    win_stay: [-5, 5] - Tendency to repeat stage-1 choice if previously rewarded.
    lose_stay: [-5, 5] - Tendency to repeat stage-1 choice if previously unrewarded.
    stickiness_2: [-5, 5] - Tendency to repeat the previous stage-2 choice.
    """
    alpha, beta, w, lambda_, decay, win_stay, lose_stay, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        ws_var = np.zeros(2)
        ls_var = np.zeros(2)
        if trial > 0:
            if reward[trial - 1] > 0:
                ws_var[action_1[trial - 1]] = 1.0
            else:
                ls_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + win_stay * ws_var + lose_stay * ls_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick_var_2 = np.zeros(2)
        if trial > 0 and state[trial] == state[trial - 1]:
            stick_var_2[action_2[trial - 1]] = 1.0
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + stickiness_2 * stick_var_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Incorporates action stickiness and state stickiness. State stickiness increases the probability 
    of choosing the stage 1 action that commonly leads to the stage 2 state visited on the previous trial.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness: [-5, 5] - Tendency to repeat the previous stage-1 choice.
    decay: [0, 1] - Rate at which unchosen action values decay toward zero.
    state_stickiness: [-5, 5] - Tendency to choose the stage-1 action that commonly leads to the previous stage-2 state.
    """
    alpha, beta, w, lambda_, stickiness, decay, state_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        state_stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            # State 0 is commonly reached by Action 0. State 1 is commonly reached by Action 1.
            state_stick_var[state[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness * stick_var + state_stickiness * state_stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
