def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model-based and model-free reinforcement learning model with 
    reward-dependent learning rates. It uses alpha_win for rewarded trials 
    and alpha_lose for unrewarded trials. Unchosen Q-values decay towards 
    a free parameter 'center'.
    
    Bounds:
    alpha_win: [0, 1] - Learning rate when reward is 1.
    alpha_lose: [0, 1] - Learning rate when reward is 0.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for model-based vs model-free control.
    lam: [0, 1] - Eligibility trace parameter determining how much stage 2 
                  prediction errors update stage 1 values.
    decay: [0, 1] - Rate at which unchosen Q-values decay towards the center.
    center: [0, 1] - Baseline value towards which Q-values decay.
    """
    alpha_win, alpha_lose, beta, w, lam, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Determine learning rate based on reward outcome
        current_alpha = alpha_win if reward[trial] == 1 else alpha_lose

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += current_alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += current_alpha * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += current_alpha * lam * delta_stage2

        # Decay unchosen action values towards center
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * center
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * center

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):

    asymmetric eligibility traces for positive and negative stage 2 prediction 
    errors (lam_pos and lam_neg). This captures differential credit assignment

    Bounds:

    w: [0, 1] - Weighting parameter for model-based vs model-free control.

    lam_neg: [0, 1] - Eligibility trace for negative stage 2 prediction errors.

    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model-based and model-free reinforcement learning model featuring a 
    continuous choice trace (autocorrelation) for stage 1 actions. This allows 
    for extended perseveration over multiple trials, governed by a decay rate 
    (tau) and weight (pers), combined with Q-value decay to a center parameter.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for model-based vs model-free control.
    lam: [0, 1] - Eligibility trace parameter.
    decay: [0, 1] - Rate at which unchosen Q-values decay towards the center.
    center: [0, 1] - Baseline value towards which Q-values decay.
    tau: [0, 1] - Decay rate of the choice trace (higher means faster decay).
    pers: [0, 10] - Weight of the choice trace on stage 1 action selection.
    """
    alpha, beta, w, lam, decay, center, tau, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)
    
    choice_trace = np.zeros(2, dtype=float)

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add choice trace to logits and standardise to prevent overflow
        logits_1 = beta * q_stage1_net + pers * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update continuous choice trace
        choice_trace = (1 - tau) * choice_trace
        choice_trace[action_1[trial]] += 1.0

        # Decay unchosen action values towards center
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * center
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * center

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
