import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF learning with separate inverse temperatures for stage 1 and stage 2.
    Builds on the best model by allowing the stochasticity of choices to differ between the first
    and second stages, reflecting different levels of decision noise or exploration.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta_1: [0,10] - inverse temperature for stage 1 softmax choice
    beta_2: [0,10] - inverse temperature for stage 2 softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    """
    learning_rate, beta_1, beta_2, w, stickiness_1, stickiness_2, decay_rate, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF learning with action bias.
    Adds an action_bias parameter to the best model to capture any baseline preference 
    for choosing spaceship 0 over spaceship 1, independent of learned values.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    action_bias: [-5,5] - bias added to the logit of action 0 at stage 1
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, decay_rate, lam, action_bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])

    p_choice_2 = np.zeros(n_trials)

    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF learning with asymmetric learning rates for positive and negative RPEs.
    Builds on the best model by allowing the participant to update Q-values differently when 
    outcomes are better vs worse than expected.
    
    Parameters:
    lr_pos: [0,1] - learning rate for positive prediction errors
    lr_neg: [0,1] - learning rate for negative prediction errors
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    """
    lr_pos, lr_neg, beta, w, stickiness_1, stickiness_2, decay_rate, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
