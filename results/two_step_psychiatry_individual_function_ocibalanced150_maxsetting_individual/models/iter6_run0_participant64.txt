Here are 3 new cognitive models that explore alternative mechanisms and parameter combinations to explain the participant's decision-making behavior. 

### Model 1: Win-Stay-Lose-Shift (WSLS) Mechanism
This model introduces a `wsls` parameter that explicitly captures the tendency to repeat a Stage 1 choice if it ultimately led to a reward on the previous trial, and to switch if it did not. This represents a more reactive, reward-driven form of stickiness compared to pure choice perseveration (`p`).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF hybrid with choice stickiness (perseveration) and 
    a Win-Stay-Lose-Shift (WSLS) parameter.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness parameter [-5,5]
    wsls: win-stay-lose-shift weight [-5,5]
    """
    alpha, beta, w, lam, p, wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        wsls_rep = np.zeros(2)
        
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            if last_reward != -1:
                # Add +1 for a win (Stay), -1 for a loss (Shift)
                wsls_rep[last_action_1] = 1.0 if last_reward > 0 else -1.0
            
        logits_1 = beta * q_net + p * rep + wsls * wsls_rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        act1 = action_1[trial]
        p_choice_1[trial] = probs_1[act1]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        r = reward[trial]

        # policy for the second choice
        if act2 != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            # ACTION VALUE UPDATING FOR CHOICE 1
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            q_stage1_mf[act1] += alpha * delta_stage1

            # ACTION VALUE UPDATING FOR CHOICE 2
            delta_stage2 = r - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            # ELIGIBILITY TRACE UPDATE
            q_stage1_mf[act1] += alpha * lam * delta_stage2

            last_reward = r
        else:
            p_choice_2[trial] = 1.0
            last_reward = -1
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Decaying Choice Trace
Instead of perseveration only depending on the immediately preceding trial, this model keeps a continuous `choice_trace` that decays over time. This accounts for participants who might show a longer-lasting bias towards actions they have chosen frequently in the recent past.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF hybrid with an exponentially decaying choice trace 
    for perseveration at stage 1, rather than just the last action.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness weight [-5,5]
    trace_decay: decay rate of the choice trace [0,1]
    """
    alpha, beta, w, lam, p, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_net + p * choice_trace
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        act1 = action_1[trial]
        p_choice_1[trial] = probs_1[act1]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        r = reward[trial]

        # policy for the second choice
        if act2 != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            # ACTION VALUE UPDATING FOR CHOICE 1
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            q_stage1_mf[act1] += alpha * delta_stage1

            # ACTION VALUE UPDATING FOR CHOICE 2
            delta_stage2 = r - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            # ELIGIBILITY TRACE UPDATE
            q_stage1_mf[act1] += alpha * lam * delta_stage2

        else:
            p_choice_2[trial] = 1.0
            
        # Update choice trace
        choice_trace *= trace_decay
        if act1 != -1:
            choice_trace[act1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Unchosen Forgetting + WSLS
Building on the strengths of previous models, this iteration integrates the forgetting (decay) of unchosen Q-values with the Win-Stay-Lose-Shift (WSLS) mechanism. It captures both the passive degradation of information over time and the immediate reflexive response to rewards.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF hybrid with choice stickiness, forgetting (decay) for 
    unchosen action values, and a Win-Stay-Lose-Shift (WSLS) mechanism.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness parameter [-5,5]
    decay: forgetting rate for unchosen actions [0,1]
    wsls: win-stay-lose-shift weight [-5,5]
    """
    alpha, beta, w, lam, p, decay, wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        wsls_rep = np.zeros(2)
        
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            if last_reward != -1:
                wsls_rep[last_action_1] = 1.0 if last_reward > 0 else -1.0
            
        logits_1 = beta * q_net + p * rep + wsls * wsls_rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        act1 = action_1[trial]
        p_choice_1[trial] = probs_1[act1]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        r = reward[trial]

        # policy for the second choice
        if act2 != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            # ACTION VALUE UPDATING FOR CHOICE 1
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            q_stage1_mf[act1] += alpha * delta_stage1

            # ACTION VALUE UPDATING FOR CHOICE 2
            delta_stage2 = r - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            # ELIGIBILITY TRACE UPDATE
            q_stage1_mf[act1] += alpha * lam * delta_stage2

            # UNCHOSEN DECAY
            unchosen_act1 = 1 - act1
            q_stage1_mf[unchosen_act1] *= (1 - decay)

            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == act2):
                        q_stage2_mf[s, a] *= (1 - decay)
                        
            last_reward = r
        else:
            p_choice_2[trial] = 1.0
            last_reward = -1
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
