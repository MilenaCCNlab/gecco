Here are 3 new cognitive models based on the observed participant behavior. The participant shows a strong preference for spaceship 0, which they repeatedly choose even after occasionally exploring spaceship 1. This suggests that in addition to perseveration (a tendency to repeat the last choice), there might be an explicit bias toward choice 0. Furthermore, the participant might exhibit different exploration/exploitation balances between the first and second stages of the task, which can be captured by separating the inverse temperature (`beta`) into `beta_1` and `beta_2`.

### Model 1: Separate Betas and Explicit Bias
This model introduces separate inverse temperatures for stage 1 and stage 2, allowing for different levels of choice stochasticity in each stage. It also includes an explicit bias parameter for choice 0, which captures the participant's strong baseline preference for spaceship 0 independent of learning and perseveration.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate betas for each stage, MB/MF mixture, choice 1 perseveration, and explicit bias for choice 0.
    
    Parameters:
    alpha: [0,1] - Learning rate.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight.
    pers1: [-5,5] - Perseveration bonus for choice 1.
    bias1: [-5,5] - Explicit bias for choice 0 in stage 1.
    """
    alpha, beta_1, beta_2, w, pers1, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_net_1
        
        if last_choice_1 != -1:
            logits_1[last_choice_1] += pers1
            
        logits_1[0] += bias1  # Explicit bias for choice 0
        
        logits_1 -= np.max(logits_1)
        probs_1 = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # Stage 2 policy
        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
  
        # Value updating
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Eligibility Trace and Separate Betas
This model incorporates an eligibility trace (`lam`) to update the stage 1 model-free values directly based on the stage 2 reward prediction error. This helps explain rapid adaptation. It retains the separate betas for each stage to account for different choice consistencies.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Eligibility trace (lambda), separate betas for each stage, MB/MF mixture, and stage 1 perseveration.
    
    Parameters:
    alpha: [0,1] - Learning rate.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight.
    pers1: [-5,5] - Perseveration bonus for choice 1.
    lam: [0,1] - Eligibility trace parameter.
    """
    alpha, beta_1, beta_2, w, pers1, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_net_1
        
        if last_choice_1 != -1:
            logits_1[last_choice_1] += pers1
            
        logits_1 -= np.max(logits_1)
        probs_1 = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # Stage 2 policy
        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
  
        # Value updating
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            # Eligibility trace update for stage 1
            q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
            
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates, Separate Betas, and Explicit Bias
This model combines the best-performing mechanism from previous iterations (asymmetric learning rates for positive vs. negative prediction errors) with separate choice stochasticity (betas) for both stages and an explicit baseline bias for spaceship 0. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates, separate betas, MB/MF mixture, perseveration, and explicit bias for choice 0.
    
    Parameters:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight.
    pers1: [-5,5] - Perseveration bonus for choice 1.
    bias1: [-5,5] - Explicit bias for choice 0 in stage 1.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, pers1, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_net_1
        
        if last_choice_1 != -1:
            logits_1[last_choice_1] += pers1
            
        logits_1[0] += bias1  # Explicit bias for choice 0
        
        logits_1 -= np.max(logits_1)
        probs_1 = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # Stage 2 policy
        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
  
        # Value updating
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
            q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
            
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
