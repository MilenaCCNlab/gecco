def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with separate temperatures for Stage 1 and Stage 2, choice stickiness, and unchosen Q-value decay.
    Builds upon the best model by allowing decision noise to vary between the two stages, which often differs due to the complexity of the stages.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta1: [0, 10] - Inverse temperature for Stage 1 softmax action selection.
    beta2: [0, 10] - Inverse temperature for Stage 2 softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for Stage 1 MF updates from Stage 2 rewards.
    rho: [-5, 5] - Choice stickiness (positive for perseveration, negative for alternation).
    decay: [0, 1] - Decay rate for unchosen Q-values (0 = no decay, 1 = complete forgetting).
    """
    alpha, beta1, beta2, w, lam, rho, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if trial > 0:
            stickiness[action_1[trial - 1]] = 1
            
        logits_1 = beta1 * q_net + rho * stickiness
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen actions
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, :] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with asymmetric learning rates (positive vs negative RPE), choice stickiness, and unchosen Q-value decay.
    Builds upon the best model by allowing participants to learn differently from positive versus negative feedback.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive reward prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative reward prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for Stage 1 MF updates from Stage 2 rewards.
    rho: [-5, 5] - Choice stickiness (positive for perseveration, negative for alternation).
    decay: [0, 1] - Decay rate for unchosen Q-values (0 = no decay, 1 = complete forgetting).
    """
    alpha_pos, alpha_neg, beta, w, lam, rho, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if trial > 0:
            stickiness[action_1[trial - 1]] = 1
            
        logits_1 = beta * q_net + rho * stickiness
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2

        # decay for unchosen actions
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, :] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with outcome-dependent choice stickiness (Win-Stay / Lose-Shift) and unchosen Q-value decay.
    Builds upon the best model by allowing the tendency to repeat a choice to depend on whether the previous trial was rewarded.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for Stage 1 MF updates from Stage 2 rewards.
    rho_win: [-5, 5] - Choice stickiness applied if the previous trial was rewarded.
    rho_lose: [-5, 5] - Choice stickiness applied if the previous trial was unrewarded.
    decay: [0, 1] - Decay rate for unchosen Q-values (0 = no decay, 1 = complete forgetting).
    """
    alpha, beta, w, lam, rho_win, rho_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if trial > 0:
            if reward[trial - 1] > 0:

            else:
                stickiness[action_1[trial - 1]] = rho_lose
            
        logits_1 = beta * q_net + stickiness
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)

        state_idx = state[trial]

        # policy for the second choice

        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen actions
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, :] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
