import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate temperatures for Stage 1 and Stage 2, and choice stickiness at both Stage 1 and Stage 2.
    
    Bounds:
    alpha: [0,1] - learning rate
    beta_1: [0,10] - inverse temperature for stage 1 choices
    beta_2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight
    lambda_: [0,1] - eligibility trace
    pers_1: [-5,5] - perseveration (stickiness) bonus for previous stage 1 choice
    pers_2: [-5,5] - perseveration (stickiness) bonus for previous stage 2 choice (same state)
    """
    alpha, beta_1, beta_2, w, lambda_, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1
    prev_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if prev_action_1 != -1:
            pers_bonus_1[prev_action_1] = pers_1
            
        logits_1 = beta_1 * q_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        pers_bonus_2 = np.zeros(2)
        if prev_action_2[state_idx] != -1:
            pers_bonus_2[prev_action_2[state_idx]] = pers_2

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        prev_action_1 = action_1[trial]
        prev_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate temperatures, stickiness, and separate learning rates for positive and negative RPEs.
    
    Bounds:
    alpha_pos: [0,1] - learning rate for positive RPE
    alpha_neg: [0,1] - learning rate for negative RPE
    beta_1: [0,10] - inverse temperature for stage 1 choices
    beta_2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight
    lambda_: [0,1] - eligibility trace
    pers: [-5,5] - perseveration (stickiness) bonus for previous stage 1 choice
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1

    for trial in range(n_trials):

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if prev_action_1 != -1:
            pers_bonus[prev_action_1] = pers
            
        logits_1 = beta_1 * q_net + pers_bonus
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Updates with separate learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2
        
        prev_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate temperatures, stickiness, and a decay (forgetting) parameter for unchosen options.
    
    Bounds:
    alpha: [0,1] - learning rate
    beta_1: [0,10] - inverse temperature for stage 1 choices
    beta_2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight
    lambda_: [0,1] - eligibility trace
    pers: [-5,5] - perseveration (stickiness) bonus for previous stage 1 choice
    decay: [0,1] - decay rate for unchosen Q-values towards 0.5
    """
    alpha, beta_1, beta_2, w, lambda_, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 for decay model
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    prev_action_1 = -1

    for trial in range(n_trials):

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if prev_action_1 != -1:
            pers_bonus[prev_action_1] = pers
            
        logits_1 = beta_1 * q_net + pers_bonus
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update chosen options
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Decay unchosen options towards 0.5
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * 0.5
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
        
        prev_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
