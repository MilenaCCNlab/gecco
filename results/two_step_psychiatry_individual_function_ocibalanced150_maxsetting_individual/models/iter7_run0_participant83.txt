def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Reward-dependent model-based weight.
    This model posits that the degree of model-based control (w) depends on whether 
    the previous trial was rewarded or not, capturing potential shifts between 
    goal-directed and habitual strategies after success vs. failure.
    
    Parameters:
    alpha1: learning rate for stage 1 [0, 1]
    alpha2: learning rate for stage 2 [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w_rew: model-based weight following a rewarded trial [0, 1]
    w_unrew: model-based weight following an unrewarded trial [0, 1]
    lambda_: eligibility trace [0, 1]
    pi1: choice perseveration for stage 1 [-5, 5]
    """
    alpha1, alpha2, beta1, beta2, w_rew, w_unrew, lambda_, pi1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1
    prev_reward = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        if prev_reward == 1:
            w = w_rew
        elif prev_reward == 0:
            w = w_unrew
        else:
            w = 0.5 
            
        net_q1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * net_q1
        if prev_action_1 != -1:
            logits_1[prev_action_1] += pi1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2
        
        prev_action_1 = action_1[trial]
        prev_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric eligibility traces.
    This model assumes that positive and negative reward prediction errors at stage 2 
    propagate back to stage 1 action values with different strengths (lambda_pos vs lambda_neg).
    
    Parameters:
    alpha1: learning rate for stage 1 [0, 1]
    alpha2: learning rate for stage 2 [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_pos: eligibility trace for positive PE [0, 1]
    lambda_neg: eligibility trace for negative PE [0, 1]
    pi1: choice perseveration for stage 1 [-5, 5]
    """
    alpha1, alpha2, beta1, beta2, w, lambda_pos, lambda_neg, pi1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * net_q1
        if prev_action_1 != -1:
            logits_1[prev_action_1] += pi1
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        lambda_curr = lambda_pos if delta_stage2 > 0 else lambda_neg
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_curr * delta_stage2
        
        prev_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Shared asymmetric learning rates across stages with dual perseveration.
    Applies separate learning rates for positive and negative prediction errors globally,
    while capturing stickiness at stage 1 and state-dependent stickiness at stage 2.
    
    Parameters:
    alpha_pos: learning rate for positive PE [0, 1]
    alpha_neg: learning rate for negative PE [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    pi1: choice perseveration for stage 1 [-5, 5]
    rep_st2: choice perseveration for stage 2 if returning to the same state [-5, 5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lambda_, pi1, rep_st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1
    prev_state = -1
    prev_action_2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * net_q1
        if prev_action_1 != -1:
            logits_1[prev_action_1] += pi1
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx]
        if prev_state == state_idx and prev_action_2 != -1:
            logits_2[prev_action_2] += rep_st2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1_curr = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1_curr * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2_curr = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_curr * delta_stage2
        
        alpha_lambda = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_lambda * lambda_ * delta_stage2
        
        prev_action_1 = action_1[trial]
        prev_state = state_idx
        prev_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
