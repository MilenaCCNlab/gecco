### Model 1: Asymmetric Learning Rates for Positive and Negative RPEs
This model explores the idea that participants might update their value expectations differently depending on whether the outcome was better or worse than expected. It introduces `alpha_pos` for positive reward prediction errors (RPEs) and `alpha_neg` for negative RPEs.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Asymmetric Learning Rates, Eligibility Trace, and Choice Perseveration.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive reward prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative reward prediction errors.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter (positive values indicate a tendency to repeat the last choice).
    """
    alpha_pos, alpha_neg, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
            
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Inverse Temperatures for Stage 1 and Stage 2
Often, decision-making stochasticity differs between the two stages of the task (e.g., participants might be more exploratory in stage 1 and more deterministic in stage 2). This model introduces separate inverse temperatures (`beta_1` and `beta_2`) for each stage.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Separate Inverse Temperatures, Eligibility Trace, and Choice Perseveration.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax choice.
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter (positive values indicate a tendency to repeat the last choice).
    """
    alpha, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
            
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta_1 * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta_2 * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Choice Perseveration for Both Stages
In previous models, perseveration (or stickiness) was modeled exclusively for the first-stage choice. This model assumes participants might also have a tendency to stick to the same alien (second-stage choice) when they revisit the same planet. It introduces separate parameters `p_1` and `p_2` for stage 1 and stage 2 perseveration.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Eligibility Trace and Choice Perseveration for both stages.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p_1: [-5, 5] - Perseveration parameter for stage 1 choice.
    p_2: [-5, 5] - Perseveration parameter for stage 2 choice (state specific).
    """
    alpha, beta, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1
    last_choice_2 = np.array([-1, -1]) # tracking last choice for each planet/state

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_1 = np.zeros(2)
        if last_choice_1 != -1:
            pers_1[last_choice_1] = 1
            
        q_stage1_combined = q_stage1_net + p_1 * pers_1
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_2 = np.zeros(2)
        if last_choice_2[state_idx] != -1:
            pers_2[last_choice_2[state_idx]] = 1

        q_stage2_combined = q_stage2_mf[state_idx] + p_2 * pers_2
        q_stage2_shifted = q_stage2_combined - np.max(q_stage2_combined)
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # update last choices
        last_choice_1 = action_1[trial]
        last_choice_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
