Here are 3 new cognitive models proposing alternative mechanisms to explain the participant's decision-making behavior.

### Model 1: Reward-dependent Stickiness (Win-Stay / Lose-Shift)
This model proposes that the participant's tendency to repeat the previous first-stage action depends heavily on whether the last trial was rewarded. Instead of a single general stickiness parameter, it uses separate stickiness bonuses for rewarded (`stick_rew`) and unrewarded (`stick_unrew`) outcomes, effectively capturing the Win-Stay/Lose-Shift heuristic.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Reward-dependent Stickiness (Win-Stay / Lose-Shift)
    
    This model extends the standard mixture model by having separate stickiness 
    parameters depending on whether the previous trial was rewarded or not.
    This captures the win-stay/lose-shift heuristic directly in the choice policy.

    Bounds:
    learning_rate: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature for stage 1 choices
    beta2: [0,10] - Inverse temperature for stage 2 choices
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace parameter
    stick_rew: [-5,5] - Choice stickiness for stage 1 if previous trial was rewarded
    stick_unrew: [-5,5] - Choice stickiness for stage 1 if previous trial was unrewarded
    """
    learning_rate, beta, beta2, w, lam, stick_rew, stick_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action != -1:
            if last_reward == 1:
                q_stage1_net[last_action] += stick_rew
            elif last_reward == 0:
                q_stage1_net[last_action] += stick_unrew
                
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Loss Penalty Model
This model assumes that unrewarded outcomes (0 coins) aren't just neutral but are perceived as actively aversive (akin to loss aversion). A loss penalty parameter maps unrewarded trials to a negative value during the update step, which could drive stronger avoidance learning after a 0-coin trial.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Loss Penalty Model
    
    This model assumes that unrewarded outcomes (0 coins) are perceived as aversive 
    losses rather than just neutral outcomes. A loss penalty parameter maps 0 rewards 
    to a negative value, driving stronger avoidance learning.

    Bounds:
    learning_rate: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature for stage 1 choices
    beta2: [0,10] - Inverse temperature for stage 2 choices
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace parameter
    stick: [-5,5] - Choice stickiness parameter for stage 1
    loss_penalty: [0,5] - Penalty applied to unrewarded trials (0 reward becomes -loss_penalty)
    """
    learning_rate, beta, beta2, w, lam, stick, loss_penalty = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += stick
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Apply loss penalty to reward
        effective_reward = 1.0 if reward[trial] == 1 else -loss_penalty

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = effective_reward - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Eligibility Trace Model
This model proposes that credit assignment to the first-stage choice depends on whether the outcome at the second stage was better or worse than expected. It uses separate eligibility traces (`lam_pos` and `lam_neg`) for positive and negative prediction errors. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Eligibility Trace Model
    
    This model allows the eligibility trace (lambda) to differ depending on whether 
    the stage 2 prediction error is positive or negative. This captures asymmetric 
    credit assignment for better-than-expected versus worse-than-expected outcomes.

    Bounds:
    learning_rate: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature for stage 1 choices
    beta2: [0,10] - Inverse temperature for stage 2 choices
    w: [0,1] - Weighting of MB vs MF control
    lam_pos: [0,1] - Eligibility trace for positive prediction errors
    lam_neg: [0,1] - Eligibility trace for negative prediction errors
    stick: [-5,5] - Choice stickiness parameter for stage 1
    """
    learning_rate, beta, beta2, w, lam_pos, lam_neg, stick = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += stick
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        current_lam = lam_pos if delta_stage2 > 0 else lam_neg

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * current_lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
