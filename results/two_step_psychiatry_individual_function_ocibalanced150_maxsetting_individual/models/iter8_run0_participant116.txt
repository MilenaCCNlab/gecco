def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates + Exploration Bonus + Value Forgetting
    Uses separate learning rates for positive and negative prediction errors. 
    Unchosen actions decay towards a fixed baseline of 0.5.
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors
    lr_neg: [0,1] - Learning rate for negative prediction errors
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness
    expl_bonus: [0,1] - Weight of the exploration bonus for unchosen actions
    decay: [0,1] - Forgetting rate for unchosen actions
    """
    lr_pos, lr_neg, beta, w, lam, st, expl_bonus, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q0 = 0.5
    q_stage1_mf = np.zeros(2) + q0
    q_stage2_mf = np.zeros((2, 2)) + q0
    
    trials_since_1 = np.zeros(2)
    trials_since_2 = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_total = q_net + expl_bonus * np.sqrt(trials_since_1)
        exp_q1 = np.exp(beta * q1_total + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_total = q_stage2_mf[state_idx] + expl_bonus * np.sqrt(trials_since_2[state_idx])
        exp_q2 = np.exp(beta * q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

        # decay unchosen values
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] = q_stage1_mf[unchosen_action_1] * (1 - decay) + q0 * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + q0 * decay

        # update trials since chosen
        trials_since_1 += 1
        trials_since_1[action_1[trial]] = 0
        
        trials_since_2 += 1
        trials_since_2[state_idx, action_2[trial]] = 0
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
