Here are 3 new cognitive models that build upon the best model so far by introducing new combinations of mechanisms such as stage-specific inverse temperatures, asymmetric learning rates for positive and negative prediction errors, and separate model-based/model-free temperature parameters. All models include the successful decay mechanism and stage-specific stickiness, while keeping the total parameter count at 8.

### Model 1
This model introduces **stage-specific inverse temperatures** (`beta_1` and `beta_2`). It allows for the possibility that the participant's choice randomness or exploration varies between the first-stage (spaceship) and second-stage (alien) decisions.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Incorporates stage-specific inverse temperatures (beta_1, beta_2), 
    stage-specific stickiness, and a decay parameter for unchosen options.
    
    Bounds:
    alpha: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stickiness_1: [-5, 5]
    stickiness_2: [-5, 5]
    decay: [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, stickiness_1, stickiness_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_stage2 = {0: None, 1: None}

    for trial in range(n_trials):

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if trial > 0:
            stickiness_bonus_1[action_1[trial-1]] = stickiness_1
            
        logits_1 = beta_1 * q_net + stickiness_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 Policy
        stickiness_bonus_2 = np.zeros(2)
        if last_action_stage2[state_idx] is not None:
            stickiness_bonus_2[last_action_stage2[state_idx]] = stickiness_2

        logits_2 = beta_2 * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility Trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_stage2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2
This model incorporates **asymmetric learning rates** (`alpha_pos` and `alpha_neg`) for positive and negative prediction errors. This captures the psychological phenomenon where individuals might learn differently from rewards (positive reinforcement) versus lack of rewards (disappointment).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Incorporates asymmetric learning rates for positive and negative prediction errors,
    along with stage-specific stickiness and unchosen option decay.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stickiness_1: [-5, 5]
    stickiness_2: [-5, 5]
    decay: [0, 1]
    """
    alpha_pos, alpha_neg, beta, w, lam, stickiness_1, stickiness_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_stage2 = {0: None, 1: None}

    for trial in range(n_trials):

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if trial > 0:
            stickiness_bonus_1[action_1[trial-1]] = stickiness_1
            
        logits_1 = beta * q_net + stickiness_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 Policy
        stickiness_bonus_2 = np.zeros(2)
        if last_action_stage2[state_idx] is not None:
            stickiness_bonus_2[last_action_stage2[state_idx]] = stickiness_2

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating with Asymmetric Alphas
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # Eligibility Trace
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_stage2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3
This model removes the restrictive combination parameter `w` and instead fits **separate inverse temperatures for the model-based (`beta_mb`) and model-free (`beta_mf`) systems**. This permits independent scaling of the decision noise associated with each system, providing a more flexible evaluation of their relative contributions. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separates the inverse temperature for model-based (beta_mb) and model-free (beta_mf) 
    systems at stage 1, avoiding the bounded weight parameter 'w'. Also includes beta_2, 
    stage-specific stickiness, and a decay parameter.
    
    Bounds:
    alpha: [0, 1]
    beta_mb: [0, 10]
    beta_mf: [0, 10]
    beta_2: [0, 10]
    lam: [0, 1]
    stickiness_1: [-5, 5]
    stickiness_2: [-5, 5]
    decay: [0, 1]
    """
    alpha, beta_mb, beta_mf, beta_2, lam, stickiness_1, stickiness_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_stage2 = {0: None, 1: None}

    for trial in range(n_trials):

        # Stage 1 Policy (Independent MB and MF Betas)
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness_bonus_1 = np.zeros(2)
        if trial > 0:
            stickiness_bonus_1[action_1[trial-1]] = stickiness_1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + stickiness_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stickiness_bonus_2 = np.zeros(2)

        logits_2 = beta_2 * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        # Decay for unchosen options

        for s in range(2):

        last_action_stage2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
```
