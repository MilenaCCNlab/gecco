import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with valence-dependent eligibility traces and unchosen Q-value decay.
    Positive and negative reward prediction errors at stage 2 are propagated back
    to stage 1 using different eligibility traces (lam_pos and lam_neg). This accounts
    for asymmetric credit assignment based on whether a reward was received or omitted.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - single inverse temperature
    w: [0, 1] - model-based weight
    lam_pos: [0, 1] - eligibility trace for positive RPEs
    lam_neg: [0, 1] - eligibility trace for negative RPEs
    p1: [-5, 5] - choice perseverance for stage 1
    p2: [-5, 5] - choice perseverance for stage 2
    decay: [0, 1] - decay rate for unchosen actions towards 0.5
    """
    alpha, beta, w, lam_pos, lam_neg, p1, p2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2 != -1:
            logits_2[last_action_2] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Valence-dependent eligibility trace
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate inverse temperatures for model-based and model-free values,
    as well as unchosen Q-value decay.
    Instead of a single beta and a weighting parameter w, this model uses beta_mb 
    for the model-based Q-values and beta_mf for the model-free Q-values at stage 1, 
    and a separate beta2 for stage 2.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_mb: [0, 10] - inverse temperature for model-based values
    beta_mf: [0, 10] - inverse temperature for model-free values
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace
    p1: [-5, 5] - choice perseverance for stage 1
    p2: [-5, 5] - choice perseverance for stage 2
    decay: [0, 1] - decay rate for unchosen actions towards 0.5
    """
    alpha, beta_mb, beta_mf, beta2, lam, p1, p2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2 != -1:
            logits_2[last_action_2] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with anti-correlated stage 2 updates and unchosen Q-value decay.
    Assumes that the reward probabilities of the two aliens on a planet are anti-correlated.
    When an alien is chosen, the unchosen alien's value in the same state is updated 
    towards the opposite reward (1 - reward).
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - single inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    p1: [-5, 5] - choice perseverance for stage 1
    p2: [-5, 5] - choice perseverance for stage 2
    decay: [0, 1] - decay rate for unchosen actions towards 0.5
    """
    alpha, beta, w, lam, p1, p2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2 != -1:
            logits_2[last_action_2] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Anti-correlated update for the unchosen alien in the same state
        unchosen_2 = 1 - action_2[trial]
        delta_stage2_anti = (1 - reward[trial]) - q_stage2_mf[state_idx, unchosen_2]
        q_stage2_mf[state_idx, unchosen_2] += alpha * delta_stage2_anti

        # Decay for stage 1 unchosen action
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        # Decay for the unvisited state's actions only
        unvisited_state = 1 - state_idx
        for a in range(2):
            q_stage2_mf[unvisited_state, a] += decay * (0.5 - q_stage2_mf[unvisited_state, a])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
