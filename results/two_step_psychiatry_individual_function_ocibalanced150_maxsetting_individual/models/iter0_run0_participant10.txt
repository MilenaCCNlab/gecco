def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model-Free SARSA(lambda) with choice perseveration.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    lam: [0,1] - Eligibility trace parameter determining how much stage 2 reward updates stage 1.
    beta_1: [0,10] - Inverse temperature for stage 1 choices (exploration vs exploitation).
    beta_2: [0,10] - Inverse temperature for stage 2 choices.
    pers: [-5,5] - Perseveration bonus for repeating the previous stage 1 choice.
    """
    learning_rate, lam, beta_1, beta_2, pers = model_parameters
    n_trials = len(action_1)
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        logits_1 = beta_1 * q_stage1_mf.copy()
        if prev_a1 != -1:
            logits_1[prev_a1] += pers
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[s].copy()
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        # Updates
        delta_1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_1
        
        delta_2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_2
        q_stage1_mf[a1] += learning_rate * lam * delta_2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based model with a single learning rate and beta.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-values.
    beta: [0,10] - Inverse temperature for both stages.
    w: [0,1] - Weighting of model-based vs model-free values (1 = pure MB, 0 = pure MF).
    lam: [0,1] - Eligibility trace.
    """
    learning_rate, beta, w, lam = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 MB values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Stage 1 net values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stage 1 policy
        logits_1 = beta * q_stage1_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # Stage 2 policy
        logits_2 = beta * q_stage2_mf[s]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        # Updates
        delta_1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_1
        
        delta_2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_2
        q_stage1_mf[a1] += learning_rate * lam * delta_2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
