def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-Step Model with Stage 2 Action Matching Bias and Stage 1 Perseveration
    Captures the participant's tendency to match their stage 2 action to their stage 1 action,
    as well as their tendency to stick with the same stage 1 action (spaceship) across consecutive trials.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    pers: stage 1 perseveration (stickiness) [-5, 5]
    """
    alpha, beta_1, beta_2, w, lam, theta, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseveration bonus to the previously chosen stage 1 action
        if last_action_1 != -1:
            q_net[last_action_1] += pers
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta_1 * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx].copy()
        
        # Add matching bias to the action that matches stage 1 choice
        q2[action_1[trial]] += theta
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta_2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2 and eligibility trace
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates with Stage 2 Action Matching Bias
    Explores whether the participant learns differently from positive versus negative 
    reward prediction errors, while still accounting for the strong stage 2 action matching.
    
    Parameters:
    alpha_pos: learning rate for positive RPE [0,1]
    alpha_neg: learning rate for negative RPE [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, theta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta_1 * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx].copy()
        
        # Add matching bias
        q2[action_1[trial]] += theta
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta_2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1_eff = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1_eff * delta_stage1
        
        # action value updating for choice 2 and eligibility trace
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2_eff = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2_eff * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_2_eff * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-Step Model with Stage 1 Choice Bias and Stage 2 Action Matching Bias
    Introduces a baseline lateral bias for stage 1 to account for the participant's 
    overwhelming preference for Spaceship 0 over the course of the session, along with 
    the stage 2 matching bias.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    bias_1: stage 1 choice bias for action 0 [-5, 5]
    """

    n_trials = len(action_1)

            continue

        # policy for the first choice

        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
