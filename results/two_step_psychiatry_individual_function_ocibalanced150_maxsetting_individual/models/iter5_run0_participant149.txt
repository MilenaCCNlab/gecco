Here are three new cognitive models based on distinct psychological mechanisms that could explain the participant's behavior in the two-step task. 

### Model 1: Transition Matrix Learning
While standard models assume participants perfectly adopt the instructed 70%/30% transition structure and keep it fixed, this model allows participants to dynamically update their internal representation of the transition probabilities based on the actual transitions they experience.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-step model with Transition Matrix Learning.
    
    Instead of assuming fixed transition probabilities (0.7/0.3), this model 
    updates the transition matrix dynamically based on observed transitions 
    using a transition learning rate.
    
    Bounds:
    learning_rate: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pi_1: [-5, 5]
    pi_2: [-5, 5]
    lr_trans: [0, 1]
    """
    import numpy as np
    learning_rate, beta, w, lam, pi_1, pi_2, lr_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize with the instructed probabilities, but allow them to change
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + pi_1 * stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[s] != -1:
            stickiness_2[last_action_2[s]] = 1.0
            
        logits_2 = beta * q_stage2_mf[s] + pi_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[s, a2] += learning_rate * delta_stage2

        # transition matrix updating
        transition_matrix[a1, s] += lr_trans * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += lr_trans * (0.0 - transition_matrix[a1, 1 - s])

        last_action_1 = a1
        last_action_2[s] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Eligibility Trace (Optimistic vs Pessimistic Credit Assignment)
This model posits that the participant may retroactively assign credit to the Stage 1 choice differently depending on whether the final reward was better or worse than expected. It uses separate eligibility traces for positive and negative Reward Prediction Errors (RPEs).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Eligibility Trace for Positive and Negative RPEs.
    
    This model allows the eligibility trace (lambda) to differ based on whether 
    the Stage 2 reward prediction error is positive or negative. This captures 
    differential credit assignment for better-than-expected vs worse-than-expected outcomes.
    
    Bounds:
    learning_rate: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam_pos: [0, 1]
    lam_neg: [0, 1]
    pi_1: [-5, 5]
    pi_2: [-5, 5]
    """
    import numpy as np
    learning_rate, beta, w, lam_pos, lam_neg, pi_1, pi_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + pi_1 * stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[s] != -1:
            stickiness_2[last_action_2[s]] = 1.0
            
        logits_2 = beta * q_stage2_mf[s] + pi_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        lam_current = lam_pos if delta_stage2 >= 0 else lam_neg

        q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam_current * delta_stage2
        q_stage2_mf[s, a2] += learning_rate * delta_stage2

        last_action_1 = a1
        last_action_2[s] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: State Perseveration (Planet Stickiness)
Participants may not just perseverate on the motor action (spaceship) they chose previously, but also on the *state* (planet) they ended up on. This model adds a bias to select the spaceship that is most likely to lead back to the planet visited on the previous trial.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: State Perseveration (Planet Stickiness).
    
    In addition to action perseveration (repeating the same spaceship), this model 
    includes state perseveration: a bias to choose the spaceship that is most 
    likely to lead to the planet visited on the previous trial.
    
    Bounds:
    learning_rate: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pi_1: [-5, 5]
    pi_2: [-5, 5]
    pi_state: [-5, 5]
    """
    import numpy as np
    learning_rate, beta, w, lam, pi_1, pi_2, pi_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action_1 = -1
    last_state = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        state_bonus = np.zeros(2)
        if last_state != -1:
            state_bonus[last_state] = 1.0
            
        # Expected state bonus for each action based on the transition matrix
        action_state_bonus = transition_matrix @ state_bonus
            
        logits_1 = beta * q_net + pi_1 * stickiness_1 + pi_state * action_state_bonus
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[s] != -1:
            stickiness_2[last_action_2[s]] = 1.0
            
        logits_2 = beta * q_stage2_mf[s] + pi_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[s, a2] += learning_rate * delta_stage2

        last_action_1 = a1
        last_state = s
        last_action_2[s] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
