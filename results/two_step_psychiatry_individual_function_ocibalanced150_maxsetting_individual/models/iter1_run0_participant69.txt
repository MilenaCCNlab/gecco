def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free reinforcement learning model with eligibility trace,
    stage 1 choice perseveration, and stage 2 choice perseveration.
    
    This model extends the baseline by adding a tendency to repeat the previous stage 2 
    choice when visiting the same state, capturing lower-level habit formation.
    
    Parameters:
    alpha: learning rate for updating Q-values
    beta_1: softmax inverse temperature for stage 1 choices
    beta_2: softmax inverse temperature for stage 2 choices
    w: mixing weight for model-based vs model-free control at stage 1
    lam: eligibility trace parameter
    p_1: choice perseveration parameter for stage 1
    p_2: choice perseveration parameter for stage 2
    
    Bounds:
    alpha: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p_1: [-5, 5]
    p_2: [-5, 5]
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p_1 * pers_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_2[last_action_2[state_idx]] = 1

        logits_2 = beta_2 * q_stage2_mf[state_idx] + p_2 * pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free reinforcement learning model with eligibility trace,
    choice perseveration, and asymmetric learning rates for positive and negative prediction errors.
    
    This model allows for different learning rates depending on whether the prediction error 
    is positive or negative, capturing optimism/pessimism biases in belief updating.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors
    alpha_neg: learning rate for negative prediction errors
    beta_1: softmax inverse temperature for stage 1 choices
    beta_2: softmax inverse temperature for stage 2 choices
    w: mixing weight for model-based vs model-free control at stage 1
    lam: eligibility trace parameter
    p: choice perseveration parameter
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p: [-5, 5]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free reinforcement learning model with eligibility trace,
    choice perseveration, and a decay/forgetting mechanism for unchosen actions.
    
    This model assumes that the values of unchosen actions slowly decay towards zero, 
    accounting for forgetting over time which encourages exploratory behavior.
    
    Parameters:
    alpha: learning rate for updating Q-values
    beta_1: softmax inverse temperature for stage 1 choices
    beta_2: softmax inverse temperature for stage 2 choices
    w: mixing weight for model-based vs model-free control at stage 1
    lam: eligibility trace parameter
    p: choice perseveration parameter
    decay: forgetting rate for unchosen actions
    
    Bounds:
    alpha: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p: [-5, 5]
    decay: [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay unchosen stage 1
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        # Decay unchosen stage 2 in the visited state
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        # Decay all actions in unvisited state
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
