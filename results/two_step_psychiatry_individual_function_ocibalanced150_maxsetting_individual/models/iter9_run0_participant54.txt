def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate inverse temperatures for Model-Based (MB), Model-Free (MF) stage 1, 
    and stage 2, eliminating the need for a weighting parameter 'w'.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_mb: inverse temperature for model-based Q-values [0, 10]
    beta_mf: inverse temperature for model-free Q-values [0, 10]
    beta_2: inverse temperature for stage 2 Q-values [0, 10]
    lam: eligibility trace [0, 1]
    pers_1: perseveration bonus for stage 1 [-5, 5]
    pers_2: perseveration bonus for stage 2 [-5, 5]
    decay: forgetting rate for unchosen options towards 0.5 [0, 1]
    """
    alpha, beta_mb, beta_mf, beta_2, lam, pers_1, pers_2, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q_1_total = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus_1
        exp_q1 = np.exp(q_1_total - np.max(q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0
            
        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers_2
                
            q_2_total = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(q_2_total - np.max(q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0
            
        if a1 != -1 and a2 != -1 and state_idx != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2
            
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1
            
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with separate learning rates for stage 1 and stage 2.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2: learning rate for stage 2 [0, 1]
    beta: shared inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers_1: perseveration bonus for stage 1 [-5, 5]
    pers_2: perseveration bonus for stage 2 [-5, 5]
    decay: forgetting rate for unchosen options towards 0.5 [0, 1]
    """
    alpha_1, alpha_2, beta, w, lam, pers_1, pers_2, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta * q_1_total - np.max(beta * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0
            
        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers_2
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta * q_2_total - np.max(beta * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0
            
        if a1 != -1 and a2 != -1 and state_idx != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            q_stage1_mf[a1] += alpha_1 * delta_stage1 + alpha_1 * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
            
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1
            
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL where the transition matrix is learned over time.
    
    Parameters:
    alpha: learning rate for Q-values [0, 1]
    beta: shared inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers_1: perseveration bonus for stage 1 [-5, 5]
    pers_2: perseveration bonus for stage 2 [-5, 5]
    alpha_trans: learning rate for the transition matrix [0, 1]
    decay: forgetting rate for unchosen options towards 0.5 [0, 1]
    """
    alpha, beta, w, lam, pers_1, pers_2, alpha_trans, decay = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix to uniform (0.5)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = np.zeros(2)
        q_stage1_mb[0] = transition_matrix[0, 0] * max_q_stage2[0] + transition_matrix[0, 1] * max_q_stage2[1]
        q_stage1_mb[1] = transition_matrix[1, 0] * max_q_stage2[0] + transition_matrix[1, 1] * max_q_stage2[1]
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta * q_1_total - np.max(beta * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0
            
        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers_2
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta * q_2_total - np.max(beta * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0
            
        if a1 != -1 and a2 != -1 and state_idx != -1:
            # Update transition matrix
            transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
            transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])
            
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2
            
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1
            
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
