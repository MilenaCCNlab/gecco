Here are three cognitive models that build upon the provided template. They introduce different well-established mechanisms from computational psychiatry to explain participant behavior in the two-step task.

### Model 1: Model-Based / Model-Free Mixture with Eligibility Trace and Stickiness
This is the standard dual-system reinforcement learning model (Daw et al., 2011). It assumes that participants mix model-based (MB) and model-free (MF) strategies. It also includes an eligibility trace (`lam`) that allows the second-stage reward to directly update the first-stage action value, and a choice `stickiness` parameter to capture perseveration (the tendency to repeat the previous first-stage choice).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Mixture of Model-Based and Model-Free learning with eligibility trace and choice stickiness.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weight given to model-based Q-values for stage 1 (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for stage 1 MF update.
    stickiness: [-5, 5] - Choice perseveration bonus for repeating the previous stage 1 action.
    """
    learning_rate, beta, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Handle missed first-stage choices
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Mix MB and MF values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add stickiness bonus for repeating last action
        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            stickiness_bonus[last_action] = stickiness
            
        logits_1 = beta * q_stage1_net + stickiness_bonus
        logits_1 -= np.max(logits_1) # Subtract max for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        # Handle missed second-stage choices (timeout)
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action = action_1[trial]
            continue
            
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates
This model explores the possibility that participants learn differently from positive versus negative prediction errors (a common phenomenon observed in cognitive science). It separates the learning rate into `alpha_pos` (better-than-expected outcomes) and `alpha_neg` (worse-than-expected outcomes). 

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Mixture of MB and MF learning with asymmetric learning rates.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (delta > 0).
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (delta <= 0).
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weight given to model-based Q-values.
    stickiness: [-5, 5] - Choice perseveration bonus for stage 1.
    """
    alpha_pos, alpha_neg, beta, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            stickiness_bonus[last_action] = stickiness
            
        logits_1 = beta * q_stage1_net + stickiness_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action = action_1[trial]
            continue
            
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        lr_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        lr_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Separate Temperatures and Value Forgetting
This model allows for different levels of exploration/exploitation between the two stages (using `beta_1` and `beta_2`). It also incorporates a `forget_rate` mechanism where the values of unchosen options naturally decay back to a neutral prior (0.5), mimicking memory decay in dynamic environments.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF mixture with separate inverse temperatures and unchosen value forgetting.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight given to model-based Q-values.
    forget_rate: [0, 1] - Decay rate for unchosen options towards 0.5.
    """
    learning_rate, beta_1, beta_2, w, forget_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 as it's the expected value of a random binary reward
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits_1 = beta_1 * q_stage1_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue
            
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Forgetting for unchosen stage 1 action
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += forget_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Forgetting for unchosen stage 2 action in the visited state
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += forget_rate * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        # Forgetting for actions in the unvisited state
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += forget_rate * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += forget_rate * (0.5 - q_stage2_mf[unvisited_state, 1])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
