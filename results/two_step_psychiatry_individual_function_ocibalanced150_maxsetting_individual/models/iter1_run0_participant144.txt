Here are 3 new cognitive models that explore alternative mechanisms for decision-making in the two-step task. 

### Model 1: Hybrid Model-Based / Model-Free Agent with Eligibility Trace
This model introduces the classic Daw et al. (2011) hybrid approach. Instead of relying purely on Model-Based (MB) values, it computes a weighted average of MB and Model-Free (MF) Q-values for Stage 1. It also includes an eligibility trace (`eligibility`), which allows the Stage 2 reward prediction error to directly update the Stage 1 MF value.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Agent with Eligibility Trace.
    Combines model-based and model-free Q-values for Stage 1 choices, and updates Stage 1 MF values with Stage 2 reward prediction error.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    w: [0, 1] - Weight on model-based vs model-free values (1 = pure MB, 0 = pure MF).
    eligibility: [0, 1] - Eligibility trace parameter (lambda) for updating Stage 1 with Stage 2 error.
    """
    learning_rate, beta, w, eligibility = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx, :])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * eligibility * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Model-Based Agent with Choice Stickiness (Perseveration)
Participants often exhibit a tendency to repeat their previous choices regardless of the reward history (perseveration) or alternate them. This model captures this by adding a "stickiness" bonus directly to the Q-value of the action that was chosen on the previous trial.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model-Based Agent with Choice Stickiness.
    Adds a bonus to the Q-values of previously chosen actions to model perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    learning_rate, beta, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_eff = q_stage1_mb.copy()
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        exp_q1 = np.exp(beta * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid MB/MF Agent with Q-value Decay
Building on the success of the "best model so far" (which demonstrated that decay improves fit) and Model 1, this model integrates the Daw et al. Hybrid mechanism with unchosen action decay. It acknowledges that memory of alien payouts slowly drifts toward baseline over time.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF Agent with Q-value Decay.
    Combines MB and MF values, and decays unchosen actions towards zero.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    w: [0, 1] - Weight on model-based vs model-free values (1 = pure MB, 0 = pure MF).
    decay_rate: [0, 1] - Decay rate for unchosen Q-values.
    """
    learning_rate, beta, w, decay_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx, :])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1.0 - decay_rate)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1.0 - decay_rate)
        q_stage2_mf[1 - state_idx, :] *= (1.0 - decay_rate)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
