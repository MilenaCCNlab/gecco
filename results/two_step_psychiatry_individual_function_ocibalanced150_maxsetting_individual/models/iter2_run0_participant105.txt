def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Model-based and Model-free mixture with eligibility trace, and separate perseveration for stage 1 and stage 2.
    This model adds a stickiness parameter for both the first-stage choices and the second-stage choices (state-dependent), 
    capturing a participant's tendency to repeat past actions regardless of reward history.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature for softmax choice [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace for updating stage 1 action values from stage 2 rewards [0,1]
    pers1: stickiness/perseveration for stage 1 choice [-5,5]
    pers2: stickiness/perseveration for stage 2 choice [-5,5]
    """
    alpha, beta, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1
    last_choice_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_choice_1 != -1:
            pers_bonus_1[last_choice_1] = pers1
            
        q1_total = beta * q_stage1_net + pers_bonus_1
        q1_max = np.max(q1_total)
        exp_q1 = np.exp(q1_total - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_choice_2[state_idx] != -1:
            pers_bonus_2[last_choice_2[state_idx]] = pers2
            
        q2_total = beta * q_stage2_mf[state_idx] + pers_bonus_2
        q2_max = np.max(q2_total)
        exp_q2 = np.exp(q2_total - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Track past choices for perseveration
        last_choice_1 = action_1[trial]
        last_choice_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate inverse temperatures for model-based and model-free values, plus eligibility trace and perseveration.
    Instead of using a weighting parameter `w`, this model separates the influence of model-based and model-free 
    values by assigning them independent inverse temperatures (beta_mb and beta_mf).
    
    Parameters:
    alpha: learning rate [0,1]
    beta_mb: inverse temperature for model-based choice [0,10]
    beta_mf: inverse temperature for model-free choice [0,10]
    lam: eligibility trace for updating stage 1 action values from stage 2 rewards [0,1]
    pers: perseveration for stage 1 choice [-5,5]
    """
    alpha, beta_mb, beta_mf, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus_1 = np.zeros(2)
        if last_choice_1 != -1:
            pers_bonus_1[last_choice_1] = pers
            
        q1_total = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus_1
        q1_max = np.max(q1_total)
        exp_q1 = np.exp(q1_total - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice (purely model-free)
        q2_total = beta_mf * q_stage2_mf[state_idx]
        q2_max = np.max(q2_total)
        exp_q2 = np.exp(q2_total - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Learns the transition matrix dynamically rather than assuming fixed probabilities.
    This model assumes the participant does not perfectly know the 70/30 transition structure beforehand 
    and updates their internal transition model over time using a transition learning rate.
    
    Parameters:
    alpha: learning rate for Q-values [0,1]
    beta: inverse temperature for softmax choice [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace for updating stage 1 action values from stage 2 rewards [0,1]
    alpha_trans: learning rate for transition probabilities [0,1]
    pers: perseveration for stage 1 choice [-5,5]
    """
    alpha, beta, w, lam, alpha_trans, pers = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix assuming 0.5 probability (uniform prior)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_choice_1 != -1:
            pers_bonus_1[last_choice_1] = pers
            
        q1_total = beta * q_stage1_net + pers_bonus_1
        q1_max = np.max(q1_total)
        exp_q1 = np.exp(q1_total - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta * q_stage2_mf[state_idx]
        q2_max = np.max(q2_total)
        exp_q2 = np.exp(q2_total - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update transition matrix
        target_trans = np.zeros(2)
        target_trans[state_idx] = 1.0
        transition_matrix[action_1[trial]] += alpha_trans * (target_trans - transition_matrix[action_1[trial]])
        
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
