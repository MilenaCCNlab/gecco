Here are three cognitive models that build upon the basic two-step task template to better capture the participant's behavior. The participant shows strong choice perseveration (often sticking with spaceship 0 even after receiving 0 coins) and occasional missed trials at the second stage (e.g., trial 51). 

### Model 1: Standard Two-Step Model (Daw et al., 2011)
This model incorporates a mixture of model-based and model-free control, an eligibility trace for updating stage 1 values from stage 2 rewards, and a choice perseveration parameter (stickiness) to account for the participant's tendency to repeat the previous first-stage choice.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Standard Two-Step Model (Daw et al., 2011)
    
    Parameters:
    learning_rate: [0,1] - learning rate for value updates
    beta: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi: [-5,5] - choice perseveration (stickiness) for stage 1
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    """
    learning_rate, beta, beta2, w, pi, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
        logits_1 = beta * q_net + pi * pers
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        if action_2[trial] != -1:
            logits_2 = beta2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            # Action value updating for choice 1
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            # Action value updating for choice 2
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
            # Eligibility trace update for choice 1
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        else:
            # Handle missed trials (e.g., trial 51)
            p_choice_2[trial] = 1.0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates (Optimism/Pessimism Bias)
This model tests whether the participant updates values differently following positive versus negative prediction errors, a common pattern in psychiatric populations (e.g., depression or addiction).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with Asymmetric Learning Rates
    
    Parameters:
    learning_rate_pos: [0,1] - learning rate for positive prediction errors
    learning_rate_neg: [0,1] - learning rate for negative prediction errors
    beta: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - model-based weight
    lam: [0,1] - eligibility trace
    """
    learning_rate_pos, learning_rate_neg, beta, beta2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        if action_2[trial] != -1:
            logits_2 = beta2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            # Action value updating for choice 1
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            lr1 = learning_rate_pos if delta_stage1 > 0 else learning_rate_neg
            q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
            
            # Action value updating for choice 2
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            lr2 = learning_rate_pos if delta_stage2 > 0 else learning_rate_neg
            q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
            
            # Eligibility trace update for choice 1
            q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
