Here are 3 new cognitive models that explore different parameter configurations and mechanisms, avoiding the combinations you've already tried. Each model is a standalone Python function.

### Model 1: Reward-Dependent Model-Based Weight and Decay
This model tests the hypothesis that participants might shift their balance between model-based and model-free strategies depending on whether they were rewarded on the previous trial, alongside decaying unchosen Q-values.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw Two-Step Model with Reward-Dependent Model-Based Weight and Decay.
    The weight 'w' on the model-based system depends on whether the previous trial was rewarded.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen actions.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.
    beta_1: [0,10] - Inverse temperature for first-stage choice.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    w_rew: [0,1] - Weight on model-based strategy if previous trial was rewarded.
    w_unrew: [0,1] - Weight on model-based strategy if previous trial was unrewarded (or for the first trial).
    lam: [0,1] - Eligibility trace parameter.
    p: [0,10] - Choice stickiness (perseveration) for the first-stage choice.
    """
    import numpy as np
    learning_rate, decay, beta_1, beta_2, w_rew, w_unrew, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        w_current = w_rew if last_reward == 1 else w_unrew
        q_net = w_current * q_stage1_mb + (1 - w_current) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + p * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay for unchosen options
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Stage Learning Rates and Reward-Dependent Stickiness
This model investigates if participants update values at different speeds for stage 1 versus stage 2, and if their likelihood to repeat a choice is contingent on whether the last trial gave them a coin.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw Two-Step Model with Separate Stage Learning Rates and Reward-Dependent Stickiness.
    Uses separate learning rates for Stage 1 and Stage 2, and choice stickiness depends on previous trial's reward.
    
    Bounds:
    alpha_1: [0,1] - Learning rate for first-stage Q-values.
    alpha_2: [0,1] - Learning rate for second-stage Q-values.
    beta_1: [0,10] - Inverse temperature for first-stage choice.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weight on model-based vs. model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    p_rew: [0,10] - First-stage choice stickiness if previous trial was rewarded.
    p_unrew: [0,10] - First-stage choice stickiness if previous trial was unrewarded.
    """
    import numpy as np
    alpha_1, alpha_2, beta_1, beta_2, w, lam, p_rew, p_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        p_current = p_rew if last_reward == 1 else p_unrew
        if last_action_1 == -1:
            p_current = 0.0

        logits_1 = beta_1 * q_net + p_current * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates and Stage-Specific Stickiness
This model checks if prediction errors of differing valences (expecting worse than what happened versus expecting better) learn at different speeds, while modeling perseveration (stickiness) effects occurring at both stages of the task independently.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw Two-Step Model with Asymmetric Learning Rates and Stage-Specific Stickiness.
    Uses separate learning rates for positive and negative prediction errors, and stickiness for both stages.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta_1: [0,10] - Inverse temperature for first-stage choice.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weight on model-based vs. model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    p_1: [0,10] - Choice stickiness for the first-stage choice.
    p_2: [0,10] - Choice stickiness for the second-stage choice (conditional on same state).
    """
    import numpy as np
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + p_1 * stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_state == state_idx and last_action_2 != -1:
            stickiness_2[last_action_2] = 1.0

        logits_2 = beta_2 * q_stage2_mf[state_idx, :] + p_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        # Eligibility trace update to stage 1 based on stage 2 PE
        alpha_trace = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_trace * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
