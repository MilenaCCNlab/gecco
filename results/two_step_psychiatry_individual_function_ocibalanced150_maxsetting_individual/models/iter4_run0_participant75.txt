def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric PE Learning Rates + Value Decay
    This model assumes that participants learn differently from better-than-expected
    (positive PE) and worse-than-expected (negative PE) outcomes, while also 
    gradually forgetting the values of unchosen options over time (decay).
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration parameter for stage 1 choice.
    decay: [0, 1] - Decay rate pulling unchosen Q-values toward 0.5.
    """
    import numpy as np
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            perseveration_1[prev_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pers * perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Decay unchosen Q-values toward 0.5
        q_stage1_mf[1 - a1] += decay * (0.5 - q_stage1_mf[1 - a1])
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha_s2 * lam * delta_stage2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-Dependent Perseveration + Value Decay
    This model assumes that the tendency to repeat the previous stage 1 choice 
    depends on whether the previous trial resulted in a reward (win-stay) or 
    no reward (lose-shift), combined with value decay for unchosen options.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Perseveration parameter for stage 1 choice after a win (reward=1).
    pers_lose: [-5, 5] - Perseveration parameter for stage 1 choice after a loss (reward=0).
    decay: [0, 1] - Decay rate pulling unchosen Q-values toward 0.5.
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_r == 1:
                perseveration_1[prev_a1] = pers_win
            else:
                perseveration_1[prev_a1] = pers_lose
            
        logits_1 = beta1 * q_stage1_net + perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Decay unchosen Q-values toward 0.5
        q_stage1_mf[1 - a1] += decay * (0.5 - q_stage1_mf[1 - a1])
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Stage 1 and Stage 2 Perseveration + Value Decay
    This model assumes that participants exhibit perseveration (stickiness) not 
    only at the first stage (spaceship choice) but also at the second stage 
    (alien choice, conditional on the state visited), alongside value decay.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers1: [-5, 5] - Perseveration parameter for stage 1 choice.
    pers2: [-5, 5] - Perseveration parameter for stage 2 choice (alien choice).
    decay: [0, 1] - Decay rate pulling unchosen Q-values toward 0.5.
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, pers1, pers2, decay = model_parameters

    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1
    prev_a2 = np.array([-1, -1])  # Keep track of previous stage 2 action for each state

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0

        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        perseveration_2 = np.zeros(2)
        if prev_a2[state_idx] != -1:
            perseveration_2[prev_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx] + pers2 * perseveration_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
