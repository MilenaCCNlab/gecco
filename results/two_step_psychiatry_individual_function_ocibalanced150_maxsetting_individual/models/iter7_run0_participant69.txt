import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free reinforcement learning model with eligibility trace,
    stage 1 and stage 2 choice perseveration, a single inverse temperature (beta), and dynamic learning of 
    the transition matrix.
    
    Parameters:
    alpha: learning rate for updating Q-values [0, 1]
    beta: softmax inverse temperature for both stage 1 and stage 2 choices [0, 10]
    w: mixing weight for model-based vs model-free control at stage 1 [0, 1]
    lam: eligibility trace parameter [0, 1]
    p_1: choice perseveration parameter for stage 1 [-5, 5]
    p_2: choice perseveration parameter for stage 2 [-5, 5]
    alpha_t: learning rate for updating transition probabilities [0, 1]
    """
    alpha, beta, w, lam, p_1, p_2, alpha_t = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p_1 * pers_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_2 = np.zeros(2)
        if last_action_2 != -1:
            pers_2[last_action_2] = 1

        logits_2 = beta * q_stage2_mf[state_idx] + p_2 * pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # transition matrix updating
        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_t * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_t * (0.0 - transition_matrix[a1, 1 - s])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free reinforcement learning model with eligibility trace,
    choice perseveration, a single inverse temperature (beta), dynamic learning of 
    the transition matrix, and asymmetric learning rates for positive and negative prediction errors.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta: softmax inverse temperature for both stage 1 and stage 2 choices [0, 10]
    w: mixing weight for model-based vs model-free control at stage 1 [0, 1]
    lam: eligibility trace parameter [0, 1]
    p: choice perseveration parameter for stage 1 [-5, 5]
    alpha_t: learning rate for updating transition probabilities [0, 1]
    """
    alpha_pos, alpha_neg, beta, w, lam, p, alpha_t = model_parameters
    n_trials = len(action_1)

    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2

        # transition matrix updating
        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_t * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_t * (0.0 - transition_matrix[a1, 1 - s])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
