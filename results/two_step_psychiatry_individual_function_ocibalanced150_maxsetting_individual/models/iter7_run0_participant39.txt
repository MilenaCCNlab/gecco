Here are 3 new cognitive models, implemented as Python functions, that explore alternative parameter configurations to explain the participant's strong behavioral biases (e.g., heavily preferring spaceship 0) and learning dynamics.

### Model 1: Stage 1 and Stage 2 Baseline Biases
This model introduces separate baseline biases for action 1 (spaceship 1 and alien 1) at both stages, along with independent inverse temperatures (`beta1`, `beta2`). Since the participant overwhelmingly favors spaceship 0, capturing a strong baseline preference is crucial.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Stage 1 and Stage 2 Baseline Biases.
    
    Parameters:
    learning_rate: learning rate for updating Q-values [0,1]
    beta1: inverse temperature for stage 1 softmax choice [0,10]
    beta2: inverse temperature for stage 2 softmax choice [0,10]
    w: weight for model-based vs model-free control [0,1]
    pers1: perseveration bonus for repeating the previous stage 1 choice [-5,5]
    bias_stage1: baseline bias for choosing action 1 in stage 1 [-5,5]
    bias_stage2: baseline bias for choosing action 1 in stage 2 [-5,5]
    """
    learning_rate, beta1, beta2, w, pers1, bias_stage1, bias_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_term = np.zeros(2)
        if last_action_1 != -1:
            pers_term[last_action_1] = pers1
            
        bias_term1 = np.array([0.0, bias_stage1])
            
        q1_vals = beta1 * q_stage1_net + pers_term + bias_term1
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        bias_term2 = np.array([0.0, bias_stage2])
        q2_vals = beta2 * q_stage2_mf[state_idx] + bias_term2
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates (Positive/Negative PE) with Stage 1 Bias
This model assumes the participant updates their values differently based on whether the outcome was better or worse than expected. It is paired with a baseline bias in stage 1 to explain the participant's overarching behavioral patterns in the absence of reward.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Learning Rates (Positive vs Negative PE) and Stage 1 Bias.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta1: inverse temperature for stage 1 softmax choice [0,10]
    beta2: inverse temperature for stage 2 softmax choice [0,10]
    w: weight for model-based vs model-free control [0,1]
    pers1: perseveration bonus for repeating the previous stage 1 choice [-5,5]
    bias_stage1: baseline bias for choosing action 1 in stage 1 [-5,5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, pers1, bias_stage1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_term = np.zeros(2)
        if last_action_1 != -1:
            pers_term[last_action_1] = pers1
            
        bias_term1 = np.array([0.0, bias_stage1])
            
        q1_vals = beta1 * q_stage1_net + pers_term + bias_term1
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
