def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative prediction errors,
    separate MB and MF temperatures, and stickiness at both stages.
    This model captures differential sensitivity to better-than-expected vs.
    worse-than-expected outcomes, which often drives win-stay/lose-shift behavior.
    
    Bounds:
    alpha_pos: [0, 1] (learning rate for positive prediction errors)
    alpha_neg: [0, 1] (learning rate for negative prediction errors)
    beta_mb: [0, 10] (inverse temperature for model-based system)
    beta_mf: [0, 10] (inverse temperature for model-free system)
    beta2: [0, 10] (inverse temperature for stage 2)
    lam: [0, 1] (eligibility trace parameter)
    pi1: [-5, 5] (choice stickiness/perseveration for stage 1)
    pi2: [-5, 5] (choice stickiness/perseveration for stage 2)
    """
    alpha_pos, alpha_neg, beta_mb, beta_mf, beta2, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        last_a1 = a1

        # Skip stage 2 if no choice was made (timeout)
        if a2 == -1:
            p_choice_2[trial] = 1.0
            continue

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            stickiness2[last_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx, :] + pi2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        last_a2[state_idx] = a2

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2

        alpha_lambda = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_lambda * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dynamic learning of the transition matrix, separate MB and MF temperatures, 
    and stickiness at both stages. Instead of assuming the transition matrix is fixed at 70/30,
    the participant updates their internal transition matrix based on observed transitions.
    
    Bounds:
    learning_rate: [0, 1] (learning rate for updating Q-values)
    alpha_trans: [0, 1] (learning rate for updating the transition matrix)
    beta_mb: [0, 10] (inverse temperature for model-based system)
    beta_mf: [0, 10] (inverse temperature for model-free system)
    beta2: [0, 10] (inverse temperature for stage 2)
    lam: [0, 1] (eligibility trace parameter)
    pi1: [-5, 5] (choice stickiness/perseveration for stage 1)
    pi2: [-5, 5] (choice stickiness/perseveration for stage 2)
    """
    learning_rate, alpha_trans, beta_mb, beta_mf, beta2, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    # Start with the instructed transition probabilities, but allow them to update over time
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        last_a1 = a1

        if a2 == -1:
            p_choice_2[trial] = 1.0
            continue

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            stickiness2[last_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx, :] + pi2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        last_a2[state_idx] = a2

        # Update transition matrix based on observed transition
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
