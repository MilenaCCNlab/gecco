Here are three novel cognitive models to explain the participant's behavior in the two-step task. These models introduce different mechanisms such as transition matrix learning, stage-specific asymmetric learning rates, and asymmetric eligibility traces. 

### Model 1: Transition Matrix Learning & Perseveration
This model relaxes the assumption that participants hold a fixed `70/30` belief about the transition structure. Instead, they dynamically update their internal transition matrix based on observed transitions. This captures potential fluctuations in participants' beliefs about the environment. It also incorporates a choice perseveration parameter (stickiness) for Stage 1.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition matrix learning and choice perseveration.
    Instead of a fixed transition matrix, the agent updates its belief about 
    the transition probabilities based on observed state transitions.
    
    Bounds:
    alpha: [0,1] - Learning rate for Q-values.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace.
    w: [0,1] - Model-based weight.
    alpha_tr: [0,1] - Learning rate for the transition matrix.
    pers: [0,5] - Perseveration (choice stickiness) for stage 1.
    """
    alpha, beta1, beta2, lam, w, alpha_tr, pers = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix belief to the true initial probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q1_total = beta1 * q_net
        
        if last_action_1 != -1:
            q1_total[last_action_1] += pers
            
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Transition matrix updating (moving belief towards observed transition)
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Asymmetric Learning Rates
This model explores the possibility that participants learn differently from positive versus negative prediction errors, and importantly, that this valence-dependent learning asymmetry differs between Stage 1 (spaceship choices) and Stage 2 (alien choices). 

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative prediction errors, 
    distinct for Stage 1 and Stage 2, capturing asymmetric learning across stages.
    
    Bounds:
    alpha1_pos: [0,1] - Stage 1 learning rate for positive prediction errors.
    alpha1_neg: [0,1] - Stage 1 learning rate for negative prediction errors.
    alpha2_pos: [0,1] - Stage 2 learning rate for positive prediction errors.
    alpha2_neg: [0,1] - Stage 2 learning rate for negative prediction errors.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace.
    w: [0,1] - Model-based weight.
    """
    alpha1_pos, alpha1_neg, alpha2_pos, alpha2_neg, beta1, beta2, lam, w = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q1_total = beta1 * q_net
        
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        a1_lr_s1 = alpha1_pos if delta_stage1 >= 0 else alpha1_neg
        a1_lr_s2 = alpha1_pos if delta_stage2 >= 0 else alpha1_neg
        a2_lr = alpha2_pos if delta_stage2 >= 0 else alpha2_neg
        
        q_stage1_mf[action_1[trial]] += a1_lr_s1 * delta_stage1 + a1_lr_s2 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += a2_lr * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Eligibility Trace
Instead of updating values differently based on valanced learning rates, this model tests whether participants propagate positive prediction errors back to Stage 1 differently than negative prediction errors via an asymmetric eligibility trace (`lam_pos` vs `lam_neg`).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric eligibility traces for positive and negative stage 2 prediction errors, 
    allowing reward-specific backwards credit assignment.
    
    Bounds:
    alpha: [0,1] - Learning rate for Q-values.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam_pos: [0,1] - Eligibility trace for positive prediction errors.
    lam_neg: [0,1] - Eligibility trace for negative prediction errors.
    w: [0,1] - Model-based weight.
    pers: [0,5] - Perseveration (choice stickiness) for stage 1.
    """
    alpha, beta1, beta2, lam_pos, lam_neg, w, pers = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q1_total = beta1 * q_net
        
        if last_action_1 != -1:
            q1_total[last_action_1] += pers
            
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Determine which trace parameter to use based on the stage 2 prediction error
        lam = lam_pos if delta_stage2 >= 0 else lam_neg
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
