Here are 3 new cognitive models, expressed as Python functions, that explore alternative mechanisms and parameter combinations to explain the participant's decision-making behavior. 

### Model 1: Separate Learning Rates and Betas for Stage 1 and Stage 2
Participants often exhibit different learning and exploitation rates at different stages of a task. This model extends the best model by allowing independent learning rates (`learning_rate_1`, `learning_rate_2`) and inverse temperatures (`beta_1`, `beta_2`) for Stage 1 and Stage 2, while retaining the Q-value decay towards 0.5.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and betas for Stage 1 and Stage 2.
    Also includes decay towards 0.5 for unchosen options, eligibility trace, and stage 1 perseveration.
    
    Parameters:
    learning_rate_1: [0, 1] - Learning rate for stage 1 Q-values.
    learning_rate_2: [0, 1] - Learning rate for stage 2 Q-values.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight on model-based vs model-free values (1 = purely MB, 0 = purely MF).
    lam: [0, 1] - Eligibility trace decay parameter (lambda).
    p: [-5, 5] - Choice perseveration (stickiness) parameter for stage 1.
    """
    learning_rate_1, learning_rate_2, decay_rate, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate_1 * delta_stage1 + learning_rate_1 * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate_2 * delta_stage2

        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Learned Transition Matrix
Rather than assuming participants perfectly know and trust the fixed transition probabilities (0.7/0.3), they might dynamically learn the transition probabilities from experience. This model introduces `lr_trans` to update the participant's internal transition matrix upon observing states.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Learning the transition matrix.
    Participants may not know or trust the fixed transition probabilities and instead learn them over time.
    Includes decay towards 0.5 for Q-values.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    lr_trans: [0, 1] - Learning rate for updating the internal transition matrix.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Choice perseveration (stickiness) parameter for stage 1.
    """
    learning_rate, lr_trans, decay_rate, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Transition matrix update based on observed state
        transition_matrix[action_1[trial], state_idx] += lr_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1.0 - transition_matrix[action_1[trial], state_idx]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2

        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Action Trace Perseveration
Observing the participant's choices, they often stick to the same spaceship for long streaks (e.g., trials 37 to 64). Instead of just perseverating on the immediately preceding action, this model maintains a decaying memory trace of past actions (`action_trace`). This allows the model to capture longer-term stickiness or habituation toward a given option.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Action Trace Perseveration.
    Maintains a decaying trace of past actions at Stage 1, capturing longer-term stickiness to a spaceship
    rather than just repeating the immediately preceding step.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Choice perseveration (stickiness) weight for the action trace.
    trace_decay: [0, 1] - Decay rate of the action trace (0 = only last action, higher values = longer memory).
    """
    learning_rate, decay_rate, beta, w, lam, p, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    action_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        exp_q1 = np.exp(beta * q_net + p * action_trace)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2

        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen
        
        # Update action trace representing habituation/long-term stickiness
        action_trace = action_trace * trace_decay
        action_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
