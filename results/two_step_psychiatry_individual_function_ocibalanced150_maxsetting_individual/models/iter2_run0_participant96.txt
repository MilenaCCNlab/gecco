import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid model with separate stage 1 and stage 2 inverse temperatures (beta_1, beta_2),
    along with stage 1 and stage 2 perseveration.
    The participant data shows long streaks of identical stage 1 choices, suggesting strong perseveration.
    Allowing different choice stochasticity (beta) at stage 1 vs stage 2 can help capture this behavior better.
    
    Bounds:
    learning_rate: [0, 1] - learning rate for both stages
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    pers_1: [-5, 5] - perseveration bonus for repeating the last stage 1 action
    pers_2: [-5, 5] - perseveration bonus for repeating the last stage 2 action in the current state
    """
    learning_rate, beta_1, beta_2, w, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -np.ones(2, dtype=int)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q1_total = beta_1 * q_stage1_net + pers_bonus_1
        q1_total -= np.max(q1_total) # for numerical stability
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2

        q2_total = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with stage 1 and stage 2 perseveration, plus an eligibility trace (lam).
    The eligibility trace allows stage 2 prediction errors to update stage 1 values directly, 
    often capturing how rewards strongly reinforce the entire sequence of choices leading to them.
    
    Bounds:
    learning_rate: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    pers_1: [-5, 5] - perseveration bonus for repeating the last stage 1 action
    pers_2: [-5, 5] - perseveration bonus for repeating the last stage 2 action in the current state
    lam: [0, 1] - eligibility trace parameter
    """
    learning_rate, beta, w, pers_1, pers_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -np.ones(2, dtype=int)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q1_total = beta * q_stage1_net + pers_bonus_1
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2

        q2_total = beta * q_stage2_mf[state_idx] + pers_bonus_2
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with asymmetric learning rates for positive and negative prediction errors,
    combined with stage 1 and stage 2 perseveration.
    People often update values differently depending on whether the outcome was better or worse than expected.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    pers_1: [-5, 5] - perseveration bonus for repeating the last stage 1 action
    pers_2: [-5, 5] - perseveration bonus for repeating the last stage 2 action in the current state
    """
    alpha_pos, alpha_neg, beta, w, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -np.ones(2, dtype=int)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q1_total = beta * q_stage1_net + pers_bonus_1
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2

        q2_total = beta * q_stage2_mf[state_idx] + pers_bonus_2
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
