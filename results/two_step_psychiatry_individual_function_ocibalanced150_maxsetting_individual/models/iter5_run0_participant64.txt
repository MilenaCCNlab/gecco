Here are 3 new cognitive models that extend the best model so far by exploring different combinations of choice stickiness (perseveration) across both stages, asymmetric learning for positive and negative prediction errors, and separate temperature parameters for each stage.

### Model 1: `alpha, beta, w, lam, p1, p2, decay`
This model builds on the best model by adding a second choice stickiness parameter (`p2`) specifically for the second stage. This allows the model to capture independent motor perseveration or alternation tendencies in both stages of the task, while retaining the forgetting mechanism (`decay`) for unchosen actions.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF hybrid with choice stickiness (perseveration) for BOTH stage 1 and stage 2,
    and a decay parameter for unchosen action values.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p1: choice stickiness parameter for stage 1 [-5,5]
    p2: choice stickiness parameter for stage 2 [-5,5]
    decay: forgetting rate for unchosen actions [0,1]
    """
    import numpy as np
    alpha, beta, w, lam, p1, p2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # Stage 1 Model-Based values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        # Stage 1 choice
        rep1 = np.zeros(2)
        if last_action_1 != -1:
            rep1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]

        # Stage 2 choice and updates
        if act2 != -1:
            rep2 = np.zeros(2)
            if last_action_2 != -1:
                rep2[last_action_2] = 1.0

            logits_2 = beta * q_stage2_mf[state_idx] + p2 * rep2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            # Stage 1 update
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            q_stage1_mf[act1] += alpha * delta_stage1

            # Stage 2 update
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            # Eligibility trace update for Stage 1
            q_stage1_mf[act1] += alpha * lam * delta_stage2

            # Decay for unchosen Stage 1 action
            unchosen_act1 = 1 - act1
            q_stage1_mf[unchosen_act1] *= (1 - decay)

            # Decay for unchosen Stage 2 actions
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == act2):
                        q_stage2_mf[s, a] *= (1 - decay)
            
            last_action_2 = act2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: `alpha_pos, alpha_neg, beta, w, lam, p1, p2`
This model incorporates asymmetric learning rates for positive and negative prediction errors. This allows the model to capture potential biases in how participants update their values after better-than-expected versus worse-than-expected outcomes, while also maintaining stage 1 and stage 2 perseveration components.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF hybrid with asymmetric learning rates for positive and negative 
    prediction errors, and choice stickiness for both stage 1 and stage 2.
    
    Parameters:
    alpha_pos: learning rate for positive PE [0,1]
    alpha_neg: learning rate for negative PE [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p1: choice stickiness parameter for stage 1 [-5,5]
    p2: choice stickiness parameter for stage 2 [-5,5]
    """
    import numpy as np
    alpha_pos, alpha_neg, beta, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep1 = np.zeros(2)
        if last_action_1 != -1:
            rep1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]

        if act2 != -1:
            rep2 = np.zeros(2)
            if last_action_2 != -1:
                rep2[last_action_2] = 1.0

            logits_2 = beta * q_stage2_mf[state_idx] + p2 * rep2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            # Stage 1 update with asymmetric learning
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
            q_stage1_mf[act1] += alpha_s1 * delta_stage1

            # Stage 2 update with asymmetric learning
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage2_mf[state_idx, act2] += alpha_s2 * delta_stage2

            # Eligibility trace update
            q_stage1_mf[act1] += alpha_s2 * lam * delta_stage2
            
            last_action_2 = act2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: `alpha, beta1, beta2, w, lam, p1, p2, decay`
This uses an 8-parameter specification where the inverse temperature (exploration vs. exploitation rate) is allowed to vary between stage 1 and stage 2. This assumes the participant might have different levels of stochasticity when making the initial structural choice versus the more immediate reward-based choice.
