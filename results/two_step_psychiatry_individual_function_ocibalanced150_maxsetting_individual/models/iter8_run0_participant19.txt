import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual-system model with forgetting (decay) and Stage 1 choice bias.
    Includes a bias parameter for Action 0 in Stage 1 to account for side/option preferences,
    which is particularly relevant given the participant's strong baseline preference for Spaceship 0.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    decay: [0, 1] (decay rate for unchosen Q-values)
    bias_1: [-5, 5] (bias towards Action 0 in Stage 1)
    """
    alpha, beta_1, beta_2, w, lam, st, decay, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        bias = np.zeros(2)
        bias[0] = bias_1
            
        logits_1 = beta_1 * q_net + st * stickiness + bias
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[1 - a1] = (1 - decay) * q_stage1_mf[1 - a1] + decay * 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
                    
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dual-system model with forgetting (decay) and an exploration bonus (kappa).
    Adds a bonus to Stage 1 options based on how many trials it has been since they were last chosen.
    This accounts for sporadic switching to the unchosen option after long periods of staying.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    decay: [0, 1] (decay rate for unchosen Q-values)
    kappa: [0, 5] (exploration bonus based on trials since last chosen)
    """
    alpha, beta_1, beta_2, w, lam, st, decay, kappa = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    trials_since_chosen = np.zeros(2)

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + st * stickiness + kappa * trials_since_chosen
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[1 - a1] = (1 - decay) * q_stage1_mf[1 - a1] + decay * 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
                    
        last_action_1 = a1
        trials_since_chosen += 1
        trials_since_chosen[a1] = 0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dual-system model with forgetting (decay), dropping the eligibility trace (lam) 
    but adding choice biases for both Stage 1 and Stage 2 to capture spatial/option preferences.
    This model aims to explain consistent selection biases at both stages of the task.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    decay: [0, 1] (decay rate for unchosen Q-values)
    bias_1: [-5, 5] (bias towards Action 0 in Stage 1)
    bias_2: [-5, 5] (bias towards Action 0 in Stage 2)
    """
    alpha, beta_1, beta_2, w, st, decay, bias_1, bias_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        b1_arr = np.zeros(2)
        b1_arr[0] = bias_1
            
        logits_1 = beta_1 * q_net + st * stickiness + b1_arr
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        b2_arr = np.zeros(2)
        b2_arr[0] = bias_2

        logits_2 = beta_2 * q_stage2_mf[state_idx] + b2_arr
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        # No lam term here, simplified to basic model-free updates
        q_stage1_mf[a1] += alpha * delta_stage1
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[1 - a1] = (1 - decay) * q_stage1_mf[1 - a1] + decay * 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
                    
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
