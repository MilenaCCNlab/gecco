Here are three cognitive models proposed to explain the participant's behavior in the two-step task. 

Given the participant data, the participant almost exclusively chose spaceship 0 for the first 153 trials before occasionally switching to spaceship 1. This strong initial bias or perseveration suggests that choice stickiness, or an asymmetric learning mechanism where positive prediction errors have a different impact than negative ones, may be at play. 

The models below build upon the standard Model-Based / Model-Free (MB/MF) hybrid reinforcement learning framework (Daw et al., 2011) to capture these dynamics.

### Model 1: Standard MB/MF Hybrid Model
This model implements a hybrid of model-based and model-free learning. It uses an eligibility trace (`lam`) to update the stage 1 model-free values based on the stage 2 reward, and a mixing weight (`w`) to balance model-based and model-free influences.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Standard Model-Based / Model-Free hybrid with eligibility trace.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    """
    alpha, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Hybrid Q-value for stage 1
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]

        # Handle missing responses (e.g., trial 30)
        if act2 != -1:
            # policy for the second choice
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]
      
            # action value updating for choice 1
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            # action value updating for choice 2
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2
            
            # eligibility trace update for choice 1
            q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: MB/MF Hybrid with Asymmetric Learning Rates
This model introduces separate learning rates for positive and negative prediction errors. This is useful to explain if the participant holds onto a preferred choice (like spaceship 0) because they learn more from successes than they unlearn from failures.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF hybrid with asymmetric learning rates for positive and negative PE.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    """
    alpha_pos, alpha_neg, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]

        if act2 != -1:
            # policy for the second choice
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]
      
            # action value updating for choice 1
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[action_1[trial]]
            alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            # action value updating for choice 2
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage2_mf[state_idx, act2] += alpha_2 * delta_stage2
            
            # eligibility trace update for choice 1
            q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: MB/MF Hybrid with Choice Stickiness (Perseveration)
This model adds a choice stickiness parameter (`p`) to the first stage policy. Given that the participant repeatedly chose spaceship 0 for over 150 trials, choice perseveration heavily influences their behavior beyond just the learned Q-values.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF hybrid with choice stickiness (perseveration) for stage 1.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness parameter [-5,5]
    """
    alpha, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add stickiness for the action taken in the previous trial
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]

        if act2 != -1:
            # policy for the second choice
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]
      
            # action value updating for choice 1
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            # action value updating for choice 2
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2
            
            # eligibility trace update for choice 1
            q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
