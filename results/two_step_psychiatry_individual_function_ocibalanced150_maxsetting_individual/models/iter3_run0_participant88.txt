def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Incorporates decay for unchosen actions and stickiness (perseveration) 
    for both Stage 1 and Stage 2 choices.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating action values.
    decay_rate: [0,1] - Rate at which unchosen Stage 2 actions decay toward 0.5.
    beta_stage1: [0,10] - Inverse temperature for Stage 1 choices.
    beta_stage2: [0,10] - Inverse temperature for Stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) systems.
    stickiness_stage1: [0,10] - Choice perseveration bonus for Stage 1.
    stickiness_stage2: [0,10] - Choice perseveration bonus for Stage 2.
    lam: [0,1] - Eligibility trace for Stage 2 reward updating Stage 1 value.
    """
    learning_rate, decay_rate, beta_stage1, beta_stage2, w, stickiness_stage1, stickiness_stage2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_stage1 * q_stage1_net
        if last_action1 != -1:
            logits_1[last_action1] += stickiness_stage1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_stage2 * q_stage2_mf[state_idx]
        if last_action2 != -1:
            logits_2[last_action2] += stickiness_stage2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += learning_rate * delta_stage2
                else:
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Differentiates learning rates for positive and negative prediction errors,
    while retaining decay for unchosen actions and the standard two-step parameters.
    
    Parameters:
    learning_rate_pos: [0,1] - Learning rate for positive prediction errors.
    learning_rate_neg: [0,1] - Learning rate for negative prediction errors.
    decay_rate: [0,1] - Rate at which unchosen Stage 2 actions decay toward 0.5.
    beta_stage1: [0,10] - Inverse temperature for Stage 1 choices.
    beta_stage2: [0,10] - Inverse temperature for Stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) systems.
    stickiness: [0,10] - Choice perseveration bonus for Stage 1.
    lam: [0,1] - Eligibility trace for Stage 2 reward updating Stage 1 value.
    """
    learning_rate_pos, learning_rate_neg, decay_rate, beta_stage1, beta_stage2, w, stickiness, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_stage1 * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += stickiness
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_stage2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = learning_rate_pos if delta_stage1 >= 0 else learning_rate_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = learning_rate_pos if delta_stage2 >= 0 else learning_rate_neg

        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += lr2 * delta_stage2
                else:
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
