def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Habit traces for choices rather than just 1-trial stickiness.
    Replaces the simple last_action stickiness with an exponentially decaying habit trace,
    allowing choices from multiple past trials to influence current perseveration.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - weight of the stage 1 habit trace
    stickiness_2: [-5,5] - weight of the stage 2 habit trace
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    habit_decay: [0,1] - rate at which the habit traces decay (0 = 1-trial stickiness, 1 = permanent habit)
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, decay_rate, lam, habit_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    habit_1 = np.zeros(2)
    habit_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy() + stickiness_1 * habit_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx].copy() + stickiness_2 * habit_2[state_idx]
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        habit_1 *= habit_decay
        habit_1[action_1[trial]] += 1
        
        habit_2 *= habit_decay
        habit_2[state_idx, action_2[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
