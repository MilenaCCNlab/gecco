Here are 3 new cognitive models that build upon the best-performing model so far. They retain the core mechanisms of model-based/model-free weighting (`w`), eligibility traces (`lam`), choice stickiness (`p`), and unchosen action decay (`decay`), while introducing new variations in learning rates and temperatures that have not yet been tested with the `decay` parameter.

### Model 1: Asymmetric Learning Rates + Decay
This model hypothesizes that participants update their values differently for positive vs. negative prediction errors, which is a common finding in reinforcement learning.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF hybrid with choice stickiness (p), unchosen action decay (decay),
    and asymmetric learning rates for positive (alpha_pos) and negative (alpha_neg) prediction errors.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness parameter [-5,5]
    decay: forgetting rate for unchosen actions [0,1]
    """
    alpha_pos, alpha_neg, beta, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]

        if act2 != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            # Stage 1 update
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
            q_stage1_mf[act1] += alpha_s1 * delta_stage1

            # Stage 2 update
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage2_mf[state_idx, act2] += alpha_s2 * delta_stage2

            # Eligibility trace update
            q_stage1_mf[act1] += alpha_s2 * lam * delta_stage2

            # Decay unchosen stage 1 actions
            unchosen_act1 = 1 - act1
            q_stage1_mf[unchosen_act1] *= (1 - decay)

            # Decay unchosen stage 2 actions
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == act2):
                        q_stage2_mf[s, a] *= (1 - decay)
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Inverse Temperatures + Decay
This model separates the inverse temperature (`beta`) into two parameters, allowing for different levels of exploration/exploitation between Stage 1 (spaceship choice) and Stage 2 (alien choice).

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF hybrid with choice stickiness (p), unchosen action decay (decay),
    and separate inverse temperatures for stage 1 (beta1) and stage 2 (beta2).
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness parameter [-5,5]
    decay: forgetting rate for unchosen actions [0,1]
    """
    alpha, beta1, beta2, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0

        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]

        if act2 != -1:
            logits_2 = beta2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            q_stage1_mf[act1] += alpha * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            q_stage1_mf[act1] += alpha * lam * delta_stage2

            unchosen_act1 = 1 - act1
            q_stage1_mf[unchosen_act1] *= (1 - decay)

            for s in range(2):
                for a in range(2):

            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage-Specific Learning Rates + Decay
This model allows for different learning rates in Stage 1 and Stage 2, accommodating the possibility that participants update their values more rapidly for the immediate reward (alien choice) compared to the abstract state transition (spaceship choice).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF hybrid with choice stickiness (p), unchosen action decay (decay),
    and separate learning rates for stage 1 (alpha1) and stage 2 (alpha2).
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness parameter [-5,5]
    decay: forgetting rate for unchosen actions [0,1]
    """
    alpha1, alpha2, beta, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]

        if act2 != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            q_stage1_mf[act1] += alpha1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha2 * delta_stage2

            # The eligibility trace updates Stage 1 using the Stage 1 learning rate
            q_stage1_mf[act1] += alpha1 * lam * delta_stage2

            unchosen_act1 = 1 - act1
            q_stage1_mf[unchosen_act1] *= (1 - decay)

            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == act2):
                        q_stage2_mf[s, a] *= (1 - decay)
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
