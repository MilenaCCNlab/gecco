def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates, shared beta, MB weight, eligibility trace, 
    action matching bias, and stage 1 perseveration.
    
    This model builds upon the best model by adding a perseveration parameter for 
    stage 1 choices, capturing the participant's tendency to repeat the same spaceship choice.
    
    Parameters:
    alpha_1: stage 1 learning rate [0,1]
    alpha_2: stage 2 learning rate [0,1]
    beta: shared inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    pers: stage 1 perseveration [-5, 5]
    """
    alpha_1, alpha_2, beta, w, lam, theta, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += pers
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        q2 = q_stage2_mf[state_idx].copy()
        q2[action_1[trial]] += theta  # action matching bias
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):

    Model 2: Comprehensive model with separate learning rates and separate betas, 
    MB weight, eligibility trace, action matching bias, and stage 1 perseveration.
    
    This model allows for different levels of stochasticity (exploration) at stage 1

    entirely deterministic based on their stage 1 choice.

    alpha_2: stage 2 learning rate [0,1]

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates (positive/negative RPE), shared beta, MB weight, 
    eligibility trace, action matching bias, and stage 1 perseveration.
    
    Instead of separate learning rates for each stage, this model uses separate learning rates
    based on the valence of the prediction error (positive vs negative), capturing potential
    differences in how the participant learns from rewards vs omissions.
    
    Parameters:
    alpha_pos: learning rate for positive RPE [0,1]
    alpha_neg: learning rate for negative RPE [0,1]
    beta: shared inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    pers: stage 1 perseveration [-5, 5]
    """
    alpha_pos, alpha_neg, beta, w, lam, theta, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += pers

        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)

        state_idx = state[trial]

        # Stage 2 policy

        q2[action_1[trial]] += theta  # action matching bias
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
