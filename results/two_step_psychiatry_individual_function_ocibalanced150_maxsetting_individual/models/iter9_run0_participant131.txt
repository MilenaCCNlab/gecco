import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Hybrid MB/MF with Asymmetric Learning Rates and Unchosen Decay.
    
    This model extends the best fitting decay model by incorporating separate learning 
    rates for positive and negative prediction errors (alpha_pos, alpha_neg). This allows 
    the model to capture differential sensitivity to wins versus losses, while still 
    forgetting unchosen option values over time.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    w: [0, 1] - Weight given to model-based vs model-free values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    stickiness_1: [-5, 5] - Log-odds bonus for repeating the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Log-odds bonus for repeating the previous Stage 2 choice.
    """
    alpha_pos, alpha_neg, decay_rate, w, beta, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay_rate * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] += decay_rate * (0.5 - q_stage2_mf[other_state, 0])
        q_stage2_mf[other_state, 1] += decay_rate * (0.5 - q_stage2_mf[other_state, 1])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Hybrid MB/MF with Eligibility Trace and Unchosen Decay.
    
    This model adds an eligibility trace (lambda_trace) to the decay model. 
    The eligibility trace allows the reward prediction error from Stage 2 
    to directly update the Stage 1 Q-value, reflecting how humans often assign 
    credit for outcomes directly to initial choices.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    lambda_trace: [0, 1] - Eligibility trace parameter for Stage 1 update.
    w: [0, 1] - Weight given to model-based vs model-free values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    stickiness_1: [-5, 5] - Log-odds bonus for repeating the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Log-odds bonus for repeating the previous Stage 2 choice.
    """
    learning_rate, decay_rate, lambda_trace, w, beta, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:

        probs_1 = exp_q1 / np.sum(exp_q1)

        state_idx = state[trial]

        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_trace * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay_rate * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] += decay_rate * (0.5 - q_stage2_mf[other_state, 0])
        q_stage2_mf[other_state, 1] += decay_rate * (0.5 - q_stage2_mf[other_state, 1])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Hybrid MB/MF with Unchosen Decay and Choice Traces.
    
    The participant data shows periods of extreme stickiness (e.g., choosing spaceship 1 
    for over 50 consecutive trials). Instead of just 1-trial stickiness, this model uses a 
    decaying choice trace to capture longer-term perseveration, providing an aggregated 
    history of recent choices.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    w: [0, 1] - Weight given to model-based vs model-free values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    alpha_c: [0, 1] - Learning rate (update rate) for the choice trace.
    stickiness_weight_1: [-5, 5] - Weight of the choice trace for Stage 1.
    stickiness_weight_2: [-5, 5] - Weight of the choice trace for Stage 2.
    """
    learning_rate, decay_rate, w, beta, alpha_c, stickiness_weight_1, stickiness_weight_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net + stickiness_weight_1 * choice_trace_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_weight_2 * choice_trace_2[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay_rate * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] += decay_rate * (0.5 - q_stage2_mf[other_state, 0])
        q_stage2_mf[other_state, 1] += decay_rate * (0.5 - q_stage2_mf[other_state, 1])
        
        # Update choice traces
        choice_trace_1 = (1 - alpha_c) * choice_trace_1
        choice_trace_1[action_1[trial]] += alpha_c
        
        choice_trace_2[state_idx] = (1 - alpha_c) * choice_trace_2[state_idx]
        choice_trace_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
