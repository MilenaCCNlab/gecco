def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates with Dual Temperature.
    Separates the learning rate into alpha_pos (for positive prediction errors) and alpha_neg (for negative).
    Uses separate inverse temperatures for model-free (beta_mf) and model-based (beta_mb) systems at stage 1.
    Includes perseveration at stage 1, eligibility trace, and stage 1 action bias.
    
    Parameters:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta_mf: [0,10] - Inverse temperature for model-free first-stage choice.
    beta_mb: [0,10] - Inverse temperature for model-based first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    p: [-5,5] - Perseveration parameter for first-stage choice.
    lambd: [0,1] - Eligibility trace parameter.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    """
    alpha_pos, alpha_neg, beta_mf, beta_mb, beta2, p, lambd, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        action_bias1 = np.array([bias1, 0.0])
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * pers + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha2 * lambd * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Outcome-dependent perseveration with Dual Temperature.
    Separates the inverse temperature for model-free and model-based systems at stage 1.
    Uses outcome-dependent perseveration (p_rew for previously rewarded trials, p_unrew for unrewarded trials).
    Includes biases for action 0 at both stages.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta_mf: [0,10] - Inverse temperature for model-free first-stage choice.
    beta_mb: [0,10] - Inverse temperature for model-based first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    p_rew: [-5,5] - Perseveration parameter for first-stage choice after a rewarded trial.
    p_unrew: [-5,5] - Perseveration parameter for first-stage choice after an unrewarded trial.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    bias2: [-5,5] - Bias towards choosing action 0 at stage 2.
    """
    learning_rate, beta_mf, beta_mb, beta2, p_rew, p_unrew, bias1, bias2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers[last_action_1] = p_rew
            elif last_reward == 0:
                pers[last_action_1] = p_unrew
            
        action_bias1 = np.array([bias1, 0.0])
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        action_bias2 = np.array([bias2, 0.0])
        logits_2 = beta2 * q_stage2_mf[state_idx] + action_bias2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
