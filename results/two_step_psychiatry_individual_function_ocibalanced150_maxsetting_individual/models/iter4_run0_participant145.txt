def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MF/MB mixture with separate perseveration for stage 1 and stage 2 choices.
    Participants often repeat their choices at both stages. This model captures that by 
    having independent perseveration parameters for the spaceship choice (stage 1) 
    and the alien choice (stage 2, conditional on the planet).
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    decay: [0,1] - decay rate for unchosen Q-values
    rep_stage1: [-5,5] - perseveration on the previous stage 1 choice
    rep_stage2: [-5,5] - perseveration on the previous stage 2 choice (in the same state)
    """
    alpha, beta1, beta2, w, lam, decay, rep_stage1, rep_stage2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_term1 = np.zeros(2)
        if last_action1 != -1:
            rep_term1[last_action1] += rep_stage1
            
        logits1 = beta1 * q_net + rep_term1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        rep_term2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            rep_term2[last_action2[state_idx]] += rep_stage2
            
        logits2 = beta2 * q_stage2_mf[state_idx] + rep_term2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MF/MB mixture with outcome-dependent perseveration and Q-value decay.
    Perseveration on the previous stage 1 choice might differ depending on whether 
    the previous trial was rewarded or not (win-stay vs lose-shift tendencies).
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    rep_win: [-5,5] - perseveration on previous stage 1 choice if rewarded
    rep_lose: [-5,5] - perseveration on previous stage 1 choice if unrewarded
    decay: [0,1] - decay rate for unchosen Q-values
    """
    alpha, beta1, beta2, w, lam, rep_win, rep_lose, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_term = np.zeros(2)
        if last_action != -1:
            if last_reward > 0:
                rep_term[last_action] += rep_win
            else:
                rep_term[last_action] += rep_lose
                
        logits1 = beta1 * q_net + rep_term
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice

        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action = action_1[trial]
        last_reward = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
