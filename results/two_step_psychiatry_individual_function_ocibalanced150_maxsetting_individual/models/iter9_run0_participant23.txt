def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Hybrid model with decay, dual betas, and outcome-dependent stickiness.
    Bounds:
    alpha: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    w: [0,1]
    lmbda: [0,1]
    decay: [0,1]
    st_win: [-5,5]
    st_lose: [-5,5]
    """
    alpha, beta1, beta2, w, lmbda, decay, st_win, st_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_reward = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add outcome-dependent stickiness bonus to the previously chosen action
        stickiness = np.zeros(2)
        if prev_a1 != -1:
            if prev_reward == 1:
                stickiness[prev_a1] = st_win
            else:
                stickiness[prev_a1] = st_lose
                
        exp_q1 = np.exp(beta1 * q_net + stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lmbda * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Decay for unchosen stage 1 action
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        # Decay for unchosen stage 2 actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        prev_a1 = action_1[trial]
        prev_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Hybrid model with decay and counterfactual updating for the unchosen stage 2 action.
    Bounds:
    alpha: [0,1]
    beta: [0,10]
    w: [0,1]
    lmbda: [0,1]
    decay: [0,1]
    alpha_cf: [0,1]
    """
    alpha, beta, w, lmbda, decay, alpha_cf = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lmbda * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Counterfactual update for the unchosen stage 2 action in the same state
        unchosen_2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_2]
        q_stage2_mf[state_idx, unchosen_2] += alpha_cf * delta_cf

        # Decay for unchosen stage 1 action
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        # Decay for stage 2 actions in the unvisited state
        for s in range(2):
            for a in range(2):
                if s != state_idx:
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Hybrid model with positive/negative alphas, dual betas, decay, and stickiness.
    Bounds:
    alpha_pos: [0,1]
    alpha_neg: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    w: [0,1]
    lmbda: [0,1]
    decay: [0,1]
    st: [-5,5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lmbda, decay, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stickiness bonus
        stickiness = np.zeros(2)
        if prev_a1 != -1:
            stickiness[prev_a1] = st
            
        exp_q1 = np.exp(beta1 * q_net + stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice

        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Asymmetric learning rates
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1 + alpha_s1 * lmbda * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        # Decay for unchosen stage 1 action
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        # Decay for unchosen stage 2 actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        prev_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
