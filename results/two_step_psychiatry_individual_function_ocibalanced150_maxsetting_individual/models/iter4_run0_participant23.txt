Here are three new cognitive models that build upon the successful `decay` mechanism (which represents forgetting for unchosen actions) by combining it with other well-established cognitive variations that have not yet been tested alongside decay. 

### Model 1: Stage-Specific Learning Rates with Decay
This model tests the hypothesis that participants update their expectations at different speeds for the first-stage choice (spaceship) versus the second-stage choice (alien). By separating `alpha1` and `alpha2`, we can capture whether learning is faster for the immediate reward (stage 2) or the delayed consequence (stage 1).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with stage-specific learning rates and decay for unchosen actions.
    Bounds:
    alpha1: [0,1] - Learning rate for stage 1
    alpha2: [0,1] - Learning rate for stage 2
    beta: [0,10] - Inverse temperature for choices
    w: [0,1] - Weight on model-based vs model-free strategy
    lmbda: [0,1] - Eligibility trace (how much stage 2 reward updates stage 1)
    decay: [0,1] - Decay rate for unchosen actions
    """
    import numpy as np
    alpha1, alpha2, beta, w, lmbda, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Prediction errors
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Action value updating with stage-specific alphas
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lmbda * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Valence-Dependent Learning Rates with Decay
This model tests the hypothesis that participants learn differently from positive prediction errors (better-than-expected outcomes) versus negative prediction errors (worse-than-expected outcomes), combined with the decay mechanism. This captures "optimistic" or "pessimistic" learning biases.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with valence-dependent learning rates and decay for unchosen actions.
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    beta: [0,10] - Inverse temperature for choices
    w: [0,1] - Weight on model-based vs model-free strategy
    lmbda: [0,1] - Eligibility trace (how much stage 2 reward updates stage 1)
    decay: [0,1] - Decay rate for unchosen actions
    """
    import numpy as np
    alpha_pos, alpha_neg, beta, w, lmbda, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Prediction errors
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Valence-dependent Action value updating
        total_delta_stage1 = delta_stage1 + lmbda * delta_stage2
        alpha1 = alpha_pos if total_delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * total_delta_stage1

        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage-Specific Exploration (Temperatures) with Decay
This model tests the hypothesis that participants exhibit different levels of exploration/exploitation at the two stages. They might be highly exploratory when picking spaceships (Stage 1) but highly exploitative when picking aliens (Stage 2), or vice-versa. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with stage-specific inverse temperatures (beta1, beta2) and decay.
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for stage 1 choices
    beta2: [0,10] - Inverse temperature for stage 2 choices
    w: [0,1] - Weight on model-based vs model-free strategy
    lmbda: [0,1] - Eligibility trace (how much stage 2 reward updates stage 1)
    decay: [0,1] - Decay rate for unchosen actions
    """
    import numpy as np
    alpha, beta1, beta2, w, lmbda, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # Stage 1 policy with beta1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta1 * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy with beta2
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Prediction errors
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Action value updating
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lmbda * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
