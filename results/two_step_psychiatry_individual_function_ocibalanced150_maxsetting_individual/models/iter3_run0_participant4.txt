Here are 3 new cognitive models that explore alternative mechanisms for decision-making in the two-step task. 

### Model 1: Choice Kernel (Exponentially Decaying Choice Trace)
Instead of a simple 1-trial stickiness parameter, this model assumes that participants maintain an exponentially decaying "choice trace" (Choice Kernel) of their past actions. This allows past choices to have a lingering effect on current decisions, decaying gradually over time.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Choice Kernel (exponentially decaying choice trace) for Stage 1.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    p: [-5, 5] - Choice kernel weight for stage 1
    lambda_: [0, 1] - Eligibility trace
    alpha_c: [0, 1] - Learning rate (decay rate) for the choice kernel
    """
    learning_rate, beta_1, beta_2, w, p, lambda_, alpha_c = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net + p * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update Stage 1 Model-Free Values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Update Stage 2 Model-Free Values
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility Trace Update for Stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        # Update Choice Trace
        choice_trace *= (1 - alpha_c)
        choice_trace[action_1[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Eligibility Trace (Sign-Dependent $\lambda$)
This model posits that the degree to which Stage 2 reward prediction errors (RPEs) influence Stage 1 values depends on whether the RPE is positive (better-than-expected) or negative (worse-than-expected). This captures participants who might strongly reinforce Stage 1 choices after a surprise win, but weakly penalize them after a surprise loss (or vice versa).

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Eligibility Trace (different lambda for positive vs negative RPEs).
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    p: [-5, 5] - Stickiness for stage 1
    lambda_pos: [0, 1] - Eligibility trace for positive Stage 2 RPE
    lambda_neg: [0, 1] - Eligibility trace for negative Stage 2 RPE
    """
    learning_rate, beta_1, beta_2, w, p, lambda_pos, lambda_neg = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update Stage 1 Model-Free Values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Update Stage 2 Model-Free Values
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Asymmetric Eligibility Trace Update
        current_lambda = lambda_pos if delta_stage2 > 0 else lambda_neg
        q_stage1_mf[action_1[trial]] += learning_rate * current_lambda * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Value Forgetting (Decay) Towards a Parameterized Center
The previous decay model forced unchosen Q-values to decay toward 0. However, in environments where rewards are strictly 0 or 1, the baseline expectation of an unchosen option naturally drifts towards the average reward rate (e.g., 0.5) rather than zero. This model adds a `decay_center` parameter to capture what baseline expected value participants regress to when they forget.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Value Forgetting (Decay) towards a parameterized center.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for chosen actions
    decay_rate: [0, 1] - Decay rate for unchosen actions
    decay_center: [0, 1] - The value towards which unchosen actions decay
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    p: [-5, 5] - Stickiness for stage 1
    lambda_: [0, 1] - Eligibility trace
    """
    learning_rate, decay_rate, decay_center, beta_1, beta_2, w, p, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to the decay center assumption
    q_stage1_mf = np.full(2, decay_center)
    q_stage2_mf = np.full((2, 2), decay_center)
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update Stage 1 Model-Free Values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Decay unchosen Stage 1 action towards decay_center
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = q_stage1_mf[unchosen_1] * (1 - decay_rate) + decay_center * decay_rate
        
        # Update Stage 2 Model-Free Values
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Decay unchosen Stage 2 actions towards decay_center
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay_rate) + decay_center * decay_rate

        # Eligibility Trace Update for Stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
