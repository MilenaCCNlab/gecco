Here are three new cognitive models that build upon the template by incorporating well-established mechanisms from the literature on two-step decision-making tasks (e.g., Daw et al., 2011). 

### Model 1: Model-Based and Model-Free Mixture (w parameter)
The template model operates as a purely model-based (MB) learner for the first stage. This model introduces a weighting parameter `w` to create a hybrid system that combines both model-based and model-free (MF) Q-values for the first-stage choice.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Model-Based and Model-Free Mixture (w parameter).
    This model combines model-based (MB) and model-free (MF) action values for the first-stage choice.
    The parameter 'w' controls the relative weight of the model-based system.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating model-free Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for model-based vs model-free influence (1 = pure MB, 0 = pure MF).
    """
    learning_rate, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice (with missed trial handling)
        if action_1[trial] == -1 or action_2[trial] == -1 or state_idx == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
        else:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid Q-learning with Eligibility Trace (lambda)
This model expands on Model 1 by adding an eligibility trace parameter (`lam`). In biological reinforcement learning, reward prediction errors at the second stage often bleed back to influence the first-stage choice directly.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Q-learning with Eligibility Trace (lambda).
    Combines MB and MF values for stage 1 using weight 'w'.
    Incorporates an eligibility trace (lam) so second-stage reward prediction errors 
    directly update the first-stage model-free action values.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating model-free Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for model-based vs model-free influence.
    lam: [0,1] - Eligibility trace decay parameter (lambda).
    """
    learning_rate, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_1[trial] == -1 or action_2[trial] == -1 or state_idx == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
        else:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            # Eligibility trace update for stage 1
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid Q-learning with Transition Matrix Learning
Standard models assume participants inherently trust the 70/30 common vs rare transition mappings. However, participants often learn or continuously update these transition probabilities iteratively over the course of the experiment. This model learns the transition matrix alongside the Q-values.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Q-learning with Transition Matrix Learning.
    In addition to learning action values, the model updates its internal transition 
    probabilities based on observed transitions using a separate learning rate (lr_trans).
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating model-free Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for model-based vs model-free influence.
    lr_trans: [0,1] - Learning rate for updating the internal transition matrix.
    """
    learning_rate, beta, w, lr_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize with the instructed base probabilities, but allow them to dynamically update
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_1[trial] == -1 or action_2[trial] == -1 or state_idx == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
        else:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            # Update transition matrix probabilities based on the observed state
            target_trans = np.zeros(2)
            target_trans[state_idx] = 1.0
            transition_matrix[action_1[trial]] += lr_trans * (target_trans - transition_matrix[action_1[trial]])
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
