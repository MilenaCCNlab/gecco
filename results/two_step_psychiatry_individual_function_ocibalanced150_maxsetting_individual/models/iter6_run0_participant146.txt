import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-learning with shared stickiness and a global response bias for action 1.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    stickiness: [-5, 5] - Shared perseveration for both stages
    response_bias: [-5, 5] - Preference for action 1 over action 0 in both stages
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness, response_bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness
            
        q_stage1_net[1] += response_bias
            
        q1_scaled = beta_1 * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_state == state_idx and last_action_2 != -1:
            q_stage2_net[last_action_2] += stickiness
            
        q_stage2_net[1] += response_bias
            
        q2_scaled = beta_2 * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Stage-specific learning rates, with asymmetric learning for positive/negative RPEs in stage 2.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1
    alpha_pos_2: [0, 1] - Learning rate for positive RPEs in stage 2
    alpha_neg_2: [0, 1] - Learning rate for negative RPEs in stage 2
    beta: [0, 10] - Inverse temperature for both stages
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    stickiness_1: [-5, 5] - Perseveration for stage 1
    stickiness_2: [-5, 5] - Perseveration for stage 2
    """
    alpha_1, alpha_pos_2, alpha_neg_2, beta, w, lam, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness_1
            
        q1_scaled = beta * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_state == state_idx and last_action_2 != -1:
            q_stage2_net[last_action_2] += stickiness_2
            
        q2_scaled = beta * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1 + alpha_1 * lam * delta_stage2
        
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_pos_2 * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_neg_2 * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Q-learning with transition matrix updating based on observed transitions.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values
    beta: [0, 10] - Inverse temperature
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    stickiness_1: [-5, 5] - Perseveration for stage 1
    stickiness_2: [-5, 5] - Perseveration for stage 2
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix
    """
    learning_rate, beta, w, lam, stickiness_1, stickiness_2, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness_1
            
        q1_scaled = beta * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_state == state_idx and last_action_2 != -1:
            q_stage2_net[last_action_2] += stickiness_2
            
        q2_scaled = beta * q_stage2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Update dynamic transition matrix towards observed transition
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
