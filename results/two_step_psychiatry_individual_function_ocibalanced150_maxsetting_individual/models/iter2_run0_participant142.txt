def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with outcome-dependent perseverance (win-stay/lose-shift) at stage 1,
    and standard perseverance at stage 2.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers_win: [-5,5] - choice perseverance for stage 1 if previous trial was rewarded
    pers_lose: [-5,5] - choice perseverance for stage 1 if previous trial was unrewarded
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_stage2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent1 = beta1 * net_q_stage1
        if trial > 0:
            if reward[trial-1] > 0:
                exponent1[action_1[trial-1]] += pers_win
            else:
                exponent1[action_1[trial-1]] += pers_lose
                
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # policy for the second choice
        state_idx = state[trial]
        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with an exponentially decaying choice trace for stage 1 perseverance,
    and a standard 1-trial perseverance for stage 2.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-values
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - perseverance weight for the choice trace at stage 1
    tau: [0,1] - decay rate of the choice trace (higher = slower decay)
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha, beta1, beta2, w, lam, pers1, tau, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)
    last_action_stage2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent1 = beta1 * net_q_stage1 + pers1 * choice_trace
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # policy for the second choice
        state_idx = state[trial]
        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # update choice trace and last action
        choice_trace *= tau
        choice_trace[action_1[trial]] += 1.0
        last_action_stage2[state_idx] = action_2[trial]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
