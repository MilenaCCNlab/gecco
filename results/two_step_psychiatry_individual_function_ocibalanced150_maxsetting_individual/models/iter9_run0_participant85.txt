def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Stage-dependent Learning Rates with Outcome-dependent Stickiness.
    
    This model uses separate learning rates for Stage 1 and Stage 2, allowing for 
    different speeds of value updating at each stage. It also separates choice stickiness 
    (perseveration) at Stage 1 based on whether the previous trial was rewarded (pi_win) 
    or unrewarded (pi_lose).

    Bounds:
    alpha1: [0, 1]
    alpha2: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    pi_win: [-5, 5]
    pi_lose: [-5, 5]
    """
    alpha1, alpha2, beta1, beta2, w, pi_win, pi_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            if last_reward == 1:
                q1_vals[last_action] += pi_win
            else:
                q1_vals[last_action] += pi_lose
                
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates with Outcome-dependent Stickiness.
    
    This model uses asymmetric learning rates for positive and negative prediction 
    errors, allowing participants to learn differently from better-than-expected 
    and worse-than-expected outcomes. It also models Stage 1 choice stickiness 
    separately for rewarded (pi_win) and unrewarded (pi_lose) previous trials.

    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    pi_win: [-5, 5]
    pi_lose: [-5, 5]

    q_stage2_mf = np.zeros((2, 2))

    last_reward = 0

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-Modulated Model-Based Weighting with Eligibility Trace.
    
    This model proposes that the balance between model-based and model-free 
    control is dynamically modulated by recent success. The weighting parameter 
    w depends on whether the previous trial was rewarded (w_win) or unrewarded 
    (w_lose). It also includes an eligibility trace (lam) for updating Stage 1 
    values based on Stage 2 rewards.

    Bounds:
    alpha: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w_win: [0, 1]
    w_lose: [0, 1]
    pi: [-5, 5]
    lam: [0, 1]
    """
    alpha, beta1, beta2, w_win, w_lose, pi, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # Determine w based on last reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = 0.5  # Default for first trial

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            q1_vals[last_action] += pi
                
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Stage 1 MF updated again by Stage 2 PE via eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
