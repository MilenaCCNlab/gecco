Here are 3 new cognitive models that build upon the best model so far by introducing alternative cognitive mechanisms.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
It is common in multistep tasks for participants to update their action values at different speeds depending on the stage. For instance, the second stage provides immediate reward feedback, whereas the first stage relies on propagated values. This model introduces separate learning rates for Stage 1 and Stage 2 updates.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Separate Stage 1 and Stage 2 Learning Rates.
    Builds upon the best model by distinguishing the learning rate for the first
    and second stage actions, allowing flexibility in how quickly participants 
    update their values at each stage.
    
    Parameters:
    learning_rate_1: [0, 1] - Learning rate for updating stage 1 Q-values.
    learning_rate_2: [0, 1] - Learning rate for updating stage 2 Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weight on model-based vs model-free values (1 = purely MB, 0 = purely MF).
    lam: [0, 1] - Eligibility trace decay parameter (lambda).
    p: [-5, 5] - Choice perseveration (stickiness) parameter for stage 1.
    """
    learning_rate_1, learning_rate_2, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2

        # Eligibility trace applies proportional to stage 1's learning rate
        q_stage1_mf[action_1[trial]] += learning_rate_1 * lam * delta_stage2

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Forgetting (Decay) of Unchosen Actions
Participants may slowly forget the values of actions they haven't chosen recently. This model includes a decay parameter where unchosen Q-values gradually drift towards zero. This acts as a mechanism to explain prolonged streaks of choosing the same action, as alternative options naturally depreciate.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Forgetting (Decay) of Unchosen Actions.
    Builds upon the best model by adding a decay parameter, which causes 
    the Q-values of unchosen actions to gradually decay towards zero over time.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weight on model-based vs model-free values (1 = purely MB, 0 = purely MF).
    lam: [0, 1] - Eligibility trace decay parameter (lambda).
    p: [-5, 5] - Choice perseveration (stickiness) parameter for stage 1.
    """
    learning_rate, decay_rate, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # calculate updates for chosen actions
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2

        # apply decay to all Q-values
        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        # assign the updated values for chosen actions (overriding the decay)
        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Transition Matrix Learning
While standard models assume participants use a fixed 0.7/0.3 representation of the environment's structure, participants may actually learn these probabilities dynamically. This model learns the transition probabilities based on the states it ends up reaching.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Transition Matrix Learning.
    Builds upon the best model by allowing the participant to learn the 
    transition probabilities between stage 1 actions and stage 2 states, 
    rather than assuming they are fixed at 0.7/0.3.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    learning_rate_trans: [0, 1] - Learning rate for updating transition probabilities.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weight on model-based vs model-free values (1 = purely MB, 0 = purely MF).
    lam: [0, 1] - Eligibility trace decay parameter (lambda).
    p: [-5, 5] - Choice perseveration (stickiness) parameter for stage 1.
    """
    learning_rate, learning_rate_trans, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to uniform probabilities
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        
        # Calculate MB values using the dynamically learned transition matrix
        q_stage1_mb = np.zeros(2)
        q_stage1_mb[0] = transition_matrix[0, 0] * max_q_stage2[0] + transition_matrix[0, 1] * max_q_stage2[1]
        q_stage1_mb[1] = transition_matrix[1, 0] * max_q_stage2[0] + transition_matrix[1, 1] * max_q_stage2[1]

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # transition matrix updating
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += learning_rate_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += learning_rate_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
