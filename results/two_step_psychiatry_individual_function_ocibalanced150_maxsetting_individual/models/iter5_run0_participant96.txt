def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Incorporates eligibility traces (lambda), decay for unchosen options at stage 2,
    and 1-trial perseveration at both stages. This builds on the best model so far by adding
    an eligibility trace to allow stage 2 rewards to directly update stage 1 values.
    
    Bounds:
    learning_rate: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    pers_1: [-5, 5] - stage 1 perseveration
    pers_2: [-5, 5] - stage 2 perseveration
    lam: [0, 1] - eligibility trace parameter
    decay_rate: [0, 1] - rate at which unchosen stage 2 values decay toward 0.5
    """
    learning_rate, beta, w, pers_1, pers_2, lam, decay_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = -np.ones(2, dtype=int)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q1_total = beta * q_stage1_net + pers_bonus_1
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2
            
        q2_total = beta * q_stage2_mf[state_idx] + pers_bonus_2
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # decay for unchosen options at stage 2
        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] = (1 - decay_rate) * q_stage2_mf[state_idx, unchosen_action_2] + decay_rate * 0.5
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] = (1 - decay_rate) * q_stage2_mf[other_state, 0] + decay_rate * 0.5
        q_stage2_mf[other_state, 1] = (1 - decay_rate) * q_stage2_mf[other_state, 1] + decay_rate * 0.5

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for positive and negative prediction errors,
    combined with perseveration and unchosen option decay. This allows the model to
    capture potential biases (e.g., ignoring negative feedback) that lead to long streaks.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    pers_1: [-5, 5] - stage 1 perseveration
    pers_2: [-5, 5] - stage 2 perseveration
    decay_rate: [0, 1] - rate at which unchosen stage 2 values decay toward 0.5
    """
    alpha_pos, alpha_neg, beta, w, pers_1, pers_2, decay_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        unchosen_action_2 = 1 - action_2[trial]

        q_stage2_mf[other_state, 0] = (1 - decay_rate) * q_stage2_mf[other_state, 0] + decay_rate * 0.5
        q_stage2_mf[other_state, 1] = (1 - decay_rate) * q_stage2_mf[other_state, 1] + decay_rate * 0.5

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10

    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Choice Kernel model. Instead of just 1-trial perseveration,
    this model maintains an exponentially decaying trace of past choices (choice kernel),
    which can effectively capture the participant's behavior of sticking to the same 
    spaceship for extended periods of time despite occasional negative feedback.
    
    Bounds:
    learning_rate: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    ck_weight_1: [-5, 5] - choice kernel weight for stage 1
    ck_weight_2: [-5, 5] - choice kernel weight for stage 2
    ck_decay: [0, 1] - decay rate for the choice kernel trace
    """
    learning_rate, beta, w, ck_weight_1, ck_weight_2, ck_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    # Choice kernels initialized to 0
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta * q_stage1_net + ck_weight_1 * ck_1
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta * q_stage2_mf[state_idx] + ck_weight_2 * ck_2[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # update choice kernels
        ck_1 = ck_1 * ck_decay
        ck_2 = ck_2 * ck_decay
        
        ck_1[action_1[trial]] += 1
        ck_2[state_idx, action_2[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
