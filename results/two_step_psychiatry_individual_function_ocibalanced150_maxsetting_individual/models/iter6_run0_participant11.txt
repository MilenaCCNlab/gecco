import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with choice perseveration at both stages and a strong motor chunking bias.
    The motor chunking bias captures the participant's tendency to repeat the same 
    motor action in stage 2 as they just executed in stage 1 (e.g., double-tapping).
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    pers_1: choice perseveration for stage 1 [-5, 5]
    pers_2: choice perseveration for stage 2 [-5, 5]
    chunking_bias: bias to choose the same motor action in stage 2 as in stage 1 [-5, 5]
    """
    alpha, beta, w, pers_1, pers_2, chunking_bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * np.copy(q_net)
        if last_action_1 != -1:
            logits_1[last_action_1] += pers_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta * np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += pers_2
            
        # Motor chunking bias: bias to repeat stage 1 action
        logits_2[action_1[trial]] += chunking_bias
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with unchosen action value decay, stage 1 perseveration, and a 
    motor chunking bias. Decay helps explain eventual switches after long streaks 
    while the chunking bias explains the near-perfect correlation between stage 1 and stage 2 choices.
    
    Parameters:
    alpha: learning rate [0, 1]
    decay: decay rate for unchosen actions [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    pers_1: choice perseveration for stage 1 [-5, 5]
    chunking_bias: bias to choose the same motor action in stage 2 as in stage 1 [-5, 5]
    """
    alpha, decay, beta, w, pers_1, chunking_bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * np.copy(q_net)
        if last_action_1 != -1:
            logits_1[last_action_1] += pers_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta * np.copy(q_stage2_mf[state_idx])
        logits_2[action_1[trial]] += chunking_bias
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Decay unchosen values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with stage 1 perseveration, win-stay/lose-shift (WSLS) bonus, 
    and a motor chunking bias. WSLS captures short-term reward-driven behavioral adjustments 
    while the chunking bias explains the action repetitions across stages.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    pers_1: choice perseveration for stage 1 [-5, 5]
    chunking_bias: bias to choose the same motor action in stage 2 as in stage 1 [-5, 5]
    wsls: win-stay/lose-shift bonus for stage 1 [-5, 5]
    """
    alpha, beta, w, pers_1, chunking_bias, wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * np.copy(q_net)
        if last_action_1 != -1:
            logits_1[last_action_1] += pers_1
            if last_reward == 1:
                logits_1[last_action_1] += wsls
            elif last_reward == 0:
                logits_1[last_action_1] -= wsls
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta * np.copy(q_stage2_mf[state_idx])
        logits_2[action_1[trial]] += chunking_bias
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
