def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice Kernel with Win/Lose Modulation
    Instead of just looking at the immediate previous trial, this model maintains a 
    continuous choice trace (Choice Kernel) for Stage 1 actions. The influence of this 
    trace on the current choice is modulated by whether the previous trial was rewarded 
    (pers_win) or unrewarded (pers_lose).
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Perseveration weight for the choice trace after a win.
    pers_lose: [-5, 5] - Perseveration weight for the choice trace after a loss.
    alpha_ck: [0, 1] - Update rate for the choice kernel.
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, alpha_ck = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck = np.zeros(2)
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_r = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_r == 1:
            perseveration_1 = pers_win * ck
        elif prev_r == 0:
            perseveration_1 = pers_lose * ck
            
        logits_1 = beta1 * q_stage1_net + perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update choice kernel
        ck[a1] += alpha_ck * (1 - ck[a1])
        ck[1 - a1] += alpha_ck * (0 - ck[1 - a1])
        
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition-Dependent Eligibility Trace + Win/Lose Perseveration
    This model hypothesizes that the model-free update to Stage 1 (eligibility trace) 
    depends on whether the transition was common or rare. Participants might attribute 
    rewards less to their Stage 1 action if a rare transition occurred.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam_com: [0, 1] - Eligibility trace for common transitions.
    lam_rare: [0, 1] - Eligibility trace for rare transitions.
    pers_win: [-5, 5] - Perseveration parameter for stage 1 choice after a win.
    pers_lose: [-5, 5] - Perseveration parameter for stage 1 choice after a loss.
    """
    alpha, beta1, beta2, w, lam_com, lam_rare, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_r == 1:
                perseveration_1[prev_a1] = pers_win
            else:
                perseveration_1[prev_a1] = pers_lose
            
        logits_1 = beta1 * q_stage1_net + perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Determine if transition was common
        # Spaceship 0 (a1=0) commonly goes to Planet 0 (state=0)
        # Spaceship 1 (a1=1) commonly goes to Planet 1 (state=1)
        is_common = (a1 == state_idx)
        current_lam = lam_com if is_common else lam_rare
        
        q_stage1_mf[a1] += alpha * current_lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Counterfactual Updating + Win/Lose Perseveration
    In addition to standard updating, this model assumes participants might infer 
    information about the unchosen Stage 2 action in the visited state. It updates 
    the unchosen Stage 2 action's value towards (1 - reward) using a counterfactual 
    learning rate (alpha_cf).
    
    Parameters:
    alpha: [0, 1] - Learning rate for chosen Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Perseveration parameter for stage 1 choice after a win.
    pers_lose: [-5, 5] - Perseveration parameter for stage 1 choice after a loss.
    alpha_cf: [0, 1] - Counterfactual learning rate for the unchosen Stage 2 action.
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, alpha_cf = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_r == 1:
                perseveration_1[prev_a1] = pers_win
            else:
                perseveration_1[prev_a1] = pers_lose
            
        logits_1 = beta1 * q_stage1_net + perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Counterfactual update for unchosen action in the SAME state
        unchosen_a2 = 1 - a2
        delta_cf = (1 - r) - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += alpha_cf * delta_cf

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
