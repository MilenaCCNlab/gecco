Here are 3 new cognitive models, each introducing a distinct psychological mechanism that builds upon the successful features of the previous best model.

### Model 1: Asymmetric Learning Rates for Positive and Negative Prediction Errors
Research often shows that individuals update their beliefs differently depending on whether the outcome was better or worse than expected. This model uses separate learning rates for positive and negative reward prediction errors, while retaining separate stickiness parameters for stage 1 and stage 2 choices.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative RPEs, with separate stickiness for stage 1 and stage 2.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta_2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    stick_1: [-5, 5] - Choice perseverance parameter for stage 1.
    stick_2: [-5, 5] - Choice perseverance parameter for stage 2 (dependent on state).
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, stick_1, stick_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    last_action_1 = -1
    last_action_2 = np.full(2, -1) 

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_term_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_term_1[last_action_1] = stick_1
            
        q1_scaled = beta_1 * q_stage1_net + stickiness_term_1
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        stickiness_term_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_term_2[last_action_2[state_idx]] = stick_2

        q2_scaled = beta_2 * q_stage2_mf[state_idx] + stickiness_term_2
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Stickiness (Win-Stay / Lose-Shift)
Instead of a static tendency to repeat the previous stage 1 choice, this model introduces reward-dependent choice perseverance. A participant might be more likely to repeat an action if it previously led to a reward (`stick_win`), and more likely to switch if it did not (`stick_lose`).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Win-stay/lose-shift behavior at stage 1 via reward-dependent stickiness, plus stage 2 stickiness.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta_2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    stick_win: [-5, 5] - Choice perseverance parameter for stage 1 if the previous trial was rewarded.
    stick_lose: [-5, 5] - Choice perseverance parameter for stage 1 if the previous trial was unrewarded.
    stick_2: [-5, 5] - Choice perseverance parameter for stage 2 (dependent on state).
    """
    learning_rate, beta_1, beta_2, w, lam, stick_win, stick_lose, stick_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.full(2, -1) 

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_term_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward > 0:
                stickiness_term_1[last_action_1] = stick_win
            else:
                stickiness_term_1[last_action_1] = stick_lose
            
        q1_scaled = beta_1 * q_stage1_net + stickiness_term_1
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        stickiness_term_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_term_2[last_action_2[state_idx]] = stick_2

        q2_scaled = beta_2 * q_stage2_mf[state_idx] + stickiness_term_2
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Dynamic Transition Matrix Learning
Instead of assuming the participant holds the transition matrix strictly at the true 0.7/0.3 probabilities throughout the experiment, this model posits that the participant dynamically updates their internal transition model based on experienced state transitions.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dynamic transition matrix updating along with separate stickiness for stage 1 and stage 2.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta_2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    stick_1: [-5, 5] - Choice perseverance parameter for stage 1.
    stick_2: [-5, 5] - Choice perseverance parameter for stage 2 (dependent on state).
    alpha_trans: [0, 1] - Learning rate for updating the internal transition matrix model.
    """
    learning_rate, beta_1, beta_2, w, lam, stick_1, stick_2, alpha_trans = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix to standard probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    last_action_1 = -1
    last_action_2 = np.full(2, -1) 

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_term_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_term_1[last_action_1] = stick_1
            
        q1_scaled = beta_1 * q_stage1_net + stickiness_term_1
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        stickiness_term_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_term_2[last_action_2[state_idx]] = stick_2

        q2_scaled = beta_2 * q_stage2_mf[state_idx] + stickiness_term_2
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Update transition matrix using Rescorla-Wagner on observed transitions
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] = 1.0 - transition_matrix[a1, state_idx]
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
