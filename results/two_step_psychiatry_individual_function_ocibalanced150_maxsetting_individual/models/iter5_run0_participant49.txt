def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for Stage 1 and Stage 2, with perseveration depending on transition type.
    
    Parameters:
    learning_rate_1: [0,1] - Learning rate for updating stage 1 values.
    learning_rate_2: [0,1] - Learning rate for updating stage 2 values.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers_c: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous transition was common.
    pers_u: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous transition was uncommon.
    """
    learning_rate_1, learning_rate_2, beta_1, beta_2, w, lam, pers_c, pers_u = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_transition_common = False

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_transition_common:
                pers_bonus[last_action_1] = pers_c
            else:
                pers_bonus[last_action_1] = pers_u
                
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage1_mf[action_1[trial]] += learning_rate_1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_transition_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Single learning rate and beta, but full factorial perseveration bonuses based on previous trial's transition and reward.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating values.
    beta: [0,10] - Inverse temperature for both stages.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter.
    pers_cr: [-5,5] - Perseveration bonus if previous trial was common and rewarded.
    pers_ur: [-5,5] - Perseveration bonus if previous trial was uncommon and rewarded.
    pers_cn: [-5,5] - Perseveration bonus if previous trial was common and unrewarded.
    pers_un: [-5,5] - Perseveration bonus if previous trial was uncommon and unrewarded.
    """
    learning_rate, beta, w, lam, pers_cr, pers_ur, pers_cn, pers_un = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_transition_common = False
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_transition_common and last_reward == 1:
                pers_bonus[last_action_1] = pers_cr
            elif not last_transition_common and last_reward == 1:
                pers_bonus[last_action_1] = pers_ur
            elif last_transition_common and last_reward == 0:
                pers_bonus[last_action_1] = pers_cn
            elif not last_transition_common and last_reward == 0:
                pers_bonus[last_action_1] = pers_un
                
        exp_q1 = np.exp(beta * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_transition_common = (action_1[trial] == state_idx)
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates for positive and negative prediction errors, with stage 1 and stage 2 perseveration.
    
    Parameters:
    lr_pos: [0,1] - Learning rate for positive prediction errors.
    lr_neg: [0,1] - Learning rate for negative prediction errors.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax.
    w: [0,1] - Weighting between model-based and model-free values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers_1: [-5,5] - Perseveration bonus for repeating stage 1 choice.
    pers_2: [-5,5] - Perseveration bonus for repeating stage 2 choice (if in the same state).
    """
    lr_pos, lr_neg, beta_1, beta_2, w, lam, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_state == state_idx and last_action_2 != -1:
            pers_bonus_2[last_action_2] = pers_2
            
        exp_q2 = np.exp(beta_2 * (q_stage2_mf[state_idx] + pers_bonus_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        lr_1_a = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_1_a * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        lr_1_b = lr_pos if delta_stage2 >= 0 else lr_neg
        lr_2 = lr_pos if delta_stage2 >= 0 else lr_neg
        
        q_stage1_mf[action_1[trial]] += lr_1_b * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
