def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with stage-specific learning rates and inverse temperatures,
    model-based weighting, eligibility trace, and stage-specific perseveration.
    
    Bounds:
    alpha1: [0, 1] (learning rate for stage 1)
    alpha2: [0, 1] (learning rate for stage 2)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (model-based weight)
    lam: [0, 1] (eligibility trace)
    pers1: [0, 5] (perseveration for stage 1)
    pers2: [0, 5] (perseveration for stage 2)
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers1
            
        exp_q1 = np.exp(beta1 * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers2
            
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Update last actions
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with stage-specific inverse temperatures, model-based weighting, eligibility trace,
    outcome-dependent stage 1 perseveration (pers_win, pers_lose), and decay for unchosen actions.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (model-based weight)
    lam: [0, 1] (eligibility trace)
    pers_win: [0, 5] (perseveration for stage 1 after a win)
    pers_lose: [0, 5] (perseveration for stage 1 after a loss)
    decay: [0, 1] (decay rate for unchosen actions towards 0.5)
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers_bonus[last_action_1] = pers_win
            elif last_reward == 0:
                pers_bonus[last_action_1] = pers_lose
                
        exp_q1 = np.exp(beta1 * q_stage1_net + pers_bonus)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Update last actions and reward
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # decay unchosen stage 2 actions
        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_action_2])
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates for positive and negative prediction errors,
    stage-specific inverse temperatures, model-based weighting, eligibility trace,
    and stage-specific perseveration.
    
    Bounds:
    alpha_pos: [0, 1] (learning rate for positive RPEs)
    alpha_neg: [0, 1] (learning rate for negative RPEs)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (model-based weight)
    lam: [0, 1] (eligibility trace)
    pers1: [0, 5] (perseveration for stage 1)
    pers2: [0, 5] (perseveration for stage 2)
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers1
            
        exp_q1 = np.exp(beta1 * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers2
            
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Update last actions
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
