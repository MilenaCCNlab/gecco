import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition-Dependent Perseveration with Unchosen Decay
    
    Instead of win-stay/lose-shift, this model uses common-stay/rare-shift tendencies.
    If the previous trial was a common transition, a 'pers_common' bonus is applied to 
    the repeated action. If the previous trial was a rare transition, a 'pers_rare' 
    bonus is applied. It also includes decay for unchosen actions towards 0.5.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based vs model-free weight
    lam: [0, 1] - Eligibility trace
    pers_common: [-5, 5] - Stage 1 perseveration bonus if previous transition was common
    pers_rare: [-5, 5] - Stage 1 perseveration bonus if previous transition was rare
    unchosen_decay: [0, 1] - Decay rate for unchosen actions towards 0.5
    """
    alpha, beta_1, beta_2, w, lam, pers_common, pers_rare, unchosen_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_transition_common = False

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_transition_common:
                pers_bonus_1[last_action_1] = pers_common
            else:
                pers_bonus_1[last_action_1] = pers_rare
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Decay unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += unchosen_decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += unchosen_decay * (0.5 - q_stage2_mf[s, a])

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        # In this task, choosing action 0 leading to state 0, or action 1 leading to state 1 is common.
        last_transition_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-Dependent Model-Based Weight with Unchosen Decay
    
    The degree of model-based control (w) depends on whether the previous trial
    was rewarded or not. This tests if participants shift between MB and MF
    strategies based on recent success or failure.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w_win: [0, 1] - Model-based vs model-free weight if previous trial was rewarded
    w_lose: [0, 1] - Model-based vs model-free weight if previous trial was unrewarded
    lam: [0, 1] - Eligibility trace
    pers: [-5, 5] - Stage 1 perseveration bonus
    unchosen_decay: [0, 1] - Decay rate for unchosen actions towards 0.5
    """
    alpha, beta_1, beta_2, w_win, w_lose, lam, pers, unchosen_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine current w based on last reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = 0.5 * (w_win + w_lose) # Default for first trial
            
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Decay unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += unchosen_decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += unchosen_decay * (0.5 - q_stage2_mf[s, a])

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Eligibility Trace with Unchosen Decay
    
    The eligibility trace (lam) for updating Stage 1 action values based on Stage 2 
    prediction errors differs for positive and negative Stage 2 prediction errors.
    This captures whether Stage 1 choices are more strongly reinforced by surprising 
    rewards vs surprising omissions of reward.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based vs model-free weight
    lam_pos: [0, 1] - Eligibility trace for positive prediction errors
    lam_neg: [0, 1] - Eligibility trace for negative prediction errors
    pers: [-5, 5] - Stage 1 perseveration bonus
    unchosen_decay: [0, 1] - Decay rate for unchosen actions towards 0.5
    """
    alpha, beta_1, beta_2, w, lam_pos, lam_neg, pers, unchosen_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        state_idx = state[trial]

        # Decay unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += unchosen_decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += unchosen_decay * (0.5 - q_stage2_mf[s, a])

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Apply asymmetric eligibility trace based on sign of prediction error
        if delta_stage2 > 0:
            lam = lam_pos
        else:
            lam = lam_neg
            
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
