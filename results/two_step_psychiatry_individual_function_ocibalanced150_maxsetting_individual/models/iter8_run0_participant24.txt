def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and inverse temperatures for stage 1 and stage 2, 
    plus separate choice stickiness (perseveration) for both stages.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1.
    alpha2: [0, 1] - Learning rate for stage 2.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight given to model-based vs. model-free values (1 = purely MB, 0 = purely MF).
    lam: [0, 1] - Eligibility trace parameter.
    rep1: [-5, 5] - Choice stickiness (perseveration) for stage 1.
    rep2: [-5, 5] - Choice stickiness (perseveration) for stage 2.
    """
    alpha1, alpha2, beta1, beta2, w, lam, rep1, rep2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # Last action taken in each stage 2 state

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness1[last_action_1] = 1
            
        logits_1 = beta1 * q_stage1_net + rep1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        state_idx = state[trial]
        
        stickiness2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness2[last_action_2[state_idx]] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + rep2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive and negative reward prediction errors,
    with a single inverse temperature, plus separate choice stickiness for both stages.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive RPEs.
    alpha_neg: [0, 1] - Learning rate for negative RPEs.
    beta: [0, 10] - Inverse temperature for both stages.
    w: [0, 1] - Weight given to model-based vs. model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rep1: [-5, 5] - Choice stickiness for stage 1.
    rep2: [-5, 5] - Choice stickiness for stage 2.
    """
    alpha_pos, alpha_neg, beta, w, lam, rep1, rep2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness1[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + rep1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        state_idx = state[trial]
        
        stickiness2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness2[last_action_2[state_idx]] = 1
            
        logits_2 = beta * q_stage2_mf[state_idx] + rep2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Updates with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_stage1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_stage2 * delta_stage2

        # Eligibility trace update to stage 1 uses stage 2 RPE sign
        q_stage1_mf[a1] += alpha_stage2 * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate inverse temperatures for stages 1 and 2, and asymmetric 
    eligibility trace parameters (lam_pos, lam_neg) based on the sign of the 
    stage 2 reward prediction error.
    
    Parameters:
    alpha: [0, 1] - Learning rate for both stages.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight given to model-based vs. model-free values.
    lam_pos: [0, 1] - Eligibility trace parameter when stage 2 RPE > 0.
    lam_neg: [0, 1] - Eligibility trace parameter when stage 2 RPE <= 0.
    rep: [-5, 5] - Choice stickiness for stage 1.
    """
    alpha, beta1, beta2, w, lam_pos, lam_neg, rep = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta1 * q_stage1_net + rep * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        state_idx = state[trial]
        
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Asymmetric eligibility trace update
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
