Here are 3 new cognitive models that explore alternative combinations of mechanisms, building on the success of the best model so far. They introduce asymmetric eligibility traces, reward-dependent perseveration, and stage-specific decay rates to better capture the participant's learning and choice dynamics.

### Model 1: Asymmetric Traces and Decay
This model extends the asymmetric learning rates by also allowing the eligibility trace (`lambda`) to differ based on whether the reward prediction error is positive or negative. This captures the possibility that the participant attributes positive outcomes to their initial choice differently than negative outcomes.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates, asymmetric eligibility traces,
    stage 1 perseveration, and decay of unchosen Q-values.
    
    Parameters:
    alpha_pos: [0,1] - learning rate for positive reward prediction errors
    alpha_neg: [0,1] - learning rate for negative reward prediction errors
    beta: [0,10] - inverse temperature for softmax choice stochasticity
    w: [0,1] - weight of model-based values
    lambda_pos: [0,1] - eligibility trace parameter for positive RPEs
    lambda_neg: [0,1] - eligibility trace parameter for negative RPEs
    p: [-5,5] - perseverance bonus for repeating the previous stage 1 action
    decay: [0,1] - decay rate for unchosen actions towards 0.5
    """
    import numpy as np
    alpha_pos, alpha_neg, beta, w, lambda_pos, lambda_neg, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta * q_stage1_net + p * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2

        lam = lambda_pos if delta_stage2 > 0 else lambda_neg
        q_stage1_mf[action_1[trial]] += lr_2 * lam * delta_stage2

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] = (1 - decay) * q_stage1_mf[a] + decay * 0.5
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Perseveration (Win-Stay/Lose-Shift)
This model tests whether the participant's tendency to repeat the previous Stage 1 choice is heavily dependent on whether the previous trial resulted in a reward or not. It also uses separate inverse temperatures for Stage 1 and Stage 2 to account for differing stochasticity between the stages.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates, separate betas for each stage,
    and reward-dependent perseveration (win-stay/lose-shift).
    
    Parameters:
    alpha_pos: [0,1] - learning rate for positive reward prediction errors
    alpha_neg: [0,1] - learning rate for negative reward prediction errors
    beta_1: [0,10] - inverse temperature for stage 1
    beta_2: [0,10] - inverse temperature for stage 2
    w: [0,1] - weight of model-based values
    lambda_: [0,1] - eligibility trace parameter
    p_win: [-5,5] - perseverance bonus for stage 1 if previous trial was rewarded
    p_lose: [-5,5] - perseverance bonus for stage 1 if previous trial was unrewarded
    """
    import numpy as np
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_reward = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        current_p = 0.0
        if last_reward == 1:
            current_p = p_win
        elif last_reward == 0:
            current_p = p_lose
            
        logits_1 = beta_1 * q_stage1_net + current_p * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_2 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage-Specific Decay Rates
This model explores the idea that memory of unchosen options degrades differently for Stage 1 (spaceships) versus Stage 2 (aliens). It utilizes separate decay rates and separate learning rates for each stage to decouple the learning dynamics of the two stages entirely.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with separate learning rates and separate decay rates for stage 1 and stage 2.
    
    Parameters:
    alpha_1: [0,1] - learning rate for stage 1 prediction errors
    alpha_2: [0,1] - learning rate for stage 2 prediction errors
    beta: [0,10] - inverse temperature for softmax choice stochasticity
    w: [0,1] - weight of model-based values
    lambda_: [0,1] - eligibility trace parameter
    p: [-5,5] - perseverance bonus for repeating the previous stage 1 action
    decay_1: [0,1] - decay rate for unchosen actions in stage 1 towards 0.5
    decay_2: [0,1] - decay rate for unchosen actions in stage 2 towards 0.5
    """
    import numpy as np
    alpha_1, alpha_2, beta, w, lambda_, p, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        q_stage1_mb = transition_matrix @ max_q_stage2

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta * q_stage1_net + p * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

                q_stage1_mf[a] = (1 - decay_1) * q_stage1_mf[a] + decay_1 * 0.5
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay_2) * q_stage2_mf[s, a] + decay_2 * 0.5
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
