Here are 3 new cognitive models, expressed as Python functions, that explore different mechanisms of learning and decision-making for the two-step task. 

### Model 1: Daw Model with Stage 1 and Stage 2 Perseveration
This model extends the baseline by adding choice stickiness (perseveration) not only to the first-stage choice but also to the second-stage choice. It captures a participant's tendency to repeat the same action when they find themselves in the same state (planet), independent of the reward.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw Model with Stage 1 and Stage 2 Perseveration
    This model adds choice stickiness (perseveration) to both the first-stage choice 
    and the second-stage choice, capturing the tendency to repeat actions in the same state.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1.
    alpha2: [0, 1] - Learning rate for stage 2.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers1: [-5, 5] - Perseveration parameter for stage 1 choice.
    pers2: [-5, 5] - Perseveration parameter for stage 2 choice.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_a2 = np.array([-1, -1])
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            perseveration_1[prev_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pers1 * perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        perseveration_2 = np.zeros(2)
        if prev_a2[state_idx] != -1:
            perseveration_2[prev_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx] + pers2 * perseveration_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        prev_a1 = a1
        prev_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Daw Model with Reward-Dependent Perseveration (Win-Stay / Lose-Shift)
This model splits the first-stage perseveration parameter into two separate parameters based on the outcome of the previous trial. It captures the distinct heuristic tendencies to repeat an action if it was rewarded ("win-stay") and switch away from an action if it was unrewarded ("lose-shift").

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw Model with Reward-Dependent Perseveration (Win-Stay / Lose-Shift)
    This model separates the perseveration tendency into two parameters based on whether
    the previous trial was rewarded or not, capturing win-stay and lose-shift heuristics.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1.
    alpha2: [0, 1] - Learning rate for stage 2.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Perseveration for stage 1 choice if previous trial was rewarded.
    pers_lose: [-5, 5] - Perseveration for stage 1 choice if previous trial was unrewarded.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_r == 1:
                perseveration_1[prev_a1] = pers_win
            elif prev_r == 0:
                perseveration_1[prev_a1] = pers_lose
            
        logits_1 = beta1 * q_stage1_net + perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Daw Model with Transition Learning
This model assumes participants do not rigidly rely on the instructed fixed transition probabilities (0.7 / 0.3) but dynamically learn and update their subjective transition structure over time based on the actual environments they encounter.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw Model with Transition Learning
    This model assumes that participants dynamically learn the transition probabilities
    of the task over time, rather than relying exclusively on fixed prior instructions.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values (same for both stages).
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration parameter for stage 1 choice.
    alpha_trans: [0, 1] - Learning rate for updating transition probabilities.
    """
    alpha, beta1, beta2, w, lam, pers, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix assuming instructed probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_1 = np.zeros(2)
        if prev_a1 != -1:
            perseveration_1[prev_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pers * perseveration_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Update transition probabilities based on observed transitions
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
