def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Stage 1 Bias, Stickiness, and Decay
    
    This model adds a constant bias parameter for the first stage choice. 
    The participant data shows a strong, almost exclusive preference for spaceship 0.
    A constant bias parameter can capture this baseline preference better than 
    just stickiness (which relies on the previous trial's action).
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax choice
    w: [0,1] - Weighting of model-based (1) vs model-free (0) values
    stickiness_1: [-5,5] - Tendency to repeat the previous stage 1 action
    stickiness_2: [-5,5] - Tendency to repeat the previous stage 2 action on the same planet
    decay_rate: [0,1] - Rate at which unchosen Q-values decay towards 0.5
    bias_1: [-5,5] - Constant bias towards choosing spaceship 0
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, decay_rate, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.full((2, 2), 0.5)
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        bias_vector_1 = np.array([bias_1, 0.0])
            
        logits_1 = beta * q_net + stickiness_1 * stick_vector_1 + bias_vector_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stick_vector_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stick_vector_2[last_action_2[state_idx]] = 1.0
            
        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_2 * stick_vector_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_action_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
