Here are 3 new cognitive models that explore alternative mechanisms for the two-step decision-making task, avoiding the parameter combinations previously tried.

### Model 1: Asymmetric Stage-2 Learning Rates
This model explores the idea that participants might learn differently from positive and negative reward prediction errors at the second stage. It uses separate learning rates (`alpha2_pos` and `alpha2_neg`) for positive and negative feedback, while maintaining a single learning rate for stage 1.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with asymmetric learning rates for positive and negative RPEs at stage 2.
    
    Parameters:
    alpha1: [0,1] - learning rate for stage 1 value updates.
    alpha2_pos: [0,1] - learning rate for stage 2 positive RPEs.
    alpha2_neg: [0,1] - learning rate for stage 2 negative RPEs.
    beta: [0,10] - inverse temperature for softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p1: [-5,5] - perseveration bonus added to the previously chosen first-stage action.
    p2: [-5,5] - perseveration bonus added to the previously chosen second-stage action.
    """
    alpha1, alpha2_pos, alpha2_neg, beta, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = p1
            
        q_net_pers1 = beta * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = p2
            
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta * q2 + pers2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += alpha2_pos * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += alpha2_neg * delta_stage2
            
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Exponentially Decaying Choice Trace
Instead of simple perseveration (which only looks one trial back), this model assumes choices leave a lingering "choice trace" that decays over time. This captures longer-term autocorrelations in a participant's choice behavior across multiple trials.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model replacing simple 1-trial perseveration with an exponentially decaying choice trace.
    
    Parameters:
    alpha1: [0,1] - learning rate for stage 1 value updates.
    alpha2: [0,1] - learning rate for stage 2 value updates.
    beta: [0,10] - inverse temperature for softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p1: [-5,5] - weight of the choice trace at stage 1.
    p2: [-5,5] - weight of the choice trace at stage 2.
    tau: [0,1] - retention rate for the choice trace (0 = no memory, 1 = perfect memory).
    """
    alpha1, alpha2, beta, w, lam, p1, p2, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_pers1 = beta * q_stage1_net + p1 * choice_trace_1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta * q2 + p2 * choice_trace_2[state_idx]
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        # Choice trace updating
        choice_trace_1 *= tau
        choice_trace_1[action_1[trial]] += 1.0
        
        choice_trace_2 *= tau
        choice_trace_2[state_idx, action_2[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Learned Transition Matrix
Rather than assuming participants perfectly know and utilize the fixed `[0.7, 0.3]` transition matrix, this model posits that participants start with an agnostic uniform prior of `[0.5, 0.5]` and dynamically update their internal transition probabilities based on observed transitions using a transition learning rate `alpha_tr`.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model where the transition matrix is learned over time, rather than fixed.
    Includes separate learning rates for stage 1 and stage 2.
    
    Parameters:
    alpha1: [0,1] - learning rate for stage 1 value updates.
    alpha2: [0,1] - learning rate for stage 2 value updates.
    beta: [0,10] - inverse temperature for softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p1: [-5,5] - perseveration bonus for stage 1.
    p2: [-5,5] - perseveration bonus for stage 2.
    alpha_tr: [0,1] - learning rate for updating the transition probabilities.
    """
    alpha1, alpha2, beta, w, lam, p1, p2, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize learned transition matrix to uniform probabilities
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = p1
            
        q_net_pers1 = beta * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = p2
            
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta * q2 + pers2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        # Transition matrix updating
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
