def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF Mixture with Forgetting and Stage 1 Perseveration
    Combines the best-performing forgetting mechanism (phi) with a 
    perseveration bonus (rep) for repeating the previous stage 1 choice.
    The participant exhibits long streaks of choosing the same spaceship, 
    making perseveration a highly relevant mechanism.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Mixing weight for MB vs MF values.
    lam: [0, 1] - Eligibility trace.
    phi: [0, 1] - Decay (forgetting) rate for unchosen actions.
    rep: [-5, 5] - Perseveration bonus for the previous stage 1 choice.
    """
    alpha, beta1, beta2, w, lam, phi, rep = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if last_action_1 != -1:
            logits_1[last_action_1] += rep
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for a in range(2):
            if a == action_1[trial]:
                q_stage1_mf[a] += alpha * delta_stage1
                q_stage1_mf[a] += alpha * lam * delta_stage2
            else:
                q_stage1_mf[a] *= (1 - phi)

        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += alpha * delta_stage2
                else:
                    q_stage2_mf[s, a] *= (1 - phi)
                    
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates with Forgetting
    Different learning rates for positive and negative prediction errors,
    combined with forgetting of unchosen actions to capture both valence-dependent
    learning and passive decay.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Mixing weight for MB vs MF values.
    lam: [0, 1] - Eligibility trace.
    phi: [0, 1] - Decay (forgetting) rate for unchosen actions.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, phi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg

        for a in range(2):
            if a == action_1[trial]:
                q_stage1_mf[a] += alpha1 * delta_stage1
                q_stage1_mf[a] += alpha2 * lam * delta_stage2
            else:
                q_stage1_mf[a] *= (1 - phi)

        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += alpha2 * delta_stage2
                else:
                    q_stage2_mf[s, a] *= (1 - phi)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF with Autoregressive Choice Trace
    Maintains a decaying trace of past stage 1 choices to model strong 
    perseveration or choice autocorrelation over long streaks, which is strongly
    evident in the participant's data (e.g. trials 22-63, 79-129).
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Mixing weight for MB vs MF values.
    lam: [0, 1] - Eligibility trace.
    tau: [0, 1] - Update rate for the choice trace.
    rho: [0, 10] - Weight of the choice trace on stage 1 choice.
    """
    alpha, beta1, beta2, w, lam, tau, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net + rho * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # update choice trace
        for a in range(2):
            if a == action_1[trial]:
                choice_trace[a] = (1 - tau) * choice_trace[a] + tau
            else:
                choice_trace[a] = (1 - tau) * choice_trace[a]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
