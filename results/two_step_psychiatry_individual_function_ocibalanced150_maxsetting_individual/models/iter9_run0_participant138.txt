import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Reward-dependent Model-Based Weight.
    
    This model assumes that the reliance on the model-based system (w) depends on 
    whether the previous trial was rewarded or not. It uses asymmetric learning rates 
    for positive and negative prediction errors.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w_rew: [0, 1] - Model-based weight after a rewarded trial.
    w_unrew: [0, 1] - Model-based weight after an unrewarded trial.
    p: [0, 10] - Perseveration for stage 1 choice.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w_rew, w_unrew, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        if last_reward == 1:
            w = w_rew
        elif last_reward == 0:
            w = w_unrew
        else:
            w = 0.5 # Default for first trial
            
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning with Choice Trace Stickiness.
    
    This model uses a decaying choice trace for stage 1 instead of just the immediate 
    last choice. This captures longer-term perseveration tendencies (e.g. sticking to one choice 
    for many trials as observed in the participant data).
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p: [0, 10] - Weight of the choice trace.
    trace_decay: [0, 1] - Decay rate of the choice trace.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, p, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits_1 = beta_1 * q_net + p * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Update choice trace
        choice_trace *= trace_decay
        choice_trace[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Learning with Reward-Dependent Stickiness and Eligibility Trace.
    
    This model assumes that the tendency to repeat a stage 1 choice (stickiness) 
    depends on whether the previous trial was rewarded (p_rew) or unrewarded (p_unrew).
    It also includes an eligibility trace (lam) to update stage 1 values directly from stage 2 rewards.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p_rew: [0, 10] - Perseveration after a rewarded trial.
    p_unrew: [0, 10] - Perseveration after an unrewarded trial.
    lam: [0, 1] - Eligibility trace for stage 1 update.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, p_rew, p_unrew, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        if last_reward == 1:
            p = p_rew
        elif last_reward == 0:
            p = p_unrew
        else:
            p = 0
            
        logits_1 = beta_1 * q_net + p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
