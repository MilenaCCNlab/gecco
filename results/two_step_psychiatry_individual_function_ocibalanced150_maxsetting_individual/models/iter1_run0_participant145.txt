import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MF/MB mixture with stage 1 and stage 2 perseveration.
    Includes stickiness to the previous first-stage and second-stage choices,
    capturing general motor perseveration at both decision stages.
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    rep1: [-5,5] - stage 1 choice perseveration
    rep2: [-5,5] - stage 2 choice perseveration
    """
    alpha, beta1, beta2, w, lam, rep1, rep2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep_term1 = np.zeros(2)
        if last_action1 != -1:
            rep_term1[last_action1] = 1
            
        logits1 = beta1 * q_net + rep1 * rep_term1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep_term2 = np.zeros(2)
        if last_action2 != -1:
            rep_term2[last_action2] = 1

        logits2 = beta2 * q_stage2_mf[state_idx] + rep2 * rep_term2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MF/MB mixture with unchosen option decay (forgetting).
    Unchosen Q-values at both stages decay towards 0 over time, which models 
    the participant forgetting the value of options they haven't recently explored.
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    rep: [-5,5] - stage 1 choice perseveration
    decay: [0,1] - decay rate for unchosen Q-values
    """
    alpha, beta1, beta2, w, lam, rep, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep_term = np.zeros(2)
        if last_action != -1:
            rep_term[last_action] = 1
            
        logits1 = beta1 * q_net + rep * rep_term
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MF/MB mixture with reward-dependent perseveration (Win-Stay/Lose-Shift).
    The tendency to repeat the previous stage 1 choice depends on whether 
    the previous trial was rewarded or not, separating perseveration into two contexts.
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    rep_rew: [-5,5] - perseveration if previously rewarded
    rep_unrew: [-5,5] - perseveration if previously unrewarded
    """
    alpha, beta1, beta2, w, lam, rep_rew, rep_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep_term = np.zeros(2)
        if last_action != -1:
            if last_reward == 1:
                rep_term[last_action] = rep_rew
            else:
                rep_term[last_action] = rep_unrew
            
        logits1 = beta1 * q_net + rep_term
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
