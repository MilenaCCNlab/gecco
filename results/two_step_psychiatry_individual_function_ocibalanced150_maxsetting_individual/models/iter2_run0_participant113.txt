Here are three new cognitive models that build upon the most successful mechanisms (like separate stickiness for stage 1 and stage 2) while exploring alternative parameter configurations, such as asymmetric learning rates, distinct decision noise for each stage, and value decay for unchosen options.

### Model 1: Asymmetric Learning Rates + Dual Stickiness
This model introduces separate learning rates for positive and negative prediction errors (`alpha_pos` and `alpha_neg`). This is a common pattern in psychiatry to capture biases in learning from rewards versus omissions, combined with the successful dual perseveration parameters.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with asymmetric learning rates for positive 
    and negative prediction errors, and separate stickiness for stage 1 and stage 2.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p_1: [0, 5]
    p_2: [0, 5]
    """
    alpha_pos, alpha_neg, beta, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = p_1
            
        exp_q1 = np.exp(beta * q_net + stickiness_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = p_2
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + stickiness_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Inverse Temperatures + Dual Stickiness
This model assumes that decision noise (exploration/exploitation balance) might fundamentally differ between the first stage (which involves model-based planning) and the second stage (which is purely model-free).

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with separate inverse temperatures (beta) 
    for stage 1 and stage 2, and separate stickiness parameters for each stage.
    
    Bounds:
    alpha: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p_1: [0, 5]
    p_2: [0, 5]
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        if last_action_1 != -1:

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = p_2
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + stickiness_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Value Decay / Forgetting + Dual Stickiness
This model captures the phenomenon where participants slowly "forget" the value of unchosen options, causing their expected values to decay towards zero over time. This acts as a mechanism to encourage exploration of options that haven't been tried recently.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free / Model-Based RL with stage-specific stickiness and a decay 
    parameter for unchosen actions to capture forgetting.
    
    Bounds:
    alpha: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p_1: [0, 5]
    p_2: [0, 5]
    decay: [0, 1]
    """
    alpha, beta, w, lam, p_1, p_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = p_1
            
        exp_q1 = np.exp(beta * q_net + stickiness_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = p_2
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + stickiness_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
