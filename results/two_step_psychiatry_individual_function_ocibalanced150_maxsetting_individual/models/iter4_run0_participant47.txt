Here are 3 new cognitive models that build upon the most successful mechanisms identified so far. 

### Model 1: Asymmetric Learning + Stage-Specific Temperatures
This model extends the best model by introducing separate inverse temperatures (`beta1` and `beta2`) for Stage 1 and Stage 2. Decision noise often differs between the two stages (e.g., participants might be more exploratory or noisy in the first stage compared to the second stage where the reward is immediately available).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF Q-learning with asymmetric learning rates for positive and negative prediction errors,
    separate inverse temperatures for stage 1 and stage 2, and separate choice stickiness.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Mixing weight for model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness for stage 1.
    pi2: [-5, 5] - Choice stickiness for stage 2.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta1 * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        last_action_1 = action_1[trial]

        # Stage 2 choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        last_action_2 = action_2[trial]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning Rates & Temperatures
Instead of separating learning rates by the sign of the prediction error (positive/negative), this model separates them by the stage of the task. Participants might update their values for the alien (Stage 2) faster or slower than they update their values for the spaceship (Stage 1).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF Q-learning with separate learning rates and inverse temperatures for stage 1 and stage 2,
    along with separate choice stickiness.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1.
    alpha2: [0, 1] - Learning rate for stage 2.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Mixing weight for model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness for stage 1.
    pi2: [-5, 5] - Choice stickiness for stage 2.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta1 * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        last_action_1 = action_1[trial]

        # Stage 2 choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        last_action_2 = action_2[trial]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Win-Stay / Lose-Shift Stickiness
This model retains the successful asymmetric learning rates but replaces the static Stage 1 stickiness (`pi1`) with a reward-dependent stickiness (`pi_win` and `pi_lose`). This captures whether the participant is more likely to repeat a Stage 1 choice if the previous trial was rewarded (win-stay) versus if it was unrewarded (lose-shift).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF Q-learning with asymmetric learning rates, and Win-stay/Lose-shift specific stickiness for stage 1.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature.
    w: [0, 1] - Mixing weight for model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pi_win: [-5, 5] - Stage 1 choice stickiness if the previous trial was rewarded.
    pi_lose: [-5, 5] - Stage 1 choice stickiness if the previous trial was unrewarded.
    pi2: [-5, 5] - Choice stickiness for stage 2.
    """
    alpha_pos, alpha_neg, beta, w, lam, pi_win, pi_lose, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        current_pi1 = pi_win if last_reward > 0 else pi_lose

        exp_q1 = np.exp(beta * q_stage1_net + current_pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        last_action_1 = action_1[trial]

        # Stage 2 choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        last_action_2 = action_2[trial]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
