Here are three new cognitive models that build upon your best model by exploring alternative mechanisms, while maintaining parsimony (8 parameters each) and avoiding previously tried combinations.

### Model 1: Initial Q-Value Parameter
This model introduces `q_init`, a parameter that determines the initial Q-values for both stages. Instead of assuming values start at 0.5, this allows the model to capture initial optimism or pessimism, which affects the magnitude of early prediction errors and exploratory behavior.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate inverse temperatures for model-based and model-free 
    stage 1 values, stage 1 and stage 2 perseveration, and an initial Q-value parameter.
    
    Bounds:
    learning_rate: [0,1]
    beta_mb: [0,10]
    beta_mf: [0,10]
    beta2: [0,10]
    pi1: [-5,5]
    pi2: [-5,5]
    lam: [0,1]
    q_init: [0,1]
    """
    learning_rate, beta_mb, beta_mf, beta2, pi1, pi2, lam, q_init = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, q_init)
    q_stage2_mf = np.full((2, 2), q_init)
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
            last_action2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Decay Rates for Stage 1 and Stage 2
This model tests the hypothesis that the tendency to forget or revert to a baseline value differs across stages. It replaces stage 2 perseveration with independent decay rates `decay1` and `decay2` while retaining the eligibility trace and separate MB/MF betas.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate decay rates for stage 1 and stage 2 values,
    eligibility trace, and separate MB/MF temperatures.
    
    Bounds:
    learning_rate: [0,1]
    decay1: [0,1]
    decay2: [0,1]
    beta_mb: [0,10]
    beta_mf: [0,10]
    beta2: [0,10]
    pi1: [-5,5]
    lam: [0,1]
    """
    learning_rate, decay1, decay2, beta_mb, beta_mf, beta2, pi1, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            logits_2 = beta2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay1 * (0.5 - q_stage1_mf[unchosen_1])
            
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == action_2[trial]):
                        q_stage2_mf[s, a] += decay2 * (0.5 - q_stage2_mf[s, a])
                        
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates + Decay (No Eligibility Trace)
This model tests whether the combination of asymmetric learning (different learning rates for positive and negative prediction errors) alongside value decay provides a better explanation than an eligibility trace mechanism (`lam`). 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate learning rates for positive and negative prediction errors,
    value decay, and separate stage 1 and stage 2 perseveration. No eligibility trace.
    
    Bounds:
    lr_pos: [0,1]
    lr_neg: [0,1]
    decay: [0,1]
    beta_mb: [0,10]
    beta_mf: [0,10]
    beta2: [0,10]
    pi1: [-5,5]
    pi2: [-5,5]
    """
    lr_pos, lr_neg, decay, beta_mb, beta_mf, beta2, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            lr1 = lr_pos if delta_stage1 > 0 else lr_neg
            q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            lr2 = lr_pos if delta_stage2 > 0 else lr_neg
            q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == action_2[trial]):
                        q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
                        
            last_action2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
