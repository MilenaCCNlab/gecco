def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with separate value decay for Stage 1 and Stage 2 unchosen actions.
    Unchosen options decay towards a neutral expected value of 0.5 over time, but at 
    different rates for the first and second stage choices.
    
    Bounds:
    alpha: [0, 1]
    decay1: [0, 1]
    decay2: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    """
    alpha, decay1, decay2, beta1, beta2, w, lam = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay1) * q_stage1_mf[unchosen_a1] + decay1 * 0.5
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay2) * q_stage2_mf[state_idx, action_idx] + decay2 * 0.5

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with value decay for unchosen actions and separate perseveration 
    for Stage 1 and Stage 2. Perseveration captures the tendency to repeat the previous 
    choice at Stage 1, and the previous choice made in the SAME state at Stage 2.
    
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pers1: [-5, 5]
    pers2: [-5, 5]
    """
    alpha, decay, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = np.full(2, -1)
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if last_a1 != -1:
            logits_1[last_a1] += pers1
            
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta2 * q_stage2_mf[s]
        if last_a2[s] != -1:
            logits_2[last_a2[s]] += pers2
            
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay) * q_stage2_mf[state_idx, action_idx] + decay * 0.5

        last_a1 = a1
        last_a2[s] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with independent weights for Model-Based and Model-Free Stage 1 values, 
    plus value decay for unchosen actions. This allows MB and MF systems to be scaled independently 
    rather than strictly trading off.
    
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w_mb: [0, 1]
    w_mf: [0, 1]
    lam: [0, 1]
    """
    alpha, decay, beta1, beta2, w_mb, w_mf, lam = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:

            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w_mb * q_stage1_mb + w_mf * q_stage1_mf
        
        logits_1 = beta1 * q_net
