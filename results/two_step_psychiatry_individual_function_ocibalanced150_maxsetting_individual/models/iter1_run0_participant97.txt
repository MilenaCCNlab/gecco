Here are 3 new cognitive models that explore different learning mechanisms and behavioral biases based on the participant's data.

### Model 1: Separate Learning Rates and Inverse Temperatures for Stage 1 and Stage 2
This model proposes that participants might learn and explore at different rates for the first and second stages. For instance, they might quickly update their preferences for the spaceships (Stage 1) but learn more slowly about the aliens (Stage 2), or vice versa.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Separate learning rates and inverse temperatures for Stage 1 and Stage 2.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for softmax action selection in Stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection in Stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = pure MB, 0 = pure MF).
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    """
    alpha1, alpha2, beta1, beta2, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_act = -1

    for trial in range(n_trials):
        # Handle missing trials
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act != -1:
            V1[last_act] += p
            
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        V2 = beta2 * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act = action_1[trial]
            continue
            
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2
        
        last_act = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Perseverance (Win-Stay / Lose-Shift)
This model differentiates the perseverance bonus based on the outcome of the previous trial. Participants often exhibit a "win-stay, lose-shift" heuristic, meaning they are more likely to repeat a choice if it was just rewarded and switch if it was unrewarded.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Reward-dependent perseverance (Win-Stay / Lose-Shift).
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p_win: [-5, 5] - Perseverance bonus if the previous trial was rewarded.
    p_lose: [-5, 5] - Perseverance bonus if the previous trial was unrewarded.
    """
    learning_rate, beta, w, lambda_, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_act = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act = -1
            last_reward = 0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        V = beta * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act != -1:
            if last_reward > 0:
                V[last_act] += p_win
            else:
                V[last_act] += p_lose
                
        V = V - np.max(V)
        exp_q1 = np.exp(V)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        V2 = beta * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act = action_1[trial]
            last_reward = 0
            continue
            
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_act = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Unchosen Option Decay (Forgetting)
This model posits that Q-values for actions that are not chosen slowly decay toward a neutral baseline (0.5, the expected value of a random choice). This accounts for "forgetting" or uncertainty growth for state-action pairs that have not been visited recently.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Unchosen Option Decay (Forgetting).
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen options.
    decay_rate: [0, 1] - Decay rate for unchosen options drifting back to 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    """
    learning_rate, decay_rate, beta, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values to neutral expected value of 0.5
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_act = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        V = beta * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act != -1:
            V[last_act] += p
            
        V = V - np.max(V)
        exp_q1 = np.exp(V)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        V2 = beta * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act = action_1[trial]
            continue
            
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Decay unchosen option in Stage 1
        unchosen_act1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_act1] += decay_rate * (0.5 - q_stage1_mf[unchosen_act1])
        
        # Decay unchosen option in Stage 2 (visited state)
        unchosen_act2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_act2] += decay_rate * (0.5 - q_stage2_mf[state_idx, unchosen_act2])
        
        # Decay both options in the unvisited Stage 2 state
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay_rate * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay_rate * (0.5 - q_stage2_mf[unvisited_state, 1])
        
        # Apply strict learning to chosen options
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_act = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
