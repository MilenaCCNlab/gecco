Here are three new cognitive models that build upon the best-performing model by exploring alternative mechanisms. They introduce separated temperatures for model-based/model-free systems, subjective transition probabilities, and dual-stage perseveration.

### Model 1: Dual Temperature Model (Separated MB and MF)
Instead of using a single weighting parameter `w`, this model assigns independent inverse temperatures (`beta_mf`, `beta_mb`) to the model-free and model-based values at stage 1. This captures whether the participant has different levels of stochasticity for the two decision systems.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual Temperature Model.
    Separates the inverse temperature for model-free and model-based systems at stage 1.
    Instead of a weighting parameter w, it uses beta_mf and beta_mb.
    Includes biases for action 0 at both stages, perseveration at stage 1, and eligibility trace.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta_mf: [0,10] - Inverse temperature for model-free first-stage choice.
    beta_mb: [0,10] - Inverse temperature for model-based first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    p: [0,10] - Perseveration parameter for first-stage choice.
    lambd: [0,1] - Eligibility trace parameter.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    bias2: [-5,5] - Bias towards choosing action 0 at stage 2.
    """
    learning_rate, beta_mf, beta_mb, beta2, p, lambd, bias1, bias2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        action_bias1 = np.array([bias1, 0.0])
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * pers + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        action_bias2 = np.array([bias2, 0.0])
        logits_2 = beta2 * q_stage2_mf[state_idx] + action_bias2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambd * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Subjective Transition Model
Participants might not perfectly represent the 70/30 transition probabilities. This model includes `pi_trans` as a parameter to estimate the participant's subjective belief regarding the common transition probability.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Subjective Transition Model.
    Incorporates a subjective transition probability parameter (pi_trans).
    Participants may not believe the transition probabilities are exactly 0.7/0.3.
    Also includes biases for action 0 at both stages and perseveration at stage 1.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta1: [0,10] - Inverse temperature for first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weighting of model-based vs. model-free values.
    p: [0,10] - Perseveration parameter for first-stage choice.
    pi_trans: [0.5, 1.0] - Subjective probability of common transition.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    bias2: [-5,5] - Bias towards choosing action 0 at stage 2.
    """
    learning_rate, beta1, beta2, w, p, pi_trans, bias1, bias2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[pi_trans, 1 - pi_trans], [1 - pi_trans, pi_trans]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        action_bias1 = np.array([bias1, 0.0])
            
        logits_1 = beta1 * q_stage1_net + p * pers + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        action_bias2 = np.array([bias2, 0.0])
        logits_2 = beta2 * q_stage2_mf[state_idx] + action_bias2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Dual-Stage Perseveration with Shared Temperature
To keep the model parsimonious while capturing motor/choice stickiness at both stages, this model assumes a shared temperature parameter (`beta`) for both stages. This frees up a parameter slot to allow for perseveration at the second stage (`p2`) alongside the eligibility trace (`lambd`) and biases.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dual-Stage Perseveration Model.
    Incorporates separate perseveration parameters for stage 1 and stage 2 choices,
    along with an eligibility trace and biases, but assumes a single shared temperature.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature shared across both stages.
    w: [0,1] - Weighting of model-based vs. model-free values.
    p1: [0,10] - Perseveration parameter for first-stage choice.
    p2: [0,10] - Perseveration parameter for second-stage choice.
    lambd: [0,1] - Eligibility trace parameter.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    bias2: [-5,5] - Bias towards choosing action 0 at stage 2.
    """
    learning_rate, beta, w, p1, p2, lambd, bias1, bias2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1
            
        action_bias1 = np.array([bias1, 0.0])
            
        logits_1 = beta * q_stage1_net + p1 * pers1 + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_action_2 != -1:
            pers2[last_action_2] = 1

        action_bias2 = np.array([bias2, 0.0])
        logits_2 = beta * q_stage2_mf[state_idx] + p2 * pers2 + action_bias2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambd * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
