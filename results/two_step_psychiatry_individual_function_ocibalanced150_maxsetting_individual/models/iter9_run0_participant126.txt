Here are 3 new cognitive models that build upon the best-performing model by introducing new mechanisms to capture participant behavior, while avoiding previously tried parameter combinations. All models maintain a maximum of 8 parameters to remain parsimonious.

### Model 1: Stage-Specific Decay
This model hypothesizes that the rate of forgetting (decay of unchosen Q-values toward 0.5) might differ between the first and second stages of the task. Participants might maintain stable valuations of the spaceships (Stage 1) but forget the values of the aliens (Stage 2) more rapidly, or vice versa.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Incorporates separate decay parameters for Stage 1 and Stage 2 unchosen options,
    allowing for different rates of forgetting at different stages of the task.
    
    Bounds:
    alpha: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stickiness_1: [-5, 5]
    stickiness_2: [-5, 5]
    decay_1: [0, 1]
    decay_2: [0, 1]
    """
    import numpy as np
    alpha, beta, w, lam, stickiness_1, stickiness_2, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_stage2 = {0: None, 1: None}

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if trial > 0:
            stickiness_bonus_1[action_1[trial-1]] = stickiness_1
            
        logits_1 = beta * q_net + stickiness_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        stickiness_bonus_2 = np.zeros(2)
        if last_action_stage2[state_idx] is not None:
            stickiness_bonus_2[last_action_stage2[state_idx]] = stickiness_2

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Unchosen Q-value decay with stage-specific rates
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_1 * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_2 * (0.5 - q_stage2_mf[s, a])
        
        last_action_stage2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Choice Kernel with Decay
Instead of simply tracking the choice on the immediate previous trial (1-trial stickiness), this model replaces the stickiness parameters with a Choice Kernel. A choice kernel acts as an exponential moving average of past choices, capturing longer-term autocorrelation in participant choices (e.g., favoring one spaceship over many trials).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Replaces simple 1-trial stickiness with a Choice Kernel that updates over time,
    capturing longer-term autocorrelation in choices, alongside unchosen Q-value decay.
    
    Bounds:
    alpha: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    ck_alpha: [0, 1]
    ck_weight_1: [-5, 5]
    ck_weight_2: [-5, 5]
    decay: [0, 1]
    """
    import numpy as np
    alpha, beta, w, lam, ck_alpha, ck_weight_1, ck_weight_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net + ck_weight_1 * ck_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta * q_stage2_mf[state_idx] + ck_weight_2 * ck_2[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Unchosen Q-value decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        # Update choice kernels
        ck_1 *= (1 - ck_alpha)
        ck_1[action_1[trial]] += ck_alpha
        
        ck_2[state_idx] *= (1 - ck_alpha)
        ck_2[state_idx, action_2[trial]] += ck_alpha

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Parameterized Decay Target
While previous models assumed unchosen action values decay toward an expected value of 0.5 (representing a neutral prior), this model allows the `decay_target` to be a free parameter. This can capture participants who have an overall pessimistic (decay target < 0.5) or optimistic (decay target > 0.5) baseline expectation of reward when they haven't sampled an option recently.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Introduces a parameterized decay target for unchosen Q-values, allowing the model
    to capture regression to a specific prior belief (not strictly 0.5) over time.
    
    Bounds:
    alpha: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stickiness_1: [-5, 5]
    stickiness_2: [-5, 5]
    decay: [0, 1]
    decay_target: [0, 1]

    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    # Initialize Q-values to the parameterized decay target
    q_stage1_mf = np.full(2, decay_target)
    q_stage2_mf = np.full((2, 2), decay_target)
    
    last_action_stage2 = {0: None, 1: None}

    for trial in range(n_trials):

        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        stickiness_bonus_2 = np.zeros(2)
        if last_action_stage2[state_idx] is not None:
            stickiness_bonus_2[last_action_stage2[state_idx]] = stickiness_2

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_bonus_2
```
