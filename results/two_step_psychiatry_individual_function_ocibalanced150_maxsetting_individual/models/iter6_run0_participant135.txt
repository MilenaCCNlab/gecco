def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate inverse temperatures for MB and MF systems, eligibility trace, 
    and reward-dependent stage 1 perseveration.
    
    Instead of a single weighting parameter `w`, this model uses independent inverse temperatures 
    (`beta_mb` and `beta_mf`) for model-based and model-free values at stage 1.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_mb: [0, 10] - Inverse temperature for model-based values
    beta_mf: [0, 10] - Inverse temperature for model-free values
    beta2: [0, 10] - Inverse temperature for stage 2 choices
    lam: [0, 1] - Eligibility trace parameter
    pers_rew: [0, 5] - Perseveration bonus if previous trial was rewarded
    pers_unrew: [0, 5] - Perseveration bonus if previous trial was unrewarded
    """
    alpha, beta_mb, beta_mf, beta2, lam, pers_rew, pers_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                pers_bonus1[last_action1] = pers_rew
            else:
                pers_bonus1[last_action1] = pers_unrew
            
        logits1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with asymmetric learning rates, eligibility trace, and stage 1 & stage 2 perseveration.
    
    This model separates perseveration into two independent parameters for stage 1 and stage 2, 
    accounting for stickiness at both decision points (stage 2 perseveration only applies if the 
    planet visited is the same as the previous trial).
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta1: [0, 10] - Inverse temperature for stage 1 choices
    beta2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers1: [0, 5] - Perseveration bonus for stage 1
    pers2: [0, 5] - Perseveration bonus for stage 2
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1
    last_state = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action1 != -1:
            pers_bonus1[last_action1] = pers1
            
        logits1 = beta1 * q_net1 + pers_bonus1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        pers_bonus2 = np.zeros(2)
        if last_state == state_idx and last_action2 != -1:
            pers_bonus2[last_action2] = pers2

        logits2 = beta2 * q_stage2_mf[state_idx] + pers_bonus2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate learning rates for stage 1 and 2, eligibility trace, 
    and a continuous choice trace for stage 1 perseveration.
    
    This model replaces single-trial perseveration with an exponentially decaying choice trace, 
    better capturing the participant's tendency to go on long streaks of choosing the same spaceship.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for stage 1
    alpha2: [0, 1] - Learning rate for stage 2
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers: [0, 5] - Perseveration weight for the choice trace
    choice_trace_decay: [0, 1] - Retention rate of the choice trace (1 = perfect memory, 0 = only last trial)
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, choice_trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits1 = beta1 * q_net1 + pers * choice_trace
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        # Update choice trace
        choice_trace *= choice_trace_decay
        choice_trace[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
