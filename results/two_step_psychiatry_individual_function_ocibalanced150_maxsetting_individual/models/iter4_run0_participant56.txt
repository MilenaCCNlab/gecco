Here are 3 new cognitive models that explore alternative mechanisms for learning and choice behavior in the two-step task. 

### Model 1: Choice Trace & Q-Value Forgetting
Instead of a simple 1-trial perseveration, this model introduces **choice traces** that decay over time. This captures longer-term stickiness to actions. It also retains the previously successful Q-value decay mechanism (forgetting unchosen options).
```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with choice traces for stage 1 and stage 2 choices,
    and a forgetting mechanism (decay) for unchosen Q-values.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based vs model-free weight
    lam: [0,1] - Eligibility trace decay
    c1: [-5,5] - Choice trace weight for stage 1
    c2: [-5,5] - Choice trace weight for stage 2
    trace_decay: [0,1] - Choice trace decay rate
    q_decay: [0,1] - Forgetting rate of unchosen options
    """
    alpha, beta, w, lam, c1, c2, trace_decay, q_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net + c1 * trace_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx] + c2 * trace_2[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Unchosen Q-value decay
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - q_decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - q_decay)

        # Choice trace updating
        trace_1 *= trace_decay
        trace_1[action_1[trial]] += 1
        
        trace_2 *= trace_decay
        trace_2[state_idx, action_2[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning Rates & Inverse Temperatures
This model explicitly separates the learning rate and exploration-exploitation tradeoff (inverse temperature) between the first and second stages, while retaining stage-specific perseveration bonuses.
```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with separate learning rates and inverse temperatures for each stage,
    along with stage-specific perseveration bonuses.
    
    Bounds:
    alpha1: [0,1] - Learning rate for stage 1
    alpha2: [0,1] - Learning rate for stage 2
    beta1: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Model-based vs model-free weight
    lam: [0,1] - Eligibility trace decay
    p1: [-5,5] - Choice perseveration bonus for stage 1
    p2: [-5,5] - State-specific choice perseveration bonus for stage 2
    """
    alpha1, alpha2, beta1, beta2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = p1
            
        logits_1 = beta1 * q_stage1_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2_in_state[state_idx] != -1:
            pers_bonus_2[last_action_2_in_state[state_idx]] = p2
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + pers_bonus_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Win-Stay / Lose-Shift Perseveration
Instead of applying a constant stickiness bonus, this model evaluates whether the previous trial was rewarded or not, and applies different perseveration bonuses based on that outcome (i.e. capturing win-stay and lose-shift dynamics) for both stages.
```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with reward-dependent perseveration (win-stay/lose-shift) 
    for both stage 1 and stage-specific stage 2 choices.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based vs model-free weight
    lam: [0,1] - Eligibility trace decay
    p1_win: [-5,5] - Perseveration bonus for stage 1 if previously rewarded
    p1_lose: [-5,5] - Perseveration bonus for stage 1 if previously unrewarded
    p2_win: [-5,5] - Perseveration bonus for stage 2 if state was previously rewarded
    p2_lose: [-5,5] - Perseveration bonus for stage 2 if state was previously unrewarded
    """
    alpha, beta, w, lam, p1_win, p1_lose, p2_win, p2_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward_1 = -1
    
    last_action_2_in_state = np.array([-1, -1])
    last_reward_in_state = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward_1 == 1:
                pers_bonus_1[last_action_1] = p1_win
            else:
                pers_bonus_1[last_action_1] = p1_lose
            
        logits_1 = beta * q_stage1_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2_in_state[state_idx] != -1:
            if last_reward_in_state[state_idx] == 1:
                pers_bonus_2[last_action_2_in_state[state_idx]] = p2_win
            else:
                pers_bonus_2[last_action_2_in_state[state_idx]] = p2_lose
            
        logits_2 = beta * q_stage2_mf[state_idx] + pers_bonus_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_reward_1 = reward[trial]
        
        last_action_2_in_state[state_idx] = action_2[trial]
        last_reward_in_state[state_idx] = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
