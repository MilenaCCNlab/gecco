Here are three cognitive models for the two-step decision-making task. Each model builds upon the base template but introduces distinct mechanisms (e.g., eligibility traces, perseveration, asymmetric learning, and decay) to capture different nuances of participant behavior.

### Model 1: Hybrid MB/MF with Eligibility Trace and Perseveration
This model implements the standard hybrid model (Daw et al., 2011). It uses an eligibility trace (`lam`) to allow the second-stage reward to directly update the first-stage action value. It also includes a perseveration parameter (`pers`) to capture the tendency of participants to repeat their previous first-stage choice regardless of the outcome.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free learner with eligibility trace and perseveration.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for choice stochasticity.
    w: [0, 1] - Mixing weight (1 = pure model-based, 0 = pure model-free).
    lam: [0, 1] - Eligibility trace (credits stage 2 reward to stage 1 action).
    pers: [0, 5] - Perseveration bonus for repeating the previous stage 1 choice.
    """
    alpha, beta, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Handle missing data trials safely
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Blend model-based and model-free values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply perseveration bonus
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        q_net_scaled = beta * (q_net + pers_bonus)
        q_net_scaled -= np.max(q_net_scaled) # for numerical stability
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating for choice 1 (includes eligibility trace)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates (Positive vs. Negative Prediction Errors)
This model considers that participants might learn differently from positive experiences (better than expected) versus negative experiences (worse than expected). It fits two separate learning rates: `alpha_pos` and `alpha_neg`.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with asymmetric learning rates for positive vs. negative prediction errors.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (PE > 0).
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (PE <= 0).
    beta: [0, 10] - Inverse temperature for choice stochasticity.
    w: [0, 1] - Mixing weight (1 = pure model-based, 0 = pure model-free).
    """
    alpha_pos, alpha_neg, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_scaled = beta * q_net
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating for choice 1
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        # Action value updating for choice 2
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Separate Stage Temperatures and Unchosen Value Decay
This model explores whether participants evaluate the two stages with different levels of noise by fitting `beta1` and `beta2`. Further, it assumes that information about unchosen options decays over time back toward the neutral expected value of 0.5. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate temperatures for each stage and decay of unchosen values.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1 choice.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    w: [0, 1] - Mixing weight (1 = pure model-based, 0 = pure model-free).
    decay: [0, 1] - Decay rate pulling all Q-values towards 0.5 before learning occurs.
    """
    alpha, beta1, beta2, w, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize at 0.5 since rewards are binary (0 or 1)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_scaled = beta1 * q_net
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Decay all values towards neutral 0.5
        q_stage1_mf += decay * (0.5 - q_stage1_mf)
        q_stage2_mf += decay * (0.5 - q_stage2_mf)
        
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
