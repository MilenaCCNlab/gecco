def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Stage 1 and Stage 2 stickiness.
    
    This model adds a perseveration bonus not only to the first-stage choice but also 
    to the second-stage choice. This accounts for participants' tendency to repeat 
    their previous actions at both stages of the task independently of the reward outcome.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting of model-based vs model-free values.
    lambda_: [0,1] - Eligibility trace parameter.
    stickiness: [0,10] - Bonus added to the logit of the previously chosen stage 1 action.
    stickiness_stage2: [0,10] - Bonus added to the logit of the previously chosen stage 2 action in the current state.
    """
    alpha, beta, w, lambda_, stickiness, stickiness_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness / (beta + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q_stage2_net[last_action_2[state_idx]] += stickiness_stage2 / (beta + 1e-10)

        max_q2 = np.max(q_stage2_net)
        exp_q2 = np.exp(beta * (q_stage2_net - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Reward-dependent Stickiness at Stage 1.
    
    This model posits that the tendency to repeat the previous first-stage choice 
    is modulated by whether the previous trial was rewarded. This effectively integrates 
    a win-stay mechanism directly into the choice perseveration parameter.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting of model-based vs model-free values.
    lambda_: [0,1] - Eligibility trace parameter.
    stickiness: [0,10] - Baseline bonus added to the logit of the previously chosen stage 1 action.
    reward_stickiness: [0,10] - Additional bonus added if the previous trial was rewarded.
    """
    alpha, beta, w, lambda_, stickiness, reward_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += (stickiness + reward_stickiness * last_reward) / (beta + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        max_q2 = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with separate learning rates for Stage 1 and Stage 2, and Q-value decay.
    
    This model allows for different speeds of learning at the first and second stages, 
    since participants might update their model-free values at different rates depending on 
    the temporal distance from the reward. It also includes decay for unchosen options.
    
    Bounds:
    alpha_1: [0,1] - Learning rate for Stage 1 Q-values.
    alpha_2: [0,1] - Learning rate for Stage 2 Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting of model-based vs model-free values.
    lambda_: [0,1] - Eligibility trace parameter.
    stickiness: [0,10] - Bonus added to the logit of the previously chosen stage 1 action.
    decay: [0,1] - Decay rate of unchosen Q-values towards 0.5.
    """
    alpha_1, alpha_2, beta, w, lambda_, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness / (beta + 1e-10)
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        max_q2 = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2

        # decay unchosen options towards 0.5
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
