Here are 3 new cognitive models that explore different mechanisms to explain the participant's choices, based on common heuristics and biases observed in the two-step task.

### Model 1: Win-Stay / Lose-Shift Perseveration
This model separates choice perseveration at the first stage into two distinct parameters depending on the outcome of the previous trial. Participants often exhibit different tendencies to repeat an action if it was recently rewarded (win-stay) compared to when it was unrewarded (lose-shift).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Win-Stay/Lose-Shift Perseveration.
    Choice perseveration at stage 1 depends on whether the previous trial was rewarded (pers_win) or unrewarded (pers_lose).
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating Q-values
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Weighting parameter for model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers_win: [-5,5] - Choice perseveration for stage 1 if previous trial was rewarded
    pers_lose: [-5,5] - Choice perseveration for stage 1 if previous trial was unrewarded
    """
    learning_rate, beta, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_reward = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            if last_reward == 1:
                logits_1[last_action] += pers_win
            elif last_reward == 0:
                logits_1[last_action] += pers_lose
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage 1 Action Bias
This model includes a parameter to capture a baseline preference (bias) for a specific action at the first stage (e.g., heavily preferring spaceship 0 over spaceship 1). This is often observed independent of the learned values.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Stage 1 Action Bias.
    Includes a bias parameter that adds a constant to the value of action 0 at stage 1, capturing baseline preference.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating Q-values
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Weighting parameter for model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers: [-5,5] - Choice perseveration (stickiness) for stage 1
    bias_stage1: [-5,5] - Constant bias added to the logit of action 0 at stage 1
    """
    learning_rate, beta, w, lam, pers, bias_stage1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        logits_1[0] += bias_stage1
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Counterfactual Updating at Stage 2
This model tests the hypothesis that participants assume an anti-correlated reward structure for the aliens on a planet (i.e. if one alien pays out, the other probably doesn't). It updates the unchosen alien's value in the opposite direction of the received reward.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Counterfactual Updating at Stage 2.
    The unchosen action at stage 2 is updated towards the opposite of the received reward, weighted by cf_weight.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating Q-values
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Weighting parameter for model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers: [-5,5] - Choice perseveration (stickiness) for stage 1
    cf_weight: [0,1] - Weight for counterfactual updating of the unchosen alien
    """
    learning_rate, beta, w, lam, pers, cf_weight = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Counterfactual update for unchosen action at stage 2
        unchosen_2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_2]
        q_stage2_mf[state_idx, unchosen_2] += learning_rate * cf_weight * delta_cf

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
