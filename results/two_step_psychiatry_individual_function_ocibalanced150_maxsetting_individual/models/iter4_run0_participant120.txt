import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based / Model-Free with Win-Stay/Lose-Shift Perseverance
    This model separates the perseverance parameter into p_win (applied if the 
    previous trial was rewarded) and p_lose (applied if the previous trial was 
    unrewarded). This captures asymmetric stickiness based on recent reinforcement.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    lam: eligibility trace [0,1]
    w: model-based weight [0,1]
    p_win: perseverance after reward [-5,5]
    p_lose: perseverance after no reward [-5,5]
    """
    alpha, beta, lam, w, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stay_bias = np.zeros(2)
        if last_action != -1:
            if last_reward == 1:
                stay_bias[last_action] = p_win
            else:
                stay_bias[last_action] = p_lose
                
        logits_1 = beta * q_net + stay_bias
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Stage 1 and Stage 2 Perseverance
    Includes separate perseverance (stay bias) parameters for the first-stage choice 
    and the second-stage choice (specific to the planet visited). This captures 
    habitual responding that might occur at both stages of the task independently.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    lam: eligibility trace [0,1]
    w: model-based weight [0,1]
    p1: perseverance for stage 1 choice [-5,5]
    p2: perseverance for stage 2 choice [-5,5]
    """
    alpha, beta, lam, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stay_bias_1 = np.zeros(2)
        if last_action_1 != -1:
            stay_bias_1[last_action_1] = 1
                
        logits_1 = beta * q_net + p1 * stay_bias_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stay_bias_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stay_bias_2[last_action_2[state_idx]] = 1
            
        logits_2 = beta * q_stage2_mf[state_idx] + p2 * stay_bias_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Asymmetric Eligibility Trace
    Features different eligibility trace parameters for positive versus negative 
    reward prediction errors at the second stage. This allows for differential 
    credit assignment to stage 1 actions based on whether the outcome was better 
    or worse than expected.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    lam_pos: eligibility trace for positive PE [0,1]
    lam_neg: eligibility trace for negative PE [0,1]
    w: model-based weight [0,1]
    p: perseverance (stay bias) [-5,5]
    """
    alpha, beta, lam_pos, lam_neg, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stay_bias = np.zeros(2)
        if last_action != -1:
            stay_bias[last_action] = 1
                
        logits_1 = beta * q_net + p * stay_bias
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Apply asymmetric eligibility trace based on sign of PE
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
