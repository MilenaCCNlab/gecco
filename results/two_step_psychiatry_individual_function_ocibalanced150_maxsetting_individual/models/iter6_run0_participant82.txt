def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Decay to Baseline and Separate Learning Rates.
    Allows different learning rates for Stage 1 and Stage 2 to capture differences 
    in update speeds between the two stages.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for Stage 1
    alpha_2: [0, 1] - Learning rate for Stage 2
    decay_rate: [0, 1] - Decay rate for unchosen actions
    baseline: [0, 1] - Baseline value to which unchosen actions decay
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Weight on model-based vs model-free strategy
    lam: [0, 1] - Eligibility trace parameter
    """
    alpha_1, alpha_2, decay_rate, baseline, beta_1, beta_2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * baseline
    q_stage2_mf = np.ones((2, 2)) * baseline

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta_1 * q_net - np.max(beta_1 * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (baseline - q_stage1_mf[unchosen_a1])

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (baseline - q_stage2_mf[s, a])

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Decay to Baseline and Reward Sensitivity.
    Includes a reward sensitivity parameter `rho` that scales the objective reward.
    This dynamically changes the asymptote of chosen Q-values, creating a unique 
    contrast against the unchosen actions decaying to the baseline.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate
    decay_rate: [0, 1] - Decay rate for unchosen actions
    baseline: [0, 1] - Baseline value to which unchosen actions decay
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Weight on model-based vs model-free strategy
    lam: [0, 1] - Eligibility trace parameter
    rho: [0, 10] - Reward sensitivity scaling factor

    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * baseline
    q_stage2_mf = np.ones((2, 2)) * baseline

    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):

    Model 3: Hybrid Model with Decay to Baseline and Asymmetric Eligibility Traces.
