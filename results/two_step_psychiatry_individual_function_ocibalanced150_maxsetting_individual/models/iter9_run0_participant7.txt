Here are 3 new cognitive models that build upon the best model by testing novel hypotheses about participant behavior, while avoiding previously tried parameter combinations. Each model is constrained to 8 parameters to remain parsimonious.

### Model 1: Stage 2 Perseveration
This model adds a `rep2` parameter to test the hypothesis that the participant exhibits perseveration in Stage 2. That is, if they arrive at the exact same planet (state) as the previous trial, they might have a tendency to stick to the same alien (Stage 2 action), regardless of the reward received.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate learning rates for Stage 1 and Stage 2,
    shared beta, eligibility trace, win/lose perseveration for Stage 1,
    and a Stage 2 perseveration parameter.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta: shared inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: stage 1 perseveration if previous trial was rewarded [-5,5]
    pers_lose: stage 1 perseveration if previous trial was unrewarded [-5,5]
    rep2: stage 2 perseveration bonus for repeating the same stage 2 action [-5,5]
    """
    alpha1, alpha2, beta, w, lam, pers_win, pers_lose, rep2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1
    last_state = -1
    last_reward = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action1 >= 0:
            if last_reward == 1:
                q_net1[last_action1] += pers_win
            elif last_reward == 0:
                q_net1[last_action1] += pers_lose
                
        exp_q1 = np.exp(beta * (q_net1 - np.max(q_net1)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q_net2 = np.copy(q_stage2_mf[state_idx])
        # Apply Stage 2 perseveration if we are in the same state as the previous trial
        if last_state == state_idx and last_action2 >= 0:
            q_net2[last_action2] += rep2
            
        exp_q2 = np.exp(beta * (q_net2 - np.max(q_net2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Choice Kernel for Stage 1
Instead of simple "last-trial" perseveration, this model introduces a full *choice kernel* for Stage 1. This tracks the participant's historical tendency to choose a particular spaceship over multiple trials, decaying slowly over time.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with a Choice Kernel for stage 1 actions.
    The choice kernel tracks the tendency to repeat actions over multiple trials.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    ck_alpha: choice kernel learning rate [0,1]
    ck_weight: choice kernel weight [-5,5]
    """
    alpha1, alpha2, beta1, beta2, w, lam, ck_alpha, ck_weight = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck = np.zeros(2)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate choice kernel into Stage 1 decision
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf + ck_weight * ck
        
        exp_q1 = np.exp(beta1 * (q_net1 - np.max(q_net1)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q_net2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q_net2 - np.max(q_net2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Update choice kernel
        ck = (1 - ck_alpha) * ck
        ck[action_1[trial]] += ck_alpha
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Model-Based Weight
A common phenomenon in the two-step task is that participants often "drop" or reduce model-based control immediately following a rare transition. This model tests this hypothesis by fitting separate model-based weights (`w_common` vs `w_rare`) depending on whether the *previous* trial experienced a common or rare transition.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate MB weights based on the previous trial's transition type.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta: shared inverse temperature [0,10]
    w_common: MB weight if previous transition was common [0,1]
    w_rare: MB weight if previous transition was rare [0,1]
    lam: eligibility trace [0,1]
    pers_win: stage 1 perseveration if previous trial was rewarded [-5,5]
    pers_lose: stage 1 perseveration if previous trial was unrewarded [-5,5]
    """
    alpha1, alpha2, beta, w_common, w_rare, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1
    last_trans_common = True # Default for the first trial

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Apply the transition-dependent MB weight
        w = w_common if last_trans_common else w_rare
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose
                
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q_net2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta * (q_net2 - np.max(q_net2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Determine if the current transition was common (Action 0 -> State 0 OR Action 1 -> State 1)
        is_common = (action_1[trial] == state_idx)
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]
        last_trans_common = is_common
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
