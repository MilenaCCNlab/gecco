Here are 3 new cognitive models that build upon the baseline and best-performing models to explore distinct psychological mechanisms:

### Model 1: Full Daw 2011 Model (Separate Betas & Eligibility Trace)
This model introduces separate inverse temperatures (`beta1` and `beta2`) for the two stages, allowing for the possibility that decision noise or exploration differs between the first (spaceship) and second (alien) choices. It also explicitly includes the eligibility trace parameter (`lam`), representing how much the second-stage reward directly reinforces the first-stage choice.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Full Daw 2011 Model with separate betas.
    Includes separate inverse temperatures for stage 1 and stage 2, eligibility trace, and choice stickiness.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    lam: [0, 1] - Eligibility trace
    w: [0, 1] - Model-based weight
    p: [0, 10] - Choice stickiness (perseveration)
    """
    import numpy as np
    alpha, beta1, beta2, lam, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        # Policy for choice 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = p
            
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for choice 2
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

It parameterizes `alpha1` (learning rate for spaceships) and `alpha2` (learning rate for aliens) separately, accommodating differences in how quickly participants update values for transition structures vs. terminal rewards.

### Model 3: Forgetting / Decay Model
This model introduces a `decay` parameter. In dynamically changing environments, participants may forget the values of unchosen options over time. This model assumes that on every trial, the expected value of all stage 2 aliens decays towards a neutral expected value (0.5), pushing participants to explore options they haven't chosen in a while.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Forgetting / Decay Model.
    Unchosen actions in stage 2 decay towards 0.5. Includes stickiness and MB/MF hybrid learning.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta: [0, 10] - Inverse temperature
    w: [0, 1] - Model-based weight
    p: [0, 10] - Choice stickiness
    decay: [0, 1] - Decay rate for unchosen options
    """
    import numpy as np
    alpha, beta, w, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2)) + 0.5 # Initialized at 0.5 for decay target
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        # Policy for choice 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = p
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for choice 2
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        # Decay all stage 2 values towards 0.5
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)
        
        # Update chosen option
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Implicit trace update for stage 1
        q_stage1_mf[a1] += alpha * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
