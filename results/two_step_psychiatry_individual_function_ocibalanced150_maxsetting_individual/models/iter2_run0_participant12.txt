Here are 3 new cognitive models that explore different mechanisms to explain participant behavior in the two-step task. 

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
It is common for learning to occur at different speeds for different stages of a task. This model extends the baseline hybrid model by disentangling the learning rate for the first stage (spaceship selection) from the second stage (alien selection).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF RL with Separate Learning Rates for Stage 1 and Stage 2.
    Incorporates choice perseveration and an eligibility trace, but allows the 
    learning speed to differ between the two stages of the task.
    
    Parameters:
    lr_stage1: [0,1] - learning rate for stage 1 Q-values
    lr_stage2: [0,1] - learning rate for stage 2 Q-values
    beta: [0,10] - inverse temperature for softmax choice (shared)
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    """
    lr_stage1, lr_stage2, beta, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] = pers
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # Stage 2 policy
        state_idx = state[trial]
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        # Eligibility trace update using Stage 1 learning rate
        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unchosen Action Decay
Participants might naturally "forget" the value of unchosen options over time, causing those values to regress toward a neutral expectation. This model introduces a `decay` parameter that pulls the Q-values of all unchosen options towards 0.5.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF RL with Unchosen Action Decay.
    Includes choice perseveration and an eligibility trace. Additionally, Q-values 
    of unchosen actions decay towards a neutral expected value of 0.5.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating chosen Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    decay: [0,1] - rate at which unchosen Q-values decay towards 0.5
    """
    learning_rate, beta, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    # Initialize values at neutral 0.5
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] = pers
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # Stage 2 policy
        state_idx = state[trial]
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Value updating for chosen actions
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen actions towards 0.5
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition Matrix Learning
Instead of assuming that participants fully trust the fixed task instructions (`0.7` and `0.3` probabilities), this model assumes they continually dynamically update their internal representation of the transition matrix based on the outcomes they experience.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF RL with Transition Matrix Learning.
    Instead of assuming a fixed transition matrix, the agent learns the transition 
    probabilities from experience.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    lr_trans: [0,1] - learning rate for updating the transition matrix
    """
    learning_rate, beta, w, lam, pers, lr_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix neutrally
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] = pers
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # Stage 2 policy
        state_idx = state[trial]
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update Transition Matrix based on observation
        a1 = action_1[trial]
        target_trans = np.zeros(2)
        target_trans[state_idx] = 1.0  # 1 for observed state, 0 for the other
        transition_matrix[a1] += lr_trans * (target_trans - transition_matrix[a1])
        
        # Value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
