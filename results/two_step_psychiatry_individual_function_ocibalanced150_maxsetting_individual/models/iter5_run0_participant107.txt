def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates and a penalty parameter for zero rewards.
    Zero rewards are replaced with a negative penalty to amplify the effect of unrewarded trials.
    This could explain why a participant might avoid an option after receiving 0 coins,
    treating it actively as a loss rather than just neutral.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Choice stickiness for stage 1.
    penalty: [0, 5] - Negative reward assigned when 0 coins are received.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pi, penalty = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits1 = beta1 * q_net_stage1 + pi * stickiness
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Apply penalty for zero rewards
        r_eff = r if r > 0 else -penalty
        
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        delta_stage2 = r_eff - q_stage2_mf[s, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha2 * lam * delta_stage2
        
        last_action_1 = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with symmetric learning rate at stage 1 and asymmetric learning rates at stage 2.
    This model assumes that model-free learning about transitions (stage 1) is symmetric, but 
    learning directly from rewards (stage 2) has separate rates for positive and negative prediction errors.
    
    Parameters:
    alpha1: [0, 1] - Symmetric learning rate for stage 1 prediction errors.
    alpha2_pos: [0, 1] - Learning rate for positive prediction errors at stage 2.
    alpha2_neg: [0, 1] - Learning rate for negative prediction errors at stage 2.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Choice stickiness for stage 1.
    """
    alpha1, alpha2_pos, alpha2_neg, beta1, beta2, w, lam, pi = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits1 = beta1 * q_net_stage1 + pi * stickiness
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha2 = alpha2_pos if delta_stage2 > 0 else alpha2_neg
        q_stage2_mf[s, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha2 * lam * delta_stage2
        
        last_action_1 = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates where the model-based weight depends on the previous trial's reward.
    This tests if the participant changes their reliance on the transition model after winning versus losing.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w_win: [0, 1] - Model-based weight after a rewarded trial.
    w_lose: [0, 1] - Model-based weight after an unrewarded trial.
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Choice stickiness for stage 1.
    """
    alpha_pos, alpha_neg, beta1, beta2, w_win, w_lose, lam, pi = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        if last_reward == 1:
            w_current = w_win
        elif last_reward == 0:
            w_current = w_lose
        else:
            w_current = (w_win + w_lose) / 2.0

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w_current * q_stage1_mb + (1 - w_current) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits1 = beta1 * q_net_stage1 + pi * stickiness
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha2 * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = r
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
