Here are three cognitive models implemented as Python functions to explain the participant's behavior in the two-step task. 

### Model 1: Model-Free SARSA($\lambda$)
This model assumes the participant is purely model-free. It ignores the transition structure of the task and updates the value of the first-stage choices based on the observed second-stage values and rewards. It uses an eligibility trace ($\lambda$) to propagate the second-stage reward back to the first-stage choice in the same trial.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model-Free SARSA(lambda) model.
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    lam: [0,1]
    """
    learning_rate, beta, lam = model_parameters
    n_trials = len(action_1)
  
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Handle missing data
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q1 = np.max(q_stage1_mf)
        exp_q1 = np.exp(beta * (q_stage1_mf - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        max_q2 = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1 (SARSA)
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # apply eligibility trace to choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Pure Model-Based Q-learning
This model assumes the participant is purely model-based. They fully understand the transition matrix and evaluate the first-stage choices by looking ahead to the maximum expected value of the second-stage planets they are likely to land on.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model-Based Q-learning model.
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    """
    learning_rate, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        max_q1 = np.max(q_stage1_mb)
        exp_q1 = np.exp(beta * (q_stage1_mb - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        max_q2 = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1 (MF values tracked but not used for policy)
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid MB/MF Model with Perseveration (Daw et al. 2011 style)
This is the standard model used to analyze the two-step task. It assumes the participant's choices are a weighted mixture (`w`) of both Model-Based and Model-Free strategies. It also includes separate temperature parameters for each stage and a perseveration parameter (`p_pers`) capturing the participant's tendency to repeat their previous first-stage choice regardless of reward.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Hybrid Model-Based / Model-Free model with perseveration.
    Bounds:
    learning_rate: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    lam: [0,1]
    w: [0,1]
    p_pers: [-5,5]
    """
    learning_rate, beta_1, beta_2, lam, w, p_pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice (weighted combination of MB and MF)
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent = beta_1 * q_stage1_net
        if last_action != -1:
            exponent[last_action] += p_pers
            
        max_exp = np.max(exponent)
        exp_q1 = np.exp(exponent - max_exp)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        max_q2 = np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta_2 * (q_stage2_mf[state_idx] - max_q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # apply eligibility trace to choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
