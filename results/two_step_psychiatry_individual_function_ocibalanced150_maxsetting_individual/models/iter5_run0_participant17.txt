Here are 3 new cognitive models that build upon the best model so far by introducing new mechanisms and parameter configurations not yet explored.

### Model 1: Hybrid Model with Decay and Stage-Specific Inverse Temperatures
This model extends the best decay model by allowing separate inverse temperatures (`beta1` and `beta2`) for stage 1 and stage 2. This accounts for the possibility that the participant's choice stochasticity (or exploration/exploitation trade-off) differs between the spaceship choice and the alien choice.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with decay for unchosen actions and separate stage betas.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen action Q-value updates.
    decay: [0, 1] - Decay rate for unchosen action Q-values toward 0.
    beta1: [0, 10] - Inverse temperature for stage 1 softmax action selection.
    beta2: [0, 10] - Inverse temperature for stage 2 softmax action selection.
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter for updating stage 1 with stage 2 RPE.
    """
    learning_rate, decay, beta1, beta2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay for unchosen action in stage 1
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] -= decay * q_stage1_mf[unchosen_1]
        
        # Decay for all unchosen actions in stage 2
        for state_i in range(2):
            for action_j in range(2):
                if not (state_i == state_idx and action_j == action_2[trial]):
                    q_stage2_mf[state_i, action_j] -= decay * q_stage2_mf[state_i, action_j]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Stage-Specific Inverse Temperatures
This model removes the decay parameter but splits the learning rate into positive (`lr_pos`) and negative (`lr_neg`) update rates. This captures the psychological phenomenon where individuals learn differently from positive prediction errors (better-than-expected outcomes) compared to negative prediction errors (worse-than-expected outcomes). It also utilizes separate betas.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with asymmetric learning rates for positive vs. negative RPEs,
    and separate inverse temperatures for stage 1 and stage 2.
    
    Parameters:
    lr_pos: [0, 1] - Learning rate for positive prediction errors (RPE >= 0).
    lr_neg: [0, 1] - Learning rate for negative prediction errors (RPE < 0).
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    """
    lr_pos, lr_neg, beta1, beta2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        # Stage 2 MF update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += lr_stage2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid Model with Stage-Specific Perseveration Bonuses
Instead of using parameter decay, this model incorporates explicit perseveration parameters for both stage 1 (`pers1`) and stage 2 (`pers2`). Perseveration is added directly to the log-odds of the action taken on the previous trial, capturing choice "stickiness" or motor-level repetition habits independent of reward histories.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate perseveration bonuses for stage 1 and stage 2 choices.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection (shared).
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    pers1: [-5, 5] - Perseveration (stickiness) bonus for stage 1 actions.
    pers2: [-5, 5] - Perseveration (stickiness) bonus for stage 2 actions.
    """
    learning_rate, beta, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus1 = np.zeros(2)
        if prev_a1 != -1:
            pers_bonus1[prev_a1] = pers1
            
        exp_q1 = np.exp(beta * q_stage1_net + pers_bonus1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        pers_bonus2 = np.zeros(2)
        if prev_a2 != -1:
            pers_bonus2[prev_a2] = pers2
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pers_bonus2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Store previous actions for perseveration on next trial
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
