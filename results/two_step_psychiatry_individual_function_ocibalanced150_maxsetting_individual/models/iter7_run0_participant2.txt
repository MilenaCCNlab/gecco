Here are 3 new cognitive models that build upon the best-performing mechanisms so far, while exploring novel ways to capture participant behavior in the two-step task.

### Model 1: Transition Learning
While the standard Daw et al. model assumes participants use fixed 0.7/0.3 transition probabilities, participants often continuously update their internal model of the environment. This model dynamically learns the transition matrix using a separate learning rate (`alpha_tr`) while maintaining state-specific perseveration at Stage 2.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Transition Learning and State-Specific Perseveration.
    Instead of assuming fixed 0.7/0.3 transition probabilities throughout the task, 
    the model continuously updates its internal transition matrix using alpha_tr.
    
    Bounds:
    alpha: [0,1] - Learning rate for Q-values
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Model-based vs Model-free weight
    lam: [0,1] - Eligibility trace parameter
    pers_1: [-5,5] - Choice perseveration bonus for stage 1
    pers_2_state: [-5,5] - State-specific choice perseveration bonus for stage 2
    alpha_tr: [0,1] - Learning rate for transition probabilities
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2_state, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize with the instructed probabilities but allow them to be updated
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    last_a2_in_state = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if prev_a1 != -1:
            pers_bonus_1[prev_a1] = pers_1
            
        logits_1 = beta_1 * q_stage1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_a2_in_state[state_idx] != -1:
            pers_bonus_2[last_a2_in_state[state_idx]] = pers_2_state

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update transition matrix based on observed transition
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1 - transition_matrix[action_1[trial], state_idx]
        
        prev_a1 = action_1[trial]
        last_a2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Stage 1 Perseveration (Win-Stay / Lose-Shift)
This model splits Stage 1 perseveration into two separate parameters based on whether the previous trial was rewarded or not. This directly models heuristic win-stay/lose-shift choice tendencies independently of Q-value updates.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Reward-Dependent Stage 1 Perseveration.
    This model features separate perseveration parameters for Stage 1 depending on whether 
    the previous trial was rewarded (pers_1_win) or unrewarded (pers_1_lose).
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Model-based vs Model-free weight
    lam: [0,1] - Eligibility trace parameter
    pers_1_win: [-5,5] - Choice perseveration if previous trial was rewarded
    pers_1_lose: [-5,5] - Choice perseveration if previous trial was unrewarded
    pers_2_state: [-5,5] - State-specific choice perseveration for stage 2
    """
    alpha, beta_1, beta_2, w, lam, pers_1_win, pers_1_lose, pers_2_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_reward = -1
    last_a2_in_state = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_reward = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if prev_a1 != -1 and prev_reward != -1:
            if prev_reward == 1:
                pers_bonus_1[prev_a1] = pers_1_win
            else:
                pers_bonus_1[prev_a1] = pers_1_lose
            
        logits_1 = beta_1 * q_stage1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_a2_in_state[state_idx] != -1:
            pers_bonus_2[last_a2_in_state[state_idx]] = pers_2_state

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_reward = reward[trial]
        last_a2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Stage 1 Perseveration
Participants often react strongly to rare transitions, sometimes abandoning their Stage 1 choice purely because the transition was unexpected, regardless of the reward outcome. This model splits Stage 1 perseveration based on whether the previous transition was common or rare.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Transition-Dependent Stage 1 Perseveration.
    Stage 1 perseveration is split into two parameters based on whether the transition 
    on the previous trial was common (pers_1_common) or rare (pers_1_rare).
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Model-based vs Model-free weight
    lam: [0,1] - Eligibility trace parameter
    pers_1_common: [-5,5] - Choice perseveration if previous transition was common
    pers_1_rare: [-5,5] - Choice perseveration if previous transition was rare
    pers_2_state: [-5,5] - State-specific choice perseveration for stage 2
    """
    alpha, beta_1, beta_2, w, lam, pers_1_common, pers_1_rare, pers_2_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_trans_common = False
    last_a2_in_state = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_trans_common:
                pers_bonus_1[prev_a1] = pers_1_common
            else:
                pers_bonus_1[prev_a1] = pers_1_rare
            
        logits_1 = beta_1 * q_stage1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_a2_in_state[state_idx] != -1:
            pers_bonus_2[last_a2_in_state[state_idx]] = pers_2_state

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        # In this task structure, action 0 -> state 0 and action 1 -> state 1 are the common transitions
        prev_trans_common = (action_1[trial] == state_idx)
        last_a2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
