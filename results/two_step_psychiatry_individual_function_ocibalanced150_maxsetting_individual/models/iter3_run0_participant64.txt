def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF hybrid with both choice stickiness and reward-dependent stickiness for stage 1.
    This captures the tendency to repeat the previous stage 1 action generally, 
    as well as the specific tendency to repeat it if it was rewarded on the previous trial.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p_c: choice stickiness [-5,5]
    p_r: reward stickiness [-5,5]
    """
    alpha, beta, w, lam, p_c, p_r = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        rep_rew = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            if last_reward > 0:
                rep_rew[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p_c * rep + p_r * rep_rew
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]
        r = reward[trial]

        # policy for the second choice
        if act2 != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            # action value updating for choice 1
            q_stage1_mf[act1] += alpha * delta_stage1

            delta_stage2 = r - q_stage2_mf[state_idx, act2]
            # action value updating for choice 2
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            q_stage1_mf[act1] += alpha * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF hybrid with choice stickiness and a reward sensitivity (scaling) parameter.
    This model allows the subjective value of the reward to be scaled by rho, 
    accommodating individual differences in reward valuation.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness [-5,5]
    rho: reward sensitivity [0,10]
    """
    alpha, beta, w, lam, p, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]
        r = reward[trial]

        # policy for the second choice
        if act2 != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            # action value updating for choice 1
            q_stage1_mf[act1] += alpha * delta_stage1

            delta_stage2 = (rho * r) - q_stage2_mf[state_idx, act2]
            # action value updating for choice 2
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            q_stage1_mf[act1] += alpha * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF hybrid with separate learning rates for positive and negative prediction errors,
    separate inverse temperatures for stage 1 and stage 2, and choice stickiness.
    
    Parameters:
    alpha_pos: learning rate for positive PE [0,1]
    alpha_neg: learning rate for negative PE [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness [-5,5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta1 * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]
        r = reward[trial]

        # policy for the second choice
        if act2 != -1:
            logits_2 = beta2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            alpha_stage1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
            # action value updating for choice 1
            q_stage1_mf[act1] += alpha_stage1 * delta_stage1

            delta_stage2 = r - q_stage2_mf[state_idx, act2]
            alpha_stage2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
            # action value updating for choice 2
            q_stage2_mf[state_idx, act2] += alpha_stage2 * delta_stage2

            q_stage1_mf[act1] += alpha_stage2 * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
