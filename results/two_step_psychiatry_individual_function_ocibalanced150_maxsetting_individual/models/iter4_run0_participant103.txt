def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Forgetting (Decay) towards a baseline q0 and Stage 1 Stickiness.
    Instead of decaying Q-values towards 0, this model decays them towards an initial 
    baseline value (q0), which can better capture regression to the mean in environments 
    with fluctuating reward probabilities.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Forgetting rate for all Q-values per trial [0, 1]
    st: Choice stickiness for stage 1 [-10, 10]
    q0: Initial Q-value and decay target [0, 1]
    """
    alpha, beta1, beta2, w, lam, decay, st, q0 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to q0
    q_stage1_mf = np.full(2, q0, dtype=float)
    q_stage2_mf = np.full((2, 2), q0, dtype=float)
    
    prev_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net
        if prev_a1 != -1:
            logit1[prev_a1] += st
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay Q-values towards q0
        q_stage1_mf = q_stage1_mf * (1 - decay) + q0 * decay
        q_stage2_mf = q_stage2_mf * (1 - decay) + q0 * decay
        
        prev_a1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Forgetting (Decay) and Stage 1 Choice Trace (Autocorrelation).
    Instead of only sticking to the immediate last choice, this model maintains an exponentially 
    decaying trace of past choices to capture longer-term perseveration and choice bursts.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Forgetting rate for all Q-values per trial [0, 1]
    st: Choice stickiness weight for stage 1 [-10, 10]
    tau: Choice trace decay rate [0, 1]
    """
    alpha, beta1, beta2, w, lam, decay, st, tau = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net + st * choice_trace
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay Q-values
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
        # update choice trace
        choice_trace *= tau
        choice_trace[action_1[trial]] += 1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Forgetting (Decay), Stage 1 Stickiness, and Counterfactual Updating.
    Updates the unchosen action at stage 2 assuming its outcome would have been the opposite 
    of the received reward (e.g., if chosen gave 1, unchosen might have given 0).
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Forgetting rate for all Q-values per trial [0, 1]
    st: Choice stickiness for stage 1 [-10, 10]
    cf_weight: Weight of counterfactual update for unchosen option [0, 1]
    """
    alpha, beta1, beta2, w, lam, decay, st, cf_weight = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net
        if prev_a1 != -1:
            logit1[prev_a1] += st
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # counterfactual update for unchosen action at stage 2
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += alpha * cf_weight * delta_cf

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay Q-values
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
        prev_a1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
