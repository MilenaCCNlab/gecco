Here are 3 new cognitive models that build upon the insights from the best-performing model so far. Given that the participant exhibits a very strong "action matching" tendency (almost always choosing the same action index in stage 2 as they did in stage 1), all models retain the `theta` parameter. We explore additional mechanisms such as value decay, asymmetric learning from positive/negative prediction errors, and dual perseverance.

### Model 1: Separate LRs and Betas with Action Matching and Value Decay
This model tests the hypothesis that the participant's values might naturally decay toward a neutral baseline (0.5) over time, which can capture forgetting of unchosen options. It also uses separate learning rates and inverse temperatures for both stages to allow maximum flexibility in how values are updated and translated into choices.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate LR and Beta with Action Matching and Decay
    Includes separate learning rates and inverse temperatures for both stages,
    model-based weight, eligibility trace, action matching bias, and a decay 
    parameter for all Q-values towards a neutral 0.5 baseline.
    
    Parameters:
    alpha_1: stage 1 learning rate [0,1]
    alpha_2: stage 2 learning rate [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    decay: decay rate of Q-values towards 0.5 [0,1]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, theta, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 to match the neutral decay baseline
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta_1 * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx].copy()
        q2[action_1[trial]] += theta  # apply action matching bias
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta_2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        # Decay all Q-values towards 0.5 at the end of the trial
        q_stage1_mf += decay * (0.5 - q_stage1_mf)
        q_stage2_mf += decay * (0.5 - q_stage2_mf)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning with Action Matching and Perseverance
This model assumes that the participant updates values differently depending on whether the prediction error is positive (better than expected) or negative (worse than expected). It also includes a stage 1 perseverance parameter to capture the consecutive streaks of stage 1 choices observed in the data.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning with Action Matching and Perseverance
    Uses separate learning rates for positive and negative prediction errors,
    separate inverse temperatures for each stage, MB weight, eligibility trace,
    action matching bias at stage 2, and stage 1 choice perseverance.
    
    Parameters:
    alpha_pos: learning rate for positive PE [0,1]
    alpha_neg: learning rate for negative PE [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    pers: stage 1 perseverance [-5, 5]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, theta, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_net[last_action_1] += pers
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta_1 * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx].copy()
        q2[action_1[trial]] += theta  # apply action matching bias
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta_2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_s1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Separate LRs with Action Matching and Dual Perseverance
This model extends the perseverance mechanic to both stages. While `theta` captures within-trial action matching (stage 1 influencing stage 2), `pers_2` captures between-trial state-specific perseverance (repeating the same action the last time the participant visited that specific planet).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate LRs with Action Matching and Dual Perseverance
    Includes separate learning rates for stage 1 and 2, a shared inverse temperature,
    MB weight, trace, action matching bias, stage 1 perseverance, and stage 2 
    state-specific perseverance.
    
    Parameters:
    alpha_1: stage 1 learning rate [0,1]
    alpha_2: stage 2 learning rate [0,1]
    beta: shared inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    pers_1: stage 1 perseverance [-5, 5]
    pers_2: stage 2 state-specific perseverance [-5, 5]
    """
    alpha_1, alpha_2, beta, w, lam, theta, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_net[last_action_1] += pers_1
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx].copy()
        q2[action_1[trial]] += theta  # apply action matching bias
        
        if last_action_2_in_state[state_idx] != -1:
            q2[last_action_2_in_state[state_idx]] += pers_2
            
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
