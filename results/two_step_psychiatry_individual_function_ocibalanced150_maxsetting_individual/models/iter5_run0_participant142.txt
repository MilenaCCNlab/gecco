def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate learning rates for stage 1 and stage 2,
    decay mechanism for unchosen options, and separate MB/MF weights.
    
    Bounds:
    alpha1: [0,1] - learning rate for stage 1
    alpha2: [0,1] - learning rate for stage 2
    decay: [0,1] - decay rate towards 0 for unchosen options
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - choice perseverance for stage 1
    """
    alpha1, alpha2, decay, beta_mb, beta_mf, beta2, lam, pers1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exponent2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.0 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.0 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate MB and MF inverse temperatures at stage 1, 
    and a decay mechanism for unchosen options towards 0.5 (neutral value).
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    decay: [0,1] - decay rate towards 0.5 for unchosen options
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - choice perseverance for stage 1
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha, decay, beta_mb, beta_mf, beta2, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values at 0.5
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_stage2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen options towards 0.5
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with decay mechanism for unchosen options, separate MB/MF weights,
    and reward-dependent choice perseverance (win-stay/lose-shift) at stage 1.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    decay: [0,1] - decay rate towards 0 for unchosen options
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers_win: [-5,5] - choice perseverance for stage 1 after a win
    pers_lose: [-5,5] - choice perseverance for stage 1 after a loss
    """
    alpha, decay, beta_mb, beta_mf, beta2, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            if reward[trial-1] > 0:
                exponent1[action_1[trial-1]] += pers_win
            else:
                exponent1[action_1[trial-1]] += pers_lose
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exponent2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.0 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.0 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
