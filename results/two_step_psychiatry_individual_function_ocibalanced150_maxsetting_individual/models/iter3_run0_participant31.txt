def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw Two-Step Model with asymmetric learning rates (positive/negative RPE), choice perseveration, and unchosen action value decay.
    
    Parameters:
    alpha_pos: learning rate for positive RPE [0,1]
    alpha_neg: learning rate for negative RPE [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers: perseveration bonus for stage 1 [0,10]
    decay: unchosen action value decay rate [0,1]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = pers
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + pers_bonus)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta_2 * q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_s1 * lam * delta_stage2

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1.0 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1.0 - decay)

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw Two-Step Model with stage 1 and stage 2 choice perseveration, and unchosen action value decay.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_1: perseveration bonus for stage 1 [0,10]
    pers_2: perseveration bonus for stage 2 [0,10]
    decay: unchosen action value decay rate [0,1]
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2

        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta_2 * q2 + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1.0 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1.0 - decay)

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
