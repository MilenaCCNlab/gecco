def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 1: Full Daw et al. model with an action bias for Stage 1.
    This bias captures the participant's strong behavioral preference for spaceship 0, 
    as observed in the data.
    
    Parameters and Bounds:
    alpha: [0, 1] - Learning rate for Q-value updates.
    beta_1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
    stickiness: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    lambda_: [0, 1] - Eligibility trace for updating Stage 1 from Stage 2 reward.
    bias_1: [-5, 5] - Constant bias added to the value of choosing spaceship 0 in Stage 1.
    """
    alpha, beta_1, beta_2, w, stickiness, lambda_, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus = np.zeros(2)
        if last_action_1 != -1:
            bonus[last_action_1] = stickiness
        bonus[0] += bias_1
        
        q1_total = beta_1 * q_stage1_net + bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 3: Asymmetric learning rates for positive and negative prediction errors, 
    combined with an action bias for Stage 1 to capture the strong preference for spaceship 0.
    
    Parameters and Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
    stickiness: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    lambda_: [0, 1] - Eligibility trace.
    bias_1: [-5, 5] - Constant bias added to the value of choosing spaceship 0 in Stage 1.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, stickiness, lambda_, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus = np.zeros(2)
        if last_action_1 != -1:

        bonus[0] += bias_1

        max_q1 = np.max(q1_total)

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10

    return log_loss
