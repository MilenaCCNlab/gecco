### Model 1: Separate Decay Rates for Stage 1 and Stage 2
It is possible that participants forget or discount unchosen options at different rates depending on the decision stage. Stage 1 options might decay differently than Stage 2 options, given the hierarchical nature of the task.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Separate Unchosen Option Decay for Stage 1 and Stage 2.
    Includes model-based/model-free weighting, eligibility trace, perseverance,
    and separate decay parameters for unchosen action values in Stage 1 vs Stage 2.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: weight of model-based vs model-free control [0, 1]
    lam: eligibility trace parameter [0, 1]
    p: choice perseverance [-5, 5]
    decay1: forgetting rate for Stage 1 unchosen options [0, 1]
    decay2: forgetting rate for Stage 2 unchosen options [0, 1]
    """
    alpha, beta, w, lam, p, decay1, decay2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        exp_q1 = np.exp(beta * q_stage1_net + p * rep)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay1)

        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay2)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay2)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay2)

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Decay Towards a Central Value
Instead of unchosen Q-values decaying towards zero, participants might regress unchosen options towards a specific "prior" or central expectation (like 0.5, representing a 50% chance of reward). This model introduces a `center` parameter as the decay target.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Decay towards a central value.
    Includes model-based/model-free weighting, eligibility trace, perseverance,
    and a decay parameter for unchosen action values to decay towards a specific target value.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: weight of model-based vs model-free control [0, 1]
    lam: eligibility trace parameter [0, 1]
    p: choice perseverance [-5, 5]
    decay: forgetting rate for unchosen options [0, 1]
    center: target expected value for decay (e.g., 0.5) [0, 1]
    """
    alpha, beta, w, lam, p, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to the central value
    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)
    last_action = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        exp_q1 = np.exp(beta * q_stage1_net + p * rep)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] = (1 - decay) * q_stage1_mf[1 - action_1[trial]] + decay * center

        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage2_mf[state_idx, 1 - action_2[trial]] = (1 - decay) * q_stage2_mf[state_idx, 1 - action_2[trial]] + decay * center
        q_stage2_mf[1 - state_idx, 0] = (1 - decay) * q_stage2_mf[1 - state_idx, 0] + decay * center
        q_stage2_mf[1 - state_idx, 1] = (1 - decay) * q_stage2_mf[1 - state_idx, 1] + decay * center

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Asymmetric Eligibility Trace Update
This model introduces asymmetric credit assignment back to Stage 1. Participants may be more likely to propagate positive reward prediction errors (unexpected rewards) back to their first-stage choices than negative prediction errors (unexpected omissions).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Asymmetric Eligibility Trace with Decay.
    Includes model-based/model-free weighting, perseverance, unchosen decay,
    and distinct eligibility traces for positive vs negative reward prediction errors.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: weight of model-based vs model-free control [0, 1]
    lam_pos: eligibility trace for positive PE [0, 1]
    lam_neg: eligibility trace for negative PE [0, 1]
    p: choice perseverance [-5, 5]
    decay: forgetting rate for unchosen options [0, 1]
    """
    alpha, beta, w, lam_pos, lam_neg, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        exp_q1 = np.exp(beta * q_stage1_net + p * rep)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)

        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

        # Asymmetric Eligibility trace update for Stage 1
        if delta_stage2 > 0:
            q_stage1_mf[action_1[trial]] += alpha * lam_pos * delta_stage2
        else:
            q_stage1_mf[action_1[trial]] += alpha * lam_neg * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
