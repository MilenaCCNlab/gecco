import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dynamic Baseline Model
    Instead of a fixed baseline, unchosen actions decay towards a dynamic baseline 
    that tracks the average reward rate over time. This reflects a shifting reference 
    point based on the general richness of the environment.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for action values
    decay_rate: [0, 1] - Decay rate for unchosen actions
    alpha_rho: [0, 1] - Learning rate for the dynamic baseline (average reward)
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Weight on model-based vs model-free strategy
    lam: [0, 1] - Eligibility trace parameter
    """
    learning_rate, decay_rate, alpha_rho, beta_1, beta_2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    baseline = 0.5
    q_stage1_mf = np.ones(2) * baseline
    q_stage2_mf = np.ones((2, 2)) * baseline

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta_1 * q_net - np.max(beta_1 * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (baseline - q_stage1_mf[unchosen_a1])

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (baseline - q_stage2_mf[s, a])

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # update dynamic baseline
        baseline += alpha_rho * (reward[trial] - baseline)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Q-learning for Stage 1 MF update with Baseline Decay
    Instead of using the value of the chosen Stage 2 action (SARSA), Stage 1 model-free 
    values are updated using the maximum Q-value of Stage 2 (Q-learning). This reflects 
    an assumption that participants evaluate Stage 1 based on the best possible outcome in Stage 2.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate
    decay_rate: [0, 1] - Decay rate for unchosen actions
    baseline: [0, 1] - Baseline value to which unchosen actions decay
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Weight on model-based vs model-free strategy
    lam: [0, 1] - Eligibility trace parameter
    """
    learning_rate, decay_rate, baseline, beta_1, beta_2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * baseline
    q_stage2_mf = np.ones((2, 2)) * baseline

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta_1 * q_net - np.max(beta_1 * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1 (Q-learning update)
        delta_stage1 = np.max(q_stage2_mf[state_idx]) - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (baseline - q_stage1_mf[unchosen_a1])

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (baseline - q_stage2_mf[s, a])

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: State-Dependent Stage 2 Temperature with Baseline Decay
    Uses different inverse temperatures for the two Stage 2 states (planets), 
    capturing differences in decision noise that might arise depending on the 
    participant's familiarity or perceived difficulty of each state.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate
    decay_rate: [0, 1] - Decay rate for unchosen actions
    baseline: [0, 1] - Baseline value to which unchosen actions decay
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2_X: [0, 10] - Inverse temperature for Stage 2 State 0
    beta_2_Y: [0, 10] - Inverse temperature for Stage 2 State 1
    w: [0, 1] - Weight on model-based vs model-free strategy
    lam: [0, 1] - Eligibility trace parameter
    """
    learning_rate, decay_rate, baseline, beta_1, beta_2_X, beta_2_Y, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * baseline
    q_stage2_mf = np.ones((2, 2)) * baseline

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        beta_2 = beta_2_X if state_idx == 0 else beta_2_Y
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (baseline - q_stage1_mf[unchosen_a1])

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
