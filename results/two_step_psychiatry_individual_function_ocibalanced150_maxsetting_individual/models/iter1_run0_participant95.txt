Here are 3 new cognitive models that explore alternative mechanisms for the two-step decision-making task, avoiding the parameter combinations you have already tested.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model relaxes the assumption that learning occurs at the same rate for both stages. Participants might update their preferences for spaceships (Stage 1) at a different rate than they update their preferences for specific aliens (Stage 2), considering that the aliens' reward probabilities are the ones actually drifting over time.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Mixed MF-MB RL model with separate learning rates for stage 1 and stage 2.
    This model assumes that updates to spaceship values and alien values occur at different speeds.
    
    Bounds:
    alpha1: [0,1] - Learning rate for stage 1 (spaceships)
    alpha2: [0,1] - Learning rate for stage 2 (aliens)
    beta1: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    lam: [0,1] - Eligibility trace (how much stage 2 reward updates stage 1)
    w: [0,1] - Weighting of model-based vs model-free values (1 = fully MB)
    stickiness: [-5,5] - Tendency to repeat the previous stage 1 choice
    """
    alpha1, alpha2, beta1, beta2, lam, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits1 = beta1 * q_net + stickiness * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update to stage 1 using stage 1 learning rate
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Choice Stickiness (Perseveration)
Often, participants demonstrate choice perseveration not just at the first stage, but also at the second stage (e.g., continually choosing the same alien on a specific planet regardless of recent rewards). This model introduces a stage 2 stickiness parameter to capture state-specific habituation.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Mixed MF-MB RL model with separate stickiness parameters for both stage 1 and stage 2 choices.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    lam: [0,1] - Eligibility trace
    w: [0,1] - Weighting of model-based vs model-free values
    st1: [-5,5] - Stickiness for stage 1 (spaceships)
    st2: [-5,5] - Stickiness for stage 2 (aliens, state-specific)
    """
    alpha, beta1, beta2, lam, w, st1, st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.full(2, -1) # tracks the last action taken in each specific state

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if last_action1 != -1:
            rep1[last_action1] = 1
            
        logits1 = beta1 * q_net + st1 * rep1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            rep2[last_action2[state_idx]] = 1

        logits2 = beta2 * q_stage2_mf[state_idx] + st2 * rep2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition Matrix Learning
Instead of assuming that the participant possesses a static, perfect understanding of the transition probabilities (0.7 / 0.3) from trial 1, this model assumes they learn the transition matrix dynamically through experience, starting from a uniform prior.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Mixed MF-MB RL model that dynamically learns the transition matrix over time.
    
    Bounds:
    alpha: [0,1] - Value learning rate
    alpha_tr: [0,1] - Transition matrix learning rate
    beta1: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    lam: [0,1] - Eligibility trace
    w: [0,1] - Weighting of model-based vs model-free values
    stickiness: [-5,5] - Tendency to repeat the previous stage 1 choice
    """
    alpha, alpha_tr, beta1, beta2, lam, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to uniform probabilities (0.5)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]]) 
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits1 = beta1 * q_net + stickiness * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Transition matrix update using experienced state transitions
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] *= (1 - alpha_tr)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
