import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Asymmetric Learning Rates + Separate Betas + Stage 1 & 2 Stickiness
    
    This model uses different learning rates for positive and negative prediction errors,
    which is common in reinforcement learning as subjects may weigh gains and losses differently.
    It also includes separate stickiness parameters for both stage 1 and stage 2 choices.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p1: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    p2: [-5, 5] - Perseverance bonus for repeating the previous second-stage action in the same state.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lambda_, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_act1 = -1
    last_act2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act1 != -1:
            V1[last_act1] += p1
            
        q_stage1_mb = V1 - np.max(V1)
        beta = 1.0
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act1 = action_1[trial]
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        if last_act2[state_idx] != -1:
            V2[last_act2[state_idx]] += p2
            
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2 * lambda_ * delta_stage2
        
        last_act1 = action_1[trial]
        last_act2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Reward-Dependent Stage 1 Stickiness + Stage 2 Stickiness
    
    This model assumes that the tendency to repeat the first-stage action depends on whether
    the previous trial was rewarded (win-stay) or unrewarded (lose-shift). 
    It also includes a general stickiness parameter for the second stage.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p1_win: [-5, 5] - Perseverance bonus for stage 1 if the previous trial was rewarded.
    p1_lose: [-5, 5] - Perseverance bonus for stage 1 if the previous trial was unrewarded.
    p2: [-5, 5] - Perseverance bonus for repeating the previous second-stage action.
    """
    learning_rate, beta1, beta2, w, lambda_, p1_win, p1_lose, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_act1 = -1
    last_act2 = np.array([-1, -1])
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act1 != -1:
            if last_reward == 1:
                V1[last_act1] += p1_win
            else:
                V1[last_act1] += p1_lose
                
        q_stage1_mb = V1 - np.max(V1)
        beta = 1.0
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act1 = action_1[trial]
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        if last_act2[state_idx] != -1:
            V2[last_act2[state_idx]] += p2
            
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_act1 = action_1[trial]
        last_act2[state_idx] = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Q-Value Decay + Separate Betas + Stage 1 & 2 Stickiness
    
    This model adds a decay parameter for unchosen options. Over time, the Q-values
    of unchosen actions decay towards 0, reflecting forgetting or uncertainty about 
    unsampled options.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p1: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    p2: [-5, 5] - Perseverance bonus for repeating the previous second-stage action.
    """
    learning_rate, decay, beta1, beta2, w, lambda_, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_act1 = -1
    last_act2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act1 != -1:
            V1[last_act1] += p1
            
        q_stage1_mb = V1 - np.max(V1)
        beta = 1.0
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act1 = action_1[trial]
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        if last_act2[state_idx] != -1:
            V2[last_act2[state_idx]] += p2
            
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        # Decay unchosen actions towards 0
        unchosen_act1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_act1] *= (1.0 - decay)
        
        unchosen_act2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_act2] *= (1.0 - decay)
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1.0 - decay)
        q_stage2_mf[other_state, 1] *= (1.0 - decay)
        
        last_act1 = action_1[trial]
        last_act2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
