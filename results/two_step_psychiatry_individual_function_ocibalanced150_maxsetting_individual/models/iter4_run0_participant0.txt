def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative RPEs, 
    a single inverse temperature for both stages, model-based weight, 
    eligibility trace, stage 1 and stage 2 stickiness, and decay rate.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive RPEs
    alpha_neg: [0, 1] - Learning rate for negative RPEs
    beta: [0, 10] - Inverse temperature for both stages
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    stickiness_1: [-5, 5] - Choice stickiness for stage 1
    stickiness_2: [-5, 5] - Choice stickiness for stage 2
    decay_rate: [0, 1] - Forgetting rate for unchosen Q-values (decay towards 0.5)
    """
    alpha_pos, alpha_neg, beta, w, lam, stickiness_1, stickiness_2, decay_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + stickiness_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        rep_2 = np.zeros(2)
        if last_action_2 != -1:
            rep_2[last_action_2] = 1.0

        logits_2 = beta * q_stage2_mf[s] + stickiness_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_stage1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha_stage2 * delta_stage2
        q_stage1_mf[a1] += alpha_stage2 * lam * delta_stage2

        # Decay for unchosen Q-values towards initial value (0.5)
        q_stage1_mf[1 - a1] += decay_rate * (0.5 - q_stage1_mf[1 - a1])
        q_stage2_mf[s, 1 - a2] += decay_rate * (0.5 - q_stage2_mf[s, 1 - a2])
        q_stage2_mf[1 - s, 0] += decay_rate * (0.5 - q_stage2_mf[1 - s, 0])
        q_stage2_mf[1 - s, 1] += decay_rate * (0.5 - q_stage2_mf[1 - s, 1])
        
        last_action_1 = a1
        last_action_2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with positive/negative learning rates, separate betas for each stage,
    reward-dependent eligibility traces (lam_win, lam_lose), and stage 1 stickiness.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive RPEs
    alpha_neg: [0, 1] - Learning rate for negative RPEs
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    lam_win: [0, 1] - Eligibility trace when rewarded
    lam_lose: [0, 1] - Eligibility trace when unrewarded
    stickiness: [-5, 5] - Choice stickiness for stage 1
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam_win, lam_lose, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits_1 = beta_1 * q_net + stickiness * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_stage1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha_stage2 * delta_stage2
        
        # Apply reward-dependent eligibility trace
        lam = lam_win if r == 1 else lam_lose
        q_stage1_mf[a1] += alpha_stage2 * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with positive/negative learning rates, separate betas,
    reward-dependent model-based weights (w_win, w_lose), eligibility trace, 
    and stage 1 stickiness.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive RPEs
    alpha_neg: [0, 1] - Learning rate for negative RPEs
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w_win: [0, 1] - Model-based weight after a rewarded trial
    w_lose: [0, 1] - Model-based weight after an unrewarded trial
    lam: [0, 1] - Eligibility trace
    stickiness: [-5, 5] - Choice stickiness for stage 1
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w_win, w_lose, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Use w_win or w_lose based on previous reward outcome
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = (w_win + w_lose) / 2.0  # Fallback for the first trial

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits_1 = beta_1 * q_net + stickiness * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_stage1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha_stage2 * delta_stage2
        q_stage1_mf[a1] += alpha_stage2 * lam * delta_stage2
        
        last_action = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
