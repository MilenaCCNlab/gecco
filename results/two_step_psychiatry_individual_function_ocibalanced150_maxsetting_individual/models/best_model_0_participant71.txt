def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual Temperature Model.
    Separates the inverse temperature for model-free and model-based systems at stage 1.
    Instead of a weighting parameter w, it uses beta_mf and beta_mb.
    Includes biases for action 0 at both stages, perseveration at stage 1, and eligibility trace.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta_mf: [0,10] - Inverse temperature for model-free first-stage choice.
    beta_mb: [0,10] - Inverse temperature for model-based first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    p: [0,10] - Perseveration parameter for first-stage choice.
    lambd: [0,1] - Eligibility trace parameter.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    bias2: [-5,5] - Bias towards choosing action 0 at stage 2.
    """
    learning_rate, beta_mf, beta_mb, beta2, p, lambd, bias1, bias2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        action_bias1 = np.array([bias1, 0.0])
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * pers + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        action_bias2 = np.array([bias2, 0.0])
        logits_2 = beta2 * q_stage2_mf[state_idx] + action_bias2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambd * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
