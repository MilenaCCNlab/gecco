def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-step model with learned transition matrix.
    Instead of assuming the transition matrix is fixed at 0.7/0.3, this model assumes 
    participants learn the transition probabilities from experience.
    
    Parameters:
    alpha1: learning rate for stage 1 [0, 1]
    alpha2: learning rate for stage 2 [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    pi1: choice perseveration for stage 1 [-5, 5]
    alpha_trans: learning rate for the transition matrix [0, 1]
    """
    alpha1, alpha2, beta1, beta2, w, lambda_, pi1, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to 0.5 (unbiased initially)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * net_q1
        if prev_action_1 != -1:
            logits_1[prev_action_1] += pi1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        a1 = action_1[trial]
        a2 = action_2[trial]
        
        # Update transition matrix based on observed transition
        transition_matrix[a1, state_idx] += alpha_trans * (1 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        q_stage1_mf[a1] += alpha1 * lambda_ * delta_stage2
        
        prev_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-step model with separate eligibility traces for common and rare transitions.
    This model tests whether participants propagate reward information differently depending 
    on if the transition was expected (common) or unexpected (rare).
    
    Parameters:
    alpha1: learning rate for stage 1 [0, 1]
    alpha2: learning rate for stage 2 [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_c: eligibility trace for common transitions [0, 1]
    lambda_r: eligibility trace for rare transitions [0, 1]
    pi1: choice perseveration for stage 1 [-5, 5]
    """
    alpha1, alpha2, beta1, beta2, w, lambda_c, lambda_r, pi1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * net_q1
        if prev_action_1 != -1:
            logits_1[prev_action_1] += pi1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        a1 = action_1[trial]
        a2 = action_2[trial]
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        # Determine if transition was common (action 0 -> state 0, action 1 -> state 1)
        is_common = (a1 == state_idx)
        current_lambda = lambda_c if is_common else lambda_r
        
        q_stage1_mf[a1] += alpha1 * current_lambda * delta_stage2
        
        prev_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates for positive and negative prediction errors at both stages,
    with a shared inverse temperature and stage 1 perseveration.
    
    Parameters:
    alpha1_pos: learning rate for positive PE at stage 1 [0, 1]
    alpha1_neg: learning rate for negative PE at stage 1 [0, 1]
    alpha2_pos: learning rate for positive PE at stage 2 [0, 1]
    alpha2_neg: learning rate for negative PE at stage 2 [0, 1]
    beta: shared softmax inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    pi1: choice perseveration for stage 1 [-5, 5]
    """
    alpha1_pos, alpha1_neg, alpha2_pos, alpha2_neg, beta, w, lambda_, pi1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * net_q1
        if prev_action_1 != -1:
            logits_1[prev_action_1] += pi1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        a1 = action_1[trial]
        a2 = action_2[trial]
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha1_st1 = alpha1_pos if delta_stage1 > 0 else alpha1_neg
        q_stage1_mf[a1] += alpha1_st1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        alpha2 = alpha2_pos if delta_stage2 > 0 else alpha2_neg
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        alpha1_st2 = alpha1_pos if delta_stage2 > 0 else alpha1_neg
        q_stage1_mf[a1] += alpha1_st2 * lambda_ * delta_stage2
        
        prev_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
