def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate learning rates for positive and negative RPEs,
    separate stickiness for stage 1 and stage 2, and eligibility trace.
    
    Bounds:
    lr_pos: [0,1] - learning rate for positive prediction errors
    lr_neg: [0,1] - learning rate for negative prediction errors
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi1: [-5,5] - choice perseveration (stickiness) for stage 1
    pi2: [-5,5] - choice perseveration (stickiness) for stage 2
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    """
    lr_pos, lr_neg, beta1, beta2, w, pi1, pi2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta1 * q_net + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        if action_2[trial] != -1:
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            lr1 = lr_pos if delta_stage1 > 0 else lr_neg
            q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            lr2 = lr_pos if delta_stage2 > 0 else lr_neg
            q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2
            
            last_action2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with win-stay / lose-shift specific stickiness parameters.
    pi_win applies to stage 1 if the previous trial was rewarded.
    pi_lose applies to stage 1 if the previous trial was not rewarded.
    
    Bounds:
    learning_rate: [0,1] - learning rate for value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi_win: [-5,5] - choice perseveration for stage 1 if previous trial rewarded
    pi_lose: [-5,5] - choice perseveration for stage 1 if previous trial unrewarded
    pi2: [-5,5] - choice perseveration (stickiness) for stage 2
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    """
    learning_rate, beta1, beta2, w, pi_win, pi_lose, pi2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action1 = -1
    last_reward = -999
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        # Apply pi_win or pi_lose depending on last_reward
        if last_reward > 0:
            current_pi1 = pi_win
        elif last_reward <= 0 and last_reward != -999:
            current_pi1 = pi_lose
        else:
            current_pi1 = 0.0 # First trial
            
        logits_1 = beta1 * q_net + current_pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        if action_2[trial] != -1:
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
            last_action2[state_idx] = action_2[trial]
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0 # No reward if no action taken
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
