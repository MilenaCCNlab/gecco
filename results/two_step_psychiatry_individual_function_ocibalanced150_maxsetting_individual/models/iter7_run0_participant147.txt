import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate stage 1 and stage 2 temperatures, and reward-dependent perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    p_rew: [-5, 5] - Perseveration for stage 1 choice if last trial was rewarded.
    p_unrew: [-5, 5] - Perseveration for stage 1 choice if last trial was unrewarded.
    lam: [0, 1] - Eligibility trace.
    """
    learning_rate, beta_1, beta_2, w, p_rew, p_unrew, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers[last_action_1] = p_rew
            else:
                pers[last_action_1] = p_unrew
            
        logits_1 = beta_1 * q_stage1_net + pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates for positive and negative prediction errors, and reward-dependent perseveration.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based versus model-free values.
    p_rew: [-5, 5] - Perseveration parameter for Stage 1 choice if last trial was rewarded.
    p_unrew: [-5, 5] - Perseveration parameter for Stage 1 choice if last trial was unrewarded.
    lam: [0, 1] - Eligibility trace parameter.
    """
    alpha_pos, alpha_neg, beta, w, p_rew, p_unrew, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers[last_action_1] = p_rew
            else:
                pers[last_action_1] = p_unrew
            
        logits_1 = beta * q_stage1_net + pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with decay for unchosen actions, separate temperatures, and reward-dependent perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting of MB vs MF.
    p_rew: [-5, 5] - Perseveration if last trial rewarded.
    p_unrew: [-5, 5] - Perseveration if last trial unrewarded.
    lam: [0, 1] - Eligibility trace.
    """
    learning_rate, decay, beta_1, beta_2, w, p_rew, p_unrew, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers[last_action_1] = p_rew
            else:
                pers[last_action_1] = p_unrew
            
        logits_1 = beta_1 * q_stage1_net + pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])

        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])

        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
