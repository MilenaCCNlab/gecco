Here are 3 new cognitive models that build upon the best model so far by introducing novel, distinct mechanisms while adhering to the parameter limits.

### Model 1: Transition Learning Model
Instead of assuming the participant fully knows and fixes the transition probabilities at 0.7/0.3, this model assumes the participant learns the transition matrix dynamically from their experience. 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition Learning Model with Decay
    The agent learns the transition matrix from experience rather than assuming it is fixed.
    
    Parameters:
    alpha: [0, 1] - learning rate for Q-values
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    alpha_trans: [0, 1] - learning rate for updating the transition matrix
    decay: [0, 1] - decay rate for unchosen options
    p: [-10, 10] - stage 1 perseveration bonus
    """
    alpha, beta_1, beta_2, w, lam, alpha_trans, decay, p = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to 0.5 (uniform expectation)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += p
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay * (0.5 - q_stage2_mf[s_idx, a_idx])
                    
        # update learned transition matrix
        transition_matrix[a1, s] += alpha_trans * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_trans * (0.0 - transition_matrix[a1, 1 - s])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unrewarded Penalty (Omega) Model
Treats receiving 0 coins not just as a lack of reward, but as an active penalty (loss aversion). This is parameterized by `omega`, mapping unrewarded outcomes to negative values.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Unrewarded Penalty Model with Decay and Perseveration
    Treats unrewarded trials as a penalty rather than neutral, controlled by an 'omega' parameter.
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    omega: [0, 1] - penalty magnitude for unrewarded trials (r=0 becomes r=-omega)
    decay: [0, 1] - decay rate for unchosen options
    p: [-10, 10] - stage 1 perseveration bonus
    """
    alpha, beta_1, beta_2, w, lam, omega, decay, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    # Initialize to 0.0 because effective rewards can be negative
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # apply penalty if unrewarded
        r_eff = 1.0 if r == 1 else -omega

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += p
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r_eff - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.0 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay * (0.0 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-dependent Perseveration
This model posits that the participant's tendency to repeat the same stage 1 action (perseveration) depends on whether the previous trial's state transition was expected (common) or unexpected (rare).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-dependent Perseveration Model with Decay
    Perseveration at stage 1 branches depending on whether the previous transition was common or rare.
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    decay: [0, 1] - decay rate for unchosen options
    p_common: [-10, 10] - perseveration bonus if previous transition was common
    p_rare: [-10, 10] - perseveration bonus if previous transition was rare
    """
    alpha, beta_1, beta_2, w, lam, decay, p_common, p_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        
        # transition-dependent perseveration
        if trial > 0:
            prev_a1 = action_1[trial - 1]
            prev_s = state[trial - 1]
            # 0->0 and 1->1 are the common transitions
            is_common = (prev_a1 == prev_s)
            if is_common:
                logits_1[prev_a1] += p_common
            else:
                logits_1[prev_a1] += p_rare
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay * (0.5 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
