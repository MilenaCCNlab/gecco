Here are 3 new cognitive models that explore alternative mechanisms for the two-step task. Each model avoids the previously tried parameter combinations and introduces novel psychological hypotheses.

### Model 1: Forgetting / Decay Model
This model tests the hypothesis that participants slowly forget the values of unchosen options over time, pulling their expectations back toward a neutral baseline (0.5).
```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with decay (forgetting) of Q-values towards 0.5.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    decay: [0,1] - Forgetting rate for all Q-values towards 0.5 before each update.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    pers: [-5,5] - Perseveration bonus for repeating stage 1 choice.
    """
    learning_rate, decay, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 since rewards are 0 or 1
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_stage1_net
        
        if prev_a1 != -1:
            logits_1[prev_a1] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        # Decay all Q-values towards 0.5 (forgetting) before updating chosen options
        q_stage1_mf = (1 - decay) * q_stage1_mf + decay * 0.5
        q_stage2_mf = (1 - decay) * q_stage2_mf + decay * 0.5

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Model-Based Control
This model proposes that the participant's reliance on planning (Model-Based weight) changes depending on whether the previous trial was successful or not. For example, individuals might lean into Model-Free habits after a win, but engage Model-Based planning after a loss.
```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with reward-dependent model-based weight.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w_win: [0,1] - Weight on model-based strategy if previous trial was rewarded.
    w_lose: [0,1] - Weight on model-based strategy if previous trial was unrewarded.
    lam: [0,1] - Eligibility trace parameter.
    pers: [-5,5] - Perseveration bonus for repeating stage 1 choice.
    """
    learning_rate, beta_1, beta_2, w_win, w_lose, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Determine current model-based weight
        if prev_r == 1:
            current_w = w_win
        elif prev_r == 0:
            current_w = w_lose
        else:
            current_w = 0.5 # default if no previous valid reward

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        logits_1 = beta_1 * q_stage1_net
        
        if prev_a1 != -1:
            logits_1[prev_a1] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Perseveration
This model tests whether choice stickiness (perseveration) depends on whether the participant experienced a common vs. rare transition on the previous trial, irrespective of the reward outcome.
```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with transition-dependent perseveration for stage 1.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    pers_common: [-5,5] - Perseveration bonus if the previous transition was common.
    pers_rare: [-5,5] - Perseveration bonus if the previous transition was rare.
    """
    learning_rate, beta_1, beta_2, w, lam, pers_common, pers_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    prev_a1 = -1
    prev_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_stage1_net
        
        if prev_a1 != -1 and prev_state != -1:
            # Common transitions are a1 == 0 -> state 0, and a1 == 1 -> state 1
            if prev_a1 == prev_state:
                logits_1[prev_a1] += pers_common
            else:
                logits_1[prev_a1] += pers_rare
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        prev_a1 = a1
        prev_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
