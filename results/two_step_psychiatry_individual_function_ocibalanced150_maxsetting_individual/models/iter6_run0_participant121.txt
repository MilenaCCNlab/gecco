def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. (2011) model with separate learning rates for Stage 1 and Stage 2, 
    along with decay of unchosen actions towards a decay_center.
    
    Bounds:
    alpha_1: [0,1] - Stage 1 learning rate
    alpha_2: [0,1] - Stage 2 learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p: [0,5] - Choice stickiness
    decay: [0,1] - Decay rate for unchosen action values
    decay_center: [0,1] - The value towards which unchosen actions decay
    """
    alpha_1, alpha_2, beta, w, lambda_, p, decay, decay_center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, decay_center)
    q_stage2_mf = np.full((2, 2), decay_center)
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2

        # Decay for unchosen actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + decay_center * decay

        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] = q_stage1_mf[unchosen_a1] * (1 - decay) + decay_center * decay
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. (2011) model with separate learning rates for positive and negative reward prediction errors, 
    along with decay of unchosen actions towards a decay_center.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive PE
    alpha_neg: [0,1] - Learning rate for negative PE
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace

    decay: [0,1] - Decay rate for unchosen action values

    p_choice_2 = np.zeros(n_trials)

    q_stage2_mf = np.full((2, 2), decay_center)

        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
