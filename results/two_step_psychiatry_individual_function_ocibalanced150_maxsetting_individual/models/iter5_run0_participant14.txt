def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Eligibility trace (lambda), choice perseveration for both stages (stickiness_1, stickiness_2), 
    and unchosen value decay. Stage 2 stickiness is state-specific.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness_1: [-5, 5] - Tendency to repeat the previous stage-1 choice.
    stickiness_2: [-5, 5] - Tendency to repeat the previous stage-2 choice in the same state.
    decay: [0, 1] - Rate at which unchosen action values decay toward zero.
    """
    alpha, beta, w, lambda_, stickiness_1, stickiness_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var_1 = np.zeros(2)
        if trial > 0:
            stick_var_1[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness_1 * stick_var_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick_var_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stick_var_2[last_action_2[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + stickiness_2 * stick_var_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2[state_idx] = action_2[trial]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates (alpha_pos, alpha_neg), eligibility trace (lambda), 
    choice perseveration (stickiness), and unchosen value decay.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness: [-5, 5] - Tendency to repeat the previous stage-1 choice.
    decay: [0, 1] - Rate at which unchosen action values decay toward zero.
    """
    alpha_pos, alpha_neg, beta, w, lambda_, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness * stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_2 * lambda_ * delta_stage2
        
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
