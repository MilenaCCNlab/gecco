Here are three new cognitive models that build on the best-performing mechanisms (like value decay) while introducing new combinations of parameters such as choice stickiness (perseveration) at both stages, separate learning rates for different stages, and asymmetric learning (positive vs. negative prediction errors).

### Model 1
This model introduces separate learning rates and separate softmax temperatures for Stage 1 and Stage 2, alongside choice stickiness parameters for both stages. This allows the model to capture differences in how participants learn and explore at the two distinct decision points.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and temperatures for Stage 1 and Stage 2,
    with an eligibility trace, model-based weight, and separate stickiness parameters 
    for Stage 1 and Stage 2 choices.
    
    Parameters:
    alpha1: Learning rate for Stage 1 [0, 1]
    alpha2: Learning rate for Stage 2 [0, 1]
    beta1: Inverse temperature for Stage 1 [0, 10]
    beta2: Inverse temperature for Stage 2 [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    stickiness_1: Stickiness for Stage 1 choice [-5, 5]
    stickiness_2: Stickiness for Stage 2 choice [-5, 5]
    """
    alpha1, alpha2, beta1, beta2, w, lam, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if prev_a1 != -1:
            logits_1[prev_a1] += stickiness_1
            
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if prev_a2 != -1:
            logits_2[prev_a2] += stickiness_2
            
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2
This model combines the highly effective mechanism of unchosen value decay (from the best model) with separate choice stickiness (perseveration) parameters for Stage 1 and Stage 2. It assesses whether temporal decay and basic motor perseveration concurrently drive behavior.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with value decay for unchosen actions, and separate 
    stickiness parameters for Stage 1 and Stage 2 choices.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta: Inverse temperature [0, 10]
    w: Model-based weight [0, 1]
    lam: Eligibility trace [0, 1]
    decay: Decay rate for unchosen actions towards 0.5 [0, 1]
    stickiness_1: Stickiness for Stage 1 choice [-5, 5]
    stickiness_2: Stickiness for Stage 2 choice [-5, 5]
    """
    alpha, beta, w, lam, decay, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if prev_a1 != -1:
            logits_1[prev_a1] += stickiness_1
            
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if prev_a2 != -1:
            logits_2[prev_a2] += stickiness_2
            
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # decay for unchosen actions in Stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
                    
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3
This model incorporates asymmetric learning rates for positive and negative prediction errors (a common hallmark in psychiatric computational models), separate inverse temperatures for Stage 1 and Stage 2, a Stage 1 choice stickiness parameter, and unchosen value decay. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates (positive/negative PE), separate temperatures 
    for Stage 1 and Stage 2, model-based weight, eligibility trace, Stage 1 stickiness, 
    and value decay for unchosen actions.
    
    Parameters:
    alpha_pos: Learning rate for positive prediction errors [0, 1]
    alpha_neg: Learning rate for negative prediction errors [0, 1]
    beta1: Inverse temperature for Stage 1 [0, 10]
    beta2: Inverse temperature for Stage 2 [0, 10]
    w: Model-based weight [0, 1]
    lam: Eligibility trace parameter [0, 1]
    stickiness: Stickiness for Stage 1 choice [-5, 5]
    decay: Decay rate for unchosen actions towards 0.5 [0, 1]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if prev_a1 != -1:
            logits_1[prev_a1] += stickiness
            
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2
        
        # decay for unchosen actions in Stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
                    
        prev_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
