Here are 3 new cognitive models that introduce novel mechanisms to explain the participant's decision-making behavior, avoiding previously tried parameter combinations. Each model is built directly upon the required template and uses exactly 8 parameters.

### Model 1: Decaying Choice Trace (Perseveration Memory)
Instead of only perseverating on the immediately preceding action (which binary 0/1 indicator captures), this model maintains a **decaying trace** of past actions for both stage 1 and stage 2. This allows the model to capture longer-term stickiness or habit formation that decays over time.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Decaying Choice Trace (Perseveration)
    
    Instead of only perseverating on the immediately preceding action, this model
    maintains a decaying trace of past actions for both stage 1 and stage 2 choices.
    
    Parameters:
    alpha: learning rate for value updates [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p1: stage 1 perseveration weight [-5, 5]
    p2: stage 2 perseveration weight [-5, 5]
    tau: choice trace decay rate [0, 1]
    """
    alpha, beta1, beta2, w, lam, p1, p2, tau = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Initialize traces for stage 1 and stage 2 actions
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net + p1 * trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * trace_2[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update choice traces
        trace_1 *= tau
        trace_1[action_1[trial]] += 1.0
        
        trace_2 *= tau
        trace_2[state_idx, action_2[trial]] += 1.0
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward Baseline (Reference Dependence)
This model introduces a subjective **reward baseline**. In environments where participants expect to win, a reward of `0` might be encoded as a negative outcome rather than a neutral one. The baseline parameter `b` shifts the reward representation, fundamentally altering the prediction errors that drive learning.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Reward Baseline (Reference Dependence)
    
    This model introduces a baseline parameter 'b' that shifts the reward representation.
    A reward of 0 is perceived as -b, and a reward of 1 is perceived as 1 - b.
    This allows the model to treat omissions as negative outcomes.
    
    Parameters:
    alpha: learning rate for value updates [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p1: stage 1 perseveration [-5, 5]
    p2: stage 2 perseveration [-5, 5]
    b: subjective reward baseline [0, 1]
    """
    alpha, beta1, beta2, w, lam, p1, p2, b = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        effective_reward = reward[trial] - b
        delta_stage2 = effective_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Eligibility Traces (Success vs. Failure Updates)
This model investigates if the participant's credit assignment differs between successful and unsuccessful trials. It uses **two different eligibility traces** (`lam_pos` and `lam_neg`) depending on whether the stage 2 prediction error is positive or negative.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Asymmetric Eligibility Traces
    
    This model uses different eligibility traces (lam_pos and lam_neg) depending on 
    whether the stage 2 prediction error is positive or negative. This captures 
    asymmetric credit assignment for successes versus failures.
    
    Parameters:
    alpha: learning rate for value updates [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam_pos: eligibility trace for positive prediction errors [0, 1]
    lam_neg: eligibility trace for negative prediction errors [0, 1]
    p1: stage 1 perseveration [-5, 5]
    p2: stage 2 perseveration [-5, 5]
    """
    alpha, beta1, beta2, w, lam_pos, lam_neg, p1, p2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    
    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
