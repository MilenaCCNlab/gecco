Here are 3 new cognitive models that build upon the best model so far by introducing new psychological mechanisms while retaining the successful value forgetting (decay) towards a baseline (`q0`) mechanism.

### Model 1: Two-Stage Stickiness with Value Forgetting
This model extends the best model by adding a stickiness parameter for the second stage (`st2`). Participants often exhibit perseveration not just at the first-stage choice, but also in their alien selection on the planets. 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-stage stickiness with value forgetting towards a baseline
    Incorporates choice stickiness at both stage 1 and stage 2, along with
    unchosen action Q-values decaying towards a baseline q0.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight
    lam: [0,1] - Eligibility trace parameter
    st1: [-5,5] - Choice stickiness for stage 1
    st2: [-5,5] - Choice stickiness for stage 2
    decay: [0,1] - Forgetting rate for unchosen actions
    q0: [0,1] - Baseline Q-value that unchosen actions decay towards
    """
    import numpy as np
    learning_rate, beta, w, lam, st1, st2, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + q0
    q_stage2_mf = np.zeros((2, 2)) + q0
    
    last_action1 = -1
    last_action2 = np.full(2, -1) # tracks last action taken in each stage 2 state

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness1 = np.zeros(2)
        if last_action1 != -1:
            stickiness1[last_action1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st1 * stickiness1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            stickiness2[last_action2[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + st2 * stickiness2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # decay unchosen actions towards baseline
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] = q_stage1_mf[unchosen_action_1] * (1 - decay) + q0 * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + q0 * decay
                    
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Value Forgetting
This model splits the learning rate into `lr_pos` (for positive prediction errors) and `lr_neg` (for negative prediction errors). Together with value forgetting to `q0`, this can help differentiate if the participant is more sensitive to unexpected rewards or unexpected omissions.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates with Value Forgetting towards a Baseline
    Uses different learning rates for positive and negative prediction errors,
    while unchosen actions decay towards a baseline q0.
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors
    lr_neg: [0,1] - Learning rate for negative prediction errors
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness for stage 1
    decay: [0,1] - Forgetting rate for unchosen actions
    q0: [0,1] - Baseline Q-value that unchosen actions decay towards
    """
    import numpy as np
    lr_pos, lr_neg, beta, w, lam, st, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + q0
    q_stage2_mf = np.zeros((2, 2)) + q0
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

        # decay unchosen actions towards baseline
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] = q_stage1_mf[unchosen_action_1] * (1 - decay) + q0 * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + q0 * decay
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Stickiness with Value Forgetting
In this model, the perseveration tendency at stage 1 is split based on the outcome of the previous trial (`st_win` vs `st_lose`). This helps capture heuristic win-stay/lose-shift behavior explicitly alongside the value decay mechanism.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent Stickiness with Value Forgetting towards a Baseline
    Stage 1 stickiness depends on whether the previous trial was rewarded (st_win) 
    or not (st_lose), capturing win-stay/lose-shift tendencies. Unchosen actions decay to q0.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight
    lam: [0,1] - Eligibility trace parameter
    st_win: [-5,5] - Choice stickiness if previous trial was rewarded
    st_lose: [-5,5] - Choice stickiness if previous trial was unrewarded
    decay: [0,1] - Forgetting rate for unchosen actions
    q0: [0,1] - Baseline Q-value that unchosen actions decay towards
    """
    import numpy as np
    learning_rate, beta, w, lam, st_win, st_lose, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + q0
    q_stage2_mf = np.zeros((2, 2)) + q0
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        current_st = st_win if last_reward == 1 else st_lose
            
        exp_q1 = np.exp(beta * q_net + current_st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # decay unchosen actions towards baseline
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] = q_stage1_mf[unchosen_action_1] * (1 - decay) + q0 * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + q0 * decay
                    
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
