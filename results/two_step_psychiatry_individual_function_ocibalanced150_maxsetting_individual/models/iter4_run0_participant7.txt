def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with reward-dependent perseveration and Q-value decay.
    Unchosen Q-values decay towards 0.5 to capture forgetting.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    decay: decay rate for unchosen Q-values towards 0.5 [0,1]
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 for the decay model
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose

        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action = action_1[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay unchosen Q-values
        unchosen_stage1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_stage1] += decay * (0.5 - q_stage1_mf[unchosen_stage1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with reward-dependent model-based weight (w)
    and reward-dependent perseveration. Allows for different degrees of model-based planning
    following wins versus losses.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w_win: model-based weight if previous trial was rewarded [0,1]
    w_lose: model-based weight if previous trial was unrewarded [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    """
    alpha, beta1, beta2, w_win, w_lose, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        current_w = 0.5
        if last_reward == 1:
            current_w = w_win
        elif last_reward == 0:
            current_w = w_lose
            
        q_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose

        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action = action_1[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with reward-dependent perseveration
    and valence-dependent eligibility traces (lam_pos, lam_neg) to distinguish credit
    assignment for positive versus negative reward prediction errors.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam_pos: eligibility trace for positive reward prediction errors [0,1]
    lam_neg: eligibility trace for negative reward prediction errors [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    """
    alpha, beta1, beta2, w, lam_pos, lam_neg, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose

        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action = action_1[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        if delta_stage2 >= 0:
            q_stage1_mf[action_1[trial]] += alpha * lam_pos * delta_stage2
        else:
            q_stage1_mf[action_1[trial]] += alpha * lam_neg * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
