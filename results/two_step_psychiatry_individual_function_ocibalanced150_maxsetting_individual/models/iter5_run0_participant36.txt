def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition-dependent eligibility traces.
    Participants might credit the first-stage action differently depending on whether 
    the transition to the second stage was common or rare, reflecting different 
    levels of causal attribution to their initial choice.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam_c: [0, 1] - eligibility trace for common transitions
    lam_r: [0, 1] - eligibility trace for rare transitions
    st: [0, 10] - choice stickiness for stage 1
    """
    alpha, beta1, beta2, w, lam_c, lam_r, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1
            
        logits_1 = beta1 * q_net + st * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        # Determine if transition was common or rare
        # Spaceship 0 -> Planet 0 and Spaceship 1 -> Planet 1 are common
        is_common = (a1 == state_idx)
        current_lam = lam_c if is_common else lam_r

        # Updating for Choice 1
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * current_lam * delta_stage2
        
        # Updating for Choice 2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates for positive and negative prediction errors,
    combined with outcome-dependent choice stickiness (win-stay/lose-shift tendencies).
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    st_win: [0, 10] - stickiness after a rewarded trial
    st_lose: [0, 10] - stickiness after an unrewarded trial
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, st_win, st_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            st = st_win if last_reward == 1 else st_lose
            stickiness[last_action] = st
            
        logits_1 = beta1 * q_net + stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

        # Updating for Choice 1
        q_stage1_mf[a1] += alpha_s1 * delta_stage1 + alpha_s2 * lam * delta_stage2
        
        # Updating for Choice 2
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2
        
        last_action = a1
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with counterfactual updating for the unchosen stage 2 action.
    When a participant receives a reward (or no reward) from the chosen alien,
    they might infer that the unchosen alien on the same planet would have 
    yielded the opposite outcome, updating its value fictitiously.
    
    Bounds:
    alpha: [0, 1] - learning rate for chosen actions
    alpha_cf: [0, 1] - learning rate for counterfactual update of unchosen stage 2 action
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    st: [0, 10] - choice stickiness for stage 1
    """
    alpha, alpha_cf, beta1, beta2, w, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1
            
        logits_1 = beta1 * q_net + st * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        # Updating for Choice 1
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # Updating for Choice 2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Counterfactual update for the unchosen action in the same state
        unchosen_a2 = 1 - a2
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += alpha_cf * delta_cf
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
