import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates with Decay
    Uses different learning rates for positive and negative prediction errors,
    along with a decay mechanism for unchosen actions. This captures both 
    valence-dependent learning and memory decay over trials.
    
    Parameters:
    lr_pos: [0,1] - learning rate for positive prediction errors
    lr_neg: [0,1] - learning rate for negative prediction errors
    decay: [0,1] - decay rate for unchosen Q-values towards 0.5
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    p: [-2,2] - choice perseverance bonus for stage 1
    """
    lr_pos, lr_neg, decay, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and state_idx != -1 and a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            lr1 = lr_pos if delta_stage1 > 0 else lr_neg
            lr2 = lr_pos if delta_stage2 > 0 else lr_neg
            
            # Decay for unchosen options
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == a2):
                        q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
            
            # ACTION VALUE UPDATING FOR CHOICE 1
            q_stage1_mf[a1] += lr1 * delta_stage1 + lr2 * lam * delta_stage2
            # ACTION VALUE UPDATING FOR CHOICE 2
            q_stage2_mf[state_idx, a2] += lr2 * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-Dependent Perseveration with Decay
    Perseveration tendency on stage 1 depends on whether the previous trial was 
    rewarded or unrewarded, combined with a decay mechanism for unchosen actions.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    decay: [0,1] - decay rate for unchosen Q-values towards 0.5
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    p_rew: [-2,2] - choice perseverance bonus if previous trial was rewarded
    p_unrew: [-2,2] - choice perseverance bonus if previous trial was unrewarded
    """
    learning_rate, decay, beta1, beta2, w, lam, p_rew, p_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                pers[last_action_1] = p_rew
            else:
                pers[last_action_1] = p_unrew
                
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and state_idx != -1 and a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            # Decay for unchosen options
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == a2):
                        q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
            
            # ACTION VALUE UPDATING FOR CHOICE 1
            q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
            # ACTION VALUE UPDATING FOR CHOICE 2
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1
            last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
