Here are 3 new cognitive models that introduce novel parameter combinations and mechanisms (such as separate stage learning rates, asymmetric learning rates for positive/negative prediction errors, and win-stay-lose-shift biases) to better capture the participant's choice behavior. 

### Model 1: Dual Learning Rates, Dual Betas, and Dual Stickiness
This model allows for separate learning rates and inverse temperatures (betas) for stage 1 and stage 2 choices, recognizing that learning and exploration/exploitation trade-offs may differ between the two decision stages. It also includes choice stickiness for both stages.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Different learning rates and betas for each stage, with stage 1 and stage 2 stickiness.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace [0,1]
    pers1: stickiness for stage 1 choices [-5,5]
    pers2: stickiness for stage 2 choices [-5,5]
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        # Handle missing trials
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick1 = np.zeros(2)
        if last_a1 != -1:
            stick1[last_a1] = 1.0
            
        q1_net = q_stage1_net + pers1 * stick1
        q1_max = np.max(q1_net)
        exp_q1 = np.exp(beta1 * (q1_net - q1_max))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stick2 = np.zeros(2)
        if last_a2 != -1:
            stick2[last_a2] = 1.0
            
        q2_net = q_stage2_mf[state_idx] + pers2 * stick2
        q2_max = np.max(q2_net)
        exp_q2 = np.exp(beta2 * (q2_net - q2_max))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_a1 = action_1[trial]
        last_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Dual Betas and Dual Stickiness
This model captures the possibility that the participant learns differently from positive and negative prediction errors (e.g., they might update values more strongly after receiving a reward than missing out on one, or vice versa). It also maintains separate inverse temperatures and choice stickiness for both decision stages.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive/negative prediction errors, 
    separate betas and stickiness for both stages.
    
    Parameters:
    alpha_pos: learning rate for positive PE [0,1]
    alpha_neg: learning rate for negative PE [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace [0,1]
    pers1: stickiness for stage 1 choices [-5,5]
    pers2: stickiness for stage 2 choices [-5,5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = -1

        if last_a2 != -1:
            stick2[last_a2] = 1.0

        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Dual Learning Rates, Dual Betas, and Win-Stay-Lose-Shift (WSLS)
This model introduces a heuristic Win-Stay-Lose-Shift (WSLS) mechanism alongside the full reinforcement learning framework. If the previous trial yielded a reward, the participant may exhibit a heuristic bias to repeat their stage 1 choice; if unrewarded, they may bias towards shifting to the alternative choice.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Different learning rates and betas for each stage, and a win-stay-lose-shift (WSLS) mechanism for stage 1.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace [0,1]
    wsls: win-stay-lose-shift bias for stage 1 choices [-5,5]
    """
    alpha1, alpha2, beta1, beta2, w, lam, wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # Handle missing trials
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        wsls_bias = np.zeros(2)
        if last_a1 != -1 and last_reward != -1:
            if last_reward > 0:
                wsls_bias[last_a1] = 1.0
            else:
                wsls_bias[1 - last_a1] = 1.0
            
        q1_net = q_stage1_net + wsls * wsls_bias
        q1_max = np.max(q1_net)
        exp_q1 = np.exp(beta1 * (q1_net - q1_max))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_net = q_stage2_mf[state_idx]
        q2_max = np.max(q2_net)
        exp_q2 = np.exp(beta2 * (q2_net - q2_max))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_a1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
