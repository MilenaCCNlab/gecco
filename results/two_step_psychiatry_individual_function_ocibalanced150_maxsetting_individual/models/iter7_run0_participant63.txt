import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Eligibility trace, separate temperatures, and win/lose stickiness.
    
    This model separates the exploration/exploitation trade-off into stage 1 and stage 2 
    via two distinct inverse temperatures. It also incorporates an eligibility trace (lam) 
    to properly credit the first stage choice from the second stage reward, and uses separate 
    stickiness parameters for trials following a win versus a loss.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    lam: [0, 1] - Eligibility trace
    w: [0, 1] - Model-based weight
    p_win: [0, 10] - Choice stickiness after a rewarded trial
    p_lose: [0, 10] - Choice stickiness after an unrewarded trial
    """
    alpha, beta1, beta2, lam, w, p_win, p_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    last_reward = 0
    
    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            if last_reward == 1:
                stickiness[last_action] = p_win
            else:
                stickiness[last_action] = p_lose
                
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action = a1
        last_reward = r
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Eligibility trace, separate temperatures, single stickiness, and decay.
    
    This model adds an eligibility trace (lam) and a general choice stickiness parameter (p) 
    that applies regardless of the previous trial's outcome. It also features a decay parameter 
    that pulls all Q-values towards the uninformative prior of 0.5 at each trial, capturing 
    forgetting over time.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    lam: [0, 1] - Eligibility trace
    w: [0, 1] - Model-based weight
    p: [0, 10] - Choice stickiness
    decay: [0, 1] - Decay rate for Q-values towards initial 0.5
    """
    alpha, beta1, beta2, lam, w, p, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    
    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = p
                
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        # Decay of Q-values
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)
        
        q_stage1_mf[a1] += alpha * delta_stage1
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
