import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with value decay and Win-Stay/Lose-Shift (WSLS) heuristics.
    WSLS adds a bonus to the previously chosen stage 1 action depending on the reward.
    
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stay_win: [-5, 5]
    stay_lose: [-5, 5]
    """
    alpha, decay, beta1, beta2, w, lam, stay_win, stay_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1
    prev_r = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_r = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if prev_a1 != -1:
            if prev_r == 1:
                logits_1[prev_a1] += stay_win
            elif prev_r == 0:
                logits_1[prev_a1] += stay_lose
                
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay) * q_stage2_mf[state_idx, action_idx] + decay * 0.5

        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with value decay and counterfactual updating.
    The unchosen alien in the visited planet is updated with the opposite reward (1-r),
    scaled by cf_weight.
    
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    cf_weight: [0, 1]
    """
    alpha, decay, beta1, beta2, w, lam, cf_weight = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Counterfactual update for the unchosen alien in the same state
        unchosen_a2 = 1 - a2
        delta_cf = (1.0 - r) - q_stage2_mf[s, unchosen_a2]
        q_stage2_mf[s, unchosen_a2] += alpha * cf_weight * delta_cf

        # Decay for stage 1
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        # Decay for stage 2 (only for the unvisited state)
        unvisited_s = 1 - s
        for action_idx in range(2):
            q_stage2_mf[unvisited_s, action_idx] = (1.0 - decay) * q_stage2_mf[unvisited_s, action_idx] + decay * 0.5

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with value decay and a decaying choice trace for perseveration.
    The choice trace accumulates for chosen actions and decays over time.
    
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pers: [-5, 5]
    trace_decay: [0, 1]
    """
    alpha, decay, beta1, beta2, w, lam, pers, trace_decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    choice_trace = np.zeros(2)
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            choice_trace = (1.0 - trace_decay) * choice_trace
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net + pers * choice_trace
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay) * q_stage2_mf[state_idx, action_idx] + decay * 0.5

        # choice trace updating
        choice_trace = (1.0 - trace_decay) * choice_trace
        choice_trace[a1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
