Here are 3 new cognitive models that introduce reward-dependent choice stickiness (`p_win` and `p_lose`) to capture the nuance of whether a participant is more likely to repeat a choice after a reward versus a non-reward. These models combine this mechanism with asymmetric learning rates, stage-specific parameters, and unchosen action decay to test different explanations for the participant's behavior.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative RPEs,
    separate inverse temperatures for stage 1 and stage 2,
    and reward-dependent choice stickiness (p_win, p_lose).
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p_win: [0,5] - Choice stickiness after a rewarded trial
    p_lose: [0,5] - Choice stickiness after an unrewarded trial
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            if last_reward > 0:
                q_net[last_action_1] += p_win
            else:
                q_net[last_action_1] += p_lose
                
        exp_q1 = np.exp(beta_1 * q_net - np.max(beta_1 * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative RPEs,
    reward-dependent choice stickiness (p_win, p_lose),
    and unchosen action value decay towards zero.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    beta: [0,10] - Inverse temperature for both stages
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p_win: [0,5] - Choice stickiness after a rewarded trial
    p_lose: [0,5] - Choice stickiness after an unrewarded trial
    decay: [0,1] - Decay rate for unchosen action values
    """
    alpha_pos, alpha_neg, beta, w, lambda_, p_win, p_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            if last_reward > 0:
                q_net[last_action_1] += p_win
            else:
                q_net[last_action_1] += p_lose
                
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2
        
        # decay for unchosen actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
