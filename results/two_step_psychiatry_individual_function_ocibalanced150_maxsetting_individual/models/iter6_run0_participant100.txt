Here are 3 new cognitive models that build upon the insights of the best model so far (which highlighted the importance of stage 1 and stage 2 stickiness) while exploring new parameter combinations and mechanisms. Each model is constrained to a maximum of 8 parameters and avoids previously tried combinations.

### Model 1: Asymmetric Learning Rates + Dual Stickiness
This model tests whether participants learn differently from positive versus negative prediction errors at both stages, while retaining the dual-stickiness mechanisms for perseveration.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative prediction errors, 
    with stage 1 and stage 2 stickiness.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    st_1: choice stickiness for stage 1 [0,10]
    st_2: choice stickiness for stage 2 [0,10]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, st_1, st_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if last_a1 != -1:
            rep1[last_a1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + st_1 * rep1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 choice
        rep2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            rep2[last_a2[state_idx]] = 1.0

        logits_2 = beta_2 * q_stage2_mf[state_idx, :] + st_2 * rep2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2
        q_stage1_mf[a1] += alpha_s2 * lam * delta_stage2

        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Subjective Transition Probability + Dual Stickiness
This model investigates whether participants hold a subjective belief about the task's transition structure (e.g., underestimating or overestimating the 0.7/0.3 probability) rather than using the objective probabilities.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Subjective transition probability with stage 1 and 2 stickiness.
    Participants may not use the exact 0.7/0.3 transition probabilities, 
    but rather a subjective probability (p_trans) for the common transitions.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
```

### Model 3: Win-Stay / Lose-Shift Stickiness for Stage 1 + Stage 2 Stickiness
This model allows the perseveration tendency at stage 1 to depend dynamically on whether the previous trial resulted in a reward or not (evaluating win-stay vs. lose-shift behaviors).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Win-stay / Lose-shift stickiness for stage 1, and general stickiness for stage 2.
    This model allows the perseveration at stage 1 to depend on whether the previous trial was rewarded.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    st_win: choice stickiness for stage 1 after a reward [0,10]
    st_lose: choice stickiness for stage 1 after no reward (loss) [0,10]
    st_2: choice stickiness for stage 2 [0,10]
    """
    alpha, beta_1, beta_2, w, lam, st_win, st_lose, st_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    last_a1 = -1
    last_reward = 0
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if last_a1 != -1:
            rep1[last_a1] = 1.0
            
        st_1 = st_win if last_reward > 0 else st_lose
        
        logits_1 = beta_1 * q_stage1_net + st_1 * rep1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 choice
        rep2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            rep2[last_a2[state_idx]] = 1.0

        logits_2 = beta_2 * q_stage2_mf[state_idx, :] + st_2 * rep2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lam * delta_stage2

        last_a1 = a1
        last_a2[state_idx] = a2
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
