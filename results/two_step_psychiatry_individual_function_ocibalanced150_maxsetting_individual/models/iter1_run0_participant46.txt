Here are three new cognitive models that build upon the baseline and the best model so far, incorporating different well-established mechanisms from the decision-making literature to explain participant behavior.

### Model 1: Full Daw 2011 Model (Eligibility Trace + Stickiness)
This model introduces an eligibility trace parameter (`lam`) alongside stickiness. The eligibility trace allows the prediction error generated at the second stage (reward) to directly update the model-free value of the first-stage action. This is the complete version of the classic Daw et al. (2011) two-step task model.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Full Daw 2011 Model
    Incorporates model-based and model-free learning (w), eligibility trace (lam), 
    and choice perseveration (st) at the first stage.
    
    Bounds:
    learning_rate: [0, 1] - Rate of updating Q-values from prediction errors
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax policy
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax policy
    w: [0, 1] - Weighting between model-based (1) and model-free (0) systems
    lam: [0, 1] - Eligibility trace parameter propagating stage 2 prediction errors to stage 1
    st: [0, 10] - Choice perseveration (stickiness) for the previous stage 1 action
    """
    import numpy as np
    learning_rate, beta_1, beta_2, w, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net
        if last_action != -1:
            logits_1[last_action] += st
            
        # Subtract max for numerical stability
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates Model
Research shows that individuals often learn differently from positive versus negative feedback (e.g., optimistic vs. pessimistic updating). This model uses two separate learning rates depending on whether the prediction error is positive or negative.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates Model
    Uses different learning rates for positive and negative prediction errors,
    which can capture optimistic or pessimistic learning tendencies, alongside stickiness.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (better than expected)
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (worse than expected)
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting between model-based (1) and model-free (0) systems
    st: [0, 10] - Choice perseveration (stickiness) parameter
    """
    import numpy as np
    alpha_pos, alpha_neg, beta_1, beta_2, w, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net
        if last_action != -1:
            logits_1[last_action] += st
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Forgetting (Decay) Model
In volatile environments where reward probabilities drift, participants often assume that unchosen options revert to a baseline expectation. This model incorporates a decay parameter that pulls the Q-values of all unchosen actions toward 0.5 on each trial.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Forgetting (Decay) Model
    Incorporates decay of Q-values for unchosen actions towards the initial value of 0.5.
    This captures the assumption that the environment is volatile and unobserved states drift.
    
    Bounds:
    learning_rate: [0, 1] - Rate of updating Q-values from prediction errors
    decay: [0, 1] - Rate at which unchosen Q-values decay towards 0.5
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting between model-based (1) and model-free (0) systems
    st: [0, 10] - Choice perseveration (stickiness) parameter
    """
    import numpy as np
    learning_rate, decay, beta_1, beta_2, w, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 for the decay model
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net
        if last_action != -1:
            logits_1[last_action] += st
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Decay for unchosen stage 1 action
        unchosen_stage1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_stage1] += decay * (0.5 - q_stage1_mf[unchosen_stage1])
        
        # Decay for unchosen stage 2 actions (all unvisited state-action pairs)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
