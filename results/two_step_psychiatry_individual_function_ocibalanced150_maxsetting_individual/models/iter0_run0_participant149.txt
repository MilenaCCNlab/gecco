def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Full Model-Free and Model-Based hybrid model (Daw et al., 2011)
    
    This model assumes participants use a mixture of model-free (MF) and model-based (MB) 
    strategies for their first-stage choice, and use MF learning for the second stage.
    An eligibility trace allows the second-stage reward to update the first-stage action value.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter connecting stage 2 reward to stage 1
    """
    learning_rate, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Handle missed trials gracefully
        if a1 == -1 or s == -1 or a2 == -1:
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Weighted combination of MB and MF values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[s] - np.max(beta * q_stage2_mf[s]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating for choice 1 (Stage 1)
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        # Action value updating for choice 2 (Stage 2)
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with choice stickiness (perseveration) at stage 1
    
    This model builds upon Model 1 by adding a stickiness parameter that captures
    the participant's tendency to repeat their previous first-stage choice regardless 
    of the Q-values, which is common in two-step tasks.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter
    pi: [-5, 5] - Stickiness parameter for repeating the previous stage 1 choice (positive = perseveration)
    """
    learning_rate, beta, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits = beta * q_net + pi * stickiness
        exp_q1 = np.exp(logits - np.max(logits))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[s] - np.max(beta * q_stage2_mf[s]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with separate learning rates for positive and negative prediction errors
    
    This model assumes participants might update their expectations differently 
    depending on whether the outcome was better or worse than expected. 
    It splits the learning rate into two components based on the sign of the prediction error.
    
    Bounds:
    lr_pos: [0, 1] - Learning rate for positive prediction errors (better than expected)
    lr_neg: [0, 1] - Learning rate for negative prediction errors (worse than expected)
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter
    """
    lr_pos, lr_neg, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[s] - np.max(beta * q_stage2_mf[s]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[s, a2] += alpha2 * delta_stage2
        q_stage1_mf[a1] += alpha2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
