Here are 3 new cognitive models that explore alternative psychological mechanisms for the two-step decision-making task. They avoid the exact parameter combinations previously tried while remaining within the 8-parameter limit.

### Model 1: Asymmetric Learning Rates for Positive and Negative Prediction Errors (Stage 2)
This model hypothesizes that participants update their values differently depending on whether they receive a reward (positive prediction error) or no reward (negative prediction error) at Stage 2. This captures differences in sensitivity to wins versus losses during learning.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative prediction errors at Stage 2.
    Bounds:
    alpha_1: [0,1] (learning rate for stage 1)
    alpha_2_pos: [0,1] (learning rate for stage 2 when rewarded)
    alpha_2_neg: [0,1] (learning rate for stage 2 when unrewarded)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd: [0,1] (eligibility trace)
    pers: [0,10] (perseveration bonus for stage 1)
    """
    alpha_1, alpha_2_pos, alpha_2_neg, beta_1, beta_2, w, lambd, pers = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2_pos * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2_neg * delta_stage2
            
        q_stage1_mf[action_1[trial]] += alpha_1 * lambd * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Loss Aversion for Omitted Rewards
This model assumes that receiving 0 coins is not just a neutral outcome but is perceived negatively (like a loss). A `loss_av` parameter scales the omitted reward to a negative value, shifting how state values are penalized after an unrewarded trial.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a loss aversion / penalty parameter for unrewarded trials.
    Bounds:
    alpha_1: [0,1] (learning rate for stage 1)
    alpha_2: [0,1] (learning rate for stage 2)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd: [0,1] (eligibility trace)
    pers: [0,10] (perseveration bonus for stage 1)
    loss_av: [0,5] (negative penalty applied when reward is 0)
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lambd, pers, loss_av = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)  # Initialized at 0 since bounds can dip below 0
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Effective reward is negative if standard reward is 0
        r = reward[trial] if reward[trial] > 0 else -loss_av
        
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lambd * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Differential Eligibility Traces for Common vs. Rare Transitions
This model tests the hypothesis that credit assignment back to Stage 1 depends on the transition type. Participants might strongly reinforce their Stage 1 choice if the transition was predictable (common), but discount the feedback if the transition was a fluke (rare).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate eligibility traces depending on whether the transition was common or rare.
    Bounds:
    alpha_1: [0,1] (learning rate for stage 1)
    alpha_2: [0,1] (learning rate for stage 2)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd_common: [0,1] (eligibility trace applied after common transitions)
    lambd_rare: [0,1] (eligibility trace applied after rare transitions)
    pers: [0,10] (perseveration bonus for stage 1)
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lambd_common, lambd_rare, pers = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Determine transition type (A0->S0 and A1->S1 are common)
        if action_1[trial] == state_idx:
            lambd = lambd_common
        else:
            lambd = lambd_rare
            
        q_stage1_mf[action_1[trial]] += alpha_1 * lambd * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
