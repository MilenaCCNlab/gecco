import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate learning rates and inverse temperatures for stage 1 and stage 2, 
    and choice perseverance (stickiness) for stage 1.
    
    This model allows for different learning dynamics and exploration/exploitation 
    trade-offs between the first and second stages, while still accounting for the 
    participant's strong tendency to repeat first-stage actions.

    Parameters:
    model_parameters: list or array containing:
        alpha1: [0, 1] - Learning rate for updating Q-values in stage 1.
        alpha2: [0, 1] - Learning rate for updating Q-values in stage 2.
        beta1: [0, 10] - Inverse temperature for softmax action selection in stage 1.
        beta2: [0, 10] - Inverse temperature for softmax action selection in stage 2.
        w: [0, 1] - Weighting of model-based vs model-free values (1=MB, 0=MF).
        lambda_: [0, 1] - Eligibility trace parameter.
        pi: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1.
    """
    alpha1, alpha2, beta1, beta2, w, lambda_, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi
            
        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # eligibility trace updating for choice 1 based on stage 2 reward
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with a continuous habit trace for stage 1 actions.
    
    Instead of a simple 1-trial stickiness, this model uses a continuous habit trace 
    that tracks the frequency of recent actions and decays over time. This can better 
    capture block-like perseverative behavior seen in the participant's data.

    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate for Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        alpha_H: [0, 1] - Learning rate for the habit trace.
        w_H: [-5, 5] - Weight of the habit trace on stage 1 choices.
    """
    alpha, beta, w, lambda_, alpha_H, w_H = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    habit = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf + w_H * habit
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace updating for choice 1 based on stage 2 reward
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # update habit trace
        habit = (1 - alpha_H) * habit
        habit[action_1[trial]] += alpha_H

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate learning rates and inverse temperatures for both stages, 
    and choice perseverance for BOTH stage 1 and stage 2 actions.
    
    This is a comprehensive model that allows for distinct learning and choice dynamics 
    at each stage, while tracking perseveration independently for the first-stage spaceships 
    and the second-stage aliens (conditional on the planet).

    Parameters:
    model_parameters: list or array containing:
        alpha1: [0, 1] - Learning rate for stage 1.
        alpha2: [0, 1] - Learning rate for stage 2.
        beta1: [0, 10] - Inverse temperature for stage 1.
        beta2: [0, 10] - Inverse temperature for stage 2.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi1: [-5, 5] - Choice perseverance for stage 1.
        pi2: [-5, 5] - Choice perseverance for stage 2 (state-dependent).
    """
    alpha1, alpha2, beta1, beta2, w, lambda_, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = None
    last_action_2 = {0: None, 1: None} # state-dependent tracking of last stage 2 action

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action_1 is not None:
            q_stage1_net[last_action_1] += pi1
            
        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] is not None:
            q_stage2_net[last_action_2[state_idx]] += pi2

        exp_q2 = np.exp(beta2 * (q_stage2_net - np.max(q_stage2_net)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # eligibility trace updating for choice 1 based on stage 2 reward
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
