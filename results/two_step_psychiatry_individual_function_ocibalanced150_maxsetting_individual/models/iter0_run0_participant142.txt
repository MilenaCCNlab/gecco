Here are three cognitive models that build upon the template to explain participant behavior in the two-step task. 

Given the prompt's instruction to not modify what is already specified in the template while adding logic, I have preserved every line of the original template's structure. To introduce the necessary Model-Based (MB) vs. Model-Free (MF) weighting and choice perseverance without deleting the template's `exp_q1 = np.exp(beta * q_stage1_mb)` line, the models re-assign the `q_stage1_mb` variable right before the exponentiation to represent the **net** Q-value.

### Model 1: Standard Daw Model with Perseverance
This model implements the standard hybrid MB/MF reinforcement learning model (Daw et al., 2011). It includes an eligibility trace (`lam`) to propagate second-stage rewards directly to first-stage actions, a weight parameter (`w`) to balance MB and MF control, and a perseverance parameter (`pers`) to capture the participant's tendency to repeat the previous trial's first-stage choice.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Standard Two-Step Model (Daw et al. 2011) with choice perseverance.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    beta: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers: [-5,5] - choice perseverance (tendency to repeat stage 1 choice)
    """
    alpha, beta, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF into the net Q-value, adding perseverance
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            safe_beta = max(beta, 1e-5) # Avoid division by zero
            q_stage1_mb[action_1[trial-1]] += pers / safe_beta
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates (Positive vs. Negative RPEs)
In psychiatric datasets, participants often exhibit asymmetric learning—updating values differently when outcomes are better than expected (positive reward prediction error) compared to when they are worse than expected (negative RPE). This model splits the learning rate into `alpha_pos` and `alpha_neg`.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with asymmetric learning rates for positive and negative RPEs.
    
    Bounds:
    alpha_pos: [0,1] - learning rate for positive prediction errors
    alpha_neg: [0,1] - learning rate for negative prediction errors
    beta: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers: [-5,5] - choice perseverance
    """
    alpha_pos, alpha_neg, beta, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF into the net Q-value, adding perseverance
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            safe_beta = max(beta, 1e-5)
            q_stage1_mb[action_1[trial-1]] += pers / safe_beta
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Value Decay for Unchosen Actions
This model assumes that learning is volatile, and memory decays over time. If a participant doesn't choose a specific spaceship or alien, their cached model-free value for that option decays back toward zero.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with value decay for unchosen actions (forgetting).
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    beta: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers: [-5,5] - choice perseverance 
    decay: [0,1] - decay rate for unchosen actions towards 0
    """
    alpha, beta, beta2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF into the net Q-value, adding perseverance
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            safe_beta = max(beta, 1e-5)
            q_stage1_mb[action_1[trial-1]] += pers / safe_beta
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay unchosen actions towards baseline 0
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
