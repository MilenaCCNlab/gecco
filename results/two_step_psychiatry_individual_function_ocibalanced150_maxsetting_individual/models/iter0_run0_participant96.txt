def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based and Model-Free learning (Daw et al., 2011).
    Combines model-based and model-free value estimates for the first-stage choice,
    capturing the participant's balance between goal-directed and habitual behavior.
    
    Bounds:
    learning_rate: [0, 1] - learning rate for value updating
    beta: [0, 10] - inverse temperature for softmax choices
    w: [0, 1] - model-based weight (1 = pure MB, 0 = pure MF)
    """
    learning_rate, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Hybrid Q-value
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta * q_stage1_net
        q1_total -= np.max(q1_total) # prevent overflow
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with choice perseveration and eligibility trace.
    Accounts for the participant's strong tendency to repeat the previous stage 1 
    choice (e.g., sticking with spaceship 0 for 164 trials before switching).
    
    Bounds:
    learning_rate: [0, 1] - learning rate
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace parameter
    pers: [-5, 5] - perseveration bonus for repeating the last stage 1 action
    """
    learning_rate, beta, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseveration
        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = pers
            
        q1_total = beta * q_stage1_net + pers_bonus
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with asymmetric learning rates.
    Uses different learning rates for positive vs negative prediction errors,
    modeling differences in how participants weigh better-than-expected vs worse-than-expected outcomes.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta: [0, 10] - inverse temperature
    w: [0, 1] - model-based weight
    """
    alpha_pos, alpha_neg, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta * q_stage1_net
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
