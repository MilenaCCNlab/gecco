def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice Trace Model.
    Instead of only perseverating on the immediately preceding action, this model maintains a choice trace 
    that decays over time. This captures longer-term tendencies to prefer one stage 1 action over another.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    p: [-2,2] - choice perseverance weight
    trace_decay: [0,1] - decay factor for the choice trace
    """
    learning_rate, beta1, beta2, w, lam, p, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta1 * (q_stage1_net + p * choice_trace))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1 and reward[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
            
        if a1 != -1:
            choice_trace *= trace_decay
            choice_trace[a1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-Dependent Model-Based Control.
    The mixing weight w (degree of model-based control) depends on whether the previous trial was rewarded.
    This tests if participants rely more on the model after receiving a reward vs. not receiving one.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w_rew: [0,1] - mixing weight for MB vs MF control following a rewarded trial
    w_unrew: [0,1] - mixing weight for MB vs MF control following an unrewarded trial
    lam: [0,1] - eligibility trace parameter
    p: [-2,2] - choice perseverance bonus
    """
    learning_rate, beta1, beta2, w_rew, w_unrew, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        w_current = w_rew if last_reward > 0 else w_unrew

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w_current * q_stage1_mb + (1 - w_current) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1 and reward[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1
            
        if reward[trial] != -1:
            last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-Dependent Dynamic Temperature Model.
    The inverse temperature for stage 1 (beta1) dynamically scales based on a trace of recent rewards.
    This captures the idea that participants might become more exploratory (lower beta) when recent rewards are low,
    and more exploitative (higher beta) when recent rewards are high.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta1_base: [0,10] - baseline inverse temperature for stage 1 choices
    beta_scale: [0,10] - how much recent reward trace increases stage 1 beta
    reward_trace_decay: [0,1] - decay rate for the reward trace
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    p: [-2,2] - choice perseverance bonus
    """
    learning_rate, beta1_base, beta_scale, reward_trace_decay, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    reward_trace = 0.5 # initialize at 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
            
        beta1_eff = beta1_base + beta_scale * reward_trace
        
        exp_q1 = np.exp(beta1_eff * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1 and reward[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1
            
        if reward[trial] != -1:
            reward_trace = reward_trace * reward_trace_decay + reward[trial] * (1 - reward_trace_decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
