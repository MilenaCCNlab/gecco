def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with Transition Matrix Learning.
    The participant dynamically updates their belief about the transition probabilities 
    between Stage 1 spaceships and Stage 2 planets based on experience.
    
    Parameters:
    alpha: [0, 1] - Learning rate for action values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Choice stickiness for stage 1.
    alpha_trans: [0, 1] - Learning rate for transition matrix probabilities.
    """
    alpha, beta1, beta2, w, lam, pi, alpha_trans = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix with default probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Handle missed trials
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits1 = beta1 * q_net_stage1 + pi * stickiness
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update transition matrix
        transition_matrix[a1, s] += alpha_trans * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_trans * (0.0 - transition_matrix[a1, 1 - s])
        
        last_action_1 = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with Reference Point for rewards.
    The participant evaluates the reward relative to a baseline reference point, 
    which can capture risk sensitivity or subjective valuation of zero rewards as negative.
    
    Parameters:
    alpha: [0, 1] - Learning rate for action values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Choice stickiness for stage 1.
    rho: [0, 1] - Reference point subtracted from the reward.
    """
    alpha, beta1, beta2, w, lam, pi, rho = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Handle missed trials
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits1 = beta1 * q_net_stage1 + pi * stickiness
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Subjective reward (shift relative to reference point)
        r_eff = r - rho

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r_eff - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with Choice Kernel (Choice Trace) for stage 1.
    Instead of a simple 1-trial stickiness, the participant maintains a moving average 
    of past choices, allowing for longer-term perseveration or autocorrelation in choices.
    
    Parameters:
    alpha: [0, 1] - Learning rate for action values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pi_c: [-5, 5] - Weight of the choice kernel in stage 1.
    alpha_c: [0, 1] - Learning rate / decay rate of the choice kernel.
    """
    alpha, beta1, beta2, w, lam, pi_c, alpha_c = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_kernel = np.zeros(2)
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Handle missed trials
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits1 = beta1 * q_net_stage1 + pi_c * choice_kernel
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update choice kernel (exponentially weighted moving average)
        choice_kernel[a1] += alpha_c * (1.0 - choice_kernel[a1])
        choice_kernel[1 - a1] += alpha_c * (0.0 - choice_kernel[1 - a1])
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
