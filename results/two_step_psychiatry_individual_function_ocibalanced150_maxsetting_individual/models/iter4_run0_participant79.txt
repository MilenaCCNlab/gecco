def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model-based and model-free reinforcement learning model with a decay 
    parameter for unchosen actions. In this model, unchosen Q-values decay 
    towards a free parameter 'center' rather than 0, accounting for a baseline 
    expectation or uncertainty over time.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for model-based vs model-free control.
    lam: [0, 1] - Eligibility trace parameter determining how much stage 2 
                  prediction errors update stage 1 values.
    decay: [0, 1] - Rate at which unchosen Q-values decay towards the center.
    center: [0, 1] - Baseline value towards which Q-values decay.
    """
    alpha, beta, w, lam, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to the center value
    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay unchosen actions towards center
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * center
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * center

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):

    Bounds:

    w: [0, 1] - Weighting parameter for model-based vs model-free control.

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model-based and model-free reinforcement learning model with 
    asymmetric learning rates for positive and negative prediction errors, 
    and a decay parameter for unchosen actions.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for model-based vs model-free control.
    lam: [0, 1] - Eligibility trace parameter.

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)

    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
