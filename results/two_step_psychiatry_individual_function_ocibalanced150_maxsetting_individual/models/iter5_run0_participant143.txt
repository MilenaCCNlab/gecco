def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Choice Kernel model. Adds a choice kernel (exponentially smoothed action history) 
    to both stage 1 and stage 2 decisions, separate from value-based reinforcement learning.
    Bounds:
    alpha: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    alpha_ck: [0, 1]
    beta_ck1: [0, 10]
    beta_ck2: [0, 10]
    """
    alpha, beta_1, beta_2, w, lam, alpha_ck, beta_ck1, beta_ck2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + beta_ck1 * ck_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx] + beta_ck2 * ck_2[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # update for choice 2 and eligibility trace for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Choice kernel updates
        ck_1 *= (1 - alpha_ck)
        ck_1[action_1[trial]] += alpha_ck
        
        ck_2 *= (1 - alpha_ck)
        ck_2[state_idx, action_2[trial]] += alpha_ck

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Transition-dependent perseveration model. The stickiness to the previous stage 1 choice 
    depends on whether the previous trial's transition was common or rare.
    Bounds:
    alpha: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p_common: [0, 5]
    p_rare: [0, 5]
    """
    alpha, beta_1, beta_2, w, lam, p_common, p_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_trans_common = False

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if last_a1 != -1:
            if last_trans_common:
                logits_1[last_a1] += p_common
            else:
                logits_1[last_a1] += p_rare
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # update for choice 2 and eligibility trace for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        last_a1 = action_1[trial]
        # Common transition is when chosen spaceship matches state
        last_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model with asymmetric learning rates for positive/negative prediction errors, 
    plus a static bias for stage 1 choices to capture baseline preferences.
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p: [0, 5]
    bias_1: [-5, 5]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if last_a1 != -1:
            logits_1[last_a1] += p
            
        # Add static bias for spaceship 0
        logits_1[0] += bias_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        # update for choice 1
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1 + lr2 * lam * delta_stage2
        # update for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        last_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
