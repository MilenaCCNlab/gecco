def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition-dependent perseveration.
    Perseveration at stage 1 depends on whether the previous trial's transition was common or rare.
    Participants may exhibit different stickiness to an action based on whether it led to the expected state.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weight on model-based vs model-free Q-values for stage 1 (1 = purely MB, 0 = purely MF).
    lambd: [0,1] - Eligibility trace parameter carrying stage 2 prediction errors to stage 1.
    p_common: [-5,5] - Perseveration bonus if the previous transition was common.
    p_rare: [-5,5] - Perseveration bonus if the previous transition was rare.
    """
    alpha, beta, w, lambd, p_common, p_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_transition_common = False

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            if last_transition_common:
                q_net[last_action_1] += p_common
            else:
                q_net[last_action_1] += p_rare
                
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        last_action_1 = action_1[trial]
        # Transition is common if action matches the planet state (e.g., action 0 -> state 0)
        last_transition_common = (action_1[trial] == state[trial])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-dependent model-based weight.
    The balance between model-based and model-free control depends on whether the previous trial was rewarded.
    Participants might rely more on model-based planning after a reward (or vice versa).
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w_rewarded: [0,1] - Weight on MB vs MF Q-values if the previous trial was rewarded.
    w_unrewarded: [0,1] - Weight on MB vs MF Q-values if the previous trial was unrewarded.
    lambd: [0,1] - Eligibility trace parameter carrying stage 2 prediction errors to stage 1.
    p: [-5,5] - Perseveration bonus for the previously chosen stage 1 action.
    """
    alpha, beta, w_rewarded, w_unrewarded, lambd, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine the model-based weight based on the reward of the previous trial
        w = w_rewarded if last_reward > 0 else w_unrewarded
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Choice trace perseveration.
    Instead of a 1-trial perseveration, a choice trace accumulates over time, capturing longer-term stickiness.
    This helps explain robust repetitive behaviors that persist beyond just the immediately preceding trial.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weight on model-based vs model-free Q-values for stage 1.
    lambd: [0,1] - Eligibility trace parameter carrying stage 2 prediction errors to stage 1.
    p: [-5,5] - Perseveration weight applied to the choice trace.
    alpha_c: [0,1] - Update rate for the choice trace.
    """
    alpha, beta, w, lambd, p, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add the perseveration bonus proportional to the accumulated choice trace
        q_net += p * choice_trace
            
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        # Update the choice trace for the chosen action
        choice_trace *= (1 - alpha_c)
        choice_trace[action_1[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
