Here are 3 new cognitive models, each implementing different theoretical mechanisms to capture the behavior observed in the two-step decision task.

### Model 1: Asymmetric Learning Rates + Stage 1 & Stage 2 Perseveration
This model tests the hypothesis that the participant learns differently from positive versus negative prediction errors (e.g., they might be more sensitive to missing a reward than receiving one). It also includes choice perseveration for both decision stages.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates with Stage 1 & Stage 2 Perseveration.
    
    Parameters:
    lr_pos: [0,1] - learning rate for positive prediction errors
    lr_neg: [0,1] - learning rate for negative prediction errors
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence
    lam: [0,1] - eligibility trace parameter
    pers_st1: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    pers_st2: [-5,5] - choice perseveration bonus for repeating the previous stage 2 action (in the same state)
    """
    lr_pos, lr_neg, beta, w, lam, pers_st1, pers_st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action2 = {-1: -1, 0: -1, 1: -1}

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if trial > 0:
            pers_bonus_1[action_1[trial-1]] = pers_st1
            
        exp_q1 = np.exp(beta * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            pers_bonus_2[last_action2[state_idx]] = pers_st2
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        last_action2[state_idx] = action_2[trial]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        lr_s1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_s1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        lr_s2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_s2 * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += lr_s2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Stage 1 & Stage 2 Inverse Temperatures + Perseveration + Decay
This model tests the hypothesis that the participant exhibits different levels of exploration/exploitation (noise) at the first stage (spaceship selection) compared to the second stage (alien selection), while unchosen values gradually decay back to a neutral baseline.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate Stage 1 and Stage 2 Inverse Temperatures with Perseveration and Decay.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating chosen Q-values
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - weight of model-based vs model-free influence
    lam: [0,1] - eligibility trace parameter
    pers_st1: [-5,5] - choice perseveration bonus for stage 1
    pers_st2: [-5,5] - choice perseveration bonus for stage 2
    decay: [0,1] - rate at which unchosen Q-values decay towards 0.5
    """
    learning_rate, beta1, beta2, w, lam, pers_st1, pers_st2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action2 = {-1: -1, 0: -1, 1: -1}

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if trial > 0:
            pers_bonus_1[action_1[trial-1]] = pers_st1
            
        exp_q1 = np.exp(beta1 * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            pers_bonus_2[last_action2[state_idx]] = pers_st2
            
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        last_action2[state_idx] = action_2[trial]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay for unchosen actions
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Dynamic Transition Learning + Perseveration + Decay
This model expands on the best model so far by letting the participant dynamically learn and update the transition matrix based on their observations (rather than assuming it stays fixed at 0.7/0.3).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF RL with Stage 1 & Stage 2 Perseveration, Decay, and Dynamic Transition Learning.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating chosen Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence
    lam: [0,1] - eligibility trace parameter
    pers_st1: [-5,5] - choice perseveration bonus for stage 1
    pers_st2: [-5,5] - choice perseveration bonus for stage 2
    decay: [0,1] - rate at which unchosen Q-values decay towards 0.5
    lr_trans: [0,1] - learning rate for transition probabilities
    """
    learning_rate, beta, w, lam, pers_st1, pers_st2, decay, lr_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action2 = {-1: -1, 0: -1, 1: -1}

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if trial > 0:
            pers_bonus_1[action_1[trial-1]] = pers_st1
            
        exp_q1 = np.exp(beta * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Update transition matrix based on observed state
        transition_matrix[action_1[trial], state_idx] += lr_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += lr_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            pers_bonus_2[last_action2[state_idx]] = pers_st2
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        last_action2[state_idx] = action_2[trial]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay for unchosen actions
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
