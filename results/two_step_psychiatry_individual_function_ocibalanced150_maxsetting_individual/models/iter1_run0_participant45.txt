def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw 2011 hybrid model with eligibility trace, stage 1 perseveration, and stage 2 perseveration.
    This model captures response stickiness at both the first and second stages, as participants often 
    repeat the same second-stage choice when encountering the same planet.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    pi1: Perseveration bonus for stage 1 [0, 5]
    pi2: Perseveration bonus for stage 2 [0, 5]
    lambd: Eligibility trace [0, 1]
    """
    alpha, beta1, beta2, w, pi1, pi2, lambd = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2_state = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action1 = -1
            continue

        state_idx = state[trial]

        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Stage 1 Net Values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stage 1 Perseveration
        rep1 = np.zeros(2)
        if last_action1 != -1:
            rep1[last_action1] = 1.0
            
        # Policy for the first choice
        logits1 = beta1 * (q_stage1_net + pi1 * rep1)
        logits1 -= np.max(logits1) # for numerical stability
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Stage 2 Perseveration
        rep2 = np.zeros(2)
        if last_action2_state[state_idx] != -1:
            rep2[last_action2_state[state_idx]] = 1.0

        # Policy for the second choice
        logits2 = beta2 * (q_stage2_mf[state_idx] + pi2 * rep2)
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw 2011 hybrid model with eligibility trace, perseveration, and value decay (forgetting).
    This model assumes that values of unchosen actions gradually decay towards zero, capturing 
    memory decay mechanisms which often influence sequential decision-making.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    pi: Perseveration bonus for stage 1 [0, 5]
    lambd: Eligibility trace [0, 1]
    decay: Decay rate for unchosen action values [0, 1]
    """
    alpha, beta1, beta2, w, pi, lambd, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        state_idx = state[trial]

        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Stage 1 Net Values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stage 1 Perseveration
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        # Policy for the first choice
        logits1 = beta1 * (q_stage1_net + pi * rep)
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        # Decay unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a2 in range(2):
                if not (s == state_idx and a2 == action_2[trial]):
                    q_stage2_mf[s, a2] *= (1 - decay)

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw 2011 hybrid model with eligibility trace, perseveration, and a stage 1 choice bias.
    This model incorporates an inherent spatial or object bias at the first stage, which is commonly 
    observed in decision-making tasks where participants favor one side/option independent of value.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    pi: Perseveration bonus for stage 1 [0, 5]
    lambd: Eligibility trace [0, 1]
    bias1: Bias for stage 1 action 1 vs action 0 [-5, 5]
    """
    alpha, beta1, beta2, w, pi, lambd, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        state_idx = state[trial]

        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Stage 1 Net Values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stage 1 Perseveration
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        # Stage 1 Bias
        bias_vec = np.array([0.0, bias1])
        
        # Policy for the first choice
        logits1 = beta1 * (q_stage1_net + pi * rep + bias_vec)
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
