def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Single Alpha, Decay, Bias, and Pers_Win.
    Uses a single learning rate for both stages to free up a parameter for a choice bias.
    Includes decay of unchosen options towards 0, and a perseveration bonus for rewarded previous actions.
    This model explicitly accounts for the participant's strong preference for a specific spaceship.
    
    Parameters:
    alpha: learning rate for both stages [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-2,2]
    decay: decay rate for unchosen actions [0,1]
    bias: constant bias added to action 0 in stage 1 [-5,5]
    """
    alpha, beta_1, beta_2, w, lam, pers_win, decay, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1 and last_reward > 0:
            pers_bonus[last_action] = 1.0
            
        bias_bonus = np.zeros(2)
        bias_bonus[0] = bias
                
        q_net_total = beta_1 * q_net + pers_win * pers_bonus + bias_bonus
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] *= (1 - decay)
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
            
            other_state = 1 - state_idx
            q_stage2_mf[other_state, 0] *= (1 - decay)
            q_stage2_mf[other_state, 1] *= (1 - decay)
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Decay towards Neutral Expectation (0.5).
    Similar to the best baseline model, but instead of decaying unchosen action values to 0, 
    they decay towards 0.5. This represents an assumption that unknown options have a 50% chance of reward,
    which is a more realistic default expectation in a binary reward task.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-2,2]
    decay_center: decay rate for unchosen actions towards 0.5 [0,1]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win, decay_center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1 and last_reward > 0:
            pers_bonus[last_action] = 1.0
                
        q_net_total = beta_1 * q_net + pers_win * pers_bonus
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] = (1 - decay_center) * q_stage1_mf[unchosen_1] + decay_center * 0.5
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] = (1 - decay_center) * q_stage2_mf[state_idx, unchosen_2] + decay_center * 0.5
            
            other_state = 1 - state_idx
            q_stage2_mf[other_state, 0] = (1 - decay_center) * q_stage2_mf[other_state, 0] + decay_center * 0.5
            q_stage2_mf[other_state, 1] = (1 - decay_center) * q_stage2_mf[other_state, 1] + decay_center * 0.5
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Stage-specific Reward Perseveration.
    Includes separate perseveration bonuses for stage 1 (spaceship) and stage 2 (alien) choices,
    applied only when the previous trial was rewarded. The stage 2 perseveration is state-dependent.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win_1: stage 1 perseveration bonus if previous trial was rewarded [-2,2]
    pers_win_2: stage 2 perseveration bonus if previous trial was rewarded and landed in same state [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win_1, pers_win_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward > 0:
            pers_bonus_1[last_action_1] = 1.0
                
        q_net_total = beta_1 * q_net + pers_win_1 * pers_bonus_1
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_reward > 0 and state_idx == last_state and last_action_2 != -1:
                pers_bonus_2[last_action_2] = 1.0

            q_stage2_total = beta_2 * q_stage2_mf[state_idx] + pers_win_2 * pers_bonus_2
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            last_reward = reward[trial]
            last_action_2 = action_2[trial]
            last_state = state_idx
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            last_action_2 = -1
            last_state = -1
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
