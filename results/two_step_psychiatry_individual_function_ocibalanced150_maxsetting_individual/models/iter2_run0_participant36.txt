Here are 3 new cognitive models that build upon the best model so far by introducing new mechanisms, while avoiding parameter configurations that have already been tested.

### Model 1: Hybrid Model with Stage 1 Action Bias
This model introduces a `bias` parameter to account for any spatial or action-specific preference at Stage 1 (e.g., a tendency to prefer spaceship 0 over spaceship 1 regardless of values).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Daw model with an added bias for action 0 at Stage 1.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    st: [0, 10] - choice stickiness
    bias: [-5, 5] - bias for action 0 at stage 1
    """
    alpha, beta1, beta2, w, lam, st, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1
            
        action_bias = np.array([bias, 0.0])
            
        logits_1 = beta1 * q_net + st * stickiness + action_bias
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates + Forgetting (Decay)
This model combines the best-performing decay mechanism with asymmetric learning rates, allowing participants to learn differently from positive versus negative prediction errors (e.g., being more sensitive to better-than-expected vs. worse-than-expected outcomes).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with asymmetric learning rates (alpha_pos, alpha_neg) and decay of unchosen options.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    decay: [0, 1] - decay rate for unchosen options
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    st: [0, 10] - choice stickiness
    """
    alpha_pos, alpha_neg, decay, beta1, beta2, w, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1
            
        logits_1 = beta1 * q_net + st * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Forgetting (Decay)
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # Asymmetric Learning Rate Selection
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg

        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1 + alpha_2 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Stickiness + Forgetting (Decay)
This model combines decay with `st_win` and `st_lose`, testing the hypothesis that choice perseveration (stickiness) is modulated by whether the previous trial was rewarded or unrewarded, in addition to values decaying over time.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with forgetting (decay) and outcome-dependent stickiness (st_win, st_lose).
    
    Bounds:
    alpha: [0, 1] - learning rate
    decay: [0, 1] - decay rate for unchosen options
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    st_win: [0, 10] - choice stickiness after a rewarded trial
    st_lose: [0, 10] - choice stickiness after an unrewarded trial
    """
    alpha, decay, beta1, beta2, w, lam, st_win, st_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            if last_reward == 1:
                stickiness[last_action] = st_win
            else:
                stickiness[last_action] = st_lose
            
        logits_1 = beta1 * q_net + stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Forgetting (Decay)
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
