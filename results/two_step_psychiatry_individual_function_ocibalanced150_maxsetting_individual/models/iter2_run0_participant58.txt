def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with separate learning rates and inverse temperatures for each stage, 
    and perseveration only at the first stage.
    
    Parameters:
    alpha1: [0,1] - learning rate for stage 1 value updates.
    alpha2: [0,1] - learning rate for stage 2 value updates.
    beta1: [0,10] - inverse temperature for softmax action selection at stage 1.
    beta2: [0,10] - inverse temperature for softmax action selection at stage 2.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p: [-5,5] - perseveration bonus added to the previously chosen first-stage action.
    """
    alpha1, alpha2, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
            
        q_net_pers1 = beta1 * q_stage1_net + pers
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        q_net2 = beta2 * q2
        exp_q2 = np.exp(q_net2 - np.max(q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with asymmetric learning rates for positive and negative prediction errors, 
    and choice perseveration at both stages.
    
    Parameters:
    alpha_pos: [0,1] - learning rate for positive prediction errors (delta > 0).
    alpha_neg: [0,1] - learning rate for negative prediction errors (delta <= 0).
    beta: [0,10] - inverse temperature for softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p1: [-5,5] - perseveration bonus added to the previously chosen first-stage action.
    p2: [-5,5] - perseveration bonus added to the previously chosen second-stage action (per state).
    """
    alpha_pos, alpha_neg, beta, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = p1
            
        q_net_pers1 = beta * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = p2
            
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta * q2 + pers2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1_current = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1_current * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2_current = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_current * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2_current * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with perseveration at both stages and a value decay parameter 
    for unchosen actions.
    
    Parameters:
    alpha: [0,1] - learning rate for value updates.
    beta: [0,10] - inverse temperature for softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p1: [-5,5] - perseveration bonus added to the previously chosen first-stage action.
    p2: [-5,5] - perseveration bonus added to the previously chosen second-stage action (per state).
    decay: [0,1] - decay rate for unchosen actions (their values decay towards 0).
    """
    alpha, beta, w, lam, p1, p2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = p1
            
        q_net_pers1 = beta * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = p2
            
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta * q2 + pers2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # decay for unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
