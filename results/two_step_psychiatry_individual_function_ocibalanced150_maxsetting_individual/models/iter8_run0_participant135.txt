import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Differentiates perseveration based on reward and transition type.
    This model posits that the tendency to repeat a rewarded action might depend on whether 
    the transition that led to the reward was common or rare, testing if participants 
    differentiate these experiences.
    
    Parameters:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    pers_rc: [0, 5] - Perseveration if last trial was rewarded and transition was common
    pers_rr: [0, 5] - Perseveration if last trial was rewarded and transition was rare
    pers_unrew: [0, 5] - Perseveration if last trial was unrewarded
    """
    alpha, beta1, beta2, w, lam, pers_rc, pers_rr, pers_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0
    last_trans_common = False

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                if last_trans_common:
                    pers_bonus[last_action1] = pers_rc
                else:
                    pers_bonus[last_action1] = pers_rr
            else:
                pers_bonus[last_action1] = pers_unrew
                
        logits1 = beta1 * q_net1 + pers_bonus
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 policy
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]
        # Spaceship 0 -> planet 0 is common, Spaceship 1 -> planet 1 is common
        last_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Model-based weight (w) depends on the reward of the previous trial.
    This model tests the hypothesis that positive or negative reinforcement modulates 
    cognitive flexibility and the balance between model-based and model-free control.
    
    Parameters:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    w_win: [0, 1] - Model-based weight if last trial was rewarded
    w_loss: [0, 1] - Model-based weight if last trial was unrewarded
    lam: [0, 1] - Eligibility trace
    pers_rew: [0, 5] - Perseveration if last trial was rewarded
    pers_unrew: [0, 5] - Perseveration if last trial was unrewarded
    """
    alpha, beta1, beta2, w_win, w_loss, lam, pers_rew, pers_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        current_w = w_win if last_reward == 1 else w_loss
        q_net1 = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                pers_bonus[last_action1] = pers_rew
            else:
                pers_bonus[last_action1] = pers_unrew
                
        logits1 = beta1 * q_net1 + pers_bonus
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 policy
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dynamic Model-based control with asymmetric learning rates.
    Combines reward-dependent model-based weighting (w_win / w_loss) with 
    separate learning rates for positive and negative prediction errors.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    w_win: [0, 1] - Model-based weight if last trial was rewarded
    w_loss: [0, 1] - Model-based weight if last trial was unrewarded
    lam: [0, 1] - Eligibility trace
    pers: [0, 5] - Stage 1 perseveration bonus
    """
    alpha_pos, alpha_neg, beta1, beta2, w_win, w_loss, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        current_w = w_win if last_reward == 1 else w_loss
        q_net1 = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action1 != -1:
            pers_bonus[last_action1] = pers
                
        logits1 = beta1 * q_net1 + pers_bonus
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 policy
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
