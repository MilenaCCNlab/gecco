Here are 3 new cognitive models that explore different combinations of learning and choice mechanisms. They build upon the insights from the previously tested models while avoiding exact duplications of the parameter sets.

### Model 1: Positive/Negative Learning Rates with Dual-Stage Stickiness
This model uses shared positive and negative learning rates across both stages (to capture differential sensitivity to better-than-expected vs. worse-than-expected outcomes) but introduces a stickiness parameter for **both** Stage 1 and Stage 2 choices. Stage 2 stickiness tracks the last action chosen in the *currently visited* state.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with positive and negative learning rates (shared across stages),
    stage 1 stickiness, and stage 2 stickiness.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p1: [-5, 5] - stage 1 stickiness (tendency to repeat previous stage 1 action)
    p2: [-5, 5] - stage 2 stickiness (tendency to repeat last action taken in current state)
    """
    import numpy as np
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = [-1, -1]

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep1 = np.zeros(2)
        if last_action_1 != -1:
            rep1[last_action_1] = 1
            
        q1 = beta1 * q_net + p1 * rep1
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if last_action_2_in_state[state_idx] != -1:
            rep2[last_action_2_in_state[state_idx]] = 1

        q2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # calculate prediction errors
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # determine learning rates based on sign of PE
        alpha1_current = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha2_current = alpha_pos if delta_stage2 > 0 else alpha_neg

        # action value updating
        q_stage1_mf[action_1[trial]] += alpha1_current * delta_stage1 + alpha1_current * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_current * delta_stage2
        
        # update trackers
        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Stage-Specific Asymmetric Learning Rates
This model explores the idea that asymmetric updating (positive vs. negative prediction errors) might be specific to the first stage where model-free and model-based systems interact, while using a standard single learning rate for the terminal model-free stage 2 choices.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate positive and negative learning rates for stage 1, 
    a single learning rate for stage 2, and basic stage 1 stickiness.
    
    Bounds:
    alpha1_pos: [0, 1] - learning rate for positive prediction errors at stage 1
    alpha1_neg: [0, 1] - learning rate for negative prediction errors at stage 1
    alpha2: [0, 1] - learning rate for stage 2
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p: [-5, 5] - stage 1 stickiness
    """
    import numpy as np
    alpha1_pos, alpha1_neg, alpha2, beta1, beta2, lam, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1

        q1 = beta1 * q_net + p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # calculate prediction errors
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1_current = alpha1_pos if delta_stage1 > 0 else alpha1_neg

        # action value updating
        q_stage1_mf[action_1[trial]] += alpha1_current * delta_stage1 + alpha1_current * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Reward-Transition Interaction Stickiness
This model captures model-based behavior directly through choice perseveration. It differentiates stage 1 stickiness not only based on whether the previous trial was rewarded, but also on whether the rewarded trial followed a common vs. rare transition (an established heuristic strategy to mimic model-based planning).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with stickiness at stage 1 dependent on whether the previous trial 
    was rewarded after a common transition, rewarded after a rare transition, 
    or unrewarded.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p_win_common: [-5, 5] - stage 1 stickiness if prev trial rewarded & common trans
    p_win_rare: [-5, 5] - stage 1 stickiness if prev trial rewarded & rare trans
    p_lose: [-5, 5] - stage 1 stickiness if prev trial unrewarded
    """
    import numpy as np
    alpha, beta1, beta2, lam, w, p_win_common, p_win_rare, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0
    last_trans_common = False

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1

        # determine stickiness based on previous reward and transition type
        if last_reward > 0:
            current_p = p_win_common if last_trans_common else p_win_rare
        else:
            current_p = p_lose
            
        q1 = beta1 * q_net + current_p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # calculate prediction errors
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # action value updating
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # update trackers for next trial
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        # Common transitions in this task structure align with action matching state
        last_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
