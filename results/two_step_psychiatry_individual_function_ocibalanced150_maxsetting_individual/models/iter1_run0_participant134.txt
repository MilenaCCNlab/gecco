def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learner with separate learning rates for Stage 1 and Stage 2.
    Includes eligibility trace and stage 1 choice stickiness.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1 Q-values
    alpha_2: [0, 1] - Learning rate for stage 2 Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1 (exploration/exploitation)
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter (how much stage 2 reward updates stage 1)
    stickiness: [-5, 5] - Choice stickiness for stage 1 (tendency to repeat last choice)
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits_1 = beta_1 * q_net + stickiness * rep
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learner with asymmetric learning rates for positive 
    and negative prediction errors. Includes eligibility trace and choice stickiness.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    stickiness: [-5, 5] - Choice stickiness for stage 1
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1

        logits_1 -= np.max(logits_1)

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        lr_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # action value updating for choice 2
        lr_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += lr_2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learner with decay for unchosen options.
    Includes eligibility trace and choice stickiness.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating chosen Q-values
    decay_rate: [0, 1] - Decay rate for unchosen Q-values pulling them towards 0.5
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    stickiness: [-5, 5] - Choice stickiness for stage 1
    """
    learning_rate, decay_rate, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 for decay model
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits_1 = beta_1 * q_net + stickiness * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # decay for unchosen options towards 0.5
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
