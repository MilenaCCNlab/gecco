import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Unchosen Option Decay, Eligibility Trace, and Reward-Dependent Stickiness.
    
    This model separates Stage 1 choice stickiness based on whether the previous trial
    was rewarded (win-stay) or unrewarded (lose-shift), which often captures participant 
    perseveration better than a single stickiness parameter.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen Q-value updates.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    beta: [0, 10] - Inverse temperature for choice probabilities.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    eligibility: [0, 1] - Eligibility trace (lambda) for updating Stage 1 values from Stage 2 prediction errors.
    stickiness_rew: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice if the previous trial was rewarded.
    stickiness_unrew: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice if the previous trial was unrewarded.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    learning_rate, decay_rate, beta, w, eligibility, stickiness_rew, stickiness_unrew, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])
    prev_reward = -1

    for trial in range(n_trials):
        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MB and MF values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply reward-dependent Stage 1 stickiness
        q1_eff = q_stage1_net.copy()
        if prev_a1 != -1:
            if prev_reward == 1:
                q1_eff[prev_a1] += stickiness_rew
            else:
                q1_eff[prev_a1] += stickiness_unrew
            
        # Stage 1 Choice Probability
        exp_q1 = np.exp(beta * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Apply Stage 2 stickiness
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        # Stage 2 Choice Probability
        exp_q2 = np.exp(beta * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen/unvisited options towards 0.5
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])
        
        unchosen_a2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_a2] += decay_rate * (0.5 - q_stage2_mf[state_idx, unchosen_a2])
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay_rate * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay_rate * (0.5 - q_stage2_mf[unvisited_state, 1])

        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility Trace Update for Stage 1 MF
        q_stage1_mf[action_1[trial]] += eligibility * learning_rate * delta_stage2
        
        # Track previous actions and reward
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]
        prev_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Unchosen Option Decay, Eligibility Trace, Stickiness, and Lapse Rate.
    
    This model introduces an epsilon parameter to account for random responding (lapses in attention)
    that can otherwise distort maximum likelihood estimates of learning rates and temperatures.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen Q-value updates.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    beta: [0, 10] - Inverse temperature for choice probabilities.
    epsilon: [0, 1] - Lapse rate (probability of making a completely random choice).
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    eligibility: [0, 1] - Eligibility trace (lambda) for updating Stage 1 values from Stage 2 prediction errors.
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    learning_rate, decay_rate, beta, epsilon, w, eligibility, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

        # Value Updates
