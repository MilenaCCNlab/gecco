Here are 3 cognitive models that explain the participant's behavior in the two-step task. 

### Model 1: Hybrid Model-Free and Model-Based Control with Eligibility Trace and Stickiness
This model implements the standard Daw et al. (2011) two-step task model. It assumes the participant uses a mixture of model-free and model-based learning, weighted by $w$. It also includes an eligibility trace ($\lambda$) to allow the second-stage reward to directly update the first-stage action value, and a stickiness parameter to capture choice perseveration (the tendency to repeat the previous trial's first-stage choice regardless of reward).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Free and Model-Based control (Daw et al., 2011).
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free)
    lam: [0, 1] - Eligibility trace parameter for stage 2 reward updating stage 1
    stickiness: [-5, 5] - Choice perseveration for stage 1
    """
    learning_rate, beta, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness
            
        q1_scaled = beta * q_stage1_net
        q1_scaled -= np.max(q1_scaled) # Prevent overflow
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action_1 = action_1[trial]
            continue

        # policy for the second choice
        q2_scaled = beta * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Asymmetric Learning Rates
Participants often exhibit different learning dynamics depending on whether the outcome was better or worse than expected. This model separates the learning rate into `alpha_pos` (for positive prediction errors) and `alpha_neg` (for negative prediction errors). It also allows for separate inverse temperatures (`beta_1` and `beta_2`) for the two stages, as decision noise often varies between the spaceship choice and the alien choice.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with asymmetric learning rates.
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weight on model-based vs model-free control
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_scaled = beta_1 * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        # Action value updating
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Q-Value Decay (Forgetting)
In environments where reward probabilities drift over time (like the aliens changing their payout odds), participants may actively discount the value of unchosen options, assuming they are slowly drifting back to a baseline. This model introduces a `decay_rate` parameter, which pulls the Q-values of all unchosen actions towards 0 on every trial.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with Q-value decay for unchosen actions.
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen action updates
    beta: [0, 10] - Inverse temperature for softmax
    w: [0, 1] - Weight on model-based vs model-free control
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0
    """
    learning_rate, beta, w, decay_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_scaled = beta * q_stage1_net
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue

        # policy for the second choice
        q2_scaled = beta * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Update chosen actions
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Decay unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay_rate)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_rate)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
