def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate Learning Rates for Stage 1 and Stage 2 with Forgetting Rate.
    
    This model extends the standard two-step model by allowing different learning rates 
    for the first and second stages, which is useful if participants update their 
    model-free values at different speeds for spaceships vs. aliens. It also includes 
    a forgetting rate for unchosen options.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence at Stage 1.
    lam: [0, 1] - Eligibility trace parameter (updates Stage 1 action with Stage 2 reward).
    pers: [-5, 5] - Perseveration bonus added to the previously chosen Stage 1 action.
    forgetting_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, forgetting_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        # Forgetting for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += forgetting_rate * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += forgetting_rate * (0.5 - q_stage2_mf[s, a])

        # Action value updating for choice 1
        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        # Action value updating for choice 2
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-Step Model with Counterfactual/Fictive Updating.
    
    This model assumes that participants might infer the payout of the unchosen alien 
    on the visited planet to be anti-correlated with the chosen alien. It updates the 
    unchosen Stage 2 action using the opposite of the received reward.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen actions.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration bonus added to the previously chosen Stage 1 action.
    fictive_lr: [0, 1] - Learning rate for counterfactual update of unchosen Stage 2 action.
    """
    learning_rate, beta1, beta2, w, lam, pers, fictive_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        # Action value updating for choice 1
        q_stage1_mf[a1] += learning_rate * delta_stage1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Action value updating for choice 2
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Counterfactual update for unchosen Stage 2 action
        unchosen_a2 = 1 - a2
        fictive_reward = 1 - reward[trial]
        delta_fictive = fictive_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += fictive_lr * delta_fictive
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Learning Rates for Positive/Negative PEs with Forgetting.
    
    This model tests whether participants learn differently from better-than-expected 
    outcomes versus worse-than-expected outcomes, while also incorporating the 
    forgetting mechanism for unchosen options.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (PE > 0).
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (PE <= 0).
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration bonus added to the previously chosen Stage 1 action.
    forgetting_rate: [0, 1] - Decay rate for unchosen options towards 0.5.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers, forgetting_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        # Forgetting for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += forgetting_rate * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += forgetting_rate * (0.5 - q_stage2_mf[s, a])

        # Action value updating for choice 1
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += lr1 * delta_stage1
        
        # Action value updating for choice 2
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += lr2 * delta_stage2
        
        # Eligibility trace update using Stage 2 PE
        q_stage1_mf[a1] += lr2 * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
