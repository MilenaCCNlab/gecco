Here are 3 new cognitive models that explore different psychological mechanisms for learning and decision-making in the two-step task. They build on the most successful features (like separate MB/MF temperatures and initialization to 0.5) while introducing novel components such as counterfactual updating, asymmetric eligibility traces combined with decay, and uncertainty-driven exploration.

### Model 1: Counterfactual Updating
This model assumes that participants might infer the outcome of the unchosen alien based on the chosen one. Since the probabilities are independent in reality, this is a cognitive bias, but a common one (e.g., "If I got a coin here, the other one probably wouldn't have given me one"). It introduces an `alpha_cf` parameter to learn from these inferred counterfactuals.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with counterfactual updating at Stage 2. 
    Participants may assume that if one alien gave a reward, the other would not (and vice versa).
    
    Bounds:
    alpha: [0,1] - learning rate for chosen options
    alpha_cf: [0,1] - learning rate for counterfactual updating of unchosen stage 2 option
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - choice perseverance for stage 1
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha, alpha_cf, beta_mb, beta_mf, beta2, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_stage2 = np.array([-1, -1])

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Counterfactual update for the unchosen action in the same state
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += alpha_cf * delta_cf

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Asymmetric Eligibility Traces + Decay
This model modifies the best model so far by introducing asymmetric eligibility traces (`lam_pos` and `lam_neg`). This captures the idea that a participant's Stage 1 choices might be more strongly reinforced by surprising rewards than they are punished by surprising omissions (or vice versa), while maintaining the highly successful decay mechanism for unchosen options.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate MB/MF inverse temperatures, decay, and asymmetric eligibility traces 
    for positive and negative prediction errors at stage 2.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    decay: [0,1] - decay rate towards 0.5 for unchosen options
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam_pos: [0,1] - eligibility trace for positive prediction errors
    lam_neg: [0,1] - eligibility trace for negative prediction errors
    pers1: [-5,5] - choice perseverance for stage 1
    """
    alpha, decay, beta_mb, beta_mf, beta2, lam_pos, lam_neg, pers1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exponent2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Asymmetric eligibility trace update
        if delta_stage2 > 0:
            q_stage1_mf[action_1[trial]] += alpha * lam_pos * delta_stage2
        else:
            q_stage1_mf[action_1[trial]] += alpha * lam_neg * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Uncertainty-Driven Exploration
Since the alien reward probabilities change slowly over time, participants might exhibit directed exploration towards aliens they haven't sampled recently. This model tracks the number of visits to a planet since each alien was last chosen and adds a bonus/penalty (`w_unc`) to the unchosen options.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate MB/MF inverse temperatures, and an uncertainty bonus 
    (or penalty) for Stage 2 options based on how many visits to the state since they were last chosen.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - choice perseverance for stage 1
    pers2: [-5,5] - choice perseverance for stage 2
    w_unc: [-5,5] - weight for uncertainty (time since last chosen) at stage 2
    """
    alpha, beta_mb, beta_mf, beta2, lam, pers1, pers2, w_unc = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_stage2 = np.array([-1, -1])
    
    # Tracks visits to a state since an action was last taken
    time_since_chosen_stage2 = np.zeros((2, 2))

    for trial in range(n_trials):
        
        state_idx = state[trial]
        time_since_chosen_stage2[state_idx] += 1

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Policy incorporates uncertainty based on time since choice
        exponent2 = beta2 * q_stage2_mf[state_idx] + w_unc * time_since_chosen_stage2[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]
        time_since_chosen_stage2[state_idx, action_2[trial]] = 0

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
