import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with different learning rates for stage 1 and stage 2, 
    and perseveration for both stage 1 and stage 2 choices.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for stage 1
    alpha2: [0, 1] - Learning rate for stage 2
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace
    pers1: [-5, 5] - Perseveration parameter for stage 1
    pers2: [-5, 5] - Perseveration parameter for stage 2
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = [-1, -1]

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep1 = np.zeros(2)
        if last_action1 != -1:
            rep1[last_action1] = 1.0
            
        logits1 = beta1 * q_net + pers1 * rep1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        rep2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            rep2[last_action2[state_idx]] = 1.0
            
        logits2 = beta2 * q_stage2_mf[state_idx] + pers2 * rep2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with reward-dependent stage 1 perseveration (win-stay vs lose-shift) 
    and stage 2 perseveration.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace
    pers_win: [-5, 5] - Stage 1 perseveration after a rewarded trial
    pers_lose: [-5, 5] - Stage 1 perseveration after an unrewarded trial
    pers2: [-5, 5] - Stage 2 perseveration
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = [-1, -1]
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep1 = np.zeros(2)
        if last_action1 != -1:
            rep1[last_action1] = 1.0
            
        pers1 = pers_win if last_reward > 0 else pers_lose
            
        logits1 = beta1 * q_net + pers1 * rep1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        rep2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            rep2[last_action2[state_idx]] = 1.0
            
        logits2 = beta2 * q_stage2_mf[state_idx] + pers2 * rep2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with different learning rates for stage 1 and stage 2, and 
    asymmetric eligibility traces for positive and negative reward prediction errors.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for stage 1
    alpha2: [0, 1] - Learning rate for stage 2
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam_pos: [0, 1] - Eligibility trace for positive prediction errors
    lam_neg: [0, 1] - Eligibility trace for negative prediction errors
    pers: [-5, 5] - Perseveration parameter for stage 1
    """
    alpha1, alpha2, beta1, beta2, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        if last_action1 != -1:

        logits1 = beta1 * q_net + pers * rep1
        logits1 -= np.max(logits1)

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        probs_2 = exp_q2 / np.sum(exp_q2)

        lam = lam_pos if delta_stage2 > 0 else lam_neg

        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

    eps = 1e-10
