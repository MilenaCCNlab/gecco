import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Asymmetric Learning + Decay + Stage 1 Stickiness
    
    This model separates the learning rate into two components depending on whether 
    the reward prediction error is positive or negative. It also incorporates 
    decay for unchosen Q-values and a perseverance bonus for stage 1.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.
    p: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lambda_, decay, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_act1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act1 = -1
            q_stage1_mf *= (1.0 - decay)
            q_stage2_mf *= (1.0 - decay)
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act1 != -1:
            V1[last_act1] += p
            
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act1 = action_1[trial]
            unchosen_act1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_act1] *= (1.0 - decay)
            q_stage2_mf *= (1.0 - decay)
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2

        unchosen_act1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_act1] *= (1.0 - decay)
        
        unchosen_act2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_act2] *= (1.0 - decay)
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1.0 - decay)
        q_stage2_mf[other_state, 1] *= (1.0 - decay)
        
        last_act1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Transition Matrix Updating + Separate Betas + Stickiness
    
    This model assumes the participant learns the transition matrix over time 
    rather than assuming it is fixed. It also features separate betas and 
    stickiness parameters for stage 1 and stage 2.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p1: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    p2: [-5, 5] - Perseverance bonus for repeating the previous second-stage action.
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix probabilities.
    """
    learning_rate, beta1, beta2, w, lambda_, p1, p2, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_act1 = -1
    last_act2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act1 != -1:
            V1[last_act1] += p1
            
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act1 = action_1[trial]
            last_act2[state_idx] = -1
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        if last_act2[state_idx] != -1:
            V2[last_act2[state_idx]] += p2
            
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

        # Update transition matrix probabilities based on observed transition
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_act1 = action_1[trial]
        last_act2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Choice Kernels for Stage 1 & 2 + Separate Betas
    
    This model utilizes a choice kernel mechanism to capture autocorrelations 
    in choice behavior over multiple trials. It includes separate choice kernel 
    weights for stage 1 and stage 2.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    ck_lr: [0, 1] - Learning rate for updating choice kernels.
    ck_weight1: [-5, 5] - Weight of the choice kernel in stage 1 action selection.
    ck_weight2: [-5, 5] - Weight of the choice kernel in stage 2 action selection.
    """
    learning_rate, beta1, beta2, w, lambda_, ck_lr, ck_weight1, ck_weight2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            ck_2 = (1 - ck_lr) * ck_2
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf) + ck_weight1 * ck_1
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            ck_1[action_1[trial]] += ck_lr
            ck_2 = (1 - ck_lr) * ck_2
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx] + ck_weight2 * ck_2[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

        # Update choice kernels
        ck_1 = (1 - ck_lr) * ck_1
        ck_1[action_1[trial]] += ck_lr
        
        ck_2[state_idx] = (1 - ck_lr) * ck_2[state_idx]
        ck_2[state_idx, action_2[trial]] += ck_lr
        
        other_state = 1 - state_idx
        ck_2[other_state] = (1 - ck_lr) * ck_2[other_state]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
