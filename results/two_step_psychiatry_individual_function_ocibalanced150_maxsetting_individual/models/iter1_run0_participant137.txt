import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) hybrid model with separate learning rates for Stage 1 and Stage 2.
    This model captures the possibility that participants update their expectations 
    at different rates for the first-stage (spaceship) and second-stage (alien) choices.
    
    Bounds:
    alpha1: [0, 1] - learning rate for stage 1
    alpha2: [0, 1] - learning rate for stage 2
    beta1: [0, 10] - stage 1 inverse temperature
    beta2: [0, 10] - stage 2 inverse temperature
    lam: [0, 1] - eligibility trace
    w: [0, 1] - model-based weight
    p: [-5, 5] - perseveration bonus
    """
    alpha1, alpha2, beta1, beta2, lam, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits_1 = beta1 * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) hybrid model with reward-dependent perseveration (win-stay/lose-shift).
    This model allows the tendency to repeat the previous Stage 1 choice to depend 
    on whether the previous trial was rewarded or unrewarded.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - stage 1 inverse temperature
    beta2: [0, 10] - stage 2 inverse temperature
    lam: [0, 1] - eligibility trace
    w: [0, 1] - model-based weight
    p_rew: [-5, 5] - perseveration bonus after a rewarded trial
    p_unrew: [-5, 5] - perseveration bonus after an unrewarded trial
    """
    alpha, beta1, beta2, lam, w, p_rew, p_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_reward = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        current_p = 0.0
        if last_reward == 1:
            current_p = p_rew
        elif last_reward == 0:
            current_p = p_unrew
            
        logits_1 = beta1 * q_net + current_p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) hybrid model with choice perseveration at both Stage 1 and Stage 2.
    This model tests whether the participant has a tendency to stick to not just the same spaceship, 
    but also the same alien when visiting a specific planet.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - stage 1 inverse temperature
    beta2: [0, 10] - stage 2 inverse temperature
    lam: [0, 1] - eligibility trace
    w: [0, 1] - model-based weight
    p1: [-5, 5] - stage 1 perseveration bonus
    p2: [-5, 5] - stage 2 perseveration bonus (state-specific)
    """
    alpha, beta1, beta2, lam, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # tracks the last action taken in each Stage 2 state

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = np.full(2, -1)
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep1 = np.zeros(2)
        if last_action_1 != -1:
            rep1[last_action_1] = 1.0
            
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            rep2[last_action_2[state_idx]] = 1.0
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
