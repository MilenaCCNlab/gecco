def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Single learning rate, two betas (one for each stage), model-based weight, 
    eligibility trace, choice stickiness, and separate decay rates for stage 1 and stage 2 unchosen options.
    
    Bounds:
    learning_rate: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    rho: [0, 5]
    decay_1: [0, 1]
    decay_2: [0, 1]
    """
    learning_rate, beta_1, beta_2, w, lam, rho, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = rho
            
        q1_vals = q_stage1_net + stickiness_1
        exp_q1 = np.exp(beta_1 * (q1_vals - np.max(q1_vals)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = rho
            
        q2_vals = q_stage2_mf[state_idx] + stickiness_2
        exp_q2 = np.exp(beta_2 * (q2_vals - np.max(q2_vals)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay for unchosen options
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay_1)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay_2)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay_2)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay_2)
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Single learning rate, two betas, model-based weight, eligibility trace, 
    and continuous choice traces for stage 1 and stage 2 with separate stickiness weights.
    
    Bounds:
    learning_rate: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    rho_1: [0, 5]
    rho_2: [0, 5]
    alpha_c: [0, 1]
    """
    learning_rate, beta_1, beta_2, w, lam, rho_1, rho_2, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = q_stage1_net + rho_1 * choice_trace_1
        exp_q1 = np.exp(beta_1 * (q1_vals - np.max(q1_vals)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = q_stage2_mf[state_idx] + rho_2 * choice_trace_2[state_idx]
        exp_q2 = np.exp(beta_2 * (q2_vals - np.max(q2_vals)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Update continuous choice traces
        choice_trace_1[action_1[trial]] += alpha_c * (1 - choice_trace_1[action_1[trial]])
        choice_trace_1[1 - action_1[trial]] += alpha_c * (0 - choice_trace_1[1 - action_1[trial]])
        
        choice_trace_2[state_idx, action_2[trial]] += alpha_c * (1 - choice_trace_2[state_idx, action_2[trial]])
        choice_trace_2[state_idx, 1 - action_2[trial]] += alpha_c * (0 - choice_trace_2[state_idx, 1 - action_2[trial]])
        
        # Unvisited state choice trace decays to 0
        choice_trace_2[1 - state_idx, 0] += alpha_c * (0 - choice_trace_2[1 - state_idx, 0])
        choice_trace_2[1 - state_idx, 1] += alpha_c * (0 - choice_trace_2[1 - state_idx, 1])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate learning rates for rewarded and unrewarded trials, two betas, 
    model-based weight, eligibility trace, and separate choice stickiness for stage 1 and stage 2.
    
    Bounds:
    lr_reward: [0, 1]
    lr_unreward: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    rho_1: [0, 5]
    rho_2: [0, 5]
    """
    lr_reward, lr_unreward, beta_1, beta_2, w, lam, rho_1, rho_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = rho_1
            
        q1_vals = q_stage1_net + stickiness_1
        exp_q1 = np.exp(beta_1 * (q1_vals - np.max(q1_vals)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = rho_2
            
        q2_vals = q_stage2_mf[state_idx] + stickiness_2
        exp_q2 = np.exp(beta_2 * (q2_vals - np.max(q2_vals)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Determine learning rate based on reward outcome
        lr = lr_reward if reward[trial] > 0 else lr_unreward

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += lr * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
