def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with reward-dependent perseveration and a spatial/action bias for Stage 1.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    bias: constant bias for choosing action 1 over action 0 in stage 1 [-5,5]
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add constant bias for action 1
        q_net[1] += bias

        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose

        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action = action_1[trial]

        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with reward-dependent perseveration in Stage 1, 
    and simple perseveration in Stage 2.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    pers2: perseveration bonus for repeating the same Stage 2 action if in the same state [-5,5]
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = -1
    last_state = -1
    last_action2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action1 >= 0:
            if last_reward == 1:
                q_net[last_action1] += pers_win
            elif last_reward == 0:
                q_net[last_action1] += pers_lose

        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # Add stage 2 perseveration if returning to the same state
        q2 = np.copy(q_stage2_mf[state_idx])
        if state_idx == last_state and last_action2 >= 0:
            q2[last_action2] += pers2
            
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_state = state_idx
        last_action2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with reward-dependent action perseveration 
    and a state-approach bonus (increasing the model-based value of the previously rewarded state).
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    state_bonus: bonus added to the model-based value of the previous state if it was rewarded [-5,5]
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, state_bonus = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = -1
    last_state = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        v_stage2 = np.copy(max_q_stage2)
        
        # State approach bonus (model-based win-stay logic)
        if last_reward == 1 and last_state >= 0:
            v_stage2[last_state] += state_bonus
            
        q_stage1_mb = transition_matrix @ v_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action1 >= 0:
            if last_reward == 1:
                q_net[last_action1] += pers_win
            elif last_reward == 0:
                q_net[last_action1] += pers_lose

        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
