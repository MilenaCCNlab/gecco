Here are three new cognitive models designed to explain the participant's behavior. 

Looking at the participant data, there is an overwhelming preference for spaceship 1 at the first stage (chosen in roughly ~90% of trials). While a stickiness parameter (perseveration) explains repeating a past choice, it doesn't fully capture an intrinsic, baseline preference or "bias" for one option over the other that might exist regardless of the reward history. 

To address this, all three proposed models incorporate an explicit **Action 1 Bias** parameter at the first stage. This acts as an intercept in the softmax function, shifting the baseline probability toward Action 1. 

### Model 1: Hybrid Model with Stickiness and Bias
This model builds on the classic hybrid model by adding an explicit bias parameter for Action 1 at the first stage, alongside the stage 1 choice stickiness parameter. 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free with Choice Stickiness and Action Bias.
    
    This model includes a bias parameter for stage 1 choice, capturing an intrinsic 
    preference for Action 1 over Action 0, which is prominent in the participant's data.
    
    Bounds:
    alpha: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p: [-5, 5]
    bias: [-5, 5]
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, p, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits_1 = beta1 * q_stage1_net + p * rep
        logits_1[1] += bias # Add bias to action 1
        
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 Policy
        logits_2 = beta2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Stickiness and Bias
Since the best model so far utilized asymmetric learning rates (different rates for positive and negative reward prediction errors), this formulation tests if combining that asymmetry with the explicit Action 1 bias provides the most complete picture of this specific participant's behavior.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free with Asymmetric Learning Rates, Stickiness, and Bias.
    
    Combines asymmetric learning rates (different rates for positive and negative RPEs) 
    with a choice stickiness parameter and a bias parameter for stage 1 choice.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p: [-5, 5]
    bias: [-5, 5]
    """
    import numpy as np
    alpha_pos, alpha_neg, beta1, beta2, w, lam, p, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits_1 = beta1 * q_stage1_net + p * rep
        logits_1[1] += bias
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 Policy
        logits_2 = beta2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha_s2 * delta_stage2

        q_stage1_mf[a1] += alpha_s1 * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Dual-Stage Stickiness and Stage 1 Bias
This model tests whether choice perseveration is occurring not just globally at stage 1, but uniquely inside the states of stage 2 as well. It provides two separate stickiness parameters—one for each stage—along with the Stage 1 Action 1 bias.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free with Dual Stickiness and Bias.
    
    This model includes separate choice stickiness parameters for stage 1 (p1) 
    and stage 2 (p2, tracked state-specifically), as well as a bias parameter for Action 1.
    
    Bounds:
    alpha: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p1: [-5, 5]
    p2: [-5, 5]
    bias: [-5, 5]
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, p1, p2, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2_state = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if last_action1 != -1:
            rep1[last_action1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + p1 * rep1
        logits_1[1] += bias
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 Policy
        rep2 = np.zeros(2)
        if last_action2_state[s] != -1:
            rep2[last_action2_state[s]] = 1.0

        logits_2 = beta2 * q_stage2_mf[s] + p2 * rep2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action1 = a1
        last_action2_state[s] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
