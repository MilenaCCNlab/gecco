Here are 3 new cognitive models that build upon the structure of the best model so far but introduce novel mechanisms for choice perseveration (stickiness). Perseveration is a major driver of variance in the two-step task, and these models explore different psychological nuances of how past choices influence future ones.

### Model 1: Reward-Dependent Stage 1 Perseveration (Win-Stay / Lose-Shift)
This model splits the Stage 1 choice perseveration into two separate parameters based on the outcome of the previous trial. Instead of a single tendency to repeat the last action, it allows for asymmetric stickiness: one parameter for repeating the action after a reward (Win-Stay) and another for repeating it after a non-reward (Lose-Stay/Shift).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. Hybrid Model with Reward-Dependent Stage 1 Perseveration (Win-Stay/Lose-Shift).
    Instead of a single perseveration parameter for stage 1, it has separate bonuses depending on whether 
    the previous trial was rewarded or not.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values
    beta_1: [0,10] - Inverse temperature for stage 1 choices
    beta_2: [0,10] - Inverse temperature for stage 2 choices
    w: [0,1] - Model-based vs Model-free weight
    lam: [0,1] - Eligibility trace parameter
    pers_1_win: [-5,5] - Choice perseveration bonus for stage 1 if previous trial was rewarded
    pers_1_lose: [-5,5] - Choice perseveration bonus for stage 1 if previous trial was unrewarded
    """
    alpha, beta_1, beta_2, w, lam, pers_1_win, pers_1_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_reward = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if prev_a1 != -1:
            if prev_reward == 1:
                pers_bonus_1[prev_a1] = pers_1_win
            else:
                pers_bonus_1[prev_a1] = pers_1_lose
            
        logits_1 = beta_1 * q_stage1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: State-Specific Stage 2 Perseveration
Rather than capturing a global tendency to repeat the last Stage 2 motor action regardless of the planet visited, this model captures the tendency to repeat the action previously chosen *in the current state*. This accurately reflects state-bound stimulus perseveration.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. Hybrid Model with State-Specific Stage 2 Perseveration.
    Instead of a global perseveration at stage 2, this model captures the tendency to repeat 
    the same action previously taken *in that specific state*.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values
    beta_1: [0,10] - Inverse temperature for stage 1 choices
    beta_2: [0,10] - Inverse temperature for stage 2 choices
    w: [0,1] - Model-based vs Model-free weight
    lam: [0,1] - Eligibility trace parameter
    pers_1: [-5,5] - Choice perseveration bonus for stage 1
    pers_2_state: [-5,5] - State-specific choice perseveration bonus for stage 2
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    last_a2_in_state = np.full(2, -1)

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2

        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]

    return log_loss
```

### Model 3: Global and State-Specific Stage 2 Perseveration
This model combines both forms of Stage 2 perseveration, recognizing that participants might have a general motor bias to repeat their last button press (global perseveration) *and* a stimulus-driven bias to select the same alien when returning to the same planet (state-specific perseveration). 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. Hybrid Model with both Global and State-Specific Stage 2 Perseveration.
    This model distinguishes between a general tendency to repeat the last motor action (global) 
    and the tendency to repeat the action previously chosen in the current state (state-specific).
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values
    beta_1: [0,10] - Inverse temperature for stage 1 choices
    beta_2: [0,10] - Inverse temperature for stage 2 choices
    w: [0,1] - Model-based vs Model-free weight
    lam: [0,1] - Eligibility trace parameter
    pers_1: [-5,5] - Choice perseveration bonus for stage 1
    pers_2_global: [-5,5] - Global choice perseveration bonus for stage 2 (last action overall)
    pers_2_state: [-5,5] - State-specific choice perseveration bonus for stage 2 (last action in this state)
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2_global, pers_2_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2_global = -1
    last_a2_in_state = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_a2_global = -1
            continue

        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if prev_a1 != -1:

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        pers_bonus_2 = np.zeros(2)
        if prev_a2_global != -1:
            pers_bonus_2[prev_a2_global] += pers_2_global
        if last_a2_in_state[state_idx] != -1:
            pers_bonus_2[last_a2_in_state[state_idx]] += pers_2_state

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2_global = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
