def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with both state-specific and global motor perseveration at stage 2.
    This captures a participant's tendency to repeat the exact same motor action at stage 2 (global), 
    as well as their tendency to repeat their choice given the specific planet (state).
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1 choice.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace for updating stage 1 based on stage 2 reward.
    rho1: [0, 10] - Perseveration bonus for repeating the previous stage 1 choice.
    rho2_state: [0, 10] - Perseveration for repeating the same action in the same state at stage 2.
    rho2_global: [0, 10] - Global stage 2 perseveration (repeating the same action, regardless of state).
    """
    learning_rate, beta1, beta2, w, lam, rho1, rho2_state, rho2_global = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_global = -1
    last_action_2_state = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1
            
        logits1 = beta1 * q_stage1_net + rho1 * pers1
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers2_global = np.zeros(2)
        if last_action_2_global != -1:
            pers2_global[last_action_2_global] = 1
            
        pers2_state = np.zeros(2)
        if last_action_2_state[state_idx] != -1:
            pers2_state[last_action_2_state[state_idx]] = 1
            
        logits2 = beta2 * q_stage2_mf[state_idx] + rho2_global * pers2_global + rho2_state * pers2_state
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2_global = action_2[trial]
        last_action_2_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):

    Model with separate learning rates for stage 1 and stage 2, and global perseveration at stage 2.
    This allows for different speeds of learning in updating the spaceship preferences vs the alien preferences.
    
    Parameters:

    rho1: [0, 10] - Perseveration bonus for repeating the previous stage 1 choice.

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates for positive and negative prediction errors, and global perseveration at stage 2.
    This separates the impact of positive surprises (gaining gold) versus negative surprises (missing gold) on value updates.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1 choice.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace for updating stage 1 based on stage 2 reward.
    rho1: [0, 10] - Perseveration bonus for repeating the previous stage 1 choice.
    rho2_global: [0, 10] - Global stage 2 perseveration (repeating the same action, regardless of state).
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, rho1, rho2_global = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_global = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1
            
        logits1 = beta1 * q_stage1_net + rho1 * pers1
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers2_global = np.zeros(2)
        if last_action_2_global != -1:
            pers2_global[last_action_2_global] = 1
            
        logits2 = beta2 * q_stage2_mf[state_idx] + rho2_global * pers2_global
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2_global = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
