def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with asymmetric learning rates, unchosen value decay, 
    and independent perseverance bonuses for stage 1 and stage 2 choices.
    
    Parameters:
    alpha_pos: [0,1] - learning rate for positive reward prediction errors
    alpha_neg: [0,1] - learning rate for negative reward prediction errors
    beta: [0,10] - inverse temperature for softmax choice stochasticity
    w: [0,1] - weight of model-based values
    lambda_: [0,1] - eligibility trace parameter 
    p_stage1: [-5,5] - perseverance bonus for repeating the previous stage 1 action
    p_stage2: [-5,5] - perseverance bonus for repeating the previous stage 2 action at the same state
    decay: [0,1] - decay rate for unchosen actions towards 0.5
    """
    import numpy as np
    alpha_pos, alpha_neg, beta, w, lambda_, p_stage1, p_stage2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1.0
            
        logits_1 = beta * q_stage1_net + p_stage1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        rep_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            rep_2[last_action_2[state_idx]] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + p_stage2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_2 * lambda_ * delta_stage2

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] = (1 - decay) * q_stage1_mf[a] + decay * 0.5
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with asymmetric learning rates, unchosen value decay, 
    and transition matrix learning.
    
    Parameters:
    alpha_pos: [0,1] - learning rate for positive reward prediction errors
    alpha_neg: [0,1] - learning rate for negative reward prediction errors
    beta: [0,10] - inverse temperature for softmax choice stochasticity
    w: [0,1] - weight of model-based values
    lambda_: [0,1] - eligibility trace parameter 
    p: [-5,5] - perseverance bonus for repeating the previous stage 1 action
    decay: [0,1] - decay rate for unchosen actions towards 0.5
    alpha_tr: [0,1] - learning rate for updating the transition matrix
    """
    import numpy as np
    alpha_pos, alpha_neg, beta, w, lambda_, p, decay, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta * q_stage1_net + p * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Transition matrix update based on observed state transition
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_2 * lambda_ * delta_stage2

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] = (1 - decay) * q_stage1_mf[a] + decay * 0.5
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
