def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF mixture model with reward-dependent perseveration.
    Instead of a single stickiness parameter, this model has separate 
    stickiness parameters depending on whether the previous trial was 
    rewarded or unrewarded, allowing for asymmetric win-stay/lose-shift behavior.
    
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    w: [0,1]
    lam: [0,1]
    pers_rew: [-5,5]
    pers_unrew: [-5,5]
    """
    learning_rate, beta, w, lam, pers_rew, pers_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            if last_reward == 1:
                q_stage1_net[last_action] += pers_rew
            else:
                q_stage1_net[last_action] += pers_unrew
                
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # update choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # update choice 2 and eligibility trace
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF mixture model with asymmetric eligibility traces.
    The eligibility trace parameter (lambda) is different depending on 
    whether the stage 2 reward prediction error is positive or negative,
    capturing potential differential reinforcement of stage 1 choices from
    better-than-expected versus worse-than-expected outcomes.
    
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    w: [0,1]
    lam_pos: [0,1]
    lam_neg: [0,1]
    stickiness: [-5,5]
    """
    learning_rate, beta, w, lam_pos, lam_neg, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += stickiness
            
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # update choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # update choice 2 and eligibility trace
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        if delta_stage2 > 0:
            q_stage1_mf[action_1[trial]] += learning_rate * lam_pos * delta_stage2
        else:
            q_stage1_mf[action_1[trial]] += learning_rate * lam_neg * delta_stage2
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF mixture model with a choice kernel.
    Instead of a simple n-1 stickiness, this model maintains a choice kernel 
    that exponentially decays past choices, capturing longer-term choice 
    perseveration trends in the first stage.
    
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    w: [0,1]
    lam: [0,1]
    ck_lr: [0,1]
    ck_weight: [-5,5]
    """
    learning_rate, beta, w, lam, ck_lr, ck_weight = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    choice_kernel = np.zeros(2)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q_stage1_net += ck_weight * choice_kernel
            
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update choice kernel
        choice_kernel *= (1 - ck_lr)
        choice_kernel[action_1[trial]] += ck_lr
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # update choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # update choice 2 and eligibility trace
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
