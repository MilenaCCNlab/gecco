Here are three new cognitive models that build upon the best-performing model by adding cognitively plausible mechanisms. All models retain the stage-specific perseveration that proved successful, but introduce new dynamics to capture additional variance in the participant's choices.

### Model 1: Q-Value Decay (Forgetting)
This model adds a decay parameter to the Q-values. When an action is not chosen, its Q-value slowly regresses towards a neutral value of 0.5. This accounts for "forgetting" over time, which can explain why participants might re-explore options they haven't chosen recently.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-value decay with stage-specific perseveration.
    Unchosen Q-values decay towards 0.5 at each trial, capturing memory decay/forgetting.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    p_stage1: [-5, 5] - Choice perseveration for stage 1
    p_stage2: [-5, 5] - State-specific choice perseveration for stage 2
    lam: [0, 1] - Eligibility trace
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5
    """
    learning_rate, beta, w, p_stage1, p_stage2, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        val_1 = beta * q_net + p_stage1 * pers_1
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        state_idx = s
        pers_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_2[last_action_2[state_idx]] = 1.0

        val_2 = beta * q_stage2_mf[state_idx] + p_stage2 * pers_2
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Decay unchosen Q-values
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for st in range(2):
            for act in range(2):
                if not (st == state_idx and act == a2):
                    q_stage2_mf[st, act] += decay * (0.5 - q_stage2_mf[st, act])
        
        last_action_1 = a1
        last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates (Positive vs. Negative RPEs)
This model splits the learning rate into two components depending on the sign of the reward prediction error (RPE). People often update their beliefs differently following better-than-expected outcomes compared to worse-than-expected outcomes (optimism/pessimism bias).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for positive and negative prediction errors,
    combined with stage-specific perseveration.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    p_stage1: [-5, 5] - Choice perseveration for stage 1
    p_stage2: [-5, 5] - State-specific choice perseveration for stage 2
    lam: [0, 1] - Eligibility trace
    """
    alpha_pos, alpha_neg, beta, w, p_stage1, p_stage2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        val_1 = beta * q_net + p_stage1 * pers_1
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        state_idx = s
        pers_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_2[last_action_2[state_idx]] = 1.0

        val_2 = beta * q_stage2_mf[state_idx] + p_stage2 * pers_2
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2
        
        q_stage1_mf[a1] += alpha_s2 * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Perseveration
This variant splits the stage 1 perseveration tendency based on whether the previous trial resulted in a reward or not. This captures win-stay/lose-shift dynamics directly within the baseline choice bias rather than just through Q-value updates.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent perseveration for stage 1, and state-specific
    perseveration for stage 2. This captures whether a participant is more
    likely to repeat their first-stage choice after a win vs after a loss.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    p_stage1_rew: [-5, 5] - Choice perseveration for stage 1 after a rewarded trial
    p_stage1_unrew: [-5, 5] - Choice perseveration for stage 1 after an unrewarded trial
    p_stage2: [-5, 5] - State-specific choice perseveration for stage 2
    lam: [0, 1] - Eligibility trace
    """
    learning_rate, beta, w, p_stage1_rew, p_stage1_unrew, p_stage2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        p_stage1 = 0
        if last_reward == 1:
            p_stage1 = p_stage1_rew
        elif last_reward == 0:
            p_stage1 = p_stage1_unrew
            
        val_1 = beta * q_net + p_stage1 * pers_1
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        state_idx = s
        pers_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_2[last_action_2[state_idx]] = 1.0

        val_2 = beta * q_stage2_mf[state_idx] + p_stage2 * pers_2
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[state_idx] = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
