def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL model with eligibility traces and stage-specific perseveration.
    This model accounts for the tendency to repeat the same first-stage choice as the previous trial,
    as well as the tendency to repeat the same second-stage choice when visiting the same state.
    
    Bounds:
    alpha: [0,1] - Learning rate.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace for updating stage 1 based on stage 2 reward.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    p1: [-5,5] - Stage 1 choice perseveration (stickiness).
    p2: [-5,5] - Stage 2 choice perseveration (stickiness) specific to each state.
    """
    alpha, beta1, beta2, lam, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta1 * q_net
        if last_action_1 != -1:
            q1_total[last_action_1] += p1
            
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            q2_total[last_action_2[state_idx]] += p2
            
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL model with asymmetric learning rates for positive and negative prediction errors.
    This allows the model to capture differences in how participants update values after unexpected rewards versus omissions.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace.
    w: [0,1] - Model-based weight.
    p: [-5,5] - Stage 1 choice perseveration.
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta1 * q_net
        if last_action_1 != -1:
            q1_total[last_action_1] += p
            
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL model with reward sensitivity scaling.
    This model includes a multiplier `rho` on the rewards to account for individual differences 
    in subjective reward valuation, alongside standard eligibility and perseveration parameters.
    
    Bounds:
    alpha: [0,1] - Learning rate.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace.
    w: [0,1] - Model-based weight.

    rho: [0,10] - Reward sensitivity multiplier.
    """
    alpha, beta1, beta2, lam, w, p, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
