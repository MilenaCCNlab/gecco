import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and temperatures for Stage 1 and Stage 2, with decay.
    This model assumes that learning processes and choice stochasticity might differ 
    between the first stage (spaceship choice) and the second stage (alien choice), 
    while retaining memory decay for unchosen options.
    
    Parameters:
    alpha1: Learning rate for Stage 1 [0, 1]
    alpha2: Learning rate for Stage 2 [0, 1]
    beta1: Inverse temperature for Stage 1 softmax choices [0, 10]
    beta2: Inverse temperature for Stage 2 softmax choices [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Decay rate for unchosen actions towards 0.5 [0, 1]
    """
    alpha1, alpha2, beta1, beta2, w, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # eligibility trace for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        # decay for unchosen actions at stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with dynamic transition matrix learning and decay.
    Instead of assuming a fixed known transition matrix, this model updates its 
    internal model of the transitions based on the observed planet arrivals.
    
    Parameters:
    alpha: Learning rate for updating Q-values [0, 1]
    beta: Inverse temperature for softmax choices [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Decay rate for unchosen actions towards 0.5 [0, 1]
    alpha_trans: Learning rate for updating the transition matrix probabilities [0, 1]
    """
    alpha, beta, w, lam, decay, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # transition matrix updating
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen actions at stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates (positive/negative RPE), with value decay and choice stickiness.
    This model captures the fact that participants may learn differently from better-than-expected 
    outcomes compared to worse-than-expected outcomes, while also exhibiting a baseline tendency 
    to repeat their previous Stage 1 choice.
    
    Parameters:
    alpha_pos: Learning rate for positive reward prediction errors [0, 1]
    alpha_neg: Learning rate for negative reward prediction errors [0, 1]
    beta: Inverse temperature for softmax choices [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Decay rate for unchosen actions towards 0.5 [0, 1]
    stickiness: Choice stickiness for stage 1 (tendency to repeat last action) [-5, 5]
    """
    alpha_pos, alpha_neg, beta, w, lam, decay, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_term = np.zeros(2)
        if last_action_1 != -1:
            stick_term[last_action_1] = stickiness
            
        exp_q1 = np.exp(beta * q_stage1_net + stick_term)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        # eligibility trace for choice 1
        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2

        # decay for unchosen actions at stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
                    
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
