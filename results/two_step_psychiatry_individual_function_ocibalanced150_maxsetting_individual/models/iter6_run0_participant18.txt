def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Separate Q-value Decay Rates for Stage 1 and Stage 2.
    Allows for differential forgetting of first-stage vs second-stage values, 
    accounting for potentially different memory decay rates for different stages of the task.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weighting of model-based vs model-free control (1=pure MB, 0=pure MF).
    pi: [-5, 5] - General perseveration bonus for repeating the previous stage 1 choice.
    lambda_: [0, 1] - Eligibility trace for stage 1 model-free update.
    decay1: [0, 1] - Rate at which Stage 1 Q-values decay towards 0.
    decay2: [0, 1] - Rate at which Stage 2 Q-values decay towards 0.
    """
    alpha, beta, w, pi, lambda_, decay1, decay2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta * q_net + pi * pers
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # decay
        q_stage1_mf = q_stage1_mf * (1 - decay1)
        q_stage2_mf = q_stage2_mf * (1 - decay2)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Q-value Decay and Slow Choice Trace (Perseveration).
    Instead of only considering the immediately previous choice, perseveration 
    is driven by a choice trace that updates slowly over time, capturing longer-term 
    habits rather than just last-trial stickiness.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature.
    w: [0, 1] - Model-based vs model-free weight.
    pi: [-5, 5] - Perseveration weight on the choice trace.
    lambda_: [0, 1] - Eligibility trace.
    decay: [0, 1] - Q-value decay rate.
    alpha_c: [0, 1] - Update rate for the choice trace.
    """
    alpha, beta, w, pi, lambda_, decay, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_net + pi * choice_trace
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # decay
        q_stage1_mf = q_stage1_mf * (1 - decay)
        q_stage2_mf = q_stage2_mf * (1 - decay)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # update choice trace
        choice_trace = (1 - alpha_c) * choice_trace
        choice_trace[action_1[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
