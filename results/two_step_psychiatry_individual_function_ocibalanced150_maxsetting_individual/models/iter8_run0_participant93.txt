def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates, separate MB/MF temperatures, and unchosen Q-value decay.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta_mb: inverse temperature for model-based Q-values [0,10]
    beta_mf: inverse temperature for model-free Q-values [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    lam: eligibility trace [0,1]
    pi: choice stickiness [0,5]
    decay: decay rate of unchosen Q-values towards 0.5 [0,1]
    """
    alpha_pos, alpha_neg, beta_mb, beta_mf, beta_2, lam, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi * stickiness
        q1_vals -= np.max(q1_vals) 
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        q2_vals = beta_2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        a2 = action_2[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_stage2 * delta_stage2

        q_stage1_mf[a1] += alpha_stage2 * lam * delta_stage2
        
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = q_stage1_mf[unchosen_a1] * (1 - decay) + 0.5 * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + 0.5 * decay
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with separate MB/MF temperatures and transition matrix learning.
    
    Parameters:
    alpha: learning rate for Q-values [0,1]
    alpha_trans: learning rate for transition probabilities [0,1]
    beta_mb: inverse temperature for model-based Q-values [0,10]
    beta_mf: inverse temperature for model-free Q-values [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    lam: eligibility trace [0,1]
    pi: choice stickiness [0,5]
    """
    alpha, alpha_trans, beta_mb, beta_mf, beta_2, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi * stickiness
        q1_vals -= np.max(q1_vals) 
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        q2_vals = beta_2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        a2 = action_2[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
