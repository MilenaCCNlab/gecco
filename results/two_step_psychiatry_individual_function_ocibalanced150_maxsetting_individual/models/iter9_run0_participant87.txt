Here are 3 new cognitive models that build upon the best-performing mechanisms (transition learning, unchosen decay, perseveration) while exploring novel, cognitively plausible hypotheses. All models stay within the 8-parameter limit.

### Model 1: Decoupled Model-Based and Model-Free Temperatures (`beta_mb`, `beta_mf`)
Instead of a weighting parameter `w` that forces a convex combination between model-based and model-free values, this model assigns independent inverse temperatures to the model-based (`beta_mb`) and model-free (`beta_mf`) Q-values. This formulation allows the optimization process more freedom to scale the distinct influences of the two systems independently.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model that parameterizes model-based and model-free weights directly via independent temperatures.
    Parameters:
    learning_rate: Learning rate for Q-values [0,1]
    decay_unchosen: Decay rate for unchosen Q-values [0,1]
    lr_trans: Learning rate for transition probabilities [0,1]
    beta_mf: Inverse temperature for model-free Q-values in stage 1 [0,10]
    beta_mb: Inverse temperature for model-based Q-values in stage 1 [0,10]
    beta_2: Inverse temperature for stage 2 [0,10]
    lam: Eligibility trace parameter [0,1]
    p_pers: Choice perseveration bonus [-5,5]
    """
    learning_rate, decay_unchosen, lr_trans, beta_mf, beta_mb, beta_2, lam, p_pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Q-value is weighted by respective independent betas
        exponent_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if last_action != -1:
            exponent_1[last_action] += p_pers
            
        max_exp_1 = np.max(exponent_1)
        exp_q1 = np.exp(exponent_1 - max_exp_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exponent_2 = beta_2 * q_stage2_mf[state_idx]
        max_exp_2 = np.max(exponent_2)
        exp_q2 = np.exp(exponent_2 - max_exp_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += lr_trans * (1.0 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += lr_trans * (0.0 - transition_matrix[a1, 1 - s2])

        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] *= (1 - decay_unchosen)
        
        a2 = action_2[trial]
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay_unchosen)

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Loss Aversion for Reward Omission (`loss_av`)
Sometimes participants treat a 0-coin reward not just as a lack of reward, but as a distinct negative outcome (loss). This model introduces a `loss_av` parameter that heavily penalizes actions resulting in 0 coins, capturing an asymmetric sensitivity to missing out on rewards. To maintain the 8-parameter limit, it shares a single temperature parameter `beta` across both decision stages.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a single temperature parameter and loss aversion for 0-coin rewards.
    Parameters:
    learning_rate: Learning rate for Q-values [0,1]
    decay_unchosen: Decay rate for unchosen Q-values [0,1]
    lr_trans: Learning rate for transition probabilities [0,1]
    beta: Inverse temperature for both stages [0,10]
    lam: Eligibility trace parameter [0,1]
    w: Model-based weight [0,1]
    p_pers: Choice perseveration bonus [-5,5]
    loss_av: Penalty applied when reward is 0 [0,5]
    """
    learning_rate, decay_unchosen, lr_trans, beta, lam, w, p_pers, loss_av = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent_1 = beta * q_stage1_net
        if last_action != -1:
            exponent_1[last_action] += p_pers
            
        max_exp_1 = np.max(exponent_1)
        exp_q1 = np.exp(exponent_1 - max_exp_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exponent_2 = beta * q_stage2_mf[state_idx]
        max_exp_2 = np.max(exponent_2)
        exp_q2 = np.exp(exponent_2 - max_exp_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += lr_trans * (1.0 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += lr_trans * (0.0 - transition_matrix[a1, 1 - s2])

        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] *= (1 - decay_unchosen)
        
        a2 = action_2[trial]
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay_unchosen)

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Adjust reward for loss aversion
        eff_reward = 1.0 if reward[trial] == 1 else -loss_av
        
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Optimistic Initialization & Decay Baseline (`q_init`)
This model explores the idea that early exploration is driven by "optimistic initialization" (starting Q-values at >0). Furthermore, instead of unchosen values decaying entirely towards 0, they decay back towards their initial baseline (`q_init`). This captures a participant's cognitive regression to a prior expectation rather than strict forgetting.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with a single temperature parameter and optimistic initialization of Q-values.
    Parameters:
    learning_rate: Learning rate for Q-values [0,1]
    decay_unchosen: Decay rate for unchosen Q-values [0,1]
    lr_trans: Learning rate for transition probabilities [0,1]
    beta: Inverse temperature for both stages [0,10]
    lam: Eligibility trace parameter [0,1]
    w: Model-based weight [0,1]
    p_pers: Choice perseveration bonus [-5,5]
    q_init: Initial Q-value for optimistic initialization [0,1]
    """
    learning_rate, decay_unchosen, lr_trans, beta, lam, w, p_pers, q_init = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Init values optimistically
    q_stage1_mf = np.full(2, q_init)
    q_stage2_mf = np.full((2, 2), q_init)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent_1 = beta * q_stage1_net
        if last_action != -1:
            exponent_1[last_action] += p_pers
            
        max_exp_1 = np.max(exponent_1)
        exp_q1 = np.exp(exponent_1 - max_exp_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exponent_2 = beta * q_stage2_mf[state_idx]
        max_exp_2 = np.max(exponent_2)
        exp_q2 = np.exp(exponent_2 - max_exp_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += lr_trans * (1.0 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += lr_trans * (0.0 - transition_matrix[a1, 1 - s2])

        # Decay towards q_init rather than 0
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] = q_stage1_mf[unchosen_1] * (1 - decay_unchosen) + q_init * decay_unchosen
        
        a2 = action_2[trial]
        for s in range(2):
            for a in range(2):
                if not (s == s2 and a == a2):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay_unchosen) + q_init * decay_unchosen

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
