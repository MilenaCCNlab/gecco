Here are 3 new cognitive models that explore alternative mechanisms and parameter combinations for the two-step decision-making task. 

### Model 1: Full Daw (2011) Model
This model separates learning rates (`alpha1`, `alpha2`) and inverse temperatures (`beta1`, `beta2`) for the first and second stages. This allows the model to capture differences in learning speed and choice stochasticity between the two stages of the task, which is a very common structure used in modeling this specific task.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Full Daw 2011 model with separate learning rates and inverse temperatures for each stage.
    
    Parameters:
    alpha1: [0,1] - Learning rate for stage 1.
    alpha2: [0,1] - Learning rate for stage 2.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free Q-values for stage 1.
    lambd: [0,1] - Eligibility trace parameter.
    p: [-5,5] - Perseveration parameter for stage 1 choice.
    """
    alpha1, alpha2, beta1, beta2, w, lambd, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lambd * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid Model with Forgetting (Value Decay)
This model introduces a `decay` parameter. Instead of assuming that unchosen options maintain their expected values perfectly, it decays all Q-values towards a neutral expectation (0.5) on every trial before updating. This captures the phenomenon where participants slowly "forget" the values of aliens or spaceships they haven't interacted with recently.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with forgetting/decay of all Q-values towards 0.5.
    
    Parameters:
    alpha: [0,1] - Learning rate for updating Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weight on model-based vs model-free Q-values for stage 1.
    lambd: [0,1] - Eligibility trace parameter.
    p: [-5,5] - Perseveration parameter for stage 1 choice.
    decay: [0,1] - Decay rate of all Q-values towards 0.5 before update.
    """
    alpha, beta, w, lambd, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Decay all Q-values towards 0.5 prior to the trial update
        q_stage1_mf = (1 - decay) * q_stage1_mf + decay * 0.5
        q_stage2_mf = (1 - decay) * q_stage2_mf + decay * 0.5

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid Model with Reward-Dependent Perseveration (Win-Stay/Lose-Shift)
Instead of a single perseveration parameter, this model splits perseveration into two separate parameters based on the outcome of the previous trial (`p_win` and `p_lose`). This allows the model to capture "win-stay" and "lose-shift" heuristics directly at the first stage, which is a common behavior pattern distinct from pure incremental Q-learning.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with reward-dependent perseveration (win-stay/lose-shift).
    
    Parameters:
    alpha: [0,1] - Learning rate for updating Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weight on model-based vs model-free Q-values for stage 1.
    lambd: [0,1] - Eligibility trace parameter.
    p_win: [-5,5] - Perseveration parameter if the previous trial was rewarded.
    p_lose: [-5,5] - Perseveration parameter if the previous trial was unrewarded.
    """
    alpha, beta, w, lambd, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            if last_reward > 0:
                q_net[last_action_1] += p_win
            else:
                q_net[last_action_1] += p_lose
            
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
