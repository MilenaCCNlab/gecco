Here are 3 new cognitive models that explore alternative mechanisms for the two-step task. Each model focuses on a unique cognitive hypothesis, such as subjective reward scaling, transition-dependent credit assignment, and reward-dependent exploration.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Reward Penalty Model
    Incorporates a parameter to represent the subjective value of an unrewarded outcome. 
    Instead of treating 0 coins simply as 0, this allows the model to actively penalize 
    zero-reward trials (if r_penalty < 0), simulating loss aversion or disappointment.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0, 1] - Eligibility trace (how much stage 2 reward updates stage 1 MF value).
    rep: [-5, 5] - Stickiness for stage 1 (tendency to repeat previous stage 1 choice).
    r_penalty: [-2, 1] - Subjective value of receiving 0 coins.
    """
    alpha, beta_1, beta_2, w, lam, rep, r_penalty = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = rep
            
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Subjective reward evaluation
        r_eff = 1.0 if reward[trial] == 1 else r_penalty

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = r_eff - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition-Dependent Eligibility Trace
    Separates the eligibility trace parameter (lambda) based on whether the transition
    from stage 1 to stage 2 was common or rare. Participants may trust the reward from 
    a common transition more to update the first-stage action.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam_common: [0, 1] - Eligibility trace if the transition was common.
    lam_rare: [0, 1] - Eligibility trace if the transition was rare.
    rep: [-5, 5] - Stickiness for stage 1.
    """
    alpha, beta_1, beta_2, w, lam_common, lam_rare, rep = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = rep
            
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Transition-dependent trace
        is_common = (action_1[trial] == state_idx)
        lam = lam_common if is_common else lam_rare

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-Dependent Stage 1 Temperature
    The inverse temperature for stage 1 choices depends on whether the previous trial 
    was rewarded or not. This reflects dynamic shifts between exploration and exploitation 
    (e.g., participants might explore more randomly after a loss).
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1_win: [0, 10] - Stage 1 inverse temperature after a reward.
    beta_1_lose: [0, 10] - Stage 1 inverse temperature after no reward.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0, 1] - Eligibility trace.
    rep: [-5, 5] - Stickiness for stage 1.
    """
    alpha, beta_1_win, beta_1_lose, beta_2, w, lam, rep = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 1  # Default starting state

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = rep
            
        # Select beta based on previous reward
        beta_1 = beta_1_win if last_reward == 1 else beta_1_lose
        
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
