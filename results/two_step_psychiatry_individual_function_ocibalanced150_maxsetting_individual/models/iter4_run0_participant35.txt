def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Pure Model-Free with Separate Betas and Action Matching Bias
    Tests if the participant has different exploration/exploitation tendencies
    at stage 1 vs stage 2, while remaining purely model-free. The data shows 
    strong action matching (repeating stage 1 action at stage 2), so theta is included.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    """
    alpha, beta_1, beta_2, lam, theta = model_parameters
    n_trials = len(action_1)
  
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # Stage 1 choice
        q_net_shifted = q_stage1_mf - np.max(q_stage1_mf)
        exp_q1 = np.exp(beta_1 * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 choice
        q2 = q_stage2_mf[state_idx].copy()
        q2[action_1[trial]] += theta  # action matching bias
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta_2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Learning
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Pure Model-Free with Shared Beta, Action Matching, and Perseveration
    Adds a stage 1 perseveration parameter to the pure MF model to capture
    tendency to repeat the first-stage choice regardless of reward, supplementing
    the strong stage 2 action matching (theta).
    
    Parameters:
    alpha: learning rate [0,1]
    beta: shared inverse temperature [0,10]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    pers: stage 1 perseveration bonus [-5, 5]
    """
    alpha, beta, lam, theta, pers = model_parameters
    n_trials = len(action_1)
  
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # Stage 1 choice
        q1 = q_stage1_mf.copy()
        if last_action_1 != -1:
            q1[last_action_1] += pers
            
        q1_shifted = q1 - np.max(q1)
        exp_q1 = np.exp(beta * q1_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 choice
        q2 = q_stage2_mf[state_idx].copy()
        q2[action_1[trial]] += theta  # action matching bias
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Learning
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF with Separate Learning Rates and Action Matching
    Uses separate learning rates for stage 1 and stage 2, along with a shared
    beta, model-based weight, eligibility trace, and action matching bias.
    
    Parameters:
    alpha_1: stage 1 learning rate [0,1]
    alpha_2: stage 2 learning rate [0,1]
    beta: shared inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    theta: stage 2 action matching bias [-5, 5]
    """
    alpha_1, alpha_2, beta, w, lam, theta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 choice
        q2 = q_stage2_mf[state_idx].copy()
        q2[action_1[trial]] += theta  # action matching bias
        
        q2_shifted = q2 - np.max(q2)
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Learning
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
