def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Hybrid MB/MF with Stage 1/2 Stickiness and Stage 1 Win-Stay Bonus.
    
    This model adds a reward-contingent stickiness (win-stay) parameter for Stage 1.
    If the previous trial was rewarded, the tendency to repeat the Stage 1 choice is 
    further increased (or decreased) by this parameter, capturing a direct effect of 
    recent reward on top of general perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    w: [0, 1] - Weight given to model-based vs model-free values (1 = pure MB, 0 = pure MF).
    beta: [0, 10] - Inverse temperature for softmax action selection.
    stickiness_1: [-5, 5] - Log-odds bonus for repeating the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Log-odds bonus for repeating the previous Stage 2 choice.
    win_stay_1: [-5, 5] - Additional log-odds bonus for Stage 1 if the previous trial was rewarded.
    """
    learning_rate, w, beta, stickiness_1, stickiness_2, win_stay_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            if last_reward == 1:
                logits_1[last_action_1] += win_stay_1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Hybrid MB/MF with Unchosen Q-Value Decay.
    
    This model incorporates forgetting by decaying the Q-values of unchosen actions 
    towards a neutral value (0.5). This captures the hypothesis that participants 
    may forget the value of options they haven't sampled recently.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    w: [0, 1] - Weight given to model-based vs model-free values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    stickiness_1: [-5, 5] - Log-odds bonus for repeating the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Log-odds bonus for repeating the previous Stage 2 choice.
    """
    learning_rate, decay_rate, w, beta, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values at 0.5 to match the decay target
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Q-learning updates for chosen actions
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay_rate * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] += decay_rate * (0.5 - q_stage2_mf[other_state, 0])
        q_stage2_mf[other_state, 1] += decay_rate * (0.5 - q_stage2_mf[other_state, 1])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Hybrid MB/MF with Separate Stage 1 and Stage 2 Learning Rates.
    
    This model allows for different learning rates at Stage 1 versus Stage 2. 
    Participants might update their planetary alien preferences (Stage 2) more or 
    less aggressively than their spaceship preferences (Stage 1).
    
    Parameters:
    lr_stage1: [0, 1] - Learning rate for Stage 1 Q-values.
    lr_stage2: [0, 1] - Learning rate for Stage 2 Q-values.
    w: [0, 1] - Weight given to model-based vs model-free values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    stickiness_1: [-5, 5] - Log-odds bonus for repeating the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Log-odds bonus for repeating the previous Stage 2 choice.

    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
