Here are 3 new cognitive models that build upon the best model so far. Since `decay` (forgetting of unchosen Q-values) was found to improve the model fit, each of these new models incorporates the `decay` mechanism alongside other well-known computational variations (e.g., separate stage learning rates, stage 2 perseveration, and asymmetric learning rates) to capture more nuanced participant behavior. 

### Model 1: Separate Stage 1 and Stage 2 Learning Rates + Decay
Participants might update their expectations at different speeds for the first-stage choices (spaceships) compared to the second-stage choices (aliens). This model splits the learning rate into `learning_rate_1` and `learning_rate_2`.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw 2011 model with separate learning rates for stage 1 and stage 2, plus Q-value decay.
    
    Parameters:
    learning_rate_1: [0, 1] - Learning rate for stage 1 Q-value updates.
    learning_rate_2: [0, 1] - Learning rate for stage 2 Q-value updates.
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax choice.
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax choice.
    w: [0, 1] - Weight given to model-based Q-values for stage 1 (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for stage 1 MF update.
    stickiness: [-5, 5] - Choice perseveration bonus for repeating the previous stage 1 action.
    decay: [0, 1] - Decay rate of Q-values towards 0.5 for unchosen actions.
    """
    learning_rate_1, learning_rate_2, beta_1, beta_2, w, lam, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            stickiness_bonus[last_action] = stickiness
            
        logits_1 = beta_1 * q_stage1_net + stickiness_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action = action_1[trial]
            continue
            
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate_1 * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage 1 & Stage 2 Perseveration (Stickiness) + Decay
Often, participants demonstrate choice perseveration (or alternation) not just for the first-stage choices, but also for the second-stage choices (aliens) on a given planet. This model introduces `stickiness_2` to capture state-dependent stage 2 perseveration.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw 2011 model with stage 1 and stage 2 choice perseveration (stickiness) and Q-value decay.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax choice.
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax choice.
    w: [0, 1] - Weight given to model-based Q-values for stage 1 (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for stage 1 MF update.
    stickiness_1: [-5, 5] - Choice perseveration bonus for repeating the previous stage 1 action.
    stickiness_2: [-5, 5] - Choice perseveration bonus for repeating the previous stage 2 action in the same state.
    decay: [0, 1] - Decay rate of Q-values towards 0.5 for unchosen actions.
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness_1, stickiness_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus_1[last_action_1] = stickiness_1
            
        logits_1 = beta_1 * q_stage1_net + stickiness_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action_1 = action_1[trial]
            continue
            
        stickiness_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_bonus_2[last_action_2[state_idx]] = stickiness_2

        logits_2 = beta_2 * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates + Decay
Participants often exhibit a positivity bias (or negativity bias) where they learn more from better-than-expected outcomes compared to worse-than-expected ones. This model splits the learning rate by the sign of the prediction error (`alpha_pos` and `alpha_neg`).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw 2011 model with asymmetric learning rates for positive/negative prediction errors and Q-value decay.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax choice.
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax choice.
    w: [0, 1] - Weight given to model-based Q-values for stage 1 (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for stage 1 MF update.
    stickiness: [-5, 5] - Choice perseveration bonus for repeating the previous stage 1 action.
    decay: [0, 1] - Decay rate of Q-values towards 0.5 for unchosen actions.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            stickiness_bonus[last_action] = stickiness
            
        logits_1 = beta_1 * q_stage1_net + stickiness_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action = action_1[trial]
            continue
            
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        # update stage 1 with eligibility trace
        q_stage1_mf[action_1[trial]] += lr_2 * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
