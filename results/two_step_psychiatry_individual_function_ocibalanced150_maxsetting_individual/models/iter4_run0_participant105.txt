def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate temperatures and perseveration for stage 1 and stage 2.
    
    This model extends the standard hybrid model-based/model-free reinforcement 
    learning algorithm by allowing for different levels of stochasticity (inverse 
    temperature) at stage 1 versus stage 2. It also includes separate choice 
    perseveration (stickiness) parameters for both stages, capturing the tendency 
    to repeat the previous first-stage action and the previous second-stage action 
    (conditional on the state).
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace for updating stage 1 action values from stage 2 rewards [0,1]
    pers1: stage 1 choice perseveration [-5,5]
    pers2: stage 2 choice perseveration [-5,5]
    """
    alpha, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus1 = np.zeros(2)
        if last_action1 != -1:
            pers_bonus1[last_action1] = pers1
            
        q1_total = beta1 * q_stage1_net + pers_bonus1
        q1_max = np.max(q1_total)
        exp_q1 = np.exp(q1_total - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            pers_bonus2[last_action2[state_idx]] = pers2
            
        q2_total = beta2 * q_stage2_mf[state_idx] + pers_bonus2
        q2_max = np.max(q2_total)
        exp_q2 = np.exp(q2_total - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive/negative prediction errors, 
    separate temperatures for stage 1 and stage 2, and stage 1 perseveration.
    
    This model allows agents to learn differently from better-than-expected versus 
    worse-than-expected outcomes by implementing separate learning rates (alpha_pos 
    and alpha_neg). It also maintains separate inverse temperatures for both stages 
    and incorporates a general choice perseveration effect for the first stage.
    
    Parameters:
    alpha_pos: learning rate for positive PE [0,1]
    alpha_neg: learning rate for negative PE [0,1]
    beta1: stage 1 inverse temperature [0,10]
    beta2: stage 2 inverse temperature [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace [0,1]
    pers: stage 1 choice perseveration [-5,5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action1 != -1:
            pers_bonus[last_action1] = pers
            
        q1_total = beta1 * q_stage1_net + pers_bonus
        q1_max = np.max(q1_total)
        exp_q1 = np.exp(q1_total - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2_total)
        exp_q2 = np.exp(q2_total - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Win-stay/Lose-shift (WSLS) mechanism alongside MB/MF learning.
    
    In addition to model-based and model-free learning, participants often employ 
    simple heuristics. This model introduces a win-stay/lose-shift (WSLS) parameter 
    that adds a direct bonus to the probability of repeating the last first-stage 
    action if it ultimately led to a reward, or penalizes it if it led to no reward.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: weight of model-based vs model-free control [0,1]
    lam: eligibility trace [0,1]
    pers: stage 1 general choice perseveration [-5,5]
    wsls: win-stay/lose-shift bonus for stage 1 [-5,5]
    """
    alpha, beta, w, lam, pers, wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus = np.zeros(2)
        if last_action1 != -1:
            bonus[last_action1] += pers
            if last_reward > 0:
                bonus[last_action1] += wsls
            else:
                bonus[last_action1] -= wsls
            
        q1_total = beta * q_stage1_net + bonus
        q1_max = np.max(q1_total)
        exp_q1 = np.exp(q1_total - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_total = beta * q_stage2_mf[state_idx]
        q2_max = np.max(q2_total)
        exp_q2 = np.exp(q2_total - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
