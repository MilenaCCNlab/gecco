import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and betas for stage 1 and stage 2, with value decay, eligibility trace, and stage 1 perseveration.
    Bounds:
    alpha_1: learning rate stage 1 [0, 1]
    alpha_2: learning rate stage 2 [0, 1]
    decay: decay rate for unchosen options [0, 1]
    beta_1: inverse temperature stage 1 [0, 10]
    beta_2: inverse temperature stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p_1: perseveration parameter for stage 1 [-5, 5]
    """
    alpha_1, alpha_2, decay, beta_1, beta_2, w, lambda_, p_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + p_1 * pers_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2

        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for positive and negative RPEs, separate decay for stage 1 and stage 2, eligibility trace, and stage 1 perseveration.
    Bounds:
    alpha_pos: learning rate for positive RPE [0, 1]
    alpha_neg: learning rate for negative RPE [0, 1]
    decay_1: decay rate for unchosen options in stage 1 [0, 1]
    decay_2: decay rate for unchosen options in stage 2 [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p_1: perseveration parameter for stage 1 [-5, 5]
    """
    alpha_pos, alpha_neg, decay_1, decay_2, beta, w, lambda_, p_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):

            p_choice_2[trial] = 1.0
            continue

            continue

        # policy for the second choice

        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

            for a in range(2):

                    q_stage2_mf[s, a] *= (1 - decay_2)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate betas for stage 1 and stage 2, learning with decay, and reward-dependent perseveration (win-stay, lose-shift) at stage 1.
    Bounds:
    alpha: learning rate [0, 1]
    decay: decay rate for unchosen options [0, 1]
    beta_1: inverse temperature stage 1 [0, 10]
    beta_2: inverse temperature stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p_win: perseveration parameter for stage 1 after a win [-5, 5]
    p_lose: perseveration parameter for stage 1 after a loss [-5, 5]
    """
    alpha, decay, beta_1, beta_2, w, lambda_, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice

        q_stage1_mb = transition_matrix @ max_q_stage2

        pers_1 = np.zeros(2)

        p_weight = p_win if last_reward > 0 else p_lose
        exp_q1 = np.exp(beta_1 * q_stage1_net + p_weight * pers_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

            continue

        # policy for the second choice

        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # eligibility trace update

        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):

        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
