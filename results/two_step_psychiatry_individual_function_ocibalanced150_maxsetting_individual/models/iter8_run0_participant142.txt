def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate learning rates for positive and negative prediction errors,
    a decay mechanism for unchosen options towards 0.5, separate MB and MF inverse temperatures
    at stage 1, and stage 1 choice perseverance.
    
    Bounds:
    alpha_pos: [0,1] - learning rate for positive Q-value updates
    alpha_neg: [0,1] - learning rate for negative Q-value updates
    decay: [0,1] - decay rate towards 0.5 for unchosen options
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers: [-5,5] - choice perseverance for stage 1
    """
    alpha_pos, alpha_neg, decay, beta_mb, beta_mf, beta2, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exponent2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # action value updating for choice 2
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with learnable transition matrix, decay mechanism for unchosen options towards 0.5,
    separate MB and MF inverse temperatures at stage 1, and stage 1 choice perseverance.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    alpha_trans: [0,1] - learning rate for transition probabilities
    decay: [0,1] - decay rate towards 0.5 for unchosen options
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers: [-5,5] - choice perseverance for stage 1
    """

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

        probs_2 = exp_q2 / np.sum(exp_q2)

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
