import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Positive/Negative learning rates, decay, eligibility trace, and separate perseveration for stage 1 and stage 2.
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    decay: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lambda_: [0, 1]
    p_1: [-5, 5]
    p_2: [-5, 5]
    """
    alpha_pos, alpha_neg, decay, beta, w, lambda_, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) 

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        logits_1 = beta * q_stage1_net + p_1 * pers_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue

        # policy for the second choice
        pers_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_2[last_action_2[state_idx]] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + p_2 * pers_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2[state_idx] = action_2[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        # eligibility trace update
        alpha_s1_lambda = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1_lambda * lambda_ * delta_stage2

        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Learning with decay, Eligibility Trace, win-stay/lose-shift for stage 1, and perseveration for stage 2.
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lambda_: [0, 1]
    p_win: [-5, 5]
    p_lose: [-5, 5]
    p_2: [-5, 5]
    """
    alpha, decay, beta, w, lambda_, p_win, p_lose, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_reward = 0
    last_action_2 = np.array([-1, -1]) 

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                pers_1[last_action_1] = p_win
            else:
                pers_1[last_action_1] = p_lose
            
        logits_1 = beta * q_stage1_net + pers_1

        last_action_2[state_idx] = action_2[trial]
        last_reward = reward[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
