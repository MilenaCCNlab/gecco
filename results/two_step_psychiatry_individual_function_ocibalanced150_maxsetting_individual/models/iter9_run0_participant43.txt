import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two Learning Rates with Eligibility Trace and Stage 1 Bias
    
    This model extends the baseline by separating the learning rates for Stage 1 
    and Stage 2, which allows for different speeds of value updating across stages. 
    It also includes an eligibility trace parameter (lam) that updates Stage 1 
    values based on the Stage 2 prediction error (reward), and a bias parameter 
    to capture the participant's strong baseline preference for spaceship 0.
    
    Parameters:
    learning_rate_1: [0,1] - Learning rate for Stage 1 Q-values
    learning_rate_2: [0,1] - Learning rate for Stage 2 Q-values
    beta: [0,10] - Inverse temperature for softmax choice
    w: [0,1] - Weighting of model-based (1) vs model-free (0) values
    stickiness_1: [-5,5] - Tendency to repeat the previous stage 1 action
    stickiness_2: [-5,5] - Tendency to repeat the previous stage 2 action on the same planet
    bias_1: [-5,5] - Constant bias towards choosing spaceship 0
    lam: [0,1] - Eligibility trace parameter for updating Stage 1 from Stage 2 reward
    """
    learning_rate_1, learning_rate_2, beta, w, stickiness_1, stickiness_2, bias_1, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        bias_vector_1 = np.array([bias_1, 0.0])
            
        logits_1 = beta * q_net + stickiness_1 * stick_vector_1 + bias_vector_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2
        stick_vector_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stick_vector_2[last_action_2[state_idx]] = 1.0
            
        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_2 * stick_vector_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates and Dual Temperatures with Stage 1 Bias
    
    This model allows learning from positive prediction errors to occur at a 
    different rate than negative prediction errors (learning_rate_pos vs 
    learning_rate_neg). This captures asymmetric learning tendencies. 
    It also separates the softmax inverse temperature for Stage 1 and Stage 2, 
    accounting for different levels of choice stochasticity in each stage.
    
    Parameters:
    learning_rate_pos: [0,1] - Learning rate for positive prediction errors
    learning_rate_neg: [0,1] - Learning rate for negative prediction errors
    beta_1: [0,10] - Inverse temperature for stage 1 choice
    beta_2: [0,10] - Inverse temperature for stage 2 choice
    w: [0,1] - Weighting of model-based (1) vs model-free (0) values
    stickiness_1: [-5,5] - Tendency to repeat the previous stage 1 action

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Win-Stay Lose-Shift Stickiness for Both Stages with Stage 1 Bias
    
    This model differentiates between stickiness following a rewarded trial versus 
    an unrewarded trial, explicitly modeling win-stay and lose-shift heuristics for 
    both stages. It retains the Stage 1 bias to account for the participant's strong 
    baseline preference for spaceship 0.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-values
    beta: [0,10] - Inverse temperature for softmax choice
    w: [0,1] - Weighting of model-based (1) vs model-free (0) values
    stick_win_1: [-5,5] - Tendency to repeat stage 1 action if previous trial was rewarded
    stick_loss_1: [-5,5] - Tendency to repeat stage 1 action if previous trial was not rewarded
    stick_win_2: [-5,5] - Tendency to repeat stage 2 action if previous trial on that planet was rewarded
    stick_loss_2: [-5,5] - Tendency to repeat stage 2 action if previous trial on that planet was not rewarded
    bias_1: [-5,5] - Constant bias towards choosing spaceship 0
    """
    learning_rate, beta, w, stick_win_1, stick_loss_1, stick_win_2, stick_loss_2, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward_1 = -1
    last_action_2 = np.full(2, -1)
    last_reward_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward_1 == 1:
                stick_vector_1[last_action_1] = stick_win_1
            elif last_reward_1 == 0:
                stick_vector_1[last_action_1] = stick_loss_1
            
        bias_vector_1 = np.array([bias_1, 0.0])
            
        logits_1 = beta * q_net + stick_vector_1 + bias_vector_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2
        stick_vector_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            if last_reward_2[state_idx] == 1:
                stick_vector_2[last_action_2[state_idx]] = stick_win_2
            elif last_reward_2[state_idx] == 0:
                stick_vector_2[last_action_2[state_idx]] = stick_loss_2
            
        logits_2 = beta * q_stage2_mf[state_idx] + stick_vector_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward_1 = reward[trial]
        last_action_2[state_idx] = action_2[trial]
        last_reward_2[state_idx] = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
