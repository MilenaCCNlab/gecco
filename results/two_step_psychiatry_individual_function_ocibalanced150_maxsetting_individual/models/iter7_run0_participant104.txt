import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual Learning Rates and Reward-Dependent Stage 1 Perseveration
    Uses separate learning rates for Stage 1 and Stage 2. Stage 1 perseveration 
    is split into win-stay (rep1_win) and lose-stay (rep1_lose) based on the 
    previous trial's reward, capturing differential repetition tendencies after 
    success versus failure.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1.
    alpha2: [0, 1] - Learning rate for Stage 2.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - MB weight.
    lam: [0, 1] - Eligibility trace.
    rep1_win: [-5, 5] - Perseveration bonus for Stage 1 if previous trial was rewarded.
    rep1_lose: [-5, 5] - Perseveration bonus for Stage 1 if previous trial was unrewarded.
    """
    alpha1, alpha2, beta1, beta2, w, lam, rep1_win, rep1_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if last_action_1 != -1:
            if last_reward == 1:
                logits_1[last_action_1] += rep1_win
            elif last_reward == 0:
                logits_1[last_action_1] += rep1_lose
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx].copy()
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
                    
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates and Dual Decay Rates
    Uses different learning rates for positive and negative prediction errors, 
    and separate decay (forgetting) rates for unchosen actions in Stage 1 and Stage 2.
    This captures differential learning from better-than-expected versus worse-than-expected
    outcomes, alongside stage-specific forgetting.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive RPEs.
    alpha_neg: [0, 1] - Learning rate for negative RPEs.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - MB weight.
    lam: [0, 1] - Eligibility trace.
    phi_stage1: [0, 1] - Decay rate for Stage 1 unchosen actions.
    phi_stage2: [0, 1] - Decay rate for Stage 2 unchosen actions.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, phi_stage1, phi_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx].copy()
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        # Update stage 1 with decay for unchosen
        for a in range(2):
            if a == action_1[trial]:
                q_stage1_mf[a] += alpha1 * delta_stage1
                q_stage1_mf[a] += alpha2 * lam * delta_stage2
            else:
                q_stage1_mf[a] *= (1 - phi_stage1)

        # Update stage 2 with decay for unchosen
        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += alpha2 * delta_stage2
                else:
                    q_stage2_mf[s, a] *= (1 - phi_stage2)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Learning Rates, Win-Stay Stage 1, and Stage 2 Perseveration
    Uses separate learning rates for positive and negative prediction errors. Includes 
    a Stage 1 perseveration bonus only after a rewarded trial (win-stay), and a general 
    Stage 2 state-specific perseveration bonus regardless of the previous outcome.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive RPE.
    alpha_neg: [0, 1] - Learning rate for negative RPE.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - MB weight.
    lam: [0, 1] - Eligibility trace.
    rep1_win: [-5, 5] - Stage 1 perseveration bonus after a reward.
    rep2: [-5, 5] - Stage 2 perseveration bonus (state-specific).
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, rep1_win, rep2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if last_action_1 != -1 and last_reward == 1:
            logits_1[last_action_1] += rep1_win
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta2 * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += rep2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha2 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
                    
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
