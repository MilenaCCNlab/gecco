def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice trace perseveration.
    Instead of just the last trial, stage 1 perseveration is based on an exponentially decaying choice trace.
    This captures long-term stickiness to a particular spaceship regardless of recent rewards.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p_c: choice trace weight [0,10]
    alpha_c: choice trace learning rate [0,1]
    """
    alpha, beta_1, beta_2, w, lam, p_c, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta_1 * q_stage1_net + p_c * choice_trace)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # update choice trace
        choice_trace *= (1 - alpha_c)
        choice_trace[action_1[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition-dependent Win-Stay.
    Differentiates perseveration after a rewarded trial based on whether the transition was common or rare.
    This captures the classic behavioral signature of model-based vs model-free control directly in the heuristic stickiness.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p_win_common: perseveration if previous trial was rewarded and transition was common [0,10]
    p_win_rare: perseveration if previous trial was rewarded and transition was rare [0,10]
    p_lose: perseveration if previous trial was not rewarded [0,10]
    """
    alpha, beta_1, beta_2, w, lam, p_win_common, p_win_rare, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0
    last_transition_common = False

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        current_p = 0.0
        if last_action_1 != -1:
            if last_reward > 0:
                if last_transition_common:
                    current_p = p_win_common
                else:
                    current_p = p_win_rare
            else:
                current_p = p_lose
                
        exp_q1 = np.exp(beta_1 * q_stage1_net + current_p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        # Common transitions are 0->0 and 1->1
        last_transition_common = (action_1[trial] == state[trial])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Valence-dependent eligibility trace.
    Allows for different eligibility traces (lam_pos, lam_neg) based on the sign of the stage 2 prediction error,
    along with reward-dependent perseveration. This captures optimism bias in reward backpropagation.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    lam_pos: eligibility trace for positive stage 2 PE [0,1]
    lam_neg: eligibility trace for negative stage 2 PE [0,1]
    p_win: stickiness if previous trial was rewarded [0,10]
    p_lose: stickiness if previous trial was not rewarded [0,10]
    """
    alpha, beta_1, beta_2, w, lam_pos, lam_neg, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        current_p = 0.0
        if last_action_1 != -1:
            current_p = p_win if last_reward > 0 else p_lose
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + current_p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Valence-dependent eligibility trace update for stage 1
        current_lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * current_lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
