def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Stage-Specific Temperatures and Perseveration
    Uses different inverse temperatures and different perseveration (stickiness) 
    parameters for stage 1 and stage 2 choices. The stage 2 perseveration is 
    state-specific (i.e., repeating the last action taken in that specific state).
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    p_1: stage 1 perseveration [0,10]
    p_2: stage 2 perseveration [0,10]
    """
    alpha, beta_1, beta_2, w, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + p_1 * pers_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_2 = np.zeros(2)
        if last_action_2_in_state[state_idx] != -1:
            pers_2[last_action_2_in_state[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2 * pers_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Stage-Specific Learning Rates with Eligibility Trace
    Uses different learning rates for stage 1 and stage 2 updates, a shared 
    inverse temperature, an eligibility trace (lambda) for stage 1 updating, 
    and stage 1 perseveration.
    
    Parameters:
    alpha_1: stage 1 learning rate [0,1]
    alpha_2: stage 2 learning rate [0,1]
    beta: shared inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: stage 1 perseveration [0,10]
    """
    alpha_1, alpha_2, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Learning with Eligibility Trace
    Differentiates learning rates for positive and negative prediction errors,
    includes an eligibility trace (lambda) for stage 1 updating, stage-specific 
    inverse temperatures, and stage 1 perseveration.
    
    Parameters:
    alpha_pos: learning rate for positive PE [0,1]
    alpha_neg: learning rate for negative PE [0,1]
    beta_1: stage 1 inverse temperature [0,10]
    beta_2: stage 2 inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: stage 1 perseveration [0,10]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + p * pers_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
