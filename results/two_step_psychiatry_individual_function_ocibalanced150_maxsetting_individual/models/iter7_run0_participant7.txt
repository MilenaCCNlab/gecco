Here are 3 new cognitive models that explore alternative mechanisms and parameter configurations to explain the participant's behavior.

### Model 1: Asymmetric Learning Rates with Shared Beta
This model uses separate learning rates for positive and negative prediction errors (which is often seen in human learning where people learn differently from better-than-expected vs. worse-than-expected outcomes). It pairs this with a shared inverse temperature and separate perseveration for wins and losses.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative prediction errors,
    shared beta, model-based weight, eligibility trace, and separate perseveration 
    for win and lose trials.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta: shared inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    """
    alpha_pos, alpha_neg, beta, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose

        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Transition-Dependent Perseveration
Instead of perseverating based on reward (win/lose), this model tests the hypothesis that the participant's tendency to repeat the Stage 1 choice depends on whether the previous transition was expected (common) or unexpected (rare), integrating separate learning rates and betas for both stages.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for Stage 1 and Stage 2, separate betas,
    and perseveration that depends on whether the previous transition was common or rare.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_common: perseveration bonus if previous transition was common [-5,5]
    pers_rare: perseveration bonus if previous transition was rare [-5,5]
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers_common, pers_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_common = False

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action >= 0:
            if last_trans_common:
                q_net[last_action] += pers_common
            else:
                q_net[last_action] += pers_rare

        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]
        # Transition is common if action 0 -> state 0 OR action 1 -> state 1
        last_trans_common = (action_1[trial] == state_idx)

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Value Forgetting
This model incorporates a `forgetting_rate` mechanism, which assumes that the Q-values of unchosen or less frequently chosen options gradually decay towards zero as the task progresses. This can explain the dynamic shifts in preference without relying on complex perseveration states.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for Stage 1 and Stage 2, shared beta,
    and a forgetting rate that decays all Q-values towards 0 on each trial.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta: shared inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers: perseveration bonus [-5,5]
    forgetting_rate: decays Q-values towards 0 [0,1]
    """
    alpha1, alpha2, beta, w, lam, pers, forgetting_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action >= 0:
            q_net[last_action] += pers

        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        # Forgetting: decay all Q-values towards 0
        q_stage1_mf *= (1 - forgetting_rate)
        q_stage2_mf *= (1 - forgetting_rate)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
