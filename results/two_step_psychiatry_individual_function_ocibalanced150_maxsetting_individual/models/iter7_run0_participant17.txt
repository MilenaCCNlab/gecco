import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with decay for unchosen actions towards a baseline (decay_target).
    This model assumes that unchosen actions do not simply decay to 0, but rather to an expected baseline value,
    reflecting the average expected reward in the environment.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen action Q-value updates.
    decay: [0, 1] - Decay rate for unchosen action Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter for stage 2 reward updating stage 1 Q-values.
    decay_target: [0, 1] - The baseline value that unchosen Q-values decay towards.
    """
    learning_rate, decay, beta, w, lam, decay_target = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to the decay target instead of 0
    q_stage1_mf = np.full(2, decay_target)
    q_stage2_mf = np.full((2, 2), decay_target)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # decay for unchosen actions towards decay_target
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (decay_target - q_stage1_mf[unchosen_1])
        
        for state_i in range(2):
            for action_j in range(2):
                if not (state_i == state_idx and action_j == action_2[trial]):
                    q_stage2_mf[state_i, action_j] += decay * (decay_target - q_stage2_mf[state_i, action_j])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with decay for unchosen actions and dynamic learning of transition probabilities.
    Instead of assuming fixed transition probabilities (0.7/0.3), the participant learns the transition structure
    over time based on the states they end up in.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen action Q-value updates.
    decay: [0, 1] - Decay rate for unchosen action Q-values toward 0.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    transition_lr: [0, 1] - Learning rate for updating transition probabilities based on observed transitions.
    """
    learning_rate, decay, beta, w, lam, transition_lr = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix at 0.5 (maximum uncertainty)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # update transition probabilities
        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += transition_lr * (1 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += transition_lr * (0 - transition_matrix[a1, 1 - s2])

        # decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] -= decay * q_stage1_mf[unchosen_1]
        
        for state_i in range(2):
            for action_j in range(2):
                if not (state_i == state_idx and action_j == action_2[trial]):
                    q_stage2_mf[state_i, action_j] -= decay * q_stage2_mf[state_i, action_j]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate decay parameters for stage 1 and stage 2 unchosen actions.
    This allows for different forgetting rates depending on the depth of the task structure.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen action Q-value updates.
    decay1: [0, 1] - Decay rate for unchosen action Q-values at stage 1 toward 0.
    decay2: [0, 1] - Decay rate for unchosen action Q-values at stage 2 toward 0.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    """
    learning_rate, decay1, decay2, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # separate decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] -= decay1 * q_stage1_mf[unchosen_1]
        
        for state_i in range(2):

                if not (state_i == state_idx and action_j == action_2[trial]):

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
