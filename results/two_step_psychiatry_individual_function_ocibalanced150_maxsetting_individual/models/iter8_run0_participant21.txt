def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Stage-Specific Learning Rates, Forgetting, and Reward-Dependent Stickiness.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1.
    alpha2: [0, 1] - Learning rate for stage 2.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) evaluations.
    lam: [0, 1] - Eligibility trace parameter for stage 1 updating.
    decay: [0, 1] - Rate at which Q-values of unchosen actions decay towards 0.
    stay_rew: [-5, 5] - Stickiness for previous stage 1 action if it was rewarded.
    stay_unrew: [-5, 5] - Stickiness for previous stage 1 action if it was unrewarded.
    """
    alpha1, alpha2, beta, w, lam, decay, stay_rew, stay_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            if last_reward > 0:
                stickiness_bonus[last_action] = stay_rew
            else:
                stickiness_bonus[last_action] = stay_unrew
                
        q_net_total = q_net + stickiness_bonus
        
        exp_q1 = np.exp(beta * (q_net_total - np.max(q_net_total)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1]
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a]
                    
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Forgetting and a Decaying Choice Trace.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) evaluations.
    lam: [0, 1] - Eligibility trace parameter for stage 1 updating.
    decay: [0, 1] - Rate at which Q-values of unchosen actions decay towards 0.
    tau: [0, 1] - Decay rate for the choice trace (autocorrelation).
    stickiness: [-5, 5] - Weight of the choice trace on stage 1 action selection.
    """
    alpha, beta, w, lam, decay, tau, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_total = q_net + stickiness * choice_trace
        
        exp_q1 = np.exp(beta * (q_net_total - np.max(q_net_total)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1]
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a]
                    
        # Update choice trace
        choice_trace = (1 - tau) * choice_trace
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Stage-Specific Asymmetric Learning Rates and Forgetting.
    
    Parameters:
    alpha_pos1: [0, 1] - Learning rate for positive PEs at stage 1.
    alpha_neg1: [0, 1] - Learning rate for negative PEs at stage 1.
    alpha_pos2: [0, 1] - Learning rate for positive PEs at stage 2.
    alpha_neg2: [0, 1] - Learning rate for negative PEs at stage 2.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) evaluations.
    lam: [0, 1] - Eligibility trace parameter for stage 1 updating.
    decay: [0, 1] - Rate at which Q-values of unchosen actions decay towards 0.
    """
    alpha_pos1, alpha_neg1, alpha_pos2, alpha_neg2, beta, w, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos1 if delta_stage1 > 0 else alpha_neg1
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos2 if delta_stage2 > 0 else alpha_neg2
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for stage 1
        alpha_lambda = alpha_pos1 if delta_stage2 > 0 else alpha_neg1
        q_stage1_mf[action_1[trial]] += alpha_lambda * lam * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1]
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
