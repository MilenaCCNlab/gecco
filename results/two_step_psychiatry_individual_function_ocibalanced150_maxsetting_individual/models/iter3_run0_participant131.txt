def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Negative Valuation of Omitted Rewards.
    
    Participants may perceive the omission of a reward (0 coins) not as a neutral 
    outcome, but as an active penalty. This model assigns a negative value (-penalty) 
    when the reward is 0, accelerating the decrease in Q-value for unrewarded actions.
    This can explain rapid switching behavior after unrewarded trials.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    penalty: [0, 1] - Magnitude of the negative value assigned to unrewarded trials.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    """
    learning_rate, penalty, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Apply penalty if reward is 0, otherwise standard reward
        actual_reward = reward[trial] - penalty * (1 - reward[trial])
        delta_stage2 = actual_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Choice-Induced Preference (Value Inflation).
    
    This model posits that the mere act of choosing an alien increases its 
    perceived value, independent of the reward received. This is modeled by 
    adding a choice_bonus to the reward signal during the Stage 2 update. 
    This mechanism naturally accounts for strong perseveration or "stickiness" 
    in participant behavior, even through unrewarded streaks.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    choice_bonus: [0, 1] - Intrinsic reward/value added to the chosen action.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    """
    learning_rate, choice_bonus, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Inflate the effective reward by the choice bonus
        effective_reward = reward[trial] + choice_bonus
        delta_stage2 = effective_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Cross-Planet Generalization.
    
    Participants might confuse the aliens across planets or assume a common 
    reward structure (e.g., "alien 0 is good everywhere"). When a reward is 
    received from an alien on one planet, this model slightly updates the value 
    of the alien with the same index on the other planet, controlled by a 
    generalization parameter.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    generalization: [0, 1] - Rate at which learning generalizes to the other planet.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    """
    learning_rate, generalization, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Generalize to the corresponding alien on the other planet
        other_state = 1 - state_idx
        delta_gen = reward[trial] - q_stage2_mf[other_state, action_2[trial]]
        q_stage2_mf[other_state, action_2[trial]] += learning_rate * generalization * delta_gen
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
