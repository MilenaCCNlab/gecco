def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual learning rates and dual temperatures.
    This model uses separate learning rates and inverse temperatures for the two stages,
    allowing for differences in learning speed and choice stochasticity between stage 1 and stage 2.
    It also includes a model-based weight and a perseveration bonus for stage 1.
    
    Parameters:
    alpha1 (float): Learning rate for stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for stage 2. Bounds: [0,1]
    beta1 (float): Inverse temperature for stage 1. Bounds: [0,10]
    beta2 (float): Inverse temperature for stage 2. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    pers (float): Perseveration bonus for repeating the previous stage 1 choice. Bounds: [0,5]
    """
    alpha1, alpha2, beta1, beta2, w, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_net[last_action] += pers
            
        exp_q1 = np.exp(beta1 * q_net - np.max(beta1 * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * q2 - np.max(beta2 * q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Stage-specific perseveration and eligibility trace.
    This model includes an eligibility trace (lam) for updating stage 1 values based on stage 2 rewards,
    which accelerates learning. It also uses separate perseveration parameters for stage 1 and stage 2 choices,
    capturing varying degrees of stickiness across the two decision stages.
    
    Parameters:
    alpha (float): Learning rate. Bounds: [0,1]
    beta (float): Inverse temperature. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers1 (float): Perseveration for stage 1. Bounds: [0,5]
    pers2 (float): Perseveration for stage 2 (state-specific). Bounds: [0,5]
    """
    alpha, beta, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers1
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers2
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates for positive/negative prediction errors with eligibility trace.
    This model updates values differently based on whether the prediction error is positive (better than expected)
    or negative (worse than expected), and incorporates an eligibility trace to update stage 1 values from stage 2 outcomes.
    
    Parameters:
    alpha_pos (float): Learning rate for positive prediction errors. Bounds: [0,1]
    alpha_neg (float): Learning rate for negative prediction errors. Bounds: [0,1]
    beta (float): Inverse temperature. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers (float): Perseveration bonus for repeating the previous stage 1 choice. Bounds: [0,5]
    """
    alpha_pos, alpha_neg, beta, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_net[last_action] += pers
            
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta * q2 - np.max(beta * q2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
