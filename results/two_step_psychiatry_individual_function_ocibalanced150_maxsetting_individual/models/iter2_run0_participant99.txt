def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free learner with separate learning rates for Stage 1 and Stage 2,
    and perseveration at both stages.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-values.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-values.
    beta: [0, 10] - Inverse temperature for choice stochasticity.
    w: [0, 1] - Mixing weight (1 = pure model-based, 0 = pure model-free).
    lam: [0, 1] - Eligibility trace.
    pers1: [0, 5] - Perseveration bonus for repeating the previous stage 1 choice.
    pers2: [0, 5] - Perseveration bonus for repeating the previous stage 2 choice (in the same state).
    """
    alpha1, alpha2, beta, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus1[last_action_1] = pers1
            
        q_net_scaled = beta * (q_net + pers_bonus1)
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus2[last_action_2[state_idx]] = pers2
            
        q2_scaled = beta * (q_stage2_mf[state_idx] + pers_bonus2)
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free learner with continuous choice traces for perseveration.
    Instead of just the last action, choices leave a decaying trace that biases future decisions.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature for choice stochasticity.
    w: [0, 1] - Mixing weight (1 = pure model-based, 0 = pure model-free).
    lam: [0, 1] - Eligibility trace.
    pers1: [0, 5] - Weight of the Stage 1 choice trace on decision making.
    pers2: [0, 5] - Weight of the Stage 2 choice trace on decision making.
    choice_alpha: [0, 1] - Update rate for the choice traces (how quickly old choices are forgotten).
    """
    alpha, beta, w, lam, pers1, pers2, choice_alpha = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    trace1 = np.zeros(2)
    trace2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        q_net_scaled = beta * (q_net + pers1 * trace1)
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta * (q_stage2_mf[state_idx] + pers2 * trace2[state_idx])
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Update choice traces
        trace1 *= (1 - choice_alpha)
        trace1[action_1[trial]] += choice_alpha
        
        trace2[state_idx] *= (1 - choice_alpha)
        trace2[state_idx, action_2[trial]] += choice_alpha

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free learner with outcome-dependent perseveration at Stage 1 (Win-Stay/Lose-Shift).
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature for choice stochasticity.
    w: [0, 1] - Mixing weight (1 = pure model-based, 0 = pure model-free).
    lam: [0, 1] - Eligibility trace.
    pers_win: [0, 5] - Perseveration bonus for repeating Stage 1 choice if last trial was rewarded.
    pers_lose: [-5, 5] - Perseveration bonus for repeating Stage 1 choice if last trial was unrewarded (negative values imply lose-shift).
    pers2: [0, 5] - Perseveration bonus for repeating the previous stage 2 choice.
    """
    alpha, beta, w, lam, pers_win, pers_lose, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus1[last_action_1] = pers_win
            else:
                pers_bonus1[last_action_1] = pers_lose
            
        q_net_scaled = beta * (q_net + pers_bonus1)
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus2[last_action_2[state_idx]] = pers2
            
        q2_scaled = beta * (q_stage2_mf[state_idx] + pers_bonus2)
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
