def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with a subjective transition probability for the common transition.
    Instead of assuming the true 0.7/0.3 transition probabilities, this model fits 
    a subjective probability 'pi' that the participant uses for model-based valuation.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    w: [0,1] - Weight given to model-based vs model-free Q-values at Stage 1
    lam: [0,1] - Eligibility trace parameter for updating Stage 1 based on Stage 2 reward
    pers1: [-5, 5] - Perseveration bonus for repeating the previous Stage 1 action
    pers2: [-5, 5] - Perseveration bonus for repeating the previous Stage 2 action
    pi: [0.5, 1.0] - Subjective probability of the common transition
    """
    alpha, beta1, beta2, w, lam, pers1, pers2, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[pi, 1.0 - pi], [1.0 - pi, pi]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers1_bonus[last_action_1] = pers1
            
        q_stage1_total = beta1 * q_stage1_net + pers1_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2_bonus = np.zeros(2)
        if last_action_2 != -1:
            pers2_bonus[last_action_2] = pers2
            
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2_bonus
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model where the transition matrix is learned over time.
    Instead of assuming fixed transition probabilities, the participant dynamically 
    updates their transition model based on observed state transitions.
    
    Bounds:
    alpha: [0,1] - Learning rate for updating Q-values
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    w: [0,1] - Weight given to model-based vs model-free Q-values at Stage 1
    lam: [0,1] - Eligibility trace parameter for updating Stage 1 based on Stage 2 reward
    pers1: [-5, 5] - Perseveration bonus for repeating the previous Stage 1 action
    pers2: [-5, 5] - Perseveration bonus for repeating the previous Stage 2 action
    alpha_trans: [0,1] - Learning rate for updating the transition matrix probabilities
    """
    alpha, beta1, beta2, w, lam, pers1, pers2, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers1_bonus[last_action_1] = pers1
            
        q_stage1_total = beta1 * q_stage1_net + pers1_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2_bonus = np.zeros(2)
        if last_action_2 != -1:
            pers2_bonus[last_action_2] = pers2
            
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2_bonus
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Transition matrix update
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
