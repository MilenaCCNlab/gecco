def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with separate learning rates for Stage 1 and Stage 2, and choice stickiness at both stages.
    This model captures potential differences in how quickly participants update values at the two different stages of the task.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1.
    alpha_2: [0, 1] - Learning rate for stage 2.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p_1: [-5, 5] - Choice stickiness for stage 1.
    p_2: [-5, 5] - Choice stickiness for stage 2.
    lam: [0, 1] - Eligibility trace.
    """
    alpha_1, alpha_2, beta_1, beta_2, w, p_1, p_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + p_1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 Choice
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness_2[last_action_2] = 1.0
            
        logits_2 = beta_2 * q_stage2_mf[s] + p_2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha_1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha_2 * delta_stage2
        
        q_stage1_mf[a1] += alpha_1 * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Stage 1 stickiness and State-Specific Stage 2 stickiness.
    This captures perseveration specific to the planet visited (e.g., sticking to alien 0 on planet 0, independently of choices on planet 1).
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p_1: [-5, 5] - Choice stickiness for stage 1.
    p_2_state: [-5, 5] - State-specific choice stickiness for stage 2.
    lam: [0, 1] - Eligibility trace.
    """
    alpha, beta_1, beta_2, w, p_1, p_2_state, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_state = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])

        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 Choice
        stickiness_2 = np.zeros(2)
        if last_action_2_state[s] != -1:
            stickiness_2[last_action_2_state[s]] = 1.0
            
        logits_2 = beta_2 * q_stage2_mf[s] + p_2_state * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]

        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2_state[s] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Stage 1 Stickiness and Reward-Modulated Stage 2 Stickiness (Win-Stay / Lose-Shift).
    This model posits that perseveration at the second stage heavily depends on whether the last trial was rewarded.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p_1: [-5, 5] - Choice stickiness for stage 1.
    p_2_win: [-5, 5] - Stage 2 stickiness after a rewarded trial.
    p_2_lose: [-5, 5] - Stage 2 stickiness after an unrewarded trial.
    lam: [0, 1] - Eligibility trace.
    """
    alpha, beta_1, beta_2, w, p_1, p_2_win, p_2_lose, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + p_1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 Choice (Reward-Modulated)
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            if last_reward == 1:
                stickiness_2[last_action_2] = p_2_win
            elif last_reward == 0:
                stickiness_2[last_action_2] = p_2_lose
            
        logits_2 = beta_2 * q_stage2_mf[s] + stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
