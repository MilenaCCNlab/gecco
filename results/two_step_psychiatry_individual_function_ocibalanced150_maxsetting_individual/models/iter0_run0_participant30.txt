import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) two-step task model.
    This model implements a mixture of model-free and model-based learning, 
    along with an eligibility trace and choice perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    p: [-5, 5] - Choice perseveration (tendency to repeat the previous stage 1 choice)
    lam: [0, 1] - Eligibility trace (how much stage 2 reward updates stage 1 Q-value)
    """
    learning_rate, beta, w, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Handle invalid/missed trials
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Mixture of MB and MF values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Choice perseveration
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        val_1 = beta * q_net + p * pers
        val_1 -= np.max(val_1) # for numerical stability
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = s

        # Policy for the second choice
        val_2 = beta * q_stage2_mf[state_idx]
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw model with separate learning rates and inverse temperatures for Stage 1 and Stage 2.
    Allows for differential learning speeds and exploration/exploitation trade-offs across stages.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1
    alpha_2: [0, 1] - Learning rate for stage 2
    beta_1: [0, 10] - Inverse temperature for stage 1 action selection
    beta_2: [0, 10] - Inverse temperature for stage 2 action selection
    w: [0, 1] - Model-based weight
    p: [-5, 5] - Choice perseveration
    lam: [0, 1] - Eligibility trace
    """
    alpha_1, alpha_2, beta_1, beta_2, w, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        val_1 = beta_1 * q_net + p * pers
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = s

        # Policy for the second choice
        val_2 = beta_2 * q_stage2_mf[state_idx]
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha_1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha_1 * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw model with separate choice perseveration for rewarded and unrewarded previous trials.
    Captures asymmetric "win-stay, lose-shift" dynamics independent of Q-learning.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Model-based weight
    p_rew: [-5, 5] - Choice perseveration after a rewarded trial
    p_unrew: [-5, 5] - Choice perseveration after an unrewarded trial
    lam: [0, 1] - Eligibility trace
    """
    learning_rate, beta, w, p_rew, p_unrew, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        p_current = p_rew if last_reward == 1 else p_unrew
            
        val_1 = beta * q_net + p_current * pers
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = s

        # Policy for the second choice
        val_2 = beta * q_stage2_mf[state_idx]
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
