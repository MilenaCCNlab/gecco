def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Model-based and Model-free RL with outcome-dependent stickiness.
    This model separates the choice stickiness parameter into two: one for trials 
    following a rewarded outcome (pi_win) and one for trials following an unrewarded 
    outcome (pi_lose). This captures Win-Stay/Lose-Shift tendencies at the first stage.
    
    Bounds:
    learning_rate: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pi_win: [-5, 5]
    pi_lose: [-5, 5]
    """
    learning_rate, beta1, beta2, w, lam, pi_win, pi_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_r = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_a1 = -1
            last_r = 0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        if last_r > 0:
            pi = pi_win
        else:
            pi = pi_lose

        logits_1 = beta1 * q_stage1_net + pi * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        if a2 == -1:
            p_choice_2[trial] = 1.0
            last_a1 = a1
            last_r = r
            continue

        logits_2 = beta2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_a1 = a1
        last_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dual-system RL with separate inverse temperatures for MB and MF systems.
    Instead of a single inverse temperature and a weighting parameter 'w', this model 
    uses separate inverse temperatures (beta_mb and beta_mf) for the model-based and 
    model-free Q-values at stage 1.
    
    Bounds:
    learning_rate: [0, 1]
    beta_mb: [0, 10]
    beta_mf: [0, 10]
    beta2: [0, 10]
    lam: [0, 1]
    pi: [-5, 5]
    """
    learning_rate, beta_mb, beta_mf, beta2, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_a1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        if a2 == -1:
            p_choice_2[trial] = 1.0
            last_a1 = a1
            continue

        logits_2 = beta2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: RL with transition probability learning.
    Instead of assuming a fixed transition matrix throughout the task, the agent 
    starts with the instructed transition probabilities (0.7/0.3) but updates them 
    based on experienced transitions using a transition learning rate (alpha_trans).
    
    Bounds:
    learning_rate: [0, 1]
    alpha_trans: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pi: [-5, 5]
    """
    learning_rate, alpha_trans, beta1, beta2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_a1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pi * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Learn transition matrix based on observed state
        if state_idx != -1:
            transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
            transition_matrix[a1, 1 - state_idx] = 1.0 - transition_matrix[a1, state_idx]

        # policy for the second choice
        if a2 == -1:
            p_choice_2[trial] = 1.0
            last_a1 = a1
            continue

        logits_2 = beta2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
