Here are three new cognitive models that build upon the most successful mechanism discovered so far (value decay for unchosen actions). These models systematically combine the decay mechanism with other established reinforcement learning principles, such as separate decision temperatures, asymmetrical learning rates for positive and negative prediction errors, and separate learning rates across stages.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid model with separate temperatures for Stage 1 and Stage 2, and value decay.
    This model allows for different levels of exploration/exploitation at each stage 
    by using separate inverse temperatures (beta1 and beta2), while incorporating 
    forgetting (decay) for unchosen Stage 2 actions.
    
    Parameters:
    alpha: Learning rate for updating values [0, 1]
    beta1: Inverse temperature for Stage 1 choices [0, 10]
    beta2: Inverse temperature for Stage 2 choices [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Decay rate for unchosen actions towards 0.5 [0, 1]
    """
    alpha, beta1, beta2, w, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stage 1 choice
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay unchosen actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with asymmetrical learning rates (positive/negative PEs) and value decay.
    This model assumes that participants update values differently depending on whether 
    the outcome was better or worse than expected (alpha_pos vs alpha_neg), 
    while also decaying unchosen Stage 2 actions.
    
    Parameters:
    alpha_pos: Learning rate for positive prediction errors [0, 1]
    alpha_neg: Learning rate for negative prediction errors [0, 1]
    beta: Inverse temperature for choices [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Decay rate for unchosen actions towards 0.5 [0, 1]
    """
    alpha_pos, alpha_neg, beta, w, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 choice

        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
        
        # Eligibility trace update uses the alpha corresponding to the stage 2 PE
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2

        # Decay unchosen actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
