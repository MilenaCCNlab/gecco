import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Stage-specific learning rates, eligibility trace, stickiness, and unchosen value decay.
    Separating learning rates by stage can capture differences in how immediately participants 
    update primary versus secondary choices.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1.
    alpha_2: [0, 1] - Learning rate for stage 2.
    beta: [0, 10] - Inverse temperature for choice stochasticity.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness: [-5, 5] - Tendency to repeat the previous stage-1 choice regardless of value.
    decay: [0, 1] - Rate at which unchosen action values decay toward zero.
    """
    alpha_1, alpha_2, beta, w, lambda_, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness * stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        # Value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2  # eligibility trace update
        
        # Decay unchosen options
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Incorporates Win-Stay/Lose-Shift (WSLS) along with stickiness, trace, and decay.
    This model separates general perseveration (stickiness) from reward-driven perseveration (WSLS).
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness: [-5, 5] - Tendency to repeat the previous stage-1 choice regardless of value.
    wsls: [-5, 5] - Tendency to repeat the previous stage-1 choice if rewarded, or switch if not.
    decay: [0, 1] - Rate at which unchosen action values decay toward zero.
    """
    alpha, beta, w, lambda_, stickiness, wsls, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        wsls_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            if reward[trial - 1] > 0:
                wsls_var[action_1[trial - 1]] = 1.0
            else:
                wsls_var[1 - action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness * stick_var + wsls * wsls_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        # Value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2  # eligibility trace update
        
        # Decay unchosen options
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates (positive/negative PEs), stage-specific temperatures, 
    eligibility trace, stickiness, and unchosen value decay. This configuration is highly relevant 
    for psychiatric modeling where sensitivity to reward vs punishment may be unequal.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1 choice stochasticity.
    beta_2: [0, 10] - Inverse temperature for stage 2 choice stochasticity.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness: [-5, 5] - Tendency to repeat the previous stage-1 choice regardless of value.
    decay: [0, 1] - Rate at which unchosen action values decay toward zero.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + stickiness * stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        # Value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_s2 * lambda_ * delta_stage2  # eligibility trace update
        
        # Decay unchosen options
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
