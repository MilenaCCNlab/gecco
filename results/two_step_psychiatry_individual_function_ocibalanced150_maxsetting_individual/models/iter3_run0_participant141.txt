Here are three new cognitive models that explore alternative mechanisms not yet evaluated. They capture different psychological phenomena often observed in the two-step task: long-term choice stickiness (choice traces), asymmetric valuation of unrewarded outcomes, and transition-dependent perseveration.

### Model 1: Choice Trace Model
This model extends the concept of perseveration. Instead of only relying on the immediately preceding action, it maintains an exponentially decaying "choice trace." This allows the model to capture longer-term habitual stickiness to a spaceship, which is common when participants develop a strong preference over multiple trials.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with an exponentially decaying choice trace for first-stage actions.
    Instead of only perseverating on the immediately preceding action, the model
    maintains a choice trace that decays over time, capturing longer-term stickiness.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updating
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers: [-5, 5] - Perseveration weight on the choice trace
    tau: [0, 1] - Decay rate of the choice trace (0 = no memory, 1 = perfect memory)
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, pers, tau = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits1 = beta1 * q_net + pers * choice_trace
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update choice trace
        choice_trace *= tau
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Reward / Unrewarded Penalty Model
This model introduces a mechanism to weigh unrewarded trials differently. Rather than treating a 0-coin outcome purely as a neutral 0, it fits an `omega` parameter which acts as a penalty, effectively allowing the model to capture loss-aversion or asymmetric learning from positive versus negative feedback.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a separate penalty for unrewarded trials (omega).
    Instead of treating non-rewards as exactly 0, this model allows unrewarded
    outcomes to act as a negative reinforcer (or differently scaled), which can
    better capture loss-aversion or asymmetric learning from positive/negative feedback.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updating
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers: [-5, 5] - Perseveration parameter
    omega: [-1, 5] - Effective penalty for receiving a reward of 0
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, pers, omega = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits1 = beta1 * q_net + pers * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Apply omega penalty for unrewarded trials
        eff_reward = 1.0 if reward[trial] == 1 else -omega

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Perseveration Model
A key signature in two-step tasks is how behavior changes depending on the transition experienced. While the `w` parameter captures this through MB/MF valuations, participants sometimes show heuristic stickiness depending strictly on whether the transition was common or rare, completely independent of reward values. This model explicitly tests if stickiness differs based on the transition type.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition-dependent perseveration.
    The tendency to repeat a first-stage action might depend on whether the previous
    trial featured a common or rare transition, separating stickiness from MB/MF valuation.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updating
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers_common: [-5, 5] - Perseveration if the last transition was common
    pers_rare: [-5, 5] - Perseveration if the last transition was rare
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, pers_common, pers_rare = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_common = True # default assumption for the first trial

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        # Determine which perseveration parameter to use based on the last transition
        pers = pers_common if last_trans_common else pers_rare
            
        logits1 = beta1 * q_net + pers * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        
        # Check if current transition was common (action 0 -> state 0 OR action 1 -> state 1)
        if (action_1[trial] == 0 and state_idx == 0) or (action_1[trial] == 1 and state_idx == 1):
            last_trans_common = True
        else:
            last_trans_common = False

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
