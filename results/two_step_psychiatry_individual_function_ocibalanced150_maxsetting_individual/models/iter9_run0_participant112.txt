def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Win-stay / Lose-shift stickiness with decaying choice trace.
    Instead of only considering the immediately preceding trial, the tendency
    to repeat past actions is based on a choice trace that decays over time.
    The effect of this trace depends on the reward of the immediately preceding trial.
    
    Bounds:
    learning_rate: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness_win: [-5,5]
    stickiness_lose: [-5,5]
    choice_trace_decay: [0,1]
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness_win, stickiness_lose, choice_trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))
    
    last_reward = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_reward != -1:
            stickiness = stickiness_win if last_reward == 1 else stickiness_lose
            q_stage1_net += stickiness * trace_1
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_reward != -1:
            stickiness = stickiness_win if last_reward == 1 else stickiness_lose
            q_stage2_net += stickiness * trace_2[state_idx]

        exp_q2 = np.exp(beta_2 * (q_stage2_net - np.max(q_stage2_net)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Update choice traces
        trace_1 *= choice_trace_decay
        trace_1[action_1[trial]] += 1
        
        trace_2 *= choice_trace_decay
        trace_2[state_idx, action_2[trial]] += 1

        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Win-stay / Lose-shift stickiness with transition-dependent MB weight.
    The weight given to the model-based system (w) depends on whether the 
    previous trial was a common (w_common) or rare (w_rare) transition.
    
    Bounds:
    learning_rate: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w_common: [0,1]
    w_rare: [0,1]
    lam: [0,1]
    stickiness_win: [-5,5]
    stickiness_lose: [-5,5]
    """
    learning_rate, beta_1, beta_2, w_common, w_rare, lam, stickiness_win, stickiness_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = -1
    last_transition_common = True

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        current_w = w_common if last_transition_common else w_rare
        q_stage1_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        if last_action_1 != -1:
            stickiness = stickiness_win if last_reward == 1 else stickiness_lose
            q_stage1_net[last_action_1] += stickiness
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1 and last_state == state_idx:
            stickiness = stickiness_win if last_reward == 1 else stickiness_lose
            q_stage2_net[last_action_2] += stickiness

        exp_q2 = np.exp(beta_2 * (q_stage2_net - np.max(q_stage2_net)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]
        last_transition_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent Stage 1 inverse temperature (beta).
    The inverse temperature for Stage 1 (beta_1) changes depending on whether 
    the previous trial was rewarded (beta_win) or unrewarded (beta_lose).
    This captures changes in exploration/exploitation based on recent success,
    along with win-stay/lose-shift stickiness.
    
    Bounds:
    learning_rate: [0,1]
    beta_win: [0,10]
    beta_lose: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness_win: [-5,5]
    stickiness_lose: [-5,5]
    """
    learning_rate, beta_win, beta_lose, beta_2, w, lam, stickiness_win, stickiness_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        current_beta_1 = beta_win if last_reward == 1 else beta_lose
        if last_reward == -1:
            current_beta_1 = (beta_win + beta_lose) / 2.0
            
        if last_action_1 != -1:
            stickiness = stickiness_win if last_reward == 1 else stickiness_lose
            q_stage1_net[last_action_1] += stickiness
            
        exp_q1 = np.exp(current_beta_1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1 and last_state == state_idx:
            stickiness = stickiness_win if last_reward == 1 else stickiness_lose
            q_stage2_net[last_action_2] += stickiness

        exp_q2 = np.exp(beta_2 * (q_stage2_net - np.max(q_stage2_net)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
