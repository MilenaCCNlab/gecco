def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-Step Model with Forgetting Rate for Unchosen Options
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence.
    lam: [0, 1] - Eligibility trace parameter (updates stage 1 action with stage 2 reward).
    pers: [-5, 5] - Perseveration bonus added to the previously chosen stage 1 action.
    forgetting_rate: [0, 1] - Rate at which unchosen Q-values decay towards their initial value of 0.5.
    """
    learning_rate, beta1, beta2, w, lam, pers, forgetting_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 (neutral expectation)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Calculate prediction errors
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        # Unchosen option forgetting (decay towards 0.5)
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += forgetting_rate * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += forgetting_rate * (0.5 - q_stage2_mf[s, a])

        # Action value updating for chosen options
        q_stage1_mf[a1] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-Step Model with Choice Trace Perseveration
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration weight applied to the choice trace.
    trace_decay: [0, 1] - Decay rate of the choice trace (1 = no decay, 0 = only last action).
    """
    learning_rate, beta1, beta2, w, lam, pers, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Initialize an exponentially decaying choice trace
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseveration based on the accumulated choice trace
        q_stage1_net += pers * choice_trace
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Update choice trace
        choice_trace *= trace_decay
        choice_trace[a1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-Step Model with Learned Transition Matrix
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration bonus added to the previously chosen stage 1 action.
    transition_lr: [0, 1] - Learning rate for updating the internal transition matrix.
    """
    learning_rate, beta1, beta2, w, lam, pers, transition_lr = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix uniformly, to be learned over time
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Update the transition matrix based on observed state transition
        transition_matrix[a1, state_idx] += transition_lr * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += transition_lr * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
