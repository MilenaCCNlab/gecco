def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with asymmetric learning rates for positive and negative prediction errors,
    unchosen option decay, and simple 1-trial perseveration.
    
    Parameters:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    unchosen_decay: [0,1] - Decay rate for unchosen options.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    pers: [-5,5] - Perseveration bonus for repeating the previous stage 1 choice.
    """
    alpha_pos, alpha_neg, unchosen_decay, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_stage1_net
        
        if prev_a1 != -1:
            logits_1[prev_a1] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += lr1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += lr2 * delta_stage2
        q_stage1_mf[a1] += lr2 * lam * delta_stage2

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - unchosen_decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - unchosen_decay)
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate inverse temperatures for model-based and model-free values,
    unchosen option decay, and outcome-dependent perseveration.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    unchosen_decay: [0,1] - Decay rate for unchosen options towards 0.
    beta_mb: [0,10] - Inverse temperature for model-based stage 1 values.
    beta_mf1: [0,10] - Inverse temperature for model-free stage 1 values.
    beta_mf2: [0,10] - Inverse temperature for stage 2 values.
    lam: [0,1] - Eligibility trace parameter.
    pers_win: [-5,5] - Perseveration bonus if previous trial was rewarded.
    pers_lose: [-5,5] - Perseveration bonus if previous trial was unrewarded.
    """
    learning_rate, unchosen_decay, beta_mb, beta_mf1, beta_mf2, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        logits_1 = beta_mb * q_stage1_mb + beta_mf1 * q_stage1_mf
        
        if prev_a1 != -1 and prev_r != -1:
            if prev_r > 0:
                logits_1[prev_a1] += pers_win
            else:
                logits_1[prev_a1] += pers_lose
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_mf2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - unchosen_decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - unchosen_decay)
        
        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with unchosen option decay and a choice trace mechanism.
    Instead of 1-trial perseveration, it uses a choice trace that exponentially
    tracks past choices.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    unchosen_decay: [0,1] - Decay rate for unchosen options.
    alpha_c: [0,1] - Choice trace update rate.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    pers_c: [-10,10] - Weight on the choice trace for stage 1 decisions.
    """
    learning_rate, unchosen_decay, alpha_c, beta_1, beta_2, w, lam, pers_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_stage1_net + pers_c * choice_trace
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - unchosen_decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - unchosen_decay)
        
        choice_trace = (1 - alpha_c) * choice_trace
        choice_trace[a1] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
