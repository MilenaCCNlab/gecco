Here are 3 new cognitive models that build upon the most successful components identified so far (like separate stage 1 and stage 2 perseveration, and eligibility traces) while introducing new, psychologically grounded mechanisms. 

### Model 1: Optimistic/Pessimistic Initialization (`q_init`)
Instead of assuming a fixed neutral initial Q-value (e.g., 0.5), this model introduces a `q_init` parameter. This allows the model to capture individual differences in optimistic or pessimistic initialization, which strongly dictates early exploratory behavior before sufficient reward statistics are gathered.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate choice perseveration for stage 1 and stage 2,
    eligibility trace, and a parameter for initial Q-values to capture 
    optimistic/pessimistic initialization.
    
    Bounds:
    learning_rate: [0,1] - learning rate for value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi1: [-5,5] - choice perseveration (stickiness) for stage 1
    pi2: [-5,5] - choice perseveration (stickiness) for stage 2
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    q_init: [0,1] - initial Q-value for all state-action pairs
    """
    learning_rate, beta1, beta2, w, pi1, pi2, lam, q_init = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, q_init)
    q_stage2_mf = np.full((2, 2), q_init)
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta1 * q_net + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # ACTION VALUE UPDATING
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
            last_action2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition Matrix Learning (`transition_lr`)
Instead of assuming the participant inherently trusts and permanently fixes the transition probabilities to 70/30, this model dynamically updates the internal transition matrix as the participant experiences rare transitions.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with transition matrix learning. Participants update their 
    internal transition probabilities based on experienced transitions, in addition 
    to standard Q-learning with eligibility traces and perseveration.
    
    Bounds:
    learning_rate: [0,1] - learning rate for value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi1: [-5,5] - choice perseveration (stickiness) for stage 1
    pi2: [-5,5] - choice perseveration (stickiness) for stage 2
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    transition_lr: [0,1] - learning rate for updating the transition matrix
    """
    learning_rate, beta1, beta2, w, pi1, pi2, lam, transition_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta1 * q_net + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Update dynamic transition matrix based on observed transition
        transition_matrix[action_1[trial], state_idx] += transition_lr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += transition_lr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        # policy for the second choice
        if action_2[trial] != -1:
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # ACTION VALUE UPDATING
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
            last_action2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Omission Penalty (`omega`)
This model captures asymmetric sensitivity to rewards (wins) and lack of rewards (omissions). When a reward of 0 is observed, it is treated as a negative utility penalty `-omega`, representing implicit loss aversion when expecting a reward.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with a penalty parameter for unrewarded trials (omega).
    When reward is 0, it is treated as a negative reward of -omega, allowing 
    asymmetric sensitivity to wins versus omissions.
    
    Bounds:
    learning_rate: [0,1] - learning rate for value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - model-based weight (1 = fully model-based, 0 = fully model-free)
    pi1: [-5,5] - choice perseveration (stickiness) for stage 1
    pi2: [-5,5] - choice perseveration (stickiness) for stage 2
    lam: [0,1] - eligibility trace (lambda) for updating stage 1 from stage 2 rewards
    omega: [0,5] - penalty applied when reward is 0 (effective reward = -omega)
    """
    learning_rate, beta1, beta2, w, pi1, pi2, lam, omega = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    # Initiated at 0.0 so that -omega omissions drive values into negatives
    q_stage1_mf = np.full(2, 0.0) 
    q_stage2_mf = np.full((2, 2), 0.0)
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta1 * q_net + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # Map a 0 reward to negative utility penalty omega
            eff_reward = 1.0 if reward[trial] == 1 else -omega

            # ACTION VALUE UPDATING
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

            delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
            last_action2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
