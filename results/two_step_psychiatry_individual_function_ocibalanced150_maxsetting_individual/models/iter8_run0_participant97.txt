Here are three new cognitive models that explore alternative mechanisms for the participant's behavior, avoiding the exact parameter combinations previously tried. 

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model tests the hypothesis that participants update their values at different rates for the first and second stages of the task. It uses separate learning rates (`alpha1` and `alpha2`) for the Q-values and Choice Kernels of Stage 1 and Stage 2, respectively.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Separate Learning Rates for Stage 1 and Stage 2
    
    This model uses different learning rates for the first and second stages.
    It includes choice kernels for both stages, where the choice kernel
    learning rate for each stage is tied to the Q-value learning rate of that stage.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1 Q-values and stage 1 choice kernel.
    alpha2: [0, 1] - Learning rate for stage 2 Q-values and stage 2 choice kernel.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace.
    ck_weight1: [-5, 5] - Choice kernel weight for stage 1.
    ck_weight2: [-5, 5] - Choice kernel weight for stage 2.
    """
    alpha1, alpha2, beta1, beta2, w, lambda_, ck_weight1, ck_weight2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            ck_1 = (1 - alpha1) * ck_1
            ck_2 = (1 - alpha2) * ck_2
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf) + ck_weight1 * ck_1
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            ck_1 = (1 - alpha1) * ck_1
            ck_1[action_1[trial]] += alpha1
            ck_2 = (1 - alpha2) * ck_2
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx] + ck_weight2 * ck_2[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2

        ck_1 = (1 - alpha1) * ck_1
        ck_1[action_1[trial]] += alpha1
        
        ck_2[state_idx] = (1 - alpha2) * ck_2[state_idx]
        ck_2[state_idx, action_2[trial]] += alpha2
        
        other_state = 1 - state_idx
        ck_2[other_state] = (1 - alpha2) * ck_2[other_state]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Dynamic Transition Matrix Learning
This model assumes that the participant does not rely on a strictly fixed representation of the transition probabilities (0.7/0.3). Instead, they continuously update their mental model of the transitions based on the states they actually experience, using a transition learning rate (`alpha_trans`).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Dynamic Transition Matrix Learning
    
    This model assumes participants dynamically update the transition probabilities 
    based on the experienced transitions over time. It initializes at 0.7/0.3 
    and updates using alpha_trans. It also includes choice kernels.
    
    Parameters:
    learning_rate: [0, 1] - Shared learning rate for Q-values and choice kernels.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace.
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix.
    ck_weight1: [-5, 5] - Choice kernel weight for stage 1.
    ck_weight2: [-5, 5] - Choice kernel weight for stage 2.
    """
    learning_rate, beta1, beta2, w, lambda_, alpha_trans, ck_weight1, ck_weight2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            ck_1 = (1 - learning_rate) * ck_1
            ck_2 = (1 - learning_rate) * ck_2
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf) + ck_weight1 * ck_1
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            ck_1 = (1 - learning_rate) * ck_1
            ck_1[action_1[trial]] += learning_rate
            ck_2 = (1 - learning_rate) * ck_2
            
            if state_idx != -1:
                transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
                transition_matrix[action_1[trial], 1 - state_idx] = 1 - transition_matrix[action_1[trial], state_idx]
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx] + ck_weight2 * ck_2[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1 - transition_matrix[action_1[trial], state_idx]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

        ck_1 = (1 - learning_rate) * ck_1
        ck_1[action_1[trial]] += learning_rate
        
        ck_2[state_idx] = (1 - learning_rate) * ck_2[state_idx]
        ck_2[state_idx, action_2[trial]] += learning_rate
        
        other_state = 1 - state_idx
        ck_2[other_state] = (1 - learning_rate) * ck_2[other_state]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Loss Penalty Representation
This model tests the hypothesis that participants interpret an unrewarded trial (0 coins) not simply as an absence of reward, but as a subjective negative event or loss. It converts 0-reward events into a `-loss_penalty` to drive stronger value updates when expectations are violated.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Loss Penalty Representation
    
    This model proposes that participants frame unrewarded trials not as 0, 
    but as a negative outcome (loss). It introduces a loss_penalty parameter 
    to represent the subjective negative value of receiving no reward.
    
    Parameters:
    learning_rate: [0, 1] - Shared learning rate for Q-values and choice kernels.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace.
    loss_penalty: [0, 5] - Subjective penalty for receiving 0 reward.
    ck_weight1: [-5, 5] - Choice kernel weight for stage 1.
    ck_weight2: [-5, 5] - Choice kernel weight for stage 2.
    """
    learning_rate, beta1, beta2, w, lambda_, loss_penalty, ck_weight1, ck_weight2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            ck_1 = (1 - learning_rate) * ck_1
            ck_2 = (1 - learning_rate) * ck_2
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            ck_1 = (1 - learning_rate) * ck_1
            ck_1[action_1[trial]] += learning_rate
            ck_2 = (1 - learning_rate) * ck_2
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx] + ck_weight2 * ck_2[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Evaluate subjective reward: map 0 to -loss_penalty
        r = reward[trial] if reward[trial] == 1 else -loss_penalty

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

        ck_1 = (1 - learning_rate) * ck_1
        ck_1[action_1[trial]] += learning_rate
        
        ck_2[state_idx] = (1 - learning_rate) * ck_2[state_idx]
        ck_2[state_idx, action_2[trial]] += learning_rate

        ck_2[other_state] = (1 - learning_rate) * ck_2[other_state]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
