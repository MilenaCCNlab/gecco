Here are 3 new cognitive models that build upon the best-performing model (which included value decay) by integrating novel psychological mechanisms: Stage 2 perseveration, asymmetric learning rates for positive/negative prediction errors, and reward-dependent perseveration (Win-Stay/Lose-Shift). 

### Model 1: Value Decay + Stage 1 & Stage 2 Perseveration
This model extends the base decay model by allowing for perseveration not just at the first stage (spaceship choice), but also at the second stage (alien choice), capturing tendencies to repeat actions in the same state regardless of reward.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid model with value decay, and perseveration at both stage 1 and stage 2.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Decay rate for unchosen actions
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weight for model-based vs model-free control
    lam: [0, 1] - Eligibility trace
    p_1: [-5, 5] - Stage 1 perseveration
    p_2: [-5, 5] - Stage 2 perseveration
    """
    alpha, decay, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial] if action_1[trial] != -1 else 1
        a2 = action_2[trial] if action_2[trial] != -1 else 1
        
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            rep = np.zeros(2)
            rep[last_action_1] = 1
            q_stage1_net += p_1 * rep
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        q2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            rep2 = np.zeros(2)
            rep2[last_action_2[state_idx]] = 1
            q2_net += p_2 * rep2
            
        q2_scaled = beta_2 * q2_net
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == state_idx and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] *= (1 - decay)
                    
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Value Decay + Asymmetric Learning Rates
This model introduces asymmetric learning rates for positive and negative prediction errors. This captures the psychological phenomenon where participants might learn differently from better-than-expected outcomes versus worse-than-expected outcomes.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with value decay, and asymmetric learning rates for positive and negative RPEs.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    decay: [0, 1] - Decay rate for unchosen actions
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weight for model-based vs model-free control
    lam: [0, 1] - Eligibility trace
    p: [-5, 5] - Stage 1 perseveration
    """
    alpha_pos, alpha_neg, decay, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial] if action_1[trial] != -1 else 1
        a2 = action_2[trial] if action_2[trial] != -1 else 1
        
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            rep = np.zeros(2)
            rep[last_action_1] = 1
            q_stage1_net += p * rep
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg

        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == state_idx and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] *= (1 - decay)
                    
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2

        q_stage1_mf[a1] += alpha_s2 * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Value Decay + Win-Stay / Lose-Shift Perseveration
This model replaces simple action perseveration with reward-dependent perseveration. It models how likely a participant is to repeat their first-stage choice based on whether the immediately preceding trial was rewarded or not.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with value decay, and reward-dependent perseveration (Win-stay/Lose-shift).
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Decay rate for unchosen actions
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weight for model-based vs model-free control
    lam: [0, 1] - Eligibility trace
    p_win: [-5, 5] - Perseveration if previous trial was rewarded
    p_lose: [-5, 5] - Perseveration if previous trial was unrewarded
    """
    alpha, decay, beta_1, beta_2, w, lam, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = action_1[trial] if action_1[trial] != -1 else 1
        a2 = action_2[trial] if action_2[trial] != -1 else 1
        
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            rep = np.zeros(2)
            rep[last_action_1] = 1
            if last_reward == 1:
                q_stage1_net += p_win * rep
            elif last_reward == 0:
                q_stage1_net += p_lose * rep
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]

        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == state_idx and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] *= (1 - decay)
                    
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
