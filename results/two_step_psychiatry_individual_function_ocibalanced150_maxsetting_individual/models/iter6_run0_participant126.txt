def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-step model with separate learning rates for Stage 1 and Stage 2,
    incorporating stage-specific stickiness and decay for unchosen options.
    Different learning rates can capture distinct levels of uncertainty or volatility 
    participants attribute to the spaceship choices vs. the alien choices.
    
    Bounds:
    alpha_1: [0, 1]
    alpha_2: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stickiness_1: [-5, 5]
    stickiness_2: [-5, 5]
    decay: [0, 1]
    """
    alpha_1, alpha_2, beta, w, lam, stickiness_1, stickiness_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_stage2 = {0: None, 1: None}

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if trial > 0:
            stickiness_bonus_1[action_1[trial-1]] = stickiness_1
            
        logits_1 = beta * q_net + stickiness_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stickiness_bonus_2 = np.zeros(2)
        if last_action_stage2[state_idx] is not None:
            stickiness_bonus_2[last_action_stage2[state_idx]] = stickiness_2

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_stage2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Includes 'model-based perseveration' (pers_mb), a tendency to choose the Stage 1 
    action that commonly leads to the Stage 2 state visited on the previous trial. This captures 
    a heuristic where participants try to return to the planet they just visited regardless of reward.
    
    Bounds:
    alpha: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stickiness_1: [-5, 5]
    stickiness_2: [-5, 5]
    decay: [0, 1]
    pers_mb: [-5, 5]
    """
    alpha, beta, w, lam, stickiness_1, stickiness_2, decay, pers_mb = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_stage2 = {0: None, 1: None}

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        pers_mb_bonus = np.zeros(2)
        
        if trial > 0:
            stickiness_bonus_1[action_1[trial-1]] = stickiness_1
            # Action 0 commonly leads to State 0, Action 1 commonly leads to State 1
            prev_state = state[trial-1]
            pers_mb_bonus[prev_state] = pers_mb
            
        logits_1 = beta * q_net + stickiness_bonus_1 + pers_mb_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stickiness_bonus_2 = np.zeros(2)
        if last_action_stage2[state_idx] is not None:
            stickiness_bonus_2[last_action_stage2[state_idx]] = stickiness_2

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_stage2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Incorporates transition probability learning. Instead of fixing the transition
    matrix at 0.7/0.3, participants update their internal transition model based on experienced
    transitions using a transition learning rate (alpha_tr). This helps capture behavior 
    after experiencing multiple rare transitions.
    
    Bounds:
    alpha: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    stickiness_1: [-5, 5]
    stickiness_2: [-5, 5]
    decay: [0, 1]
    alpha_tr: [0, 1]
    """
    alpha, beta, w, lam, stickiness_1, stickiness_2, decay, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize with the standard task instructions
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_stage2 = {0: None, 1: None}

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if trial > 0:
            stickiness_bonus_1[action_1[trial-1]] = stickiness_1
            
        logits_1 = beta * q_net + stickiness_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stickiness_bonus_2 = np.zeros(2)
        if last_action_stage2[state_idx] is not None:
            stickiness_bonus_2[last_action_stage2[state_idx]] = stickiness_2

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value learning
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Transition matrix learning
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_stage2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
