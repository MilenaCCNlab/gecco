def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual-system model with outcome-dependent stickiness (Win-Stay/Lose-Shift).
    Stage 1 choice stickiness depends on whether the previous trial was rewarded, 
    allowing the model to capture asymmetric perseveration after wins vs. losses.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st_win: [-5, 5] (choice stickiness after a rewarded trial)
    st_loss: [-5, 5] (choice stickiness after an unrewarded trial)
    """
    alpha, beta_1, beta_2, w, lam, st_win, st_loss = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0
    
    for trial in range(n_trials):
        # Stage 1 model-based values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Outcome-dependent stickiness
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            st = st_win if last_reward == 1 else st_loss
            stickiness[last_action_1] = st
            
        # Policy for first choice
        logits_1 = beta_1 * q_net + stickiness
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = r
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dual-system model with an exponentially decaying choice trace.
    Instead of just tracking the immediately preceding action, choice perseveration 
    is driven by a decaying history of past actions.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st: [-5, 5] (choice stickiness weight)
    tau_c: [0, 1] (decay rate of the choice trace, 0 = no memory, 1 = full memory)
    """
    alpha, beta_1, beta_2, w, lam, st, tau_c = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)
    
    for trial in range(n_trials):
        # Stage 1 model-based values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Policy for first choice
        logits_1 = beta_1 * q_net + st * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update choice trace
        choice_trace *= tau_c
        choice_trace[a1] += 1.0
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dual-system model with transition matrix learning.
    The agent updates its internal model of the transition probabilities based on observed 
    transitions, accommodating dynamic beliefs about the task structure.
    
    Bounds:
    alpha: [0, 1] (learning rate for Q-values)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    alpha_tr: [0, 1] (learning rate for the transition matrix)
    """
    alpha, beta_1, beta_2, w, lam, st, alpha_tr = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix based on typical instructions
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        # Stage 1 model-based values using dynamic transition matrix
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        # Policy for first choice
        logits_1 = beta_1 * q_net + st * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Update transition matrix probabilities based on observed transition
        transition_matrix[a1, state_idx] += alpha_tr * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
