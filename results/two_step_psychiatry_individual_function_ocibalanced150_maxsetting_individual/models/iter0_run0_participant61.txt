### Model 1: Full Daw et al. (2011) Model
This is the standard model used to analyze the two-step task. It assumes participants use a mixture of model-free (habitual) and model-based (goal-directed) learning. It also includes an eligibility trace to update stage 1 values based on stage 2 rewards, and a perseveration (stickiness) parameter to account for the tendency to repeat the previous first-stage choice regardless of reward.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Full Daw et al. (2011) two-step model with Model-Free and Model-Based control.
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p: [-5,5] (choice perseveration / stickiness)
    """
    alpha, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Handle missed trials
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine Model-Free and Model-Based values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Choice stickiness/perseveration
        perseveration = np.zeros(2)
        if last_action != -1:
            perseveration[last_action] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p * perseveration)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for Choice 1 based on Choice 2 reward
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates (Positive vs Negative RPE)
This model extends the classical model by assuming participants learn differently from better-than-expected versus worse-than-expected outcomes. It fits separate learning rates for positive and negative reward prediction errors (RPEs), which is a common pattern in reinforcement learning tasks.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-step model with asymmetric learning rates for positive and negative reward prediction errors.
    
    Bounds:
    alpha_pos: [0,1] (learning rate for positive RPE)
    alpha_neg: [0,1] (learning rate for negative RPE)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p: [-5,5] (choice perseveration / stickiness)
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Handle missed trials
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration = np.zeros(2)
        if last_action != -1:
            perseveration[last_action] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p * perseveration)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
        
        # Eligibility trace update for Choice 1
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Distinct Stage Learning Rates
This model posits that learning might unfold at different paces across the two stages of the task. For example, participants might adapt their values for the terminal aliens (stage 2) much faster than the spaceships (stage 1) due to direct reward vs. sequential state transitions.
