Here are 3 new cognitive models based on variations of the best-performing model. They introduce distinct, psychologically plausible mechanisms: state-specific perseveration at stage 2, forgetting (decay) of unchosen options, and outcome-dependent perseveration (win-stay/lose-shift).

### Model 1: Stage-Specific Perseveration
This model builds on the best model by adding a separate perseveration parameter for stage 2 choices. Crucially, the stage 2 perseveration is state-specific (i.e., participants might have a tendency to repeat their choice of alien *on that specific planet*).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. 2011 Two-Step Task Model with Stage-Specific Learning Rates and Stage-Specific Perseveration.
    Includes separate choice perseveration bonuses for stage 1 and stage 2. Stage 2 perseveration is state-specific.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_1: choice perseveration bonus for stage 1 [-2,2]
    pers_2: choice perseveration bonus for stage 2 (state-specific) [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])  # Tracks last action for each of the 2 states

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = 1.0
            
        q_net_total = beta_1 * q_net + pers_1 * pers_bonus_1
        exp_q1 = np.exp(q_net_total - np.max(q_net_total)) # Safe softmax
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = 1.0
                
            q_stage2_total = beta_2 * q_stage2_mf[state_idx] + pers_2 * pers_bonus_2
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            last_action_2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Forgetting (Decay)
This model adds a `decay` parameter. Unchosen Q-values passively decay towards 0.5 (neutral expectation) on each trial, capturing the idea that participants may forget the value of options they haven't sampled recently, thus naturally encouraging exploration over time.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. 2011 Two-Step Task Model with Stage-Specific Learning Rates, Perseveration, and Forgetting (Decay).
    Unchosen options decay towards 0.5.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers: choice perseveration bonus for stage 1 [-2,2]
    decay: decay rate for all unchosen Q-values towards 0.5 [0,1]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = 1.0
            
        q_net_total = beta_1 * q_net + pers * pers_bonus
        exp_q1 = np.exp(q_net_total - np.max(q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            # Decay for unchosen options towards 0.5
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            for s in range(2):
                for a in range(2):
                    if s != state_idx or a != action_2[trial]:
                        q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        else:
            p_choice_2[trial] = 1.0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Perseveration (Win-Stay/Lose-Shift)
This model splits the standard perseveration parameter into two context-dependent components: `pers_win` and `pers_lose`. Participants often exhibit higher stickiness (stay behavior) immediately following a reward, and lower stickiness (shift behavior) after an unrewarded trial. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. 2011 Two-Step Task Model with Outcome-Dependent Perseveration.
    Includes separate perseveration bonuses for stage 1 depending on whether the 
    previous trial was rewarded or unrewarded (capturing Win-Stay/Lose-Shift dynamics).
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-2,2]
    pers_lose: perseveration bonus if previous trial was unrewarded [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        pers = 0.0
        if last_action != -1:
            pers_bonus[last_action] = 1.0
            if last_reward > 0:
                pers = pers_win
            else:
                pers = pers_lose
                
        q_net_total = beta_1 * q_net + pers * pers_bonus
        exp_q1 = np.exp(q_net_total - np.max(q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
