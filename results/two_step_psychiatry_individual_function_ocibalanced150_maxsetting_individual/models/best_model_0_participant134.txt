def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with decay for unchosen options and transition-dependent choice stickiness.
    Stickiness for Stage 1 depends on whether the previous trial's transition was common or rare.
    
    Parameters:
    alpha: [0, 1] - Learning rate
    decay_rate: [0, 1] - Decay rate for unchosen Q-values towards 0.5
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace
    stick_common: [-5, 5] - Stage 1 stickiness following a common transition
    stick_rare: [-5, 5] - Stage 1 stickiness following a rare transition
    """
    alpha, decay_rate, beta_1, beta_2, w, lam, stick_common, stick_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    last_trans_common = False
    
    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net
        if last_action != -1:
            if last_trans_common:
                logits_1[last_action] += stick_common
            else:
                logits_1[last_action] += stick_rare
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
                    
        last_action = action_1[trial]

        last_trans_common = (action_1[trial] == state_idx) 

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
