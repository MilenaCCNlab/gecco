def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual Stickiness Model
    Incorporates stickiness for both Stage 1 and Stage 2 choices. 
    Stage 2 stickiness is state-specific, meaning participants tend to repeat 
    the action they last took on that specific planet, independent of reward.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight (1 = fully model-based, 0 = fully model-free)
    lam: [0,1] - Eligibility trace parameter
    st1: [-5,5] - Stage 1 choice stickiness
    st2: [-5,5] - Stage 2 choice stickiness (state-specific)
    """
    learning_rate, beta, w, lam, st1, st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness1 = np.zeros(2)
        if last_action1 != -1:
            stickiness1[last_action1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st1 * stickiness1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            stickiness2[last_action2[state_idx]] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + st2 * stickiness2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # update last actions
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition Learning Model
    Participant dynamically updates their internal transition matrix based on observed state transitions,
    rather than assuming fixed 0.7/0.3 probabilities throughout the entire task.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight (1 = fully model-based, 0 = fully model-free)
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Stage 1 choice stickiness
    lr_trans: [0,1] - Learning rate for updating the transition matrix
    """
    learning_rate, beta, w, lam, st, lr_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize with the instructed probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Update transition matrix based on observed transition
        transition_matrix[action_1[trial], state_idx] += lr_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1.0 - transition_matrix[action_1[trial], state_idx]
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Counterfactual Stage 2 Learning Model
    Assumes participants update the unchosen Stage 2 action on the visited planet, 
    simulating a belief that the two aliens have anti-correlated reward probabilities 
    (i.e., if one pays out, the other likely wouldn't have).
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight (1 = fully model-based, 0 = fully model-free)
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Stage 1 choice stickiness
    anti_lr: [0,1] - Learning rate for counterfactual update of unchosen Stage 2 action
    """
    learning_rate, beta, w, lam, st, anti_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Counterfactual update for the unchosen action in the same state
        unchosen_action_2 = 1 - action_2[trial]
        counterfactual_reward = 1.0 - reward[trial]
        delta_unchosen = counterfactual_reward - q_stage2_mf[state_idx, unchosen_action_2]
        q_stage2_mf[state_idx, unchosen_action_2] += anti_lr * delta_unchosen
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
