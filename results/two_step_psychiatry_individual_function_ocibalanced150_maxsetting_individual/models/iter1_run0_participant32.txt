def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model-free and model-based RL with choice perseveration and an eligibility trace.
    This model adds an eligibility trace parameter (lam) which allows the stage 2 reward 
    prediction error to directly update the stage 1 action value, in addition to the 
    standard TD learning. It retains the perseveration parameter to capture choice stickiness.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta: [0, 10] (inverse temperature)
    w: [0, 1] (weighting of model-based vs model-free values)
    pers: [0, 5] (perseveration bonus added to the previously chosen stage 1 action)
    lam: [0, 1] (eligibility trace parameter for updating stage 1 value from stage 2 reward)
    """
    alpha, beta, w, pers, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = pers
            
        exp_q1 = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        last_action = action_1[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):

    Hybrid model-free and model-based RL with separate learning rates and inverse

    This model allows for the possibility that learning and choice stochasticity

    alpha1: [0, 1] (learning rate for stage 1)
    alpha2: [0, 1] (learning rate for stage 2)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)

    pers: [0, 5] (perseveration bonus added to the previously chosen stage 1 action)

    for trial in range(n_trials):

        q_stage1_mb = transition_matrix @ max_q_stage2

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model-free and model-based RL with perseveration bonuses at both stage 1 
    and stage 2. This model accounts for choice stickiness not only for the spaceships 
    but also for the specific aliens on each planet.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta: [0, 10] (inverse temperature)
    w: [0, 1] (weighting of model-based vs model-free values)
    pers1: [0, 5] (perseveration bonus added to the previously chosen stage 1 action)
    pers2: [0, 5] (perseveration bonus added to the previously chosen stage 2 action in the current state)
    """
    alpha, beta, w, pers1, pers2 = model_parameters
    n_trials = len(action_1)

    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
