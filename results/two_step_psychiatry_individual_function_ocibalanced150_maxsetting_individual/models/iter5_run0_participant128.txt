Here are 3 new cognitive models that build on the best model so far, proposing new mechanisms while keeping the single-beta constraint and parsimony in mind.

### Model 1: Asymmetric Learning Rates (Positive vs. Negative Prediction Errors)
This model extends the best model by differentiating how participants learn from better-than-expected versus worse-than-expected outcomes. People often exhibit asymmetric learning, updating their beliefs more strongly for positive prediction errors compared to negative ones (or vice versa).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative prediction errors,
    a single inverse temperature (beta) for both stages, and separate choice 
    stickiness (p1, p2) for each stage.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta: [0, 10] - single inverse temperature for both stages
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    p1: [-5, 5] - choice perseverance (stickiness) for stage 1
    p2: [-5, 5] - choice perseverance (stickiness) for stage 2
    """
    alpha_pos, alpha_neg, beta, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2 != -1:
            logits_2[last_action_2] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Choice Trace (Exponentially Decaying Perseveration)
Instead of perseveration depending exclusively on the very last trial's action, this model calculates a slowly decaying "Choice Trace". This accounts for longer-term habitual responding, which is quite visible in the participant data where they stick to a single spaceship or alien for long streaks.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a choice trace that decays over time, capturing longer-term 
    perseveration rather than just the last trial. Uses a single beta.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta: [0, 10] - single inverse temperature for both stages
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    alpha_c: [0, 1] - choice trace update rate (decay)
    p1: [-5, 5] - choice trace weight for stage 1
    p2: [-5, 5] - choice trace weight for stage 2
    """
    alpha, beta, w, lam, alpha_c, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros(2)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_stage1_net + p1 * choice_trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx] + p2 * choice_trace_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update choice traces
        choice_trace_1 *= (1 - alpha_c)
        choice_trace_1[action_1[trial]] += alpha_c
        
        choice_trace_2 *= (1 - alpha_c)
        choice_trace_2[action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage-Specific Learning Rates
This model introduces separate learning rates for Stage 1 and Stage 2 but maintains a single inverse temperature (beta) for both stages. This captures instances where participants update their values rapidly for the aliens (Stage 2) but remain highly resistant to updating values for the spaceships (Stage 1).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for stage 1 and stage 2, a single 
    inverse temperature (beta) for both stages, and separate choice 
    stickiness (p1, p2) for each stage.
    
    Bounds:
    alpha1: [0, 1] - learning rate for stage 1
    alpha2: [0, 1] - learning rate for stage 2
    beta: [0, 10] - single inverse temperature for both stages
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    p1: [-5, 5] - choice perseverance (stickiness) for stage 1
    p2: [-5, 5] - choice perseverance (stickiness) for stage 2
    """
    alpha1, alpha2, beta, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2 != -1:
            logits_2[last_action_2] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
