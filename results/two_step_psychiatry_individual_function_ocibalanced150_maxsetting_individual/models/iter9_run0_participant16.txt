Here are 3 new cognitive models that build upon the success of the `penalty_zero` parameter by combining it with other mechanisms that have not yet been tested alongside it in this specific configuration. All models adhere to the 8-parameter limit and are fully self-contained.

### Model 1: Outcome-Dependent Perseveration with Subjective Penalty
This model tests whether the tendency to repeat the previous stage 1 choice (perseveration) differs depending on whether the previous trial resulted in a reward or a zero outcome. It combines this with the subjective penalty for zero rewards.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Outcome-Dependent Perseveration with Subjective Penalty
    Uses different perseveration parameters for stage 1 based on whether the previous 
    trial was rewarded or not, combined with a subjective penalty for zero rewards.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers_win: [-5, 5] - Perseveration for stage 1 if previous trial was rewarded
    pers_lose: [-5, 5] - Perseveration for stage 1 if previous trial was unrewarded
    penalty_zero: [0, 2] - Penalty applied when 0 coins are received
    """
    alpha, beta_1, beta_2, w, lam, pers_win, pers_lose, penalty_zero = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 Values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        # Outcome-dependent perseveration
        pers = pers_win if last_reward == 1 else pers_lose
        if last_action_1 == -1:
            pers = 0.0
            
        # Policy for the first choice
        logits_1 = beta_1 * q_stage1_net + pers * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        subj_reward = 1.0 if reward[trial] == 1 else -penalty_zero
        delta_stage2 = subj_reward - q_stage2_mf[state_idx, action_2[trial]]

        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Subjective Penalty
This model investigates whether the participant learns differently from positive versus negative prediction errors, especially when a zero-reward outcome is subjectively perceived as a negative penalty rather than just neutral.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates with Subjective Penalty
    Uses separate learning rates for positive and negative prediction errors,
    along with a general perseveration parameter and penalty for zero reward.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers: [-5, 5] - Perseveration for stage 1
    penalty_zero: [0, 2] - Penalty applied when 0 coins are received
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers, penalty_zero = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 Values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        # Policy for the first choice
        logits_1 = beta_1 * q_stage1_net + pers * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # Action value updating for choice 2
        subj_reward = 1.0 if reward[trial] == 1 else -penalty_zero
        delta_stage2 = subj_reward - q_stage2_mf[state_idx, action_2[trial]]

        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Model-Based Weighting with Decay and Penalty
This model tests if the participant relies more heavily on the transition structure (model-based planning) after a win compared to after a loss (or vice versa), while utilizing Q-value decay for unchosen alternatives.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Outcome-Dependent Model-Based Weight with Decay and Subjective Penalty
    The weighting between model-based and model-free values depends on whether the 
    previous trial was rewarded. It also includes value decay and a penalty for zero reward.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Retention rate for unchosen Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w_win: [0, 1] - Model-based weight if previous trial was rewarded
    w_lose: [0, 1] - Model-based weight if previous trial was unrewarded
    lam: [0, 1] - Eligibility trace parameter
    penalty_zero: [0, 2] - Penalty applied when 0 coins are received
    """
    alpha, decay, beta_1, beta_2, w_win, w_lose, lam, penalty_zero = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_reward = 0
    trial_count = 0

    for trial in range(n_trials):
        # Stage 1 Model-Based Values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Outcome-dependent model-based weight
        w = w_win if last_reward == 1 else w_lose
        if trial_count == 0:
            w = 0.5  # default for first trial
            
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        # Policy for the first choice
        logits_1 = beta_1 * q_stage1_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        subj_reward = 1.0 if reward[trial] == 1 else -penalty_zero
        delta_stage2 = subj_reward - q_stage2_mf[state_idx, action_2[trial]]

        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen values
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= decay
        
        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] *= decay
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= decay
        q_stage2_mf[other_state, 1] *= decay
        
        last_reward = reward[trial]
        trial_count += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
