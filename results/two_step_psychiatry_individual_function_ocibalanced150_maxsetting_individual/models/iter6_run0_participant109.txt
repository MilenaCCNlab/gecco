def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate Learning Rates with Lapsing (Epsilon-Greedy).
    
    Includes a lapse rate parameter that blends the softmax probabilities 
    with a uniform random distribution. This accounts for random exploratory 
    or accidental choices, preventing the model from overly penalizing rare actions.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence at Stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration bonus added to the previously chosen Stage 1 action.
    lapse_rate: [0, 1] - Probability of making a random choice at either stage.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, lapse_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # Apply lapse rate (epsilon-greedy)
        probs_1 = (1 - lapse_rate) * probs_1 + lapse_rate * 0.5
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        # Apply lapse rate (epsilon-greedy)
        probs_2 = (1 - lapse_rate) * probs_2 + lapse_rate * 0.5
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2

        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-Dependent Perseveration (Win-Stay / Lose-Shift).
    
    Extends the perseveration mechanism by allowing different perseveration 
    strengths depending on whether the previous trial was rewarded or not.
    This captures the common win-stay, lose-shift heuristic.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence at Stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Perseveration on previous Stage 1 action if rewarded.
    pers_lose: [-5, 5] - Perseveration on previous Stage 1 action if not rewarded.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            if last_reward == 1:
                q_stage1_net[last_action] += pers_win
            else:
                q_stage1_net[last_action] += pers_lose
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2

        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        last_action = a1
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Eligibility Traces for Positive and Negative RPEs.
    
    This model allows the eligibility trace (lambda) to differ depending on whether 
    the Stage 2 reward prediction error is positive or negative. This captures 
    differences in credit assignment for better-than-expected vs worse-than-expected outcomes.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence at Stage 1.
    lam_pos: [0, 1] - Eligibility trace for positive Stage 2 RPE.
    lam_neg: [0, 1] - Eligibility trace for negative Stage 2 RPE.
    pers: [-5, 5] - Perseveration on previous Stage 1 action.
    """
    alpha1, alpha2, beta1, beta2, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        # Determine which lambda to use based on the sign of delta_stage2
        current_lam = lam_pos if delta_stage2 > 0 else lam_neg

        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[a1] += alpha1 * current_lam * delta_stage2

        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
