Here are three new cognitive models to explain the participant's behavior in the two-step task. These models introduce different cognitive mechanisms (transition learning, habit traces, and forgetting) while staying within the parameter limit.

### Model 1: Learned Transition Matrix
Instead of assuming the participant fully knows and fixes the transition probabilities at 0.7/0.3, this model assumes the participant learns the transition matrix over time based on the states they actually end up in. This is controlled by a new parameter `alpha_trans`.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with a learned transition matrix for the model-based system.
    Bounds:
    alpha: [0,1] (learning rate for Q-values)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd: [0,1] (eligibility trace)
    pers: [0,10] (perseveration bonus)
    alpha_trans: [0,1] (learning rate for the transition matrix)
    """
    alpha, beta_1, beta_2, w, lambd, pers, alpha_trans = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix at 0.5 (maximum uncertainty)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Update transition matrix based on the state reached
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Decaying Choice Trace (Habit)
The participant data shows long streaks of choosing the same spaceship. Instead of a simple 1-trial perseveration, this model uses a choice trace that accumulates and decays over time, capturing stronger habitual behavior.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a decaying choice trace for Stage 1 (habitual perseveration).
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd: [0,1] (eligibility trace)
    pers: [0,10] (weight of the choice trace)
    tau: [0,1] (decay rate of the choice trace)
    """
    alpha, beta_1, beta_2, w, lambd, pers, tau = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + pers * choice_trace
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Update choice trace
        choice_trace *= tau
        choice_trace[action_1[trial]] += 1
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Unchosen Action Decay (Forgetting)
This model introduces a mechanism of forgetting. Q-values of actions that are not chosen at Stage 2 slowly decay back towards a neutral expected value of 0.5. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with forgetting (decay) for unchosen Stage 2 actions.
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd: [0,1] (eligibility trace)
    pers: [0,10] (perseveration bonus)
    decay: [0,1] (decay rate for unchosen actions)
    """
    alpha, beta_1, beta_2, w, lambd, pers, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        # Decay unchosen actions in Stage 2 towards 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
