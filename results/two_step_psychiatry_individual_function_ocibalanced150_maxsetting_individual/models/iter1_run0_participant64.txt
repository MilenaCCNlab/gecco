def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF hybrid with choice stickiness for stage 1 and separate inverse temperatures (beta) for stage 1 and stage 2.
    This allows the model to capture differences in exploration/exploitation behavior between the two stages.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness parameter [-5,5]
    """
    alpha, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]

        # policy for the second choice
        if act2 != -1:
            exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            # action value updating
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF hybrid with separate choice stickiness parameters for stage 1 and stage 2.
    This model captures the tendency to repeat choices (perseveration) not just at the first stage but also at the second stage, conditionally on the planet.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p_1: choice stickiness for stage 1 [-5,5]
    p_2: choice stickiness for stage 2 [-5,5]
    """
    alpha, beta, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # track last action taken in each stage 2 state

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + p_1 * rep_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]

        # policy for the second choice
        if act2 != -1:
            rep_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                rep_2[last_action_2[state_idx]] = 1.0
                
            logits_2 = beta * q_stage2_mf[state_idx] + p_2 * rep_2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            # action value updating
            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
            
            last_action_2[state_idx] = act2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
