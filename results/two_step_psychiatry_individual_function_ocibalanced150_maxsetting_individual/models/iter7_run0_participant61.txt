import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-step model with stage 1 and stage 2 perseveration, plus a 'state perseveration' parameter.
    State perseveration captures the tendency to choose the stage 1 action that commonly leads to the 
    planet visited on the previous trial, regardless of whether a reward was received.
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p_1: [-5,5] (stage 1 choice perseveration)
    p_2: [-5,5] (stage 2 state-specific choice perseveration)
    p_state: [-5,5] (state perseveration - bias towards action commonly leading to previous state)
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2, p_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            persev_1[last_action_1] = 1.0
            
        state_persev = np.zeros(2)
        if last_state != -1:
            # Action 0 commonly leads to State 0, Action 1 commonly leads to State 1
            state_persev[last_state] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p_1 * persev_1 + p_state * state_persev)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        persev_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            persev_2[last_action_2[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2 * persev_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-step model with a subjective value parameter for zero-reward outcomes.
    Instead of assuming a reward of 0 has a neutral value of 0, it is mapped to a subjective 
    value parameter `omega`, which allows modeling loss aversion (if omega < 0) or 
    baseline reinforcement (if omega > 0).
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p_1: [-5,5] (stage 1 choice perseveration)
    p_2: [-5,5] (stage 2 state-specific choice perseveration)
    omega: [-1,1] (subjective value of receiving 0 coins)
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2, omega = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            persev_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p_1 * persev_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        persev_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            persev_2[last_action_2[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2 * persev_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Substitute 0 rewards with subjective value omega
        subj_reward = reward[trial] if reward[trial] > 0 else omega

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = subj_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-step model incorporating a constant side bias for stage 1 choices.
    Participants often exhibit a baseline preference for one of the two spaceships, independent of 
    learning or perseveration. This model adds a `bias_1` parameter to the value of action 0.
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p_1: [-5,5] (stage 1 choice perseveration)
    p_2: [-5,5] (stage 2 state-specific choice perseveration)
    bias_1: [-5,5] (constant bias towards stage 1 action 0)
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            persev_1[last_action_1] = 1.0
            
        # Add side bias to action 0
        bias_arr = np.array([bias_1, 0.0])
            
        exp_q1 = np.exp(beta_1 * q_net + p_1 * persev_1 + bias_arr)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        persev_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            persev_2[last_action_2[state_idx]] = 1.0
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + p_2 * persev_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
