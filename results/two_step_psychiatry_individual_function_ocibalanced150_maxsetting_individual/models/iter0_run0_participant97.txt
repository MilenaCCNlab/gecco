def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Hybrid Model-Based / Model-Free Learning with Perseverance
    
    This model implements the standard Daw et al. (2011) hybrid reinforcement learning model. 
    It combines model-based (MB) and model-free (MF) Q-values for the first-stage choice. 
    
    Parameters:
    learning_rate (alpha): [0, 1] - Learning rate for updating Q-values.
    beta_param (beta): [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = pure MB, 0 = pure MF).
    lambda_: [0, 1] - Eligibility trace decay parameter for second-stage rewards updating first-stage values.
    p: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    """
    learning_rate, beta_param, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_act = -1

    for trial in range(n_trials):
        # Handle completely missed trials
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF values, apply perseverance, and stabilize for softmax
        V = beta_param * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act != -1:
            V[last_act] += p
        q_stage1_mb = V - np.max(V)
        beta = 1.0  # Set to 1.0 to integrate seamlessly with the template's exp_q1 line
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        V2 = beta_param * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        # Handle trials where only the second stage was missed (e.g., Trial 100)
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act = action_1[trial]
            continue
            
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_act = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Hybrid Model with Asymmetric Learning Rates
    
    This model extends the standard hybrid model by allowing different learning rates 
    for positive and negative prediction errors, which can capture optimism or pessimism biases.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (when delta > 0).
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (when delta <= 0).
    beta_param: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    """
    alpha_pos, alpha_neg, beta_param, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_act = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V = beta_param * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act != -1:
            V[last_act] += p
        q_stage1_mb = V - np.max(V)
        beta = 1.0
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        V2 = beta_param * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act = action_1[trial]
            continue
            
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2
        
        last_act = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Hybrid Model with Distinct Temperatures and Choice Kernel
    
    This model features separate inverse temperatures for stage 1 and stage 2, 
    and replaces the simple perseverance parameter with an exponentially updating choice kernel.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta1: [0, 10] - Inverse temperature for first-stage action selection.
    beta2: [0, 10] - Inverse temperature for second-stage action selection.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    ck_lr: [0, 1] - Learning rate for the choice kernel (how fast past choices are forgotten).
    ck_weight: [-5, 5] - Weight of the choice kernel on first-stage choices.
    """
    learning_rate, beta1, beta2, w, lambda_, ck_lr, ck_weight = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    choice_kernel = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            choice_kernel *= (1 - ck_lr)
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf) + ck_weight * choice_kernel
        q_stage1_mb = V - np.max(V)
        beta = 1.0
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        V2 = beta2 * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            choice_kernel *= (1 - ck_lr)
            choice_kernel[action_1[trial]] += ck_lr
            continue
            
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        # Update choice kernel
        choice_kernel *= (1 - ck_lr)
        choice_kernel[action_1[trial]] += ck_lr

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
