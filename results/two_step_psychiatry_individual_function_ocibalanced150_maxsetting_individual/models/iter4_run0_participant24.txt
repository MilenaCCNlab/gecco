Here are 3 new cognitive models that explore alternative parameter configurations and learning mechanisms for the two-step task.

### Model 1: Dual Inverse Temperature & Dual Stickiness
This model expands on the standard Daw et al. (2011) model by assuming that the participant's choice stochasticity and perseveration tendencies might differ structurally between the first and second stages of the task. It includes separate inverse temperatures (`beta1`, `beta2`) and separate choice stickiness parameters (`rep1`, `rep2`) for Stage 1 and Stage 2.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step task model with separate inverse temperatures and stickiness for both stages.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight given to model-based vs. model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rep1: [-5, 5] - Choice stickiness for stage 1.
    rep2: [-5, 5] - Choice stickiness for stage 2.
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, rep1, rep2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness1[last_action_1] = 1
            
        logits_1 = beta1 * q_stage1_net + rep1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness2[last_action_2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + rep2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates & Dual Inverse Temperature
This model incorporates learning rate asymmetry (differing updates for better-than-expected vs. worse-than-expected outcomes) by splitting `alpha` into `alpha_pos` and `alpha_neg` based on the sign of the prediction error. It also allows for varying choice stochasticity across the two decision stages via `beta1` and `beta2`.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step task model with asymmetric learning rates for positive and negative prediction errors,
    and separate inverse temperatures for stage 1 and stage 2.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight given to model-based vs. model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rep: [-5, 5] - Choice stickiness for stage 1.
    """
    import numpy as np
    alpha_pos, alpha_neg, beta1, beta2, w, lam, rep = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta1 * q_stage1_net + rep * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[a1] += alpha_2 * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Choice Stickiness
This model investigates whether the participants are more likely to perseverate on their Stage 1 choices depending on the nature of the transition they just experienced. It introduces `rep_c` and `rep_u` to distinguish the baseline stickiness applied after a common transition versus an uncommon (rare) transition. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step task model with transition-dependent choice stickiness.
    The stickiness on stage 1 depends on whether the transition on the previous trial
    was common or uncommon.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature.
    w: [0, 1] - Weight given to model-based vs. model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rep_c: [-5, 5] - Choice stickiness for stage 1 following a common transition.
    rep_u: [-5, 5] - Choice stickiness for stage 1 following an uncommon transition.
    """
    import numpy as np
    alpha, beta, w, lam, rep_c, rep_u = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_trans_common = False

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            rep = rep_c if last_trans_common else rep_u
            stickiness[last_action_1] = rep
            
        logits_1 = beta * q_stage1_net + stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        # Check if transition was common (a1 == 0 -> state 0 OR a1 == 1 -> state 1)
        last_trans_common = (a1 == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
