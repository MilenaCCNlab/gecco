def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition-dependent Perseveration
    
    This model separates the stage 1 perseveration into two parameters based on 
    whether the previous trial's transition was common or rare. This captures 
    the idea that participants might be more or less likely to repeat an action 
    depending on the expectedness of the transition.
    
    Parameters:
    alpha: learning rate for value updates [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p_common: perseveration if last transition was common [-5, 5]
    p_rare: perseveration if last transition was rare [-5, 5]
    p2: perseveration parameter for stage 2 choice [-5, 5]
    """
    alpha, beta1, beta2, w, lam, p_common, p_rare, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    last_transition_was_common = -1 

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            if last_transition_was_common == 1:
                logits_1[last_action_1] += p_common
            elif last_transition_was_common == 0:
                logits_1[last_action_1] += p_rare
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # update memory for next trial
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        
        if action_1[trial] == state_idx:
            last_transition_was_common = 1
        else:
            last_transition_was_common = 0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate Stage Learning Rates with Unchosen Decay
    
    This model uses different learning rates for stage 1 and stage 2, acknowledging 
    that participants might update their model-free values at different speeds for 
    the two stages. It also includes a decay parameter for unchosen actions as values
    might drift towards neutrality (0.5) over time.
    
    Parameters:
    alpha1: learning rate for stage 1 value updates [0, 1]
    alpha2: learning rate for stage 2 value updates [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p1: perseveration parameter for stage 1 choice [-5, 5]
    decay: decay rate for unchosen values towards 0.5 [0, 1]
    """
    alpha1, alpha2, beta1, beta2, w, lam, p1, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        # decay unchosen options towards 0.5
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-dependent Eligibility Trace
    
    This model allows the eligibility trace (lambda) to differ depending on whether 
    the transition from stage 1 to stage 2 was common or rare. Participants might 
    be more willing to credit the stage 1 action for a stage 2 reward if the 
    transition was expected (common) compared to unexpected (rare).
    
    Parameters:
    alpha: learning rate for value updates [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam_common: eligibility trace for common transitions [0, 1]
    lam_rare: eligibility trace for rare transitions [0, 1]
    p1: perseveration parameter for stage 1 choice [-5, 5]
    p2: perseveration parameter for stage 2 choice [-5, 5]
    """
    alpha, beta1, beta2, w, lam_common, lam_rare, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # determine transition type for eligibility trace
        is_common = (action_1[trial] == state_idx)
        lam = lam_common if is_common else lam_rare

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # update memory
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
