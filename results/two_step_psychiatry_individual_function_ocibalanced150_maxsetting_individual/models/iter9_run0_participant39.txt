Here are 3 new cognitive models that explore different combinations of learning mechanisms, eligibility traces, choice perseveration, and biases. They are designed to capture the participant's behavior, such as their strong preference for spaceship 0 and how they update values based on rewards.

### Model 1: Eligibility Trace with Stage-Specific Perseveration
This model includes an eligibility trace (`lam`) to allow stage 2 rewards to directly update stage 1 action values. It also separates choice perseveration into stage 1 (`pers1`) and stage 2 (`pers2`), as participants might have different tendencies to stick to their previous choices at each stage.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with eligibility trace and stage-specific perseveration.
    
    Parameters:
    learning_rate: learning rate for updating Q-values [0,1]
    beta1: inverse temperature for stage 1 softmax [0,10]
    beta2: inverse temperature for stage 2 softmax [0,10]
    w: weight for model-based vs model-free control [0,1]
    pers1: stage 1 choice perseveration bonus [-5,5]
    pers2: stage 2 choice perseveration bonus [-5,5]
    lam: eligibility trace for stage 1 updating from stage 2 reward [0,1]
    """
    learning_rate, beta1, beta2, w, pers1, pers2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action_1 != -1:
            q1_vals[last_action_1] += pers1
            
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        if last_state == state_idx and last_action_2 != -1:
            q2_vals[last_action_2] += pers2
            
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Stage-Specific Perseveration
Research shows humans often learn differently from positive and negative feedback. This model applies separate learning rates for positive vs. negative reward prediction errors (RPEs), combined with stage-specific perseveration.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates for positive vs negative RPEs.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    pers1: stage 1 choice perseveration [-5,5]
    pers2: stage 2 choice perseveration [-5,5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action_1 != -1:
            q1_vals[last_action_1] += pers1
            
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        if last_state == state_idx and last_action_2 != -1:
            q2_vals[last_action_2] += pers2
            
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage-Specific Learning Rates with Spatial Bias and Eligibility Trace
The participant heavily favored spaceship 0. This model incorporates a baseline spatial bias (`bias1`) for choosing spaceship 0, allowing the model to naturally capture this preference while still utilizing model-based/model-free learning, eligibility traces, and stage-specific learning rates.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with separate learning rates for stage 1 and 2, eligibility trace, 
    stage 1 perseveration, and an explicit spatial bias for stage 1.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    pers1: stage 1 choice perseveration [-5,5]
    lam: eligibility trace [0,1]
    bias1: baseline bias for choosing action 0 at stage 1 [-5,5]
    """
    alpha1, alpha2, beta1, beta2, w, pers1, lam, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action_1 != -1:
            q1_vals[last_action_1] += pers1
            
        # Add baseline bias towards action 0
        q1_vals[0] += bias1
            
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
