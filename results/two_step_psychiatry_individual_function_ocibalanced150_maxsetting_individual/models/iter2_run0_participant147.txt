Here are three novel cognitive models for the two-step decision-making task. These models incorporate mechanisms such as asymmetric learning rates, stage-specific exploration parameters, and reward-dependent choice perseveration, expanding on the parameter combinations explored previously.

### Model 1: Asymmetric Learning Rates with Choice Perseveration
This model updates action values differently depending on whether the prediction error is positive (better than expected) or negative (worse than expected). Unlike previous combinations, it also incorporates a choice perseveration parameter (`p`) for Stage 1, which is critical for capturing the tendency of participants to repeat their previous choices regardless of the outcome.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative prediction errors.
    Includes Model-Free and Model-Based RL, eligibility trace, and choice perseveration.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based versus model-free values (1 = MB, 0 = MF).
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    lam: [0, 1] - Eligibility trace parameter.
    """
    alpha_pos, alpha_neg, beta, w, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p * pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Inverse Temperatures with Choice Perseveration
This model assumes that participants might exhibit different degrees of stochasticity or exploration when making their first-stage versus their second-stage choices. It uses two separate inverse temperature parameters (`beta_1` and `beta_2`) and reintroduces the essential perseveration parameter (`p`).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate inverse temperatures (beta) for Stage 1 and Stage 2.
    Includes Model-Free and Model-Based RL, eligibility trace, and choice perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta_1: [0, 10] - Inverse temperature for Stage 1 softmax.
    beta_2: [0, 10] - Inverse temperature for Stage 2 softmax.
    w: [0, 1] - Weighting of model-based versus model-free values (1 = MB, 0 = MF).
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    lam: [0, 1] - Eligibility trace parameter.
    """
    learning_rate, beta_1, beta_2, w, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p * pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Perseveration (Win-Stay / Lose-Shift)
This model differentiates between choice perseveration following a rewarded trial versus an unrewarded trial. This directly captures a common behavioral heuristic in two-step tasks: participants often show a strong bias to repeat the exact same Stage 1 choice if they were just rewarded (Win-Stay), but might switch more readily if they received no reward (Lose-Shift).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with reward-dependent choice perseveration at Stage 1 (Win-Stay/Lose-Shift).
    Includes Model-Free and Model-Based RL, eligibility trace, and separate perseveration
    for previously rewarded vs unrewarded trials.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based versus model-free values (1 = MB, 0 = MF).
    p_rew: [-5, 5] - Perseveration parameter for Stage 1 choice if previous trial was rewarded.
    p_unrew: [-5, 5] - Perseveration parameter for Stage 1 choice if previous trial was unrewarded.
    lam: [0, 1] - Eligibility trace parameter.
    """
    learning_rate, beta, w, p_rew, p_unrew, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        p_current = 0
        if last_action_1 != -1:
            pers[last_action_1] = 1
            if last_reward == 1:
                p_current = p_rew
            else:
                p_current = p_unrew
            
        logits_1 = beta * q_stage1_net + p_current * pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
