Here are 3 new cognitive models that build upon the most successful mechanisms identified so far (such as asymmetric learning rates and decay) while exploring novel parameter configurations like choice kernels, dual-stage perseveration, and shared temperature parameters to remain within the 8-parameter limit.

### Model 1: Asymmetric Learning with Stage 1 Choice Kernel
This model replaces simple "last-action" stickiness with a **choice kernel**. A choice kernel captures a longer-term running average of past actions, which is particularly useful for participants who show extended streaks of choosing the same stage 1 action (as seen in this participant's data).
```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates and a choice kernel for stage 1.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    alpha_c: [0, 1] - Choice kernel learning rate.
    beta_c: [0, 10] - Choice kernel inverse temperature (weight).
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, alpha_c, beta_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    ck = np.zeros(2)

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + beta_c * ck
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

        q_stage1_mf[a1] += lr1 * delta_stage1 + lr2 * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += lr2 * delta_stage2
        
        # update choice kernel
        ck[a1] = ck[a1] + alpha_c * (1 - ck[a1])
        ck[1-a1] = ck[1-a1] + alpha_c * (0 - ck[1-a1])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning with Dual-Stage Perseveration
This model adds an independent perseveration parameter for stage 2 decisions. Participants often show a tendency to repeat the same alien choice when visiting the same planet, completely independent of the reward history.
```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates and perseveration at both stages.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p_1: [0, 5] - Perseveration for stage 1.
    p_2: [0, 5] - Perseveration for stage 2.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2_in_state = np.full(2, -1)

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_a1 != -1:
            pers_1[last_a1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p_1 * pers_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers_2 = np.zeros(2)
        if last_a2_in_state[state_idx] != -1:
            pers_2[last_a2_in_state[state_idx]] = 1

        logits_2 = beta_2 * q_stage2_mf[state_idx] + p_2 * pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

        q_stage1_mf[a1] += lr1 * delta_stage1 + lr2 * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += lr2 * delta_stage2
        
        last_a1 = a1
        last_a2_in_state[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Unified Inverse Temperature with Decay and Dual Perseveration
This model combines the strongest features identified so far—asymmetric learning rates, unchosen option decay, and dual-stage perseveration. To accommodate all these features without exceeding the 8-parameter limit, it utilizes a single shared inverse temperature (`beta`) for both stages, leveraging the fact that stage 1 and stage 2 stochasticity are often highly correlated in human participants.
```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates, unchosen option decay, dual perseveration, 
    and a single shared inverse temperature for both stages.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    decay: [0, 1] - Decay rate for unchosen actions (decays towards 0.5).
    beta: [0, 10] - Shared inverse temperature for stage 1 and stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace for stage 1 update from stage 2 reward.
    p_1: [0, 5] - Perseveration for stage 1.
    p_2: [0, 5] - Perseveration for stage 2.
    """
    alpha_pos, alpha_neg, decay, beta, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2_in_state = np.full(2, -1)

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_a1 != -1:
            pers_1[last_a1] = 1
            
        logits_1 = beta * q_stage1_net + p_1 * pers_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers_2 = np.zeros(2)
        if last_a2_in_state[state_idx] != -1:
            pers_2[last_a2_in_state[state_idx]] = 1

        logits_2 = beta * q_stage2_mf[state_idx] + p_2 * pers_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

        q_stage1_mf[a1] += lr1 * delta_stage1 + lr2 * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += lr2 * delta_stage2
        
        # Decay unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_a1 = a1
        last_a2_in_state[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
