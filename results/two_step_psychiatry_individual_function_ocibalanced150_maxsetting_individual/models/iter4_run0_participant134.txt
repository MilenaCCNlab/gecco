Here are 3 new cognitive models that explore alternative parameter configurations and mechanisms to explain the participant's behavior, based on the provided data and feedback.

### Model 1: Distinct Learning Rates for Stage 1 and Stage 2 + Decay
Participants may update their values at different speeds for the first-stage choice (spaceship) versus the second-stage choice (alien). This model uses two separate learning rates (`alpha_1` and `alpha_2`), while retaining the successful decay mechanism that pulls unchosen options towards 0.5.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learner with distinct learning rates for 
    Stage 1 and Stage 2, decay for unchosen options, eligibility trace, and choice stickiness.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for Stage 1 Q-values
    alpha_2: [0, 1] - Learning rate for Stage 2 Q-values
    decay_rate: [0, 1] - Decay rate for unchosen Q-values pulling them towards 0.5
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    stickiness: [-5, 5] - Choice stickiness for stage 1
    """
    alpha_1, alpha_2, decay_rate, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    # Initialize Q-values at 0.5
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits_1 = beta_1 * q_net + stickiness * rep
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Stage 1 and Stage 2 Choice Stickiness
The participant data shows strings of identical alien choices when landing on the same planet. This model introduces `stick_2`, evaluating a tendency to perseverate on the second-stage action (alien) independently of the first-stage choice stickiness. 

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learner with decay for unchosen options,
    eligibility trace, and choice stickiness parameterized separately for both Stage 1 and Stage 2.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating Q-values
    decay_rate: [0, 1] - Decay rate for unchosen Q-values pulling them towards 0.5
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    stick_1: [-5, 5] - Choice stickiness for stage 1
    stick_2: [-5, 5] - Choice stickiness for stage 2 (alien perseveration)
    """
    alpha, decay_rate, beta_1, beta_2, w, lam, stick_1, stick_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) # Keeps track of last action taken in each specific state

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + stick_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        rep_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            rep_2[last_action_2[state_idx]] = 1

        logits_2 = beta_2 * q_stage2_mf[state_idx] + stick_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Learned Transition Matrix
Instead of assuming the participant possesses complete knowledge of the transition matrix (0.7 / 0.3) from the beginning, this model lets the participant dynamically learn the transition probabilities trial-by-trial using a state prediction error.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learner where the transition matrix is 
    learned over time via state prediction errors rather than assumed fixed.
    
    Parameters:
    alpha_q: [0, 1] - Learning rate for Q-values
    alpha_t: [0, 1] - Learning rate for updating the transition matrix
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    stickiness: [-5, 5] - Choice stickiness for stage 1
    """
    alpha_q, alpha_t, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix as completely uniform (uncertain)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits_1 = beta_1 * q_net + stickiness * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update transition matrix using state prediction errors
        transition_matrix[action_1[trial], state_idx] += alpha_t * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_t * (0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_q * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_q * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_q * lam * delta_stage2

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
