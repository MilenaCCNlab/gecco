def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based / Model-Free learner (SARSA(0)).
    This model integrates model-based and model-free values for the first-stage choice.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy (1 = pure MB, 0 = pure MF)
    """
    learning_rate, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Hybrid weighting of model-based and model-free values
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            # Handle missing data / timeout for stage 2
            p_choice_2[trial] = 1.0
  
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            # Action value updating for choice 1 (SARSA)
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            # Action value updating for choice 2
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with choice perseveration.
    Builds on Model 1 by adding a stickiness parameter to capture the participant's 
    tendency to repeat the previous first-stage choice.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    """
    learning_rate, beta, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Hybrid weighting and choice perseveration
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_mb[last_action] += p
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
  
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            # Action value updating for choice 1
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            # Action value updating for choice 2
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with choice perseveration and eligibility trace (Daw et al., 2011).
    Builds on Model 2 by allowing the second-stage reward prediction error to directly 
    update the first-stage action value via an eligibility trace.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter (lambda) linking stage 2 reward to stage 1
    """
    learning_rate, beta, w, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Hybrid weighting and choice perseveration
        q_stage1_mb = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_mb[last_action] += p
            
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
  
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            # Action value updating for choice 1
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            # Action value updating for choice 2
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
            # Eligibility trace update for choice 1 based on stage 2 prediction error
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
