def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition-dependent Stage 1 perseveration combined with Stage 2 perseveration.
    Builds on the best model by adding a tendency to repeat the same Stage 2 action 
    if the same planet is visited again.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    pi_common: Perseveration bonus if previous transition was common [-5, 5]
    pi_rare: Perseveration bonus if previous transition was rare [-5, 5]
    lambd: Eligibility trace [0, 1]
    st2_persev: Stage 2 perseveration bonus for repeating actions in the same state [-5, 5]
    """
    alpha, beta1, beta2, w, pi_common, pi_rare, lambd, st2_persev = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_common = -1
    last_action2_state0 = -1
    last_action2_state1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_trans_common = -1
            continue

        state_idx = state[trial]

        # Stage 1 Model-Based Q-values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Stage 1 Net Q-values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stage 1 Perseveration
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        if last_trans_common == 1:
            pi = pi_common
        elif last_trans_common == 0:
            pi = pi_rare
        else:
            pi = 0.0
            
        logits1 = beta1 * (q_stage1_net + pi * rep)
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Stage 2 Perseveration
        rep2 = np.zeros(2)
        if state_idx == 0 and last_action2_state0 != -1:
            rep2[last_action2_state0] = 1.0
        elif state_idx == 1 and last_action2_state1 != -1:
            rep2[last_action2_state1] = 1.0

        logits2 = beta2 * (q_stage2_mf[state_idx] + st2_persev * rep2)
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        # Track history
        last_action = action_1[trial]
        if (action_1[trial] == 0 and state_idx == 0) or (action_1[trial] == 1 and state_idx == 1):
            last_trans_common = 1
        else:
            last_trans_common = 0
            
        if state_idx == 0:
            last_action2_state0 = action_2[trial]
        elif state_idx == 1:
            last_action2_state1 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates (positive vs negative prediction errors) 
    with transition-dependent perseveration.
    
    Parameters:
    alpha_pos: Learning rate for positive prediction errors [0, 1]
    alpha_neg: Learning rate for negative prediction errors [0, 1]

    w: Model-based weight [0, 1]

    lambd: Eligibility trace [0, 1]

    n_trials = len(action_1)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
