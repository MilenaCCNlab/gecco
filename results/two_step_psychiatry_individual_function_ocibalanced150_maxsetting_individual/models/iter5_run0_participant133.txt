Here are 3 new cognitive models that build upon the best-performing mechanisms so far (especially the decay parameter) while exploring combinations of learning rates, inverse temperatures, and choice persistence (stickiness) that haven't been tested yet.

### Model 1: Separate Learning Rates and Inverse Temperatures with Decay
This model separates both the learning rate and the inverse temperature (softmax beta) for Stage 1 and Stage 2 decisions. It maintains the decay mechanism to pull unchosen options towards 0.5. Separating the parameters accounts for potential differences in how participants update values and explore/exploit at the different stages of the task.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates and inverse temperatures for stage 1 and stage 2,
    along with model-based weight, stage 1 stickiness, and decay for unchosen options.
    
    Parameters/Bounds:
    alpha_1: [0, 1] - learning rate for stage 1
    alpha_2: [0, 1] - learning rate for stage 2
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stickiness: [-5, 5] - bonus for repeating the previous stage 1 action
    decay: [0, 1] - decay rate for all action values towards 0.5 before updating
    """
    alpha_1, alpha_2, beta_1, beta_2, w, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus = np.zeros(2)
        if last_action != -1:
            stick_bonus[last_action] = stickiness
            
        q_vals_1 = beta_1 * q_net + stick_bonus
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q_vals_2 = beta_2 * q_stage2_mf[state_idx]
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen values towards 0.5
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # TD(1) style update for stage 1 based on stage 2 reward
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage 1 and Stage 2 Stickiness with Decay
This model explores the idea that perseveration (stickiness) isn't just limited to the first-stage choice. Participants might also have a tendency to pick the same alien (Stage 2) they chose the last time they were on that specific planet.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with stickiness for both stage 1 and stage 2 choices, separate inverse temperatures,
    and decay for unchosen options.
    
    Parameters/Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stick_1: [-5, 5] - bonus for repeating the previous stage 1 action
    stick_2: [-5, 5] - bonus for repeating the previous stage 2 action (in the same state)
    decay: [0, 1] - decay rate for all action values towards 0.5 before updating
    """
    alpha, beta_1, beta_2, w, stick_1, stick_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_bonus_1[last_action_1] = stick_1
            
        q_vals_1 = beta_1 * q_net + stick_bonus_1
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        stick_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stick_bonus_2[last_action_2[state_idx]] = stick_2

        q_vals_2 = beta_2 * q_stage2_mf[state_idx] + stick_bonus_2
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen values towards 0.5
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Win-Stay / Lose-Shift Stickiness with Decay
This model posits that perseveration depends on the outcome of the previous trial. A participant may be highly likely to repeat a Stage 1 choice if they were rewarded (`stick_win`), but might have a distinct tendency to switch (or stick differently) if they were not rewarded (`stick_lose`).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with win-stay/lose-shift tendencies at stage 1, separate inverse temperatures,
    and decay for unchosen options. The stickiness bonus depends on whether the previous
    trial was rewarded (stick_win) or unrewarded (stick_lose).
    
    Parameters/Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    stick_win: [-5, 5] - bonus for repeating the previous stage 1 action if rewarded
    stick_lose: [-5, 5] - bonus for repeating the previous stage 1 action if unrewarded
    decay: [0, 1] - decay rate for all action values towards 0.5 before updating
    """
    alpha, beta_1, beta_2, w, stick_win, stick_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        stick_bonus = np.zeros(2)
        if last_action != -1:
            if last_reward == 1:
                stick_bonus[last_action] = stick_win
            else:
                stick_bonus[last_action] = stick_lose
            
        q_vals_1 = beta_1 * q_net + stick_bonus
        q_vals_1 -= np.max(q_vals_1)
        exp_q1 = np.exp(q_vals_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q_vals_2 = beta_2 * q_stage2_mf[state_idx]
        q_vals_2 -= np.max(q_vals_2)
        exp_q2 = np.exp(q_vals_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen values towards 0.5
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
