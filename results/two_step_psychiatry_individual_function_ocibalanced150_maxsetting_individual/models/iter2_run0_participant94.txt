def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) with Stage 2 state-specific choice stickiness and decay.
    This model captures perseveration at both decision stages, tracking the last action 
    taken in each specific Stage 2 state, along with forgetting of unchosen options.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi_1: [-5, 5] - stage 1 choice stickiness
    pi_2: [-5, 5] - stage 2 state-specific choice stickiness
    decay: [0, 1] - decay rate for unchosen options
    """
    alpha, beta_1, beta_2, w, lambda_, pi_1, pi_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1
    last_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial]

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_a1 != -1:
            rep_1[last_a1] = 1.0
            
        q1_scaled = beta_1 * q_stage1_net + pi_1 * rep_1
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        rep_2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            rep_2[last_a2[state_idx]] = 1.0
            
        q2_scaled = beta_2 * q_stage2_mf[state_idx] + pi_2 * rep_2
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        # Decay unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) with Win-Stay/Lose-Shift (WSLS) effect on Stage 1 and decay.
    The tendency to repeat the last Stage 1 action depends on whether the previous trial was rewarded.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi_win: [-5, 5] - stage 1 choice stickiness after a win (reward > 0)
    pi_lose: [-5, 5] - stage 1 choice stickiness after a loss (reward <= 0)
    decay: [0, 1] - decay rate for unchosen options
    """
    alpha, beta_1, beta_2, w, lambda_, pi_win, pi_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial]

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_a1 != -1:
            rep[last_a1] = 1.0
            
        pi_current = 0.0
        if last_reward > 0:
            pi_current = pi_win
        elif last_reward == 0 or last_reward < 0:
            pi_current = pi_lose
            
        q1_scaled = beta_1 * q_stage1_net + pi_current * rep
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        # Decay unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_a1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) with transition matrix learning and decay.
    Instead of assuming fixed 0.7/0.3 probabilities, the participant updates their 
    internal model of the transition matrix based on experienced state transitions.
    
    Bounds:
    alpha: [0, 1] - learning rate for Q-values
    alpha_trans: [0, 1] - learning rate for transition probabilities
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi: [-5, 5] - stage 1 choice stickiness
    decay: [0, 1] - decay rate for unchosen options
    """
    alpha, alpha_trans, beta_1, beta_2, w, lambda_, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial]

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_a1 != -1:
            rep[last_a1] = 1.0
            
        q1_scaled = beta_1 * q_stage1_net + pi * rep
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2
        
        # Update transition matrix based on experience
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Decay unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
