def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for stage 1 and stage 2, eligibility trace, MB/MF weighting, and choice stickiness.
    
    Parameters:
    model_parameters (tuple):
        lr_1: [0, 1] - Learning rate for stage 1 Q-value updates.
        lr_2: [0, 1] - Learning rate for stage 2 Q-value updates.
        w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
        lambda_: [0, 1] - Eligibility trace for updating stage 1 based on stage 2 reward prediction error.
        stickiness: [-5, 5] - Bonus for repeating the previous stage 1 action.
        beta: [0, 10] - Inverse temperature for both stages.
    """
    lr_1, lr_2, w, lambda_, stickiness, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_net[last_action] += stickiness
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # policy for the second choice
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        # eligibility trace update for stage 1 based on stage 2 prediction error
        q_stage1_mf[action_1[trial]] += lr_1 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Win-stay/lose-shift stickiness, MB/MF weighting, and separate inverse temperatures for each stage.
    
    Parameters:
    model_parameters (tuple):
        learning_rate: [0, 1] - Learning rate for Q-value updates.
        w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
        stay_win: [-5, 5] - Bonus for repeating the previous stage 1 action if the last trial was rewarded.
        stay_lose: [-5, 5] - Bonus for repeating the previous stage 1 action if the last trial was unrewarded.
        beta1: [0, 10] - Inverse temperature for stage 1.
        beta2: [0, 10] - Inverse temperature for stage 2.
    """
    learning_rate, w, stay_win, stay_lose, beta1, beta2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            if last_reward > 0:
                q_net[last_action] += stay_win
            else:
                q_net[last_action] += stay_lose
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta1 * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # policy for the second choice
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):

    Parameters:
    model_parameters (tuple):
        alpha_pos: [0, 1] - Learning rate for positive prediction errors.

            p_choice_2[trial] = 1.0
