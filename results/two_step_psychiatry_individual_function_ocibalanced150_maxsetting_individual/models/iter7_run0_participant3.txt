def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid RL with State-Specific and Motor-Specific Stage 2 Perseveration
    
    This model separates stage 2 perseveration into a state-specific component (choosing the
    same alien on the same planet) and a motor-specific component (repeating the same action
    index regardless of the planet). This captures a common tendency to simply press the same
    button again.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam: [0, 1] - Eligibility trace parameter for stage 1 model-free update.
    p1: [-5, 5] - Perseveration parameter for stage 1 choice.
    p2_state: [-5, 5] - Perseveration parameter for stage 2 choice (specific to state).
    p2_motor: [-5, 5] - Perseveration parameter for stage 2 choice (independent of state).
    """
    learning_rate, beta, w, lam, p1, p2_state, p2_motor = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2_state = np.array([-1, -1])
    last_a2_motor = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1.0
            
        logits_1 = beta * q_stage1_net + p1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers2_state = np.zeros(2)
        if last_a2_state[state_idx] != -1:
            pers2_state[last_a2_state[state_idx]] = 1.0
            
        pers2_motor = np.zeros(2)
        if last_a2_motor != -1:
            pers2_motor[last_a2_motor] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + p2_state * pers2_state + p2_motor * pers2_motor
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_a1 = a1
        last_a2_state[state_idx] = a2
        last_a2_motor = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid RL with Stage 1 & 2 Perseveration and Unrewarded Punishment
    
    Treats unrewarded trials (reward=0) as a negative outcome (punishment) rather than just zero,
    allowing the model to capture asymmetric sensitivity to missing a reward.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam: [0, 1] - Eligibility trace parameter for stage 1 model-free update.
    p1: [-5, 5] - Perseveration parameter for stage 1 choice.
    p2: [-5, 5] - Perseveration parameter for stage 2 choice (specific to state).
    punishment: [-5, 5] - Effective value of an unrewarded outcome (reward = 0).
    """
    learning_rate, beta, w, lam, p1, p2, punishment = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1.0
            
        logits_1 = beta * q_stage1_net + p1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            pers2[last_a2[state_idx]] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + p2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        eff_reward = reward[trial] if reward[trial] > 0 else punishment

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = eff_reward - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid RL with Stage 1 & 2 Perseveration and Stage 1 & 2 Choice Biases
    
    Introduces baseline choice biases for both stages, which can capture intrinsic preferences
    for specific spaceships or aliens regardless of their learned values.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam: [0, 1] - Eligibility trace parameter for stage 1 model-free update.
    p1: [-5, 5] - Perseveration parameter for stage 1 choice.
    p2: [-5, 5] - Perseveration parameter for stage 2 choice (specific to state).
    bias_1: [-5, 5] - Baseline choice bias for spaceship 0 at stage 1.
    bias_2: [-5, 5] - Baseline choice bias for alien 0 at stage 2.
    """
    learning_rate, beta, w, lam, p1, p2, bias_1, bias_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1.0
            
        b1 = np.array([bias_1, 0.0])
            
        logits_1 = beta * q_stage1_net + p1 * pers1 + b1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            pers2[last_a2[state_idx]] = 1.0

        b2 = np.array([bias_2, 0.0])

        logits_2 = beta * q_stage2_mf[state_idx] + p2 * pers2 + b2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
