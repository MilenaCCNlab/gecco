Here are three new cognitive models to explain the participant's decision-making behavior. They explore different mechanisms such as stage-specific perseverance, separate learning rates, and unchosen action value decay, while avoiding the previously tested parameter combinations.

### Model 1: Stage 1 and Stage 2 Choice Perseverance
This model extends the hybrid model by incorporating stickiness not just in the first-stage choice, but also in the second-stage choices, tracking the participant's tendency to repeat actions independently across the two planets.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Stage 1 and Stage 2 Choice Perseverance.
    
    This model adds perseverance (stickiness) not only to the first-stage choice 
    but also to the second-stage choices, tracking the last action taken in each state.

    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate for updating Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi_1: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1.
        pi_2: [-5, 5] - Choice perseverance (stickiness) parameter for stage 2.
    """
    alpha, beta, w, lambda_, pi_1, pi_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = None
    last_action_2 = [None, None]

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action_1 is not None:
            q_stage1_net[last_action_1] += pi_1
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] is not None:
            q_stage2_net[last_action_2[state_idx]] += pi_2

        exp_q2 = np.exp(beta * (q_stage2_net - np.max(q_stage2_net)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Learning Rates for Stage 1 and Stage 2
This model allows for different speeds of learning (updating action values) at the first and second stages, combined with choice perseverance at the first stage. This captures differences in how quickly participants integrate feedback for the spaceship choice versus the alien choice.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Separate Learning Rates for Stage 1 and Stage 2.
    
    This model allows for different speeds of updating action values at the first 
    and second stages, combined with choice perseverance at the first stage.

    Parameters:
    model_parameters: list or array containing:
        alpha1: [0, 1] - Learning rate for updating Stage 1 Q-values.
        alpha2: [0, 1] - Learning rate for updating Stage 2 Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1.
    """
    alpha1, alpha2, beta, w, lambda_, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        if last_action is not None:
            q_stage1_net[last_action] += pi

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Decay (Forgetting) of Unchosen Action Values
This model assumes that the values of unchosen second-stage actions decay towards a neutral value (`0.5`) over time, capturing forgetting mechanisms. This is particularly relevant when a participant stays on one side of the task for a long time and might "forget" the learned values for the other planet.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Decay (Forgetting) of Unchosen Action Values.
    
    This model assumes that the values of unchosen second-stage actions decay 
    towards a neutral value (0.5) over time, capturing forgetting mechanisms.

    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate for updating Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1.
        decay: [0, 1] - Decay rate for unchosen Stage 2 actions towards 0.5.
    """
    alpha, beta, w, lambda_, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.ones((2, 2)) * 0.5  # Initialize to neutral value 0.5
    
    last_action = None

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Decay unchosen stage 2 actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
