Here are 3 new cognitive models that explore alternative configurations and mechanisms based on the participant's data. 

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
Given that the participant's behavior may update at different speeds for the overarching spaceship choice versus the specific alien choice, this model uses independent learning rates for Stage 1 (`alpha1`) and Stage 2 (`alpha2`), while maintaining the full Daw et al. (2011) hybrid architecture.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with separate learning rates for stage 1 and stage 2.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers: perseveration bonus for repeating the previous stage 1 choice [-2, 2]
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        q_stage1_total = beta1 * q_stage1_net + pers_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_total = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # eligibility trace uses alpha1 because it's updating the stage 1 MF value
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates for Positive/Negative Prediction Errors
Participants often update values differently when they are pleasantly surprised (rewarded) versus disappointed (omission). This model implements asymmetric learning rates (`alpha_pos` and `alpha_neg`) that apply dynamically based on the sign of the reward prediction error (RPE) at each stage.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with asymmetric learning rates for positive and negative prediction errors.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers: perseveration bonus for repeating the previous stage 1 choice [-2, 2]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        q_stage1_total = beta1 * q_stage1_net + pers_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_total = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1_eff = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1_eff * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2_eff = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_eff * delta_stage2
        
        # eligibility trace
        q_stage1_mf[action_1[trial]] += alpha2_eff * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Choice Perseveration at Both Stages
Observing the participant's sequence, they tend to consistently match the same alien actions (frequently choosing alien 1 when moving in spaceship 0 and alien 0 when moving in spaceship 1). This behavior heavily indicates cross-trial second-stage action dependency. This model adds a `pers2` parameter to explicitly quantify perseveration on the Stage 2 choices.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with perseveration on both stage 1 and stage 2 choices.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers1: perseveration bonus for repeating the previous stage 1 choice [-2, 2]
    pers2: perseveration bonus for repeating the previous stage 2 choice [-2, 2]
    """
    alpha, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers1_bonus[last_action_1] = pers1
            
        q_stage1_total = beta1 * q_stage1_net + pers1_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2_bonus = np.zeros(2)
        if last_action_2 != -1:
            pers2_bonus[last_action_2] = pers2
            
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2_bonus
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
