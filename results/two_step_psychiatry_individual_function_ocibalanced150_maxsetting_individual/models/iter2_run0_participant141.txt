def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with reward-dependent perseveration (Win-Stay / Lose-Shift).
    Instead of a single perseveration parameter, this model has separate parameters
    for repeating a choice after a rewarded trial vs an unrewarded trial.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updating
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers_win: [-5, 5] - Perseveration parameter after a rewarded trial
    pers_lose: [-5, 5] - Perseveration parameter after an unrewarded trial
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        pers_effect = 0
        if last_reward == 1:
            pers_effect = pers_win
        elif last_reward == 0:
            pers_effect = pers_lose
            
        logits1 = beta1 * q_net + pers_effect * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Action value updating
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric eligibility traces for positive and negative RPEs.
    This model assumes that credit assignment to the first stage differs depending
    on whether the second stage outcome was better or worse than expected.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updating
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam_pos: [0, 1] - Eligibility trace parameter for positive RPEs
    lam_neg: [0, 1] - Eligibility trace parameter for negative RPEs
    pers: [-5, 5] - Perseveration parameter
    """
    alpha, beta1, beta2, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits1 = beta1 * q_net + pers * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Action value updating
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with a spatial/action bias and forgetting.
    This model includes a general preference (bias) for one of the first-stage actions,
    along with forgetting (decay) of unchosen action values.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updating
    decay: [0, 1] - Forgetting rate for unchosen actions (values revert toward 0.5)
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers: [-5, 5] - Perseveration parameter
    bias: [-5, 5] - Bias for action 0 over action 1
    """
    alpha, decay, beta1, beta2, w, lam, pers, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        action_bias = np.array([bias, 0.0])
            
        logits1 = beta1 * q_net + pers * rep + action_bias
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Decay unchosen actions
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
