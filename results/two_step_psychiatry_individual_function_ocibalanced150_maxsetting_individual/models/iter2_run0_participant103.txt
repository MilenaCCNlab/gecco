import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Asymmetric Learning and Stage 1 Stickiness.
    Uses separate learning rates for positive and negative prediction errors,
    which captures differential sensitivity to better-than-expected vs 
    worse-than-expected outcomes, alongside choice perseveration.
    
    Parameters:
    alpha_pos: Learning rate for positive RPE [0, 1]
    alpha_neg: Learning rate for negative RPE [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    lam: Eligibility trace [0, 1]
    st: Stage 1 choice stickiness [-10, 10]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, st = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net
        if prev_a1 != -1:
            logit1[prev_a1] += st
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Forgetting (Decay) and Dual Stickiness.
    Builds upon the best performing model so far by adding choice perseveration 
    at both stage 1 and stage 2, along with Q-value decay.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    lam: Eligibility trace [0, 1]
    st1: Stage 1 choice stickiness [-10, 10]
    st2: Stage 2 choice stickiness [-10, 10]
    decay: Forgetting rate for all Q-values per trial [0, 1]
    """
    alpha, beta1, beta2, w, lam, st1, st2, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1
    prev_state = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logit1 = beta1 * q_net
        if prev_a1 != -1:
            logit1[prev_a1] += st1
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logit2 = beta2 * q_stage2_mf[state_idx]
        if prev_state == state_idx and prev_a2 != -1:
            logit2[prev_a2] += st2
            
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]
        prev_state = state_idx
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
