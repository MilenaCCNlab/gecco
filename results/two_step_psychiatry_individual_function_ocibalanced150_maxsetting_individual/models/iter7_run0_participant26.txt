import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Single learning rate, two betas (one for each stage), model-based weight, 
    eligibility trace, and a choice trace (exponential moving average) for stickiness.
    This model captures block-like choice behavior by maintaining a persistent trace of past choices.
    
    Bounds:
    learning_rate: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    rho: [0, 5]
    alpha_c: [0, 1]
    """
    learning_rate, beta_1, beta_2, w, lam, rho, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = q_stage1_net + rho * choice_trace_1
        exp_q1 = np.exp(beta_1 * (q1_vals - np.max(q1_vals)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = q_stage2_mf[state_idx] + rho * choice_trace_2[state_idx]
        exp_q2 = np.exp(beta_2 * (q2_vals - np.max(q2_vals)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Choice trace updating
        choice_trace_1 *= (1 - alpha_c)
        choice_trace_1[action_1[trial]] += alpha_c
        
        choice_trace_2[state_idx] *= (1 - alpha_c)
        choice_trace_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Single learning rate, two betas, model-based weight, eligibility trace, 
    decay for unchosen options, and separate stickiness for previous win (rho_win) 
    and previous loss (rho_lose) at stage 1. This evaluates whether participant switching 
    behavior is heavily driven by outcome-dependent perseveration.
    
    Bounds:
    learning_rate: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    rho_win: [0, 5]
    rho_lose: [-5, 5]
    decay: [0, 1]
    """
    learning_rate, beta_1, beta_2, w, lam, rho_win, rho_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_1[last_action_1] = rho_win
            else:
                stickiness_1[last_action_1] = rho_lose
            
        q1_vals = q_stage1_net + stickiness_1
        exp_q1 = np.exp(beta_1 * (q1_vals - np.max(q1_vals)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta_2 * (q2_vals - np.max(q2_vals)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay for unchosen options
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate learning rates for positive and negative prediction errors, 
    two betas, model-based weight, eligibility trace, choice trace (alpha_c) and stickiness (rho).
    This model investigates if subjects learn differently from positive versus negative feedback 
    while preserving long-term choice stickiness.
    
    Bounds:
    lr_pos: [0, 1]
    lr_neg: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    rho: [0, 5]
    alpha_c: [0, 1]
    """
    lr_pos, lr_neg, beta_1, beta_2, w, lam, rho, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = q_stage1_net + rho * choice_trace_1
        exp_q1 = np.exp(beta_1 * (q1_vals - np.max(q1_vals)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_vals = q_stage2_mf[state_idx] + rho * choice_trace_2[state_idx]
        exp_q2 = np.exp(beta_2 * (q2_vals - np.max(q2_vals)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2
        
        # Choice trace updating
        choice_trace_1 *= (1 - alpha_c)
        choice_trace_1[action_1[trial]] += alpha_c
        
        choice_trace_2[state_idx] *= (1 - alpha_c)
        choice_trace_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
