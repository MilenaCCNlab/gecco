Here are 3 new cognitive models exploring alternative mechanisms for the two-step task. 

### Model 1: Asymmetric Learning Rates Model
This model tests the hypothesis that participants might update their expectations at different speeds for the first and second stages. It separates the learning rate into `alpha_1` for stage 1 and `alpha_2` for stage 2.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for stage 1 and stage 2.
    Different learning rates can capture distinct learning dynamics at each stage, 
    reflecting differences in how credit assignment is processed across temporal steps.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for stage 1.
    alpha_2: [0, 1] - Learning rate for stage 2.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseveration parameter for stage 1.
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            rep = np.zeros(2)
            rep[last_action_1] = 1
            q_stage1_net += p * rep
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating using asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Dual-Stage Perseveration Model
This model extends the concept of perseveration (the tendency to repeat the previous action regardless of reward) from just stage 1 to stage 2. It tracks the last action taken in each specific second-stage state and provides a parameter `p_2` to capture habitual responding at the alien-selection level.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate perseveration parameters for stage 1 and stage 2 choices.
    Individuals may exhibit habitual behavior (perseveration) not only at the first stage
    but also at the second stage, repeating previous actions in a given state.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    p_1: [-5, 5] - Perseveration parameter for stage 1.
    p_2: [-5, 5] - Perseveration parameter for stage 2.
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            rep1 = np.zeros(2)
            rep1[last_action_1] = 1
            q_stage1_net += p_1 * rep1
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            rep2[last_action_2[state_idx]] = 1
        q2_net = q_stage2_mf[state_idx] + p_2 * rep2
            
        q2_scaled = beta_2 * q2_net
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Choice Trace (Long-term Habit) Model
Instead of merely looking at trial `n-1` for perseveration (which only captures immediate repeats), this model integrates an exponentially decaying choice trace. This allows the model to capture longer-term autocorrelation in actions (e.g., sticking with a spaceship for several trials out of habit rather than calculated value).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Choice trace model.
    Instead of just trial n-1 perseveration, this model includes an exponentially decaying 
    choice trace that captures longer-term tendencies to repeat previous stage 1 actions.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    alpha_c: [0, 1] - Choice trace decay rate.
    beta_c: [-5, 5] - Weight of the choice trace on stage 1 choices.
    """
    alpha, beta_1, beta_2, w, lam, alpha_c, beta_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        q1_scaled = beta_1 * q_stage1_net + beta_c * choice_trace
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # update choice trace
        choice_trace = (1 - alpha_c) * choice_trace
        choice_trace[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
