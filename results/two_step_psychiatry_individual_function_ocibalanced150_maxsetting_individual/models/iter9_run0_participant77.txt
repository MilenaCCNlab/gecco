import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Stage 1 Response Bias and Win-Stay Perseveration.
    Includes a constant bias for Stage 1 action 1 (which can be negative to favor action 0),
    along with a win-stay perseveration bonus for Stage 1. This addresses the participant's 
    strong preference for choosing spaceship 0 observed in the data.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win_1: stage 1 perseveration bonus if previous trial was rewarded [-2,2]
    bias_1: constant bias added to the value of Stage 1 action 1 [-5,5]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win_1, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseveration and bias
        q_net_total = beta_1 * q_net
        if last_action_1 != -1 and last_reward > 0:
            q_net_total[last_action_1] += pers_win_1
            
        q_net_total[1] += bias_1
        
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Stage 2 Learning and Stage 1 Win-Stay.
    Uses separate learning rates for positive and negative reward prediction errors in Stage 2,
    while keeping standard learning for Stage 1, plus a win-stay bonus for Stage 1.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_pos_2: learning rate for stage 2 positive RPEs [0,1]
    alpha_neg_2: learning rate for stage 2 negative RPEs [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win_1: stage 1 perseveration bonus if previous trial was rewarded [-2,2]
    """
    alpha_1, alpha_pos_2, alpha_neg_2, beta_1, beta_2, w, lam, pers_win_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_total = beta_1 * q_net
        if last_action_1 != -1 and last_reward > 0:
            q_net_total[last_action_1] += pers_win_1
            
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            
            if delta_stage2 > 0:
                q_stage2_mf[state_idx, action_2[trial]] += alpha_pos_2 * delta_stage2
            else:
                q_stage2_mf[state_idx, action_2[trial]] += alpha_neg_2 * delta_stage2
                
            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Subjective Transition Probability and Stage 1 Win-Stay.
    Instead of fixing the transition probability at 0.7, this model fits a 
    subjective common transition probability parameter, along with stage 1 win-stay.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_win_1: stage 1 perseveration bonus if previous trial was rewarded [-2,2]
    p_common: subjective probability of common transition [0.5,1.0]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win_1, p_common = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[p_common, 1 - p_common], [1 - p_common, p_common]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_total = beta_1 * q_net
        if last_action_1 != -1 and last_reward > 0:
            q_net_total[last_action_1] += pers_win_1
            
        q_net_total = q_net_total - np.max(q_net_total)
        exp_q1 = np.exp(q_net_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            q_stage2_total = q_stage2_total - np.max(q_stage2_total)
            exp_q2 = np.exp(q_stage2_total)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = 0
            
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
