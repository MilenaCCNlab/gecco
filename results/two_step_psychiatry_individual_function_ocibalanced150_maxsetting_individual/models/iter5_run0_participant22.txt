def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with separate perseveration for rewarded and unrewarded previous trials.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for Stage 1 updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p_rew: [-5, 5] - Perseveration parameter for Stage 1 choice after a rewarded trial.
    p_unrew: [-5, 5] - Perseveration parameter for Stage 1 choice after an unrewarded trial.
    """
    alpha_1, alpha_2, beta, w, lam, p_rew, p_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = p_rew if last_reward == 1 else p_unrew
            
        q_stage1_combined = q_stage1_net + pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        # Track history for next trial
        last_choice = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with separate perseveration based on whether the previous transition was common or rare.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for Stage 1 updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p_common: [-5, 5] - Perseveration parameter for Stage 1 choice after a common transition.
    p_rare: [-5, 5] - Perseveration parameter for Stage 1 choice after a rare transition.
    """
    alpha_1, alpha_2, beta, w, lam, p_common, p_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1
    last_trans_common = False

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = p_common if last_trans_common else p_rare
            
        q_stage1_combined = q_stage1_net + pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        # Track history for next trial
        last_choice = action_1[trial]
        # In this task structure, action 0 -> state 0 and action 1 -> state 1 are the common transitions (0.7 probability)
        last_trans_common = (action_1[trial] == state[trial])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with separate eligibility traces for rewarded and unrewarded trials.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for Stage 1 updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam_rew: [0, 1] - Eligibility trace for rewarded trials.
    lam_unrew: [0, 1] - Eligibility trace for unrewarded trials.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    """
    alpha_1, alpha_2, beta, w, lam_rew, lam_unrew, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
            
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)

        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating

        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for Stage 1 depending on whether trial was rewarded

        # Track history for next trial
        last_choice = action_1[trial]
