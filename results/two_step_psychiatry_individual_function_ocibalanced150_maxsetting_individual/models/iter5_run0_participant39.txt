Here are three new cognitive models that explore alternative mechanisms and parameter combinations not yet tested. 

### Model 1: Separate Temperatures and Perseveration for Stage 1 and Stage 2
This model allows for different choice temperatures (exploration vs. exploitation balance) and different stickiness (perseveration) parameters for the first and second stages. Participants often exhibit different decision mechanisms at the two stages (e.g., highly sticky at stage 1, but more exploratory at stage 2).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate temperatures and perseveration for Stage 1 and Stage 2.
    
    Parameters:
    learning_rate: learning rate for updating Q-values [0,1]
    beta1: inverse temperature for softmax choice at stage 1 [0,10]
    beta2: inverse temperature for softmax choice at stage 2 [0,10]
    w: weight for model-based vs model-free control [0,1]
    pers1: stickiness/perseveration bonus for repeating the previous stage 1 choice [-5, 5]
    pers2: stickiness/perseveration bonus for repeating the previous stage 2 choice [-5, 5]
    """
    learning_rate, beta1, beta2, w, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action_1 != -1:
            q1_vals[last_action_1] += pers1
            
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        if last_action_2 != -1:
            q2_vals[last_action_2] += pers2
            
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Decay for Unchosen Actions
This model introduces a `decay` parameter. Unchosen Q-values decay towards zero, meaning that options not chosen for a long time lose their cached value. This can explain bursts of exploratory behavior after extended periods of exploiting one option (like we see in the participant's data sparingly choosing spaceship 1).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model featuring decay for unchosen Q-values.
    
    Parameters:
    learning_rate: learning rate for updating chosen Q-values [0,1]
    decay: decay rate for unchosen Q-values [0,1]
    beta1: inverse temperature for softmax choice at stage 1 [0,10]
    beta2: inverse temperature for softmax choice at stage 2 [0,10]
    w: weight for model-based vs model-free control [0,1]
    pers: stickiness/perseveration bonus for repeating previous stage 1 choice [-5, 5]
    """
    learning_rate, decay, beta1, beta2, w, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action_1 != -1:
            q1_vals[last_action_1] += pers
            
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update chosen Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Decay unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Eligibility Trace and Stage 2 Win-Stay-Lose-Shift (WSLS)
This model combines an eligibility trace (`lam`) to pass Stage 2 reward info directly to Stage 1, alongside an outcome-dependent Win-Stay/Lose-Shift heuristic at Stage 2. If the participant receives a reward, they get a bonus to pick the identical alien; if no reward, they get a penalty to that alien (encouraging a shift).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with eligibility trace and Stage 2 Win-Stay-Lose-Shift (WSLS) effect.
    
    Parameters:
    learning_rate: learning rate for updating Q-values [0,1]
    beta1: inverse temperature for softmax choice at stage 1 [0,10]
    beta2: inverse temperature for softmax choice at stage 2 [0,10]
    w: weight for model-based vs model-free control [0,1]
    pers: stickiness/perseveration bonus for repeating the previous stage 1 choice [-5, 5]
    lam: eligibility trace parameter linking stage 2 rewards to stage 1 updates [0, 1]
    wsls2: Win-Stay-Lose-Shift bonus for stage 2 based on previous outcome [-5, 5]
    """
    learning_rate, beta1, beta2, w, pers, lam, wsls2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action_1 != -1:
            q1_vals[last_action_1] += pers
            
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        
        # Apply Stage 2 WSLS if returning to the exact same state (planet)
        if last_action_2 != -1 and last_state == state_idx:
            if last_reward == 1:
                q2_vals[last_action_2] += wsls2
            else:
                q2_vals[last_action_2] -= wsls2
                
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # TD learning with eligibility trace
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Stage 1 eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
