def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Win-Stay / Lose-Shift perseveration.
    Perseveration in Stage 1 depends on whether the previous trial was rewarded or not.
    This captures the tendency of participants to repeat choices differently after a win versus a loss.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace. Bounds: [0,1]
    pers_win (float): Perseveration if previous trial was rewarded. Bounds: [-5,5]
    pers_lose (float): Perseveration if previous trial was unrewarded. Bounds: [-5,5]
    """
    alpha1, alpha2, beta, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # Handle invalid trials
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply win-stay / lose-shift perseveration
        if last_action1 != -1:
            if last_reward == 1:
                q_net1[last_action1] += pers_win
            elif last_reward == 0:
                q_net1[last_action1] += pers_lose
                
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update for choice 1 based on stage 2 PE
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate eligibility traces for positive and negative prediction errors.
    This model allows the credit assignment to Stage 1 to differ depending on whether 
    the outcome was better or worse than expected.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam_pos (float): Eligibility trace for positive PE. Bounds: [0,1]
    lam_neg (float): Eligibility trace for negative PE. Bounds: [0,1]
    pers (float): Perseveration for Stage 1. Bounds: [-5,5]
    """
    alpha1, alpha2, beta, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
                
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Apply different lambda based on the sign of the Stage 2 PE
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates for positive and negative Stage 2 PEs, with Q-value decay.
    Captures differing sensitivity to rewards vs omissions, alongside forgetting of unchosen options.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2_pos (float): Stage 2 learning rate for positive PE. Bounds: [0,1]
    alpha2_neg (float): Stage 2 learning rate for negative PE. Bounds: [0,1]
    beta (float): Inverse temperature. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace. Bounds: [0,1]
    pers (float): Perseveration for Stage 1. Bounds: [-5,5]
    decay (float): Q-value decay rate. Bounds: [0,1]
    """
    alpha1, alpha2_pos, alpha2_neg, beta, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
                
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Decay before update
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2 with asymmetric learning rates
        alpha2 = alpha2_pos if delta_stage2 > 0 else alpha2_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
