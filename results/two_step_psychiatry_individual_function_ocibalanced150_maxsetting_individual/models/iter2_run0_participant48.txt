Here are three new cognitive models based on the two-step decision-making task. Each model introduces a distinct cognitive mechanism to capture different aspects of participant behavior, such as asymmetric learning for positive/negative prediction errors, separate exploration rates for different task stages, and learning the task's transition structure.

### Model 1: Asymmetric Learning Rates
This model assumes that participants update their expectations differently depending on whether the outcome was better or worse than expected. It uses separate learning rates for positive and negative prediction errors (`lr_pos` and `lr_neg`), along with a choice perseveration parameter to capture the tendency to stick to the previously chosen spaceship.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Learning Rates.
    Different learning rates for positive and negative prediction errors, with stage 1 choice perseveration.
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors
    lr_neg: [0,1] - Learning rate for negative prediction errors
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Weighting parameter for model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers: [-5,5] - Choice perseveration (stickiness) for stage 1
    """
    lr_pos, lr_neg, beta, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Handle missing trial data
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta * q_net
        
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = lr_pos if delta_stage2 > 0 else lr_neg
        
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr_stage2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Stage Inverse Temperatures and Decay
Participants might exhibit different levels of exploration/exploitation depending on whether they are making the first stage (spaceship) or second stage (alien) decision. This model separates `beta` into `beta1` and `beta2`. It also incorporates passive memory decay where unchosen options regress toward a neutral value (0.5).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Separate Stage Inverse Temperatures and Decay.
    Different softmax inverse temperatures for stage 1 and stage 2, with unchosen value decay and stage 1 choice perseveration.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta1: [0,10] - Inverse temperature for stage 1 action selection
    beta2: [0,10] - Inverse temperature for stage 2 action selection
    w: [0,1] - Weighting parameter for model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers: [-5,5] - Choice perseveration (stickiness) for stage 1
    decay: [0,1] - Decay rate for unchosen Q-values towards 0.5
    """
    learning_rate, beta1, beta2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_net
        
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Unchosen value decay towards 0.5
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition Learning
Rather than assuming the participant has perfect knowledge of the fixed 70/30 transition probabilities from the start, this model dynamically updates the expected transition structure based on the observed transitions using a dedicated transition learning rate (`lr_trans`). 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Transition Learning and Perseveration.
    Updates the expected transition probabilities based on observed transitions.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Weighting parameter for model-based vs model-free values
    lam: [0,1] - Eligibility trace parameter for stage 1 updates
    pers: [-5,5] - Choice perseveration (stickiness) for stage 1
    lr_trans: [0,1] - Learning rate for updating the transition matrix
    """
    learning_rate, beta, w, lam, pers, lr_trans = model_parameters
    n_trials = len(action_1)
  
    # Start with a uniform prior for the transition matrix
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta * q_net
        
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update transition matrix probabilities
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += lr_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += lr_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
