def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with asymmetric learning rates for positive and negative prediction errors,
    separate inverse temperatures for stage 1 and stage 2, and stage 1 perseveration.
    
    Parameters:
    alpha_pos: [0,1] - learning rate for positive prediction errors.
    alpha_neg: [0,1] - learning rate for negative prediction errors.
    beta1: [0,10] - inverse temperature for stage 1 softmax action selection.
    beta2: [0,10] - inverse temperature for stage 2 softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p: [-5,5] - perseveration bonus for stage 1.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = p
            
        q_net_pers1 = beta1 * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta2 * q2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with separate learning rates and inverse temperatures for stage 1 and stage 2,
    and choice perseveration at both stages.
    
    Parameters:
    alpha1: [0,1] - learning rate for stage 1 value updates.
    alpha2: [0,1] - learning rate for stage 2 value updates.
    beta1: [0,10] - inverse temperature for stage 1 softmax action selection.
    beta2: [0,10] - inverse temperature for stage 2 softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p1: [-5,5] - perseveration bonus added to the previously chosen first-stage action.
    p2: [-5,5] - perseveration bonus added to the previously chosen second-stage action (per state).
    """
    alpha1, alpha2, beta1, beta2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = p1
            
        q_net_pers1 = beta1 * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = p2
            
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta2 * q2 + pers2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with separate learning rates for stage 1 and stage 2,
    and reward-dependent perseveration at stage 1 (different bonus if previous trial was rewarded vs unrewarded).
    
    Parameters:
    alpha1: [0,1] - learning rate for stage 1 value updates.
    alpha2: [0,1] - learning rate for stage 2 value updates.
    beta: [0,10] - inverse temperature for softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p_win: [-5,5] - perseveration bonus added to the previously chosen first-stage action if the previous trial was rewarded.
    p_loss: [-5,5] - perseveration bonus added to the previously chosen first-stage action if the previous trial was unrewarded.
    """
    alpha1, alpha2, beta, w, lam, p_win, p_loss = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                pers1[last_action_1] = p_win
            else:
                pers1[last_action_1] = p_loss
            
        q_net_pers1 = beta * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta * q2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
