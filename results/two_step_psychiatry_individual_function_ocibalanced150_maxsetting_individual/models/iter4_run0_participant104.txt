Here are 3 new cognitive models that introduce novel mechanisms and parameter combinations to explain the participant's decision-making behavior.

### Model 1: Asymmetric Learning Rates with Dual Perseveration
This model tests whether the participant learns differently from positive versus negative prediction errors (asymmetric learning) while also exhibiting perseveration (tendency to repeat choices) at both stages of the task.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates with Dual Perseveration
    Separate learning rates for positive and negative prediction errors,
    along with perseveration bonuses for both Stage 1 and Stage 2 choices.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Mixing weight for MB vs MF values.
    lam: [0, 1] - Eligibility trace.
    rep1: [-5, 5] - Perseveration bonus for the previous stage 1 choice.
    rep2: [-5, 5] - Perseveration bonus for the previous stage 2 choice (state-specific).
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, rep1, rep2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if last_action_1 != -1:
            logits_1[last_action_1] += rep1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += rep2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_s1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning Rates with Dual Perseveration
This model posits that learning occurs at a different speed for the spaceship choice compared to the alien choice (different temporal distances to the reward), while maintaining Stage 1 and Stage 2 perseveration components.

```python

    Model 2: Stage-Specific Learning Rates with Dual Perseveration
    Separate learning rates for Stage 1 and Stage 2 updates,
    along with perseveration bonuses for both Stage 1 and Stage 2 choices.
    
    Parameters:

    alpha2: [0, 1] - Learning rate for Stage 2.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
```

### Model 3: Continuous Choice Traces for Dual Perseveration
Instead of applying a binary perseveration bonus based solely on the immediate previous trial, this model keeps a decaying, continuous "choice trace". This accounts for perseveration that can build up over several repeated choices (habit-like behavior).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Continuous Choice Traces for Dual Perseveration
    Maintains decaying choice traces for Stage 1 and Stage 2 actions instead of
    just the immediate previous choice, allowing perseveration to accumulate over repeated choices.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Mixing weight for MB vs MF values.
    lam: [0, 1] - Eligibility trace.
    tau: [0, 1] - Decay rate of the choice traces (0 = no trace retained, 1 = no decay).
    rho1: [-5, 5] - Weight of the choice trace on Stage 1 logits.
    rho2: [-5, 5] - Weight of the choice trace on Stage 2 logits.
    """
    alpha, beta1, beta2, w, lam, tau, rho1, rho2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net + rho1 * choice_trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx].copy() + rho2 * choice_trace_2[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # update continuous choice traces
        choice_trace_1 = choice_trace_1 * tau
        choice_trace_1[action_1[trial]] += 1.0
        
        choice_trace_2[state_idx] = choice_trace_2[state_idx] * tau
        choice_trace_2[state_idx, action_2[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
