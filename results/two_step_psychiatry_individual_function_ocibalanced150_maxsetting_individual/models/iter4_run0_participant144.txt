Here are 3 new cognitive models that explore different mechanisms to explain the strong perseveration and learning patterns observed in the participant's data.

### Model 1: Hybrid Model with Eligibility Trace and Stickiness
This model builds on the classic Daw et al. (2011) two-step task model. It incorporates an eligibility trace (`eligibility` or $\lambda$), which allows the Stage 2 reward prediction error to directly update the Stage 1 action value. This helps the agent learn faster across stages. It also uses stage-specific stickiness parameters to account for the participant's tendency to repeat choices.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Agent with Eligibility Trace and Stage-Specific Stickiness.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    eligibility: [0, 1] - Eligibility trace (lambda) for updating Stage 1 values from Stage 2 prediction errors.
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    learning_rate, beta, w, eligibility, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q_stage1_net.copy()
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        exp_q1 = np.exp(beta * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += eligibility * learning_rate * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid Model with Asymmetric Learning Rates
Since the participant forms long streaks of identical Stage 1 choices regardless of minor reward fluctuations, they might process positive and negative feedback differently. This model separates the learning rate into `lr_pos` (for better-than-expected outcomes) and `lr_neg` (for worse-than-expected outcomes) to capture asymmetric value updating.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Agent with Asymmetric Learning Rates and Stage-Specific Stickiness.
    
    Parameters:
    lr_pos: [0, 1] - Learning rate for positive prediction errors.
    lr_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    lr_pos, lr_neg, beta, w, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q_stage1_net.copy()
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        exp_q1 = np.exp(beta * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid Model with Choice Autocorrelation Trace
In the data, the participant shows very long stretches of repeating the same Stage 1 action (e.g. choosing spaceship 1 for 20+ trials). Standard stickiness only considers the single immediate preceding trial. This model introduces a running `choice_trace` that accumulates over time, capturing deeper, long-term perseveration patterns.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Agent with Choice Autocorrelation Trace.
    Maintains a running trace of past Stage 1 choices to model longer-term perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    alpha_c: [0, 1] - Update rate for the Stage 1 choice trace.
    stickiness_c: [-5, 5] - Weight of the Stage 1 choice trace added to Stage 1 Q-values.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    learning_rate, beta, w, alpha_c, stickiness_c, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add the choice autocorrelation trace to the Stage 1 values
        q1_eff = q_stage1_net + stickiness_c * choice_trace
            
        exp_q1 = np.exp(beta * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # update choice trace (increment chosen action, decay unchosen)
        choice_trace *= (1 - alpha_c)
        choice_trace[action_1[trial]] += alpha_c
        
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
