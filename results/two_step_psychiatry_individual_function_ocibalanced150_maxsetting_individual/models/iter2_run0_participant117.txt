Here are three new cognitive models that explore alternative mechanisms for participant behavior in the two-step task. 

### Model 1: Choice Trace Model
This model extends the concept of choice stickiness by using an exponentially weighted moving average (EWMA) of past choices (a "choice trace") rather than just the single most recent choice. This captures longer-term perseveration or alternation tendencies, which is useful for participants who exhibit extended streaks of selecting the same spaceship.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice Trace Model
    Maintains an exponentially decaying choice trace for both stages to capture 
    long-term perseveration or alternation, rather than just 1-trial stickiness.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for Stage 1
    beta2: [0,10] - Inverse temperature for Stage 2
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    tau: [0,1] - Decay rate of the choice trace (0 = only previous choice matters, 1 = never updates)
    c1: [-5,5] - Weight of the choice trace for Stage 1
    c2: [-5,5] - Weight of the choice trace for Stage 2
    """
    alpha, beta1, beta2, lam, w, tau, c1, c2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    trace1 = np.zeros(2)
    trace2 = np.zeros(2)

    for trial in range(n_trials):
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net + c1 * trace1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2
        logits_2 = beta2 * q_stage2_mf[state_idx] + c2 * trace2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Choice Trace Updates (EWMA)
        trace1 *= tau
        trace1[action_1[trial]] += (1.0 - tau)
        
        trace2 *= tau
        trace2[action_2[trial]] += (1.0 - tau)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Stickiness
Participants often weigh positive prediction errors (unexpected rewards) differently than negative prediction errors (unexpected omissions). This model incorporates separate learning rates for positive and negative updates, combined with 1-trial choice stickiness.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates with Stickiness
    Uses separate learning rates for positive and negative prediction errors,
    along with choice stickiness for both stages.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    beta1: [0,10] - Inverse temperature for Stage 1
    beta2: [0,10] - Inverse temperature for Stage 2
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Choice stickiness for Stage 1
    p2: [-5,5] - Choice stickiness for Stage 2
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2
        rep2 = np.zeros(2)
        if prev_a2 != -1:
            rep2[prev_a2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value Updates with Asymmetric Learning Rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        alpha_lam = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_lam * lam * delta_stage2

        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Learned Transitions Model
Instead of assuming the participant inherently knows and trusts the 0.7/0.3 transition probabilities, this model treats the transition probabilities as something the participant learns dynamically over time through their own experience.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Learned Transitions Model
    Participants might not assume fixed 0.7/0.3 transition probabilities, but instead 
    dynamically learn the transition matrix over time based on observed outcomes.
    
    Bounds:
    alpha: [0,1] - Learning rate for action values
    alpha_trans: [0,1] - Learning rate for transition probabilities
    beta1: [0,10] - Inverse temperature for Stage 1
    beta2: [0,10] - Inverse temperature for Stage 2
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Choice stickiness for Stage 1
    p2: [-5,5] - Choice stickiness for Stage 2
    """
    alpha, alpha_trans, beta1, beta2, lam, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix as uniform (0.5)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        # Stage 1
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2
        rep2 = np.zeros(2)
        if prev_a2 != -1:
            rep2[prev_a2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Transition matrix update
        # Observed transition: action_1[trial] -> state_idx
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
