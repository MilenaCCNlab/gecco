def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Transition-Dependent Perseveration.
    Choice perseveration differs based on whether the previous transition was common or rare.
    Includes forgetting (decay) for unchosen Q-values.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi_com: [-5, 5] (Perseveration after common transition)
    pi_rare: [-5, 5] (Perseveration after rare transition)
    decay: [0, 1] (Forgetting rate for unchosen actions)
    """
    alpha, beta1, beta2, w, lam, pi_com, pi_rare, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_com = False

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = pi_com if last_trans_com else pi_rare
            
        logits_1 = beta1 * q_net + persev
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action = action_1[trial]
        last_trans_com = (state_idx == action_1[trial])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with Reward-Dependent Eligibility Trace and Decay.
    The eligibility trace (lambda) depends on whether the second-stage outcome was rewarded.
    Unchosen Q-values decay towards zero.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam_win: [0, 1] (Eligibility trace when rewarded)
    lam_lose: [0, 1] (Eligibility trace when unrewarded)
    pi: [-5, 5] (Choice perseveration)
    decay: [0, 1] (Forgetting rate for unchosen actions)
    """
    alpha, beta1, beta2, w, lam_win, lam_lose, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1
            
        logits_1 = beta1 * q_net + pi * persev
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        lam_current = lam_win if reward[trial] > 0 else lam_lose
        q_stage1_mf[action_1[trial]] += alpha * lam_current * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with Transition-Dependent Learning Rates and Decay.
    The learning rate (alpha) depends on whether the transition from stage 1 to stage 2 was common or rare.
    Includes decay for unchosen options.
    
    Bounds:
    alpha_com: [0, 1] (Learning rate after common transitions)
    alpha_rare: [0, 1] (Learning rate after rare transitions)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi: [-5, 5] (Choice perseveration)
    decay: [0, 1] (Forgetting rate for unchosen actions)
    """
    alpha_com, alpha_rare, beta1, beta2, w, lam, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        alpha_current = alpha_com if (state_idx == action_1[trial]) else alpha_rare

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_current * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_current * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_current * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
