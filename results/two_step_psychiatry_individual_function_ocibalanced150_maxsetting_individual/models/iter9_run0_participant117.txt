import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates with Decay.
    Features separate learning rates for positive and negative reward prediction errors,
    combined with decay for unchosen options. Participants might update values differently
    when receiving a reward versus not receiving one.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    decay: [0,1] - Decay rate for unchosen options
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Stage 1 stickiness (perseveration)
    """
    alpha_pos, alpha_neg, decay, beta1, beta2, lam, w, p1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1

        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2

        # Decay unchosen options
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        prev_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-dependent Model-based Weight with Decay.
    The balance between model-based and model-free control (w) depends on whether
    the previous trial was rewarded or unrewarded. Also includes decay for unchosen values.
    
    Bounds:
    alpha: [0,1] - Learning rate
    decay: [0,1] - Decay rate for unchosen options
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    w_rew: [0,1] - Model-based vs model-free weight following a rewarded trial
    w_unrew: [0,1] - Model-based vs model-free weight following an unrewarded trial
    p1: [-5,5] - Stage 1 stickiness
    """
    alpha, decay, beta1, beta2, lam, w_rew, w_unrew, p1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_rew = False

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        w = w_rew if prev_rew else w_unrew
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1

        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay unchosen options
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        prev_a1 = action_1[trial]
        prev_rew = (reward[trial] == 1)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-dependent Stage 1 Stickiness with Stage 2 Stickiness.
    Stage 1 stickiness depends on whether the previous transition was common or rare,
    which builds upon the best model so far but additionally includes a stickiness parameter 
    for Stage 2 choices (assuming the participant lands on the same planet).
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p_common: [-5,5] - Stage 1 stickiness after a common transition
    p_rare: [-5,5] - Stage 1 stickiness after a rare transition
    p2: [-5,5] - Stage 2 stickiness (if arriving at the same state)
    """
    alpha, beta1, beta2, lam, w, p_common, p_rare, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1
    prev_state = -1
    prev_trans_common = True

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1

        p1 = p_common if prev_trans_common else p_rare
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice
        rep2 = np.zeros(2)
        if prev_state == state_idx and prev_a2 != -1:
            rep2[prev_a2] = 1

        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]
        prev_state = state_idx
        prev_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
