def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with decay (forgetting) of Q-values for unchosen actions.
    This model assumes that the values of actions not chosen (or states not visited) 
    decay over time towards a neutral value (0.5), reflecting memory decay.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    decay: [0,1] - decay rate pulling unchosen Q-values toward 0.5
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter (lambda) linking stage 2 reward to stage 1
    """
    learning_rate, decay, beta, w, p, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    last_action = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += p
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
