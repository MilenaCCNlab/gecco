def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates and inverse temperatures for Stage 1 and Stage 2.
    This model allows for different learning speeds and choice stochasticity at each stage.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Model-Free Q-values.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-values.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Model-based weight (1 = purely MB, 0 = purely MF).
    lam: [0, 1] - Eligibility trace for Stage 2 reward updating Stage 1 Q-value.
    rep: [-5, 5] - Perseveration/stickiness for Stage 1 choices.
    """
    alpha1, alpha2, beta1, beta2, w, lam, rep = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta1 * q_stage1_net + rep * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with reward-dependent perseveration (win-stay/lose-shift) and separate betas.
    This model explicitly captures how choice stickiness at Stage 1 might differ based on whether 
    the previous trial was rewarded or not, combined with distinct choice stochasticity at each stage.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    rep_win: [-5, 5] - Perseveration for Stage 1 if previous trial was rewarded.
    rep_lose: [-5, 5] - Perseveration for Stage 1 if previous trial was unrewarded.
    """
    alpha, beta1, beta2, w, lam, rep_win, rep_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        rep = rep_win if last_reward > 0 else rep_lose
        
        logits_1 = beta1 * q_stage1_net + rep * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition probability learning and separate betas for Stage 1 and Stage 2.
    Instead of assuming a fixed 70/30 transition matrix, the model learns the transition 
    probabilities dynamically based on observed state transitions.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    rep: [-5, 5] - Perseveration/stickiness for Stage 1 choices.
    """
    alpha, alpha_trans, beta1, beta2, w, lam, rep = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix assuming no prior knowledge (50/50 chance)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta1 * q_stage1_net + rep * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update transition matrix probabilities
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
