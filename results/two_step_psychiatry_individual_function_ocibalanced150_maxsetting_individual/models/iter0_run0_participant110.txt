def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based and Model-Free learning (Daw et al., 2011).
    This model computes a weighted average of model-based and model-free values 
    for the first stage choice, updates values via prediction errors, includes an 
    eligibility trace for stage 2 rewards to update stage 1, and models choice perseveration.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta1: inverse temperature for stage 1 [0, 10]
    beta2: inverse temperature for stage 2 [0, 10]
    w: model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lambda_: eligibility trace [0, 1]
    p: perseveration bonus for repeating the previous stage 1 choice [0, 5]
    """
    alpha, beta1, beta2, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        # Handle missing data
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
        q_net_scaled = beta1 * q_net + p * pers
        q_net_scaled -= np.max(q_net_scaled) # numerical stability
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        q2_scaled = beta2 * q_stage2_mf[s]
        q2_scaled -= np.max(q2_scaled) # numerical stability
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF with asymmetric learning rates.
    This model assumes that individuals learn differently from positive and negative 
    prediction errors (PEs), a common finding in reinforcement learning tasks.
    
    Parameters:
    alpha_pos: learning rate for positive PEs [0, 1]
    alpha_neg: learning rate for negative PEs [0, 1]
    beta1: inverse temperature for stage 1 [0, 10]
    beta2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Handle missing data
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_scaled = beta1 * q_net
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        q2_scaled = beta2 * q_stage2_mf[s]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha_s2 * delta_stage2
        
        # eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha_s2 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF with decay (forgetting) of unchosen action values.
    Since reward probabilities slowly drift over time, unchosen options might 
    be forgotten, drifting back towards initial values (zero).
    
    Parameters:
    alpha: learning rate [0, 1]
    beta1: inverse temperature for stage 1 [0, 10]
    beta2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    decay_rate: rate at which unchosen values decay toward 0 [0, 1]
    """
    alpha, beta1, beta2, w, decay_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Handle missing data
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_scaled = beta1 * q_net
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        q2_scaled = beta2 * q_stage2_mf[s]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Apply decay to unchosen actions
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay_rate)
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] *= (1 - decay_rate)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
