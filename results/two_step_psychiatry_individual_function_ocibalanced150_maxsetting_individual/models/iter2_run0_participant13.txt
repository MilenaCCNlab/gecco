def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Average-based Model-Based Planning.
    Instead of assuming the participant calculates the maximum value of the second stage
    options (max_q_stage2), this model assumes they calculate the average value of the 
    second stage options. This reflects a less optimal, heuristic model-based planning 
    where the participant does not fully maximize at the second stage during planning.
    
    Bounds:
    alpha: [0,1] (Learning rate)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    lam: [0,1] (Eligibility trace parameter lambda)
    pi: [-5,5] (Choice stickiness/perseveration)
    """
    alpha, beta_1, beta_2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        # Average instead of max for model-based values
        avg_q_stage2 = np.mean(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ avg_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if trial > 0:
            rep[action_1[trial-1]] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Learned Transition Matrix Model.
    This model learns the transition probabilities from experience using a dedicated 
    transition learning rate, rather than assuming fixed 0.7/0.3 probabilities.
    
    Bounds:
    alpha: [0,1] (Learning rate for Q-values)
    alpha_tr: [0,1] (Learning rate for transition probabilities)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    lam: [0,1] (Eligibility trace parameter lambda)
    pi: [-5,5] (Choice stickiness/perseveration)
    """
    alpha, alpha_tr, beta_1, beta_2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    # Initialize with instructed probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if trial > 0:
            rep[action_1[trial-1]] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update transition matrix based on observed state transition
        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_tr * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_tr * (0.0 - transition_matrix[a1, 1 - s])
        # Re-normalize to ensure it remains a valid probability distribution
        transition_matrix[a1] = transition_matrix[a1] / np.sum(transition_matrix[a1])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Direct Model-Free Update with Decay.
    The Stage 1 model-free system updates directly based on the final reward (TD(1)), 
    ignoring the intermediate Stage 2 state value. Unchosen options decay towards zero.
    
    Bounds:
    alpha: [0,1] (Learning rate)
    decay: [0,1] (Decay rate for unchosen options)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    pi: [-5,5] (Choice stickiness/perseveration)
    """
    alpha, decay, beta_1, beta_2, w, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if trial > 0:
            rep[action_1[trial-1]] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Direct MF update for Stage 1 using final reward
        q_stage1_mf[action_1[trial]] += alpha * (reward[trial] - q_stage1_mf[action_1[trial]])
        
        # Standard update for Stage 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
