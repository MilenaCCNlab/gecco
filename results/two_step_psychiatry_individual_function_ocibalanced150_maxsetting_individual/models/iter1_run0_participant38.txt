def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw 2011 6-parameter model with separate inverse temperatures for stage 1 and stage 2.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax choice.
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax choice.
    w: [0, 1] - Weight given to model-based Q-values for stage 1 (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for stage 1 MF update.
    stickiness: [-5, 5] - Choice perseveration bonus for repeating the previous stage 1 action.
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            stickiness_bonus[last_action] = stickiness
            
        logits_1 = beta_1 * q_stage1_net + stickiness_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]


        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action = action_1[trial]
            continue
            
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Mixture of Model-Based and Model-Free learning with choice stickiness for both stages.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weight given to model-based Q-values for stage 1 (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for stage 1 MF update.
    stickiness_1: [-5, 5] - Choice perseveration bonus for repeating the previous stage 1 action.
    stickiness_2: [-5, 5] - Choice perseveration bonus for repeating the previous stage 2 action in the same state.
    """
    learning_rate, beta, w, lam, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_stage1_net + stickiness_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)

        state_idx = state[trial]

            last_action_1 = action_1[trial]
            continue
            
        stickiness_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_bonus_2[last_action_2[state_idx]] = stickiness_2

        logits_2 = beta * q_stage2_mf[state_idx] + stickiness_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
