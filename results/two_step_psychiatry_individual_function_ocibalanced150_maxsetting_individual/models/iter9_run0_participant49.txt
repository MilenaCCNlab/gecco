Here are 3 new cognitive models based on the participant's data and the feedback provided. They explore novel mechanisms that haven't been tested yet, including choice kernels (exponential moving averages of past actions), dynamic transition matrix learning, and asymmetric eligibility traces for positive versus negative rewards.

### Model 1: Choice Kernels for Stage 1 and Stage 2
Instead of a simple 1-trial perseveration, this model implements a **Choice Kernel** at both stages. A choice kernel tracks the exponential moving average of past choices, capturing a longer-term habit or tendency to repeat choices that decays over time.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-learning with Choice Kernels for both Stage 1 and Stage 2.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta_1: [0,10] - Inverse temperature for stage 1 action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter.
    alpha_c: [0,1] - Update rate for the choice kernels (how fast past choices are forgotten).
    theta_1: [-5,5] - Weight of the choice kernel at stage 1 (positive = perseveration).
    theta_2: [-5,5] - Weight of the choice kernel at stage 2.
    """
    learning_rate, beta_1, beta_2, w, lam, alpha_c, theta_1, theta_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Initialize choice kernels
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + theta_1 * ck_1
        logits_1 -= np.max(logits_1) # For numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        logits_2 = beta_2 * q_stage2_mf[state_idx] + theta_2 * ck_2[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Update Choice Kernels
        ck_1 *= (1 - alpha_c)
        ck_1[action_1[trial]] += alpha_c
        
        ck_2 *= (1 - alpha_c)
        ck_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Value Decay + Dynamic Transition Matrix Learning
Building on the best model so far, this model introduces **dynamic learning of the transition matrix**. Instead of assuming the participant holds a fixed 70/30 internal model of the task's transitions throughout, this model allows the participant to update their internal transition probabilities over time based on the actual planets they land on.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Q-learning with decay, 1-trial perseveration, and dynamic transition matrix learning.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-values.
    decay: [0,1] - Decay rate for all Q-values towards zero.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight for stage 1.
    lam: [0,1] - Eligibility trace.
    pers: [-5,5] - Perseveration bonus for repeating the previous stage 1 choice.
    alpha_trans: [0,1] - Learning rate for updating the internal transition matrix.
    """
    learning_rate, decay, beta_1, beta_2, w, lam, pers, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Start with the standard instructed transition probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        logits_1 = beta_1 * (q_stage1_net + pers_bonus)
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Decay all Q-values towards zero
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Update Transition Matrix (Dynamic Learning)
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Eligibility Traces for Reward vs. Non-Reward
This model proposes that participants might propagate information back to Stage 1 differently depending on whether they won or lost. It splits the eligibility trace `lam` into `lam_pos` (applied when a gold coin is found) and `lam_neg` (applied when 0 coins are found).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Q-learning with decay and separate eligibility traces for rewarded vs unrewarded trials.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-values.
    decay: [0,1] - Decay rate for all Q-values.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight for stage 1.
    lam_pos: [0,1] - Eligibility trace scaling when reward > 0 (Success backpropagation).
    lam_neg: [0,1] - Eligibility trace scaling when reward == 0 (Failure backpropagation).
    pers: [-5,5] - Perseveration bonus for repeating stage 1 choice.
    """
    learning_rate, decay, beta_1, beta_2, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        logits_1 = beta_1 * (q_stage1_net + pers_bonus)
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Asymmetric eligibility trace mapping
        lam = lam_pos if reward[trial] > 0 else lam_neg

        # Decay unchosen/all values
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
