def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Hybrid MB/MF RL with exponentially decaying choice traces.
    
    Instead of a simple n-1 stickiness, this model uses a continuous choice trace 
    that decays over time, allowing for longer-term perseveration which might better 
    capture extended streaks in participant behavior.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updating.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 prediction error.
    rho_1: [0,10] - Choice trace weight for stage 1 actions.
    rho_2: [0,10] - Choice trace weight for stage 2 actions.
    alpha_c: [0,1] - Update rate for the choice traces.
    """
    alpha, beta, w, lam, rho_1, rho_2, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 choice probability
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        v1 = beta * q_net + rho_1 * trace_1
        exp_q1 = np.exp(v1 - np.max(v1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice probability
        v2 = beta * q_stage2_mf[state_idx, :] + rho_2 * trace_2[state_idx, :]
        exp_q2 = np.exp(v2 - np.max(v2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Trace updates
        trace_1 = (1 - alpha_c) * trace_1
        trace_1[action_1[trial]] += alpha_c
        
        trace_2[state_idx] = (1 - alpha_c) * trace_2[state_idx]
        trace_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Hybrid MB/MF RL with separate learning rates for stages and Q-value decay.
    
    This model uses a simpler n-1 stickiness representation but introduces separate 
    learning rates for stage 1 and stage 2, combined with a decay mechanism for unchosen 
    options. This helps to account for potentially different speeds of learning in the 
    different stages of the decision tree.
    
    Parameters:
    alpha_1: [0,1] - Learning rate for stage 1 Q-value updating.
    alpha_2: [0,1] - Learning rate for stage 2 Q-value updating.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 prediction error.
    rho_1: [0,10] - Choice stickiness parameter for stage 1 actions.
    rho_2: [0,10] - Choice stickiness parameter for stage 2 actions.
    decay: [0,1] - Decay rate for all Q-values towards 0.5 before updating.
    """
    alpha_1, alpha_2, beta, w, lam, rho_1, rho_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_choice_1 = -1
    last_choice_2 = [-1, -1] 

    for trial in range(n_trials):
        # Stage 1 choice probability
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_choice_1 != -1:
            stickiness_1[last_choice_1] = 1.0
            
        v1 = beta * q_net + rho_1 * stickiness_1
        exp_q1 = np.exp(v1 - np.max(v1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice probability
        stickiness_2 = np.zeros(2)
        if last_choice_2[state_idx] != -1:
            stickiness_2[last_choice_2[state_idx]] = 1.0

        v2 = beta * q_stage2_mf[state_idx, :] + rho_2 * stickiness_2
        exp_q2 = np.exp(v2 - np.max(v2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen values
        q_stage1_mf = (1 - decay) * q_stage1_mf + decay * 0.5
        q_stage2_mf = (1 - decay) * q_stage2_mf + decay * 0.5

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        # Update stickiness trackers
        last_choice_1 = action_1[trial]
        last_choice_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
