def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based / Model-Free (Daw et al. 2011)
    This model captures the mixture of model-based and model-free control, 
    which is essential for the two-step task. It also includes an eligibility 
    trace for multi-step credit assignment and a perseverance (stay bias) parameter 
    to capture the participant's tendency to repeat choices regardless of reward.
    
    Parameters:
    alpha: learning rate [0,1]
    beta: inverse temperature [0,10]
    lam: eligibility trace [0,1]
    w: model-based weight [0,1]
    p: perseverance (stay bias) [-5,5]
    """
    alpha, beta, lam, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Mix Model-Based and Model-Free values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add stay bias for the previous choice
        stay_bias = np.zeros(2)
        if last_action != -1:
            stay_bias[last_action] = 1
            
        logits_1 = beta * q_net + p * stay_bias
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for Choice 1 based on Stage 2 prediction error
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Asymmetric Learning Rates.
    Participants in psychiatric datasets often exhibit differences in how they 
    learn from positive versus negative feedback. This model separates the 
    learning rate into alpha_pos (for positive RPEs) and alpha_neg (for negative RPEs).
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    """
    alpha_pos, alpha_neg, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta * q_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Separate Temperatures.
    Decision noise often varies between the first stage (more abstract/planning) 
    and the second stage (immediate reward evaluation). This model uses separate 
    inverse temperatures for stage 1 and stage 2 choices.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    """
    alpha, beta1, beta2, w, lam = model_parameters
    n_trials = len(action_1)

    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice

        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for Choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
