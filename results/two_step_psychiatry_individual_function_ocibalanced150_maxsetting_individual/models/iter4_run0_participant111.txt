import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with positive and negative learning rates (shared across stages) 
    and reward-dependent choice stickiness (Win-Stay/Lose-Shift) at stage 1.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p_win: [-5, 5] - stage 1 stickiness if previous trial was rewarded
    p_lose: [-5, 5] - stage 1 stickiness if previous trial was unrewarded
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1

        current_p = p_win if last_reward > 0 else p_lose
            
        q1 = beta1 * q_net + current_p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1_current = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha2_current = alpha_pos if delta_stage2 > 0 else alpha_neg

        q_stage1_mf[action_1[trial]] += alpha1_current * delta_stage1 + alpha1_current * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_current * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with reward-dependent model-based weight and reward-dependent stickiness.
    Shares a single learning rate across both stages to maintain parsimony.
    
    Bounds:
    alpha: [0, 1] - learning rate for both stages
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w_win: [0, 1] - model-based weight if previous trial was rewarded
    w_lose: [0, 1] - model-based weight if previous trial was unrewarded
    p_win: [-5, 5] - stage 1 stickiness if previous trial was rewarded
    p_lose: [-5, 5] - stage 1 stickiness if previous trial was unrewarded
    """
    alpha, beta1, beta2, lam, w_win, w_lose, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        current_w = w_win if last_reward > 0 else w_lose
        q_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1

        current_p = p_win if last_reward > 0 else p_lose
            
        q1 = beta1 * q_net + current_p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)

        state_idx = state[trial]

        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with reward-dependent stage 1 stickiness and a stage 2 choice stickiness.
    Uses a single learning rate for both stages to keep parameter count at 8.
    
    Bounds:
    alpha: [0, 1] - learning rate for both stages
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p_win: [-5, 5] - stage 1 stickiness if previous trial was rewarded
    p_lose: [-5, 5] - stage 1 stickiness if previous trial was unrewarded
    p2: [-5, 5] - stage 2 stickiness (tendency to repeat stage 2 action in the same state)
    """
    alpha, beta1, beta2, lam, w, p_win, p_lose, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0
    last_action_2 = [-1, -1] 

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep1 = np.zeros(2)
        if last_action_1 != -1:
            rep1[last_action_1] = 1

        current_p = p_win if last_reward > 0 else p_lose
            
        q1 = beta1 * q_net + current_p * rep1
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        rep2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            rep2[last_action_2[state_idx]] = 1

        q2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
