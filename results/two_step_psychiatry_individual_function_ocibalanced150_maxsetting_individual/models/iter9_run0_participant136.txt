Here are 3 new cognitive models, each proposing a unique variation on the hybrid Model-Based/Model-Free learning framework with decay, introducing different mechanisms that participants might use to handle the transition structure and choice stickiness in the two-step task.

### Model 1: Transition-dependent Eligibility Trace
Participants may trust and back-propagate rewards differently depending on whether the transition to the planet was expected (common) or unexpected (rare). This model splits the eligibility trace parameter ($\lambda$) into two separate parameters based on the transition type.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition-dependent Eligibility Trace.
    Eligibility trace (lambda) differs depending on whether the transition from stage 1 to stage 2 was common or rare.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weight on model-based vs model-free values (1 = purely MB, 0 = purely MF).
    lam_common: [0, 1] - Eligibility trace for common transitions.
    lam_rare: [0, 1] - Eligibility trace for rare transitions.
    p: [-5, 5] - Choice perseveration (stickiness) parameter for stage 1.
    """
    learning_rate, decay_rate, beta, w, lam_common, lam_rare, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # Determine if transition was common or rare
        # Spaceship 0 commonly goes to Planet 0, Spaceship 1 commonly goes to Planet 1
        is_common = (action_1[trial] == state_idx)
        current_lam = lam_common if is_common else lam_rare

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * current_lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2

        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition-dependent Learning Rate
Similar to Model 1, but posits that the fundamental learning rate (rather than just the eligibility trace) differs depending on whether a transition was common or rare. A rare transition might be perceived as a less reliable signal, thus carrying a different weight in value updating.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition-dependent Learning Rate.
    The learning rate for updating Q-values differs depending on whether the transition from stage 1 to stage 2 was common or rare.
    
    Parameters:
    learning_rate_common: [0, 1] - Learning rate for common transitions.
    learning_rate_rare: [0, 1] - Learning rate for rare transitions.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weight on model-based vs model-free values (1 = purely MB, 0 = purely MF).
    lam: [0, 1] - Eligibility trace decay parameter (lambda).
    p: [-5, 5] - Choice perseveration (stickiness) parameter for stage 1.
    """
    learning_rate_common, learning_rate_rare, decay_rate, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)

            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        is_common = (action_1[trial] == state_idx)
        current_lr = learning_rate_common if is_common else learning_rate_rare

        new_q1_chosen = q_stage1_mf[action_1[trial]] + current_lr * delta_stage1 + current_lr * lam * delta_stage2

        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Choice and State Perseveration
In addition to standard choice stickiness, participants might exhibit "state stickiness"—a tendency to try and return to the planet they just visited, regardless of the explicit reward outcome. This model includes a parameter that bumps the probability of choosing the stage 1 action that commonly leads to the previously visited stage 2 state.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Choice and State Perseveration.
    Includes standard choice perseveration (sticking to the last stage 1 action) 
    and state perseveration (choosing the stage 1 action that commonly leads to the last visited stage 2 state).
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay_rate: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace decay parameter (lambda).
    p_choice: [-5, 5] - Choice perseveration parameter for stage 1.
    p_state: [-5, 5] - State perseveration parameter (tendency to try to return to the last stage 2 state).
    """
    learning_rate, decay_rate, beta, w, lam, p_choice, p_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_choice = np.zeros(2)
        if last_action_1 != -1:
            pers_choice[last_action_1] = 1.0
            
        pers_state = np.zeros(2)
        if last_state != -1:
            # Action 0 commonly leads to State 0, Action 1 to State 1
            pers_state[last_state] = 1.0
            
        exp_q1 = np.exp(beta * q_net + p_choice * pers_choice + p_state * pers_state)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2

        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
