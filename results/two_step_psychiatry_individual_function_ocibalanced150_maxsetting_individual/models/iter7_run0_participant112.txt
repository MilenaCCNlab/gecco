Here are 3 new cognitive models, each proposing a distinct psychological mechanism to explain the participant's choices, particularly focusing on how they integrate past rewards, transitions, and stage-specific behaviors.

### Model 1: Stage-Specific Stickiness
**Rationale:** The participant exhibits strong win-stay/lose-shift behavior, but this heuristic may primarily apply to the first-stage choice (the spaceship) which dictates the overall strategy. The second-stage choice (the alien) might simply be driven by a general tendency to repeat the last action on that planet (constant stickiness), rather than being strongly modulated by win/lose states. This model separates win/lose stickiness for Stage 1 and uses a constant stickiness for Stage 2.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Stage-specific stickiness.
    Win/lose stickiness is applied only to the first-stage choice, which often shows 
    stronger win-stay/lose-shift behavior. The second stage has a simple constant stickiness.
    
    Bounds:
    learning_rate: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness_win_1: [-5,5]
    stickiness_lose_1: [-5,5]
    stickiness_2: [-5,5]
    """
    import numpy as np
    learning_rate, beta_1, beta_2, w, lam, stickiness_win_1, stickiness_lose_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            stick_1 = stickiness_win_1 if last_reward == 1 else stickiness_lose_1
            q_stage1_net[last_action_1] += stick_1
            
        exp_q1 = np.exp(beta_1 * q_stage1_net - np.max(beta_1 * q_stage1_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1 and last_state == state_idx:
            q_stage2_net[last_action_2] += stickiness_2

        exp_q2 = np.exp(beta_2 * q_stage2_net - np.max(beta_2 * q_stage2_net))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Model-Based Weighting
**Rationale:** The balance between model-based (MB) and model-free (MF) control might not be static. Participants may rely more heavily on MB planning after experiencing a loss (to deliberately find a better alternative) or after a win (to carefully exploit the known transition structure). This model allows the weighting parameter `w` to change depending on whether the previous trial was rewarded or unrewarded.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-dependent Model-Based weight (w).
    The balance between model-based and model-free control (w) depends on 
    whether the previous trial was rewarded. Participants might shift their strategy 
    dynamically in response to positive or negative feedback.
    
    Bounds:
    learning_rate: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w_win: [0,1]
    w_lose: [0,1]
    lam: [0,1]
    stickiness: [-5,5]
    """
    import numpy as np
    learning_rate, beta_1, beta_2, w_win, w_lose, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine w based on last reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = 0.5 # default for the very first trial
            
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness
            
        exp_q1 = np.exp(beta_1 * q_stage1_net - np.max(beta_1 * q_stage1_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1 and last_state == state_idx:
            q_stage2_net[last_action_2] += stickiness

        exp_q2 = np.exp(beta_2 * q_stage2_net - np.max(beta_2 * q_stage2_net))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Lose-Shift
**Rationale:** A hallmark of model-based planning is reacting differently to unrewarded outcomes depending on the state transition. If a participant makes a choice, experiences a *rare* transition, and loses, they might actually be *more* likely to repeat their first-stage choice (since it usually leads to the other, potentially better state). This model splits the `stickiness_lose` parameter based on whether the preceding transition was common or rare.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-dependent Lose-Shift.
    Win-stay is modeled with a single parameter, but lose-shift is split into 
    common vs. rare transitions. This allows the model to capture model-based 
    strategies directly in the choice autocorrelation.
    
    Bounds:
    learning_rate: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness_win: [-5,5]
    stickiness_lose_common: [-5,5]
    stickiness_lose_rare: [-5,5]
    """
    import numpy as np
    learning_rate, beta_1, beta_2, w, lam, stickiness_win, stickiness_lose_common, stickiness_lose_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness = stickiness_win
            else:
                # Action 0 commonly leads to State 0; Action 1 commonly leads to State 1.
                is_common = (last_action_1 == last_state)
                stickiness = stickiness_lose_common if is_common else stickiness_lose_rare
            q_stage1_net[last_action_1] += stickiness
            
        exp_q1 = np.exp(beta_1 * q_stage1_net - np.max(beta_1 * q_stage1_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1 and last_state == state_idx:
            if last_reward == 1:
                stickiness2 = stickiness_win
            else:
                is_common = (last_action_1 == last_state)
                stickiness2 = stickiness_lose_common if is_common else stickiness_lose_rare
            q_stage2_net[last_action_2] += stickiness2

        exp_q2 = np.exp(beta_2 * q_stage2_net - np.max(beta_2 * q_stage2_net))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
