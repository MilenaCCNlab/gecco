def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for Stage 1 and Stage 2 with unchosen option decay.
    This model allows for different learning rates at each stage, while keeping decay and stage 1 stickiness.
    
    Bounds:
    alpha1: [0, 1] (learning rate for stage 1)
    alpha2: [0, 1] (learning rate for stage 2)
    decay: [0, 1] (decay rate for unchosen options)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (weight of model-based vs model-free control)
    lam: [0, 1] (eligibility trace parameter)
    pi1: [-5, 5] (choice stickiness/perseveration for stage 1)
    """
    alpha1, alpha2, decay, beta1, beta2, w, lam, pi1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pi1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        last_a1 = a1

        # Handle missed second stage
        if a2 == -1:
            p_choice_2[trial] = 1.0
            continue

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha1 * lam * delta_stage2

        # Unchosen option decay
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive/negative prediction errors with two-stage stickiness.
    This model tests whether participants learn differently from positive and negative prediction errors, while maintaining stickiness at both stages.
    
    Bounds:
    alpha_pos: [0, 1] (learning rate for positive prediction errors)
    alpha_neg: [0, 1] (learning rate for negative prediction errors)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (weight of model-based vs model-free control)
    lam: [0, 1] (eligibility trace parameter)
    pi1: [-5, 5] (choice stickiness/perseveration for stage 1)
    pi2: [-5, 5] (choice stickiness/perseveration for stage 2)
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pi1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        last_a1 = a1

        # Handle missed second stage
        if a2 == -1:
            p_choice_2[trial] = 1.0
            continue

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            stickiness2[last_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx, :] + pi2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        last_a2[state_idx] = a2

        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[a1] += lr1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += lr2 * delta_stage2

        # Eligibility trace uses stage 1's learning rate logic
        q_stage1_mf[a1] += lr1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Win-stay/lose-shift stickiness for Stage 1 with unchosen option decay.
    This model separates stage 1 stickiness based on whether the previous trial was rewarded or unrewarded.
    
    Bounds:
    learning_rate: [0, 1] (learning rate for updating values)
    decay: [0, 1] (decay rate for unchosen options)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (weight of model-based vs model-free control)
    lam: [0, 1] (eligibility trace parameter)
    pi_win: [-5, 5] (stage 1 stickiness after a rewarded trial)
    pi_lose: [-5, 5] (stage 1 stickiness after an unrewarded trial)
    """
    learning_rate, decay, beta1, beta2, w, lam, pi_win, pi_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        pi_current = 0.0
        if last_reward == 1:
            pi_current = pi_win
        elif last_reward == 0:
            pi_current = pi_lose
            
        logits_1 = beta1 * q_stage1_net + pi_current * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        last_a1 = a1

        # Handle missed second stage
        if a2 == -1:
            p_choice_2[trial] = 1.0
            last_reward = -1
            continue

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        # Unchosen option decay
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
