import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative RPEs, with reward-dependent perseveration.
    
    Parameters:
    alpha_pos: Learning rate for positive RPEs [0, 1]
    alpha_neg: Learning rate for negative RPEs [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    pi_win: Perseveration bonus if previous trial was rewarded [-5, 5]
    pi_lose: Perseveration bonus if previous trial was unrewarded [-5, 5]
    lambd: Eligibility trace [0, 1]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, pi_win, pi_lose, lambd = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_reward = -1
            continue

        state_idx = state[trial]
        act1 = action_1[trial]
        act2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        if last_reward == 1:
            pi = pi_win
        elif last_reward == 0:
            pi = pi_lose
        else:
            pi = 0.0
            
        logits1 = beta1 * (q_stage1_net + pi * rep)
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[act1]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[act2]

        delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[act1] += alpha_s1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, act2]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, act2] += alpha_s2 * delta_stage2

        q_stage1_mf[act1] += alpha_s2 * lambd * delta_stage2

        last_action = act1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Different learning rates for stage 1 and stage 2, transition-dependent perseveration, and unchosen Q-value decay.
    
    Parameters:
    alpha1: Learning rate for stage 1 [0, 1]
    alpha2: Learning rate for stage 2 [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    pi_common: Perseveration bonus if previous transition was common [-5, 5]
    pi_rare: Perseveration bonus if previous transition was rare [-5, 5]
    decay: Decay rate for unchosen options [0, 1]
    """
    alpha1, alpha2, beta1, beta2, w, pi_common, pi_rare, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_common = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_trans_common = -1
            continue

        state_idx = state[trial]
        act1 = action_1[trial]
        act2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        if last_trans_common == 1:
            pi = pi_common
        elif last_trans_common == 0:
            pi = pi_rare
        else:
            pi = 0.0
            
        logits1 = beta1 * (q_stage1_net + pi * rep)
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[act1]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[act2]

        delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
        q_stage1_mf[act1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, act2]
        q_stage2_mf[state_idx, act2] += alpha2 * delta_stage2

        # Unchosen decay
        unchosen_1 = 1 - act1
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - act2
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1 - decay)
        q_stage2_mf[other_state, 1] *= (1 - decay)
        
        last_action = act1
        if (act1 == 0 and state_idx == 0) or (act1 == 1 and state_idx == 1):
            last_trans_common = 1
        else:
            last_trans_common = 0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Choice kernels for both stage 1 and stage 2, combined with unchosen Q-value decay.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    ck_weight1: Choice kernel weight for stage 1 [-5, 5]
    ck_weight2: Choice kernel weight for stage 2 [-5, 5]
    ck_alpha: Choice kernel update rate [0, 1]
    decay: Decay rate for unchosen Q-values [0, 1]
    """
    alpha, beta1, beta2, w, ck_weight1, ck_weight2, ck_alpha, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck1 = np.zeros(2)
    ck2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        state_idx = state[trial]
        act1 = action_1[trial]
        act2 = action_2[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits1 = beta1 * q_stage1_net + ck_weight1 * ck1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[act1]

        logits2 = beta2 * q_stage2_mf[state_idx] + ck_weight2 * ck2[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[act2]

        delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
        q_stage1_mf[act1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, act2]
        q_stage2_mf[state_idx, act2] += alpha * delta_stage2

        # Unchosen decay
        unchosen_1 = 1 - act1
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - act2
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1 - decay)
        q_stage2_mf[other_state, 1] *= (1 - decay)
        
        # Choice kernel update
        ck1 = (1 - ck_alpha) * ck1
        ck1[act1] += ck_alpha
        
        ck2[0] = (1 - ck_alpha) * ck2[0]
        ck2[1] = (1 - ck_alpha) * ck2[1]
        ck2[state_idx, act2] += ck_alpha

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
