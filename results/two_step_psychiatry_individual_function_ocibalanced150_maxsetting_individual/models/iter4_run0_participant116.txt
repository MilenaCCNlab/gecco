def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual Learning Rates for Stage 2
    Uses separate learning rates for the Q-values that feed into the model-based 
    and model-free systems. This hypothesizes that explicit (model-based) and 
    implicit (model-free) value representations update at different speeds.
    
    Bounds:
    lr_mf: [0,1] - Learning rate for model-free Q-values
    lr_mb: [0,1] - Learning rate for model-based Q-values
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness
    """
    lr_mf, lr_mb, beta, w, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    q_stage2_mb = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2_mb = np.max(q_stage2_mb, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2_mb
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr_mf * delta_stage1
        
        # action value updating for choice 2
        delta_stage2_mf = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr_mf * delta_stage2_mf
        
        delta_stage2_mb = reward[trial] - q_stage2_mb[state_idx, action_2[trial]]
        q_stage2_mb[state_idx, action_2[trial]] += lr_mb * delta_stage2_mb
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += lr_mf * lam * delta_stage2_mf
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Eligibility Trace
    The eligibility trace parameter (lambda) differs for positive vs negative 
    Stage 2 prediction errors, allowing credit assignment to Stage 1 to vary 
    based on whether the outcome was better or worse than expected.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w: [0,1] - Model-based weight
    lam_pos: [0,1] - Eligibility trace for positive prediction errors
    lam_neg: [0,1] - Eligibility trace for negative prediction errors
    st: [-5,5] - Choice stickiness
    """
    learning_rate, beta, w, lam_pos, lam_neg, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # asymmetric eligibility trace update
        if delta_stage2 > 0:
            q_stage1_mf[action_1[trial]] += learning_rate * lam_pos * delta_stage2
        else:
            q_stage1_mf[action_1[trial]] += learning_rate * lam_neg * delta_stage2
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent Model-Based Weight
    The reliance on the model-based system (w) depends dynamically on whether 
    the previous trial was rewarded, modeling shifts between goal-directed and 
    habitual behavior following success or failure.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates
    beta: [0,10] - Inverse temperature for softmax action selection
    w_win: [0,1] - Model-based weight after a rewarded trial
    w_lose: [0,1] - Model-based weight after an unrewarded trial
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness
    """
    learning_rate, beta, w_win, w_lose, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # determine current w based on previous reward
        if last_reward == 1:
            current_w = w_win
        elif last_reward == 0:
            current_w = w_lose
        else:
            current_w = (w_win + w_lose) / 2.0
            
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        exp_q1 = np.exp(beta * q_net + st * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
