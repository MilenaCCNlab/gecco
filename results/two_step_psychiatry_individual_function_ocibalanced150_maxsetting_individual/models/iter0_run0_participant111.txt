def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. 2011 Two-Step Task Model.
    This model implements a mixture of model-free and model-based learning,
    eligibility traces, and choice stickiness (perseveration).
    
    Bounds:
    alpha1: [0, 1] - learning rate for stage 1
    alpha2: [0, 1] - learning rate for stage 2
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p: [0, 5] - choice stickiness (perseveration)
    """
    alpha1, alpha2, beta1, beta2, lam, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        q1 = beta1 * q_net + p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Asymmetric Learning Rate Model.
    This model separates learning rates for positive and negative prediction errors,
    which is useful for capturing differential sensitivity to better-than-expected
    versus worse-than-expected outcomes.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    """
    alpha_pos, alpha_neg, beta1, beta2, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q1 = beta1 * q_net
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # ACTION VALUE UPDATING FOR CHOICE 1
        if delta_stage1 > 0:
            q_stage1_mf[action_1[trial]] += alpha_pos * delta_stage1
        else:
            q_stage1_mf[action_1[trial]] += alpha_neg * delta_stage1
            
        # ACTION VALUE UPDATING FOR CHOICE 2
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_pos * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_neg * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Forgetting Q-Learning Model.
    This model incorporates a decay parameter that pulls the values of unchosen
    actions towards zero over time, modeling memory decay and exploration of
    less frequently chosen options.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    decay: [0, 1] - decay rate for unchosen options
    """
    alpha, beta1, beta2, w, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q1 = beta1 * q_net
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
