def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate inverse temperatures for model-based and model-free Q-values at Stage 1.
    Instead of a single inverse temperature and a weighting parameter `w`, this model parameterizes 
    the contribution of the MB and MF systems with independent `beta_mb` and `beta_mf`.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta_mb: inverse temperature for model-based Q-values at stage 1 [0,10]
    beta_mf: inverse temperature for model-free Q-values at stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    """
    alpha1, alpha2, beta_mb, beta_mf, beta2, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF values using separate inverse temperatures
        q_net = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        
        # Add perseveration bonus based on the outcome of the previous trial
        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose
                
        exp_q1 = np.exp(q_net - np.max(q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate eligibility traces for rewarded and unrewarded trials.
    This captures the possibility that the reinforcement of Stage 1 actions by Stage 2 outcomes
    differs depending on whether the outcome was a reward or an omission.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam_win: eligibility trace for rewarded trials [0,1]
    lam_lose: eligibility trace for unrewarded trials [0,1]
    pers: perseveration bonus for stage 1 choice [-5,5]
    """
    alpha1, alpha2, beta1, beta2, w, lam_win, lam_lose, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action >= 0:
            q_net[last_action] += pers
            
        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # policy for the second choice
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # ACTION VALUE UPDATING FOR CHOICE 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Update Stage 1 MF value using outcome-dependent eligibility trace
        current_lam = lam_win if reward[trial] == 1 else lam_lose
        q_stage1_mf[action_1[trial]] += alpha1 * current_lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates for positive and negative prediction errors at Stage 2.
    This allows for differential updating (e.g., optimism vs. pessimism) when expectations are 
    exceeded versus when they are disappointed.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2_pos: learning rate for positive RPEs at stage 2 [0,1]
    alpha2_neg: learning rate for negative RPEs at stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers: perseveration bonus for stage 1 choice [-5,5]
    """
    alpha1, alpha2_pos, alpha2_neg, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha2 = alpha2_pos if delta_stage2 > 0 else alpha2_neg

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
