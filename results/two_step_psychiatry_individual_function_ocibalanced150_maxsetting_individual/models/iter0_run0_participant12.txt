def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Model-Free Reinforcement Learning (SARSA).
    This model assumes choices are driven purely by model-free Q-values, 
    updating based on temporal difference errors.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    """
    learning_rate, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        # OVERRIDE choice 1 policy to be pure Model-Free
        exp_q1_mf = np.exp(beta * q_stage1_mf)
        probs_1_mf = exp_q1_mf / np.sum(exp_q1_mf)
        p_choice_1[trial] = probs_1_mf[action_1[trial]]
        
        # Stage 2 policy
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model-Free and Model-Based RL with Perseveration and Eligibility Trace.
    Standard Daw et al. (2011) model that mixes MB and MF values, incorporates 
    choice perseveration, and updates stage 1 values via an eligibility trace.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    """
    learning_rate, beta, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        # OVERRIDE choice 1 policy to be Hybrid with Perseveration
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] = pers
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]
        
        # Stage 2 policy
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
