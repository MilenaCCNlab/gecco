Here are 3 new cognitive models, each adding a unique and theoretically grounded mechanism to the best model so far, while avoiding previously tried parameter combinations.

### Model 1: Choice Trace Stickiness
Instead of only perseverating on the immediately preceding action, participants often exhibit a longer-lasting bias towards actions they have chosen frequently in the recent past. This model replaces the simple `last_action` stickiness with an exponentially decaying `choice_trace` for both Stage 1 actions.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF mixture model with choice trace stickiness.
    Instead of only sticking to the immediately previous action, this model maintains 
    an exponentially decaying choice trace for both stage 1 actions.
    
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness: [-5,5]
    trace_decay: [0,1]
    """
    learning_rate, beta, w, lam, stickiness, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add stickiness based on historical choice trace
        q_stage1_net = q_stage1_net + stickiness * choice_trace
            
        exp_q1 = np.exp(beta * q_stage1_net - np.max(beta * q_stage1_net)) # Max subtraction for numeric stability
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Update choice trace for the next trial
        choice_trace = choice_trace * trace_decay
        choice_trace[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Model-Based Weight
Participants may dynamically adjust their reliance on the model-based system based on recent success. This model splits the model-based mixing weight `w` into `w_pos` (used if the previous trial was rewarded) and `w_neg` (used if the previous trial was unrewarded). 

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF mixture model with reward-dependent MB weight.
    This model allows the model-based weight to vary depending on whether the 
    previous trial was rewarded (w_pos) or unrewarded (w_neg).
    
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    w_pos: [0,1]
    w_neg: [0,1]
    lam: [0,1]
    stickiness: [-5,5]
    """
    learning_rate, beta, w_pos, w_neg, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        # Determine current w based on last reward
        if last_reward == 1:
            current_w = w_pos
        elif last_reward == 0:
            current_w = w_neg
        else:
            current_w = (w_pos + w_neg) / 2.0

        q_stage1_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += stickiness
            
        exp_q1 = np.exp(beta * q_stage1_net - np.max(beta * q_stage1_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Anti-Correlated (Fictitious) Updating
In a two-alternative forced-choice task, participants often infer information about the unchosen option. If they receive a reward from one alien, they might assume the other alien would not have rewarded them (and vice versa). This model introduces an `unchosen_lr` parameter to update the unchosen Stage 2 action in the opposite direction of the received reward.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF mixture model with anti-correlated updating for unchosen Stage 2 actions.
    This model assumes that participants update the unchosen Stage 2 action 
    in the opposite direction of the chosen action (fictitious update).
    
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness: [-5,5]
    unchosen_lr: [0,1]
    """
    learning_rate, beta, w, lam, stickiness, unchosen_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += stickiness
            
        exp_q1 = np.exp(beta * q_stage1_net - np.max(beta * q_stage1_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Fictitious update for the unchosen action at Stage 2
        unchosen_action_2 = 1 - action_2[trial]
        opp_reward = 1 - reward[trial] # Assume the opposite outcome
        delta_unchosen = opp_reward - q_stage2_mf[state_idx, unchosen_action_2]
        q_stage2_mf[state_idx, unchosen_action_2] += unchosen_lr * delta_unchosen

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
