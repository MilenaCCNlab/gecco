import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 1: Daw et al. model with asymmetric learning rates, memory decay, 
    eligibility trace, and a side bias for Stage 1 (omitting stickiness to maintain parameter limits).
    
    Parameters and Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    decay_rate: [0, 1] - Rate at which Q-values decay towards 0.5.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    lambda_: [0, 1] - Eligibility trace.
    bias_1: [-5, 5] - Bias added to Stage 1 action 0, capturing strong preference.
    """
    alpha_pos, alpha_neg, decay_rate, beta_1, beta_2, w, lambda_, bias_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bias_bonus = np.zeros(2)
        bias_bonus[0] = bias_1
            
        q1_total = beta_1 * q_stage1_net + bias_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # value decay
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 2: Daw et al. model with asymmetric learning rates, memory decay, 
    and outcome-dependent stickiness (win-stay vs lose-stay) for Stage 1. 
    
    Parameters and Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    decay_rate: [0, 1] - Rate at which Q-values decay towards 0.5.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness_win: [-5, 5] - Bonus to previous Stage 1 action if rewarded.
    stickiness_lose: [-5, 5] - Bonus to previous Stage 1 action if not rewarded.
    """
    alpha_pos, alpha_neg, decay_rate, beta_1, beta_2, w, stickiness_win, stickiness_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                stickiness_bonus[last_action_1] = stickiness_win
            else:
                stickiness_bonus[last_action_1] = stickiness_lose
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # value decay
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
