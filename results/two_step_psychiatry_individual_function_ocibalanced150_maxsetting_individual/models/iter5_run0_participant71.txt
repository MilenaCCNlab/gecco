def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid model with decay of Q-values towards 0.5 and stage 1 action bias.
    This model assumes that the participant's memory of alien reward probabilities 
    decays over time, pulling unchosen Q-values towards the neutral 0.5 value.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta1: [0,10] - Inverse temperature for first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weighting of model-based vs. model-free values (1 = fully model-based).
    p: [0,10] - Perseveration parameter for first-stage choice.
    lambd: [0,1] - Eligibility trace parameter.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    decay: [0,1] - Decay rate for stage 2 Q-values towards 0.5.
    """
    learning_rate, beta1, beta2, w, p, lambd, bias1, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values at 0.5 to match the decay target
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Decay Q-values towards 0.5 before making a choice
        q_stage2_mf = (1 - decay) * q_stage2_mf + decay * 0.5

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        action_bias1 = np.array([bias1, 0.0])
            
        logits_1 = beta1 * q_stage1_net + p * pers + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambd * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with state-specific biases at stage 2 and a bias at stage 1.
    Allows for different preferences for alien 0 depending on which planet the participant is on,
    capturing distinct choice strategies across different environments.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta1: [0,10] - Inverse temperature for first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weighting of model-based vs. model-free values (1 = fully model-based).
    p: [0,10] - Perseveration parameter for first-stage choice.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    bias_s0: [-5,5] - Bias towards choosing action 0 at state 0 (planet 0).
    bias_s1: [-5,5] - Bias towards choosing action 0 at state 1 (planet 1).
    """
    learning_rate, beta1, beta2, w, p, bias1, bias_s0, bias_s1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        action_bias1 = np.array([bias1, 0.0])
            
        logits_1 = beta1 * q_stage1_net + p * pers + action_bias1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice (with state-specific bias)
        if state_idx == 0:
            action_bias2 = np.array([bias_s0, 0.0])
        else:
            action_bias2 = np.array([bias_s1, 0.0])
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + action_bias2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating (no eligibility trace to maintain parameter limit)
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with separate learning rates for stage 1 and stage 2, plus action biases.
    Accounts for the possibility that participants update their estimates of spaceship transitions
    at a different rate than their estimates of alien reward probabilities.
    
    Parameters:
    alpha1: [0,1] - Learning rate for stage 1 Q-value updates.
    alpha2: [0,1] - Learning rate for stage 2 Q-value updates.
    beta1: [0,10] - Inverse temperature for first-stage choice.
    beta2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weighting of model-based vs. model-free values (1 = fully model-based).
    p: [0,10] - Perseveration parameter for first-stage choice.
    bias1: [-5,5] - Bias towards choosing action 0 at stage 1.
    bias2: [-5,5] - Bias towards choosing action 0 at stage 2.

    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

        q_stage1_mb = transition_matrix @ max_q_stage2
