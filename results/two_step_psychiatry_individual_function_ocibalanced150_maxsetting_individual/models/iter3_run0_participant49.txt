Here are three new cognitive models that introduce different theoretically grounded mechanisms to explain the participant's decision-making behavior.

### Model 1: Win-Stay / Lose-Shift Perseveration
This model extends the baseline by separating the perseveration (stickiness) parameter into two distinct parameters based on the outcome of the previous trial. Participants often exhibit a "win-stay, lose-shift" heuristic, meaning they are more likely to repeat a choice if it was rewarded, and less likely (or even prone to alternating) if it was unrewarded.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Free and Model-Based learning with reward-dependent perseveration.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating values.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers_win: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous trial was rewarded.
    pers_lose: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous trial was unrewarded.
    """
    learning_rate, beta_1, beta_2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers_bonus[last_action_1] = pers_win
            else:
                pers_bonus[last_action_1] = pers_lose
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Outcome-Dependent Learning Rates
Research in reinforcement learning often shows that individuals learn at different rates from positive outcomes (successes) versus negative outcomes (failures). This model introduces separate learning rates for trials that resulted in a reward versus those that did not.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model-Free and Model-Based learning with outcome-dependent learning rates.
    
    Parameters:
    lr_reward: [0,1] - Learning rate for trials where a reward was received.
    lr_noreward: [0,1] - Learning rate for trials where no reward was received.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers: [-5,5] - Perseveration bonus for repeating the previous stage 1 choice.
    """
    lr_reward, lr_noreward, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Select learning rate based on reward outcome
        lr = lr_reward if reward[trial] == 1 else lr_noreward

        q_stage1_mf[action_1[trial]] += lr * delta_stage1 + lr * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += lr * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Exponential Choice Trace (Multi-trial Habits)
Instead of relying strictly on a 1-back perseveration (only remembering the immediate last trial), this model integrates an exponentially decaying choice trace. This allows the model to capture longer-term habitual responding or multi-trial stickiness, where frequent past choices exert a cumulative pull on future behavior.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model-Free and Model-Based learning with an exponential choice trace for perseveration.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers: [-5,5] - Perseveration weight applied to the choice trace.
    tau_choice: [0,1] - Decay rate for the choice trace (1 = 1-back memory only, 0 = perfect memory of all past choices).
    """
    learning_rate, beta_1, beta_2, w, lam, pers, tau_choice = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        # Perseveration bonus is based on the decaying choice trace
        pers_bonus = pers * choice_trace
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Update choice trace for the next trial
        choice_trace = choice_trace * (1 - tau_choice)
        choice_trace[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
