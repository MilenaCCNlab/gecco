import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dynamic Transition Matrix Updating.
    
    This model allows the transition matrix to be updated trial-by-trial based on observed transitions,
    rather than assuming fixed transition probabilities.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence at Stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration bonus added to the previously chosen Stage 1 action.
    transition_lr: [0, 1] - Learning rate for updating the transition matrix.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, transition_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        
        # action value updating for choice 1
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        # eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        # update transition matrix based on observed state transition
        transition_matrix[a1, state_idx] += transition_lr * (1 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += transition_lr * (0 - transition_matrix[a1, 1 - state_idx])
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Reward Scaling (Loss Penalty).
    
    This model treats unrewarded outcomes (0 coins) as a negative penalty rather than just zero,
    which can capture loss aversion or disappointment in the participant's updates.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence at Stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration bonus added to the previously chosen Stage 1 action.
    loss_penalty: [0, 5] - Negative value assigned to unrewarded trials (reward=0 becomes -loss_penalty).
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, loss_penalty = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += pers
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]

        # policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        
        # action value updating for choice 1
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        eff_reward = reward[trial] if reward[trial] > 0 else -loss_penalty
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, a2]
        
        # eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-stage Perseveration.
    
    This model includes separate perseveration parameters for Stage 1 (spaceship choice) 
    and Stage 2 (alien choice), capturing stickiness at both decision points.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    alpha2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence at Stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    pers1: [-5, 5] - Perseveration bonus added to the previously chosen Stage 1 action.
    pers2: [-5, 5] - Perseveration bonus added to the previously chosen Stage 2 action (state-specific).
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_stage1_net[last_action1] += pers1
            
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action2[state_idx] != -1:
            q_stage2_net[last_action2[state_idx]] += pers2

        q2_centered = q_stage2_net - np.max(q_stage2_net)
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        
        # action value updating for choice 1
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        # eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        last_action1 = a1
        last_action2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
