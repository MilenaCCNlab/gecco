Here are 3 new cognitive models that build upon the best model so far by introducing new mechanisms to capture participant behavior.

### Model 1: Dynamic Transition Matrix Learning
While the baseline models assume the participant knows the transition probabilities (0.7/0.3) from the start, this model assumes the participant learns the transition matrix from experience. It uses a `transition_lr` parameter to update the model-based transition expectations after each trial.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF learning with dynamic transition matrix learning.
    Includes eligibility traces (lam) and stickiness, but learns the transition matrix 
    from experience rather than assuming it is fixed at 0.7/0.3.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    transition_lr: [0,1] - learning rate for the transition matrix
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, lam, transition_lr = model_parameters
    n_trials = len(action_1)
  
    # Start with uniform expectations about transitions
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update transition matrix based on observed state
        transition_matrix[action_1[trial], state_idx] += transition_lr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += transition_lr * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Stage Learning Rates
This model introduces separate learning rates for Stage 1 and Stage 2. It's plausible that participants update their overarching spaceship preferences (Stage 1) at a different speed than their planet-specific alien preferences (Stage 2), especially since alien rewards drift over time but the transition structure is fixed.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for stage 1 and stage 2, with decay and eligibility trace.
    Uses learning_rate_1 for stage 1 MF updates and learning_rate_2 for stage 2 updates.
    Includes decay_rate for unchosen Q-values and lam for eligibility traces.
    
    Parameters:
    learning_rate_1: [0,1] - learning rate for stage 1 Q-values
    learning_rate_2: [0,1] - learning rate for stage 2 Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    """
    learning_rate_1, learning_rate_2, beta, w, stickiness_1, stickiness_2, decay_rate, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay all Q-values
        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        # Update Q-values with separate learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Decaying Habit Traces
Instead of 1-trial stickiness (which only rewards repeating the very last action), this model implements a decaying habit trace. This allows perseverative choices to build up over multiple trials, capturing deep habits formed when a participant stays on the same path for a long streak.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Habit traces for choices rather than just 1-trial stickiness.
    Replaces the simple last_action stickiness with an exponentially decaying habit trace,
    allowing choices from multiple past trials to influence current perseveration.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - weight of the stage 1 habit trace
    stickiness_2: [-5,5] - weight of the stage 2 habit trace
    decay_rate: [0,1] - rate at which all Q-values decay towards 0 on each trial
    lam: [0,1] - eligibility trace parameter carrying stage 2 RPE to stage 1
    habit_decay: [0,1] - rate at which the habit traces decay (0 = 1-trial stickiness, 1 = permanent habit)
    """
    learning_rate, beta, w, stickiness_1, stickiness_2, decay_rate, lam, habit_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    # Initialize habit traces
    habit_1 = np.zeros(2)
    habit_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy() + stickiness_1 * habit_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta * q_stage2_mf[state_idx].copy() + stickiness_2 * habit_2[state_idx]
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values
        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Update habit traces
        habit_1 *= habit_decay
        habit_1[action_1[trial]] += 1
        
        habit_2 *= habit_decay
        habit_2[state_idx, action_2[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
