Here are 3 new cognitive models that build upon the most successful model so far (`alpha, decay, beta1, beta2, w, lam`) by incorporating theoretically grounded mechanisms: reward penalties for zero-reward outcomes, dynamic learning of transition probabilities, and counterfactual updating.

### Model 1: Value Decay + Perseveration + Reward Penalty
This model introduces an `omega` parameter to penalize unrewarded trials. In tasks where rewards are strictly 1 or 0, participants often perceive the omission of reward (0) as a loss. This model treats a 0 reward as `-omega` during the prediction error calculation. It also includes a stage 1 choice perseveration parameter (`pers`).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with value decay, stage 1 perseveration, and reward penalty.
    Unrewarded trials (reward=0) are penalized by -omega, acting as loss aversion.
    
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pers: [-5, 5]
    omega: [0, 1]
    """
    alpha, decay, beta1, beta2, w, lam, pers, omega = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if last_a1 != -1:
            logits_1[last_a1] += pers
            
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating with reward penalty
        r_eff = 1.0 if r > 0.5 else -omega
        
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r_eff - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay) * q_stage2_mf[state_idx, action_idx] + decay * 0.5

        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Value Decay + Perseveration + Transition Learning
Instead of assuming a static `[[0.7, 0.3], [0.3, 0.7]]` transition matrix, this model dynamically updates its internal representation of the transition matrix using a transition learning rate (`alpha_tr`). This accounts for participants who continuously update their world-model over the course of the task.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with value decay, transition matrix learning, and stage 1 perseveration.
    The internal model of transitions dynamically updates based on observed states.
    
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    alpha_tr: [0, 1]
    pers: [-5, 5]
    """
    alpha, decay, beta1, beta2, w, lam, alpha_tr, pers = model_parameters
    n_trials = len(action_1)
    
    # Starting with instructed probabilities, but allowed to change
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if last_a1 != -1:
            logits_1[last_a1] += pers
            
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Transition matrix updating
        transition_matrix[a1, s] += alpha_tr * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_tr * (0.0 - transition_matrix[a1, 1 - s])

        # Decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay) * q_stage2_mf[state_idx, action_idx] + decay * 0.5

        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Value Decay + Perseveration + Counterfactual Learning
This model incorporates counterfactual learning (`cf_weight`). When a participant receives a reward (or no reward) from an alien on a planet, they might update their value for the *other* unchosen alien on that same planet in the opposite direction, assuming anti-correlated reward probabilities. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with value decay, stage 1 perseveration, and counterfactual learning.
    Updates the unchosen stage 2 action in the visited state assuming anti-correlated rewards.
    
    Bounds:
    alpha: [0, 1]
    decay: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    pers: [-5, 5]
    cf_weight: [0, 1]
    """
    alpha, decay, beta1, beta2, w, lam, pers, cf_weight = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if last_a1 != -1:
            logits_1[last_a1] += pers
            
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Direct Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Counterfactual update for the unchosen action in the SAME state
        unchosen_a2 = 1 - a2
        r_cf = 1.0 - r
        delta_cf = r_cf - q_stage2_mf[s, unchosen_a2]
        q_stage2_mf[s, unchosen_a2] += alpha * cf_weight * delta_cf

        # Decay for unchosen Stage 1 options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        # Decay only for the unvisited state in Stage 2 (since both actions in the visited state were updated)
        for state_idx in range(2):
            for action_idx in range(2):
                if state_idx != s:
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay) * q_stage2_mf[state_idx, action_idx] + decay * 0.5

        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
