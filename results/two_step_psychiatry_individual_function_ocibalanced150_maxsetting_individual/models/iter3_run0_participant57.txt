def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with perseveration on both stages and decay for unchosen actions.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers1: perseveration bonus for repeating the previous stage 1 choice [-2, 2]
    pers2: perseveration bonus for repeating the previous stage 2 choice [-2, 2]
    decay: decay rate for unchosen actions [0,1]
    """
    alpha, beta1, beta2, w, lam, pers1, pers2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers1_bonus[last_action_1] = pers1
            
        q_stage1_total = beta1 * q_stage1_net + pers1_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        pers2_bonus = np.zeros(2)
        if last_action_2 != -1:
            pers2_bonus[last_action_2] = pers2
            
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2_bonus
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay for unchosen actions
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != action_2[trial]:
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with both motor and state-specific stage 2 perseveration.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers1: perseveration bonus for repeating the previous stage 1 choice [-2, 2]
    pers2_motor: motor perseveration bonus for repeating previous stage 2 choice [-2, 2]
    pers2_state: state-specific perseveration bonus for stage 2 [-2, 2]
    """
    alpha, beta1, beta2, w, lam, pers1, pers2_motor, pers2_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_action_2_by_state = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers1_bonus[last_action_1] = pers1
            
        q_stage1_total = beta1 * q_stage1_net + pers1_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        pers2_bonus = np.zeros(2)
        # Apply motor perseveration (regardless of state)
        if last_action_2 != -1:
            pers2_bonus[last_action_2] += pers2_motor
        # Apply state-specific perseveration
        if last_action_2_by_state[state_idx] != -1:
            pers2_bonus[last_action_2_by_state[state_idx]] += pers2_state
            
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2_bonus
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_action_2_by_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with reward-dependent stage 2 perseveration.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers1: perseveration bonus for repeating the previous stage 1 choice [-2, 2]
    pers_win2: perseveration bonus for stage 2 if previous trial was rewarded [-2, 2]
    pers_lose2: perseveration bonus for stage 2 if previous trial was unrewarded [-2, 2]
    """
    alpha, beta1, beta2, w, lam, pers1, pers_win2, pers_lose2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])

    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0

            last_reward = -1

            if last_reward == 1:
                pers2_bonus[last_action_2] = pers_win2
            else:
                pers2_bonus[last_action_2] = pers_lose2
            
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2_bonus

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
