def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Standard two-step model with model-based and model-free control, 
    eligibility trace, stage-specific temperatures, and stickiness for both stages.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weight of model-based vs model-free control
    lam: [0,1] - Eligibility trace parameter
    st_1: [-5,5] - Choice stickiness for stage 1
    st_2: [-5,5] - Choice stickiness for stage 2
    """
    alpha, beta_1, beta_2, w, lam, st_1, st_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus_1[last_action_1] = st_1
            
        q1_val = beta_1 * q_net + stickiness_bonus_1
        q1_val -= np.max(q1_val) # Prevent overflow
        exp_q1 = np.exp(q1_val)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_bonus_2[last_action_2[state_idx]] = st_2
            
        q2_val = beta_2 * q_stage2_mf[state_idx] + stickiness_bonus_2
        q2_val -= np.max(q2_val) # Prevent overflow
        exp_q2 = np.exp(q2_val)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Standard two-step model with MB/MF control, eligibility trace, 
    stage-specific temperatures, first-stage stickiness, and Q-value decay for unchosen actions.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weight of model-based vs model-free control
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness for stage 1
    decay: [0,1] - Decay rate for unchosen actions towards 0
    """
    alpha, beta_1, beta_2, w, lam, st, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = st
            
        q1_val = beta_1 * q_net + stickiness_bonus
        q1_val -= np.max(q1_val) # Prevent overflow
        exp_q1 = np.exp(q1_val)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_val = beta_2 * q_stage2_mf[state_idx]
        q2_val -= np.max(q2_val) # Prevent overflow
        exp_q2 = np.exp(q2_val)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # decay unchosen actions towards 0
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Standard two-step model with MB/MF control, eligibility trace, 
    stage-specific temperatures, first-stage stickiness, and a Win-Stay-Lose-Shift (WSLS) bonus.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weight of model-based vs model-free control
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness for stage 1
    wsls: [-5,5] - Win-stay-lose-shift bonus for stage 1
    """
    alpha, beta_1, beta_2, w, lam, st, wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus = np.zeros(2)
        if last_action_1 != -1:
            bonus[last_action_1] += st
            if last_reward == 1:
                bonus[last_action_1] += wsls
            else:
                bonus[1 - last_action_1] += wsls
                
        q1_val = beta_1 * q_net + bonus
        q1_val -= np.max(q1_val) # Prevent overflow
        exp_q1 = np.exp(q1_val)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_val = beta_2 * q_stage2_mf[state_idx]
        q2_val -= np.max(q2_val) # Prevent overflow
        exp_q2 = np.exp(q2_val)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
