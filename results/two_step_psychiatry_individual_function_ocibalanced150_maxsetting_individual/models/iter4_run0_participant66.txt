Here are 3 new cognitive models that explore alternative parameter configurations that have not been tested yet. They build upon the findings of the best model so far by introducing separate learning rates for different contexts (positive/negative prediction errors, or stage 1/stage 2) while keeping the parameter counts to 8.

### Model 1: Asymmetric Learning with Dual Stage Perseveration
This model tests whether participants learn differently from positive versus negative prediction errors at both stages, while also maintaining separate stickiness (perseveration) parameters for stage 1 and stage 2 choices.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model with separate learning rates for positive and negative prediction errors, separate stage betas, and separate stage perseveration.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice perseveration (stickiness) for stage 1
    pers_2: [-2, 2] - choice perseveration (stickiness) for stage 2
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambd, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers_1
            
        logits_1 = beta_1 * q_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_a2 != -1:
                pers_bonus_2[last_a2] = pers_2
                
            logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
            q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_s2 * lambd * delta_stage2
            
        last_a1 = action_1[trial]
        last_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning with Dual Stage Perseveration
Participants might learn at different speeds depending on the stage of the task. This model tests separate learning rates for stage 1 and stage 2, combined with independent perseveration bonuses for each stage.

### Model 3: Stage-Specific Learning with Q-Value Decay
Combining the findings of your best decay model with independent stage learning rates. Instead of dual perseveration, we keep only stage 1 stickiness, but allow the learning rates for stage 1 and stage 2 to vary independently while incorporating decay towards 0.5 for unchosen options.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model with separate learning rates for stages, Q-value decay, separate stage betas, and stage 1 perseveration.
    
    Bounds:
    alpha_1: [0, 1] - learning rate for stage 1
    alpha_2: [0, 1] - learning rate for stage 2
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0.5)
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambd: [0, 1] - eligibility trace
    pers: [-2, 2] - choice perseveration (stickiness) for stage 1
    """
    alpha_1, alpha_2, decay, beta_1, beta_2, w, lambd, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers
            
        logits_1 = beta_1 * q_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_1 * lambd * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
        last_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
