Here are 3 new cognitive models that explore alternative mechanisms for decision-making in the two-step task. 

### Model 1: Stage-Specific Inverse Temperatures and Perseveration
This model separates the decision noise (inverse temperature) and the perseveration tendency (stickiness) for Stage 1 and Stage 2. Since participants often show different levels of exploration and habituation at different stages of the task, treating them independently might provide a better fit.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Free and Model-Based RL with stage-specific inverse temperatures and stage-specific perseveration.
    
    Parameters:
    alpha: [0,1] - Learning rate for updating Q-values.
    beta1: [0,10] - Inverse temperature for softmax action selection in stage 1.
    beta2: [0,10] - Inverse temperature for softmax action selection in stage 2.
    w: [0,1] - Weight on model-based vs model-free Q-values for stage 1 (1 = purely MB, 0 = purely MF).
    lambd: [0,1] - Eligibility trace parameter carrying stage 2 prediction errors to stage 1.
    p1: [-5,5] - Perseveration parameter, adding a bonus to the previously chosen stage 1 action.
    p2: [-5,5] - Perseveration parameter, adding a bonus to the previously chosen stage 2 action in the current state.
    """
    alpha, beta1, beta2, w, lambd, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net1[last_action_1] += p1
            
        exp_q1 = np.exp(beta1 * (q_net1 - np.max(q_net1)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q_net2 = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q_net2[last_action_2[state_idx]] += p2

        exp_q2 = np.exp(beta2 * (q_net2 - np.max(q_net2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition Matrix Learning
In the typical two-step task, models assume participants magically know the fixed transition matrix (0.7 / 0.3). In this variant, the participant dynamically updates their internal model of the transition matrix over time using a transition learning rate `alpha_trans` starting from an unbiased prior (0.5 / 0.5).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF model with perseveration, where the transition matrix is learned over time 
    rather than assumed fixed.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weight on model-based vs model-free Q-values for stage 1.
    lambd: [0,1] - Eligibility trace parameter carrying stage 2 prediction errors to stage 1.
    p: [-5,5] - Perseveration parameter, adding a bonus to the previously chosen stage 1 action.
    alpha_trans: [0,1] - Learning rate for updating the transition matrix probabilities.
    """
    alpha, beta, w, lambd, p, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to 0.5 for all transitions
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update transition matrix based on observed transition
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Continuous Choice Trace
Instead of single-trial perseveration where only the immediate *last* action matters, this model implements an exponentially decaying "choice trace". This means that strong habits built up over several trials will carry more momentum and take longer to override than a choice made only once.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF model with a continuous choice trace for stage 1 perseveration.
    Instead of only the immediately preceding action, an exponentially decaying trace of past actions 
    biases current choices.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-values.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weight on model-based vs model-free Q-values for stage 1.
    lambd: [0,1] - Eligibility trace parameter carrying stage 2 prediction errors to stage 1.
    p: [-5,5] - Weight of the choice trace (perseveration strength).
    tau: [0,1] - Decay rate of the choice trace (1 = no decay/infinite memory, 0 = only last action).
    """
    alpha, beta, w, lambd, p, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseveration based on choice trace
        q_net += p * choice_trace
            
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
        
        # Update choice trace
        choice_trace *= tau
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
