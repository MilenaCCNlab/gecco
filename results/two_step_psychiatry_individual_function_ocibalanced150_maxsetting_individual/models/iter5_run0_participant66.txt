Here are 3 new cognitive models that introduce distinct mechanisms to explain the participant's choices. 

### Model 1: Lapse Rate Model
This model extends the basic learning and stickiness mechanisms by incorporating a **lapse rate**. It assumes that on a certain fraction of trials, the participant's attention lapses, leading them to make a completely random choice rather than one guided by their learned values or perseveration. 

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model with separate stage betas, perseveration at both stages, and a lapse rate.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice perseveration (stickiness) for stage 1
    pers_2: [-2, 2] - choice perseveration (stickiness) for stage 2
    lapse: [0, 1] - lapse rate (probability of making a random choice)
    """
    alpha, beta_1, beta_2, w, lambd, pers_1, pers_2, lapse = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers_1
            
        logits_1 = beta_1 * q_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # Apply lapse rate
        probs_1 = lapse * 0.5 + (1 - lapse) * probs_1
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_a2 != -1:
                pers_bonus_2[last_a2] = pers_2
                
            logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            
            # Apply lapse rate
            probs_2 = lapse * 0.5 + (1 - lapse) * probs_2
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        # Updates
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            # Eligibility trace update
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

        last_a1 = action_1[trial]
        last_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Win-Stay / Lose-Shift Perseveration with Decay
This model expands on the best model you've found so far (which uses value decay towards 0.5). However, instead of a simple general stickiness, it splits stickiness into **reward-dependent perseveration**. The participant's tendency to repeat the previous action is modulated by whether the last trial yielded a reward (`pers_win`) or did not yield a reward (`pers_lose`).

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model with Q-value decay, separate stage betas, and reward-dependent perseveration (win-stay/lose-shift).
    
    Bounds:
    alpha: [0, 1] - learning rate
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0.5)
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambd: [0, 1] - eligibility trace
    pers_win: [-2, 2] - choice perseveration if previous trial was rewarded
    pers_lose: [-2, 2] - choice perseveration if previous trial was not rewarded
    """
    alpha, decay, beta_1, beta_2, w, lambd, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = -1
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            if last_reward > 0:
                pers_bonus_1[last_a1] = pers_win
            else:
                pers_bonus_1[last_a1] = pers_lose
            
        logits_1 = beta_1 * q_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_a2 != -1:
                if last_reward > 0:
                    pers_bonus_2[last_a2] = pers_win
                else:
                    pers_bonus_2[last_a2] = pers_lose
                
            logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        # Updates
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

            # Decay unchosen/unvisited Q-values towards 0.5
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
        last_a1 = action_1[trial]
        last_a2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Choice Trace Model
Instead of only considering the immediately preceding trial for stickiness, this model implements a **choice trace**. Over time, the participant builds a habit towards frequently chosen options. The trace tracks the exponential moving average of past choices governed by the `alpha_c` parameter, providing a "habitual" bonus to the logits rather than just a 1-trial perseveration bonus.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model with choice traces for perseveration at both stages.
    Instead of just the immediate past choice, a trace of past choices is maintained.
    
    Bounds:
    alpha: [0, 1] - learning rate for Q-values
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice trace weight for stage 1
    pers_2: [-2, 2] - choice trace weight for stage 2
    alpha_c: [0, 1] - learning rate for the choice trace
    """
    alpha, beta_1, beta_2, w, lambd, pers_1, pers_2, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    # Choice traces
    c_stage1 = np.zeros(2)
    c_stage2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net + pers_1 * c_stage1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_2 * c_stage2[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        # Updates
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2
            
        # Update choice traces regardless of whether stage 2 was completed
        c_stage1 *= (1 - alpha_c)
        c_stage1[action_1[trial]] += alpha_c
        
        c_stage2 *= (1 - alpha_c)
        if action_2[trial] != -1:
            c_stage2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
