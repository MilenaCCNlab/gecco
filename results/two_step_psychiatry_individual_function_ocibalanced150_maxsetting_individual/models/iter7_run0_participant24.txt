Here are 3 new cognitive models that introduce distinct computational mechanisms to explain participant behavior in the two-step task. 

### Model 1: Choice Trace (Autocorrelation) Model
Instead of only considering the immediate previous choice (lag-1 stickiness), this model maintains an exponentially decaying "choice trace." This allows the model to capture a buildup of preference for a spaceship over multiple consecutive choices, which is common in human decision-making.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with Choice Trace.
    Extends the standard model by replacing the lag-1 choice stickiness 
    with an exponential moving average of past choices (choice trace).
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight given to model-based vs. model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rep: [-5, 5] - Weight for the choice trace (stickiness).
    tau: [0, 1] - Decay rate for the choice trace (1 = only last choice matters, 0 = no update).
    """
    alpha, beta1, beta2, w, lam, rep, tau = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net + rep * choice_trace
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Value updating
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update choice trace
        choice_trace *= (1 - tau)
        choice_trace[a1] += tau

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Option Bias Model
Looking closely at the participant data, this participant selects Alien 0 (146 times) vastly more frequently than Alien 1 (54 times). This model introduces static option biases to capture inherent spatial or stimulus-specific preferences independent of learned values.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with Option Biases.
    Includes baseline preferences (biases) for specific options at both stages,
    which can account for spatial or stimulus-specific preferences.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weight given to model-based vs. model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rep: [-5, 5] - Choice stickiness (perseveration) for stage 1 choices.
    bias1: [-5, 5] - Inherent bias for spaceship 0 (added to its logit).
    bias2: [-5, 5] - Inherent bias for alien 0 (added to its logit).
    """
    alpha, beta1, beta2, w, lam, rep, bias1, bias2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta1 * q_stage1_net + rep * stickiness
        logits_1[0] += bias1
        
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        
        logits_2_biased = np.zeros(2)
        logits_2_biased[0] = logits_2[0] + bias2
        logits_2_biased[1] = logits_2[1]
        
        logits_2_biased -= np.max(logits_2_biased)
        exp_q2 = np.exp(logits_2_biased)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Value updating
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Model-Based Weight Model
This model tests the hypothesis that cognitive control (and thus model-based planning) is modulated by recent success. It allows the weight assigned to the model-based system (`w`) to increase or decrease depending on whether the previous trial was rewarded.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with Reward-Dependent Model-Based Weight.
    Allows the weight given to the model-based system (w) to vary 
    depending on whether the previous trial was rewarded or not.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w_win: [0, 1] - Model-based weight on trials following a reward.
    w_lose: [0, 1] - Model-based weight on trials following no reward.
    lam: [0, 1] - Eligibility trace parameter.
    rep: [-5, 5] - Choice stickiness (perseveration) for stage 1 choices.
    """
    alpha, beta1, beta2, w_win, w_lose, lam, rep = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1 

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine w based on previous reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = (w_win + w_lose) / 2.0 # Default for the very first trial
            
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta1 * q_stage1_net + rep * stickiness
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]

        # Value updating
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
