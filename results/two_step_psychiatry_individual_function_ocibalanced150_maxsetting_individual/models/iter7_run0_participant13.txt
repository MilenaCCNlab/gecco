def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Direct Model-Free Update with Asymmetric Learning, Decay, and Reward Scaling.
    Uses separate learning rates for positive and negative prediction errors,
    along with decay for unchosen options, and a reward scaling parameter to separate
    the magnitude of the learned values from the inverse temperatures.
    
    Bounds:
    alpha_pos: [0,1] (Learning rate for positive PE)
    alpha_neg: [0,1] (Learning rate for negative PE)
    decay: [0,1] (Decay rate for unchosen options)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    pi: [-5,5] (Choice stickiness for stage 1)
    rho: [0,10] (Reward scaling factor)
    """
    alpha_pos, alpha_neg, decay, beta_1, beta_2, w, pi, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if trial > 0:
            rep[action_1[trial-1]] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        r = reward[trial] * rho
        
        pe_1 = r - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if pe_1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * pe_1

        pe_2 = r - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if pe_2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * pe_2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Direct Model-Free Update with Asymmetric Learning, Decay, and Action Trace Stickiness.
    Uses an exponentially decaying trace of past actions instead of just the last trial's action,
    capturing longer-term choice perseveration.
    
    Bounds:
    alpha_pos: [0,1] (Learning rate for positive PE)
    alpha_neg: [0,1] (Learning rate for negative PE)
    decay: [0,1] (Decay rate for unchosen options)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    pi: [-5,5] (Weight of action trace)
    tau: [0,1] (Decay rate of action trace)
    """
    alpha_pos, alpha_neg, decay, beta_1, beta_2, w, pi, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    action_trace = np.zeros(2)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits_1 = beta_1 * q_net + pi * action_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        pe_1 = reward[trial] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if pe_1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * pe_1

        pe_2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if pe_2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * pe_2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)
        
        # Update action trace for perseveration
        action_trace *= (1 - tau)
        action_trace[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
