Here are three new cognitive models that explore alternative mechanisms for learning and decision-making in the two-step task. They avoid the exact parameter combinations you have already tested and introduce distinct, theoretically grounded psychological mechanisms.

### Model 1: Outcome-Dependent Perseveration (Win-Stay / Lose-Shift)
This model splits the stage 1 choice perseveration into two separate parameters based on the outcome of the previous trial. Participants often exhibit "win-stay, lose-shift" heuristics, meaning their tendency to repeat a choice might be much stronger after a reward than after an omission.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Outcome-Dependent Stage 1 Perseveration
    
    This model separates the stage 1 perseveration into two parameters based on 
    the outcome of the previous trial, capturing win-stay/lose-shift heuristics.
    
    Parameters:
    alpha: learning rate for value updates [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p1_win: stage 1 perseveration if the previous trial was rewarded [-5, 5]
    p1_lose: stage 1 perseveration if the previous trial was unrewarded [-5, 5]
    p2: stage 2 perseveration [-5, 5]
    """
    alpha, beta1, beta2, w, lam, p1_win, p1_lose, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    last_reward = -1
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            if last_reward == 1:
                logits_1[last_action_1] += p1_win
            elif last_reward == 0:
                logits_1[last_action_1] += p1_lose
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        last_reward = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Dynamic Transition Matrix Learning
This model assumes participants do not inherently know or trust the fixed 70%/30% transition structure. Instead, they dynamically learn the transition probabilities from experience, updating their internal model of the environment on every trial.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dynamic Transition Matrix Learning
    
    Instead of assuming fixed knowledge of the transition probabilities, 
    participants learn the transition matrix over time from experience.
    
    Parameters:
    alpha: learning rate for values [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p1: stage 1 perseveration [-5, 5]
    p2: stage 2 perseveration [-5, 5]
    alpha_tr: learning rate for transition probabilities [0, 1]
    """
    alpha, beta1, beta2, w, lam, p1, p2, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix uniformly
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # update dynamic transition matrix
        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_tr * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_tr * (0.0 - transition_matrix[a1, 1 - s])
        
        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Counterfactual Updating at Stage 2
This model incorporates counterfactual thinking. When participants receive a reward (or lack thereof) from the chosen alien, they may assume the unchosen alien on that same planet would have yielded the opposite outcome, updating its value simultaneously.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Counterfactual Updating at Stage 2
    
    This model assumes that participants update the unchosen alien in the 
    visited state, assuming its reward probability might be anti-correlated 
    with the chosen alien (e.g., updating unchosen with 1 - reward).
    
    Parameters:
    alpha: learning rate for chosen values [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p1: stage 1 perseveration [-5, 5]
    p2: stage 2 perseveration [-5, 5]
    alpha_cf: counterfactual learning rate for unchosen stage 2 action [0, 1]
    """
    alpha, beta1, beta2, w, lam, p1, p2, alpha_cf = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
                
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # counterfactual update for unchosen action at stage 2
        unchosen_2 = 1 - action_2[trial]
        cf_reward = 1.0 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_2]
        q_stage2_mf[state_idx, unchosen_2] += alpha_cf * delta_cf
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
