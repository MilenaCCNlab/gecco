def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model that dynamically learns the transition matrix instead of assuming fixed 0.7/0.3 probabilities.
    Includes a learning rate for Q-values, a separate learning rate for transitions, 
    decay rate for unchosen actions towards 0.5, separate betas, MB weight, and stage 1 stickiness.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-values
    lr_trans: [0,1] - Learning rate for updating the transition matrix
    decay_rate: [0,1] - Rate at which unchosen action values decay towards 0.5
    beta1: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weight on model-based vs model-free strategy
    stickiness: [-5,5] - Perseveration parameter for stage 1
    """
    learning_rate, lr_trans, decay_rate, beta1, beta2, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to 0.5 for all transitions (uniform prior)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness

        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
                    
        # Update transition matrix probabilities based on observed transition
        transition_matrix[action_1[trial], state_idx] += lr_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += lr_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates (lr_pos, lr_neg) for Q-values, decay rate, separate betas, 
    MB weight, and outcome-dependent stage 1 stickiness (win-stay, lose-shift mechanism).
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors
    lr_neg: [0,1] - Learning rate for negative prediction errors
    decay_rate: [0,1] - Rate at which unchosen action values decay towards 0.5
    beta1: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weight on model-based vs model-free strategy
    win_stickiness: [-5,5] - Perseveration for stage 1 if previous trial was rewarded
    lose_stickiness: [-5,5] - Perseveration for stage 1 if previous trial was unrewarded
    """
    lr_pos, lr_neg, decay_rate, beta1, beta2, w, win_stickiness, lose_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            if reward[trial-1] == 1:
                q_stage1_net[action_1[trial-1]] += win_stickiness
            else:
                q_stage1_net[action_1[trial-1]] += lose_stickiness

        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model that combines Model-Based and Model-Free strategies at the choice probability level 
    rather than the Q-value level. Includes asymmetric learning rates, decay rate, 
    separate betas, and stage 1 stickiness.
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors
    lr_neg: [0,1] - Learning rate for negative prediction errors
    decay_rate: [0,1] - Rate at which unchosen action values decay towards 0.5
    beta1: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w_prob: [0,1] - Weight on MB vs MF choice probabilities
    stickiness: [-5,5] - Perseveration parameter for stage 1
    """
    lr_pos, lr_neg, decay_rate, beta1, beta2, w_prob, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Add stickiness to both MB and MF Q-values before softmax
        q_mb_net = np.copy(q_stage1_mb)
        q_mf_net = np.copy(q_stage1_mf)
        if trial > 0:
            q_mb_net[action_1[trial-1]] += stickiness
            q_mf_net[action_1[trial-1]] += stickiness

        exp_q1_mb = np.exp(beta1 * q_mb_net)
        probs_1_mb = exp_q1_mb / np.sum(exp_q1_mb)
        
        exp_q1_mf = np.exp(beta1 * q_mf_net)
        probs_1_mf = exp_q1_mf / np.sum(exp_q1_mf)
        
        # Combine at probability level
        probs_1 = w_prob * probs_1_mb + (1 - w_prob) * probs_1_mf
        
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
