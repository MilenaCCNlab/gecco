Here are three new cognitive models that build upon the best-performing model by introducing distinct cognitive mechanisms. Each model stays within the 8-parameter limit and explores configurations that have not been tested yet.

### Model 1: Stage-Specific Decay Rates
Different stages of the task may involve different forgetting or decay processes. This model introduces separate decay rates for the first stage (`decay_1`) and the second stage (`decay_2`), allowing for the possibility that participants discount unchosen overarching strategies (spaceships) at a different rate than specific state-action pairs (aliens).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Incorporates separate decay rates for stage 1 and stage 2 Q-values.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weight of model-based vs model-free control
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness for stage 1
    decay_1: [0,1] - Decay rate for unchosen actions in stage 1
    decay_2: [0,1] - Decay rate for unchosen actions in stage 2
    """
    alpha, beta_1, beta_2, w, lam, st, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = st
            
        q1_val = beta_1 * q_net + stickiness_bonus
        q1_val -= np.max(q1_val) # Prevent overflow
        exp_q1 = np.exp(q1_val)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_val = beta_2 * q_stage2_mf[state_idx]
        q2_val -= np.max(q2_val) # Prevent overflow
        exp_q2 = np.exp(q2_val)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Decay for unchosen actions with separate rates
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay_1)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_2)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Model-Based Control
Participants may adapt their decision-making strategy based on recent success. This model assumes the weighting between model-based and model-free control shifts depending on whether the previous trial resulted in a reward (`w_rew`) or no reward (`w_unrew`).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: State-dependent model-based weight based on the previous trial's reward.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w_rew: [0,1] - Weight of model-based control after a rewarded trial
    w_unrew: [0,1] - Weight of model-based control after an unrewarded trial
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness for stage 1
    decay: [0,1] - Decay rate for unchosen actions towards 0
    """
    alpha, beta_1, beta_2, w_rew, w_unrew, lam, st, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 1 # Assume initial positive state

    for trial in range(n_trials):
        # Determine MB weight based on previous reward
        w = w_rew if last_reward == 1 else w_unrew
        
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = st
            
        q1_val = beta_1 * q_net + stickiness_bonus
        q1_val -= np.max(q1_val) 
        exp_q1 = np.exp(q1_val)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_val = beta_2 * q_stage2_mf[state_idx]
        q2_val -= np.max(q2_val) 
        exp_q2 = np.exp(q2_val)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Regression to the Mean (Decay to Center)
Instead of unchosen Q-values decaying strictly towards zero, this model assumes they decay towards a parameterized central value (`center`). This represents a "regression to the mean" bias where participants assume unknown or unchosen options tend towards an average expected value rather than being completely worthless.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Unchosen Q-values decay towards a parameterized central value rather than 0.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weight of model-based vs model-free control
    lam: [0,1] - Eligibility trace parameter
    st: [-5,5] - Choice stickiness for stage 1
    decay: [0,1] - Decay rate for unchosen actions
    center: [0,1] - The value to which unchosen actions decay
    """
    alpha, beta_1, beta_2, w, lam, st, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values at the center rather than 0
    q_stage1_mf = np.full(2, center)
    q_stage2_mf = np.full((2, 2), center)
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = st
            
        q1_val = beta_1 * q_net + stickiness_bonus
        q1_val -= np.max(q1_val) 
        exp_q1 = np.exp(q1_val)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_val = beta_2 * q_stage2_mf[state_idx]
        q2_val -= np.max(q2_val) 
        exp_q2 = np.exp(q2_val)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Decay for unchosen actions towards the center value
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = q_stage1_mf[unchosen_1] * (1 - decay) + center * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + center * decay
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
