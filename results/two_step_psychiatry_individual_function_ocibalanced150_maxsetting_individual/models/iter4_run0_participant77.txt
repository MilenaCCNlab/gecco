def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. 2011 Two-Step Task Model with Stage 2 Perseveration.
    Includes standard stage 1 perseveration as well as stage 2 perseveration 
    (stickiness to the last action chosen in the current stage 2 state).
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers: perseveration bonus for stage 1 [-2,2]
    pers_stage2: perseveration bonus for stage 2 [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers, pers_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action1 != -1:
            pers_bonus[last_action1] = 1.0
            
        q_net_total = beta_1 * q_net + pers * pers_bonus
        
        exp_q1 = np.exp(q_net_total - np.max(q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            pers2_bonus = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2_bonus[last_action2[state_idx]] = 1.0
                
            q_stage2_total = beta_2 * q_stage2_mf[state_idx] + pers_stage2 * pers2_bonus
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            last_action2[state_idx] = action_2[trial]
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-Step Task Model with Asymmetric Eligibility Trace.
    The eligibility trace parameter (lambda) differs depending on whether the 
    stage 2 reward prediction error is positive or negative.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam_pos: eligibility trace for positive prediction errors [0,1]
    lam_neg: eligibility trace for negative prediction errors [0,1]
    pers: perseveration bonus for stage 1 [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action1 != -1:
            pers_bonus[last_action1] = 1.0
            
        q_net_total = beta_1 * q_net + pers * pers_bonus
        
        exp_q1 = np.exp(q_net_total - np.max(q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            lam = lam_pos if delta_stage2 > 0 else lam_neg
            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-Step Task Model with Transition-Dependent Learning Rates.
    The learning rate for stage 1 updates differs depending on whether the 
    transition experienced was common or rare.
    
    Parameters:
    alpha_common: stage 1 learning rate after common transitions [0,1]
    alpha_rare: stage 1 learning rate after rare transitions [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers: perseveration bonus for stage 1 [-2,2]
    """
    alpha_common, alpha_rare, alpha_2, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action1 != -1:
            pers_bonus[last_action1] = 1.0
            
        q_net_total = beta_1 * q_net + pers * pers_bonus
        
        exp_q1 = np.exp(q_net_total - np.max(q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
            
            # Action value updating
            is_common = (action_1[trial] == state_idx)
            alpha_1 = alpha_common if is_common else alpha_rare
            
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
