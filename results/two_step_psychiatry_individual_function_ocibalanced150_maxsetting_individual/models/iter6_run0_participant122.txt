Here are 3 new cognitive models that explore different mechanisms for learning and decision-making in the two-step task. 

### Model 1: Dual-Stage Perseveration Model
This model introduces separate inverse temperatures (stochasticity) and choice perseveration (stickiness) for both Stage 1 and Stage 2 decisions. Participants often exhibit different levels of exploration and motor-level repetition at different stages of a task.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual-Stage Perseveration Model
    Includes separate inverse temperatures for stage 1 and stage 2, and independent 
    perseveration (stickiness) bonuses for repeating choices at both stages.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta_1: [0,10] - inverse temperature for stage 1 choice stochasticity
    beta_2: [0,10] - inverse temperature for stage 2 choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    st_1: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    st_2: [-5,5] - perseveration bonus for repeating the previous stage 2 action
    lam: [0,1] - eligibility trace parameter linking stage 2 reward to stage 1
    """
    import numpy as np
    learning_rate, beta_1, beta_2, w, st_1, st_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += st_1
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta_1 * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            q_stage2_net = np.copy(q_stage2_mf[state_idx])
            if last_action_2 != -1:
                q_stage2_net[last_action_2] += st_2
                
            max_q2 = np.max(q_stage2_net)
            exp_q2 = np.exp(beta_2 * (q_stage2_net - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        if action_2[trial] != -1:
            last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition Matrix Learning Model
Rather than assuming the participant perfectly knows and fixes the 0.7/0.3 transition probabilities, this model assumes the participant learns the environmental structure (the transition matrix) dynamically from experience.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition Matrix Learning Model
    Instead of assuming a fixed transition matrix, the agent learns the transition 
    probabilities from experience dynamically using a separate learning rate.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    lr_tr: [0,1] - learning rate for updating transition probabilities
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    st: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter
    """
    import numpy as np
    learning_rate, lr_tr, beta, w, st, lam = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix as uniform (0.5 for both states)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += st
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Update the empirical transition matrix based on observed transition
        transition_matrix[action_1[trial], state_idx] += lr_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += lr_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: UCB Directed Exploration Model
This model adds an Upper Confidence Bound (UCB) term to Stage 1 decision-making. Instead of purely random exploration driven by the softmax temperature, the agent receives an explicit mathematical bonus for choosing actions that have been sampled less frequently over time.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: UCB Directed Exploration Model
    Incorporates an explicit Upper Confidence Bound (UCB) exploration bonus for Stage 1 
    actions, encouraging the participant to sample less frequently chosen actions.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    st: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter
    c_ucb: [0,10] - exploration bonus parameter (Upper Confidence Bound scaling)
    """
    import numpy as np
    learning_rate, beta, w, st, lam, c_ucb = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    
    action_counts = np.zeros(2)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add pure perseveration
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += st
            
        # Add UCB exploration bonus
        total_trials = trial + 1
        for a in range(2):
            bonus = c_ucb * np.sqrt(np.log(total_trials) / (action_counts[a] + 1))
            q_stage1_net[a] += bonus
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            max_q2 = np.max(q_stage2_mf[state_idx])
            exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        action_counts[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
