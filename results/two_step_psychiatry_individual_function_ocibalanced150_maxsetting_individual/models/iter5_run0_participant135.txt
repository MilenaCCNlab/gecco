Here are 3 new cognitive models based on the participant data and the feedback. The participant showed strong reward-dependent perseveration at Stage 1 (as seen in the best model) but also exhibited sequences of repeatedly choosing the same alien when returning to the same planet (e.g., trials 62-66). Therefore, Models 1 and 2 introduce a Stage 2 perseveration parameter (`pers2`) while removing either the eligibility trace (`lam`) or `decay` to strictly remain within the 8-parameter limit. Model 3 explores asymmetric learning rates for positive and negative prediction errors instead of an eligibility trace.

### Model 1
This model includes value decay, reward-dependent Stage 1 perseveration, and Stage 2 perseveration, but drops the eligibility trace (`lam`).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with decay, reward-dependent stage 1 perseveration, and stage 2 perseveration.
    Drops the eligibility trace (lam) to accommodate stage 2 perseveration within 8 parameters.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating values.
    decay: [0, 1] - Forgetting/decay rate for unchosen actions.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pers_rew: [0, 5] - Perseveration bonus if previous trial was rewarded.
    pers_unrew: [0, 5] - Perseveration bonus if previous trial was unrewarded.
    pers2: [0, 5] - Stage 2 perseveration bonus for repeating the same alien choice.
    """
    alpha, decay, beta1, beta2, w, pers_rew, pers_unrew, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1
    last_state = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                pers_bonus1[last_action1] = pers_rew
            else:
                pers_bonus1[last_action1] = pers_unrew
            
        logits1 = beta1 * q_net1 + pers_bonus1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus2 = np.zeros(2)
        if last_state == state_idx and last_action2 != -1:
            pers_bonus2[last_action2] = pers2
            
        logits2 = beta2 * q_stage2_mf[state_idx] + pers_bonus2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # decay unchosen values
        unchosen_stage1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_stage1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action1 = action_1[trial]
        last_action2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2
This model retains the eligibility trace (`lam`) and introduces Stage 2 perseveration, but drops value `decay` to keep parameters at 8.

```python
import numpy as np

    Two-Step Model with eligibility trace, reward-dependent stage 1 perseveration, and stage 2 perseveration.
    Drops decay to accommodate stage 2 perseveration within 8 parameters.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating values.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace for stage 1 MF update.
    pers_rew: [0, 5] - Perseveration bonus if previous trial was rewarded.
    pers_unrew: [0, 5] - Perseveration bonus if previous trial was unrewarded.

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action2 = action_2[trial]

        last_reward = reward[trial]

    return log_loss
```

### Model 3
This model investigates asymmetric learning rates for positive vs. negative prediction errors (which can often capture shifts in choice behavior after unexpected unrewarded outcomes) alongside value decay and reward-dependent Stage 1 perseveration.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with asymmetric learning rates (positive/negative PE), decay, and reward-dependent stage 1 perseveration.
    Drops eligibility trace to fit within 8 parameters.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    decay: [0, 1] - Forgetting/decay rate for unchosen actions.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pers_rew: [0, 5] - Perseveration bonus if previous trial was rewarded.
    pers_unrew: [0, 5] - Perseveration bonus if previous trial was unrewarded.
    """
    alpha_pos, alpha_neg, decay, beta1, beta2, w, pers_rew, pers_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                pers_bonus1[last_action1] = pers_rew
            else:
                pers_bonus1[last_action1] = pers_unrew
            
        logits1 = beta1 * q_net1 + pers_bonus1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # decay unchosen values
        unchosen_stage1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_stage1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
