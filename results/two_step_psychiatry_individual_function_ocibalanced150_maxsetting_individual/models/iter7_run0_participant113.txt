Here are three new cognitive models based on the findings from the best model so far, but introducing variations on learning rates (stage-specific and valence-specific) and inverse temperatures (stage-specific).

### Model 1: Stage-Specific Learning and Inverse Temperatures with Choice Kernels
This model extends the best model by allowing independent learning rates (`alpha_1`, `alpha_2`) and inverse temperatures (`beta_1`, `beta_2`) for the first and second stages. It retains the exponential choice kernel to model action history.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF RL with separate stage learning rates and betas, and a choice kernel.
    
    Bounds:
    alpha_1: [0, 1]
    alpha_2: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    alpha_c: [0, 1]
    beta_c: [0, 5]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, alpha_c, beta_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta_1 * q_net + beta_c * ck_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + beta_c * ck_2[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        ck_1 *= (1 - alpha_c)
        ck_1[action_1[trial]] += alpha_c
        
        ck_2 *= (1 - alpha_c)
        ck_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Stage-Specific Choice Kernels
This model introduces asymmetric learning rates for positive and negative prediction errors (`alpha_pos`, `alpha_neg`) to capture differences in how participants update values following better-than-expected versus worse-than-expected outcomes, combined with stage-specific choice kernel weights (`beta_c1`, `beta_c2`).

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF RL with asymmetric learning rates for positive and negative prediction errors,
    and stage-specific choice kernels.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    alpha_c: [0, 1]
    beta_c1: [0, 5]
    beta_c2: [0, 5]
    """
    alpha_pos, alpha_neg, beta, w, lam, alpha_c, beta_c1, beta_c2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net + beta_c1 * ck_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + beta_c2 * ck_2[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1_eff = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1_eff * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2_eff = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2_eff * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_1_eff * lam * delta_stage2

        ck_1 *= (1 - alpha_c)
        ck_1[action_1[trial]] += alpha_c
        
        ck_2 *= (1 - alpha_c)
        ck_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Single Learning Rate with Stage-Specific Betas and Choice Kernels
This model simplifies learning into a single `alpha` but provides maximum flexibility in decision-making by having independent inverse temperatures (`beta_1`, `beta_2`) and choice kernel weights (`beta_c1`, `beta_c2`) for both decision stages.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF RL with single learning rate, but separate stage betas and separate stage choice kernel weights.
    
    Bounds:
    alpha: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    alpha_c: [0, 1]
    beta_c1: [0, 5]
    beta_c2: [0, 5]

    n_trials = len(action_1)

    p_choice_2 = np.zeros(n_trials)

    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
```
