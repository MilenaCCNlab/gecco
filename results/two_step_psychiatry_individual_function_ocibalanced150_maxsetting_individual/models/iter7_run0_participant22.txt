def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Separate Learning Rates for Positive/Negative RPEs at Stage 1.
    
    Parameters:
    alpha_1_pos: [0, 1] - Stage 1 learning rate for positive RPEs.
    alpha_1_neg: [0, 1] - Stage 1 learning rate for negative RPEs.
    alpha_2: [0, 1] - Stage 2 learning rate.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    """
    alpha_1_pos, alpha_1_neg, alpha_2, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
            
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1_eff = alpha_1_pos if delta_stage1 > 0 else alpha_1_neg
        q_stage1_mf[action_1[trial]] += alpha_1_eff * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for choice 1
        alpha_1_lam_eff = alpha_1_pos if delta_stage2 > 0 else alpha_1_neg
        q_stage1_mf[action_1[trial]] += alpha_1_lam_eff * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Separate Betas and Perseveration for Both Stages.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for Stage 1 updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta_1: [0, 10] - Inverse temperature for Stage 1 choice.
    beta_2: [0, 10] - Inverse temperature for Stage 2 choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p_1: [-5, 5] - Perseveration parameter for Stage 1 choice.
    p_2: [-5, 5] - Perseveration parameter for Stage 2 choice (state-specific).
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1
    last_choice_2 = np.full(2, -1)

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_1 = np.zeros(2)
        if last_choice_1 != -1:
            pers_1[last_choice_1] = 1
            
        q_stage1_combined = q_stage1_net + p_1 * pers_1
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta_1 * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        pers_2 = np.zeros(2)
        if last_choice_2[state_idx] != -1:
            pers_2[last_choice_2[state_idx]] = 1

        q_stage2_combined = q_stage2_mf[state_idx] + p_2 * pers_2
        q_stage2_shifted = q_stage2_combined - np.max(q_stage2_combined)
        exp_q2 = np.exp(beta_2 * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        last_choice_1 = action_1[trial]
        last_choice_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Transition-Dependent Eligibility Traces.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for Stage 1 updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam_common: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE for common transitions.
    lam_rare: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE for rare transitions.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    """
    alpha_1, alpha_2, beta, w, lam_common, lam_rare, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
            
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Transition-dependent eligibility trace update for choice 1
        if action_1[trial] == state_idx:
            lam = lam_common
        else:
            lam = lam_rare

        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
