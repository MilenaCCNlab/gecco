def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) with Win-Stay/Lose-Shift for Stage 2.
    
    This model adds a specialized mechanism for stage 2 where stickiness depends on 
    whether the previous visit to that state was rewarded (win-stay) or unrewarded (lose-stay).
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    stickiness_1: [-5, 5] - Tendency to repeat the previous first-stage choice.
    win_stay_2: [-5, 5] - Tendency to repeat stage 2 choice if previously rewarded in that state.
    lose_stay_2: [-5, 5] - Tendency to repeat stage 2 choice if previously unrewarded in that state.
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness_1, win_stay_2, lose_stay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    last_reward_2 = np.array([0.0, 0.0])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        if last_action_2[state_idx] != -1:
            if last_reward_2[state_idx] > 0:
                logits_2[last_action_2[state_idx]] += win_stay_2
            else:
                logits_2[last_action_2[state_idx]] += lose_stay_2
                
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        last_reward_2[state_idx] = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) with Choice Trace (accumulating stickiness).
    
    Instead of just looking at the single previous trial (t-1), this model assumes that 
    choices leave a trace that decays over time. This captures long-term perseveration blocks.
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    trace_decay: [0, 1] - Decay rate for the choice traces.
    trace_weight_1: [-5, 5] - Weight of the stage 1 choice trace on logits.
    trace_weight_2: [-5, 5] - Weight of the stage 2 choice trace on logits.
    """
    learning_rate, beta_1, beta_2, w, lam, trace_decay, trace_weight_1, trace_weight_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + trace_weight_1 * choice_trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :] + trace_weight_2 * choice_trace_2[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # choice trace updates
        choice_trace_1 *= trace_decay
        choice_trace_1[action_1[trial]] += 1.0
        
        choice_trace_2 *= trace_decay
        choice_trace_2[state_idx, action_2[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) with separate decay rates for Stage 1 and Stage 2 Q-values.
    
    This model tests if Q-values of unchosen options decay towards 0 at different speeds 
    for the abstract first stage (spaceships) versus the concrete second stage (aliens).
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    decay_1: [0, 1] - Rate at which unchosen stage 1 Q-values decay towards 0.
    decay_2: [0, 1] - Rate at which unchosen stage 2 Q-values decay towards 0.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    stickiness_1: [-5, 5] - Tendency to repeat the previous first-stage choice.
    stickiness_2: [-5, 5] - Tendency to repeat the previous second-stage choice in the same state.
    """
    learning_rate, decay_1, decay_2, beta, w, lam, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx, :]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1.0 - decay_1)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1.0 - decay_2)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, :] *= (1.0 - decay_2)
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
