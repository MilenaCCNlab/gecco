def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for stage 1 and stage 2, transition learning, and perseveration.
    Parameters:
    lr_1: Learning rate for stage 1 Q-values [0,1]
    lr_2: Learning rate for stage 2 Q-values [0,1]
    lr_trans: Learning rate for transition probabilities [0,1]
    beta_1: Inverse temperature for stage 1 [0,10]
    beta_2: Inverse temperature for stage 2 [0,10]
    lam: Eligibility trace parameter [0,1]
    w: Model-based weight [0,1]
    p_pers: Choice perseveration bonus [-5,5]
    """
    lr_1, lr_2, lr_trans, beta_1, beta_2, lam, w, p_pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent_1 = beta_1 * q_stage1_net
        if last_action != -1:
            exponent_1[last_action] += p_pers
            
        max_exp_1 = np.max(exponent_1)
        exp_q1 = np.exp(exponent_1 - max_exp_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        exponent_2 = beta_2 * q_stage2_mf[state_idx]
        max_exp_2 = np.max(exponent_2)
        exp_q2 = np.exp(exponent_2 - max_exp_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # transition matrix update
        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += lr_trans * (1.0 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += lr_trans * (0.0 - transition_matrix[a1, 1 - s2])

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[a1]
        q_stage1_mf[a1] += lr_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2

        q_stage1_mf[a1] += lr_1 * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition learning, value decay, and separate perseveration for both stages (no eligibility trace).
    Parameters:
    learning_rate: Learning rate for Q-values [0,1]
    lr_trans: Learning rate for transition probabilities [0,1]
    decay_rate: Rate at which Q-values decay towards 0 [0,1]
    beta_1: Inverse temperature for stage 1 [0,10]
    beta_2: Inverse temperature for stage 2 [0,10]
    w: Model-based weight [0,1]
    p_pers_1: Choice perseveration bonus for stage 1 [-5,5]
    p_pers_2: Choice perseveration bonus for stage 2 [-5,5]
    """
    learning_rate, lr_trans, decay_rate, beta_1, beta_2, w, p_pers_1, p_pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent_1 = beta_1 * q_stage1_net
        if last_action_1 != -1:
            exponent_1[last_action_1] += p_pers_1
            
        max_exp_1 = np.max(exponent_1)
        exp_q1 = np.exp(exponent_1 - max_exp_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        exponent_2 = beta_2 * q_stage2_mf[state_idx]
        if last_action_2 != -1:
            exponent_2[last_action_2] += p_pers_2
            
        max_exp_2 = np.max(exponent_2)
        exp_q2 = np.exp(exponent_2 - max_exp_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # transition matrix update
        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += lr_trans * (1.0 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += lr_trans * (0.0 - transition_matrix[a1, 1 - s2])

        # decay
        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = a1
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition learning, value decay, and a continuous choice trace for stage 1.
    Parameters:
    learning_rate: Learning rate for Q-values [0,1]
    lr_trans: Learning rate for transition probabilities [0,1]
    decay_rate: Rate at which Q-values decay towards 0 [0,1]
    beta_1: Inverse temperature for stage 1 [0,10]
    beta_2: Inverse temperature for stage 2 [0,10]
    lam: Eligibility trace parameter [0,1]
    w: Model-based weight [0,1]
    choice_trace_weight: Weight of the continuous choice trace on stage 1 choices [-5,5]
    """
    learning_rate, lr_trans, decay_rate, beta_1, beta_2, lam, w, choice_trace_weight = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent_1 = beta_1 * q_stage1_net + choice_trace_weight * choice_trace
            
        max_exp_1 = np.max(exponent_1)
        exp_q1 = np.exp(exponent_1 - max_exp_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        exponent_2 = beta_2 * q_stage2_mf[state_idx]
        max_exp_2 = np.max(exponent_2)
        exp_q2 = np.exp(exponent_2 - max_exp_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # transition matrix update
        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += lr_trans * (1.0 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += lr_trans * (0.0 - transition_matrix[a1, 1 - s2])

        # decay
        q_stage1_mf *= (1 - decay_rate)
        q_stage2_mf *= (1 - decay_rate)

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Choice trace update (decay by 0.5, add 1 to chosen)
        choice_trace *= 0.5
        choice_trace[a1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
