Here are 3 new cognitive models exploring different mechanisms for the two-step decision-making task. 

### Model 1: Reward-dependent Model-Based Weight
This model tests the hypothesis that participants might rely more on model-based planning (understanding the transition structure) after a win compared to after a loss, or vice versa. It replaces a single model-based weight `w` with `w_win` and `w_lose`.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Reward-dependent Model-Based Weight
    The weight given to model-based vs model-free Q-values depends on whether 
    the previous trial was rewarded (w_win) or unrewarded (w_lose).
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Retention rate for unchosen Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w_win: [0, 1] - Weighting of model-based vs model-free values after a win
    w_lose: [0, 1] - Weighting of model-based vs model-free values after a loss
    lam: [0, 1] - Eligibility trace parameter
    pers: [-5, 5] - Perseveration for stage 1
    """
    alpha, decay, beta_1, beta_2, w_win, w_lose, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = 0.5 # Default for the first trial
            
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + pers * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= decay
        
        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] *= decay
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= decay
        q_stage2_mf[other_state, 1] *= decay
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Dual Perseveration with Subjective Penalty for Zero Reward
This model takes the successful `penalty_zero` mechanism and combines it with explicit perseveration for both Stage 1 and Stage 2 choices, omitting the decay parameter to remain within the 8-parameter limit.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dual Perseveration with Subjective Penalty for Zero Reward
    Introduces perseveration for both stage 1 and stage 2 choices, along with a 
    subjective penalty for receiving 0 coins. Omits decay to maintain parameter limit.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers_1: [-5, 5] - Perseveration for stage 1
    pers_2: [-5, 5] - Perseveration for stage 2
    penalty_zero: [0, 2] - Penalty applied when 0 coins are received
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2, penalty_zero = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + pers_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep_2 = np.zeros(2)
        if last_state == state_idx and last_action_2 != -1:
            rep_2[last_action_2] = 1

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        subj_reward = 1.0 if reward[trial] == 1 else -penalty_zero
        delta_stage2 = subj_reward - q_stage2_mf[state_idx, action_2[trial]]

        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Asymmetric Learning Rates with Decay and Zero Penalty
This model merges the `penalty_zero` mechanism with asymmetric learning rates (`alpha_pos` and `alpha_neg`) and `decay`. It tests if participants have inherently different updating speeds for positive vs negative prediction errors while continuing to treat zero coins subjectively as a penalty.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Learning Rates with Decay and Zero Penalty
    Uses different learning rates for positive and negative prediction errors,
    combined with a decay mechanism and subjective penalty for zero reward.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    decay: [0, 1] - Retention rate for unchosen Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    penalty_zero: [0, 2] - Penalty applied when 0 coins are received
    """
    alpha_pos, alpha_neg, decay, beta_1, beta_2, w, lam, penalty_zero = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits_1 = beta_1 * q_stage1_net
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        subj_reward = 1.0 if reward[trial] == 1 else -penalty_zero
        delta_stage2 = subj_reward - q_stage2_mf[state_idx, action_2[trial]]

        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2

        # decay for unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= decay
        
        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] *= decay
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= decay
        q_stage2_mf[other_state, 1] *= decay

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
