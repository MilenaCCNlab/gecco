def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Learned Transition Matrix + Stage 1 Choice Kernel
    
    This model relaxes the assumption of a fixed transition matrix (commonly assumed 
    to be 0.7/0.3) and instead learns it dynamically over time via a Rescorla-Wagner 
    update. It also includes a choice kernel for stage 1 to capture autocorrelation 
    in choice behavior.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    ck_lr: [0, 1] - Learning rate for updating the choice kernel.
    ck_weight1: [-5, 5] - Weight of the choice kernel in stage 1 action selection.
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix.
    """
    learning_rate, beta1, beta2, w, lambda_, ck_lr, ck_weight1, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix at 0.5 to allow learning from scratch
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf) + ck_weight1 * ck_1
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            ck_1[action_1[trial]] += ck_lr
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update transition matrix probabilities based on observed transition
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1 - transition_matrix[action_1[trial], state_idx]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

        ck_1 = (1 - ck_lr) * ck_1
        ck_1[action_1[trial]] += ck_lr

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Asymmetric Learning Rates + Stage 1 Choice Kernel
    
    This model introduces asymmetric learning rates for positive and negative 
    prediction errors, which is a common phenomenon in reinforcement learning 
    tasks. It retains a choice kernel for stage 1 to account for basic choice 
    autocorrelation.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    ck_lr: [0, 1] - Learning rate for updating the choice kernel.
    ck_weight1: [-5, 5] - Weight of the choice kernel in stage 1 action selection.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lambda_, ck_lr, ck_weight1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)

            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2

        ck_1 = (1 - ck_lr) * ck_1
        ck_1[action_1[trial]] += ck_lr

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Q-Value Forgetting + Stage 1 Choice Kernel
    
    This model assumes that Q-values for unchosen actions are not static but 
    decay toward zero over time, representing a forgetting process. It also 
    includes a choice kernel for stage 1 to capture choice autocorrelation.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    decay: [0, 1] - Decay rate for unchosen Q-values.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    ck_lr: [0, 1] - Learning rate for updating the choice kernel.
    ck_weight1: [-5, 5] - Weight of the choice kernel in stage 1 action selection.
    """
    learning_rate, decay, beta1, beta2, w, lambda_, ck_lr, ck_weight1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            # Decay all Q-values since no action was chosen
            q_stage1_mf *= (1 - decay)
            q_stage2_mf *= (1 - decay)
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf) + ck_weight1 * ck_1
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            ck_1[action_1[trial]] += ck_lr
            # Decay all stage 2 Q-values and unchosen stage 1
            q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
            q_stage2_mf *= (1 - decay)
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

        # Forgetting for unchosen options
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, :] *= (1 - decay)

        ck_1 = (1 - ck_lr) * ck_1
        ck_1[action_1[trial]] += ck_lr

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
