### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model tests whether participants update their beliefs at different speeds for the first choice (spaceship) versus the second choice (alien). It retains the outcome-dependent choice stickiness for stage 1.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate learning rates for stage 1 and stage 2,
    separate inverse temperatures, and outcome-dependent choice stickiness for stage 1.
    
    Bounds:
    alpha1: [0,1] - learning rate for stage 1
    alpha2: [0,1] - learning rate for stage 2
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - model-based weight
    lam: [0,1] - eligibility trace parameter
    st_win: [-5,5] - stickiness for stage 1 if previous trial was rewarded
    st_lose: [-5,5] - stickiness for stage 1 if previous trial was unrewarded
    """
    alpha1, alpha2, beta1, beta2, w, lam, st_win, st_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                stickiness_bonus[last_action_1] = st_win
            else:
                stickiness_bonus[last_action_1] = st_lose
            
        val_1 = q_net + stickiness_bonus
        exp_q1 = np.exp(beta1 * (val_1 - np.max(val_1)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        val_2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (val_2 - np.max(val_2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning (Positive vs. Negative Prediction Errors)
This model tests whether participants learn differently from positive surprises (better than expected) versus negative surprises (worse than expected), which is a common pattern in reward learning tasks.

### Model 3: Multi-Stage Choice Stickiness
This model investigates whether participants exhibit perseverative behavior not only at the first stage (spaceship selection) but also independently at the second stage (alien selection depending on the planet they arrive at).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with outcome-dependent choice stickiness for stage 1 (st_win/st_lose)
    and state-specific choice stickiness for stage 2 (st2).
    
    Bounds:
    alpha: [0,1] - learning rate
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - model-based weight
    lam: [0,1] - eligibility trace parameter
    st_win: [-5,5] - stickiness for stage 1 if previous trial was rewarded
    st_lose: [-5,5] - stickiness for stage 1 if previous trial was unrewarded
    st2: [-5,5] - choice stickiness for stage 2
    """
    alpha, beta1, beta2, w, lam, st_win, st_lose, st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = 0
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                stickiness_bonus_1[last_action_1] = st_win
            else:
                stickiness_bonus_1[last_action_1] = st_lose
            
        val_1 = q_net + stickiness_bonus_1
        exp_q1 = np.exp(beta1 * (val_1 - np.max(val_1)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        stickiness_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_bonus_2[last_action_2[state_idx]] = st2

        val_2 = q_stage2_mf[state_idx] + stickiness_bonus_2
        exp_q2 = np.exp(beta2 * (val_2 - np.max(val_2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
