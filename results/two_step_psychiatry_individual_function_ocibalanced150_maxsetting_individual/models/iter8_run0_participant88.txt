def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: 6 parameters. Simplifies the best model by removing the eligibility trace (lam).
    Stage 1 model-free values are updated only via the Stage 1 prediction error (the difference 
    between the chosen Stage 2 value and Stage 1 value), without direct update from the Stage 2 reward.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating action values.
    decay_rate: [0,1] - Rate at which unchosen Stage 2 actions decay toward 0.5.
    beta: [0,10] - Inverse temperature for both Stage 1 and Stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) systems.
    stickiness: [0,10] - Choice perseveration bonus for Stage 1.
    bias_action0: [0,10] - Intrinsic bias/bonus for choosing Action 0 in Stage 1.
    """
    learning_rate, decay_rate, beta, w, stickiness, bias_action0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += stickiness
        logits_1[0] += bias_action0
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += learning_rate * delta_stage2
                else:
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: 8 parameters. Extends the best model by allowing separate learning rates 
    for Stage 1 and Stage 2 updates. This can capture differences in learning speed 
    between the two stages while retaining decay, perseveration, and action bias.
    
    Parameters:
    learning_rate_1: [0,1] - Learning rate for Stage 1 prediction errors.
    learning_rate_2: [0,1] - Learning rate for Stage 2 prediction errors.
    decay_rate: [0,1] - Rate at which unchosen Stage 2 actions decay toward 0.5.
    beta: [0,10] - Inverse temperature for both Stage 1 and Stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) systems.
    stickiness: [0,10] - Choice perseveration bonus for Stage 1.
    lam: [0,1] - Eligibility trace for Stage 2 reward updating Stage 1 value.
    bias_action0: [0,10] - Intrinsic bias/bonus for choosing Action 0 in Stage 1.
    """
    learning_rate_1, learning_rate_2, decay_rate, beta, w, stickiness, lam, bias_action0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += stickiness
        logits_1[0] += bias_action0
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1

        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += learning_rate_2 * delta_stage2
                else:
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += learning_rate_1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: 8 parameters. Extends the best model by allowing asymmetric learning rates 
    for positive and negative prediction errors. This captures valence-dependent biases in learning.
    
    Parameters:
    learning_rate_pos: [0,1] - Learning rate for positive prediction errors.
    learning_rate_neg: [0,1] - Learning rate for negative prediction errors.
    decay_rate: [0,1] - Rate at which unchosen Stage 2 actions decay toward 0.5.
    beta: [0,10] - Inverse temperature for both Stage 1 and Stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) systems.
    stickiness: [0,10] - Choice perseveration bonus for Stage 1.
    lam: [0,1] - Eligibility trace for Stage 2 reward updating Stage 1 value.
    bias_action0: [0,10] - Intrinsic bias/bonus for choosing Action 0 in Stage 1.
    """
    learning_rate_pos, learning_rate_neg, decay_rate, beta, w, stickiness, lam, bias_action0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:

        logits_1[0] += bias_action0

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = learning_rate_pos if delta_stage1 >= 0 else learning_rate_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1

        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

            for a in range(2):

                else:
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += lr_stage2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
