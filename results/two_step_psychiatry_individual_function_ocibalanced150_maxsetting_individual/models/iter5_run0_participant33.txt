Here are 3 new cognitive models that explore alternative mechanisms for how participants might make choices and update values in the two-step task. 

### Model 1: Choice Trace (Autocorrelation) Model
Instead of just repeating the immediate last action (a simple "stickiness" parameter), this model maintains an exponentially decaying trace of past stage 1 choices. This captures longer-term choice autocorrelation, where participants might drift toward one spaceship over multiple trials.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Choice Trace (Autocorrelation).
    
    Instead of only tracking the immediate last action, this model maintains 
    an exponentially decaying trace of past stage 1 choices.
    
    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate for updating Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi: [-5, 5] - Weight of the choice trace on stage 1 preferences.
        alpha_c: [0, 1] - Decay/update rate of the choice trace.
    """
    alpha, beta, w, lambda_, pi, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf + pi * choice_trace
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Update choice trace
        choice_trace = (1 - alpha_c) * choice_trace
        choice_trace[action_1[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition-Dependent Eligibility Trace Model
Often, participants do not use rare transitions to update stage 1 values as much because they attribute the outcome to the rare "jump" rather than their initial choice. This model assigns separate eligibility traces depending on whether the transition experienced was common or rare.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Transition-Dependent Eligibility Trace.
    
    This model allows the eligibility trace (lambda) to differ depending on whether 
    the transition from stage 1 to stage 2 was common or rare.
    
    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate for updating Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_common: [0, 1] - Eligibility trace for common transitions.
        lambda_rare: [0, 1] - Eligibility trace for rare transitions.
        pi: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1.
    """
    alpha, beta, w, lambda_common, lambda_rare, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Transition-dependent eligibility trace
        is_common = (action_1[trial] == state_idx)
        current_lambda = lambda_common if is_common else lambda_rare
        
        q_stage1_mf[action_1[trial]] += alpha * current_lambda * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Action vs Goal Perseveration Model
This model disentangles two types of perseveration at stage 1: motor-action perseveration (repeating the same spaceship) and goal perseveration (attempting to return to the same planet). If a participant wants to return to the planet they were just on, they will exhibit a bias towards the action that commonly leads to that planet.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Action and Goal Perseveration.
    
    Includes standard action stickiness (repeating the previous stage 1 action)
    as well as goal stickiness (preferring the stage 1 action that commonly leads 
    to the planet visited on the previous trial).
    
    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate for updating Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi_action: [-5, 5] - Action perseverance (stickiness to previous stage 1 action).
        pi_goal: [-5, 5] - Goal perseverance (stickiness to previous stage 2 state).
    """
    alpha, beta, w, lambda_, pi_action, pi_goal = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None
    last_state = None

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi_action
            
        if last_state is not None:
            # Action 0 commonly leads to state 0, Action 1 commonly leads to state 1.
            q_stage1_net[last_state] += pi_goal
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action = action_1[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
