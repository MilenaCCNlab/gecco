def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Hybrid RL with Separate Stage 1 and Stage 2 Perseverance
    
    This model builds upon the standard hybrid model by adding a perseverance (stickiness) 
    parameter not just for the first-stage choice, but also for the second-stage choice. 
    Participant data shows long streaks of choosing the same alien on the same planet.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = pure MB, 0 = pure MF).
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p1: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    p2: [-5, 5] - Perseverance bonus for repeating the previous second-stage action in the same state.
    """
    learning_rate, beta1, beta2, w, lambda_, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_act1 = -1
    last_act2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act1 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act1 != -1:
            V1[last_act1] += p1
        
        q_stage1_mb = V1 - np.max(V1)
        beta = 1.0 
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act1 = action_1[trial]
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        if last_act2[state_idx] != -1:
            V2[last_act2[state_idx]] += p2
            
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_act1 = action_1[trial]
        last_act2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Hybrid RL with Unchosen Option Decay
    
    This model incorporates a decay mechanism where the Q-values of unchosen options 
    (both at stage 1 and stage 2) gradually decay towards a neutral expected value of 0.5. 
    This accounts for forgetting or increased uncertainty about unchosen actions over time.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta_param: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = pure MB, 0 = pure MF).
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    decay: [0, 1] - Rate at which unchosen Q-values decay towards 0.5.
    """
    learning_rate, beta_param, w, lambda_, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_act = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta_param * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act != -1:
            V1[last_act] += p
            
        q_stage1_mb = V1 - np.max(V1)
        beta = 1.0 
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act = action_1[trial]
            continue
            
        V2 = beta_param * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        # Decay unchosen options towards 0.5
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * 0.5
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5
                    
        last_act = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 3: Hybrid RL with Transition Matrix Learning
    
    Instead of assuming the transition matrix is fixed at 0.7/0.3, this model assumes 
    the participant dynamically updates their internal model of the transition probabilities 
    based on the observed state transitions.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta_param: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = pure MB, 0 = pure MF).
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseverance bonus for repeating the previous first-stage action.
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix based on observed transitions.
    """
    learning_rate, beta_param, w, lambda_, p, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_act = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_act = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta_param * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        if last_act != -1:
            V1[last_act] += p
            
        q_stage1_mb = V1 - np.max(V1)
        beta = 1.0 
        
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Update transition matrix based on observation
        transition_matrix[action_1[trial], state_idx] = (1 - alpha_trans) * transition_matrix[action_1[trial], state_idx] + alpha_trans * 1.0
        transition_matrix[action_1[trial], 1 - state_idx] = (1 - alpha_trans) * transition_matrix[action_1[trial], 1 - state_idx] + alpha_trans * 0.0

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_act = action_1[trial]
            continue
            
        V2 = beta_param * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_act = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
