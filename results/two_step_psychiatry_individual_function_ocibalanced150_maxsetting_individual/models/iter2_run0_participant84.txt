def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 1: Model-Free and Model-Based mixture (w).
    Stage 1 choices are determined by a weighted mixture of model-based (MB) 
    and model-free (MF) action values.
    
    Parameters and Bounds:
    learning_rate: [0, 1] - Learning rate for updating MF values.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    """
    learning_rate, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Mix MB and MF values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta * q_stage1_net
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1) # Max trick for numerical stability
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 Policy
        q2_total = beta * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 2: MB/MF mixture + Stage 1 Stickiness.
    Builds upon Model 1 by adding a perseveration (stickiness) bonus to the 
    Stage 1 action chosen in the previous trial.
    
    Parameters and Bounds:
    learning_rate: [0, 1] - Learning rate for updating MF values.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness: [0, 10] - Bonus added to the previously chosen Stage 1 action.
    """
    learning_rate, beta, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = stickiness
            
        q1_total = beta * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 Policy
        q2_total = beta * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 3: Full Daw et al. (2011) two-step model.
    Includes separate inverse temperatures for Stage 1 and Stage 2, MB/MF weighting (w), 
    Stage 1 stickiness, and an eligibility trace (lambda_) that allows Stage 2 rewards 
    to directly update Stage 1 MF values.
    
    Parameters and Bounds:
    learning_rate: [0, 1] - Learning rate for updating MF values.
    beta_1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness: [0, 10] - Bonus added to the previously chosen Stage 1 action.
    lambda_: [0, 1] - Eligibility trace scaling Stage 2 PE updates to Stage 1.
    """
    learning_rate, beta_1, beta_2, w, stickiness, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = stickiness
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 Policy
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update: Stage 2 PE updates Stage 1 MF value
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
