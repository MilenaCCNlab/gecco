Here are 3 new cognitive models that expand upon the hybrid Model-Based (MB) / Model-Free (MF) reinforcement learning framework. They introduce new mechanisms to better capture variations in participant behavior, such as differences in exploration/exploitation between stages, asymmetric learning from positive vs. negative feedback, and perseveration across both stages of the task.

### Model 1: Separate Temperatures for Stage 1 and Stage 2
This model hypothesizes that the participant might exhibit different levels of exploration/exploitation (randomness) in the first choice compared to the second choice. It uses two separate inverse temperature parameters (`beta_1` and `beta_2`).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF RL with separate temperatures for Stage 1 and Stage 2.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta_1: [0,10] - Inverse temperature for Stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for Stage 2 softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Choice stickiness (perseveration) parameter for Stage 1.
    """
    learning_rate, beta_1, beta_2, w, lam, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits1 = beta_1 * q_stage1_net + rho * stickiness
        logits1 -= np.max(logits1) # for numerical stability
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        # Stage 2 Policy
        logits2 = beta_2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates (Positive vs. Negative Prediction Errors)
This model tests the hypothesis that the participant learns differently from better-than-expected outcomes compared to worse-than-expected outcomes. It introduces separate learning rates for positive (`alpha_pos`) and negative (`alpha_neg`) prediction errors.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF RL with asymmetric learning rates for positive and negative prediction errors.
    
    Parameters:
    alpha_pos: [0,1] - Learning rate for positive prediction errors (delta > 0).
    alpha_neg: [0,1] - Learning rate for negative prediction errors (delta <= 0).
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Choice stickiness (perseveration) parameter for Stage 1.
    """
    alpha_pos, alpha_neg, beta, w, lam, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits1 = beta * q_stage1_net + rho * stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        # Stage 2 Policy
        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # Eligibility trace update applied to Stage 1 MF value
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Choice Stickiness for both Stage 1 and Stage 2
This model incorporates perseveration at the alien choice level as well. Often participants will exhibit habitual responding not just for the spaceship (Stage 1), but also for the specific alien they ask for gold (Stage 2) depending on the planet they land on. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF RL with choice stickiness for both Stage 1 and Stage 2.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho_1: [-5,5] - Choice stickiness (perseveration) parameter for Stage 1.
    rho_2: [-5,5] - Choice stickiness (perseveration) parameter for Stage 2.
    """
    learning_rate, beta, w, lam, rho_1, rho_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.full(2, -1) # Tracks the last Stage 2 action taken in each state

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_1 = np.zeros(2)
        if last_a1 != -1:
            stickiness_1[last_a1] = 1.0
            
        logits1 = beta * q_stage1_net + rho_1 * stickiness_1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        # Stage 2 Policy
        stickiness_2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            stickiness_2[last_a2[state_idx]] = 1.0

        logits2 = beta * q_stage2_mf[state_idx] + rho_2 * stickiness_2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_a2[state_idx] = action_2[trial]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
