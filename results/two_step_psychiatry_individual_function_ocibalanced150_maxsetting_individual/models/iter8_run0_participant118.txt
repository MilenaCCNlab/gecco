### Model 1: Separate Decay Rates and Betas
This model introduces separate decay mechanisms for Stage 1 and Stage 2 actions. It tests whether participants forget or discount unchosen options at different rates depending on the decision stage, while also separating the inverse temperature (beta) for each stage to account for different levels of choice stochasticity.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate decay rates for Stage 1 and Stage 2, with separate betas.
    This model assumes that the values of actions not taken on a given trial will decay 
    towards a neutral expected value of 0.5, but the rate of decay may differ between 
    the first and second stages.
    
    Parameters:
    alpha: Learning rate for updating values [0, 1]
    beta1: Inverse temperature for Stage 1 choices [0, 10]
    beta2: Inverse temperature for Stage 2 choices [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay1: Decay rate for unchosen Stage 1 actions [0, 1]
    decay2: Decay rate for unchosen Stage 2 actions [0, 1]
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, decay1, decay2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen Stage 1
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay1 * (0.5 - q_stage1_mf[unchosen_a1])

        # Decay for unchosen Stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay2 * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Win-Stay-Lose-Shift (WSLS) Bias
This model incorporates a heuristic Win-Stay-Lose-Shift tendency into the Stage 1 decision process. In addition to MB/MF learning and general decay, the participant might exhibit a bias to simply repeat the previous Stage 1 action if they were rewarded, or switch if they were not.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with decay and Win-Stay-Lose-Shift (WSLS) bias.
    This model adds a heuristic WSLS bias to Stage 1 choices, where individuals 
    are more likely to repeat the previous Stage 1 choice if rewarded, and switch 
    if not rewarded, in addition to the standard MB/MF learning and value decay.
    
    Parameters:
    alpha: Learning rate for updating values [0, 1]
    beta1: Inverse temperature for Stage 1 choices [0, 10]
    beta2: Inverse temperature for Stage 2 choices [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Decay rate for unchosen actions [0, 1]
    wsls: Win-stay-lose-shift tendency for Stage 1 [0, 10]
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, decay, wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_reward = 0
    prev_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        wsls_bias = np.zeros(2)
        if prev_a1 != -1:
            if prev_reward == 1:
                wsls_bias[prev_a1] = wsls
            else:
                wsls_bias[1 - prev_a1] = wsls
                
        exp_q1 = np.exp(beta1 * q_stage1_net + wsls_bias)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
                    
        prev_a1 = action_1[trial]
        prev_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Dynamic Transition Matrix Learning
This model relaxes the assumption that the participant perfectly maintains the fixed 0.7/0.3 transition probabilities. Instead, it allows the participant to update their internal transition matrix based on observed transitions, effectively testing if their model-based reasoning adapts to recent transition experiences.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with decay and learning of the transition matrix.
    This model relaxes the assumption of fixed transition probabilities, allowing 
    the agent to update their internal transition matrix based on observed transitions, 
    while also utilizing separate betas and value decay.
    
    Parameters:
    alpha: Learning rate for updating values [0, 1]
    beta1: Inverse temperature for Stage 1 choices [0, 10]
    beta2: Inverse temperature for Stage 2 choices [0, 10]
    w: Model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Decay rate for unchosen actions [0, 1]
    alpha_trans: Learning rate for updating transition matrix [0, 1]
    """
    import numpy as np
    alpha, beta1, beta2, w, lam, decay, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update transition matrix based on experienced state transition
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
