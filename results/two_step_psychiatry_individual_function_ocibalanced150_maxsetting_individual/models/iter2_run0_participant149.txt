Here are 3 new cognitive models that build upon the best model (which included stage 1 and stage 2 stickiness) by adding theoretically motivated mechanisms often observed in psychiatric populations: asymmetric learning rates, separate exploration temperatures, and reward-dependent stickiness (Win-stay/Lose-shift).

### Model 1: Asymmetric Learning Rates + Two-Stage Stickiness
This model separates the learning rate into `lr_pos` (for positive prediction errors) and `lr_neg` (for negative prediction errors). Individuals often show biases in how they learn from better-than-expected vs. worse-than-expected outcomes.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative prediction errors + stickiness at both stages.
    
    Bounds:
    lr_pos: [0, 1] - Learning rate for positive prediction errors
    lr_neg: [0, 1] - Learning rate for negative prediction errors
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter
    pi_1: [-5, 5] - Stickiness parameter for repeating the previous stage 1 choice
    pi_2: [-5, 5] - Stickiness parameter for repeating the previous stage 2 choice in the given state
    """
    lr_pos, lr_neg, beta, w, lam, pi_1, pi_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Skip missed trials
        if a1 == -1 or s == -1 or a2 == -1:
            continue

        # Stage 1 choice probabilities
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + pi_1 * stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 choice probabilities
        stickiness_2 = np.zeros(2)
        if last_action_2[s] != -1:
            stickiness_2[last_action_2[s]] = 1.0
            
        logits_2 = beta * q_stage2_mf[s] + pi_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Update Q-values with asymmetric learning rates
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        lr1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[a1] += lr1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        lr2 = lr_pos if delta_stage2 > 0 else lr_neg
        
        q_stage2_mf[s, a2] += lr2 * delta_stage2
        q_stage1_mf[a1] += lr2 * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[s] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Stage Temperatures + Two-Stage Stickiness
This model allows for different inverse temperatures (`beta_1` and `beta_2`) for the two stages. Participants often exhibit different levels of exploration/exploitation depending on whether they are making the high-level planning choice (stage 1) or the immediate reward choice (stage 2).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate temperatures for stage 1 and stage 2 + stickiness at both stages.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax action selection
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter
    pi_1: [-5, 5] - Stickiness parameter for repeating the previous stage 1 choice
    pi_2: [-5, 5] - Stickiness parameter for repeating the previous stage 2 choice in the given state
    """

    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)

    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]

        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[s] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Stickiness (Win-Stay / Lose-Shift)
This model replaces the static stage-1 stickiness parameter with two separate parameters: `pi_win` and `pi_lose`. This captures the heuristic tendency to repeat choices that were just rewarded (win-stay) and switch away from choices that were unrewarded (lose-shift), independent of the Q-values.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent stickiness for stage 1 (Win-stay/Lose-shift) + stage 2 stickiness.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for Q-value updates
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Mixing weight for model-based vs model-free (1 = purely MB, 0 = purely MF)
    lam: [0, 1] - Eligibility trace decay parameter
    pi_win: [-5, 5] - Stage 1 stickiness after a rewarded trial
    pi_lose: [-5, 5] - Stage 1 stickiness after an unrewarded trial
    pi_2: [-5, 5] - Stickiness parameter for repeating the previous stage 2 choice in the given state
    """
    learning_rate, beta, w, lam, pi_win, pi_lose, pi_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Skip missed trials
        if a1 == -1 or s == -1 or a2 == -1:
            continue

        # Stage 1 choice probabilities
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            # Apply different stickiness depending on if last trial was rewarded
            current_pi_1 = pi_win if last_reward > 0 else pi_lose
            stickiness_1[last_action_1] = current_pi_1
            
        logits_1 = beta * q_net + stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 choice probabilities
        stickiness_2 = np.zeros(2)
        if last_action_2[s] != -1:
            stickiness_2[last_action_2[s]] = 1.0
            
        logits_2 = beta * q_stage2_mf[s] + pi_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Update Q-values
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[s] = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
