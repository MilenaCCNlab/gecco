import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF hybrid with separate learning rates and betas for stage 1 and 2,
    choice stickiness for stage 1, and a decay parameter for unchosen action values.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: choice stickiness parameter for stage 1 [-5,5]
    decay: forgetting rate for unchosen actions [0,1]
    """
    alpha1, alpha2, beta1, beta2, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1.0
            
        logits_1 = beta1 * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]

        if act2 != -1:
            logits_2 = beta2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            q_stage1_mf[act1] += alpha1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
            q_stage2_mf[state_idx, act2] += alpha2 * delta_stage2

            q_stage1_mf[act1] += alpha1 * lam * delta_stage2

            unchosen_act1 = 1 - act1
            q_stage1_mf[unchosen_act1] *= (1 - decay)

            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == act2):
                        q_stage2_mf[s, a] *= (1 - decay)
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF hybrid with separate learning rates for positive and negative 
    prediction errors, choice stickiness for both stage 1 AND stage 2, and a 
    decay parameter for unchosen action values.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p1: choice stickiness parameter for stage 1 [-5,5]
    p2: choice stickiness parameter for stage 2 [-5,5]
    decay: forgetting rate for unchosen actions [0,1]
    """
    alpha_pos, alpha_neg, beta, w, lam, p1, p2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) 

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep1 = np.zeros(2)
        if last_action_1 != -1:
            rep1[last_action_1] = 1.0

        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        act2 = action_2[trial]
        act1 = action_1[trial]

        if act2 != -1:
            rep2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                rep2[last_action_2[state_idx]] = 1.0
                
            logits_2 = beta * q_stage2_mf[state_idx] + p2 * rep2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[act2]

            delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
            alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg

            q_stage1_mf[act1] += alpha_stage1 * lam * delta_stage2

            unchosen_act1 = 1 - act1

            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == act2):
                        q_stage2_mf[s, a] *= (1 - decay)
                        
            last_action_2[state_idx] = act2
        else:
            p_choice_2[trial] = 1.0
            
        last_action_1 = act1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
