def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for Stage 1 and Stage 2, with Stage 1 Choice Kernel.
    
    This model assumes that participants might update their values at different speeds 
    for the first and second stages of the task. Additionally, it incorporates a choice 
    kernel specifically for stage 1 to capture autocorrelation in initial choices.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1 Q-value and eligibility trace updates.
    alpha2: [0, 1] - Learning rate for stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values at stage 1.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    ck_lr: [0, 1] - Learning rate for updating the stage 1 choice kernel.
    ck_weight1: [-5, 5] - Weight of the choice kernel in stage 1 action selection.
    """
    alpha1, alpha2, beta1, beta2, w, lambda_, ck_lr, ck_weight1 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf) + ck_weight1 * ck_1
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            ck_1 = (1 - ck_lr) * ck_1
            ck_1[action_1[trial]] += ck_lr
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2

        ck_1 = (1 - ck_lr) * ck_1
        ck_1[action_1[trial]] += ck_lr
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-dependent perseveration at Stage 1.
    
    This model evaluates whether a participant's tendency to repeat their previous 
    Stage 1 choice is modulated by whether that choice previously led to a reward 
    (win-stay) or no reward (lose-shift).
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values at stage 1.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p_stay: [-5, 5] - Baseline perseveration tendency for stage 1.
    p_win_stay: [-5, 5] - Additional perseveration if the previous trial was rewarded.
    p_lose_shift: [-5, 5] - Shift (negative perseveration) if the previous trial was unrewarded.
    """
    learning_rate, beta1, beta2, w, lambda_, p_stay, p_win_stay, p_lose_shift = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action1 = -1
    prev_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        
        if prev_action1 != -1:
            pers_weight = p_stay
            if prev_reward == 1:
                pers_weight += p_win_stay
            else:
                pers_weight -= p_lose_shift
            V1[prev_action1] += pers_weight
            
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            prev_action1 = action_1[trial]
            prev_reward = 0
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

        prev_action1 = action_1[trial]
        prev_reward = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-dependent perseveration at Stage 1 and separate Stage 2 perseveration.
    
    This model assumes that a participant's tendency to repeat their stage 1 choice is 
    strongly influenced by whether the previous trial's state transition was expected 
    (common) or unexpected (rare). It also includes standard perseveration at stage 2.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values at stage 1.
    lambda_: [0, 1] - Eligibility trace decay parameter.
    p1_common: [-5, 5] - Stage 1 perseveration if the previous transition was common.
    p1_rare: [-5, 5] - Stage 1 perseveration if the previous transition was rare.
    p2: [-5, 5] - Stage 2 perseveration (if returning to the same state).
    """
    learning_rate, beta1, beta2, w, lambda_, p1_common, p1_rare, p2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action1 = -1
    prev_action2 = -1
    prev_state = -1
    prev_trans_common = True

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        V1 = beta1 * (w * q_stage1_mb + (1 - w) * q_stage1_mf)
        
        if prev_action1 != -1:
            if prev_trans_common:
                V1[prev_action1] += p1_common
            else:
                V1[prev_action1] += p1_rare
                
        V1 = V1 - np.max(V1)
        exp_q1 = np.exp(V1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            prev_action1 = action_1[trial]
            prev_action2 = -1
            prev_state = state_idx
            # Common transitions: action 0 -> state 0, action 1 -> state 1
            prev_trans_common = (state_idx == action_1[trial])
            continue
            
        V2 = beta2 * q_stage2_mf[state_idx]
        if prev_action2 != -1 and prev_state == state_idx:
            V2[prev_action2] += p2
            
        V2 = V2 - np.max(V2)
        exp_q2 = np.exp(V2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

        prev_action1 = action_1[trial]
        prev_action2 = action_2[trial]
        prev_state = state_idx
        prev_trans_common = (state_idx == action_1[trial])
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
