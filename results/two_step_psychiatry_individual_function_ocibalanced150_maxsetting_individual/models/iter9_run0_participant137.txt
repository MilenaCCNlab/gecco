Here are 3 new cognitive models that build upon the best model so far (the Daw et al. 2011 hybrid model with an exponentially decaying choice trace). Each model proposes a distinct psychological mechanism to explain the participant's behavior, while adhering to the specified parameter limits and avoiding previously tried combinations.

### Model 1: Asymmetric Eligibility Traces for Positive and Negative RPEs
This model proposes that the participant's model-free system assigns credit differently depending on whether the outcome was better or worse than expected. It uses separate eligibility traces (`lam_pos` and `lam_neg`) based on the sign of the Stage 2 reward prediction error.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) hybrid model with an exponentially decaying perseveration trace
    and asymmetric eligibility traces for positive and negative reward prediction errors.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - stage 1 inverse temperature
    beta2: [0, 10] - stage 2 inverse temperature
    lam_pos: [0, 1] - eligibility trace for positive RPE
    lam_neg: [0, 1] - eligibility trace for negative RPE
    w: [0, 1] - model-based weight
    p: [-5, 5] - perseveration bonus
    tau: [0, 1] - decay rate of the choice trace
    """
    alpha, beta1, beta2, lam_pos, lam_neg, w, p, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits_1 = beta1 * q_net + p * choice_trace
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Apply asymmetric eligibility trace based on RPE
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update exponentially decaying choice trace
        choice_trace = choice_trace * tau
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Independent MB and MF Inverse Temperatures
Instead of a single inverse temperature (`beta1`) scaling a weighted sum of model-based and model-free values (`w`), this model proposes that the model-based and model-free systems have independent scaling factors (`beta_mb` and `beta_mf`). This allows the sum of the system influences to not be strictly constrained to 1, representing varying total decision noise based on system agreement.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) hybrid model with an exponentially decaying perseveration trace,
    but separating the model-based and model-free weights into independent inverse temperatures 
    (beta_mb and beta_mf) for stage 1, rather than a single beta1 and weight w.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_mb: [0, 10] - model-based inverse temperature at stage 1
    beta_mf: [0, 10] - model-free inverse temperature at stage 1
    beta2: [0, 10] - stage 2 inverse temperature
    lam: [0, 1] - eligibility trace
    p: [-5, 5] - perseveration bonus
    tau: [0, 1] - decay rate of the choice trace
    """
    alpha, beta_mb, beta_mf, beta2, lam, p, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * choice_trace
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update exponentially decaying choice trace
        choice_trace = choice_trace * tau
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Model-Based Weight
This model proposes that the participant's reliance on the computationally expensive model-based system fluctuates depending on the outcome of the previous trial. It utilizes separate model-based weights (`w_pos` and `w_neg`) depending on whether the previous trial resulted in a reward or not. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) hybrid model with an exponentially decaying perseveration trace,
    where the model-based weight (w) depends on whether the previous trial was rewarded or not.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - stage 1 inverse temperature
    beta2: [0, 10] - stage 2 inverse temperature
    lam: [0, 1] - eligibility trace
    w_pos: [0, 1] - model-based weight following a rewarded trial
    w_neg: [0, 1] - model-based weight following an unrewarded trial
    p: [-5, 5] - perseveration bonus
    tau: [0, 1] - decay rate of the choice trace
    """
    alpha, beta1, beta2, lam, w_pos, w_neg, p, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)
    prev_reward = 0  # Initialize state for first trial

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # State-dependent model-based weight
        w = w_pos if prev_reward > 0 else w_neg
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits_1 = beta1 * q_net + p * choice_trace
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update exponentially decaying choice trace
        choice_trace = choice_trace * tau
        choice_trace[action_1[trial]] += 1.0
        
        prev_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
