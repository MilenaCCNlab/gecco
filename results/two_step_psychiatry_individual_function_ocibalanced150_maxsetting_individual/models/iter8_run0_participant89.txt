def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 1: Decay towards a central value with reward sensitivity.
    Includes model-based/model-free weighting, eligibility trace, perseverance,
    decay for unchosen options, a target value for decay, and reward sensitivity.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: weight of model-based vs model-free control [0, 1]
    lam: eligibility trace parameter [0, 1]
    p: choice perseverance [-5, 5]
    decay: forgetting rate for unchosen options [0, 1]
    center: target expected value for decay [0, 1]
    rho: reward sensitivity [0, 5]
    """
    alpha, beta, w, lam, p, decay, center, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        exp_q1 = np.exp(beta * q_stage1_net + p * rep)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] = (1 - decay) * q_stage1_mf[1 - action_1[trial]] + decay * center

        # Action value updating for Choice 2 with reward sensitivity
        eff_reward = rho * reward[trial]
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay for unchosen options in Stage 2
        q_stage2_mf[state_idx, 1 - action_2[trial]] = (1 - decay) * q_stage2_mf[state_idx, 1 - action_2[trial]] + decay * center
        q_stage2_mf[1 - state_idx, 0] = (1 - decay) * q_stage2_mf[1 - state_idx, 0] + decay * center
        q_stage2_mf[1 - state_idx, 1] = (1 - decay) * q_stage2_mf[1 - state_idx, 1] + decay * center

        # Eligibility trace update for Choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive Model 2: Decay towards a central value with win/loss specific perseverance.
    Includes model-based/model-free weighting, eligibility trace, win-perseverance,
    loss-perseverance, and a decay parameter for unchosen action values.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: weight of model-based vs model-free control [0, 1]
    lam: eligibility trace parameter [0, 1]
    p_win: choice perseverance after a win [-5, 5]
    p_lose: choice perseverance after a loss [-5, 5]
    decay: forgetting rate for unchosen options [0, 1]
    center: target expected value for decay [0, 1]
    """
    alpha, beta, w, lam, p_win, p_lose, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        p = 0.0
        if last_action != -1:
            rep[last_action] = 1
            if last_reward == 1:
                p = p_win
            else:
                p = p_lose
            
        exp_q1 = np.exp(beta * q_stage1_net + p * rep)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] = (1 - decay) * q_stage1_mf[1 - action_1[trial]] + decay * center

        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay for unchosen options in Stage 2
        q_stage2_mf[state_idx, 1 - action_2[trial]] = (1 - decay) * q_stage2_mf[state_idx, 1 - action_2[trial]] + decay * center
        q_stage2_mf[1 - state_idx, 0] = (1 - decay) * q_stage2_mf[1 - state_idx, 0] + decay * center
        q_stage2_mf[1 - state_idx, 1] = (1 - decay) * q_stage2_mf[1 - state_idx, 1] + decay * center

        # Eligibility trace update for Choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
