def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate learning rates for stage 1 and stage 2.
    
    Bounds:
    alpha1: [0,1] - learning rate for stage 1 Q-value updates
    alpha2: [0,1] - learning rate for stage 2 Q-value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers: [-5,5] - choice perseverance for stage 1
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        net_q_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            safe_beta1 = max(beta1, 1e-5)
            net_q_stage1[action_1[trial-1]] += pers / safe_beta1
            
        exp_q1 = np.exp(beta1 * net_q_stage1 - np.max(beta1 * net_q_stage1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] - np.max(beta2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with choice perseverance at both stage 1 and stage 2.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - choice perseverance for stage 1
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_stage2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        net_q_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            safe_beta1 = max(beta1, 1e-5)
            net_q_stage1[action_1[trial-1]] += pers1 / safe_beta1
            
        exp_q1 = np.exp(beta1 * net_q_stage1 - np.max(beta1 * net_q_stage1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        net_q_stage2 = q_stage2_mf[state_idx].copy()
        if last_action_stage2[state_idx] != -1:
            safe_beta2 = max(beta2, 1e-5)
            net_q_stage2[last_action_stage2[state_idx]] += pers2 / safe_beta2
            
        exp_q2 = np.exp(beta2 * net_q_stage2 - np.max(beta2 * net_q_stage2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model that dynamically updates the transition matrix based on experience.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    alpha_trans: [0,1] - learning rate for transition probabilities
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers: [-5,5] - choice perseverance for stage 1
    """
    alpha, alpha_trans, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        net_q_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            safe_beta1 = max(beta1, 1e-5)
            net_q_stage1[action_1[trial-1]] += pers / safe_beta1
            
        exp_q1 = np.exp(beta1 * net_q_stage1 - np.max(beta1 * net_q_stage1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] - np.max(beta2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update transition matrix
        a1 = action_1[trial]
        target_trans = np.zeros(2)
        target_trans[state_idx] = 1.0
        transition_matrix[a1] += alpha_trans * (target_trans - transition_matrix[a1])
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
