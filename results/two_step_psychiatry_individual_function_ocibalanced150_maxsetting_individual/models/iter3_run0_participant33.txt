def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with a reward sensitivity parameter.
    Scales the rewards by 'rho' before calculating prediction errors.
    This allows the model to adjust the relative magnitude of learned Q-values 
    compared to the fixed choice stickiness (pi).
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta: [0, 10] - Inverse temperature
    w: [0, 1] - Model-based weight
    lambda_: [0, 1] - Eligibility trace
    pi: [-5, 5] - Stage 1 choice stickiness
    rho: [0, 10] - Reward sensitivity multiplier
    """
    alpha, beta, w, lambda_, pi, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2 with scaled reward
        scaled_reward = reward[trial] * rho
        delta_stage2 = scaled_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with baseline stickiness and reward-modulated (WSLS) stickiness.
    The tendency to repeat the previous stage 1 action is modulated by whether 
    it was rewarded on the previous trial.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta: [0, 10] - Inverse temperature
    w: [0, 1] - Model-based weight
    lambda_: [0, 1] - Eligibility trace
    pi: [-5, 5] - Baseline stage 1 choice stickiness
    pi_wsls: [-5, 5] - Reward-modulated stickiness (added if last trial rewarded, subtracted if not)
    """
    alpha, beta, w, lambda_, pi, pi_wsls = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None
    last_reward = None

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi
            if last_reward == 1:
                q_stage1_net[last_action] += pi_wsls
            else:
                q_stage1_net[last_action] -= pi_wsls
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with fictitious updating for the unchosen Stage 1 action.
    Updates the unchosen action's model-free value assuming it would have 
    resulted in the opposite reward (1 - actual reward).
    
    Bounds:
    alpha: [0, 1] - Learning rate for chosen actions
    beta: [0, 10] - Inverse temperature
    w: [0, 1] - Model-based weight
    lambda_: [0, 1] - Eligibility trace
    pi: [-5, 5] - Stage 1 choice stickiness
    alpha_unchosen: [0, 1] - Learning rate for fictitious update of unchosen action
    """
    alpha, beta, w, lambda_, pi, alpha_unchosen = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action is not None:
            q_stage1_net[last_action] += pi
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Fictitious update for unchosen action
        unchosen_action = 1 - action_1[trial]
        fictitious_reward = 1 - reward[trial]
        delta_unchosen = fictitious_reward - q_stage1_mf[unchosen_action]
        q_stage1_mf[unchosen_action] += alpha_unchosen * delta_unchosen
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
