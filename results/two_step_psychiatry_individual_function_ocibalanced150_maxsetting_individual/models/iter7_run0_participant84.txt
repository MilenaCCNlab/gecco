import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 1: Daw et al. model with separate learning rates for Stage 1 and Stage 2,
    combined with memory decay (forgetting) for all actions.
    
    Parameters and Bounds:
    alpha_1: [0, 1] - Learning rate for Stage 1.
    alpha_2: [0, 1] - Learning rate for Stage 2.
    decay_rate: [0, 1] - Rate at which Q-values decay towards 0.5.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    lambda_: [0, 1] - Eligibility trace.
    """
    alpha_1, alpha_2, decay_rate, beta_1, beta_2, w, stickiness, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = stickiness
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        
        # policy for the second choice
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # decay unchosen and chosen options towards 0.5
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 2: Daw et al. model with asymmetric learning rates for positive and negative 
    prediction errors, combined with outcome-dependent stickiness (win-stay, lose-shift).
    
    Parameters and Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness_win: [-5, 5] - Bonus added to the previously chosen Stage 1 action if the last trial was rewarded.
    stickiness_lose: [-5, 5] - Bonus added to the previously chosen Stage 1 action if the last trial was unrewarded.
    lambda_: [0, 1] - Eligibility trace.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, stickiness_win, stickiness_lose, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_bonus[last_action_1] = stickiness_win
            else:
                stickiness_bonus[last_action_1] = stickiness_lose
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        
        # policy for the second choice
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 3: Daw et al. model with an exponentially decaying choice trace for Stage 1 actions.
    This captures long-term perseveration beyond just the previous trial.
    
    Parameters and Bounds:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    lambda_: [0, 1] - Eligibility trace.
    choice_trace_decay: [0, 1] - Decay rate of the choice trace.
    choice_trace_weight: [-5, 5] - Weight of the choice trace on Stage 1 preferences.
    """
    alpha, beta_1, beta_2, w, stickiness, lambda_, choice_trace_decay, choice_trace_weight = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1
    choice_trace = np.zeros(2)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = stickiness
            
        trace_bonus = choice_trace_weight * choice_trace
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus + trace_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        
        # policy for the second choice
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        
        # Update choice trace
        choice_trace = choice_trace * choice_trace_decay
        choice_trace[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
