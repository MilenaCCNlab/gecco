Here are three new cognitive models based on established mechanisms for the two-step decision-making task (such as the Daw et al., 2011 framework). They introduce model-based/model-free weighting, eligibility traces, and choice perseveration (stickiness). 

### Model 1: Model-Based and Model-Free Mixture
This model introduces the classic mixture parameter `w`, which weights the contribution of model-based versus model-free values during the first-stage choice. 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Standard two-step model with a mixture of model-based and model-free learning.
    
    Parameters:
    model_parameters (tuple):
        learning_rate: [0, 1] - Learning rate for Q-value updates.
        w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
        beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    """
    learning_rate, w, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Skip missed trials
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Mix Model-Based and Model-Free values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Shift values for numerical stability before exponentiation
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Mixture Model with Eligibility Trace
Building on Model 1, this model includes an eligibility trace parameter (`lambda_`). The eligibility trace allows the reward prediction error from the second stage to directly update the model-free value of the first-stage choice, speeding up learning.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with model-based/model-free weighting and an eligibility trace.
    
    Parameters:
    model_parameters (tuple):
        learning_rate: [0, 1] - Learning rate for Q-value updates.
        w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
        lambda_: [0, 1] - Eligibility trace parameter for stage 1 update from stage 2 reward.
        beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    """
    learning_rate, w, lambda_, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1 (model-free value updated by stage 2 error)
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Mixture Model with Choice Stickiness
This model captures choice perseveration by incorporating a `stickiness` parameter. It adds a direct value bonus to whichever action was chosen on the previous trial, independent of the reward outcome. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with model-based/model-free weighting and a choice stickiness parameter.
    
    Parameters:
    model_parameters (tuple):
        learning_rate: [0, 1] - Learning rate for Q-value updates.
        w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
        stickiness: [-5, 5] - Bonus added to the previously chosen stage 1 action.
        beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    """
    learning_rate, w, stickiness, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add stickiness to the previously chosen action
        if last_action != -1:
            q_net[last_action] += stickiness
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # Policy for the second choice
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
