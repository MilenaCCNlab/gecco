Here are 3 new cognitive models that explore alternative parameter configurations and mechanisms to explain the participant's behavior.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model tests the hypothesis that the participant updates their values at different speeds for the first-stage choices (spaceships) versus the second-stage choices (aliens). This can account for behavior where a participant is quick to switch aliens when unrewarded, but slower to switch spaceships (or vice versa).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate learning rates for Stage 1 and Stage 2.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for Stage 1 (spaceships).
    alpha2: [0, 1] - Learning rate for Stage 2 (aliens).
    beta1: [0, 10] - Inverse temperature for Stage 1 choices.
    beta2: [0, 10] - Inverse temperature for Stage 2 choices.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Choice perseveration bonus for Stage 1.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = 1
            
        logits1 = beta1 * q_net + pers * pers_bonus
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Perseveration for Both Stage 1 and Stage 2
Participants often exhibit choice stickiness ("perseveration") irrespective of reward. This model applies a perseveration bonus not only to the first-stage choice (spaceship) but also to the second-stage choice (alien), tracking the last chosen alien for each specific planet.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with perseveration bonuses for both Stage 1 and Stage 2 choices.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pers1: [-5, 5] - Choice perseveration bonus for Stage 1.
    pers2: [-5, 5] - Choice perseveration bonus for Stage 2.
    """
    alpha, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = 1
            
        logits1 = beta1 * q_net + pers1 * pers_bonus_1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = 1
            
        logits2 = beta2 * q_stage2_mf[state_idx] + pers2 * pers_bonus_2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Perseveration (Win-Stay / Lose-Shift)
Instead of a single perseveration parameter, this model separates perseveration based on the outcome of the previous trial. This explicitly models "Win-Stay" (positive stickiness after a reward) and "Lose-Shift" (negative stickiness/alternation after no reward) behaviors at the first stage.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with outcome-dependent perseveration (Win-Stay/Lose-Shift).
    Instead of a single perseveration parameter, it uses pers_win if the previous 
    trial was rewarded, and pers_lose if the previous trial was unrewarded.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Perseveration bonus if previous trial was rewarded.
    pers_lose: [-5, 5] - Perseveration bonus if previous trial was unrewarded.
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        
        if last_action != -1:
            if last_reward == 1:
                pers_bonus[last_action] = pers_win
            else:
                pers_bonus[last_action] = pers_lose
                
        logits1 = beta1 * q_net + pers_bonus
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
