Here are three new cognitive models exploring alternative mechanisms based on the best model so far.

### Model 1: Choice Kernel + Decay
This model expands on choice perseveration by using an exponentially decaying choice kernel for Stage 1. Rather than only perseverating on the immediately preceding trial, the choice kernel retains a fading memory of the participant's past choice history.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF RL with Choice Kernel and Unchosen Action Decay.
    Includes an exponentially decaying choice kernel for Stage 1 perseveration.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating chosen Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence
    lam: [0,1] - eligibility trace parameter
    pers: [-5,5] - choice kernel weight for stage 1
    decay: [0,1] - rate at which unchosen Q-values decay towards 0.5
    ck_lr: [0,1] - learning rate (update rate) for the choice kernel
    """
    learning_rate, beta, w, lam, pers, decay, ck_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    ck = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers * ck)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update Choice Kernel
        ck = ck * (1 - ck_lr)
        ck[action_1[trial]] += ck_lr

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen actions towards 0.5
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Win-Stay / Lose-Shift Perseveration + Decay
This model modifies perseveration to depend on the reward outcome of the previous trial. It adds separate perseveration weights for when the previous trial was rewarded (`pers_win`) and when it was unrewarded (`pers_lose`).

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF RL with Win-Stay/Lose-Shift Perseveration and Unchosen Action Decay.
    Separate perseveration parameters for when the previous trial was rewarded vs unrewarded.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating chosen Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence
    lam: [0,1] - eligibility trace parameter
    pers_win: [-5,5] - perseveration bonus if previous trial was rewarded
    pers_lose: [-5,5] - perseveration bonus if previous trial was unrewarded
    decay: [0,1] - rate at which unchosen Q-values decay towards 0.5
    """
    learning_rate, beta, w, lam, pers_win, pers_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if trial > 0:
            if reward[trial-1] > 0:
                pers_bonus[action_1[trial-1]] = pers_win
            else:
                pers_bonus[action_1[trial-1]] = pers_lose
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen actions towards 0.5
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage 1 and Stage 2 Perseveration + Decay
This model adds an additional perseverative parameter specifically for Stage 2 choices. If the participant visits a specific planet, they might be biased towards choosing the same alien they chose the last time they were on that planet.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF RL with Stage 1 & Stage 2 Perseveration and Unchosen Action Decay.
    Includes separate choice perseveration bonuses for Stage 1 and Stage 2.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating chosen Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence
    lam: [0,1] - eligibility trace parameter
    pers_st1: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    pers_st2: [-5,5] - choice perseveration bonus for repeating the previous stage 2 action (in the same state)
    decay: [0,1] - rate at which unchosen Q-values decay towards 0.5
    """
    learning_rate, beta, w, lam, pers_st1, pers_st2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action2 = {-1: -1, 0: -1, 1: -1}

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if trial > 0:
            pers_bonus_1[action_1[trial-1]] = pers_st1
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus_1)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        pers_bonus_2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            pers_bonus_2[last_action2[state_idx]] = pers_st2
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # update last action tracker for Stage 2 state
        last_action2[state_idx] = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen actions towards 0.5
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
