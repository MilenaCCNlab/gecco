Here are three new cognitive models based on the participant data, incorporating different cognitive mechanisms like asymmetric learning rates, fictitious updating, and reward-dependent model-based weighting.

### Model 1: Asymmetric Learning Rates with Stage 1 & Stage 2 Perseveration
This model tests whether the participant learns differently from positive versus negative prediction errors, and whether they exhibit motor perseveration (stickiness) independently in both Stage 1 and Stage 2 choices.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 1: Daw model with asymmetric learning rates for positive and negative prediction errors,
    and separate motor perseveration (stickiness) parameters for Stage 1 and Stage 2.
    
    Parameters and Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness_1: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    stickiness_2: [-5, 5] - Bonus added to the previously chosen Stage 2 action.
    lambda_: [0, 1] - Eligibility trace.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, stickiness_1, stickiness_2, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus_1[last_action_1] = stickiness_1
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus_1
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Stage 2 Choice
        state_idx = state[trial]
        
        stickiness_bonus_2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness_bonus_2[last_action_2] = stickiness_2
            
        q2_total = beta_2 * q_stage2_mf[state_idx] + stickiness_bonus_2
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr2 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Q-Value Decay with Fictitious Updating
This model introduces both a decay factor (pulling unchosen values towards a neutral 0.5) and fictitious updating. It assumes the participant knows that reward probabilities across aliens are often anti-correlated, updating the unchosen Stage 2 action with the counterfactual outcome.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 2: Daw model with learning rate, decay rate, and fictitious updating for the unchosen Stage 2 action.
    
    Parameters and Bounds:
    alpha: [0, 1] - Learning rate for chosen actions.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting parameter (1 = fully MB, 0 = fully MF).
    stickiness: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    lambda_: [0, 1] - Eligibility trace.
    decay_rate: [0, 1] - Decay rate for Q-values towards 0.5.
    fictitious_lr: [0, 1] - Learning rate for updating the unchosen Stage 2 action.
    """
    alpha, beta_1, beta_2, w, stickiness, lambda_, decay_rate, fictitious_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = stickiness
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Stage 2 Choice
        state_idx = state[trial]
        
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Fictitious update for unchosen action in Stage 2
        unchosen_action_2 = 1 - action_2[trial]
        delta_unchosen = (1 - reward[trial]) - q_stage2_mf[state_idx, unchosen_action_2]
        q_stage2_mf[state_idx, unchosen_action_2] += fictitious_lr * delta_unchosen
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Model-Based Control
This model proposes that the participant's balance between model-based (MB) and model-free (MF) control strategies actively shifts depending on the outcome of the previous trial (i.e. whether they won or lost), testing if goal-directed planning is reinforced by recent success.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Cognitive model 3: Daw model with reward-dependent model-based weighting (w_win and w_lose),
    along with Q-value decay.
    
    Parameters and Bounds:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w_win: [0, 1] - Model-based weight if the previous trial was rewarded.
    w_lose: [0, 1] - Model-based weight if the previous trial was NOT rewarded.
    stickiness: [-5, 5] - Bonus added to the previously chosen Stage 1 action.
    lambda_: [0, 1] - Eligibility trace.
    decay_rate: [0, 1] - Rate at which Q-values decay towards 0.5.
    """
    alpha, beta_1, beta_2, w_win, w_lose, stickiness, lambda_, decay_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # Determine weighting based on previous reward
        if last_reward == 1:
            current_w = w_win
        elif last_reward == 0:
            current_w = w_lose
        else:
            current_w = (w_win + w_lose) / 2.0  # Average for the first trial
            
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = stickiness
            
        q1_total = beta_1 * q_stage1_net + stickiness_bonus
        max_q1 = np.max(q1_total)
        exp_q1 = np.exp(q1_total - max_q1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # Stage 2 Choice
        state_idx = state[trial]
        
        q2_total = beta_2 * q_stage2_mf[state_idx]
        max_q2 = np.max(q2_total)
        exp_q2 = np.exp(q2_total - max_q2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
