def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition-Dependent Learning Rates with Eligibility Trace.
    
    This model uses different learning rates for common and rare transitions, 
    and includes an eligibility trace (lambda) that allows the second-stage 
    reward prediction error to directly update the first-stage action value.
    
    Bounds:
    alpha_c: [0, 1] - Learning rate after a common transition.
    alpha_r: [0, 1] - Learning rate after a rare transition.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi: [-5, 5] - Choice stickiness (perseveration) parameter.
    lam: [0, 1] - Eligibility trace parameter.
    """
    alpha_c, alpha_r, beta1, beta2, w, pi, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            q1_vals[last_action] += pi
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        is_common = (action_1[trial] == state_idx)
        alpha_t = alpha_c if is_common else alpha_r

        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_t * delta_stage1

        # Stage 2 MF update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_t * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha_t * lam * delta_stage2

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition-Dependent Learning Rates with Decay and Eligibility Trace.
    
    Combines transition-dependent learning rates (alpha_c, alpha_r) with both 
    an eligibility trace (lam) to carry second-stage prediction errors to the 
    first stage, and a decay parameter that pulls unchosen option values 
    towards 0.5.
    
    Bounds:
    alpha_c: [0, 1] - Learning rate after a common transition.
    alpha_r: [0, 1] - Learning rate after a rare transition.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi: [-5, 5] - Choice stickiness (perseveration) parameter.
    lam: [0, 1] - Eligibility trace parameter.
    decay: [0, 1] - Decay rate for unchosen options towards 0.5.
    """
    alpha_c, alpha_r, beta1, beta2, w, pi, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            q1_vals[last_action] += pi
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        is_common = (action_1[trial] == state_idx)
        alpha_t = alpha_c if is_common else alpha_r

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_t * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_t * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_t * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] += decay * (0.5 - q_stage2_mf[other_state, 0])
        q_stage2_mf[other_state, 1] += decay * (0.5 - q_stage2_mf[other_state, 1])

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-Dependent Learning Rates with Reward-Dependent Perseveration.
    
    This model uses different learning rates based on whether the transition 
    was common or rare. Additionally, it splits the stickiness (perseveration) 
    parameter into two based on the reward received in the previous trial: 
    pi_win for when the previous trial was rewarded, and pi_lose for when it wasn't.
    
    Bounds:
    alpha_c: [0, 1] - Learning rate after a common transition.
    alpha_r: [0, 1] - Learning rate after a rare transition.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi_win: [-5, 5] - Choice stickiness after receiving a reward.
    pi_lose: [-5, 5] - Choice stickiness after not receiving a reward.
    """
    alpha_c, alpha_r, beta1, beta2, w, pi_win, pi_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            if last_reward == 1:
                q1_vals[last_action] += pi_win
            else:
                q1_vals[last_action] += pi_lose
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        is_common = (action_1[trial] == state_idx)
        alpha_t = alpha_c if is_common else alpha_r

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_t * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_t * delta_stage2

        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
