Here are 3 new cognitive models for the two-step decision-making task, exploring alternative parameter configurations and mechanisms (such as asymmetric learning rates, win-stay/lose-shift stickiness, and proper eligibility trace implementation).

### Model 1: Asymmetric Learning Rates with Stage-Specific Temperatures and Stickiness
This model separates learning rates based on the sign of the reward prediction error (RPE). It also includes independent softmax temperatures and choice stickiness for both Stage 1 and Stage 2 decisions.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Asymmetric learning rates with stage-specific temperatures and stickiness.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive RPE
    alpha_neg: [0, 1] - Learning rate for negative RPE
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    w: [0, 1] - Model-based weight
    p1: [0, 10] - Stage 1 choice stickiness
    p2: [0, 10] - Stage 2 choice stickiness
    """
    alpha_pos, alpha_neg, beta1, beta2, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness1 = np.zeros(2)
        if last_action1 != -1:
            stickiness1[last_action1] = p1
            
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2 != -1:
            stickiness2[last_action2] = p2

        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + stickiness2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2
        q_stage1_mf[a1] += alpha_s2 * delta_stage2
        
        last_action1 = a1
        last_action2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Win-Stay, Lose-Shift Stickiness with Decay
This model captures reward-dependent habit formation by applying different stickiness parameters based on whether the previous trial resulted in a win (`p_win`) or a loss (`p_lose`). It also includes a value decay mechanism that progressively pulls Q-values back to their uninformative initial state (0.5).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Win-stay, Lose-shift stickiness with decay and separate temperatures.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    w: [0, 1] - Model-based weight
    p_win: [0, 10] - Stickiness after a rewarded trial
    p_lose: [0, 10] - Stickiness after an unrewarded trial
    decay: [0, 1] - Decay rate for Q-values towards initial 0.5
    """
    alpha, beta1, beta2, w, p_win, p_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize at 0.5 to serve as a proper decay target
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_action != -1:
            if last_reward == 1:
                stickiness[last_action] = p_win
            else:
                stickiness[last_action] = p_lose
            
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        # Decay step applied to all values
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Action value updating
        q_stage1_mf[a1] += alpha * delta_stage1
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * delta_stage2
        
        last_action = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Q(λ) Full Trace with Stage-Specific Learning Rates
This model introduces separate stage learning rates (`alpha1`, `alpha2`) and temperatures, while correctly applying an eligibility trace (`lam`) strictly to the way the Stage 2 RPE updates the Stage 1 Q-value.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Full Q(lambda) with separate stage learning rates, temperatures, and stage stickiness.
    
    Bounds:
    alpha1: [0, 1] - Stage 1 learning rate
    alpha2: [0, 1] - Stage 2 learning rate
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    lam: [0, 1] - Eligibility trace
    w: [0, 1] - Model-based weight
    p1: [0, 10] - Stage 1 choice stickiness
    p2: [0, 10] - Stage 2 choice stickiness
    """
    alpha1, alpha2, beta1, beta2, lam, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness1 = np.zeros(2)
        if last_action1 != -1:
            stickiness1[last_action1] = p1
            
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2 != -1:
            stickiness2[last_action2] = p2

        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + stickiness2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        # Apply eligibility trace on the stage 1 update from stage 2 feedback
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        last_action1 = a1
        last_action2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
