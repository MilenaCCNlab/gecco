def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF with asymmetric learning rates for positive and negative prediction errors.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: weight of model-based values vs model-free values [0,1]
    lam: eligibility trace [0,1]
    st: choice stickiness (perseveration) for the first stage [0,10]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_a1 != -1:
            rep[last_a1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + st * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[a1] += alpha_s1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage2_mf[state_idx, a2] += alpha_s2 * delta_stage2
        q_stage1_mf[a1] += alpha_s2 * lam * delta_stage2
        
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF with separate choice stickiness for stage 1 and stage 2.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: weight of model-based values vs model-free values [0,1]
    lam: eligibility trace [0,1]
    st_1: choice stickiness (perseveration) for the first stage [0,10]
    st_2: choice stickiness (perseveration) for the second stage [0,10]
    """
    alpha, beta_1, beta_2, w, lam, st_1, st_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_a1 != -1:
            rep_1[last_a1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + st_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        rep_2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            rep_2[last_a2[state_idx]] = 1.0

        logits_2 = beta_2 * q_stage2_mf[state_idx, :] + st_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF with decay (forgetting) for unchosen options.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: weight of model-based values vs model-free values [0,1]
    lam: eligibility trace [0,1]
    st: choice stickiness (perseveration) for the first stage [0,10]
    decay: decay rate for unchosen options towards 0 [0,1]
    """
    alpha, beta_1, beta_2, w, lam, st, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.ones(n_trials)
    p_choice_2 = np.ones(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_a1 != -1:
            rep[last_a1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + st * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1 - decay) * q_stage1_mf[unchosen_a1]
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a]
        
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
