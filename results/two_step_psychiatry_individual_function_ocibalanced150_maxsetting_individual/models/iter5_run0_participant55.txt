Here are 3 new cognitive models that build upon the best model by exploring well-established mechanisms in the literature, such as separate learning rates and temperatures for each stage, eligibility traces, stage-2 choice stickiness, and dynamic transition learning.

### Model 1: Full Daw et al. (2011) Model
This model implements the canonical 7-parameter model from the original Daw et al. (2011) paper. It expands on the best model by using separate learning rates (`alpha_1`, `alpha_2`) and separate inverse temperatures (`beta_1`, `beta_2`) for each stage, alongside an eligibility trace (`lambda_`) that allows the stage 2 reward prediction error to directly update the stage 1 model-free value.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. 2011 full model with separate learning rates and betas for each stage,
    an eligibility trace, model-based weighting, and choice stickiness.
    
    Parameters:
    model_parameters (tuple):
        alpha_1: [0, 1] - Learning rate for stage 1 Q-value updates.
        alpha_2: [0, 1] - Learning rate for stage 2 Q-value updates.
        w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
        lambda_: [0, 1] - Eligibility trace parameter for stage 1 update from stage 2 reward.
        stickiness: [-5, 5] - Bonus added to the previously chosen stage 1 action.
        beta_1: [0, 10] - Inverse temperature controlling choice stochasticity for stage 1.
        beta_2: [0, 10] - Inverse temperature controlling choice stochasticity for stage 2.
    """
    import numpy as np
    alpha_1, alpha_2, w, lambda_, stickiness, beta_1, beta_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_net[last_action] += stickiness
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta_1 * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # Stage 2 policy
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta_2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage 1 and Stage 2 Dual-Stickiness Model
This model extends the best model by applying the concept of "stickiness" (perseveration) to both the first and second stages, as subjects frequently reuse identical alien choices in stage 2 independently of their Q-values. It also untangles the choice stochasticity by utilizing separate `beta` values for each stage.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with model-based/model-free weighting, separate stage 1 and stage 2 betas,
    and separate choice stickiness parameters for stage 1 and stage 2.
    
    Parameters:
    model_parameters (tuple):
        learning_rate: [0, 1] - Learning rate for Q-value updates.
        w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
        stickiness_1: [-5, 5] - Bonus added to the previously chosen stage 1 action.
        stickiness_2: [-5, 5] - Bonus added to the previously chosen stage 2 action (state-specific).
        beta_1: [0, 10] - Inverse temperature for stage 1.
        beta_2: [0, 10] - Inverse temperature for stage 2.
    """
    import numpy as np
    learning_rate, w, stickiness_1, stickiness_2, beta_1, beta_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) # Track last action taken in each stage 2 state

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action_1 != -1:
            q_net_1[last_action_1] += stickiness_1
            
        q_net_1_shifted = q_net_1 - np.max(q_net_1)
        exp_q1 = np.exp(beta_1 * q_net_1_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action_1 = action_1[trial]

        # Stage 2 policy
        q_net_2 = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q_net_2[last_action_2[state_idx]] += stickiness_2

        q2_shifted = q_net_2 - np.max(q_net_2)
        exp_q2 = np.exp(beta_2 * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2[state_idx] = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition Learning Model
Rather than assuming that participants inherently know and maintain the fixed 0.7/0.3 transition probabilities, this model tests if participants actively learn the transition probabilities of the spaceships over time based on the actual planets they land on.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model that dynamically learns the transition matrix instead of assuming fixed 
    probabilities, paired with model-based/model-free weighting and choice stickiness.
    
    Parameters:
    model_parameters (tuple):
        learning_rate: [0, 1] - Learning rate for Q-value updates.
        w: [0, 1] - Weighting parameter (1 = fully model-based, 0 = fully model-free).
        stickiness: [-5, 5] - Bonus added to the previously chosen stage 1 action.
        beta: [0, 10] - Inverse temperature controlling choice stochasticity.
        alpha_tr: [0, 1] - Learning rate for updating the subjective transition matrix.
    """
    import numpy as np
    learning_rate, w, stickiness, beta, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize learned transition matrix to uniform (0.5)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 policy using dynamically learned transition matrix
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            q_net[last_action] += stickiness
            
        q_net_shifted = q_net - np.max(q_net)
        exp_q1 = np.exp(beta * q_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # Stage 2 policy
        q2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Transition matrix updating based on observed state
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        other_state = 1 - state_idx
        transition_matrix[action_1[trial], other_state] += alpha_tr * (0.0 - transition_matrix[action_1[trial], other_state])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
