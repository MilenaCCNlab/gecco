def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Full Daw Model (Daw et al., 2011)
    This model implements the standard hybrid reinforcement learning model for the two-step task.
    It combines model-free and model-based learning to evaluate the first-stage actions.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1 prediction errors.
    alpha2: [0, 1] - Learning rate for stage 2 prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1 (exploration/exploitation).
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight (1 = pure model-based, 0 = pure model-free).
    lam: [0, 1] - Eligibility trace parameter connecting stage 2 reward to stage 1 action.
    pers: [-5, 5] - Perseveration parameter (stickiness) to the previous stage 1 choice.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_bonus = np.zeros(2)
        if prev_a1 != -1:
            perseveration_bonus[prev_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pers * perseveration_bonus
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Action value updating for choice 1 (Stage 1 PE)
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        # Action value updating for choice 2 (Stage 2 PE)
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw Model with Asymmetric Learning Rates
    This model assumes that individuals learn differently from positive and negative 
    prediction errors, which is a common pattern in reward learning tasks.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (better than expected).
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (worse than expected).
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration parameter to the previous stage 1 choice.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    prev_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        perseveration_bonus = np.zeros(2)
        if prev_a1 != -1:
            perseveration_bonus[prev_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pers * perseveration_bonus
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Action value updating for choice 1 (Stage 1 PE)
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        # Action value updating for choice 2 (Stage 2 PE)
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha2 * lam * delta_stage2
        
        prev_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw Model with Decaying Choice Trace
    Instead of only considering the immediately preceding trial for perseveration, 
    this model builds an exponentially decaying choice trace to capture longer-term 
    autocorrelations in stage 1 choices.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1 prediction errors.
    alpha2: [0, 1] - Learning rate for stage 2 prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration weight for the choice trace.
    decay: [0, 1] - Decay rate of the choice trace (0 = only last trial, 1 = no decay).
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net + pers * choice_trace
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        a1 = action_1[trial]
        r = reward[trial]

        # Action value updating for choice 1 (Stage 1 PE)
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        # Action value updating for choice 2 (Stage 2 PE)
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[a1] += alpha1 * lam * delta_stage2
        
        # Update choice trace
        choice_trace *= decay
        choice_trace[a1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
