Here are three new cognitive models exploring alternative mechanisms for the two-step decision-making task. 

### Model 1: Transition Matrix Learning
While standard models assume participants use the fixed, instructed transition probabilities (70%/30%), this model allows participants to dynamically update their internal transition matrix based on the transitions they actually experience.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Transition Matrix Learning.
    Starts with the default transition probabilities but updates them based on experienced transitions,
    allowing the model-based system to adapt to local transition statistics.
    
    Bounds:
    alpha: [0, 1] (Learning rate for Q-values)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi: [-5, 5] (Choice perseveration)
    alpha_trans: [0, 1] (Learning rate for transition probabilities)
    """
    alpha, beta1, beta2, w, lam, pi, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1
            
        logits_1 = beta1 * q_net + pi * persev
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # update learned transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Eligibility Trace
This model assumes that the extent to which stage 2 prediction errors are propagated back to stage 1 depends on whether the outcome was positive (reward) or negative (omission). It fits separate lambda values for rewarded and unrewarded trials.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Reward-Dependent Eligibility Trace.
    Uses different eligibility trace parameters for rewarded (lam_win) vs unrewarded (lam_lose) trials,
    allowing asymmetric credit assignment to the first stage based on the valence of the outcome.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam_win: [0, 1] (Eligibility trace for rewarded trials)
    lam_lose: [0, 1] (Eligibility trace for unrewarded trials)
    pi: [-5, 5] (Choice perseveration)
    """
    alpha, beta1, beta2, w, lam_win, lam_lose, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1
            
        logits_1 = beta1 * q_net + pi * persev
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        current_lam = lam_win if reward[trial] == 1 else lam_lose
        q_stage1_mf[action_1[trial]] += alpha * current_lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Counterfactual (Anti-Correlated) Stage 2 Learning
Participants often assume that if one alien pays out, the other alien on the same planet likely does not (anti-correlated outcomes). This model introduces a counterfactual learning rate that updates the unchosen alien at stage 2 as if it yielded the exact opposite reward of the chosen alien.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Counterfactual Learning at Stage 2.
    In addition to updating the chosen action, it updates the unchosen action in the same state
    using a counterfactual learning rate, assuming its reward is the opposite of the received reward.
    
    Bounds:
    alpha: [0, 1] (Learning rate for chosen actions)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi: [-5, 5] (Choice perseveration)
    alpha_cf: [0, 1] (Counterfactual learning rate for unchosen action)
    """
    alpha, beta1, beta2, w, lam, pi, alpha_cf = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1
            
        logits_1 = beta1 * q_net + pi * persev
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for chosen actions
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # action value updating for unchosen action at stage 2 (Counterfactual update)
        unchosen_2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_2]
        q_stage2_mf[state_idx, unchosen_2] += alpha_cf * delta_cf
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
