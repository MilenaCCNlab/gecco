Here are three new cognitive models that explore alternative mechanisms not present in the previously tried combinations. They introduce reward sensitivity, decay towards a baseline expectation (rather than zero), and a long-term choice perseveration trace.

### Model 1: Reward Sensitivity Model
This model introduces a reward sensitivity parameter (`rho`). Individuals may subjectively value the collected coins differently. Scaling the reward by `rho` helps capture inter-individual differences in reward valuation.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) model with reward sensitivity.
    The reward is scaled by rho, which can account for individual differences in reward valuation.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi: [-5, 5] - stage 1 choice stickiness
    rho: [0, 5] - reward sensitivity
    """
    alpha, beta_1, beta_2, w, lambda_, pi, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial] * rho

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_a1 != -1:
            rep[last_a1] = 1.0
            
        q1_scaled = beta_1 * q_stage1_net + pi * rep
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2
        
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Decay Towards Baseline Model
Instead of unchosen options decaying to zero (which implies they become completely worthless), this model assumes unchosen values drift towards a baseline prior expectation `q0` (often near 0.5 in similar tasks). 

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) model with decay (forgetting) towards a baseline value.
    Unchosen Q-values decay towards q0 over time, representing forgetting of stale information towards an initial prior.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi: [-5, 5] - stage 1 choice stickiness
    decay: [0, 1] - decay rate for unchosen options
    q0: [0, 1] - baseline value to decay towards
    """
    alpha, beta_1, beta_2, w, lambda_, pi, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to the baseline prior q0
    q_stage1_mf = np.full(2, float(q0))
    q_stage2_mf = np.full((2, 2), float(q0))
    last_a1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_a1 != -1:
            rep[last_a1] = 1.0
            
        q1_scaled = beta_1 * q_stage1_net + pi * rep
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        # Unchosen options decay towards q0
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (q0 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (q0 - q_stage2_mf[s, a])
        
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Accumulating Choice Trace Model
Instead of stickiness depending solely on the immediately preceding trial, this model keeps an exponentially decaying trace of past choices. This formulation captures longer-term perseveration trends and habituation rather than just immediate lag-1 behavior.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) model with an accumulating choice trace for stage 1 perseverance.
    Instead of only depending on the immediately preceding trial, stickiness is based on an
    exponentially decaying trace of past choices, capturing longer-term choice perseveration.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambda_: [0, 1] - eligibility trace
    pi: [-5, 5] - stage 1 choice stickiness multiplier
    tau_c: [0, 1] - decay rate for the choice trace
    """
    alpha, beta_1, beta_2, w, lambda_, pi, tau_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Choice trace starts at 0
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]
        r = reward[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_scaled = beta_1 * q_stage1_net + pi * choice_trace
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        # Update choice trace for the next trial
        choice_trace *= tau_c
        choice_trace[a1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
