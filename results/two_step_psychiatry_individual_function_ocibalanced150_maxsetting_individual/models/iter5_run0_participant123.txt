def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for Stage 1 and Stage 2, with Stage 1 and Stage 2 stickiness.
    This model allows for different learning speeds at each stage of the task, while also 
    capturing choice perseveration (stickiness) independently at both stages.
    
    Bounds:
    alpha1: [0, 1] (learning rate for stage 1 value updates)
    alpha2: [0, 1] (learning rate for stage 2 value updates)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (weight of model-based vs model-free control)
    lam: [0, 1] (eligibility trace parameter)
    pi1: [-5, 5] (choice stickiness/perseveration for stage 1)
    pi2: [-5, 5] (choice stickiness/perseveration for stage 2)
    """
    alpha1, alpha2, beta1, beta2, w, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + pi1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        last_a1 = a1

        # policy for the second choice
        if a2 == -1:
            p_choice_2[trial] = 1.0
            continue

        stickiness2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            stickiness2[last_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx, :] + pi2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        last_a2[state_idx] = a2

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2

        q_stage1_mf[a1] += alpha1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate inverse temperatures for MB and MF systems, with decay and stickiness at both stages.
    This replaces the weighting parameter 'w' with distinct temperatures for MB and MF values, 
    allowing independent scaling of their contributions to the final choice, while retaining decay and perseveration.
    
    Bounds:
    learning_rate: [0, 1] (learning rate for updating values)
    decay: [0, 1] (decay rate for unchosen options)
    beta_mb: [0, 10] (inverse temperature for model-based system)
    beta_mf: [0, 10] (inverse temperature for model-free system)
    beta2: [0, 10] (inverse temperature for stage 2)
    lam: [0, 1] (eligibility trace parameter)
    pi1: [-5, 5] (choice stickiness/perseveration for stage 1)
    pi2: [-5, 5] (choice stickiness/perseveration for stage 2)
    """
    learning_rate, decay, beta_mb, beta_mf, beta2, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.full(2, -1)

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness1 = np.zeros(2)
        if last_a1 != -1:
            stickiness1[last_a1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * stickiness1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        last_a1 = a1

        # policy for the second choice
        if a2 == -1:
            p_choice_2[trial] = 1.0
            continue

        stickiness2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            stickiness2[last_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx, :] + pi2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        last_a2[state_idx] = a2

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        # Decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Choice trace model.
    Instead of 1-trial stickiness, this model maintains an exponentially decaying choice trace 
    for both stage 1 and stage 2 actions, capturing longer-term choice perseveration tendencies.
    
    Bounds:
    learning_rate: [0, 1] (learning rate for updating values)
    tau: [0, 1] (retention rate for choice traces)
    beta1: [0, 10] (inverse temperature for stage 1)
    beta2: [0, 10] (inverse temperature for stage 2)
    w: [0, 1] (weight of model-based vs model-free control)
    lam: [0, 1] (eligibility trace parameter)
    pi1: [-5, 5] (weight of choice trace for stage 1)
    pi2: [-5, 5] (weight of choice trace for stage 2)
    """
    learning_rate, tau, beta1, beta2, w, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace1 = np.zeros(2)
    choice_trace2 = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta1 * q_stage1_net + pi1 * choice_trace1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # update choice trace 1
        choice_trace1 *= tau
        choice_trace1[a1] += 1.0

        # policy for the second choice
        if a2 == -1:
            p_choice_2[trial] = 1.0
            continue

        logits_2 = beta2 * q_stage2_mf[state_idx, :] + pi2 * choice_trace2[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        # update choice trace 2
        choice_trace2[state_idx, :] *= tau
        choice_trace2[state_idx, a2] += 1.0

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
