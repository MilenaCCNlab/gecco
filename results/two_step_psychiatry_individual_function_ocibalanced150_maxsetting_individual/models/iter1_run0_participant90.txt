def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-learning with counterfactual updating for the unchosen action in the chosen state,
    and decay for the actions in the unchosen state. This models a participant who assumes that 
    the rewards of the two aliens on a given planet are somewhat anti-correlated.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen action.
    counterfactual_rate: [0,1] - Learning rate for the unchosen action in the chosen state.
    decay_rate: [0,1] - Decay rate for Q-values in the unvisited state.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    """
    learning_rate, counterfactual_rate, decay_rate, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay_rate)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Counterfactual update for unchosen action in chosen state
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        cf_delta = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += counterfactual_rate * cf_delta
        
        # Decay for unchosen state
        unchosen_state = 1 - state_idx
        q_stage2_mf[unchosen_state, 0] *= (1 - decay_rate)
        q_stage2_mf[unchosen_state, 1] *= (1 - decay_rate)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive and negative prediction errors,
    combined with decay for all unchosen options. Models risk-sensitive learning 
    where participants weigh positive and negative feedback differently.
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors.
    lr_neg: [0,1] - Learning rate for negative prediction errors.
    decay_rate: [0,1] - Decay rate for unchosen Q-values towards zero.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    """
    lr_pos, lr_neg, decay_rate, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay_rate)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Decay for unchosen options
        unchosen_a2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_a2] *= (1 - decay_rate)
        
        unchosen_state = 1 - state_idx
        q_stage2_mf[unchosen_state, 0] *= (1 - decay_rate)
        q_stage2_mf[unchosen_state, 1] *= (1 - decay_rate)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Q-learning with decay for unchosen options, plus a choice stickiness
    (perseveration) parameter for the Stage 2 choice. This captures the tendency 
    to repeat the same action independent of the reward history.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen action.
    decay_rate: [0,1] - Decay rate for unchosen Q-values.
    beta: [0,10] - Inverse temperature for softmax choice rule.
    stickiness: [-5,5] - Bonus added to the Q-value of the previously chosen action in the current state.
    """
    learning_rate, decay_rate, beta, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2_net = q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            q2_net[last_action_2[state_idx]] += stickiness
            
        exp_q2 = np.exp(beta * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay_rate)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Decay for unchosen options
        unchosen_a2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_a2] *= (1 - decay_rate)
        
        unchosen_state = 1 - state_idx
        q_stage2_mf[unchosen_state, 0] *= (1 - decay_rate)
        q_stage2_mf[unchosen_state, 1] *= (1 - decay_rate)
        
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
