Here are 3 new cognitive models that build upon the best-performing model (which featured decay for unchosen actions) by incorporating novel psychological mechanisms such as asymmetric learning rates, stage-specific learning, and choice perseveration (stickiness).

### Model 1: Stage-Specific Learning Rates + Decay
This model modifies the best model by allowing different learning rates for Stage 1 and Stage 2 updates. Participants often show varying degrees of value updating depending on how temporally proximal the state is to the actual reward. 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate learning rates for Stage 1 and Stage 2, and decay for unchosen actions.
    
    Parameters:
    lr1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    lr2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    decay: [0, 1] - Decay rate for unchosen action Q-values toward 0.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    """
    lr1, lr2, decay, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # Eligibility trace to Stage 1 uses the Stage 1 learning rate
        q_stage1_mf[action_1[trial]] += lr1 * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] -= decay * q_stage1_mf[unchosen_1]
        
        for state_i in range(2):
            for action_j in range(2):
                if not (state_i == state_idx and action_j == action_2[trial]):
                    q_stage2_mf[state_i, action_j] -= decay * q_stage2_mf[state_i, action_j]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Reward Learning + Decay
This model introduces asymmetric learning rates to the base decay model. Humans often learn at different rates from positive prediction errors (better-than-expected outcomes) versus negative prediction errors (worse-than-expected outcomes), which is a hallmark of dopamine-driven reinforcement learning.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with asymmetric learning rates for positive and negative RPEs, and decay for unchosen actions.
    
    Parameters:
    lr_pos: [0, 1] - Learning rate for positive reward prediction errors.
    lr_neg: [0, 1] - Learning rate for negative reward prediction errors.
    decay: [0, 1] - Decay rate for unchosen action Q-values toward 0.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    """
    lr_pos, lr_neg, decay, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # Stage 2 MF update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # Eligibility trace for Stage 1
        q_stage1_mf[action_1[trial]] += lr1 * lam * delta_stage2

        # Decay unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] -= decay * q_stage1_mf[unchosen_1]
        
        for state_i in range(2):
            for action_j in range(2):
                if not (state_i == state_idx and action_j == action_2[trial]):
                    q_stage2_mf[state_i, action_j] -= decay * q_stage2_mf[state_i, action_j]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Choice Stickiness + Decay
Participants frequently exhibit simple behavioral perseveration (or "stickiness") independent of reward outcomes. This model adds a stickiness parameter to the Stage 1 choice structure, temporarily boosting the probability of repeating the last action, while keeping the successful unchosen decay structure.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with decay for unchosen actions and choice stickiness at Stage 1.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen action Q-value updates.
    decay: [0, 1] - Decay rate for unchosen action Q-values toward 0.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for Model-Based (w) vs Model-Free (1-w) control at stage 1.
    lam: [0, 1] - Eligibility trace parameter.
    stick: [-5, 5] - Tendency to repeat the previous Stage 1 choice, regardless of reward.
    """
    learning_rate, decay, beta, w, lam, stick = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply stickiness parameter to the previous Stage 1 action
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stick
            
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] -= decay * q_stage1_mf[unchosen_1]
        
        for state_i in range(2):
            for action_j in range(2):
                if not (state_i == state_idx and action_j == action_2[trial]):
                    q_stage2_mf[state_i, action_j] -= decay * q_stage2_mf[state_i, action_j]
                    
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
