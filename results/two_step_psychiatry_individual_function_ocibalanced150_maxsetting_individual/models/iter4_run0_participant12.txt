Here are three new cognitive models exploring different combinations of mechanisms (such as separate stage temperatures, asymmetric learning rates for positive and negative prediction errors, and unchosen action decay). They are built upon the successful hybrid MB/MF architecture.

### Model 1: Separate Temperatures and Perseveration
This model tests the hypothesis that the participant may have different choice stochasticity (exploration) and perseveration tendencies at Stage 1 compared to Stage 2.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF RL with separate temperatures and perseverance for both stages.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta1: [0,10] - inverse temperature for stage 1 softmax choice
    beta2: [0,10] - inverse temperature for stage 2 softmax choice
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers1: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    pers2: [-5,5] - choice perseveration bonus for repeating the previous stage 2 action
    """
    learning_rate, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if trial > 0:
            pers1_bonus[action_1[trial-1]] = pers1
            
        exp_q1_net = np.exp(beta1 * q_stage1_net + pers1_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # Stage 2 policy
        state_idx = state[trial]
        pers2_bonus = np.zeros(2)
        if trial > 0:
            pers2_bonus[action_2[trial-1]] = pers2
            
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pers2_bonus)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates + Decay
This model modifies the best model by adding asymmetric learning rates. The participant might learn differently from positive (better-than-expected) vs negative (worse-than-expected) prediction errors.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF RL with asymmetric learning rates and unchosen action decay.
    
    Parameters:
    lr_pos: [0,1] - learning rate for positive prediction errors
    lr_neg: [0,1] - learning rate for negative prediction errors
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    decay: [0,1] - rate at which unchosen Q-values decay towards neutral 0.5
    """
    lr_pos, lr_neg, beta, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] = pers
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # Stage 2 policy
        state_idx = state[trial]
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

        # Decay unchosen actions
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Decoupled Stage Parameters + Decay
This model completely isolates the learning and choice stochasticity properties between Stage 1 and Stage 2, while retaining the successful decay mechanism. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF RL with separate learning rates and betas for each stage, plus decay.
    
    Parameters:
    lr1: [0,1] - learning rate for stage 1 updates
    lr2: [0,1] - learning rate for stage 2 updates
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    decay: [0,1] - rate at which unchosen Q-values decay towards 0.5
    """
    lr1, lr2, beta1, beta2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] = pers
            
        exp_q1_net = np.exp(beta1 * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]

        # Stage 2 policy
        state_idx = state[trial]
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # Eligibility trace update using Stage 1 learning rate
        q_stage1_mf[action_1[trial]] += lr1 * lam * delta_stage2

        # Decay unchosen actions
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
