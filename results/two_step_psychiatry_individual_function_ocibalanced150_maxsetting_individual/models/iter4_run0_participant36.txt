def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition matrix learning.
    Instead of assuming a fixed 0.7/0.3 transition matrix, the participant updates 
    their internal model of the transition probabilities between stage 1 actions 
    and stage 2 states based on experience.
    
    Bounds:
    alpha: [0, 1] - learning rate for Q-values
    alpha_T: [0, 1] - learning rate for the transition matrix
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    st: [0, 10] - choice stickiness
    """
    alpha, alpha_T, beta1, beta2, w, lam, st = model_parameters
    n_trials = len(action_1)
  
    # Initial transition matrix
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1
            
        logits_1 = beta1 * q_net + st * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # transition matrix updating
        transition_matrix[a1, state_idx] += alpha_T * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_T * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric eligibility traces for positive and negative reward prediction errors.
    This captures the idea that the credit assignment (updating stage 1 based on stage 2 reward)
    might differ depending on whether the outcome was better or worse than expected.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam_pos: [0, 1] - eligibility trace for positive RPEs
    lam_neg: [0, 1] - eligibility trace for negative RPEs
    st: [0, 10] - choice stickiness
    """
    alpha, beta1, beta2, w, lam_pos, lam_neg, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1
            
        logits_1 = beta1 * q_net + st * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        # asymmetric eligibility trace
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with forgetting (decay) of unchosen options towards a specific center value.
    Instead of decaying to 0, which might not be the baseline expectation, values decay 
    towards a parameterized mean (e.g., 0.5 for random reward outcomes).
    
    Bounds:
    alpha: [0, 1] - learning rate
    decay: [0, 1] - decay rate for unchosen options
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    st: [0, 10] - choice stickiness
    center: [0, 1] - the value towards which unchosen options decay
    """
    alpha, decay, beta1, beta2, w, lam, st, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to the center value
    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1
            
        logits_1 = beta1 * q_net + st * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

        # Decay unchosen options towards the center
        unchosen_1 = 1 - a1
        q_stage1_mf[unchosen_1] = q_stage1_mf[unchosen_1] * (1 - decay) + center * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + center * decay

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
