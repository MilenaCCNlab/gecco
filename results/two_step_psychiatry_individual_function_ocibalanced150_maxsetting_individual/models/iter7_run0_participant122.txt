Here are three new cognitive models that explore different mechanisms for learning and decision-making, while avoiding previously tried parameter combinations. 

### Model 1: Choice Traces for Stage 1 and Stage 2
Instead of a simple perseveration bonus for only the immediately preceding trial, this model maintains an exponentially decaying "choice trace" for both stage 1 and stage 2 actions. This allows the model to capture longer-term habits or autocorrelations in choices that persist over multiple trials.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice Traces for Stage 1 and Stage 2
    Maintains an exponentially decaying choice trace for both stage 1 and stage 2 actions,
    allowing for perseveration that lasts longer than a single trial.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    lam: [0,1] - eligibility trace parameter linking stage 2 reward to stage 1
    trace_decay: [0,1] - decay factor for the choice traces (0 = no memory, 1 = perfect memory)
    trace_weight_1: [-5,5] - weight of the stage 1 choice trace on stage 1 decisions
    trace_weight_2: [-5,5] - weight of the stage 2 choice trace on stage 2 decisions
    """
    learning_rate, beta, w, lam, trace_decay, trace_weight_1, trace_weight_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))
    
    for trial in range(n_trials):
        # Stage 1 choice policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf + trace_weight_1 * trace_1
        
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 choice policy
        if action_2[trial] != -1:
            q_stage2_net = q_stage2_mf[state_idx] + trace_weight_2 * trace_2[state_idx]
            max_q2 = np.max(q_stage2_net)
            exp_q2 = np.exp(beta * (q_stage2_net - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
            
        # Action value updating
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
            # Update choice traces
            trace_1 *= trace_decay
            trace_1[action_1[trial]] += 1
            
            trace_2 *= trace_decay
            trace_2[state_idx, action_2[trial]] += 1
            
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Side Bias and Alien Bias
Participants often exhibit idiosyncratic, reward-independent preferences for a specific side (e.g., constantly favoring spaceship 0) or specific stimuli (e.g., favoring alien 0). This model isolates and captures these static biases, preventing them from being misattributed to learning processes.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Side Bias and Alien Bias
    Adds a static preference (bias) for choosing spaceship 0 over spaceship 1, 
    and alien 0 over alien 1, capturing idiosyncratic preferences independent of reward.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter linking stage 2 reward to stage 1
    bias_space0: [-5,5] - static bias added to the value of spaceship 0
    bias_alien0: [-5,5] - static bias added to the value of alien 0
    """
    learning_rate, beta, w, p, lam, bias_space0, bias_alien0 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1
    
    for trial in range(n_trials):
        # Stage 1 choice policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += p
        
        # Add bias for spaceship 0
        q_stage1_net[0] += bias_space0
        
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 choice policy
        if action_2[trial] != -1:
            q_stage2_net = np.copy(q_stage2_mf[state_idx])
            # Add bias for alien 0
            q_stage2_net[0] += bias_alien0
            
            max_q2 = np.max(q_stage2_net)
            exp_q2 = np.exp(beta * (q_stage2_net - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
            
        # Action value updating
        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
            q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
            
        last_action = action_1[trial]
            
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Surprise-modulated Learning Rate (Pearce-Hall approximation)
This model assumes that learning is driven by attention. When a participant experiences a large absolute reward prediction error (a highly surprising outcome), their learning rate temporarily increases for that update.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Surprise-modulated Learning Rate
    The learning rate is dynamic and increases when the absolute reward prediction 
    error (surprise) is larger, capturing attention-driven learning where unexpected 
    outcomes boost subsequent updating.
    
    Parameters:
    lr_base: [0,1] - baseline learning rate
    lr_surprise: [0,1] - scaling factor for how much absolute RPE increases learning rate
    beta: [0,10] - inverse temperature for choice stochasticity
    w: [0,1] - weight on model-based vs model-free strategy
    p: [-5,5] - perseveration bonus for repeating the previous stage 1 action
    lam: [0,1] - eligibility trace parameter linking stage 2 reward to stage 1
    """
    lr_base, lr_surprise, beta, w, p, lam = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1
    
    for trial in range(n_trials):
        # Stage 1 choice policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += p
            
        max_q1 = np.max(q_stage1_net)
        exp_q1 = np.exp(beta * (q_stage1_net - max_q1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 choice policy
        if action_2[trial] != -1:
            q_stage2_net = q_stage2_mf[state_idx]
            max_q2 = np.max(q_stage2_net)
            exp_q2 = np.exp(beta * (q_stage2_net - max_q2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
            
        # Action value updating
        if action_2[trial] != -1:
            # Calculate final RPE to determine dynamic learning rate
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            
            # Dynamic learning rate based on surprise (bounded to 1.0)
            current_lr = min(1.0, lr_base + lr_surprise * abs(delta_stage2))
            
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            
            # Apply updates with the dynamic learning rate
            q_stage1_mf[action_1[trial]] += current_lr * delta_stage1
            q_stage2_mf[state_idx, action_2[trial]] += current_lr * delta_stage2
            q_stage1_mf[action_1[trial]] += current_lr * lam * delta_stage2
            
        last_action = action_1[trial]
            
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
