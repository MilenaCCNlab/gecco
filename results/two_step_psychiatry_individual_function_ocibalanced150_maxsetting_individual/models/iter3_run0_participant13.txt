import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-Step Model with Stage 1 and Stage 2 Choice Stickiness.
    This model extends the standard Daw et al. (2011) model by including a separate
    stickiness (perseveration) parameter for Stage 2 choices. This captures motor 
    or choice habits at both the first and second stages of the decision process.
    
    Bounds:
    alpha: [0,1] (Learning rate)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    lam: [0,1] (Eligibility trace)
    pi_1: [-5,5] (Stage 1 choice stickiness)
    pi_2: [-5,5] (Stage 2 choice stickiness)
    """
    alpha, beta_1, beta_2, w, lam, pi_1, pi_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + pi_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 Policy
        rep_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            rep_2[last_action_2[state_idx]] = 1

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pi_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update last actions
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-Step Model with Reward Sensitivity.
    This model separates the inverse temperature (which controls stochasticity) 
    from reward sensitivity by scaling the objective reward by a parameter rho. 
    This allows for individual differences in how subjectively valuable the coins are.
    
    Bounds:
    alpha: [0,1] (Learning rate)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    lam: [0,1] (Eligibility trace)
    pi: [-5,5] (Choice stickiness)
    rho: [0,10] (Reward sensitivity/scaling)
    """
    alpha, beta_1, beta_2, w, lam, pi, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Scale reward
        eff_reward = reward[trial] * rho

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update last action
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-Step Model with Unchosen Option Drift.
    Instead of standard decay towards 0, the values of unchosen options drift towards 
    a baseline expectation of 0.5 with a separate learning rate. This reflects the 
    participant's prior that reward probabilities tend to revert to the mean.
    
    Bounds:
    alpha_c: [0,1] (Learning rate for chosen options)
    alpha_u: [0,1] (Learning rate for unchosen options/drift rate)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    lam: [0,1] (Eligibility trace)
    pi: [-5,5] (Choice stickiness)
    """
    alpha_c, alpha_u, beta_1, beta_2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values to 0.5 (mean expectation)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates for chosen options
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_c * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_c * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_c * lam * delta_stage2
        
        # Updates for unchosen options (drifting towards 0.5)
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += alpha_u * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += alpha_u * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += alpha_u * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += alpha_u * (0.5 - q_stage2_mf[unvisited_state, 1])

        # Update last action
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
