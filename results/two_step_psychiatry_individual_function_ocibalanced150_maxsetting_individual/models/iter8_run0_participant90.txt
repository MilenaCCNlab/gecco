Here are 3 new cognitive models that build upon the best-performing mechanisms so far, introducing new parameter combinations and nuanced cognitive processes like eligibility traces, stage-specific stickiness, and counterfactual updating.

### Model 1: Asymmetric Learning with Eligibility Traces and Decay
This model takes the best-performing model (which includes asymmetric learning rates, decay, and stickiness) and adds an **eligibility trace (`lambda_trace`)**. The eligibility trace allows the reward prediction error from the second stage to directly update the model-free value of the first-stage choice, which is a core mechanism in the original Daw et al. (2011) two-step task model.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free learning with asymmetric learning rates,
    decay rate for unchosen actions, separate inverse temperatures (betas) for each stage,
    choice stickiness for stage 1, and an eligibility trace (lambda_trace).
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors.
    lr_neg: [0,1] - Learning rate for negative prediction errors.
    decay_rate: [0,1] - Rate at which unchosen action values decay towards 0.5.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness: [-5,5] - Perseveration parameter for stage 1.
    lambda_trace: [0,1] - Eligibility trace for stage 1 updating from stage 2 reward.
    """
    lr_pos, lr_neg, decay_rate, beta1, beta2, w, stickiness, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 MB values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness

        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # Stage 2 MF update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # Eligibility trace update for Stage 1 MF
        lr_trace = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lambda_trace * lr_trace * delta_stage2

        # Decay unchosen actions
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Stickiness with Eligibility Traces
This model hypothesizes that perseveration (stickiness) might occur differently at Stage 1 versus Stage 2. It introduces `stickiness2` to capture tendencies to repeat choices on a specific planet, complementing the standard learning rate, decay, and eligibility trace.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free learning with a single learning rate, decay rate for unchosen actions,
    separate inverse temperatures (betas) and stickiness parameters for each stage, and an eligibility trace.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for prediction errors.
    decay_rate: [0,1] - Rate at which unchosen action values decay towards 0.5.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness1: [-5,5] - Perseveration parameter for stage 1.
    stickiness2: [-5,5] - Perseveration parameter for stage 2.
    lambda_trace: [0,1] - Eligibility trace for stage 1 updating from stage 2 reward.
    """
    learning_rate, decay_rate, beta1, beta2, w, stickiness1, stickiness2, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 MB values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness1

        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if trial > 0 and state[trial-1] == state_idx:
            q_stage2_net[action_2[trial-1]] += stickiness2

        exp_q2 = np.exp(beta2 * (q_stage2_net - np.max(q_stage2_net)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Stage 2 MF update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update for Stage 1 MF
        q_stage1_mf[action_1[trial]] += lambda_trace * learning_rate * delta_stage2

        # Decay unchosen actions
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Counterfactual Updating with Decay and Traces
This model incorporates **counterfactual updating** for the unchosen alien on the *same* planet. Participants may assume that the probabilities of the two aliens on the same planet are anti-correlated (i.e., if one doesn't give gold, the other likely will). This is combined with decay for the aliens on the completely unvisited planet.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free learning with counterfactual updating for the unchosen alien 
    on the same planet, decay for other unchosen states/actions, separate betas, stickiness, 
    and an eligibility trace.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for prediction errors.
    counterfactual_rate: [0,1] - Learning rate for counterfactual update of unchosen alien.
    decay_rate: [0,1] - Rate at which unchosen action values decay towards 0.5.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness: [-5,5] - Perseveration parameter for stage 1.
    lambda_trace: [0,1] - Eligibility trace for stage 1 updating from stage 2 reward.
    """
    learning_rate, counterfactual_rate, decay_rate, beta1, beta2, w, stickiness, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 MB values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness

        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Stage 2 MF update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Counterfactual update for the other alien on the same planet
        unchosen_a2 = 1 - action_2[trial]
        # Assuming anti-correlated rewards: counterfactual reward is 1 - reward
        cf_delta_stage2 = (1 - reward[trial]) - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += counterfactual_rate * cf_delta_stage2

        # Eligibility trace update for Stage 1 MF
        q_stage1_mf[action_1[trial]] += lambda_trace * learning_rate * delta_stage2

        # Decay unchosen actions (Stage 1)
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        # Decay unchosen actions (Stage 2 - only for the unvisited planet)
        other_state = 1 - state_idx
        for a in range(2):
            q_stage2_mf[other_state, a] += decay_rate * (0.5 - q_stage2_mf[other_state, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
