Here are three new cognitive models that explore alternative parameter configurations and mechanisms to explain the participant's choices. 

Given the participant's data, we can observe a very strong preference for spaceship `0`. To capture this, Models 1 and 3 introduce a `bias1` parameter, which models a baseline choice bias (e.g., a side bias or a preference for the visual representation of spaceship 0). Model 2 explores how previous rewards might modulate choice perseveration (win-stay/lose-shift heuristics) coupled with value decay. Model 3 explores asymmetric learning rates for positive and negative prediction errors alongside the choice bias.

### Model 1: Standard RL with Stage 1 Choice Bias
```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Standard RL with Stage 1 Choice Bias.
    Includes a side/choice bias parameter for Stage 1, as participants often exhibit 
    a baseline preference for one spaceship over the other, along with standard 
    model-based/model-free weighting and stickiness at both stages.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Stage 1 stickiness (perseveration)
    p2: [-5,5] - Stage 2 stickiness (perseveration)
    bias1: [-5,5] - Baseline bias for choosing spaceship 0
    """
    alpha, beta1, beta2, lam, w, p1, p2, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        bias_vec = np.array([bias1, 0.0])
            
        logits_1 = beta1 * q_net + p1 * rep1 + bias_vec
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if prev_a2 != -1:
            rep2[prev_a2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-dependent Stage 1 Stickiness with Decay
```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-dependent Stage 1 Stickiness with Decay.
    Stickiness at Stage 1 depends on whether the previous trial was rewarded or unrewarded, 
    capturing win-stay/lose-shift tendencies, combined with value decay for unchosen options.
    
    Bounds:
    alpha: [0,1] - Learning rate
    decay: [0,1] - Decay rate for unchosen options
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p_rew: [-5,5] - Stage 1 stickiness after a reward
    p_unrew: [-5,5] - Stage 1 stickiness after no reward
    """
    alpha, decay, beta1, beta2, lam, w, p_rew, p_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        p1 = p_rew if prev_reward == 1 else p_unrew
            
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay unchosen options
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        prev_a1 = action_1[trial]
        prev_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning with Choice Bias
```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Learning with Choice Bias.
    Separates learning rates for positive and negative prediction errors (to capture 
    differential sensitivity to better- vs worse-than-expected outcomes), combined with 
    Stage 1 stickiness and a baseline bias for spaceship 0.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Stage 1 stickiness (perseveration)
    bias1: [-5,5] - Baseline bias for choosing spaceship 0
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p1, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        bias_vec = np.array([bias1, 0.0])
            
        logits_1 = beta1 * q_net + p1 * rep1 + bias_vec
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        prev_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
