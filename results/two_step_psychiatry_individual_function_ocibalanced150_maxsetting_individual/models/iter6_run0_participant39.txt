def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Reward-Dependent Choice Perseveration.
    This model splits the stage 1 perseveration into two parameters based on whether 
    the previous trial was rewarded or unrewarded, capturing win-stay/lose-shift tendencies.
    
    Parameters:
    learning_rate: [0,1] learning rate for updating Q-values
    beta1: [0,10] inverse temperature for stage 1 softmax choice
    beta2: [0,10] inverse temperature for stage 2 softmax choice
    w: [0,1] weight for model-based vs model-free control
    pers_rew: [-5,5] perseveration bonus for repeating the previous stage 1 choice if rewarded
    pers_unrew: [-5,5] perseveration bonus for repeating the previous stage 1 choice if unrewarded
    """
    learning_rate, beta1, beta2, w, pers_rew, pers_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_term = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                pers_term[last_action_1] = pers_rew
            else:
                pers_term[last_action_1] = pers_unrew
                
        q1_vals = beta1 * q_stage1_net + pers_term
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Learning Rates and Eligibility Trace.
    This model uses different learning rates for positive versus negative prediction errors,
    and includes an eligibility trace (lambda) for updating stage 1 Q-values from stage 2 rewards.
    
    Parameters:
    alpha_pos: [0,1] learning rate for positive prediction errors
    alpha_neg: [0,1] learning rate for negative prediction errors
    beta1: [0,10] inverse temperature for stage 1 softmax choice
    beta2: [0,10] inverse temperature for stage 2 softmax choice
    w: [0,1] weight for model-based vs model-free control
    pers: [-5,5] perseveration bonus for repeating the previous stage 1 choice
    lam: [0,1] eligibility trace parameter for stage 2 reward updating stage 1
    """
    alpha_pos, alpha_neg, beta1, beta2, w, pers, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_term = np.zeros(2)
        if last_action_1 != -1:
            pers_term[last_action_1] = pers
                
        q1_vals = beta1 * q_stage1_net + pers_term
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Comprehensive Daw (2011) Model with Stage-Specific Parameters.
    Includes separate learning rates, inverse temperatures, and perseveration parameters
    for stage 1 and stage 2, along with an eligibility trace and model-based weight.
    
    Parameters:
    alpha1: [0,1] learning rate for stage 1 Q-values
    alpha2: [0,1] learning rate for stage 2 Q-values
    beta1: [0,10] inverse temperature for stage 1
    beta2: [0,10] inverse temperature for stage 2
    w: [0,1] weight for model-based vs model-free control
    pers1: [-5,5] perseveration bonus for stage 1 choice
    pers2: [-5,5] perseveration bonus for stage 2 choice (state-specific)
    lam: [0,1] eligibility trace parameter
    """
    alpha1, alpha2, beta1, beta2, w, pers1, pers2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_term1 = np.zeros(2)
        if last_action_1 != -1:
            pers_term1[last_action_1] = pers1
                
        q1_vals = beta1 * q_stage1_net + pers_term1
        exp_q1 = np.exp(q1_vals - np.max(q1_vals))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        pers_term2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_term2[last_action_2[state_idx]] = pers2
            
        q2_vals = beta2 * q_stage2_mf[state_idx] + pers_term2
        exp_q2 = np.exp(q2_vals - np.max(q2_vals))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
