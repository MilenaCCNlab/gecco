Here are 3 new cognitive models that incorporate the successful features of the best model so far (such as separate stage beta parameters, perseveration, and unchosen option decay) while introducing new distinct mechanisms.

### Model 1: Asymmetric Learning Rates + Decay
This model separates the learning rate for positive and negative prediction errors. Participants often update their expectations differently when receiving a reward versus a non-reward. It combines this asymmetry with unchosen option decay.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with separate learning rates for positive and negative prediction errors.
    Includes perseveration, eligibility trace, and unchosen option decay.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers: perseveration bonus for both stages [-5, 5]
    decay: forgetting rate for unchosen options towards 0.5 [0, 1]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and a2 != -1 and state_idx != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            lr1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
            lr2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

            q_stage1_mf[a1] += lr1 * delta_stage1 + lr2 * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += lr2 * delta_stage2

            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])

            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])

            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition Learning + Decay
Instead of assuming the transition probabilities remain fixed at exactly 0.7/0.3 internally, this model dynamically updates the internal transition matrix based on experienced transitions, making it sensitive to local stretches of rare transitions.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL where the transition matrix is learned over time.
    Includes perseveration, eligibility trace, and unchosen option decay.
    
    Parameters:
    alpha: learning rate for Q-values [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers: perseveration bonus for both stages [-5, 5]
    decay: forgetting rate for unchosen options towards 0.5 [0, 1]
    alpha_trans: learning rate for the transition matrix [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, pers, decay, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and a2 != -1 and state_idx != -1:
            # Update transition matrix
            transition_matrix[a1, state_idx] += alpha_trans * (1 - transition_matrix[a1, state_idx])
            transition_matrix[a1, 1 - state_idx] += alpha_trans * (0 - transition_matrix[a1, 1 - state_idx])

            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2

            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])

            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])

            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Win-Stay Lose-Shift (WSLS) Heuristic + Decay
Participants often exhibit a heuristic approach in parallel with standard associative learning architectures. This model adds a direct "Win-Stay, Lose-Shift" parameter reflecting a pure heuristic bonus on top of their perseveration behavior.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with Win-Stay Lose-Shift (WSLS) heuristic.
    Includes perseveration, eligibility trace, and unchosen option decay.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers: perseveration bonus for both stages [-5, 5]
    wsls: Win-Stay Lose-Shift bonus for stage 1 [-5, 5]
    decay: forgetting rate for unchosen options towards 0.5 [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, pers, wsls, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        wsls_bonus_1 = np.zeros(2)
        
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers
            if last_reward == 1:
                wsls_bonus_1[last_action_1] = wsls
            else:
                wsls_bonus_1[1 - last_action_1] = wsls
            
        q_1_total = q_net_1 + pers_bonus_1 + wsls_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a1 != -1 and a2 != -1 and state_idx != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2

            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_a2])

            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])

            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
