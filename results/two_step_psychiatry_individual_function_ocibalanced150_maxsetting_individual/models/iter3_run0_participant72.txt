def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates with Decay and Stickiness.
    
    This model separates the learning rate into positive and negative prediction 
    errors (alpha_pos and alpha_neg). It also includes a decay mechanism where 
    unchosen options drift towards 0.5, and a stickiness parameter for stage 1.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta_1: [0,10] - Inverse temperature for first-stage choice.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weight on model-based vs. model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    p: [0,10] - Choice stickiness (perseveration) for the first-stage choice.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + p * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw Two-Step Model with Subjective Transition Probability.
    
    This model introduces a subjective transition probability parameter (pi_trans). 
    Instead of assuming the participant fully internalized the true 0.7/0.3 
    transition probabilities, it estimates their internal model's transition mapping.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen actions.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.

    beta_2: [0,10] - Inverse temperature for second-stage choice.
    w: [0,1] - Weight on model-based vs. model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
