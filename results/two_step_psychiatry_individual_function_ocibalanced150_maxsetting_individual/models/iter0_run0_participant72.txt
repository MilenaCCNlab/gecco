Here are three cognitive models that build upon the template to explain participant behavior in the two-step decision-making task. They represent standard and variant models in the reinforcement learning literature (such as the Daw et al., 2011 model). 

### Model 1: Full Daw Two-Step Model
This is the standard model-free and model-based reinforcement learning model. It integrates both model-based (transition-aware) and model-free (direct caching) learning to make decisions in the first stage, and includes an eligibility trace to update first-stage values from second-stage rewards. It also incorporates a stickiness parameter to capture choice perseveration.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Full Daw Two-Step Model (Model-Free / Model-Based RL).
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    beta_1: [0,10] - Inverse temperature for first-stage choice softmax.
    beta_2: [0,10] - Inverse temperature for second-stage choice softmax.
    w: [0,1] - Weight on model-based vs. model-free strategy (1 = pure MB, 0 = pure MF).
    lam: [0,1] - Eligibility trace parameter updating stage 1 from stage 2 reward.
    p: [0,10] - Choice stickiness (perseveration) for the first-stage choice.
    """
    learning_rate, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine model-based and model-free Q-values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply stickiness (perseveration) to the previous choice
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + p * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates Model
Some participants update their beliefs differently based on whether outcomes are better or worse than expected. This model separates the learning rate into `alpha_pos` (for positive prediction errors) and `alpha_neg` (for negative prediction errors).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates Model.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta_1: [0,10] - Inverse temperature for first-stage choices.
    beta_2: [0,10] - Inverse temperature for second-stage choices.
    w: [0,1] - Weight on model-based vs. model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # action value updating for choice 1
        lr_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # action value updating for choice 2
        lr_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += lr_2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Forgetting (Decay) Q-learning Model
This model accounts for the phenomenon where unchosen options are gradually forgotten over time. Before applying the prediction errors to chosen options, all action values slowly decay toward zero.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Forgetting (Decay) Q-learning Model.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    beta_1: [0,10] - Inverse temperature for first-stage choices.
    beta_2: [0,10] - Inverse temperature for second-stage choices.
    w: [0,1] - Weight on model-based vs. model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    decay: [0,1] - Rate at which all Q-values decay towards 0 on each trial.
    """
    learning_rate, beta_1, beta_2, w, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # decay all values
        q_stage1_mf = q_stage1_mf * (1 - decay)
        q_stage2_mf = q_stage2_mf * (1 - decay)
        
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
