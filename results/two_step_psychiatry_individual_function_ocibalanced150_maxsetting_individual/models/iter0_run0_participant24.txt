def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Standard Daw et al. 2011 Two-Step Task Model.
    This model integrates model-based (MB) and model-free (MF) learning mechanisms, 
    an eligibility trace for updating stage 1 values from stage 2 rewards, 
    and a choice stickiness (perseveration) parameter.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weight given to model-based vs. model-free values (1 = purely MB, 0 = purely MF).
    lam: [0, 1] - Eligibility trace parameter for stage 2 reward updating stage 1 Q-value.
    rep: [-5, 5] - Choice stickiness (perseveration) for stage 1 choices.
    """
    alpha, beta, w, lam, rep = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + rep * stickiness
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        # Policy for the second choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
        
        # Learning / Action value updating
        r = reward[trial]
        
        # Stage 1 MF update based on stage 2 state value
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # Stage 2 MF update based on reward
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Eligibility trace: Stage 2 prediction error updates Stage 1 MF value
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with Asymmetric Learning Rates.
    This model assumes participants might learn differently from positive and negative 
    prediction errors (e.g., weighing rewards more heavily than omissions).
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors (delta > 0).
    alpha_neg: [0, 1] - Learning rate for negative prediction errors (delta <= 0).
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weight given to model-based vs. model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    rep: [-5, 5] - Choice stickiness (perseveration) for stage 1 choices.
    """
    alpha_pos, alpha_neg, beta, w, lam, rep = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + rep * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        # Policy for the second choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
        
        # Learning / Action value updating
        r = reward[trial]
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        q_stage1_mf[a1] += alpha2 * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
