def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Stage-specific learning rates, inverse temperatures, stage 1 perseveration, 
    and state-specific stage 2 perseveration.
    
    Parameters:
    alpha1: learning rate for stage 1 [0, 1]
    alpha2: learning rate for stage 2 [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    pi1: choice perseveration for stage 1 [-5, 5]
    pi2_state: state-specific choice perseveration for stage 2 [-5, 5]
    """
    alpha1, alpha2, beta1, beta2, w, lambda_, pi1, pi2_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_action_1 = -1
    prev_action_2_in_state = [-1, -1]

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * net_q1
        if prev_action_1 != -1:
            logits_1[prev_action_1] += pi1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if prev_action_2_in_state[state_idx] != -1:
            logits_2[prev_action_2_in_state[state_idx]] += pi2_state
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2
        
        prev_action_1 = action_1[trial]
        prev_action_2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
