def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates and inverse temperatures for Stage 1 and Stage 2,
    along with Model-Based weight and Stage 1 stickiness. This model captures the possibility 
    that participants learn and explore at different rates in the two stages of the task.
    
    Bounds:
    learning_rate1: [0,1] - Learning rate for Stage 1 prediction errors.
    learning_rate2: [0,1] - Learning rate for Stage 2 prediction errors.
    beta1: [0,10] - Inverse temperature for Stage 1 choices.
    beta2: [0,10] - Inverse temperature for Stage 2 choices.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness: [-5,5] - Perseveration parameter for Stage 1.
    """
    learning_rate1, learning_rate2, beta1, beta2, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness
            
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate2 * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Dual-beta model that uses separate inverse temperatures for Model-Based and Model-Free values 
    at Stage 1, effectively eliminating the explicit 'w' parameter. It also includes an 
    eligibility trace for Stage 1 updating based on Stage 2 rewards, and choice stickiness.
    
    Bounds:
    learning_rate: [0,1] - General learning rate for both stages.
    beta_mb: [0,10] - Inverse temperature scaling Model-Based values.
    beta_mf: [0,10] - Inverse temperature scaling Model-Free values.
    beta2: [0,10] - Inverse temperature for Stage 2 choices.
    stickiness: [-5,5] - Choice stickiness for Stage 1.
    lambda_trace: [0,1] - Eligibility trace for Stage 1 MF update from Stage 2 reward.
    """
    learning_rate, beta_mb, beta_mf, beta2, stickiness, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine using separate betas
        q_stage1_net = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness
            
        # Shift for numerical stability before exponentiation
        q_stage1_net_shifted = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(q_stage1_net_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Stage 1 MF update from transition
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Stage 2 MF update from reward
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for Stage 1 MF from Stage 2 reward
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_trace * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates (positive vs. negative RPEs), counterfactual 
    updating for unchosen Stage 2 actions, separate inverse temperatures for Stage 1 
    and Stage 2, Model-Based weight, and Stage 1 stickiness.
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors.
    lr_neg: [0,1] - Learning rate for negative prediction errors.
    counterfactual_rate: [0,1] - Learning rate for unchosen action in the chosen state.
    beta1: [0,10] - Inverse temperature for Stage 1.
    beta2: [0,10] - Inverse temperature for Stage 2.
    w: [0,1] - Model-based vs model-free weight (1 = pure MB).
    stickiness: [-5,5] - Choice stickiness for Stage 1.
    """
    lr_pos, lr_neg, counterfactual_rate, beta1, beta2, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness
            
        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        # Counterfactual updating for unchosen Stage 2 action
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        cf_delta = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += counterfactual_rate * cf_delta

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
