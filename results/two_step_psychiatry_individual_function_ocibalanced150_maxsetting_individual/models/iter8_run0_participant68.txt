Here are 3 new cognitive models that build upon the mechanisms of the best model so far while exploring novel parameter configurations that have not yet been tested. 

### Model 1: Asymmetric Eligibility Traces with Q-value Decay
This model tests the hypothesis that participants might update their Stage 1 actions differently depending on whether the outcome in Stage 2 was better or worse than expected (positive vs. negative Reward Prediction Error). It separates the eligibility trace parameter into `lam_pos` and `lam_neg`, while retaining the highly successful `decay` parameter.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric eligibility traces based on RPE sign with Q-value decay.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam_pos (float): Eligibility trace for positive RPEs. Bounds: [0,1]
    lam_neg (float): Eligibility trace for negative RPEs. Bounds: [0,1]
    pers (float): Perseveration for both stages. Bounds: [-5,5]
    decay (float): Decay rate for Q-values. Bounds: [0,1]
    """
    alpha1, alpha2, beta, w, lam_pos, lam_neg, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values towards zero
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Apply asymmetric eligibility trace based on RPE sign
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Beta and Perseveration for Stage 1 vs. Stage 2
This model tests whether participants have distinct exploration/exploitation balances (`beta`) and distinct tendencies to stick to previous choices (`pers`) in Stage 1 compared to Stage 2. It unifies stage-separated noise and stickiness into a single model without exceeding the parameter limit.

```python

    beta2 (float): Inverse temperature for Stage 2. Bounds: [0,10]
```

### Model 3: Reward-Dependent Perseveration with Q-value Decay
This model proposes that a participant's likelihood to repeat a past action (perseveration) is conditional on whether they were just rewarded or not (win-stay/lose-shift tendency). It introduces `pers_win` and `pers_lose` parameters, combined with the successful `decay` parameter.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent perseveration with Q-value decay.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers_win (float): Perseveration after a rewarded trial. Bounds: [-5,5]
    pers_lose (float): Perseveration after an unrewarded trial. Bounds: [-5,5]
    decay (float): Decay rate for Q-values. Bounds: [0,1]
    """
    alpha1, alpha2, beta, w, lam, pers_win, pers_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Determine perseveration based on last reward
        pers = pers_win if last_reward > 0 else pers_lose

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values towards zero
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
