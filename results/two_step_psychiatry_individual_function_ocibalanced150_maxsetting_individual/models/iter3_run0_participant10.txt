def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with asymmetric learning rates for positive and negative RPEs,
    single beta, w, lam, and separate perseveration for stage 1 and stage 2.
    
    Parameters:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta: [0,10] - Inverse temperature for both stages.
    w: [0,1] - Weight on model-based vs model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    pers_1: [-5,5] - Perseveration bonus for repeating the previous stage 1 choice.
    pers_2: [-5,5] - Perseveration bonus for repeating the previous stage 2 choice (in the same state).
    """
    alpha_pos, alpha_neg, beta, w, lam, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta * q_stage1_net
        if prev_a1 != -1:
            logits_1[prev_a1] += pers_1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if prev_a2[state_idx] != -1:
            logits_2[prev_a2[state_idx]] += pers_2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[a1] += alpha_1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
        q_stage1_mf[a1] += alpha_2 * lam * delta_stage2
        
        prev_a1 = a1
        prev_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with separate inverse temperatures (betas) for stage 1 and stage 2,
    single learning rate, w, lam, and separate perseveration for stage 1 and stage 2.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    beta_1: [0,10] - Inverse temperature for stage 1.

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with outcome-dependent perseveration for stage 1 (win-stay, lose-shift).
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    beta: [0,10] - Inverse temperature for both stages.
    w: [0,1] - Weight on model-based vs model-free strategy.
    lam: [0,1] - Eligibility trace parameter.
    pers_win: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous trial was rewarded.
    pers_lose: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous trial was unrewarded.
    """
    learning_rate, beta, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    prev_a1 = -1
    prev_r = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1 or reward[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta * q_stage1_net
        
        if prev_a1 != -1 and prev_r != -1:
            if prev_r > 0:
                logits_1[prev_a1] += pers_win
            else:
                logits_1[prev_a1] += pers_lose
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        prev_a1 = a1
        prev_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
