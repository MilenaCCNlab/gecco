def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free RL with separate learning rates for Stage 1 and Stage 2.
    This model posits that participants might update their values at different speeds 
    depending on the decision stage.
    
    Parameters:
    alpha1: learning rate for Stage 1 [0,1]
    alpha2: learning rate for Stage 2 [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free strategy [0,1]
    lam: eligibility trace parameter for stage 2 rewards [0,1]
    pers: perseveration bonus for repeating the last stage 1 action [0,1]
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        
        # Handle missing responses
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            pers_bonus = np.zeros(2)
            pers_bonus[last_action] = pers
            q_stage1_net += pers_bonus

        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free RL with perseveration at both Stage 1 and Stage 2.
    This model assumes that participants have a tendency to repeat their previous choices 
    not only at the first stage, but also at the second stage (state-dependent stickiness).
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free strategy [0,1]
    lam: eligibility trace parameter for stage 2 rewards [0,1]
    pers1: perseveration bonus for repeating the last stage 1 action [0,1]
    pers2: perseveration bonus for repeating the last stage 2 action in the same state [0,1]
    """
    alpha, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):

        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action1 != -1:
            pers_bonus1 = np.zeros(2)
            pers_bonus1[last_action1] = pers1
            q_stage1_net += pers_bonus1

        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            pers_bonus2 = np.zeros(2)
            pers_bonus2[last_action2[state_idx]] = pers2
            q_stage2_net += pers_bonus2

        exp_q2 = np.exp(beta2 * (q_stage2_net - np.max(q_stage2_net)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free RL with asymmetric learning rates for positive and 
    negative prediction errors, and a perseveration bonus for stage 1.
    This model assumes participants learn differently from better-than-expected versus 
    worse-than-expected outcomes, while still maintaining choice stickiness.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free strategy [0,1]
    lam: eligibility trace parameter for stage 2 rewards [0,1]
    pers: perseveration bonus for repeating the last stage 1 action [0,1]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            pers_bonus = np.zeros(2)
            pers_bonus[last_action] = pers
            q_stage1_net += pers_bonus

        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
