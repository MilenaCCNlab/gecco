import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid RL with Asymmetric Eligibility Traces and Stage 1/2 Perseveration
    
    This model extends the best model by allowing the eligibility trace (lambda) 
    to differ depending on whether the stage 2 reward prediction error is positive 
    or negative. This captures asymmetric credit assignment for better-than-expected 
    vs. worse-than-expected outcomes.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam_pos: [0, 1] - Eligibility trace parameter for positive stage 2 prediction errors.
    lam_neg: [0, 1] - Eligibility trace parameter for negative stage 2 prediction errors.
    p1: [-5, 5] - Perseveration parameter for stage 1 choice.
    p2: [-5, 5] - Perseveration parameter for stage 2 choice (specific to state).
    """
    learning_rate, beta, w, lam_pos, lam_neg, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1.0
            
        logits_1 = beta * q_stage1_net + p1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            pers2[last_a2[state_idx]] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + p2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        # Asymmetric eligibility trace update
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid RL with Forgetting of Unchosen Options and Perseveration
    
    This model introduces a forgetting mechanism where the values of unchosen actions 
    at both stage 1 and stage 2 decay towards their initial value of 0.5 over time. 
    This accounts for the regression to the mean of unexplored options.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam: [0, 1] - Eligibility trace parameter for stage 1 model-free update.
    p1: [-5, 5] - Perseveration parameter for stage 1 choice.
    p2: [-5, 5] - Perseveration parameter for stage 2 choice (specific to state).
    alpha_f: [0, 1] - Forgetting rate for unchosen actions (decay to 0.5).
    """
    learning_rate, beta, w, lam, p1, p2, alpha_f = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1.0
            
        logits_1 = beta * q_stage1_net + p1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            pers2[last_a2[state_idx]] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + p2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        # Eligibility trace
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Forgetting for unchosen action in stage 1
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += alpha_f * (0.5 - q_stage1_mf[unchosen_a1])
        
        # Forgetting for unchosen action in the visited state (stage 2)
        unchosen_a2 = 1 - a2
        q_stage2_mf[state_idx, unchosen_a2] += alpha_f * (0.5 - q_stage2_mf[state_idx, unchosen_a2])
        
        # Forgetting for actions in the unvisited state (stage 2)
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += alpha_f * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += alpha_f * (0.5 - q_stage2_mf[unvisited_state, 1])
        
        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid RL with Dynamic Transition Learning and Perseveration
    
    This model assumes that participants do not rely on a fixed transition matrix (0.7/0.3).
    Instead, they dynamically learn the state transition probabilities based on their 
    experienced transitions, updating the model-based system continuously.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam: [0, 1] - Eligibility trace parameter for stage 1 model-free update.
    p1: [-5, 5] - Perseveration parameter for stage 1 choice.
    p2: [-5, 5] - Perseveration parameter for stage 2 choice (specific to state).
    lr_trans: [0, 1] - Learning rate for updating the transition matrix.
    """
    learning_rate, beta, w, lam, p1, p2, lr_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1.0
            
        logits_1 = beta * q_stage1_net + p1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            pers2[last_a2[state_idx]] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + p2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Update transition matrix dynamically
        transition_matrix[a1, state_idx] += lr_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += lr_trans * (0.0 - transition_matrix[a1, 1 - state_idx])
        
        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
