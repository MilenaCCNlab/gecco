def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Baseline model with transition matrix learning.
    This model allows the participant to update their internal representation of the 
    transition probabilities between stage 1 actions and stage 2 states, rather than 
    assuming it stays fixed at 0.7/0.3.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for model-free Q-value updates.
    beta_1: [0,10] - Inverse temperature for stage 1 choices.
    beta_2: [0,10] - Inverse temperature for stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 values from stage 2 outcomes.
    stickiness: [-5,5] - General tendency to repeat the previous stage 1 action.
    transition_lr: [0,1] - Learning rate for updating the internal transition matrix.
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness, transition_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness
            
        exp_q1 = np.exp(beta_1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # transition matrix updating
        transition_matrix[action_1[trial], state_idx] += transition_lr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += transition_lr * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition-dependent stickiness.
    Stickiness parameter varies based on whether the previous transition was common or rare.
    Common transitions might encourage repeating a choice, while rare transitions might 
    encourage switching to test the environment.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for model-free Q-value updates.
    beta_1: [0,10] - Inverse temperature for stage 1 choices.
    beta_2: [0,10] - Inverse temperature for stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 values from stage 2 outcomes.
    stickiness_common: [-5,5] - Tendency to repeat the stage 1 action if the last transition was common.
    stickiness_rare: [-5,5] - Tendency to repeat the stage 1 action if the last transition was rare.
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness_common, stickiness_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_transition_common = False

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            stickiness = stickiness_common if last_transition_common else stickiness_rare
            q_stage1_net[last_action_1] += stickiness
            
        exp_q1 = np.exp(beta_1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        # Check if the transition was common (action 0 -> state 0, or action 1 -> state 1)
        last_transition_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Win-stay/Lose-shift stickiness combined with transition matrix learning.
    Builds on the best model so far by allowing the participant to update their internal
    transition matrix based on observed state transitions, while maintaining separate
    stickiness parameters for rewarded and unrewarded previous trials.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for model-free Q-value updates.
    beta_1: [0,10] - Inverse temperature for stage 1 choices.
    beta_2: [0,10] - Inverse temperature for stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 values from stage 2 outcomes.
    stickiness_win: [-5,5] - Tendency to repeat actions if the previous trial was rewarded.
    stickiness_lose: [-5,5] - Tendency to repeat actions if the previous trial was unrewarded.
    transition_lr: [0,1] - Learning rate for updating the internal transition matrix.
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness_win, stickiness_lose, transition_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            stickiness = stickiness_win if last_reward == 1 else stickiness_lose
            q_stage1_net[last_action_1] += stickiness
            
        exp_q1 = np.exp(beta_1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2 != -1 and last_state == state_idx:
            stickiness = stickiness_win if last_reward == 1 else stickiness_lose
            q_stage2_net[last_action_2] += stickiness

        exp_q2 = np.exp(beta_2 * q_stage2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # transition matrix updating
        transition_matrix[action_1[trial], state_idx] += transition_lr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += transition_lr * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
