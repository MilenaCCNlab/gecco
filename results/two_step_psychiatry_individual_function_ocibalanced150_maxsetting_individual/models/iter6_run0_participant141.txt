def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for stage 1 and stage 2, and reward-dependent perseveration.
    This tests if learning occurs at different rates for the two stages and if choice stickiness
    depends on whether the previous trial was rewarded or not.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for stage 1 prediction errors
    alpha2: [0, 1] - Learning rate for stage 2 prediction errors
    beta1: [0, 10] - Inverse temperature for stage 1 choices
    beta2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers_win: [-5, 5] - Perseveration parameter after a rewarded trial
    pers_lose: [-5, 5] - Perseveration parameter after an unrewarded trial
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = 0.0
        if last_action != -1:
            if last_reward == 1:
                pers = pers_win
            else:
                pers = pers_lose
                
        logits1 = beta1 * q_net
        if last_action != -1:
            logits1[last_action] += pers
            
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates and asymmetric eligibility traces for positive and negative 
    prediction errors. This evaluates whether the propagation of stage 2 prediction errors to stage 1 
    differs depending on the valence of the prediction error.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta1: [0, 10] - Inverse temperature for stage 1 choices
    beta2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam_pos: [0, 1] - Eligibility trace parameter for positive stage 2 prediction errors
    lam_neg: [0, 1] - Eligibility trace parameter for negative stage 2 prediction errors
    pers: [-5, 5] - Perseveration parameter for stage 1 choices
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)

        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg

    return log_loss
