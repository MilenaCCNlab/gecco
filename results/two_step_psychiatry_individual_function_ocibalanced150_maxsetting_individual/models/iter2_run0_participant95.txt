Here are 3 new cognitive models that build upon the most successful mechanism so far (value decay/forgetting) by combining it with other well-known reinforcement learning mechanisms. 

### Model 1: Value Decay + Asymmetric Learning Rates
This model combines value decay with asymmetric learning rates for positive and negative reward prediction errors. Participants might learn differently from better-than-expected outcomes compared to worse-than-expected outcomes, while still gradually forgetting the values of unchosen options.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Mixed MF-MB RL model with value decay and asymmetric learning rates.
    Values decay towards zero over time. Learning rates differ for positive and negative prediction errors.
    
    Bounds:
    alpha_pos: [0,1]
    alpha_neg: [0,1]
    decay: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    lam: [0,1]
    w: [0,1]
    stickiness: [-5,5]
    """
    alpha_pos, alpha_neg, decay, beta1, beta2, lam, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits1 = beta1 * q_net + stickiness * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value decay
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Value Decay + Dual Stage Stickiness
This model investigates whether perseveration (stickiness) applies not just to the first-stage choices (spaceships), but also independently to the second-stage choices (aliens). It combines this dual stickiness mechanism with the successful value decay mechanism.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Mixed MF-MB RL model with value decay and choice stickiness at both stages.
    Values decay towards zero over time. Stickiness is applied to both stage 1 and stage 2 choices independently.
    
    Bounds:
    alpha: [0,1]
    decay: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    lam: [0,1]
    w: [0,1]
    st1: [-5,5]
    st2: [-5,5]
    """
    alpha, decay, beta1, beta2, lam, w, st1, st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if last_action1 != -1:
            rep1[last_action1] = 1
            
        logits1 = beta1 * q_net + st1 * rep1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        rep2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            rep2[last_action2[state_idx]] = 1
            
        logits2 = beta2 * q_stage2_mf[state_idx] + st2 * rep2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value decay
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Regression to the Mean (Decay to 0.5)
In the previous best model, unchosen Q-values decayed to 0. In an environment where rewards are 0 or 1, the true expected value of a random choice is 0.5. This model initializes values at 0.5 and decays unchosen values back to this uninformative prior (0.5), capturing a "regression to the mean" assumption for unexplored options.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Mixed MF-MB RL model with value decay towards an uninformative prior (0.5).
    Instead of decaying to 0, Q-values decay towards 0.5, reflecting a regression to the mean for unchosen options.
    
    Bounds:
    alpha: [0,1]
    decay: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    lam: [0,1]
    w: [0,1]
    stickiness: [-5,5]
    """
    alpha, decay, beta1, beta2, lam, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits1 = beta1 * q_net + stickiness * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value decay (regression to 0.5)
        q_stage1_mf = q_stage1_mf * (1 - decay) + 0.5 * decay
        q_stage2_mf = q_stage2_mf * (1 - decay) + 0.5 * decay
        
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
