Here are 3 new cognitive models that build upon the Daw et al. (2011) hybrid model to explore different psychological mechanisms. They avoid the parameter combinations you have already tested while remaining parsimonious (each has 7 parameters).

### Model 1: Asymmetric Eligibility Traces (Positive vs. Negative RPE)
This model tests whether participants propagate credit back to the first-stage choice differently depending on whether the second-stage reward prediction error (RPE) was positive (better than expected) or negative (worse than expected). 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) hybrid model with asymmetric eligibility traces.
    The eligibility trace parameter depends on the sign of the stage 2 reward prediction error (RPE).
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - stage 1 inverse temperature
    beta2: [0, 10] - stage 2 inverse temperature
    lam_pos: [0, 1] - eligibility trace for positive RPEs
    lam_neg: [0, 1] - eligibility trace for negative RPEs
    w: [0, 1] - model-based weight
    p: [-5, 5] - perseveration bonus
    """
    alpha, beta1, beta2, lam_pos, lam_neg, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits_1 = beta1 * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Apply asymmetric eligibility trace to stage 1 MF value
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Choice Trace Perseveration
Standard models assume perseveration only applies to the immediately preceding trial. This model introduces a decaying choice trace, testing whether stickiness is influenced by a longer history of past actions (auto-correlated choices) rather than just the $t-1$ trial.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) hybrid model with an exponentially decaying perseveration trace.
    Instead of only the immediate past action, choice stickiness is based on a decaying trace of past actions.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - stage 1 inverse temperature
    beta2: [0, 10] - stage 2 inverse temperature
    lam: [0, 1] - eligibility trace
    w: [0, 1] - model-based weight
    p: [-5, 5] - perseveration bonus
    tau: [0, 1] - decay rate of the choice trace
    """
    alpha, beta1, beta2, lam, w, p, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits_1 = beta1 * q_net + p * choice_trace
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update choice trace
        choice_trace = choice_trace * tau
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-dependent Eligibility Traces
This model hypothesizes that the degree to which a participant updates their first-stage MF values based on second-stage outcomes depends on the *type of transition* (common vs. rare). Participants may be more likely to propagate credit backward if the sequence of events was expected (common transition) vs unexpected (rare transition).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) hybrid model with transition-dependent eligibility traces.
    The eligibility trace depends on whether the transition to stage 2 was common or rare.
    
    Bounds:
    alpha: [0, 1] - learning rate
    beta1: [0, 10] - stage 1 inverse temperature
    beta2: [0, 10] - stage 2 inverse temperature
    lam_c: [0, 1] - eligibility trace for common transitions
    lam_r: [0, 1] - eligibility trace for rare transitions
    w: [0, 1] - model-based weight
    p: [-5, 5] - perseveration bonus
    """
    alpha, beta1, beta2, lam_c, lam_r, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits_1 = beta1 * q_net + p * rep
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Determine if transition was common or rare (action 0 -> state 0, action 1 -> state 1 are common)
        is_common = (action_1[trial] == state_idx)
        lam = lam_c if is_common else lam_r
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
