def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with positive/negative learning rates, eligibility trace,
    and win-stay stickiness for both stage 1 and stage 2 choices.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive prediction errors
    alpha_neg: [0, 1] - learning rate for negative prediction errors
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p_win1: [-5, 5] - stage 1 stickiness if previous trial was rewarded
    p_win2: [-5, 5] - stage 2 stickiness if last visit to this state was rewarded
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p_win1, p_win2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward_1 = 0
    
    last_action_2 = np.array([-1, -1])
    last_reward_2 = np.array([0, 0])

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep1 = np.zeros(2)
        if last_action_1 != -1 and last_reward_1 > 0:
            rep1[last_action_1] = 1
            
        q1 = beta1 * q_net + p_win1 * rep1
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        rep2 = np.zeros(2)
        if last_action_2[state_idx] != -1 and last_reward_2[state_idx] > 0:
            rep2[last_action_2[state_idx]] = 1
            
        q2 = beta2 * q_stage2_mf[state_idx] + p_win2 * rep2
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1_current = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha2_current = alpha_pos if delta_stage2 > 0 else alpha_neg

        q_stage1_mf[action_1[trial]] += alpha1_current * delta_stage1 + alpha1_current * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_current * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward_1 = reward[trial]
        
        last_action_2[state_idx] = action_2[trial]
        last_reward_2[state_idx] = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
