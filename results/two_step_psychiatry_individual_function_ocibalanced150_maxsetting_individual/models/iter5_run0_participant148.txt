### Model 1: Asymmetric Learning with Stage-Specific Inverse Temperatures
This model proposes that participants not only learn differently from positive versus negative prediction errors, but they also exhibit different levels of exploration/exploitation (randomness) in Stage 1 compared to Stage 2. 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates with stage-specific inverse temperatures.
    Builds upon the best model by allowing different exploration/exploitation trade-offs in Stage 1 vs Stage 2.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive reward prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative reward prediction errors.
    beta1: [0, 10] - Inverse temperature for Stage 1.
    beta2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for Stage 1 MF updates from Stage 2 rewards.
    rho: [-5, 5] - Choice stickiness for Stage 1 (positive for perseveration, negative for alternation).
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if trial > 0:
            stickiness[action_1[trial - 1]] = 1
            
        logits_1 = beta1 * q_net + rho * stickiness
        logits_1 -= np.max(logits_1) # Numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Numerical stability
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Stickiness and Unchosen Decay
Participants might have a strong tendency to repeat spaceships (Stage 1) but not necessarily aliens (Stage 2), or vice-versa. This model introduces independent stickiness parameters for both stages, paired with a decay mechanism for unchosen options.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Single learning rate but separate choice stickiness for Stage 1 and Stage 2, along with unchosen Q-value decay.
    Captures differing tendencies to repeat actions at the first vs second stage of decision-making.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for Stage 1 MF updates from Stage 2 rewards.
    rho1: [-5, 5] - Choice stickiness for Stage 1.
    rho2: [-5, 5] - Choice stickiness for Stage 2 (applied if the same state is visited).
    decay: [0, 1] - Decay rate for unchosen Q-values.
    """
    alpha, beta, w, lam, rho1, rho2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if trial > 0:
            stickiness_1[action_1[trial - 1]] = 1
            
        logits_1 = beta * q_net + rho1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        stickiness_2 = np.zeros(2)
        # Apply stage 2 stickiness only if the participant landed on the same planet in the previous trial
        if trial > 0 and state[trial - 1] == state_idx:
            stickiness_2[action_2[trial - 1]] = 1
            
        logits_2 = beta * q_stage2_mf[state_idx] + rho2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen Q-values
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, :] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning with Outcome-Dependent Stickiness (Win-Stay / Lose-Shift)
This model combines the asymmetric learning rates (`alpha_pos` and `alpha_neg`) with a choice stickiness mechanism that strictly depends on whether the previous trial was rewarded or not (`rho_win` vs `rho_lose`), capturing both value-driven asymmetry and heuristic-driven asymmetry.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates combined with outcome-dependent stickiness (Win-Stay / Lose-Shift).
    Differentiates both learning and choice repetition based on whether the previous trial was rewarded.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive reward prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative reward prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based vs model-free values (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for Stage 1 MF updates from Stage 2 rewards.
    rho_win: [-5, 5] - Choice stickiness for Stage 1 following a reward.
    rho_lose: [-5, 5] - Choice stickiness for Stage 1 following no reward.
    """
    alpha_pos, alpha_neg, beta, w, lam, rho_win, rho_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        rho_current = 0
        if trial > 0:
            stickiness[action_1[trial - 1]] = 1
            if reward[trial - 1] > 0:
                rho_current = rho_win
            else:
                rho_current = rho_lose
            
        logits_1 = beta * q_net + rho_current * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Policy
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
