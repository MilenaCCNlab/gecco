import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Rare Transition Switch Bonus Model with Unchosen Decay.
    Participants may interpret rare transitions as a signal that the environment 
    has changed or that their action was "wrong", leading to a tendency to switch 
    actions on the subsequent trial specifically following a rare transition.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta1: inverse temperature for stage 1 [0, 10]
    beta2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p: regular perseveration bonus for stage 1 [0, 5]
    decay_rate: rate at which unchosen options decay to 0.5 [0, 1]
    p_rare_switch: bonus to switch actions after a rare transition [0, 5]
    """
    alpha, beta1, beta2, w, lambda_, p, decay_rate, p_rare_switch = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_state = -1
            continue

        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 2 policy
        q2_scaled = beta2 * q_stage2_mf[s]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
            # Check for rare transition on the previous trial
            if (last_action == 0 and last_state == 1) or (last_action == 1 and last_state == 0):
                pers[1 - last_action] += p_rare_switch
            
        q_net_scaled = beta1 * q_net + p * pers
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Updates
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        # Decay unchosen options
        q_stage1_mf[1 - a1] += decay_rate * (0.5 - q_stage1_mf[1 - a1])
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] += decay_rate * (0.5 - q_stage2_mf[state_idx, action_idx])
        
        last_action = a1
        last_state = s

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: State Generalization Model with Unchosen Decay.
    Participants may generalize learning from an action in one state (planet) 
    to the exact same action in the other state, as they share the same motor 
    response (e.g. left vs right button).
    
    Parameters:
    alpha: learning rate [0, 1]
    beta1: inverse temperature for stage 1 [0, 10]
    beta2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p: perseveration bonus for stage 1 [0, 5]
    decay_rate: rate at which unchosen options decay to 0.5 [0, 1]
    gen_rate: rate of generalizing the stage 2 prediction error to the other state [0, 1]
    """
    alpha, beta1, beta2, w, lambda_, p, decay_rate, gen_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 2 policy
        q2_scaled = beta2 * q_stage2_mf[s]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
        q_net_scaled = beta1 * q_net + p * pers
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Updates
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Generalization to the other state for the SAME action
        delta_stage2_gen = r - q_stage2_mf[1 - s, a2]
        q_stage2_mf[1 - s, a2] += gen_rate * alpha * delta_stage2_gen
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        # Decay unchosen options
        q_stage1_mf[1 - a1] += decay_rate * (0.5 - q_stage1_mf[1 - a1])
        for state_idx in range(2):
            for action_idx in range(2):
                # Decay the action that was not chosen in stage 2 (in both states)
                if action_idx != a2:
                    q_stage2_mf[state_idx, action_idx] += decay_rate * (0.5 - q_stage2_mf[state_idx, action_idx])
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Goal-directed (Planet) Perseveration Model.
    In addition to basic motor action perseveration, participants might perseverate 
    on the *state* (planet) they visited last, adding a choice bonus to the action 
    that commonly leads to that specific planet (goal-directed stickiness).
    
    Parameters:
    alpha: learning rate [0, 1]
    beta1: inverse temperature for stage 1 [0, 10]
    beta2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p: action perseveration bonus for stage 1 [0, 5]
    decay_rate: rate at which unchosen options decay to 0.5 [0, 1]
    p_planet: goal-directed perseveration bonus for the last visited planet [0, 5]
    """
    alpha, beta1, beta2, w, lambda_, p, decay_rate, p_planet = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_state = -1
            continue

        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 2 policy
        q2_scaled = beta2 * q_stage2_mf[s]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
        planet_bonus = np.zeros(2)
        if last_state != -1:
            # Action 0 commonly leads to Planet 0, Action 1 to Planet 1
            planet_bonus[last_state] = 1.0
            
        q_net_scaled = beta1 * q_net + p * pers + p_planet * planet_bonus
        q_net_scaled -= np.max(q_net_scaled)
        exp_q1 = np.exp(q_net_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Updates
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lambda_ * delta_stage2

        # Decay unchosen options
        q_stage1_mf[1 - a1] += decay_rate * (0.5 - q_stage1_mf[1 - a1])
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] += decay_rate * (0.5 - q_stage2_mf[state_idx, action_idx])
        
        last_action = a1
        last_state = s

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
