def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual-system model with asymmetric eligibility trace.
    The eligibility trace parameter (lambda) differs depending on whether the Stage 2 prediction error is positive or negative, capturing differences in credit assignment for positive vs negative surprises.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam_pos: [0, 1] (eligibility trace for positive PE)
    lam_neg: [0, 1] (eligibility trace for negative PE)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    """
    alpha, beta_1, beta_2, w, lam_pos, lam_neg, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + st * stickiness
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        lam = lam_pos if delta_stage2 > 0 else lam_neg

        # action value updating
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dual-system model with reward-dependent model-based weight.
    The weight placed on model-based control (w) depends on whether the previous trial was rewarded (w_win) or unrewarded (w_loss), capturing variations in cognitive flexibility based on recent rewards.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w_win: [0, 1] (MB weight after a win)
    w_loss: [0, 1] (MB weight after a loss)
    lam: [0, 1] (eligibility trace)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    """
    alpha, beta_1, beta_2, w_win, w_loss, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine w based on previous reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_loss
        else:
            w = (w_win + w_loss) / 2.0 # Default for first trial
            
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + st * stickiness
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        # action value updating
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dual-system model with forgetting/decay to a parameterized baseline.
    Unchosen actions decay towards a baseline value q0, capturing regression to a prior expectation rather than a neutral 0.5.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_1: [0, 10] (inverse temperature for Stage 1)
    beta_2: [0, 10] (inverse temperature for Stage 2)
    w: [0, 1] (weight for model-based vs model-free)
    lam: [0, 1] (eligibility trace parameter)
    st: [-5, 5] (choice stickiness/perseveration for Stage 1)
    decay: [0, 1] (decay rate for unchosen Q-values)
    q0: [0, 1] (baseline Q-value to which unchosen options decay)
    """
    alpha, beta_1, beta_2, w, lam, st, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    # Initialize Q-values to q0
    q_stage1_mf = np.ones(2) * q0
    q_stage2_mf = np.ones((2, 2)) * q0
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_net + st * stickiness
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]

        # action value updating
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Decay unchosen options towards q0
        q_stage1_mf[1 - a1] = (1 - decay) * q_stage1_mf[1 - a1] + decay * q0
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * q0

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
