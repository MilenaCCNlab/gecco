def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF with Decay, Perseveration, and Choice Bias.
    Includes a 'bias' parameter that captures a global preference for action 1 over action 0
    in both stages. This accounts for spatial or stimulus-specific biases often observed in participants.
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    decay: [0, 1] - decay rate for unchosen options
    p: [-5, 5] - stage 1 perseveration
    bias: [-5, 5] - global preference for action 1 over action 0
    """
    alpha, beta_1, beta_2, w, lam, decay, p, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += p
        logits_1[1] += bias  # Apply choice bias for action 1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2[1] += bias  # Apply choice bias for action 1
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Value updates
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay * (0.5 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF with Stage-Specific Decay and Perseveration.
    Separates the decay rate for unchosen options into stage 1 and stage 2,
    recognizing that memory for first-stage options (spaceships) and second-stage
    options (aliens) might decay at different rates over time.
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    decay_1: [0, 1] - decay rate for unchosen options in stage 1
    decay_2: [0, 1] - decay rate for unchosen options in stage 2
    p: [-5, 5] - stage 1 perseveration
    """
    alpha, beta_1, beta_2, w, lam, decay_1, decay_2, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:

        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Value updates
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2

            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay_2 * (0.5 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF with Reward-Dependent Model-Based Weight.
    The weight given to the model-based system depends on whether the previous
    trial was rewarded or not, capturing potential shifts in strategy (e.g., relying more 
    on the transition model after a success or becoming more model-free after a failure).
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w_rew: [0, 1] - model-based weight following a reward
    w_unrew: [0, 1] - model-based weight following no reward
    lam: [0, 1] - eligibility trace
    decay: [0, 1] - decay rate for unchosen options
    p: [-5, 5] - stage 1 perseveration
    """
    alpha, beta_1, beta_2, w_rew, w_unrew, lam, decay, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Determine current model-based weight
        if trial == 0:
            current_w = 0.5
        else:
            current_w = w_rew if reward[trial - 1] == 1 else w_unrew

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += p
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Value updates
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2

        # Decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay * (0.5 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
