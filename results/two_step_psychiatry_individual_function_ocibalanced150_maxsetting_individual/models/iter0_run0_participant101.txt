Here are three cognitive models proposed for the two-step decision-making task. These models build upon the foundational Daw et al. (2011) hybrid reinforcement learning framework, which is the gold standard for this task. The models integrate both Model-Free (MF) and Model-Based (MB) learning, while exploring different mechanisms for learning and choice perseverance.

### Model 1: Standard Hybrid Model with Eligibility Trace
This model implements the standard hybrid reinforcement learning model. It assumes that participants' stage 1 choices are driven by a weighted mixture (`w`) of model-based and model-free values. It also includes an eligibility trace (`lam`) which allows the stage 2 reward prediction error to directly update the stage 1 chosen action, and a choice perseverance parameter (`p`) to capture the tendency to repeat the previous stage 1 choice.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) Hybrid Model with Eligibility Trace.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control (1 = fully MB)
    lam: [0,1] - eligibility trace parameter
    p: [-2,2] - choice perseverance bonus (tendency to repeat previous stage 1 choice)
    """
    learning_rate, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MB and MF values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseverance bonus for the previously chosen action
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0  # Handle missing response

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0
  
        # action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1:
            # Stage 1 MF update (SARSA)
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1
            
            # Stage 2 MF update
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
            
            # Eligibility trace update for stage 1
            q_stage1_mf[a1] += learning_rate * lam * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid Model with Asymmetric Learning Rates
This variant proposes that participants learn differently from positive versus negative feedback. It splits the learning rate into `lr_pos` (used when the prediction error is positive) and `lr_neg` (used when the prediction error is negative), which is a common pattern observed in human learning biases.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Asymmetric Learning Rates.
    
    Parameters:
    lr_pos: [0,1] - learning rate for positive prediction errors
    lr_neg: [0,1] - learning rate for negative prediction errors
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    p: [-2,2] - choice perseverance bonus
    """
    lr_pos, lr_neg, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0
  
        # action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            lr1 = lr_pos if delta_stage1 > 0 else lr_neg
            q_stage1_mf[a1] += lr1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            lr2 = lr_pos if delta_stage2 > 0 else lr_neg
            q_stage2_mf[state_idx, a2] += lr2 * delta_stage2
            
            q_stage1_mf[a1] += lr2 * lam * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid Model with Reward-Dependent Perseverance
This model assumes that a participant's likelihood to repeat a Stage 1 choice is heavily influenced by the immediate reward outcome of the previous trial. Instead of a single perseverance parameter, it employs `p_rew` and `p_unrew` to isolate the "stay" behavior following successes vs. failures.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Reward-Dependent Perseverance.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    p_rew: [-2,2] - choice perseverance bonus after a rewarded trial
    p_unrew: [-2,2] - choice perseverance bonus after an unrewarded trial
    """
    learning_rate, beta1, beta2, w, lam, p_rew, p_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                pers[last_action_1] = p_rew
            else:
                pers[last_action_1] = p_unrew
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0
  
        # action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
            
            q_stage1_mf[a1] += learning_rate * lam * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1
            last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
