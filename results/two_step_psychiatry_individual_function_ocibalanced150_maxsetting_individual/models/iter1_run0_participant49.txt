Here are 3 new cognitive models that build upon the baseline hybrid model by incorporating additional well-known mechanisms in two-step task behavior, such as separate inverse temperatures for each stage, choice perseveration (stickiness), and unchosen option decay (forgetting).

### Model 1: Separate Inverse Temperatures & Stage 1 Perseveration
This model extends the baseline by allowing decision noise to vary between Stage 1 and Stage 2 (using `beta_1` and `beta_2`). It also includes a perseveration parameter (`pers`), which captures a participant's tendency to repeat their previous Stage 1 choice regardless of the outcome, a common pattern in behavioral data.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Free and Model-Based learning with separate inverse 
    temperatures for stage 1 and stage 2, eligibility trace, and stage 1 perseveration.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating values.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers: [0,1] - Perseveration (stickiness) bonus for repeating the previous stage 1 choice.
    """
    learning_rate, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseveration bonus to the previously chosen stage 1 action
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Value Decay (Forgetting) & Perseveration
This model introduces a `decay` parameter. In dynamic environments, participants often "forget" the values of unchosen options, pulling their Q-values back towards a neutral expected reward of 0.5. This model combines this decay mechanism with the perseveration and dual-beta structure.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with separate inverse temperatures, eligibility trace, 
    stage 1 perseveration, and a decay parameter that pulls all Q-values towards 0.5.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for value updates.
    decay: [0,1] - Decay rate pulling Q-values towards 0.5 on each trial.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter.
    pers: [0,1] - Perseveration bonus for repeating the previous stage 1 choice.
    """
    learning_rate, decay, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values to 0.5 when using decay towards 0.5
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Decay values towards 0.5 before choice
        q_stage1_mf += decay * (0.5 - q_stage1_mf)
        q_stage2_mf += decay * (0.5 - q_stage2_mf)

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Dual-Stage Perseveration
Sometimes participants exhibit habitual stickiness not just at the first stage (spaceship), but also at the second stage (aliens). This model introduces a secondary perseveration parameter (`pers_2`) specifically for repeating the alien choice when landing on the same planet.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with separate inverse temperatures, eligibility trace, 
    and perseveration parameters for both stage 1 and stage 2 choices.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for value updates.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter.
    pers_1: [0,1] - Perseveration bonus for repeating the previous stage 1 choice.
    pers_2: [0,1] - Perseveration bonus for repeating the previous stage 2 choice (in the same state).
    """
    learning_rate, beta_1, beta_2, w, lam, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) # Tracks the last action taken in each of the two states

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2

        exp_q2 = np.exp(beta_2 * (q_stage2_mf[state_idx] + pers_bonus_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
