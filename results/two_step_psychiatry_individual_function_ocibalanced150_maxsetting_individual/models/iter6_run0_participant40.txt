def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Hybrid MB/MF with decay, bias at stage 1, and perseveration at both stage 1 and stage 2.
    This model adds a stickiness parameter for the second stage to capture any tendency to repeat 
    the last chosen second-stage action, regardless of the state.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating values.
    beta: [0,10] - Inverse temperature for softmax choice.
    w: [0,1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0,1] - Eligibility trace for updating stage 1 with stage 2 prediction error.
    rep1: [-5,5] - Perseveration (stickiness) to the previously chosen stage 1 action.
    rep2: [-5,5] - Perseveration (stickiness) to the previously chosen stage 2 action.
    decay: [0,1] - Decay rate for unchosen action values towards 0.
    bias: [-5,5] - Intrinsic bias for choosing action 0 at stage 1.
    """
    import numpy as np
    learning_rate, beta, w, lam, rep1, rep2, decay, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness1[last_action_1] = 1.0
            
        logits_1 = beta * q_net + rep1 * stickiness1
        logits_1[0] += bias  
        logits_1 -= np.max(logits_1) 
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        stickiness2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness2[last_action_2] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + rep2 * stickiness2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.0 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.0 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):

    This model allows for different learning rates for positive and negative prediction errors, 
    accounting for potential biases in how participants learn from better-than-expected vs worse-than-expected outcomes.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta: [0,10] - Inverse temperature for softmax choice.
    w: [0,1] - Model-based weight (1 = fully model-based, 0 = fully model-free).

    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])

    p_choice_2 = np.zeros(n_trials)

    learning_rate: [0,1] - Learning rate for updating values.

    rep_lose: [-5,5] - Perseveration to the previously chosen stage 1 action if previous trial was unrewarded.
