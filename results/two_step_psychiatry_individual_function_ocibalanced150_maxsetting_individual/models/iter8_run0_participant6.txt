def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with Shared Win/Lose Stickiness across both stages.
    Reduces parameters by using the same win-stay/lose-shift parameters for Stage 1 and Stage 2.
    
    Parameters:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Stage 1 inverse temperature
    beta_2: [0, 10] - Stage 2 inverse temperature
    w: [0, 1] - Model-based weight
    p_win: [-5, 5] - Shared stickiness after a rewarded trial (applied to both stages)
    p_lose: [-5, 5] - Shared stickiness after an unrewarded trial (applied to both stages)
    lam: [0, 1] - Eligibility trace
    """
    alpha, beta_1, beta_2, w, p_win, p_lose, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_1[last_action_1] = p_win
            elif last_reward == 0:
                stickiness_1[last_action_1] = p_lose
                
        logits_1 = beta_1 * q_stage1_net + stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            if last_reward == 1:
                stickiness_2[last_action_2] = p_win
            elif last_reward == 0:
                stickiness_2[last_action_2] = p_lose
                
        logits_2 = beta_2 * q_stage2_mf[s] + stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with Win/Lose Stage 1 Stickiness and State-Dependent Stage 2 Stickiness.
    Stage 2 stickiness is only applied if the participant lands on the same planet as the previous trial.
    
    Parameters:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Stage 1 inverse temperature
    beta_2: [0, 10] - Stage 2 inverse temperature
    w: [0, 1] - Model-based weight
    p_1_win: [-5, 5] - Stage 1 stickiness after a rewarded trial
    p_1_lose: [-5, 5] - Stage 1 stickiness after an unrewarded trial
    p_2_state: [-5, 5] - Stage 2 stickiness applied only if the state is the same as the previous trial
    lam: [0, 1] - Eligibility trace
    """
    alpha, beta_1, beta_2, w, p_1_win, p_1_lose, p_2_state, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1 and last_state == s:
            stickiness_2[last_action_2] = p_2_state
                
        logits_2 = beta_2 * q_stage2_mf[s] + stickiness_2
        logits_2 -= np.max(logits_2)

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2
        last_state = s
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with Asymmetric Learning Rates for positive and negative RPEs, and Win/Lose Stage 1 Stickiness.
    Uses a single beta parameter for both stages to keep the parameter count at 8.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta: [0, 10] - Shared inverse temperature for both stages
    w: [0, 1] - Model-based weight
    p_1_win: [-5, 5] - Stage 1 stickiness after a rewarded trial
    p_1_lose: [-5, 5] - Stage 1 stickiness after an unrewarded trial
    p_2: [-5, 5] - General stage 2 stickiness
    lam: [0, 1] - Eligibility trace
    """
    alpha_pos, alpha_neg, beta, w, p_1_win, p_1_lose, p_2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_1[last_action_1] = p_1_win
            elif last_reward == 0:
                stickiness_1[last_action_1] = p_1_lose
                
        logits_1 = beta * q_stage1_net + stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness_2[last_action_2] = p_2
                
        logits_2 = beta * q_stage2_mf[s] + stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[a1] += alpha_1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha_2 * delta_stage2
        
        q_stage1_mf[a1] += alpha_2 * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
