import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF with different inverse temperatures for Stage 1 and Stage 2.
    Builds on the best model by allowing beta_1 and beta_2 to differ, while 
    retaining Stage 1 stickiness, reward stickiness, and Stage 2 action stickiness.
    This allows for different levels of choice stochasticity in the two stages.
    
    Bounds:
    learning_rate: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    stickiness: [-5, 5]
    reward_stickiness: [-5, 5]
    action_2_stickiness: [-5, 5]
    """
    learning_rate, beta_1, beta_2, w, stickiness, reward_stickiness, action_2_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            bonus_1[last_action_1] += stickiness
            if last_reward == 1:
                bonus_1[last_action_1] += reward_stickiness
                
        exp_q1 = np.exp(beta_1 * q_stage1_net + bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        bonus_2 = np.zeros(2)
        if last_action_2 != -1:
            bonus_2[last_action_2] += action_2_stickiness
            
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] + bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF with eligibility traces and multiple stickiness parameters.
    Includes the best model's features (w, stickiness, reward_stickiness, action_2_stickiness)
    and adds an eligibility trace (lam) to propagate stage 2 reward prediction errors 
    back to stage 1 model-free values.
    
    Bounds:
    learning_rate: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    stickiness: [-5, 5]
    reward_stickiness: [-5, 5]
    action_2_stickiness: [-5, 5]
    lam: [0, 1]
    """
    learning_rate, beta, w, stickiness, reward_stickiness, action_2_stickiness, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            bonus_1[last_action_1] += stickiness
            if last_reward == 1:
                bonus_1[last_action_1] += reward_stickiness
                
        exp_q1 = np.exp(beta * q_stage1_net + bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        bonus_2 = np.zeros(2)
        if last_action_2 != -1:
            bonus_2[last_action_2] += action_2_stickiness
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for choice 1 based on stage 2 prediction error
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
