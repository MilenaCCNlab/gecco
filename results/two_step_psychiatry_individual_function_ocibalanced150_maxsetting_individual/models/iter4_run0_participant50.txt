import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate inverse temperatures (beta) for Stage 1 and Stage 2 with a forgetting rate.
    This allows the model to capture different levels of exploration/exploitation at each stage.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating values.
    beta_1: [0, 10] - Inverse temperature for softmax action selection in stage 1.
    beta_2: [0, 10] - Inverse temperature for softmax action selection in stage 2.
    w: [0, 1] - Model-based weight (1 = pure model-based, 0 = pure model-free).
    eligibility: [0, 1] - Eligibility trace for updating stage 1 values from stage 2 rewards.
    stickiness: [-5, 5] - Choice stickiness (perseverance) for stage 1.
    forgetting_rate: [0, 1] - Decay rate for unchosen actions towards 0.5.
    """
    learning_rate, beta_1, beta_2, w, eligibility, stickiness, forgetting_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 Policy and Updates
        if a2 != -1:
            logits_2 = beta_2 * q_stage2_mf[s]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = r - q_stage2_mf[s, a2]
            q_stage2_mf[s, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += eligibility * learning_rate * delta_stage2

            # Decay unchosen stage 2 actions
            for state_idx in range(2):
                for action_idx in range(2):
                    if not (state_idx == s and action_idx == a2):
                        q_stage2_mf[state_idx, action_idx] = (1 - forgetting_rate) * q_stage2_mf[state_idx, action_idx] + forgetting_rate * 0.5
        else:
            p_choice_2[trial] = 1.0
            q_stage2_mf = (1 - forgetting_rate) * q_stage2_mf + forgetting_rate * 0.5

        # Decay unchosen stage 1 action
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1 - forgetting_rate) * q_stage1_mf[unchosen_a1] + forgetting_rate * 0.5

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Choice Kernel for Stage 1 and a forgetting rate for Q-values.
    Replaces static stickiness with an exponentially updating choice kernel to better capture dynamic response perseveration.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Model-based weight (1 = pure model-based, 0 = pure model-free).
    eligibility: [0, 1] - Eligibility trace for updating stage 1 values from stage 2 rewards.
    ck_alpha: [0, 1] - Learning rate for the choice kernel.
    ck_weight: [-5, 5] - Weight of the choice kernel in stage 1 action selection.
    forgetting_rate: [0, 1] - Decay rate for unchosen actions towards 0.5.
    """
    learning_rate, beta, w, eligibility, ck_alpha, ck_weight, forgetting_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    ck = np.zeros(2)

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Stage 1 Policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net + ck_weight * ck
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 Policy and Updates
        if a2 != -1:
            logits_2 = beta * q_stage2_mf[s]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = r - q_stage2_mf[s, a2]
            q_stage2_mf[s, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += eligibility * learning_rate * delta_stage2

            # Decay unchosen stage 2 actions
            for state_idx in range(2):
                for action_idx in range(2):
                    if not (state_idx == s and action_idx == a2):
                        q_stage2_mf[state_idx, action_idx] = (1 - forgetting_rate) * q_stage2_mf[state_idx, action_idx] + forgetting_rate * 0.5
        else:
            p_choice_2[trial] = 1.0
            q_stage2_mf = (1 - forgetting_rate) * q_stage2_mf + forgetting_rate * 0.5

        # Decay unchosen stage 1 action
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1 - forgetting_rate) * q_stage1_mf[unchosen_a1] + forgetting_rate * 0.5

        # Update choice kernel
        ck[a1] = (1 - ck_alpha) * ck[a1] + ck_alpha * 1
        ck[unchosen_a1] = (1 - ck_alpha) * ck[unchosen_a1]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
