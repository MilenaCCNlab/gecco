Here are 3 new cognitive models that build upon the best model so far while exploring alternative mechanisms for learning and choice behavior. 

### Model 1: Stage-specific Learning Rates + Decay + Stage 1 Stickiness
This model separates the learning rate for the first stage (`alpha1`) and the second stage (`alpha2`). Participants might update the values of the spaceships differently from the values of the aliens, given that the spaceships involve a transition probability while the aliens provide direct rewards. It also retains unchosen decay and stage 1 stickiness.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Stage-specific Learning Rates + Decay + Stage 1 Stickiness
    
    This model assumes separate learning rates for stage 1 and stage 2 updates, 
    allowing for differential learning speeds for spaceships vs. aliens.
    
    Bounds:
    alpha1: [0,1] - Learning rate for stage 1
    alpha2: [0,1] - Learning rate for stage 2
    decay: [0,1] - Decay rate for unchosen actions
    beta: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace
    stick1: [-5,5] - Choice stickiness for stage 1
    """
    import numpy as np
    alpha1, alpha2, decay, beta, beta2, w, lam, stick1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stick1
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward Sensitivity + Decay + Stage 1 Stickiness
This model adds a reward sensitivity parameter (`rho`). The participant might subjectively experience or value the 1 coin reward as larger or smaller than 1, thereby scaling the prediction errors globally and altering how deterministic their choices become without solely relying on inverse temperature. 

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward Sensitivity + Decay + Stage 1 Stickiness
    
    This model introduces a reward sensitivity parameter (rho) that scales the 
    magnitude of the reward before calculating prediction errors.
    
    Bounds:
    learning_rate: [0,1] - Learning rate
    decay: [0,1] - Decay rate for unchosen actions
    beta: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace
    stick1: [-5,5] - Choice stickiness for stage 1
    rho: [0,5] - Reward sensitivity scaling factor
    """

    learning_rate, decay, beta, beta2, w, lam, stick1, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

            q_stage1_net[last_action_1] += stick1

        state_idx = state[trial]

        q2_net = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        
        # Scale reward by rho
        scaled_reward = reward[trial] * rho
        delta_stage2 = scaled_reward - q_stage2_mf[state_idx, action_2[trial]]

        unchosen_1 = 1 - action_1[trial]

        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-dependent Stage 1 Stickiness + Decay
Rather than assigning a uniform stickiness parameter for stage 1 (or separately modelling stage 2 stickiness), this model distinguishes stage 1 stickiness based on whether the previous trial was rewarded or unrewarded. This captures win-stay/lose-shift heuristics directly at the top-level choice.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent Stage 1 Stickiness + Decay
    
    This model assumes that the tendency to repeat a stage 1 choice depends on 
    whether the previous trial resulted in a reward or no reward, capturing 
    win-stay/lose-shift heuristics independent of Q-value learning.
    
    Bounds:
    learning_rate: [0,1] - Learning rate
    decay: [0,1] - Decay rate for unchosen actions
    beta: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace
    stick_rew1: [-5,5] - Choice stickiness for stage 1 if previously rewarded
    stick_unrew1: [-5,5] - Choice stickiness for stage 1 if previously unrewarded
    """
    import numpy as np
    learning_rate, decay, beta, beta2, w, lam, stick_rew1, stick_unrew1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            if last_reward > 0:
                q_stage1_net[last_action_1] += stick_rew1
            else:
                q_stage1_net[last_action_1] += stick_unrew1
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
