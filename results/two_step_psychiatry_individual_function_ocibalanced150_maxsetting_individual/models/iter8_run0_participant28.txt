def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF RL with Choice Stickiness and Initial Q-values.
    Adds a free parameter for the initial Q-values to capture prior expectations
    which can influence early exploratory behavior and perseveration.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Choice stickiness (perseveration) parameter.
    q_init: [0,1] - Initial Q-value for all actions and states.
    """
    learning_rate, beta, w, lam, rho, q_init = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values with q_init
    q_stage1_mf = np.full(2, q_init, dtype=float)
    q_stage2_mf = np.full((2, 2), q_init, dtype=float)
    
    last_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits1 = beta * q_stage1_net + rho * stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF RL with Choice Stickiness and Planet Stickiness.
    Adds a parameter `phi` that increases the probability of choosing the spaceship
    that commonly leads to the planet visited in the previous trial, capturing a 
    preference for state-based perseveration over purely action-based perseveration.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Action choice stickiness (perseveration) parameter.
    phi: [-5,5] - Planet (state) stickiness parameter.
    """
    learning_rate, beta, w, lam, rho, phi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_state = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        action_stickiness = np.zeros(2)
        if last_a1 != -1:
            action_stickiness[last_a1] = 1.0
            
        planet_stickiness = np.zeros(2)
        if last_state != -1:
            # Add a bonus proportional to the probability of reaching the last state
            planet_stickiness = transition_matrix[:, last_state]
            
        logits1 = beta * q_stage1_net + rho * action_stickiness + phi * planet_stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]
        last_state = state_idx

        # policy for the second choice
        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF RL with Transition-Dependent Learning Rates.
    This model uses different learning rates for trials with common versus rare
    transitions, capturing the possibility that surprise modulates the update size.
    
    Parameters:
    lr_common: [0,1] - Learning rate for common transitions.
    lr_rare: [0,1] - Learning rate for rare transitions.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Choice stickiness (perseveration) parameter.
    """
    lr_common, lr_rare, beta, w, lam, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits1 = beta * q_stage1_net + rho * stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        # Determine if transition was common or rare
        is_common = (action_1[trial] == state_idx)
        current_lr = lr_common if is_common else lr_rare

        # policy for the second choice
        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += current_lr * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += current_lr * delta_stage2

        q_stage1_mf[action_1[trial]] += current_lr * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
