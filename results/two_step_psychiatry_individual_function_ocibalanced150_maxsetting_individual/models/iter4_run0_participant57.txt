def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with exponentially decaying choice traces to capture long-term perseveration.
    
    Bounds:
    alpha: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers1: [-5, 5]
    pers2: [-5, 5]
    trace_decay: [0,1]
    """
    alpha, beta1, beta2, w, lam, pers1, pers2, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            choice_trace_1 *= trace_decay
            choice_trace_2 *= trace_decay
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        q_stage1_total = beta1 * q_stage1_net + pers1 * choice_trace_1
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2 * choice_trace_2[state_idx]
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # update choice traces
        choice_trace_1 *= trace_decay
        choice_trace_1[action_1[trial]] += 1.0
        
        choice_trace_2 *= trace_decay
        choice_trace_2[state_idx, action_2[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model that dynamically learns the transition matrix between stages.
    
    Bounds:
    alpha: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers1: [-5, 5]
    pers2: [-5, 5]
    alpha_T: [0,1]
    """
    alpha, beta1, beta2, w, lam, pers1, pers2, alpha_T = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers1_bonus[last_action_1] = pers1
            
        q_stage1_total = beta1 * q_stage1_net + pers1_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2_bonus = np.zeros(2)
        if last_action_2 != -1:
            pers2_bonus[last_action_2] = pers2
            
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2_bonus
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # transition matrix updating
        transition_matrix[action_1[trial], state_idx] += alpha_T * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_T * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with a forgetting mechanism where unchosen Q-values decay towards a neutral value (0.5).
    
    Bounds:
    alpha: [0,1]
    beta1: [0,10]
    beta2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers1: [-5, 5]
    pers2: [-5, 5]
    forget_rate: [0,1]
    """
    alpha, beta1, beta2, w, lam, pers1, pers2, forget_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers1_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers1_bonus[last_action_1] = pers1
            
        q_stage1_total = beta1 * q_stage1_net + pers1_bonus
        
        exp_q1 = np.exp(q_stage1_total - np.max(q_stage1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2_bonus = np.zeros(2)
        if last_action_2 != -1:
            pers2_bonus[last_action_2] = pers2
            
        q_stage2_total = beta2 * q_stage2_mf[state_idx] + pers2_bonus
        
        exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # forgetting for unchosen actions
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += forget_rate * (0.5 - q_stage1_mf[a])
                
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != action_2[trial]:
                    q_stage2_mf[s, a] += forget_rate * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
