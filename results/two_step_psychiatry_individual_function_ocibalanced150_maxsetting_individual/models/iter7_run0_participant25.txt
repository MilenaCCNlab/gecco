def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free RL with a lapse rate (epsilon).
    Incorporates a probability of making a completely random choice at both stages,
    accounting for attention lapses or exploration not captured by softmax.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free strategy [0,1]
    lam: eligibility trace parameter for stage 2 rewards [0,1]
    pers: perseveration bonus for repeating the last stage 1 action [0,1]
    epsilon: lapse rate, probability of random choice [0,1]
    """
    alpha, beta1, beta2, w, lam, pers, epsilon = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            pers_bonus = np.zeros(2)
            pers_bonus[last_action] = pers
            q_stage1_net += pers_bonus

        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        probs_1 = (1 - epsilon) * probs_1 + epsilon * 0.5
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        probs_2 = (1 - epsilon) * probs_2 + epsilon * 0.5
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free RL with transition-dependent perseveration.
    The perseveration bonus at stage 1 depends on whether the previous trial's 
    transition was common or rare, capturing more nuanced stay/switch behavior.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free strategy [0,1]
    lam: eligibility trace parameter for stage 2 rewards [0,1]
    pers_common: perseveration bonus if previous transition was common [0,1]
    pers_rare: perseveration bonus if previous transition was rare [0,1]
    """
    alpha, beta1, beta2, w, lam, pers_common, pers_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_is_common = False

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        if last_action != -1:
            pers_bonus = np.zeros(2)
            if last_trans_is_common:
                pers_bonus[last_action] = pers_common
            else:
                pers_bonus[last_action] = pers_rare
            q_stage1_net += pers_bonus

        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        # Transition is common if action == state (0->0 or 1->1)
        last_trans_is_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free RL with subjective transition probabilities.
    Instead of assuming the fixed objective 0.7/0.3 transition probabilities,
    this model fits a subjective common transition probability `p_common` to 
    account for individual differences in mental models of the task structure.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: weight of model-based vs model-free strategy [0,1]
    lam: eligibility trace parameter for stage 2 rewards [0,1]
    pers: perseveration bonus for repeating the last stage 1 action [0,1]

    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
