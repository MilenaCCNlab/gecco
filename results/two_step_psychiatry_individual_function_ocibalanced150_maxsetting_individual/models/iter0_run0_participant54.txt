Here are three cognitive models as Python functions, designed to explain participant behavior in the two-step decision-making task. The models build on the standard hybrid reinforcement learning model (Daw et al., 2011), incorporating different cognitive mechanisms frequently observed in human behavior.

### Model 1: Hybrid RL with Eligibility Trace
This model assumes the participant's first-stage choice is a weighted mixture of model-free and model-based values. Crucially, it includes an eligibility trace ($\lambda$), meaning that the reward prediction error from the second stage also updates the model-free value of the first-stage action, tying distal rewards to earlier choices.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with eligibility trace (Daw et al., 2011).
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    """
    alpha, beta, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        if a2 != -1:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
            q_stage1_mf[a1] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2
            
            # Eligibility trace update for choice 1
            q_stage1_mf[a1] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid RL with Choice Perseverance (Stickiness)
Participants often exhibit "stickiness," a tendency to repeat their previous first-stage choice regardless of the reward outcome. This model introduces a perseverance parameter that acts as a bonus to the logits of the previously chosen action at stage 1. 

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Choice Perseverance (Stickiness).
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    pers: perseverance (stickiness) for stage 1 choice [0, 5]
    """
    alpha, beta, w, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net
        if last_a1 != -1:
            logits_1[last_a1] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        if a2 != -1:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
            q_stage1_mf[a1] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2
            
        last_a1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid RL with Asymmetric Learning Rates
This model explores the possibility that the participant learns differently from positive versus negative feedback. It disentangles the learning rate into two separate parameters: `alpha_pos` for positive prediction errors (better than expected) and `alpha_neg` for negative prediction errors (worse than expected).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Learning Rates.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    """
    alpha_pos, alpha_neg, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        if a2 != -1:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        if a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
            q_stage1_mf[a1] += (alpha_pos if delta_stage1 > 0 else alpha_neg) * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
            q_stage2_mf[state_idx, a2] += (alpha_pos if delta_stage2 > 0 else alpha_neg) * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
