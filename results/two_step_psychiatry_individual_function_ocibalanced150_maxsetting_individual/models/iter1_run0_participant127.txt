def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Separate Learning Rates for Stage 1 and Stage 2, and Choice Stickiness.
    This model extends the baseline hybrid model by allowing learning to occur at different rates 
    for the first-stage (spaceship) and second-stage (alien) choices, while maintaining a stickiness 
    parameter for first-stage choices to capture perseveration.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2: learning rate for stage 2 [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p: stickiness parameter for stage 1 [0, 10]
    """
    alpha_1, alpha_2, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + p * stickiness)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx, :])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Choice Stickiness for both Stage 1 and Stage 2.
    This model adds an independent stickiness parameter for the second-stage choices, capturing 
    tendencies to persevere on specific aliens within a given planet, in addition to spaceships.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p_1: stickiness parameter for stage 1 [0, 10]
    p_2: stickiness parameter for stage 2 [0, 10]
    """
    alpha, beta, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) # Tracks last action taken in each stage 2 state

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + p_1 * stickiness_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx, :] + p_2 * stickiness_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
