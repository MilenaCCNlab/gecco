### Model 1: Separate Stage Learning Rates + Reward-Dependent Perseveration
This model extends the best model by allowing different learning rates for Stage 1 and Stage 2. This accounts for the possibility that the participant updates their values at different speeds for the spaceships compared to the aliens.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for Stage 1 and Stage 2, with reward-dependent perseveration.
    
    Parameters:
    learning_rate_1: [0,1] - Learning rate for updating stage 1 values.
    learning_rate_2: [0,1] - Learning rate for updating stage 2 values.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers_win: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous trial was rewarded.
    pers_lose: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous trial was unrewarded.
    """
    learning_rate_1, learning_rate_2, beta_1, beta_2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers_bonus[last_action_1] = pers_win
            else:
                pers_bonus[last_action_1] = pers_lose
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1 + learning_rate_1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Stage 1 Perseveration + Stage 2 Perseveration
This model adds a stickiness parameter for the second stage. Often, participants not only perseverate on their first-stage choices (spaceships) but also show a tendency to repeat their second-stage choices (aliens) when they land on the same planet again.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Best model with an additional perseveration parameter for Stage 2.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating values.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weighting between model-based (1) and model-free (0) at stage 1.
    lam: [0,1] - Eligibility trace.
    pers_win: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous trial was rewarded.
    pers_lose: [-5,5] - Perseveration bonus for repeating stage 1 choice if previous trial was unrewarded.
    pers_stage2: [-5,5] - Perseveration bonus for repeating stage 2 choice in the same state.
    """
    learning_rate, beta_1, beta_2, w, lam, pers_win, pers_lose, pers_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.full(2, -1) # Track last action 2 for each state

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers_bonus_1[last_action_1] = pers_win
            else:
                pers_bonus_1[last_action_1] = pers_lose
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_stage2

        exp_q2 = np.exp(beta_2 * (q_stage2_mf[state_idx] + pers_bonus_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Model-Based vs. Model-Free Weighting
This model tests the hypothesis that the participant's reliance on model-based planning fluctuates based on recent outcomes. A rewarded trial might lead to more automatic (model-free) behavior on the next trial, while an unrewarded trial might trigger more effortful (model-based) planning, or vice versa.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent model-based weighting (w_win, w_lose).
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating values.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w_win: [0,1] - Weighting between model-based and model-free at stage 1 if previous trial rewarded.
    w_lose: [0,1] - Weighting between model-based and model-free at stage 1 if previous trial unrewarded.
    lam: [0,1] - Eligibility trace.
    pers: [-5,5] - General perseveration bonus for repeating stage 1 choice.
    """
    learning_rate, beta_1, beta_2, w_win, w_lose, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        # Reward-dependent weighting
        current_w = w_win if (last_reward == 1 or last_reward == -1) else w_lose
        q_stage1_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
