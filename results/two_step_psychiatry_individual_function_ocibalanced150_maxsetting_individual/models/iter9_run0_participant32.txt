Here are 3 new cognitive models that introduce novel mechanisms (lapse rate, asymmetric eligibility traces, and decoupled model-based/model-free inverse temperatures) while avoiding previously tried parameter combinations. Each function is standalone and assumes `numpy as np` is already imported.

### Model 1: Lapse Rate (Epsilon-Greedy) Model
This model builds on the best-performing mechanisms but incorporates an `epsilon` parameter to account for random lapses in decision-making (e.g., due to momentary inattention or exploration), which is a common source of noise in participant data.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with eligibility trace, MB/MF weighting, stage 1 and 2 perseveration, 
    and a lapse rate (epsilon) for random choices.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta: [0, 10] (inverse temperature)
    w: [0, 1] (model-based weight)
    lam: [0, 1] (eligibility trace)
    pers1: [-5, 5] (stage 1 perseveration bonus)
    pers2: [-5, 5] (stage 2 perseveration bonus)
    epsilon: [0, 1] (lapse rate / random choice probability)
    """
    alpha, beta, w, lam, pers1, pers2, epsilon = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers1
            
        exp_q1 = np.exp(beta * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # Apply lapse rate
        probs_1 = (1 - epsilon) * probs_1 + (epsilon / 2.0)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        last_action_1 = action_1[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers2

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        # Apply lapse rate
        probs_2 = (1 - epsilon) * probs_2 + (epsilon / 2.0)
        p_choice_2[trial] = probs_2[action_2[trial]]
        last_action_2[state_idx] = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Eligibility Trace Model
This model splits the eligibility trace parameter into two (`lam_pos` and `lam_neg`) depending on whether the reward prediction error at Stage 2 was positive or negative. This captures behavior where participants credit Stage 1 choices differently for unexpected rewards versus unexpected omissions.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric eligibility trace based on the sign of the Stage 2 reward prediction error.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta: [0, 10] (inverse temperature)
    w: [0, 1] (model-based weight)
    lam_pos: [0, 1] (eligibility trace for positive Stage 2 RPE)
    lam_neg: [0, 1] (eligibility trace for negative Stage 2 RPE)
    pers1: [-5, 5] (stage 1 perseveration bonus)
    pers2: [-5, 5] (stage 2 perseveration bonus)
    """
    alpha, beta, w, lam_pos, lam_neg, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers1
            
        exp_q1 = np.exp(beta * q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        last_action_1 = action_1[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers2

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        last_action_2[state_idx] = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Asymmetric Eligibility trace update
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Decoupled MB/MF Inverse Temperatures Model
Instead of a single weighting parameter `w`, this model fits separate inverse temperatures for the Model-Based (`beta_mb`) and Model-Free (`beta_mf`) values at Stage 1, alongside an independent Stage 2 inverse temperature (`beta2`). This allows the overall determinism of Stage 1 choices to scale dynamically depending on the alignment of MB and MF systems.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate inverse temperatures for model-based (beta_mb) and model-free (beta_mf) 
    values at Stage 1, allowing independent scaling of their contributions.
    
    Bounds:
    alpha: [0, 1] (learning rate)
    beta_mb: [0, 10] (inverse temp for MB values at Stage 1)
    beta_mf: [0, 10] (inverse temp for MF values at Stage 1)
    beta2: [0, 10] (inverse temp for Stage 2)
    lam: [0, 1] (eligibility trace)
    pers1: [-5, 5] (stage 1 perseveration bonus)
    pers2: [-5, 5] (stage 2 perseveration bonus)
    """
    alpha, beta_mb, beta_mf, beta2, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Linear combination using separate betas
        q_stage1_net = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers1
            
        exp_q1 = np.exp(q_stage1_net + pers_bonus_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        last_action_1 = action_1[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers2

        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        last_action_2[state_idx] = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
