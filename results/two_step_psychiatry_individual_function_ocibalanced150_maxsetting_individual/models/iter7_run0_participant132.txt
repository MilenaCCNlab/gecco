def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition matrix learning and separate beta_mb, beta_mf.
    Instead of assuming a fixed transition matrix, participants update their 
    internal transition model based on experienced state transitions.
    
    Bounds:
    learning_rate: [0, 1]
    transition_lr: [0, 1]
    beta_mb: [0, 10]
    beta_mf: [0, 10]
    beta2: [0, 10]
    pi1: [-5, 5]
    pi2: [-5, 5]
    lam: [0, 1]
    """
    learning_rate, transition_lr, beta_mb, beta_mf, beta2, pi1, pi2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            pers1[last_action1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            a2 = action_2[trial]
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += learning_rate * lam * delta_stage2

            last_action2[state_idx] = a2
        else:
            p_choice_2[trial] = 1.0
            
        # Update transition matrix probabilities based on experienced transitions
        transition_matrix[a1, state_idx] += transition_lr * (1 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += transition_lr * (0 - transition_matrix[a1, 1 - state_idx])
        
        last_action1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with continuous choice traces (alpha_c) instead of 1-trial perseveration,
    combined with decoupled beta_mb and beta_mf. This models a longer-term habit
    or stickiness to recent choices rather than just the immediately preceding one.
    
    Bounds:
    learning_rate: [0, 1]
    alpha_c: [0, 1]
    beta_mb: [0, 10]
    beta_mf: [0, 10]
    beta2: [0, 10]
    pi1: [-5, 5]
    pi2: [-5, 5]
    lam: [0, 1]
    """
    learning_rate, alpha_c, beta_mb, beta_mf, beta2, pi1, pi2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    choice_trace1 = np.zeros(2)
    choice_trace2 = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi1 * choice_trace1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            a2 = action_2[trial]
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * choice_trace2[state_idx]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += learning_rate * lam * delta_stage2

            choice_trace2[state_idx, a2] += alpha_c * (1 - choice_trace2[state_idx, a2])
            choice_trace2[state_idx, 1 - a2] += alpha_c * (0 - choice_trace2[state_idx, 1 - a2])
        else:
            p_choice_2[trial] = 1.0
            
        choice_trace1[a1] += alpha_c * (1 - choice_trace1[a1])
        choice_trace1[1 - a1] += alpha_c * (0 - choice_trace1[1 - a1])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with win-stay/lose-shift perseveration for stage 1 (pi_win, pi_lose).
    This captures the tendency to repeat an action based on whether the previous 
    trial resulted in a positive or negative outcome, decoupled from Q-learning.
    
    Bounds:
    learning_rate: [0, 1]
    beta_mb: [0, 10]
    beta_mf: [0, 10]
    beta2: [0, 10]
    pi_win: [-5, 5]
    pi_lose: [-5, 5]
    pi2: [-5, 5]
    lam: [0, 1]
    """
    learning_rate, beta_mb, beta_mf, beta2, pi_win, pi_lose, pi2, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action1 = -1
    last_reward = 0
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward > 0:
                pers1[last_action1] = pi_win
            else:
                pers1[last_action1] = pi_lose
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]
        
        if action_2[trial] != -1:
            a2 = action_2[trial]
            pers2 = np.zeros(2)
            if last_action2[state_idx] != -1:
                pers2[last_action2[state_idx]] = 1.0
                
            logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * pers2
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += learning_rate * lam * delta_stage2

            last_action2[state_idx] = a2
        else:
            p_choice_2[trial] = 1.0
            
        last_action1 = a1
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
