def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with separate stage 1 and stage 2 choice stickiness,
    and a decay parameter for unchosen actions to capture forgetting.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness (perseveration) for stage 1.
    pi2: [-5, 5] - Choice stickiness (perseveration) for stage 2.
    decay: [0, 1] - Forgetting rate for unchosen Q-values (shrinks towards 0).
    """
    alpha, beta, w, lam, pi1, pi2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # Policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with separate learning rates for stage 1 and stage 2,
    choice stickiness for stage 1, and state-specific choice stickiness (stimulus stickiness) for stage 2.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1.
    alpha2: [0, 1] - Learning rate for stage 2.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for MB vs MF.
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness for stage 1.
    pi2_stim: [-5, 5] - State-specific choice stickiness (stimulus stickiness) for stage 2.
    """
    alpha1, alpha2, beta, w, lam, pi1, pi2_stim = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_state = [-1, -1]

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # Policy for the second choice (stimulus stickiness based on current state)
        stick_vector_2 = np.zeros(2)
        if last_action_2_state[state_idx] != -1:
            stick_vector_2[last_action_2_state[state_idx]] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2_stim * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2_state[state_idx] = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with asymmetric learning rates for positive and negative prediction errors,
    separate choice stickiness for stage 1 and stage 2, and a decay parameter for unchosen Q-values.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness (perseveration) for stage 1.
    pi2: [-5, 5] - Choice stickiness (perseveration) for stage 2.
    decay: [0, 1] - Decay rate for unchosen Q-values.
    """
    alpha_pos, alpha_neg, beta, w, lam, pi1, pi2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # Policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1 + lr_stage1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2
        
        # Decay unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
