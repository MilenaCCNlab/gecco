### Model 1: Dynamic Transition Learning
While standard models assume participants know the fixed transition probabilities (0.7/0.3), this model assumes participants dynamically learn the transition matrix over time. It introduces `alpha_trans` to update the subjective transition probabilities after every trial.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model-based and model-free reinforcement learning model with decay, 
    center, and dynamic transition probability learning. The agent starts with a 
    uniform prior and updates its internal transition model based on experience.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for model-based vs model-free control.
    lam: [0, 1] - Eligibility trace parameter.
    decay: [0, 1] - Rate at which unchosen Q-values decay.
    center: [0, 1] - Baseline value towards which Q-values decay.
    alpha_trans: [0, 1] - Learning rate for the transition matrix.
    """
    alpha, beta, w, lam, decay, center, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Start with uniform expectations about transitions
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]]) 
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)

    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Softmax with max subtraction for numerical stability
        exp_q1 = np.exp(beta * q_stage1_net - np.max(beta * q_stage1_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 Choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Update transition matrix based on observed transition
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        # Decay for unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * center
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * center

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Counterfactual Updating
This model introduces `kappa`, which assumes participants infer information about the unchosen alien on the planet they visited. Assuming the payout probabilities of the two aliens might be inversely related (anti-correlated), getting a reward on one implies the other might not pay out, and vice versa.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with decay, center, and counterfactual updating at stage 2. 
    The agent updates the unchosen stage 2 action assuming its outcome is inversely 
    related to the chosen action's outcome.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    decay: [0, 1] - Q-value decay rate.
    center: [0, 1] - Decay center.
    kappa: [0, 1] - Counterfactual updating rate for unchosen stage 2 action.
    """
    alpha, beta, w, lam, decay, center, kappa = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net - np.max(beta * q_stage1_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * center
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * center
                    
        # Counterfactual update for unchosen action in the chosen state (applied after decay)
        unchosen_2 = 1 - action_2[trial]
        cf_reward = 1.0 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_2]
        q_stage2_mf[state_idx, unchosen_2] += kappa * alpha * delta_cf

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Temporal Discounting
This model adds a temporal discount factor (`gamma`). Instead of assuming the full value of the stage 2 state perfectly propagates back to stage 1, it allows the model to scale down the value of the future expected reward during both model-based calculations and model-free state updating.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with decay, center, and a temporal discount factor (gamma). 
    The discount factor scales the expected value of stage 2 states when propagating 
    back to stage 1, allowing for temporal discounting of future rewards.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    decay: [0, 1] - Q-value decay rate.
    center: [0, 1] - Decay center.
    gamma: [0, 1] - Temporal discount factor.
    """
    alpha, beta, w, lam, decay, center, gamma = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center, dtype=float)
    q_stage2_mf = np.full((2, 2), center, dtype=float)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        # Apply gamma to Model-Based calculation
        q_stage1_mb = transition_matrix @ (gamma * max_q_stage2)
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net - np.max(beta * q_stage1_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Apply gamma to Model-Free temporal difference error
        delta_stage1 = gamma * q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * center
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * center

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
