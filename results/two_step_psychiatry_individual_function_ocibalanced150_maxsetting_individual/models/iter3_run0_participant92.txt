def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Choice Trace (Momentum) Model:
    Extends the standard hybrid model by replacing the single-trial choice persistence 
    with a decaying choice trace. This captures a longer-term 'momentum' or habit 
    for Stage 1 choices, rather than just repeating the immediately preceding action.
    
    Bounds:
    alpha: [0,1] - Learning rate for both stages
    beta_1: [0,10] - Inverse temperature for Stage 1
    beta_2: [0,10] - Inverse temperature for Stage 2
    w: [0,1] - Model-based weight (0 = pure MF, 1 = pure MB)
    lam: [0,1] - Eligibility trace for Stage 1 update from Stage 2 reward
    pers: [-5,5] - Persistence weight for the choice trace
    trace_decay: [0,1] - Decay rate of the choice trace (0 = only last trial, 1 = perfectly integrates all past choices)
    """
    alpha, beta_1, beta_2, w, lam, pers, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + pers * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Update choice trace
        choice_trace *= trace_decay
        choice_trace[a1] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Counterfactual Updating Model:
    Assumes participants update the unchosen alien on the visited planet based on 
    an anti-correlated reward assumption (if chosen alien gives 1, unchosen gives 0, and vice versa).
    
    Bounds:
    alpha: [0,1] - Learning rate for both stages
    beta_1: [0,10] - Inverse temperature for Stage 1
    beta_2: [0,10] - Inverse temperature for Stage 2
    w: [0,1] - Model-based weight (0 = pure MF, 1 = pure MB)
    lam: [0,1] - Eligibility trace for Stage 1 update from Stage 2 reward
    pers: [-5,5] - Choice persistence bonus for repeating last action
    alpha_cf: [0,1] - Counterfactual learning rate for the unchosen alien on the same planet
    """
    alpha, beta_1, beta_2, w, lam, pers, alpha_cf = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + pers * pers_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Counterfactual update for unchosen alien on the same planet
        unchosen_a2 = 1 - a2
        r_cf = 1.0 - r
        delta_cf = r_cf - q_stage2_mf[s, unchosen_a2]
        q_stage2_mf[s, unchosen_a2] += alpha_cf * delta_cf
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Asymmetric Eligibility Trace Model:
    Eligibility trace parameter (lambda) differs depending on whether the 
    Stage 2 reward prediction error is positive or negative. This captures 
    differential propagation of positive vs negative surprises to Stage 1.
    
    Bounds:
    alpha: [0,1] - Learning rate for both stages
    beta_1: [0,10] - Inverse temperature for Stage 1
    beta_2: [0,10] - Inverse temperature for Stage 2
    w: [0,1] - Model-based weight (0 = pure MF, 1 = pure MB)
    lam_pos: [0,1] - Eligibility trace for positive reward prediction errors
    lam_neg: [0,1] - Eligibility trace for negative reward prediction errors
    pers: [-5,5] - Choice persistence bonus for repeating last action
    """
    alpha, beta_1, beta_2, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + pers * pers_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        # Use different lambda depending on sign of Stage 2 RPE
        lam = lam_pos if delta_stage2 >= 0 else lam_neg

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
