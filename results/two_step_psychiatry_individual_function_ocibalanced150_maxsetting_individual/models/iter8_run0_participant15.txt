def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice Trace Model
    Instead of a 1-trial stickiness, this model uses an exponentially decaying choice trace 
    to capture perseveration over multiple past trials for both stage 1 and stage 2 choices.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta_2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    alpha_choice: [0, 1] - Update rate for the choice traces.
    w_choice1: [-5, 5] - Weight of the choice trace on stage 1 action logits.
    w_choice2: [-5, 5] - Weight of the choice trace on stage 2 action logits.
    """
    learning_rate, beta_1, beta_2, w, lam, alpha_choice, w_choice1, w_choice2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    C1 = np.zeros(2)
    C2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_scaled = beta_1 * q_stage1_net + w_choice1 * C1
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx] + w_choice2 * C2[state_idx]
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # ACTION VALUE UPDATING
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # CHOICE TRACE UPDATING
        C1 = (1 - alpha_choice) * C1
        C1[action_1[trial]] += alpha_choice
        
        C2 = (1 - alpha_choice) * C2
        C2[state_idx, action_2[trial]] += alpha_choice

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dual Model-Based and Model-Free Learning Rates
    This model assumes that the model-based and model-free systems update their value 
    estimates of stage 2 states at different learning rates.
    
    Parameters:
    alpha_mf: [0, 1] - Learning rate for model-free Q-value updates.
    alpha_mb: [0, 1] - Learning rate for model-based Q-value updates.
    beta_1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta_2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    stick_1: [-5, 5] - Choice perseverance parameter for stage 1.
    stick_2: [-5, 5] - Choice perseverance parameter for stage 2 (dependent on state).
    """
    alpha_mf, alpha_mb, beta_1, beta_2, w, lam, stick_1, stick_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    q_stage2_mb = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2_mb = np.max(q_stage2_mb, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2_mb
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_term_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_term_1[last_action_1] = stick_1
            
        q1_scaled = beta_1 * q_stage1_net + stickiness_term_1
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = w * q_stage2_mb[state_idx] + (1 - w) * q_stage2_mf[state_idx]
        
        stickiness_term_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_term_2[last_action_2[state_idx]] = stick_2
            
        q2_scaled = beta_2 * q_stage2_net + stickiness_term_2
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # ACTION VALUE UPDATING
        delta_stage1_mf = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_mf * delta_stage1_mf
        
        delta_stage2_mf = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_mf * delta_stage2_mf
        
        q_stage1_mf[action_1[trial]] += alpha_mf * lam * delta_stage2_mf
        
        delta_stage2_mb = reward[trial] - q_stage2_mb[state_idx, action_2[trial]]
        q_stage2_mb[state_idx, action_2[trial]] += alpha_mb * delta_stage2_mb
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-Dependent Learning Rates
    This model assumes that participants update their Q-values differently 
    depending on whether the transition from stage 1 to stage 2 was common or rare.
    
    Parameters:
    alpha_com: [0, 1] - Learning rate for updates following a common transition.
    alpha_rare: [0, 1] - Learning rate for updates following a rare transition.
    beta_1: [0, 10] - Inverse temperature for softmax action selection at stage 1.
    beta_2: [0, 10] - Inverse temperature for softmax action selection at stage 2.
    w: [0, 1] - Weight on model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    stick_1: [-5, 5] - Choice perseverance parameter for stage 1.
    stick_2: [-5, 5] - Choice perseverance parameter for stage 2 (dependent on state).
    """
    alpha_com, alpha_rare, beta_1, beta_2, w, lam, stick_1, stick_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_term_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_term_1[last_action_1] = stick_1
            
        q1_scaled = beta_1 * q_stage1_net + stickiness_term_1
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_term_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_term_2[last_action_2[state_idx]] = stick_2

        q2_scaled = beta_2 * q_stage2_mf[state_idx] + stickiness_term_2
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # ACTION VALUE UPDATING
        is_common = (action_1[trial] == state_idx)
        lr = alpha_com if is_common else alpha_rare
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr * delta_stage2
        
        q_stage1_mf[action_1[trial]] += lr * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
