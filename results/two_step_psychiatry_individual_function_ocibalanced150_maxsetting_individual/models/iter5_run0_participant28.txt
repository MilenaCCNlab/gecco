def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF RL with Transition-Dependent Stickiness
    Stickiness to the previous Stage 1 action depends on whether the previous 
    trial's transition was common or rare. Rare transitions might disrupt 
    automatic perseveration, leading to different stickiness behavior.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho_common: [-5,5] - Choice stickiness if previous transition was common.
    rho_rare: [-5,5] - Choice stickiness if previous transition was rare.
    """
    learning_rate, beta, w, lam, rho_common, rho_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_trans_type = -1 # 0 for common, 1 for rare

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        if last_trans_type == 0:
            current_rho = rho_common
        elif last_trans_type == 1:
            current_rho = rho_rare
        else:
            current_rho = 0.0
            
        logits1 = beta * q_stage1_net + current_rho * stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Determine transition type for next trial
        # Action 0 -> State 0 is common, Action 0 -> State 1 is rare
        # Action 1 -> State 1 is common, Action 1 -> State 0 is rare
        if action_1[trial] == state_idx:
            last_trans_type = 0 # common
        else:
            last_trans_type = 1 # rare

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF RL with Counterfactual Learning at Stage 2
    Participants may infer that if one alien gave a reward, the other might not,
    updating the unchosen alien's value with (1 - reward) to learn faster.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for chosen actions.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Choice stickiness (perseveration) parameter.
    cf_learning_rate: [0,1] - Counterfactual learning rate for unchosen Stage 2 action.
    """
    learning_rate, beta, w, lam, rho, cf_learning_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits1 = beta * q_stage1_net + rho * stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Counterfactual update for unchosen action at stage 2
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1.0 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += cf_learning_rate * delta_cf

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF RL with Epsilon-Greedy Lapses
    Adds an epsilon parameter to capture random lapses in decision-making, 
    mixing a uniform distribution with the softmax probabilities.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Choice stickiness (perseveration) parameter.
    epsilon: [0,1] - Probability of making a random choice (lapse rate).
    """
    learning_rate, beta, w, lam, rho, epsilon = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits1 = beta * q_stage1_net + rho * stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # Mix with uniform distribution for random lapses
        probs_1 = (1 - epsilon) * probs_1 + epsilon * 0.5
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        # Mix with uniform distribution for random lapses
        probs_2 = (1 - epsilon) * probs_2 + epsilon * 0.5
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
