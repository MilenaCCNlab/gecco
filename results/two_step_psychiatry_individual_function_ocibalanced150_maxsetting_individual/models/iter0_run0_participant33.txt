def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free learning model (Daw et al., 2011).
    
    This model assumes participants use a mixture of model-based (MB) and model-free (MF) 
    strategies for the first-stage choice. It also includes an eligibility trace to update 
    first-stage actions based on second-stage rewards.

    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate for updating Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values (1 = pure MB, 0 = pure MF).
        lambda_: [0, 1] - Eligibility trace parameter for updating stage 1 values from stage 2 prediction errors.
    """
    alpha, beta, w, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Softmax with max-subtraction for numerical stability
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for choice 1 based on stage 2 prediction error
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Choice Perseverance (Stickiness).
    
    This model extends the hybrid MB/MF model by adding a perseverance parameter (pi) 
    that captures the participant's tendency to repeat the previous first-stage action 
    regardless of the reward history.

    Parameters:
    model_parameters: list or array containing:
        alpha: [0, 1] - Learning rate for updating Q-values.
        beta: [0, 10] - Inverse temperature for softmax action selection.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
        pi: [-5, 5] - Choice perseverance (stickiness) parameter for stage 1. Positive values indicate a tendency to repeat the last action.
    """
    alpha, beta, w, lambda_, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = None

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseverance bonus to the previously chosen action
        if last_action is not None:
            q_stage1_net[last_action] += pi
            
        exp_q1 = np.exp(beta * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Separate Learning Rates and Inverse Temperatures.
    
    This model allows for differential learning and decision noise between the two stages 
    of the task, which can account for varying degrees of uncertainty or attention across stages.

    Parameters:
    model_parameters: list or array containing:
        alpha1: [0, 1] - Learning rate for stage 1.
        alpha2: [0, 1] - Learning rate for stage 2.
        beta1: [0, 10] - Inverse temperature for stage 1.
        beta2: [0, 10] - Inverse temperature for stage 2.
        w: [0, 1] - Weighting of model-based vs model-free values.
        lambda_: [0, 1] - Eligibility trace parameter.
    """
    alpha1, alpha2, beta1, beta2, w, lambda_ = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta1 * (q_stage1_net - np.max(q_stage1_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update utilizing alpha1
        q_stage1_mf[action_1[trial]] += alpha1 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
