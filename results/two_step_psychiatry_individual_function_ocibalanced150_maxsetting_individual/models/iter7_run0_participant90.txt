import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for stage 1 and stage 2, decay for unchosen actions,
    separate inverse temperatures for both stages, a model-based weight, and stage 1 stickiness.
    
    Bounds:
    lr1: [0,1] - Learning rate for stage 1 model-free updates.
    lr2: [0,1] - Learning rate for stage 2 updates.
    decay_rate: [0,1] - Rate at which unchosen action values decay towards 0.5.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness: [-5,5] - Perseveration parameter for stage 1.
    """
    lr1, lr2, decay_rate, beta1, beta2, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness

        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a single learning rate, an eligibility trace (lambda_trace) for updating 
    stage 1 values based on stage 2 rewards, decay for unchosen actions, separate 
    inverse temperatures, model-based weight, and stage 1 stickiness.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updates.
    decay_rate: [0,1] - Rate at which unchosen action values decay towards 0.5.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness: [-5,5] - Perseveration parameter for stage 1.
    lambda_trace: [0,1] - Eligibility trace for updating stage 1 with stage 2 reward.
    """
    learning_rate, decay_rate, beta1, beta2, w, stickiness, lambda_trace = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness

        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update to stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_trace * delta_stage2

        # Decay unchosen actions
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based/Model-Free learning with asymmetric learning rates,
    separate inverse temperatures for each stage, and choice stickiness for both 
    stage 1 and stage 2 (persevering on the same alien at a specific planet).
    
    Bounds:
    lr_pos: [0,1] - Learning rate for positive prediction errors.
    lr_neg: [0,1] - Learning rate for negative prediction errors.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Weight on model-based vs model-free strategy (1 = pure MB).
    stickiness1: [-5,5] - Perseveration parameter for stage 1.
    stickiness2: [-5,5] - Perseveration parameter for stage 2.
    """
    lr_pos, lr_neg, beta1, beta2, w, stickiness1, stickiness2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if trial > 0:
            q_stage1_net[action_1[trial-1]] += stickiness1

        exp_q1 = np.exp(beta1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q_stage2_net[last_action_2[state_idx]] += stickiness2

        exp_q2 = np.exp(beta2 * q_stage2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
