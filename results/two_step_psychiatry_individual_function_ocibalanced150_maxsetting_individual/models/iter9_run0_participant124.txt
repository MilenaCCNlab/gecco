Here are 3 new cognitive models that explore alternative mechanisms for learning and decision-making in the two-step task, keeping the parameter count to 8 to maintain parsimony.

### Model 1: Asymmetric Stage 2 Learning
This model hypothesizes that participants learn differently from positive and negative feedback at stage 2. It splits the stage 2 learning rate into `alpha_2_pos` (for positive reward prediction errors) and `alpha_2_neg` (for negative reward prediction errors). It omits the eligibility trace (`lam`) to accommodate the extra learning rate while keeping the parameter count at 8.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for stage 1, separate positive and negative 
    learning rates for stage 2, a shared decay rate for unchosen Q-values, 
    separate model-based and model-free inverse temperatures, and stage 1 perseveration.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2_pos: learning rate for positive reward prediction errors at stage 2 [0, 1]
    alpha_2_neg: learning rate for negative reward prediction errors at stage 2 [0, 1]
    decay: decay rate for unchosen Q-values at both stages [0, 1]
    beta_mb: inverse temperature for model-based Q-values at stage 1 [0, 10]
    beta_mf: inverse temperature for model-free Q-values at stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    p_1: choice perseveration for stage 1 [0, 10]
    """
    alpha_1, alpha_2_pos, alpha_2_neg, decay, beta_mb, beta_mf, beta_2, p_1 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2_pos * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2_neg * delta_stage2
            
        last_action_1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Perseveration
This model hypothesizes that participants exhibit perseveration not only in their spaceship choice (stage 1) but also in their alien choice (stage 2), nested by planet. It tracks the last alien chosen on each specific planet and adds a perseveration bonus `p_2` to that alien.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for stage 1 and stage 2, a shared decay rate 
    for unchosen Q-values, separate model-based and model-free inverse temperatures, 
    and choice perseveration for both stage 1 and state-specific stage 2.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2: learning rate for stage 2 [0, 1]
    decay: decay rate for unchosen Q-values at both stages [0, 1]
    beta_mb: inverse temperature for model-based Q-values at stage 1 [0, 10]
    beta_mf: inverse temperature for model-free Q-values at stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    p_1: choice perseveration for stage 1 [0, 10]
    p_2: choice perseveration for stage 2 [0, 10]
    """
    alpha_1, alpha_2, decay, beta_mb, beta_mf, beta_2, p_1, p_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # Track last action for each planet independently
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            rep_2[last_action_2[state_idx]] = 1

        logits_2 = beta_2 * q_stage2_mf[state_idx] + p_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition Matrix Learning
This model hypothesizes that the participant does not treat the transition matrix as strictly fixed. Instead, they dynamically update their internal transition probabilities based on the actual spaceship-to-planet transitions experienced during the task.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for stage 1 and stage 2, a shared decay rate 
    for unchosen Q-values, separate model-based and model-free inverse temperatures, 
    stage 1 perseveration, and dynamic learning of the transition matrix.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2: learning rate for stage 2 [0, 1]
    decay: decay rate for unchosen Q-values at both stages [0, 1]
    beta_mb: inverse temperature for model-based Q-values at stage 1 [0, 10]
    beta_mf: inverse temperature for model-free Q-values at stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    p_1: choice perseveration for stage 1 [0, 10]
    alpha_trans: learning rate for transition matrix [0, 1]
    """
    alpha_1, alpha_2, decay, beta_mb, beta_mf, beta_2, p_1, alpha_trans = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Update transition matrix probabilities based on observed transition
        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += alpha_trans * (1 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += alpha_trans * (0 - transition_matrix[a1, 1 - s2])

        last_action_1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
