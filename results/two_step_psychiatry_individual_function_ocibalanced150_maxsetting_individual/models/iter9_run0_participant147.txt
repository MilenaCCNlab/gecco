Here are 3 new cognitive models that explore alternative parameter configurations and behavioral mechanisms, avoiding previously tried combinations:

### Model 1: Separate Temperatures & Transition Matrix Learning
This model features separate inverse temperatures (`beta_1` and `beta_2`) for stage 1 and stage 2, allowing for different levels of stochasticity/exploration in the two stages. It also introduces `alpha_trans` to dynamically learn the transition matrix from observed state transitions, rather than assuming it is fixed at 0.7/0.3.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate temperatures for stage 1 and 2, and dynamic learning of the transition matrix.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 action selection.
    beta_2: [0, 10] - Inverse temperature for stage 2 action selection.
    w: [0, 1] - Weighting of model-based versus model-free values (1 = MB, 0 = MF).
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    lam: [0, 1] - Eligibility trace parameter.
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix probabilities.
    """
    learning_rate, beta_1, beta_2, w, p, lam, alpha_trans = model_parameters
    n_trials = len(action_1)
    
    # Initialize uniform transition matrix to be learned
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p * pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Update transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        # Value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates & Outcome-Dependent Perseveration
This model evaluates if participants learn differently from positive and negative reward prediction errors (`alpha_pos` vs `alpha_neg`) and whether their tendency to repeat a stage 1 choice depends on whether they were rewarded on the previous trial (`p_rew` vs `p_unrew`). It also retains unchosen value decay.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates for positive/negative RPEs, decay for unchosen actions, 
    and outcome-dependent perseveration (p_rew and p_unrew).
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive RPE.
    alpha_neg: [0, 1] - Learning rate for negative RPE.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based versus model-free values (1 = MB, 0 = MF).
    p_rew: [-5, 5] - Perseveration parameter after a rewarded trial.
    p_unrew: [-5, 5] - Perseveration parameter after an unrewarded trial.
    lam: [0, 1] - Eligibility trace parameter.
    """
    alpha_pos, alpha_neg, decay, beta, w, p_rew, p_unrew, lam = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = -1
    
    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers[last_action_1] = p_rew
            elif last_reward == 0:
                pers[last_action_1] = p_unrew
            
        logits_1 = beta * q_stage1_net + pers
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2
        
        # Decay unchosen/unvisited states
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])

        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])

        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Two-Stage Choice Perseveration & Transition Learning
This model considers that the participant might show "stickiness" or perseveration in both Stage 1 and Stage 2 choices (`p_stage1` and `p_stage2`). Furthermore, it leverages the combination of transition matrix learning and unchosen Q-value decay towards 0.5.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model incorporating decay, transition matrix learning, and choice perseveration for both stage 1 and stage 2.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting of model-based versus model-free values (1 = MB, 0 = MF).
    p_stage1: [-5, 5] - Perseveration parameter for Stage 1 choice.
    p_stage2: [-5, 5] - Perseveration parameter for Stage 2 choice.
    lam: [0, 1] - Eligibility trace parameter.
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix probabilities.
    """
    learning_rate, decay, beta, w, p_stage1, p_stage2, lam, alpha_trans = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) # Stores last action taken for both stage 2 states
    
    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Stage 1 policy
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = 1
            
        logits_1 = beta * q_stage1_net + p_stage1 * pers_1
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Stage 2 policy
        pers_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_2[last_action_2[state_idx]] = 1
            
        logits_2 = beta * q_stage2_mf[state_idx] + p_stage2 * pers_2
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Update transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        # Value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Decay unchosen/unvisited states
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])

        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])

        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
