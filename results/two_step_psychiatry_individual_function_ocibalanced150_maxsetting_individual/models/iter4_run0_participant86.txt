Here are 3 new cognitive models, building on the mechanisms from the best model so far while introducing novel parameter configurations to capture distinct psychological processes (such as asymmetric learning, reward-dependent perseveration, and reward sensitivity).

### Model 1: Asymmetric Learning Rates + Stage 1 & 2 Stickiness
This model separates learning into two distinct rates depending on whether the prediction error is positive (better than expected) or negative (worse than expected). This accounts for optimism or pessimism biases in learning, combined with choice stickiness at both stages.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric Learning Rates + Stage 1 & 2 Stickiness
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    beta: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace
    stick1: [-5,5] - Choice stickiness for stage 1
    stick2: [-5,5] - Choice stickiness for stage 2
    """
    alpha_pos, alpha_neg, beta, beta2, w, lam, stick1, stick2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stick1
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q2_net[last_action_2[state_idx]] += stick2
            
        q2_net = q2_net - np.max(q2_net)
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg

        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unchosen Decay + Reward-Dependent Stickiness
Instead of a fixed stickiness parameter, perseveration on the first stage depends on whether the previous trial resulted in a reward. This models win-stay/lose-shift heuristics interacting with model-based and model-free reinforcement learning, combined with memory decay for unchosen options.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Unchosen Decay + Reward-Dependent Stickiness
    
    Bounds:
    learning_rate: [0,1] - Learning rate
    decay: [0,1] - Decay rate for unchosen actions
    beta: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace
    stick_rew: [-5,5] - Choice stickiness for stage 1 if previously rewarded
    stick_unrew: [-5,5] - Choice stickiness for stage 1 if previously unrewarded
    """
    learning_rate, decay, beta, beta2, w, lam, stick_rew, stick_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            if last_reward > 0:
                q_stage1_net[last_action_1] += stick_rew
            else:
                q_stage1_net[last_action_1] += stick_unrew
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_net = np.copy(q_stage2_mf[state_idx])
        q2_net = q2_net - np.max(q2_net)
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward Sensitivity + Stage 1 & 2 Stickiness
This model includes a reward sensitivity parameter (`rho`) that scales the subjective value of the reward. Scaling the Q-values interacts directly with the fixed `stick1` and `stick2` parameters, allowing the model to disentangle the sheer valuation of coins from the individual's tendency to perseverate on actions.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward Sensitivity + Stage 1 & 2 Stickiness
    
    Bounds:
    learning_rate: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature for stage 1
    beta2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Weighting of MB vs MF control
    lam: [0,1] - Eligibility trace
    stick1: [-5,5] - Choice stickiness for stage 1
    stick2: [-5,5] - Choice stickiness for stage 2
    rho: [0,10] - Subjective reward sensitivity multiplier
    """
    learning_rate, beta, beta2, w, lam, stick1, stick2, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stick1
            
        q_stage1_net = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q2_net[last_action_2[state_idx]] += stick2
            
        q2_net = q2_net - np.max(q2_net)
        exp_q2 = np.exp(beta2 * q2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # subjective scale of reward
        scaled_reward = reward[trial] * rho

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = scaled_reward - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
