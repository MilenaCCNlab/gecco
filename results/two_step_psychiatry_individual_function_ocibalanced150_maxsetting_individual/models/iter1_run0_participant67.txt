def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Forgetting (Decay).
    Unchosen Q-values decay towards 0 at each trial, capturing forgetting of unchosen options.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi: [-5, 5] (Choice perseveration)
    decay: [0, 1] (Forgetting rate for unchosen actions)
    """
    alpha, beta1, beta2, w, lam, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1
            
        logits_1 = beta1 * q_net + pi * persev
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 Choice
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay for unchosen actions
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Stage 1 and Stage 2 Perseveration.
    Includes separate perseveration parameters for first-stage and second-stage choices.
    Stage 2 perseveration is state-specific.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi1: [-5, 5] (Stage 1 choice perseveration)
    pi2: [-5, 5] (Stage 2 choice perseveration)
    """
    alpha, beta1, beta2, w, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1
    last_state = -1

    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev1 = np.zeros(2)
        if last_action1 != -1:
            persev1[last_action1] = 1
            
        logits_1 = beta1 * q_net + pi1 * persev1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 Choice
        state_idx = state[trial]
        persev2 = np.zeros(2)
        if last_state == state_idx and last_action2 != -1:
            persev2[last_action2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * persev2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Outcome-Dependent Perseveration.
    Perseveration at stage 1 depends on whether the previous trial was rewarded (win-stay) or not (lose-shift).
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi_win: [-5, 5] (Perseveration if previous trial was rewarded)
    pi_lose: [-5, 5] (Perseveration if previous trial was unrewarded)
    """
    alpha, beta1, beta2, w, lam, pi_win, pi_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            if last_reward > 0:
                persev[last_action] = pi_win
            else:
                persev[last_action] = pi_lose
            
        logits_1 = beta1 * q_net + persev
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 Choice
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action Value Updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
