import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate learning rates for Stage 1 and Stage 2,
    and outcome-dependent perseverance (win-stay/lose-shift) at Stage 1.
    
    Bounds:
    alpha1: [0,1] - learning rate for stage 1 Q-value updates
    alpha2: [0,1] - learning rate for stage 2 Q-value updates
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers_win: [-5,5] - choice perseverance for stage 1 if previous trial was rewarded
    pers_lose: [-5,5] - choice perseverance for stage 1 if previous trial was unrewarded
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent1 = beta1 * net_q_stage1
        if trial > 0:
            if reward[trial-1] > 0:
                exponent1[action_1[trial-1]] += pers_win
            else:
                exponent1[action_1[trial-1]] += pers_lose
                
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        exponent2 = beta2 * q_stage2_mf[state_idx]
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with asymmetric learning rates for positive and negative prediction errors,
    and outcome-dependent perseverance (win-stay/lose-shift) at Stage 1.
    
    Bounds:
    alpha_pos: [0,1] - learning rate for positive prediction errors
    alpha_neg: [0,1] - learning rate for negative prediction errors
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight for model-based vs model-free control
    lam: [0,1] - eligibility trace parameter
    pers_win: [-5,5] - choice perseverance for stage 1 if previous trial was rewarded
    pers_lose: [-5,5] - choice perseverance for stage 1 if previous trial was unrewarded
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        net_q_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exponent1 = beta1 * net_q_stage1
        if trial > 0:
            if reward[trial-1] > 0:
                exponent1[action_1[trial-1]] += pers_win
            else:
                exponent1[action_1[trial-1]] += pers_lose
                
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        exponent2 = beta2 * q_stage2_mf[state_idx]
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1_curr = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1_curr * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2_curr = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_curr * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2_curr * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model using separate inverse temperatures for model-based (beta_mb) 
    and model-free (beta_mf) values at stage 1, instead of a mixing weight w.
    Includes outcome-dependent perseverance at stage 1 and standard at stage 2.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers_win: [-5,5] - choice perseverance for stage 1 if previous trial was rewarded
    pers_lose: [-5,5] - choice perseverance for stage 1 if previous trial was unrewarded
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha, beta_mb, beta_mf, beta2, lam, pers_win, pers_lose, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_stage2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            if reward[trial-1] > 0:
                exponent1[action_1[trial-1]] += pers_win
            else:
                exponent1[action_1[trial-1]] += pers_lose
                
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
