import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for stage 1 and stage 2, with decay for unchosen options.
    This model allows for different learning speeds at the two stages of the task, while 
    incorporating a decay mechanism for unchosen options to capture forgetting.
    
    Bounds:
    alpha_1: [0,1] (Learning rate for stage 1)
    alpha_2: [0,1] (Learning rate for stage 2)
    decay: [0,1] (Decay rate for unchosen options)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    lam: [0,1] (Eligibility trace)
    pi: [-5,5] (Choice stickiness for stage 1)
    """
    alpha_1, alpha_2, decay, beta_1, beta_2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if trial > 0:
            rep[action_1[trial-1]] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1 + alpha_1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for common and rare transitions, with decay.
    This model assumes that participants might update their values differently depending on 
    whether the transition they just experienced was expected (common) or unexpected (rare).
    
    Bounds:
    alpha_c: [0,1] (Learning rate for common transitions)
    alpha_u: [0,1] (Learning rate for rare transitions)
    decay: [0,1] (Decay rate for unchosen options)
    beta_1: [0,10] (Inverse temperature for stage 1)
    beta_2: [0,10] (Inverse temperature for stage 2)
    w: [0,1] (Model-based weight)
    lam: [0,1] (Eligibility trace)
    pi: [-5,5] (Choice stickiness for stage 1)
    """
    alpha_c, alpha_u, decay, beta_1, beta_2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if trial > 0:
            rep[action_1[trial-1]] = 1
            
        logits_1 = beta_1 * q_net + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        is_common = (action_1[trial] == state_idx)
        alpha = alpha_c if is_common else alpha_u
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Direct model-free update with asymmetric learning rates, decay, and separate beta parameters for MB and MF.
    This model utilizes a direct Monte Carlo update for stage 1 (based purely on final reward), 
    and replaces the mixing weight 'w' with separate inverse temperatures for model-based and model-free Q-values.
    
    Bounds:
    alpha_pos: [0,1] (Learning rate for positive PE)
    alpha_neg: [0,1] (Learning rate for negative PE)
    decay: [0,1] (Decay rate for unchosen options)
    beta_mb: [0,10] (Inverse temperature for model-based)
    beta_mf: [0,10] (Inverse temperature for model-free)
    beta_2: [0,10] (Inverse temperature for stage 2)
    pi: [-5,5] (Choice stickiness for stage 1)
    """
    alpha_pos, alpha_neg, decay, beta_mb, beta_mf, beta_2, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        rep = np.zeros(2)
        if trial > 0:
            rep[action_1[trial-1]] = 1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Direct MC update for stage 1 MF
        pe_1 = reward[trial] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if pe_1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * pe_1

        # Standard TD update for stage 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] *= (1 - decay)
        q_stage2_mf[unvisited_state, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
