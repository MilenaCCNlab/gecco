import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative RPEs,
    a single inverse temperature, a choice kernel for both stages, and Q-value decay.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    alpha_c: [0, 1]
    beta_c: [0, 5]
    decay: [0, 1]
    """
    alpha_pos, alpha_neg, beta, w, lam, alpha_c, beta_c, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck_1 = np.zeros(2)
    ck_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net + beta_c * ck_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + beta_c * ck_2[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        ck_1 *= (1 - alpha_c)
        ck_1[action_1[trial]] += alpha_c
        
        ck_2 *= (1 - alpha_c)
        ck_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):

    Bounds:
    alpha_pos: [0, 1]

    q_stage2_mf = np.zeros((2, 2))

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates and inverse temperatures for stage 1 and 2,
    and simple 1-trial win-stay/lose-shift perseveration parameters for stage 1 choices
    instead of a continuous choice kernel.
    
    Bounds:
    alpha_1: [0, 1]
    alpha_2: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p_win: [-5, 5]
    p_lose: [-5, 5]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers_bonus[last_action_1] = p_win
            else:
                pers_bonus[last_action_1] = p_lose
                
        exp_q1 = np.exp(beta_1 * q_net + pers_bonus)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
