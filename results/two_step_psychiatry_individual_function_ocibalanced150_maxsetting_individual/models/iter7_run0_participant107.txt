def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates and optimistic initialization.
    Initial Q-values are set to q0, which can capture initial exploration or prolonged 
    stickiness to an option due to optimistic expectations before it is fully unlearned.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Choice stickiness.
    q0: [0, 1] - Initial Q-value for all states and actions.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pi, q0 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values with q0
    q_stage1_mf = np.full(2, float(q0))
    q_stage2_mf = np.full((2, 2), float(q0))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits1 = beta1 * q_net_stage1 + pi * stickiness
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[a1] += alpha2 * lam * delta_stage2
        
        last_action_1 = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates and forgetting of unchosen options.
    Unchosen Q-values decay towards 0.5 over time, which models the forgetting of 
    reward probabilities for options that are not frequently sampled.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Choice stickiness.
    phi: [0, 1] - Forgetting rate for unchosen actions (decays to 0.5).
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pi, phi = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits1 = beta1 * q_net_stage1 + pi * stickiness
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha2 * delta_stage2

        # eligibility trace
        q_stage1_mf[a1] += alpha2 * lam * delta_stage2
        
        # decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += phi * (0.5 - q_stage1_mf[unchosen_a1])
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] += phi * (0.5 - q_stage2_mf[state_idx, action_idx])
        
        last_action_1 = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates and a continuous choice trace.
    Instead of just sticking to the immediate last action, the participant maintains 
    a choice trace that accumulates and decays over time, capturing long-term perseveration.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Weight of the choice trace on stage 1 choices.
    tau: [0, 1] - Decay rate of the choice trace (0 = fast decay, 1 = no decay).
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pi, tau = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_stage1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits1 = beta1 * q_net_stage1 + pi * choice_trace
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[s]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += alpha2 * delta_stage2

        # eligibility trace
        q_stage1_mf[a1] += alpha2 * lam * delta_stage2
        
        # update choice trace
        choice_trace *= tau
        choice_trace[a1] += 1.0
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
