def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF mixture model with separate learning rates and inverse temperatures 
    for stage 1 and stage 2, along with eligibility trace and stage 1 stickiness.
    This model posits that learning and exploration may happen at different rates 
    for the first and second stages of the task.
    
    Bounds:
    lr_1: [0,1]
    lr_2: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness: [-5,5]
    """
    lr_1, lr_2, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action != -1:
            q_stage1_net[last_action] += stickiness
            
        exp_q1 = np.exp(beta_1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr_1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF mixture model with asymmetric learning rates for positive and negative RPEs,
    separate inverse temperatures for stage 1 and 2, eligibility trace, and stage 1 stickiness.
    This model assumes that participants might update their values differently depending 
    on whether the outcome was better or worse than expected.
    
    Bounds:
    lr_pos: [0,1]
    lr_neg: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness: [-5,5]

    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF mixture model with separate inverse temperatures for stage 1 and stage 2,
    stage 1 stickiness, stage 2 stickiness (perseveration on second stage choice), and eligibility trace.
    This model captures motor/choice perseveration not just at the primary decision stage, 
    but also on the specific planets (stage 2) independent of reward.
    
    Bounds:
    learning_rate: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    stickiness_1: [-5,5]
    stickiness_2: [-5,5]
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) # Track last action for each state

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += stickiness_1
            
        exp_q1 = np.exp(beta_1 * q_stage1_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q_stage2_net = np.copy(q_stage2_mf[state_idx])
        if last_action_2[state_idx] != -1:
            q_stage2_net[last_action_2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta_2 * q_stage2_net)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
