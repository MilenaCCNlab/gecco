def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with forgetting (decay) and perseveration at both stages.
    This model captures the tendency to repeat the same first-stage action and 
    the same second-stage action (conditional on the state).
    
    Bounds:
    alpha: [0, 1] - Learning rate
    decay: [0, 1] - Forgetting rate for all actions (values revert toward 0.5)
    beta1: [0, 10] - Inverse temperature for stage 1
    beta2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace
    pers1: [-5, 5] - Perseveration parameter for stage 1
    pers2: [-5, 5] - Perseveration parameter for stage 2
    """
    alpha, decay, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep1 = np.zeros(2)
        if last_action1 != -1:
            rep1[last_action1] = 1.0
            
        logits1 = beta1 * q_net + pers1 * rep1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        rep2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            rep2[last_action2[state_idx]] = 1.0
            
        logits2 = beta2 * q_stage2_mf[state_idx] + pers2 * rep2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with forgetting (decay) and asymmetric learning rates for positive 
    and negative prediction errors. This tests whether participants update values 
    differently when receiving a reward vs when a reward is omitted.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    decay: [0, 1] - Forgetting rate for all actions (values revert toward 0.5)
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers: [-5, 5] - Perseveration parameter
    """
    alpha_pos, alpha_neg, decay, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        logits1 = beta1 * q_net + pers * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg

        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with forgetting (decay) and reward-dependent perseveration.
    Perseveration differs based on whether the previous trial was rewarded or unrewarded, 
    accounting for distinct win-stay vs lose-shift heuristics.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-value updating
    decay: [0, 1] - Forgetting rate for unchosen actions (values revert toward 0.5)
    beta1: [0, 10] - Inverse temperature for first-stage choices
    beta2: [0, 10] - Inverse temperature for second-stage choices
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers_win: [-5, 5] - Perseveration parameter after a rewarded trial
    pers_lose: [-5, 5] - Perseveration parameter after an unrewarded trial
    """
    alpha, decay, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        
        pers = 0.0
        if last_action != -1:
            rep[last_action] = 1.0
            if last_reward == 1:
                pers = pers_win
            elif last_reward == 0:
                pers = pers_lose
            
        logits1 = beta1 * q_net + pers * rep
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
