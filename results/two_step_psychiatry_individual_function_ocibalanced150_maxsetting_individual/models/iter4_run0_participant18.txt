Here are three new cognitive models that build upon the best performing mechanisms (such as Q-value decay) by introducing novel parameter combinations and cognitive hypotheses to better explain the participant's choices.

### Model 1: Stage 1 & Stage 2 Perseveration with Q-value Decay
This model expands on the base model by introducing a separate perseveration (stickiness) parameter for Stage 2. Participants often repeat choices not just at the first stage (spaceship), but also at the second stage (aliens). The model tracks the last alien chosen at each specific planet and applies a stickiness bonus.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Stage 1 & Stage 2 Perseveration and Q-value Decay.
    Includes stickiness parameters for both stages to capture repeating choices,
    along with forgetting (decay) of all Q-values.
    
    Bounds:
    alpha: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weighting of model-based vs model-free control.
    pi1: [-5, 5] - Perseveration bonus for repeating the previous stage 1 choice.
    pi2: [-5, 5] - Perseveration bonus for repeating the previous stage 2 choice (state-specific).
    lambda_: [0, 1] - Eligibility trace for stage 1 model-free update.
    decay: [0, 1] - Rate at which Q-values decay towards 0 on each trial.
    """
    alpha, beta, w, pi1, pi2, lambda_, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1
            
        logits_1 = beta * q_net + pi1 * pers1
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 choice
        state_idx = state[trial]
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = 1

        logits_2 = beta * q_stage2_mf[state_idx] + pi2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay all Q-values towards 0 prior to update
        q_stage1_mf = q_stage1_mf * (1 - decay)
        q_stage2_mf = q_stage2_mf * (1 - decay)

        # Q-learning updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates with Q-value Decay
Humans frequently update their beliefs differently depending on whether outcomes are better or worse than expected. This model combines learning rate asymmetry (`alpha_pos` for positive prediction errors and `alpha_neg` for negative prediction errors) with the successful Q-value decay mechanism.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Learning Rates and Q-value Decay.
    Learning rates differ for positive and negative prediction errors.
    All Q-values decay towards 0 prior to the learning update.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weighting of model-based vs model-free control.
    pi: [-5, 5] - Perseveration bonus for repeating previous stage 1 choice.
    lambda_: [0, 1] - Eligibility trace.
    decay: [0, 1] - Forgetting rate (decay towards 0).
    """
    alpha_pos, alpha_neg, beta, w, pi, lambda_, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta * q_net + pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay
        q_stage1_mf = q_stage1_mf * (1 - decay)
        q_stage2_mf = q_stage2_mf * (1 - decay)

        # Stage 1 Update
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        # Stage 2 Update
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update using Stage 2's prediction error
        q_stage1_mf[action_1[trial]] += alpha_2 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Perseveration with Q-value Decay
This model modifies the perseveration mechanism by distinguishing between repeating an action following a reward (win-stay) versus following no reward (lose-shift). This is combined with the successful Q-value decay parameter to provide a comprehensive view of the participant's choice inertia.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Reward-Dependent Perseveration and Q-value Decay.
    The perseveration parameter is split based on whether the previous trial 
    was rewarded or unrewarded (capturing Win-Stay / Lose-Shift tendencies).
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature.
    w: [0, 1] - Weighting of model-based vs model-free control.
    pi_rew: [-5, 5] - Perseveration bonus if previous trial was rewarded.
    pi_unrew: [-5, 5] - Perseveration bonus if previous trial was unrewarded.
    lambda_: [0, 1] - Eligibility trace.
    decay: [0, 1] - Forgetting rate (decay towards 0).
    """
    alpha, beta, w, pi_rew, pi_unrew, lambda_, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        # Select perseveration parameter based on the last trial's reward
        pi = 0
        if last_reward == 1:
            pi = pi_rew
        elif last_reward == 0:
            pi = pi_unrew

        logits_1 = beta * q_net + pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay
        q_stage1_mf = q_stage1_mf * (1 - decay)
        q_stage2_mf = q_stage2_mf * (1 - decay)

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
