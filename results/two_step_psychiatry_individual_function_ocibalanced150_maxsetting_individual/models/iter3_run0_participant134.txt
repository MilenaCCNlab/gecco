Here are 3 new cognitive models that explore alternative mechanisms for how participants might make decisions and learn in this two-step task. Each model implements a unique combination of learning features, such as lapse rates (epsilon-greedy), asymmetric learning for positive vs. negative prediction errors, and separate inverse temperatures for Model-Based and Model-Free systems.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learner with choice stickiness and a lapse rate (epsilon).
    Epsilon accounts for random choices (e.g., due to lapses in attention) independent of the model values.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    stickiness: [-5, 5] - Choice stickiness for stage 1
    epsilon: [0, 1] - Lapse rate (probability of making a random uniform choice)
    """
    alpha, beta_1, beta_2, w, lam, stickiness, epsilon = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits_1 = beta_1 * q_net + stickiness * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # Apply lapse rate for random exploration / inattention
        probs_1 = (1 - epsilon) * probs_1 + epsilon * 0.5
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        # Apply lapse rate for random exploration / inattention
        probs_2 = (1 - epsilon) * probs_2 + epsilon * 0.5
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learner with asymmetric learning rates for Stage 2.
    Separates learning from positive vs negative reward prediction errors to capture valence biases.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1
    alpha_2_pos: [0, 1] - Learning rate for positive prediction errors in stage 2
    alpha_2_neg: [0, 1] - Learning rate for negative prediction errors in stage 2
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0, 1] - Eligibility trace parameter
    stickiness: [-5, 5] - Choice stickiness for stage 1
    """
    alpha_1, alpha_2_pos, alpha_2_neg, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1
            
        logits_1 = beta_1 * q_net + stickiness * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Asymmetric update for stage 2 based on valence of prediction error
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2_pos * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2_neg * delta_stage2
            
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate inverse temperatures for Model-Based and Model-Free systems,
    and choice stickiness applied to both Stage 1 and Stage 2 decisions.
    
    Parameters:
    alpha: [0, 1] - Learning rate for all Q-values
    beta_mb: [0, 10] - Inverse temperature for Model-Based stage 1
    beta_mf: [0, 10] - Inverse temperature for Model-Free stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    lam: [0, 1] - Eligibility trace parameter
    stickiness_1: [-5, 5] - Choice stickiness for stage 1
    stickiness_2: [-5, 5] - Choice stickiness for stage 2
    """
    alpha, beta_mb, beta_mf, beta_2, lam, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        # Using separate inverse temperatures for MB and MF instead of a mixing weight (w)
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + stickiness_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep_2 = np.zeros(2)
        if last_action_2 != -1:
            rep_2[last_action_2] = 1  # Button-level perseveration for stage 2
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + stickiness_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
