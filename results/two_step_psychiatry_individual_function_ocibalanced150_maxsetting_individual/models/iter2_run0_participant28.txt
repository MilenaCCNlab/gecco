Here are 3 new cognitive models that explore alternative mechanisms to explain the participant's behavior.

### Model 1: Hybrid MB/MF RL with Separate Learning Rates
This model explores the possibility that participants update their expectations at different speeds for Stage 1 (spaceships) and Stage 2 (aliens). This is a common extension in two-step tasks, as the first stage involves an abstract transition matrix while the second stage involves direct reward probabilities.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF RL with Separate Learning Rates for Stage 1 and Stage 2.
    
    Parameters:
    alpha_1: [0,1] - Learning rate for Stage 1 Q-value updates.
    alpha_2: [0,1] - Learning rate for Stage 2 Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Choice stickiness (perseveration) parameter for Stage 1.
    """
    alpha_1, alpha_2, beta, w, lam, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits1 = beta * q_stage1_net + rho * stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Hybrid MB/MF RL with Forgetting (Decay)
This model introduces a decay parameter (`decay`) that pulls the values of unchosen options back towards a neutral baseline (0.5) over time. This helps capture behavior where an option ignored for a long time starts to seem appealing again, or its previously learned negative value is "forgotten," promoting sudden exploration.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF RL with Forgetting (Decay) for Unchosen Options.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    decay: [0,1] - Decay rate for unchosen Q-values towards neutral (0.5).
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho: [-5,5] - Choice stickiness (perseveration) parameter for Stage 1.
    """
    learning_rate, decay, beta, w, lam, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 to align with the decay baseline
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness = np.zeros(2)
        if last_a1 != -1:
            stickiness[last_a1] = 1.0
            
        logits1 = beta * q_stage1_net + rho * stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Decay unchosen options
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] = (1 - decay) * q_stage1_mf[a] + decay * 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * 0.5

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Hybrid MB/MF RL with Outcome-Dependent Perseveration (Win-Stay/Lose-Shift)
This model splits the stickiness parameter into two: one for when the participant received a reward on the previous trial (`rho_win`), and one for when they did not (`rho_lose`). This captures the classic "win-stay, lose-shift" heuristic, which is highly relevant given the participant's tendency to switch aliens or spaceships immediately following a 0 coin outcome.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF RL with Outcome-Dependent Perseveration (Win-Stay/Lose-Shift).
    
    Parameters:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting parameter for MB (1) vs MF (0) control.
    lam: [0,1] - Eligibility trace decay parameter.
    rho_win: [-5,5] - Choice stickiness for Stage 1 if the previous trial was rewarded.
    rho_lose: [-5,5] - Choice stickiness for Stage 1 if the previous trial was not rewarded.
    """
    learning_rate, beta, w, lam, rho_win, rho_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        # Determine stickiness based on previous trial's outcome
        stickiness = np.zeros(2)
        if last_a1 != -1:
            if last_reward > 0:
                stickiness[last_a1] = rho_win
            else:
                stickiness[last_a1] = rho_lose
            
        logits1 = beta * q_stage1_net + stickiness
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        last_a1 = action_1[trial]
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
