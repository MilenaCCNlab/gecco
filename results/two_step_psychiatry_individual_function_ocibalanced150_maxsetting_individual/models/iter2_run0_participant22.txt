Here are three novel cognitive models that build upon the baseline Hybrid Agent but introduce new mechanisms to better capture participant behavior in a two-step task.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
It is often observed that participants update their expectations differently depending on the stage of the task. This model introduces two separate learning rates: `alpha_1` for updating Stage 1 model-free values, and `alpha_2` for updating Stage 2 values.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Separate Learning Rates for Stage 1 and Stage 2.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for Stage 1 updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice (positive values indicate a tendency to repeat the last choice).
    """
    import numpy as np
    alpha_1, alpha_2, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
            
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update using Stage 1 learning rate
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Forgetting/Decay for Unchosen Options
Because the probabilities of the aliens returning coins slowly change over trials, participants might assume that unchosen options drift back to a baseline probability. This model introduces a `gamma` parameter that decays the Q-values of unchosen Stage 2 state-action pairs toward 0.5 on each trial. Initial Q-values are set to 0.5 to match this expectation.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Forgetting/Decay for unchosen options.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    gamma: [0, 1] - Decay parameter pulling unchosen Stage 2 state-action pairs toward 0.5.
    """
    import numpy as np
    alpha, beta, w, lam, p, gamma = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values to 0.5 to match the decay baseline
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_choice = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
            
        q_stage1_combined = q_stage1_net + p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        exp_q1 = np.exp(beta * q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay unchosen stage 2 state-action pairs toward 0.5
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += gamma * (0.5 - q_stage2_mf[s, a])

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Separate Inverse Temperatures for Model-Based, Model-Free, and Stage 2 Choices
Instead of a single inverse temperature (`beta`) and a weight parameter (`w`), this model parameterizes the decision noise separately for the Model-Based (`beta_mb`) and Model-Free (`beta_mf`) systems at Stage 1, as well as a separate decision noise for Stage 2 (`beta_2`). This follows the original parameterization by Daw et al. (2011).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Agent with Separate Inverse Temperatures for MB, MF, and Stage 2.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_mf: [0, 10] - Inverse temperature for model-free Stage 1 values.
    beta_mb: [0, 10] - Inverse temperature for model-based Stage 1 values.
    beta_2: [0, 10] - Inverse temperature for Stage 2 choice.
    lam: [0, 1] - Eligibility trace scaling the update of Stage 1 values by Stage 2 RPE.
    p: [-5, 5] - Perseveration parameter for Stage 1 choice.
    """
    import numpy as np
    alpha, beta_mf, beta_mb, beta_2, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF using their respective inverse temperatures directly
        q_stage1_combined = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf

        pers = np.zeros(2)
        if last_choice != -1:
            pers[last_choice] = 1
            
        q_stage1_combined += p * pers
        q_stage1_combined_shifted = q_stage1_combined - np.max(q_stage1_combined)
        
        # No beta multiplier here because it is already absorbed in the Q-values
        exp_q1 = np.exp(q_stage1_combined_shifted)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q_stage2_shifted = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta_2 * q_stage2_shifted)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
