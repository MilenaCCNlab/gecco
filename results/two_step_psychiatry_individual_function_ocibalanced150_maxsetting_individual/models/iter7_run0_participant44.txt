Here are 3 new cognitive models that extend the best-performing model by introducing novel mechanisms often observed in sequential decision-making tasks.

### Model 1: Asymmetric Stage Learning Rates
Participants often update their values at different speeds for the first and second stages because Stage 1 involves abstract, state-transition values, while Stage 2 involves direct, reward-based values. This model assigns separate learning rates to Stage 1 and Stage 2.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF with separate learning rates for Stage 1 and Stage 2, and reward-dependent stickiness.
    This model allows the agent to update values at different speeds for the first and second stage choices,
    while maintaining the general and reward-dependent stickiness for the first stage.
    
    Bounds:
    learning_rate_1: [0, 1] - Learning rate for Stage 1 Q-values.
    learning_rate_2: [0, 1] - Learning rate for Stage 2 Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for model-based vs model-free influence.
    stickiness: [-5, 5] - General choice stickiness bonus.
    reward_stickiness: [-5, 5] - Additional stickiness bonus applied only if the previous trial was rewarded.
    """
    learning_rate_1, learning_rate_2, beta, w, stickiness, reward_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus = np.zeros(2)
        if last_action_1 != -1:
            bonus[last_action_1] += stickiness
            if last_reward > 0:
                bonus[last_action_1] += reward_stickiness
                
        q1_total = beta * q_stage1_net + bonus
        q1_total -= np.max(q1_total) # Prevent overflow
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q2_total = beta * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage 2 State-Specific Stickiness
While stickiness is heavily tracked at Stage 1, participants also tend to repeat their exact choice at Stage 2 when revisiting the exact same planet (state). This model incorporates a separate stickiness parameter for Stage 2.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF with Stage 1 reward-stickiness and Stage 2 general stickiness.
    This model adds a stickiness parameter for the second-stage choice, capturing the tendency
    to repeat choices at the second stage regardless of reward, in addition to Stage 1 mechanisms.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for model-based vs model-free influence.
    stickiness_1: [-5, 5] - General choice stickiness bonus for Stage 1.
    reward_stickiness_1: [-5, 5] - Additional stickiness bonus for Stage 1 if the previous trial was rewarded.
    stickiness_2: [-5, 5] - General choice stickiness bonus for Stage 2 state-action repetition.
    """
    learning_rate, beta, w, stickiness_1, reward_stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0
    last_action_2 = np.full(2, -1) # Track last action for each state

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            bonus_1[last_action_1] += stickiness_1
            if last_reward > 0:
                bonus_1[last_action_1] += reward_stickiness_1
            
        q1_total = beta * q_stage1_net + bonus_1
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            bonus_2[last_action_2[state_idx]] += stickiness_2
            
        q2_total = beta * q_stage2_mf[state_idx] + bonus_2
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Stickiness
Often in two-step tasks, participants exhibit a strong bias to repeat the previous Stage 1 action if it resulted in a "common" transition (regardless of reward), and they are more likely to shift if they were taken to a "rare" transition because they blame the rare transition for messing up their plans. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF with transition-dependent stickiness.
    In addition to general and reward stickiness, this model adds a bonus to repeat the previous 
    Stage 1 choice if it led to a common transition (action 0 -> state 0, or action 1 -> state 1).
    This captures the tendency of participants to be reinforced by expected state transitions.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter for model-based vs model-free influence.
    stickiness: [-5, 5] - General choice stickiness bonus.
    reward_stickiness: [-5, 5] - Additional stickiness bonus if the previous trial was rewarded.
    transition_stickiness: [-5, 5] - Stickiness bonus if the previous trial resulted in a common transition.
    """
    learning_rate, beta, w, stickiness, reward_stickiness, transition_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0
    last_is_common = 0

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            bonus_1[last_action_1] += stickiness
            if last_reward > 0:
                bonus_1[last_action_1] += reward_stickiness
            if last_is_common > 0:
                bonus_1[last_action_1] += transition_stickiness
            
        q1_total = beta * q_stage1_net + bonus_1
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q2_total = beta * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Tracking for the next trial
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        
        # Check if transition was common (Act 0 -> State 0 OR Act 1 -> State 1)
        if (action_1[trial] == 0 and state_idx == 0) or (action_1[trial] == 1 and state_idx == 1):
            last_is_common = 1
        else:
            last_is_common = 0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
