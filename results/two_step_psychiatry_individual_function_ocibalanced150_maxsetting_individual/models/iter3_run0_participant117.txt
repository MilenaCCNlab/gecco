import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and decay to a neutral prior (0.5).
    Often, participants do not decay unchosen values to 0, but rather to a neutral expectation (0.5).
    This model also separates the learning rate for Stage 1 and Stage 2 to capture differences in updating.
    
    Bounds:
    alpha1: [0,1] - Learning rate for Stage 1
    alpha2: [0,1] - Learning rate for Stage 2
    decay: [0,1] - Decay rate for unchosen options towards 0.5
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Choice stickiness for Stage 1
    """
    alpha1, alpha2, decay, beta1, beta2, lam, w, p1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values to neutral expectation (0.5)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        # Decay unchosen options towards 0.5
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        prev_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Outcome-dependent eligibility trace.
    Participants may assign credit to Stage 1 choices asymmetrically depending on whether 
    the outcome in Stage 2 was better or worse than expected. This model uses separate 
    eligibility traces (lam_pos and lam_neg) based on the sign of the Stage 2 prediction error.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for Stage 1
    beta2: [0,10] - Inverse temperature for Stage 2
    lam_pos: [0,1] - Eligibility trace for positive Stage 2 prediction errors
    lam_neg: [0,1] - Eligibility trace for negative Stage 2 prediction errors
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Choice stickiness for Stage 1
    p2: [-5,5] - Choice stickiness for Stage 2
    """
    alpha, beta1, beta2, lam_pos, lam_neg, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        rep2 = np.zeros(2)
        if prev_a2 != -1:
            rep2[prev_a2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Outcome-dependent backward update
        current_lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha * current_lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Counterfactual updating for Stage 2.
    Participants may assume that the two aliens on a given planet have anti-correlated payout probabilities.
    If they receive a reward from one, they might decrease the value of the other (and vice versa).
    This model uses a counterfactual learning rate (alpha_cf) to update the unchosen alien's value.
    
    Bounds:
    alpha: [0,1] - Learning rate for chosen options
    alpha_cf: [0,1] - Learning rate for counterfactual updating of unchosen Stage 2 option
    beta1: [0,10] - Inverse temperature for Stage 1
    beta2: [0,10] - Inverse temperature for Stage 2
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Choice stickiness for Stage 1
    p2: [-5,5] - Choice stickiness for Stage 2
    """
    alpha, alpha_cf, beta1, beta2, lam, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        rep2 = np.zeros(2)
        if prev_a2 != -1:
            rep2[prev_a2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Counterfactual update for the unchosen option in Stage 2
        unchosen_a2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial] # Invert the reward (1 becomes 0, 0 becomes 1)
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_a2]
        q_stage2_mf[state_idx, unchosen_a2] += alpha_cf * delta_cf
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
