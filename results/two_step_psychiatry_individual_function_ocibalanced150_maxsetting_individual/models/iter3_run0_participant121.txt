Here are 3 new cognitive models that build upon the best-performing mechanisms (such as model-based/model-free weighting, eligibility traces, and choice stickiness) while exploring alternative configurations like separate learning rates, separate inverse temperatures (betas), and unchosen value decay.

### Model 1: Separate Stage Learning Rates & Inverse Temperatures
This model introduces separate learning rates (`alpha_1`, `alpha_2`) and separate inverse temperatures (`beta_1`, `beta_2`) for the first and second decision stages. This allows the model to capture differences in learning speed and exploration/exploitation tendencies between the two stages.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. (2011) model with separate learning rates and inverse temperatures 
    for stage 1 and stage 2, along with eligibility trace and choice stickiness.
    
    Bounds:
    alpha_1: [0,1] - Learning rate for stage 1
    alpha_2: [0,1] - Learning rate for stage 2
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p: [0,5] - Choice stickiness
    """
    import numpy as np
    alpha_1, alpha_2, beta_1, beta_2, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta_1 * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning & Separate Stage Betas
This model combines asymmetric learning rates for positive and negative prediction errors (`alpha_pos`, `alpha_neg`) with separate inverse temperatures for each stage (`beta_1`, `beta_2`). It assesses whether participants not only learn differently from gains vs. non-gains but also exploit these values differently depending on the stage of the task.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. (2011) model with asymmetric learning rates for positive and 
    negative prediction errors, and separate inverse temperatures for both stages.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    beta_1: [0,10] - Inverse temperature for stage 1
    beta_2: [0,10] - Inverse temperature for stage 2
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p: [0,5] - Choice stickiness
    """
    import numpy as np
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta_1 * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        # Eligibility trace update for stage 1 based on stage 2 PE
        alpha_lambda = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_lambda * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Value Decay (Forgetting) of Unchosen Actions
This model incorporates a `decay` parameter. In dynamic environments, participants might naturally "forget" the values of unchosen actions, causing their estimated values to decay back toward zero over time. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. (2011) model with an active decay (forgetting) mechanism for unchosen 
    actions. Unchosen actions in both stages decay towards zero on every trial.
    
    Bounds:
    learning_rate: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p: [0,5] - Choice stickiness
    decay: [0,1] - Decay rate for unchosen action values
    """
    import numpy as np
    learning_rate, beta, w, lambda_, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        # Decay unchosen stage 2 actions towards 0
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        # Decay unchosen stage 1 action towards 0
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
