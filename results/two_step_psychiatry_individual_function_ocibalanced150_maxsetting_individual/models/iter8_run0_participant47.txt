def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with asymmetric learning rates for stage 1 only.
    This model posits that individuals might update their stage 1 choices differently following 
    positive vs. negative prediction errors, while maintaining a single learning rate for stage 2.
    
    Parameters:
    alpha1_pos: [0, 1] - Learning rate for positive prediction errors in stage 1.
    alpha1_neg: [0, 1] - Learning rate for negative prediction errors in stage 1.
    alpha2: [0, 1] - Learning rate for stage 2 prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness (perseveration) for stage 1.
    pi2: [-5, 5] - Choice stickiness (perseveration) for stage 2.
    """
    alpha1_pos, alpha1_neg, alpha2, beta, w, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        lr1 = alpha1_pos if delta_stage1 > 0 else alpha1_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # eligibility trace update for stage 1
        lr1_lam = alpha1_pos if delta_stage2 > 0 else alpha1_neg
        q_stage1_mf[action_1[trial]] += lr1_lam * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with a choice trace (momentum) for stage 1.
    Given the participant's tendency to repeat stage 1 choices in long blocks, this model 
    replaces a simple last-action stickiness with an exponentially decaying choice trace.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Weight of the stage 1 choice trace (momentum).
    pi2: [-5, 5] - Choice stickiness (perseveration) for stage 2.
    tau: [0, 1] - Decay rate for the stage 1 choice trace (1 = no decay, 0 = last action only).
    """
    alpha_pos, alpha_neg, beta, w, lam, pi1, pi2, tau = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    trace_1 = np.zeros(2)
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * trace_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # update choice trace for stage 1
        trace_1 *= tau
        trace_1[action_1[trial]] += 1.0

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2
        
        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with transition-dependent stage 1 stickiness.
    This model assumes that experiencing a rare transition may disrupt a participant's choice
    perseveration, leading to different stickiness weights depending on the previous transition type.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi1_comm: [-5, 5] - Stage 1 stickiness after a common transition.
    pi1_rare: [-5, 5] - Stage 1 stickiness after a rare transition.
    pi2: [-5, 5] - Choice stickiness (perseveration) for stage 2.
    """
    alpha_pos, alpha_neg, beta, w, lam, pi1_comm, pi1_rare, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_trans_common = True

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        pi1 = pi1_comm if last_trans_common else pi1_rare
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]
        
        # Check if current transition is common to set state for next trial
        if (action_1[trial] == 0 and state_idx == 0) or (action_1[trial] == 1 and state_idx == 1):
            last_trans_common = True
        else:
            last_trans_common = False

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2
        
        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
