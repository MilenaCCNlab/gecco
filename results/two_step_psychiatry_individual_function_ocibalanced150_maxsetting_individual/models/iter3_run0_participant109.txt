def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Outcome-dependent Perseveration Model
    Adds a perseveration bonus to the last stage-1 action, which differs depending on whether the last trial was rewarded (pers_win) or not (pers_lose).
    This captures "win-stay, lose-shift" tendencies at the first-stage choice.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence.
    lam: [0, 1] - Eligibility trace parameter.
    pers_win: [-5, 5] - Perseveration bonus if the last trial was rewarded.
    pers_lose: [-5, 5] - Perseveration bonus if the last trial was not rewarded.
    """
    learning_rate, beta1, beta2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values at 0.5 (middle of reward range 0-1)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine MB and MF values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add outcome-dependent perseveration
        if last_action != -1:
            if last_reward == 1:
                q_stage1_net[last_action] += pers_win
            elif last_reward == 0:
                q_stage1_net[last_action] += pers_lose
                
        # Softmax for Stage 1
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Eligibility trace update for stage 1
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Store last action and reward for next trial
        last_action = a1
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Discount Factor Model
    Introduces a temporal discount factor (gamma) that scales the prospective value of stage 2 
    when updating stage 1, and when computing model-based values.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight for model-based vs model-free influence.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration bonus added to the previously chosen stage 1 action.
    gamma: [0, 1] - Discount factor for stage 2 values.
    """
    learning_rate, beta1, beta2, w, lam, pers, gamma = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        # Apply discount factor to MB stage 1 value
        q_stage1_mb = gamma * (transition_matrix @ max_q_stage2)
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add basic perseveration
        if last_action != -1:
            q_stage1_net[last_action] += pers
                
        # Softmax for Stage 1
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(beta1 * q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
        
        # Action value updating with discount factor for stage 1 prediction error
        delta_stage1 = gamma * q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Independent MB/MF Temperatures Model
    Uses separate inverse temperatures (beta_mb, beta_mf) for the model-based and model-free 
    systems at stage 1, replacing the weighting parameter w. This allows for independent 
    levels of choice stochasticity for the two learning systems.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_mb: [0, 10] - Inverse temperature for model-based stage 1 values.
    beta_mf: [0, 10] - Inverse temperature for model-free stage 1 values.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    lam: [0, 1] - Eligibility trace parameter.
    pers: [-5, 5] - Perseveration bonus added to the previously chosen stage 1 action.
    """
    learning_rate, beta_mb, beta_mf, beta2, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    last_action = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine Q-values using their respective betas directly
        q_stage1_net = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        
        if last_action != -1:
            q_stage1_net[last_action] += pers
                
        # Since beta is already incorporated into q_stage1_net, we do not multiply by beta again
        q1_centered = q_stage1_net - np.max(q_stage1_net)
        exp_q1 = np.exp(q1_centered)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_centered = q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])
        exp_q2 = np.exp(beta2 * q2_centered)
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
        
        # Eligibility trace update
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_action = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
