import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Hybrid MB/MF with choice trace and stage 1 bias.
    Instead of just a 1-trial perseveration, this model keeps an exponentially 
    decaying trace of past stage 1 choices to capture longer-term streaks.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating values.
    beta: [0,10] - Inverse temperature for softmax choice.
    w: [0,1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0,1] - Eligibility trace for updating stage 1 with stage 2 prediction error.
    trace_decay: [0,1] - Decay rate of the choice trace (0 = only last trial matters).
    trace_weight: [-5,5] - Weight of the choice trace on stage 1 logits.
    bias: [-5,5] - Intrinsic bias for choosing action 0 at stage 1.
    """
    learning_rate, beta, w, lam, trace_decay, trace_weight, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            choice_trace = np.zeros(2)
            continue

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net + trace_weight * choice_trace
        logits_1[0] += bias
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Trace update
        choice_trace = choice_trace * trace_decay
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Hybrid MB/MF with win-stay/lose-shift perseveration and stage 1 bias.
    This model separates the perseveration parameter into rep_win (applied if the 
    previous trial was rewarded) and rep_lose (applied if the previous trial was unrewarded).
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating values.
    beta: [0,10] - Inverse temperature for softmax choice.
    w: [0,1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0,1] - Eligibility trace for updating stage 1 with stage 2 prediction error.
    rep_win: [-5,5] - Perseveration to last stage 1 action if last trial was rewarded.
    rep_lose: [-5,5] - Perseveration to last stage 1 action if last trial was unrewarded.
    bias: [-5,5] - Intrinsic bias for choosing action 0 at stage 1.
    """
    learning_rate, beta, w, lam, rep_win, rep_lose, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                stickiness[last_action_1] = rep_win
            else:
                stickiness[last_action_1] = rep_lose
                
        logits_1 = beta * q_net + stickiness
        logits_1[0] += bias
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Hybrid MB/MF with transition matrix learning and stage 1 bias.
    Instead of assuming a fixed 0.7/0.3 transition matrix, this model learns the 
    transition probabilities based on observed stage 1 action to state transitions.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for updating Q-values.
    beta: [0,10] - Inverse temperature for softmax choice.
    w: [0,1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0,1] - Eligibility trace for updating stage 1 with stage 2 prediction error.
    rep: [-5,5] - Perseveration (stickiness) to the previously chosen stage 1 action.
    bias: [-5,5] - Intrinsic bias for choosing action 0 at stage 1.
    transition_lr: [0,1] - Learning rate for the transition matrix probabilities.
    """
    learning_rate, beta, w, lam, rep, bias, transition_lr = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to uniform distributions
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta * q_net + rep * stickiness
        logits_1[0] += bias
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 choice
        state_idx = state[trial]
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Transition matrix updates
        transition_matrix[action_1[trial], state_idx] += transition_lr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1.0 - transition_matrix[action_1[trial], state_idx]

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
