def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF RL with separate unchosen decay rates for Stage 1 and Stage 2.
    Builds on the best model by allowing the unchosen action values in Stage 1 and Stage 2 
    to decay towards 0.5 at different rates, capturing potential differences in forgetting 
    dynamics between the two stages.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating chosen Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence
    lam: [0,1] - eligibility trace parameter
    pers_st1: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    pers_st2: [-5,5] - choice perseveration bonus for repeating the previous stage 2 action
    decay_st1: [0,1] - rate at which unchosen Q-values decay towards 0.5 in Stage 1
    decay_st2: [0,1] - rate at which unchosen Q-values decay towards 0.5 in Stage 2
    """
    learning_rate, beta, w, lam, pers_st1, pers_st2, decay_st1, decay_st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action2 = {-1: -1, 0: -1, 1: -1}

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if trial > 0:
            pers_bonus_1[action_1[trial-1]] = pers_st1
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus_1)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            pers_bonus_2[last_action2[state_idx]] = pers_st2
            
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pers_bonus_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        last_action2[state_idx] = action_2[trial]

        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update for Choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Unchosen action decay for Stage 1
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] += decay_st1 * (0.5 - q_stage1_mf[unchosen_a1])
        
        # Unchosen action decay for Stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay_st2 * (0.5 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
