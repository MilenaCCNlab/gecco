Here are three new cognitive models that extend the baseline hybrid model. Given the participant's data, which shows long streaks of choosing the same spaceship (e.g., trials 8–35 for spaceship 0, and 36–53 for spaceship 1), incorporating choice perseveration (stickiness) is a highly plausible mechanism. 

These models explore different configurations of choice stickiness and combine it with the successful decay mechanism identified in your best model.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Stage 1 Choice Stickiness.
    Incorporates a stickiness parameter to capture the participant's tendency 
    to repeat the previous stage 1 choice (perseveration), which is evident in the data.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    st: Choice stickiness for stage 1 [-10, 10]
    """
    alpha, beta1, beta2, w, lam, st = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Policy for the first choice (with stickiness)
        logit1 = beta1 * q_net
        if prev_a1 != -1:
            logit1[prev_a1] += st
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # Policy for the second choice
        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Stage 1 and Stage 2 Choice Stickiness.
    Adds stickiness parameters for both stages to capture perseveration across 
    the entire decision-making process. Stage 2 stickiness is state-specific.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    st1: Choice stickiness for stage 1 [-10, 10]
    st2: Choice stickiness for stage 2 [-10, 10]
    """
    alpha, beta1, beta2, w, lam, st1, st2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = np.full(2, -1)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Policy for the first choice (with stickiness)
        logit1 = beta1 * q_net
        if prev_a1 != -1:
            logit1[prev_a1] += st1
            
        exp_q1 = np.exp(logit1 - np.max(logit1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # Policy for the second choice (with state-specific stickiness)
        logit2 = beta2 * q_stage2_mf[state_idx]
        if prev_a2[state_idx] != -1:
            logit2[prev_a2[state_idx]] += st2
            
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Forgetting (Decay) and Stage 1 Stickiness.
    Combines the Q-value decay (forgetting) mechanism that proved successful 
    previously, with choice perseveration at stage 1.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Weight given to model-based vs model-free values [0, 1]
    lam: Eligibility trace parameter [0, 1]
    decay: Forgetting rate for all Q-values per trial [0, 1]
    st: Choice stickiness for stage 1 [-10, 10]
    """
    alpha, beta1, beta2, w, lam, decay, st = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Policy for the first choice (with stickiness)
        logit1 = beta1 * q_net

        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        # Policy for the second choice
        logit2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logit2 - np.max(logit2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay (forgetting)
        q_stage1_mf *= (1 - decay)
        q_stage2_mf *= (1 - decay)
        
        prev_a1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
