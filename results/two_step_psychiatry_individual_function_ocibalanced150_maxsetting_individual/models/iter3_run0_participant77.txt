Here are 3 new cognitive models that explore alternative mechanisms for how participants might learn and make decisions in the two-step task. Each model uses 8 parameters and investigates a different well-founded hypothesis from the cognitive science literature.

### Model 1: Reward-Dependent Model-Based Control
This model tests the hypothesis that the degree of model-based (goal-directed) control fluctuates based on recent success. Specifically, the model-based weight $w$ depends on whether the previous trial was rewarded or not, reflecting state-dependent cognitive control.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Two-Step Task Model with Reward-Dependent Model-Based Weight.
    The model-based weight w depends on whether the previous trial was rewarded (w_win) 
    or unrewarded (w_lose), capturing changes in cognitive control based on recent success.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w_win: model-based weight if previous trial was rewarded [0,1]
    w_lose: model-based weight if previous trial was unrewarded [0,1]
    lam: eligibility trace parameter [0,1]
    pers: perseveration bonus for stage 1 [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w_win, w_lose, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        # Determine w based on last reward
        if trial == 0 or last_reward == -1:
            current_w = (w_win + w_lose) / 2.0  # Average for first trial or after timeout
        elif last_reward > 0:
            current_w = w_win
        else:
            current_w = w_lose

        q_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = 1.0
                
        q_net_total = beta_1 * q_net + pers * pers_bonus
        exp_q1 = np.exp(q_net_total - np.max(q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            last_reward = reward[trial]
        else:
            p_choice_2[trial] = 1.0
            last_reward = -1 
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition-Dependent Perseveration
This model evaluates if participants are more likely to repeat their stage-1 choices depending on the state transition they experienced. People often stick to their choices more strongly when the environment behaves as expected (a common transition) compared to when it violates expectations (a rare transition).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-Step Task Model with Transition-Dependent Perseveration.
    The perseveration bonus depends on whether the previous trial's state transition 
    was common or rare.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace parameter [0,1]
    pers_common: perseveration bonus if previous transition was common [-2,2]
    pers_rare: perseveration bonus if previous transition was rare [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_common, pers_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_common = True

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = 1.0
            pers = pers_common if last_trans_common else pers_rare
        else:
            pers = 0.0
                
        q_net_total = beta_1 * q_net + pers * pers_bonus
        exp_q1 = np.exp(q_net_total - np.max(q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
            
            # Check if transition was common (action 0 -> state 0 OR action 1 -> state 1)
            last_trans_common = (action_1[trial] == state_idx)
        else:
            p_choice_2[trial] = 1.0
            last_trans_common = True 
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Eligibility Trace
This model assumes that credit assignment back to stage-1 choices is asymmetric. It tests whether prediction errors from a rewarding outcome propagate back to the first stage differently than prediction errors from an unrewarding outcome (i.e., different $\lambda$ traces for wins vs. losses).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Two-Step Task Model with Outcome-Dependent Eligibility Trace.
    The eligibility trace parameter lambda depends on whether the stage 2 outcome 
    was rewarding or not, capturing asymmetric credit assignment for successes vs failures.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam_win: eligibility trace if reward > 0 [0,1]
    lam_lose: eligibility trace if reward == 0 [0,1]
    pers: perseveration bonus for stage 1 [-2,2]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam_win, lam_lose, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = 1.0
                
        q_net_total = beta_1 * q_net + pers * pers_bonus
        exp_q1 = np.exp(q_net_total - np.max(q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            q_stage2_total = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(q_stage2_total - np.max(q_stage2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]

            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

            current_lam = lam_win if reward[trial] > 0 else lam_lose
            q_stage1_mf[action_1[trial]] += alpha_1 * current_lam * delta_stage2
            
        else:
            p_choice_2[trial] = 1.0
            
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
