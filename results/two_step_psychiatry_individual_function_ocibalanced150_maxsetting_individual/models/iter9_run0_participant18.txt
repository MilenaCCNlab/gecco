def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Q-value Decay towards a Baseline (Center) and Stage 2 Perseveration.
    Includes perseveration for both Stage 1 and Stage 2 choices.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weighting of model-based vs model-free control.
    pi1: [-5, 5] - Perseveration bonus for repeating previous stage 1 choice.
    pi2: [-5, 5] - Perseveration bonus for repeating previous stage 2 choice (in the same state).
    lambda_: [0, 1] - Eligibility trace for stage 1 model-free update.
    decay: [0, 1] - Rate at which Q-values decay towards the center.
    center: [0, 1] - Baseline value towards which Q-values decay.
    """
    alpha, beta, w, pi1, pi2, lambda_, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, float(center))
    q_stage2_mf = np.full((2, 2), float(center))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1
            
        logits_1 = beta * q_net + pi1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = 1

        logits_2 = beta * q_stage2_mf[state_idx] + pi2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values before learning updates
        q_stage1_mf += decay * (center - q_stage1_mf)
        q_stage2_mf += decay * (center - q_stage2_mf)

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Q-value Decay to Center and Choice Perseveration Trace.
    Uses an exponentially decaying choice trace for perseveration rather than just the last trial.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weighting of model-based vs model-free control.
    pi: [-5, 5] - Perseveration weight for the choice trace.
    lambda_: [0, 1] - Eligibility trace for stage 1 model-free update.
    decay: [0, 1] - Rate at which Q-values decay towards the center.
    center: [0, 1] - Baseline value towards which Q-values decay.
    trace_decay: [0, 1] - Decay rate of the choice trace (1 = no decay, 0 = only last trial).
    """
    alpha, beta, w, pi, lambda_, decay, center, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, float(center))
    q_stage2_mf = np.full((2, 2), float(center))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_net + pi * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values
        q_stage1_mf += decay * (center - q_stage1_mf)
        q_stage2_mf += decay * (center - q_stage2_mf)

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Update choice trace
        choice_trace *= trace_decay
        choice_trace[action_1[trial]] = 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Q-value Decay to Center and Transition Learning.
    The transition matrix is learned over time based on observed transitions.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature for softmax choice.
    w: [0, 1] - Weighting of model-based vs model-free control.
    pi: [-5, 5] - Perseveration bonus for repeating previous stage 1 choice.
    lambda_: [0, 1] - Eligibility trace for stage 1 model-free update.
    decay: [0, 1] - Rate at which Q-values decay towards the center.
    center: [0, 1] - Baseline value towards which Q-values decay.
    alpha_tr: [0, 1] - Learning rate for updating the transition matrix.
    """
    alpha, beta, w, pi, lambda_, decay, center, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, float(center))
    q_stage2_mf = np.full((2, 2), float(center))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1
            
        logits_1 = beta * q_net + pi * pers
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values
        q_stage1_mf += decay * (center - q_stage1_mf)
        q_stage2_mf += decay * (center - q_stage2_mf)

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        # Update learned transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] -= alpha_tr * transition_matrix[action_1[trial], 1 - state_idx]
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
