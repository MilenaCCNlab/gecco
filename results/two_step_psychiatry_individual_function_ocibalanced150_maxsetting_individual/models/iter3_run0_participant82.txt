import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Separate Stage Learning Rates and Decay.
    This model extends the baseline hybrid model by using distinct learning rates 
    for stage 1 and stage 2 updates. This acknowledges that participants may update 
    their expectations at different speeds for different task stages. It also 
    includes a decay mechanism for unchosen actions.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for stage 1 actions
    alpha_2: [0, 1] - Learning rate for stage 2 actions
    decay_rate: [0, 1] - Rate at which unchosen action values decay to 0
    beta_1: [0, 10] - Inverse temperature for stage 1 choices
    beta_2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weight on model-based vs model-free strategy (1=MB, 0=MF)
    lam: [0, 1] - Eligibility trace parameter
    """
    alpha_1, alpha_2, decay_rate, beta_1, beta_2, w, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta_1 * q_net - np.max(beta_1 * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay_rate)
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_rate)

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Decay and Loss Aversion.
    This model introduces a loss aversion parameter to account for differential 
    sensitivity to negative rewards (e.g., -1 coins in trial 157). It scales 
    negative rewards by the loss_aversion parameter.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for chosen actions
    decay_rate: [0, 1] - Rate at which unchosen action values decay to 0
    beta_1: [0, 10] - Inverse temperature for stage 1 choices
    beta_2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weight on model-based vs model-free strategy (1=MB, 0=MF)
    lam: [0, 1] - Eligibility trace parameter
    loss_aversion: [0, 10] - Multiplier for negative rewards
    """
    learning_rate, decay_rate, beta_1, beta_2, w, lam, loss_aversion = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta_1 * q_net - np.max(beta_1 * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay_rate)
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        r = reward[trial]
        if r < 0:
            r = r * loss_aversion
            
        delta_stage2 = r - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_rate)

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Decay and Transition Matrix Learning.
    Instead of assuming fixed transition probabilities (0.7/0.3), this model updates 
    its internal transition matrix based on observed state transitions, accounting 
    for participants learning the transition structure dynamically over time.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for chosen actions
    decay_rate: [0, 1] - Rate at which unchosen action values decay to 0
    beta_1: [0, 10] - Inverse temperature for stage 1 choices
    beta_2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weight on model-based vs model-free strategy (1=MB, 0=MF)
    lam: [0, 1] - Eligibility trace parameter
    alpha_tr: [0, 1] - Learning rate for transition probabilities
    """
    learning_rate, decay_rate, beta_1, beta_2, w, lam, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize with default transition probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta_1 * q_net - np.max(beta_1 * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx] - np.max(beta_2 * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay_rate)
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_rate)

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        # Update empirical transition matrix based on experienced transitions
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_tr * (1 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_tr * (0 - transition_matrix[a1, 1 - state_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
