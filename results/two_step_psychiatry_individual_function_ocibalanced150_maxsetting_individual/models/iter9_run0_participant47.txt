def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Q-learning model with transition matrix learning.
    Instead of assuming a fixed transition matrix, the participant learns the transition probabilities
    dynamically based on observed transitions.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness (perseveration) for stage 1.
    pi2: [-5, 5] - Choice stickiness (perseveration) for stage 2.
    alpha_trans: [0, 1] - Learning rate for the transition matrix.
    """
    alpha_pos, alpha_neg, beta, w, lam, pi1, pi2, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to uniform probabilities
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]
  
        # Update transition matrix probabilities based on observation
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1.0 - transition_matrix[action_1[trial], state_idx]

        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with transition-dependent stage 1 stickiness.
    Uses separate learning rates and inverse temperatures for stage 1 and stage 2.
    
    Parameters:
    alpha1: [0, 1] - Learning rate for stage 1.
    alpha2: [0, 1] - Learning rate for stage 2.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Mixing weight for model-based vs model-free values.
    lam: [0, 1] - Eligibility trace parameter.
    pi1_comm: [-5, 5] - Stage 1 stickiness after a common transition.
    pi1_rare: [-5, 5] - Stage 1 stickiness after a rare transition.
    """
    alpha1, alpha2, beta1, beta2, w, lam, pi1_comm, pi1_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_trans_common = True

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        pi1 = pi1_comm if last_trans_common else pi1_rare
            
        exp_q1 = np.exp(beta1 * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # Determine if the current transition was common or rare for the next trial
        if (action_1[trial] == 0 and state_idx == 0) or (action_1[trial] == 1 and state_idx == 1):
            last_trans_common = True
        else:
            last_trans_common = False

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update for choice 1
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition-dependent eligibility trace.
    Participants may discount rewards obtained after a rare transition when updating stage 1 values,
    attributing the outcome to "bad luck" in the transition rather than a bad stage 1 choice.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam_comm: [0, 1] - Eligibility trace parameter for common transitions.
    lam_rare: [0, 1] - Eligibility trace parameter for rare transitions.
    pi1: [-5, 5] - Choice stickiness for stage 1.
    pi2: [-5, 5] - Choice stickiness for stage 2.
    """
    alpha_pos, alpha_neg, beta, w, lam_comm, lam_rare, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]
  
        # ACTION VALUE UPDATING FOR CHOICE 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        # ACTION VALUE UPDATING FOR CHOICE 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        # Eligibility trace update for choice 1 depends on transition type
        trans_common = (action_1[trial] == 0 and state_idx == 0) or (action_1[trial] == 1 and state_idx == 1)
        lam = lam_comm if trans_common else lam_rare
        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
