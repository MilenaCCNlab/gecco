Here are three new cognitive models that build upon the success of the outcome-dependent perseveration (WSLS) mechanism, while exploring alternative cognitive processes instead of unchosen decay. 

### Model 1: Separate Learning Rates for Stage 1 and Stage 2 with WSLS
This model explores the idea that participants might update their model-free values at different speeds for the first stage (spaceship choice) versus the second stage (alien choice). It retains the successful win-stay/lose-shift (WSLS) perseveration mechanism for the first stage.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate Learning Rates for Stage 1 and Stage 2 with WSLS
    
    This model uses distinct learning rates for updating Stage 1 and Stage 2 
    model-free Q-values. It retains the outcome-dependent perseveration (WSLS) 
    where a previous reward encourages repeating the Stage 1 choice, and a 
    previous non-reward encourages switching (or staying, depending on pers_lose).
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for Stage 1
    alpha_2: [0, 1] - Learning rate for Stage 2
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Model-based vs model-free weight
    lam: [0, 1] - Eligibility trace
    pers_win: [-5, 5] - Stage 1 perseveration bonus if previous trial rewarded
    pers_lose: [-5, 5] - Stage 1 perseveration bonus if previous trial unrewarded
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus_1[last_action_1] = pers_win
            else:
                pers_bonus_1[last_action_1] = pers_lose
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: WSLS with Stage 2 Perseveration
This model tests whether participants exhibit perseveration not only at the first stage (spaceship choice) but also at the second stage (alien choice). It tracks the last alien chosen on each specific planet and adds a perseveration bonus to it.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: WSLS for Stage 1 and Perseveration for Stage 2
    
    This model incorporates the successful outcome-dependent perseveration (WSLS) 
    for Stage 1 choices, but adds a standard perseveration parameter for Stage 2. 
    If a participant visits a planet they have visited before, they may have a 
    tendency to repeat their last choice on that specific planet.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Model-based vs model-free weight
    lam: [0, 1] - Eligibility trace
    pers_win: [-5, 5] - Stage 1 perseveration bonus if previous trial rewarded
    pers_lose: [-5, 5] - Stage 1 perseveration bonus if previous trial unrewarded
    pers_2: [-5, 5] - Stage 2 perseveration bonus for repeating the last action in that state
    """
    alpha, beta_1, beta_2, w, lam, pers_win, pers_lose, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.full(2, -1) 

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus_1[last_action_1] = pers_win
            else:
                pers_bonus_1[last_action_1] = pers_lose
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        state_idx = state[trial]
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: WSLS with Dynamic Transition Learning
This model posits that instead of maintaining a strictly fixed belief about the transition structure (70/30), participants actively learn and update the transition matrix dynamically based on the transitions they experience.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: WSLS with Dynamic Transition Learning
    
    This model assumes participants update their internal model of the transition 
    probabilities over time using a transition learning rate (alpha_trans). This 
    is combined with the WSLS perseveration logic for Stage 1.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Model-based vs model-free weight
    lam: [0, 1] - Eligibility trace
    pers_win: [-5, 5] - Stage 1 perseveration bonus if previous trial rewarded
    pers_lose: [-5, 5] - Stage 1 perseveration bonus if previous trial unrewarded
    alpha_trans: [0, 1] - Learning rate for updating the transition matrix
    """
    alpha, beta_1, beta_2, w, lam, pers_win, pers_lose, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus_1[last_action_1] = pers_win
            else:
                pers_bonus_1[last_action_1] = pers_lose
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Dynamic update of the transition matrix
        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_trans * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_trans * (0.0 - transition_matrix[a1, 1 - s])

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
