def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for Stage 1 and Stage 2, separate inverse temperatures, and separate perseveration for each stage.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for Stage 1
    alpha2: [0, 1] - Learning rate for Stage 2
    beta1: [0, 10] - Inverse temperature for Stage 1
    beta2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    pers1: [-5, 5] - Perseveration for Stage 1
    pers2: [-5, 5] - Perseveration for Stage 2
    """
    alpha1, alpha2, beta1, beta2, w, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_state = -1
    last_action_2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_state = -1
            last_action_2 = -1
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus1[last_action_1] = pers1
            
        q1_scaled = beta1 * (q_net1 + pers_bonus1)
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q2_net = np.copy(q_stage2_mf[state_idx])
        pers_bonus2 = np.zeros(2)

        if last_state == state_idx and last_action_2 != -1:
            pers_bonus2[last_action_2] = pers2
                
        q2_scaled = beta2 * (q2_net + pers_bonus2)
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_state = state_idx
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with reward-dependent perseveration at both stages, without eligibility trace.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Inverse temperature for Stage 1
    beta2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Model-based weight
    pers_win1: [-5, 5] - Perseveration for Stage 1 if previous trial was rewarded
    pers_lose1: [-5, 5] - Perseveration for Stage 1 if previous trial was unrewarded
    pers_win2: [-5, 5] - Perseveration for Stage 2 if previous trial was rewarded
    pers_lose2: [-5, 5] - Perseveration for Stage 2 if previous trial was unrewarded
    """
    alpha, beta1, beta2, w, pers_win1, pers_lose1, pers_win2, pers_lose2 = model_parameters
    n_trials = len(action_1)
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_state = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_state = -1
            last_action_2 = -1
            last_reward = -1
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus1[last_action_1] = pers_win1
            else:
                pers_bonus1[last_action_1] = pers_lose1
            
        q1_scaled = beta1 * (q_net1 + pers_bonus1)
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q2_net = np.copy(q_stage2_mf[state_idx])
        pers_bonus2 = np.zeros(2)

        if last_state == state_idx and last_action_2 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus2[last_action_2] = pers_win2
            else:
                pers_bonus2[last_action_2] = pers_lose2
                
        q2_scaled = beta2 * (q2_net + pers_bonus2)
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_state = state_idx
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates for positive and negative prediction errors, single beta, and reward-dependent Stage 2 perseveration.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive PE
    alpha_neg: [0, 1] - Learning rate for negative PE
    beta: [0, 10] - Inverse temperature for both stages
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    pers1: [-5, 5] - Perseveration for Stage 1
    pers_win2: [-5, 5] - Perseveration for Stage 2 if previous trial was rewarded
    pers_lose2: [-5, 5] - Perseveration for Stage 2 if previous trial was unrewarded
    """
    alpha_pos, alpha_neg, beta, w, lam, pers1, pers_win2, pers_lose2 = model_parameters
    n_trials = len(action_1)
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_state = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_state = -1
            last_action_2 = -1
            last_reward = -1
            continue

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus1[last_action_1] = pers1
            
        q1_scaled = beta * (q_net1 + pers_bonus1)
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        q2_net = np.copy(q_stage2_mf[state_idx])
        pers_bonus2 = np.zeros(2)

        if last_state == state_idx and last_action_2 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus2[last_action_2] = pers_win2
            else:
                pers_bonus2[last_action_2] = pers_lose2
                
        q2_scaled = beta * (q2_net + pers_bonus2)
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        alpha_lam = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1 + alpha_lam * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_state = state_idx
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
