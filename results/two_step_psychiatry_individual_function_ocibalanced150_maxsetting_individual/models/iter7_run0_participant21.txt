Here are 3 new cognitive models that build upon your best model by introducing novel mechanisms and combinations, avoiding the ones you have already tested.

### Model 1: Hybrid Model with Decay Towards a Baseline (`q0`)
Instead of decaying unchosen Q-values towards 0, this model decays them towards a baseline value `q0`. In environments where rewards are common (e.g., probability > 0.5), decaying to 0 might underestimate the true value of unchosen options, so allowing the model to learn its own decay asymptote can improve fit.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model with Forgetting (Decay) towards a baseline value.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) evaluations.
    lam: [0, 1] - Eligibility trace parameter for stage 1 updating.
    decay: [0, 1] - Rate at which Q-values of unchosen actions decay.
    q0: [0, 1] - Baseline value towards which unchosen options decay.
    """
    alpha, beta, w, lam, decay, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to the baseline q0
    q_stage1_mf = np.ones(2) * q0
    q_stage2_mf = np.ones((2, 2)) * q0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1] + decay * q0
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a] + decay * q0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid Model with Decay, Transition Learning, and Choice Stickiness
This model combines standard decay with two highly effective mechanisms you have tried independently but not together with decay: transition matrix learning (`alpha_tr`) and choice perseveration (`stickiness`). This allows the model to capture rare-transition adaptations and habitual repeating of choices, on top of value decay.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Forgetting (Decay), Transition Learning, and Choice Stickiness.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) evaluations.
    lam: [0, 1] - Eligibility trace parameter for stage 1 updating.
    decay: [0, 1] - Rate at which Q-values of unchosen actions decay towards 0.
    alpha_tr: [0, 1] - Learning rate for the transition matrix.
    stickiness: [-5, 5] - Choice stickiness (perseveration) for stage 1.
    """
    alpha, beta, w, lam, decay, alpha_tr, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_choice = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add stickiness bonus to the previously chosen option
        stickiness_bonus = np.zeros(2)
        if last_choice != -1:
            stickiness_bonus[last_choice] = stickiness
            
        logits_1 = beta * q_net + stickiness_bonus
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update transition matrix based on observed state
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_tr * (1 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_tr * (0 - transition_matrix[a1, 1 - state_idx])
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1]
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a]

        last_choice = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid Model with Separate Temperatures (Beta) for MB, MF, and Stage 2
Instead of forcing the model to share a single temperature parameter `beta` and balancing MB/MF strictly via a weighting parameter `w`, this model breaks down the decision logic into three distinct inverse temperatures (`beta_mb`, `beta_mf`, `beta_stage2`). This provides a more flexible scaling of how deterministically the participant behaves based on model-based value vs model-free value independently.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Separate Temperatures for MB, MF, and Stage 2, plus Decay.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_mb: [0, 10] - Inverse temperature for model-based Q-values in stage 1.
    beta_mf: [0, 10] - Inverse temperature for model-free Q-values in stage 1.
    beta_stage2: [0, 10] - Inverse temperature for stage 2.
    lam: [0, 1] - Eligibility trace parameter for stage 1 updating.
    decay: [0, 1] - Rate at which Q-values of unchosen actions decay towards 0.
    """
    alpha, beta_mb, beta_mf, beta_stage2, lam, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Independent scaling for MB and MF via separate betas
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Separate beta for stage 2
        logits_2 = beta_stage2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = (1 - decay) * q_stage1_mf[unchosen_1]
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = (1 - decay) * q_stage2_mf[s, a]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
