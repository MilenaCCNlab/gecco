def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Reward-Dependent MB Weight.
    The weight given to the model-based system depends on whether the previous trial was rewarded.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w_win: [0, 1] (MB weight after rewarded trial)
    w_lose: [0, 1] (MB weight after unrewarded trial)
    lam: [0, 1] (Eligibility trace)
    pi: [-5, 5] (Choice perseveration)
    """
    alpha, beta1, beta2, w_win, w_lose, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        
        if last_reward == 1:
            current_w = w_win
        elif last_reward == 0:
            current_w = w_lose
        else:
            current_w = (w_win + w_lose) / 2.0

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1
            
        logits_1 = beta1 * q_net + pi * persev
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Stage 1 & Stage 2 Perseveration and Decay.
    Includes choice stickiness for both decision stages and forgetting of unchosen options over time.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi1: [-5, 5] (Stage 1 choice perseveration)
    pi2: [-5, 5] (Stage 2 choice perseveration)
    decay: [0, 1] (Forgetting rate for unchosen actions)
    """
    alpha, beta1, beta2, w, lam, pi1, pi2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = -1
    last_state = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev1 = np.zeros(2)
        if last_action1 != -1:
            persev1[last_action1] = 1
            
        logits_1 = beta1 * q_net + pi1 * persev1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]

        # policy for the second choice
        persev2 = np.zeros(2)
        if last_state == state_idx and last_action2 != -1:
            persev2[last_action2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + pi2 * persev2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Decay unchosen
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action1 = action_1[trial]
        last_action2 = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Action and State Perseveration.
    Includes standard action perseveration (repeating the last Stage 1 choice) and state perseveration 
    (choosing the Stage 1 action that commonly leads to the previous trial's Stage 2 state).
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi_action: [-5, 5] (Perseveration on previous Stage 1 action)
    pi_state: [-5, 5] (Perseveration on action that commonly leads to previous Stage 2 state)
    """
    alpha, beta1, beta2, w, lam, pi_action, pi_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_state = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev_act = np.zeros(2)
        if last_action != -1:
            persev_act[last_action] = 1
            
        persev_st = np.zeros(2)
        if last_state != -1:
            # Action 0 commonly leads to State 0; Action 1 commonly leads to State 1
            persev_st[last_state] = 1
            
        logits_1 = beta1 * q_net + pi_action * persev_act + pi_state * persev_st
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
