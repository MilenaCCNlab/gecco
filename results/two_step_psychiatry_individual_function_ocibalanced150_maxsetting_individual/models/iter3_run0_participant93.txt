Here are three new cognitive models that explore alternative mechanisms for how participants make decisions in the two-step task.

### Model 1: Stage-Specific Learning and Stickiness
This model investigates whether participants learn and perseverate differently at the two stages of the task. It includes separate learning rates (`alpha_1`, `alpha_2`) and choice stickiness parameters (`pi_1`, `pi_2`) for Stage 1 and Stage 2 decisions.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates and stickiness parameters for Stage 1 and Stage 2.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0,1]
    alpha_2: learning rate for stage 2 [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pi_1: choice stickiness for stage 1 [0,5]
    pi_2: choice stickiness for stage 2 [0,5]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pi_1, pi_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # Handle missing trials
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_action_2 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        q1_vals = beta_1 * q_net + pi_1 * stickiness_1
        q1_vals -= np.max(q1_vals) 
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness_2[last_action_2] = 1.0

        q2_vals = beta_2 * q_stage2_mf[state_idx] + pi_2 * stickiness_2
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Dynamic Transition Matrix Learning
This model drops the assumption that participants maintain a fixed belief of the 70%/30% transition probabilities. Instead, it introduces a learning rate (`alpha_tr`) allowing the participant to dynamically update their internal `transition_matrix` representation based on the state transitions they actually experience.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model that dynamically updates the transition matrix based on experienced state transitions.
    
    Parameters:
    alpha: learning rate for Q-values [0,1]
    alpha_tr: learning rate for transition matrix [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pi: choice stickiness [0,5]
    """
    alpha, alpha_tr, beta_1, beta_2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta_1 * q_net + pi * stickiness
        q1_vals -= np.max(q1_vals) 
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_vals = beta_2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update transition matrix probabilistically
        a1 = action_1[trial]
        s = state_idx
        transition_matrix[a1, s] += alpha_tr * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1-s] += alpha_tr * (0.0 - transition_matrix[a1, 1-s])

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Choice Stickiness (Win-Stay/Lose-Shift)
This model tests whether perseverance acts independently of Q-values but is still directly responsive to recent outcomes. It introduces separate parameters for stickiness following a rewarded trial (`pi_win`) and an unrewarded trial (`pi_lose`).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with outcome-dependent choice stickiness (Win-Stay/Lose-Shift tendencies) for Stage 1.
    
    Parameters:
    alpha: learning rate [0,1]
    beta_1: inverse temperature for stage 1 [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pi_win: choice stickiness after a rewarded trial [0,5]
    pi_lose: choice stickiness after an unrewarded trial [-5,5]
    """
    alpha, beta_1, beta_2, w, lam, pi_win, pi_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_reward = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply Win-Stay or Lose-Shift stickiness
        stickiness = np.zeros(2)
        if last_action != -1:
            if last_reward == 1:
                stickiness[last_action] = pi_win
            elif last_reward == 0:
                stickiness[last_action] = pi_lose
            
        q1_vals = beta_1 * q_net + stickiness
        q1_vals -= np.max(q1_vals) 
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_vals = beta_2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
