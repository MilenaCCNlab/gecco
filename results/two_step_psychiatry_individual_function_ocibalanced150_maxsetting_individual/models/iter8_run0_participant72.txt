import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw Two-Step Model with Separate MB/MF Inverse Temperatures and Reward-Dependent Stickiness.
    This model separates choice stickiness into two parameters based on whether the previous trial was rewarded or unrewarded.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen actions.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.
    beta_mb: [0,10] - Inverse temperature for model-based Q-values at stage 1.
    beta_mf: [0,10] - Inverse temperature for model-free Q-values at stage 1.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    lam: [0,1] - Eligibility trace parameter.
    p_rew: [0,10] - Choice stickiness (perseveration) for the first-stage choice if the previous trial was rewarded.
    p_unrew: [0,10] - Choice stickiness for the first-stage choice if the previous trial was unrewarded.
    """
    learning_rate, decay, beta_mb, beta_mf, beta_2, lam, p_rew, p_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        p = p_rew if last_reward == 1 else p_unrew
        
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * stickiness
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate MB/MF Inverse Temperatures with Asymmetric Learning Rates (Positive/Negative PE).
    This model assumes that participants update their Q-values differently for positive versus negative prediction errors.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.
    beta_mb: [0,10] - Inverse temperature for model-based Q-values at stage 1.
    beta_mf: [0,10] - Inverse temperature for model-free Q-values at stage 1.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    lam: [0,1] - Eligibility trace parameter.
    p: [0,10] - Choice stickiness (perseveration) for the first-stage choice.
    """
    alpha_pos, alpha_neg, decay, beta_mb, beta_mf, beta_2, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        stickiness = np.zeros(2)
        if last_action_1 != -1:

        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate MB/MF Inverse Temperatures with Transition Matrix Learning.
    Instead of assuming a fixed transition matrix, this model dynamically updates the individual's 
    internal representation of the transition probabilities based on experience.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen actions.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.
    beta_mb: [0,10] - Inverse temperature for model-based Q-values at stage 1.
    beta_mf: [0,10] - Inverse temperature for model-free Q-values at stage 1.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    lam: [0,1] - Eligibility trace parameter.
    p: [0,10] - Choice stickiness (perseveration) for the first-stage choice.
    alpha_trans: [0,1] - Learning rate for updating the transition matrix.
    """
    learning_rate, decay, beta_mb, beta_mf, beta_2, lam, p, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * stickiness
        logits_1 = logits_1 - np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        logits_2 = logits_2 - np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update transition matrix dynamically based on observed state transitions
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
