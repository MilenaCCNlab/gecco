Here are 3 new cognitive models that explore different mechanisms and parameter configurations not yet tried, while adhering to the 8-parameter limit.

### Model 1: Asymmetric Learning Rates + Stage-Specific Perseveration
This model tests whether participants learn differently from positive versus negative reward prediction errors (RPEs). It incorporates separate learning rates for positive and negative RPEs, an eligibility trace, separate win/lose perseveration for Stage 1, and general perseveration for Stage 2.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates (positive/negative RPEs), shared inverse temperature,
    eligibility trace, win/lose perseveration for Stage 1, and general perseveration for Stage 2.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive RPEs
    alpha_neg: [0, 1] - Learning rate for negative RPEs
    beta: [0, 10] - Shared inverse temperature
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    pers_win1: [-5, 5] - Stage 1 perseveration if previous trial was rewarded
    pers_lose1: [-5, 5] - Stage 1 perseveration if previous trial was unrewarded
    pers2: [-5, 5] - Stage 2 general perseveration
    """
    alpha_pos, alpha_neg, beta, w, lam, pers_win1, pers_lose1, pers2 = model_parameters
    n_trials = len(action_1)
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_state = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_state = -1
            last_action_2 = -1
            last_reward = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus1[last_action_1] = pers_win1
            else:
                pers_bonus1[last_action_1] = pers_lose1
            
        q1_scaled = beta * (q_net1 + pers_bonus1)
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = np.copy(q_stage2_mf[state_idx])
        pers_bonus2 = np.zeros(2)

        if last_state == state_idx and last_action_2 != -1:
            pers_bonus2[last_action_2] = pers2
                
        q2_scaled = beta * (q2_net + pers_bonus2)
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_state = state_idx
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning Rates + Win-Only Stage 2 Perseveration
This model isolates learning rates by stage instead of valence. It applies separate learning rates for Stage 1 and Stage 2 updates, keeps a shared decision noise parameter, and evaluates if Stage 2 perseveration is purely driven by previous wins.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for Stage 1 and Stage 2, shared inverse temperature,
    eligibility trace, win/lose perseveration for Stage 1, and win-only perseveration for Stage 2.
    
    Bounds:
    alpha1: [0, 1] - Stage 1 learning rate
    alpha2: [0, 1] - Stage 2 learning rate
    beta: [0, 10] - Shared inverse temperature
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    pers_win1: [-5, 5] - Stage 1 perseveration if previous trial was rewarded
    pers_lose1: [-5, 5] - Stage 1 perseveration if previous trial was unrewarded
    pers_win2: [-5, 5] - Stage 2 perseveration if previous trial was rewarded
    """
    alpha1, alpha2, beta, w, lam, pers_win1, pers_lose1, pers_win2 = model_parameters
    n_trials = len(action_1)
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_state = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_state = -1
            last_action_2 = -1
            last_reward = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus1[last_action_1] = pers_win1
            else:
                pers_bonus1[last_action_1] = pers_lose1

        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)

        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = np.copy(q_stage2_mf[state_idx])
        pers_bonus2 = np.zeros(2)

        if last_state == state_idx and last_action_2 != -1 and last_reward == 1:
            pers_bonus2[last_action_2] = pers_win2
                
        q2_scaled = beta * (q2_net + pers_bonus2)
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_state = state_idx
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Explicit Transition-Dependent Perseveration
Some subjects show model-based-like behavior not strictly through Q-learning, but by directly biasing their choices based on the previous trial's transition and reward. This model decomposes Stage 1 perseveration into 4 distinct parameters based on whether the last transition was *common* or *rare*, and whether it was *rewarded* or *unrewarded*.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with shared learning rate and inverse temperature, eligibility trace, and 
    Stage 1 perseveration decomposed by both previous reward and previous transition type 
    (common vs rare).
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta: [0, 10] - Shared inverse temperature
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    pers_wc: [-5, 5] - Stage 1 perseveration if prev trial rewarded & common transition
    pers_wr: [-5, 5] - Stage 1 perseveration if prev trial rewarded & rare transition
    pers_lc: [-5, 5] - Stage 1 perseveration if prev trial unrewarded & common transition
    pers_lr: [-5, 5] - Stage 1 perseveration if prev trial unrewarded & rare transition
    """
    alpha, beta, w, lam, pers_wc, pers_wr, pers_lc, pers_lr = model_parameters
    n_trials = len(action_1)
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_state = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_state = -1
            last_action_2 = -1
            last_reward = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        
        # Determine if the previous transition was common (action_1 == state)
        if last_action_1 != -1 and last_reward != -1 and last_state != -1:
            is_common = (last_action_1 == last_state)
            if last_reward == 1:
                pers_bonus1[last_action_1] = pers_wc if is_common else pers_wr
            else:
                pers_bonus1[last_action_1] = pers_lc if is_common else pers_lr
            
        q1_scaled = beta * (q_net1 + pers_bonus1)
        q1_scaled -= np.max(q1_scaled)
        exp_q1 = np.exp(q1_scaled)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q2_net = np.copy(q_stage2_mf[state_idx])
        
        q2_scaled = beta * q2_net
        q2_scaled -= np.max(q2_scaled)
        exp_q2 = np.exp(q2_scaled)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_state = state_idx
        last_action_2 = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
