def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with Model-Based / Model-Free weighting and choice stickiness for both stages.
    Builds upon the best model so far by introducing a mixing weight (w) to blend 
    Model-Based and Model-Free Q-values at Stage 1, capturing participants who use a hybrid strategy.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax choice probabilities.
    w: [0, 1] - Weighting of Model-Based vs Model-Free values for Stage 1 (1=pure MB, 0=pure MF).
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    learning_rate, beta, w, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Mix MB and MF values
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q1_net.copy()
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        exp_q1 = np.exp(beta * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
