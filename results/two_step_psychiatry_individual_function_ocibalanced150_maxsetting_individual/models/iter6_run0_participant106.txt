def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition matrix learning. The participant updates their internal 
    model of the transition probabilities based on observed transitions.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    alpha_t: [0, 1] - Learning rate for updating transition probabilities.
    beta1: [0, 10] - Inverse temperature for stage 1 choice.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace for updating stage 1 based on stage 2 reward.
    rho1: [0, 10] - Perseveration bonus for repeating the previous stage 1 choice.
    rho2: [0, 10] - Perseveration bonus for repeating the previous stage 2 choice in the same state.
    """
    learning_rate, alpha_t, beta1, beta2, w, lam, rho1, rho2 = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix, which will be updated trial-by-trial
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1
            
        logits1 = beta1 * q_stage1_net + rho1 * pers1
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = 1

        logits2 = beta2 * q_stage2_mf[state_idx] + rho2 * pers2
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Transition matrix updating
        transition_matrix[action_1[trial], state_idx] += alpha_t * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_t * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a free parameter for initial Q-values (q_init), separating temperatures 
    and perseveration, but without Q-value decay.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    beta1: [0, 10] - Inverse temperature for stage 1 choice.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    w: [0, 1] - Weighting of model-based vs model-free values.
    lam: [0, 1] - Eligibility trace for updating stage 1 based on stage 2 reward.
    rho1: [0, 10] - Perseveration bonus for repeating the previous stage 1 choice.
    rho2: [0, 10] - Perseveration bonus for repeating the previous stage 2 choice in the same state.
    q_init: [0, 1] - Initial value for all Q-values.
    """
    learning_rate, beta1, beta2, w, lam, rho1, rho2, q_init = model_parameters
    n_trials = len(action_1)

    # Initialize Q-values with the free parameter q_init
    q_stage1_mf = np.full(2, float(q_init))
    q_stage2_mf = np.full((2, 2), float(q_init))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model where model-based weight (w) depends dynamically on the reward of the previous trial (w_win, w_lose).
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.
    beta1: [0, 10] - Inverse temperature for stage 1 choice.
    beta2: [0, 10] - Inverse temperature for stage 2 choice.
    w_win: [0, 1] - Model-based weight after a rewarded trial.
    w_lose: [0, 1] - Model-based weight after an unrewarded trial.
    lam: [0, 1] - Eligibility trace for updating stage 1 based on stage 2 reward.
    rho1: [0, 10] - Perseveration bonus for repeating the previous stage 1 choice.
    """
    learning_rate, decay, beta1, beta2, w_win, w_lose, lam, rho1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        
        # Determine model-based weight depending on last reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = (w_win + w_lose) / 2.0

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = 1
            
        logits1 = beta1 * q_stage1_net + rho1 * pers1
        exp_q1 = np.exp(logits1 - np.max(logits1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits2 - np.max(logits2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Unchosen Q-value decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
