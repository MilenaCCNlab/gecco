def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition probability learning with separate learning rates for Stage 1 and 2.
    Instead of assuming a fixed 0.7/0.3 transition matrix, this model learns the 
    transition probabilities dynamically based on experienced state transitions.
    
    Parameters:
    learning_rate_1: [0,1] - Learning rate for updating stage 1 values.
    learning_rate_2: [0,1] - Learning rate for updating stage 2 values.
    lr_trans: [0,1] - Learning rate for updating transition probabilities.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers: [-5,5] - Perseveration bonus for repeating stage 1 choice.
    """
    learning_rate_1, learning_rate_2, lr_trans, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix naively (0.5 for all transitions)
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1 + learning_rate_1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        
        # Update transition matrix
        transition_matrix[action_1[trial], state_idx] += lr_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += lr_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for Stage 1 and 2, with decay for unchosen options.
    This model assumes that the values of actions not selected on a given trial 
    decay back towards zero, reflecting forgetting.
    
    Parameters:
    learning_rate_1: [0,1] - Learning rate for updating stage 1 values.
    learning_rate_2: [0,1] - Learning rate for updating stage 2 values.
    decay: [0,1] - Decay rate for unchosen actions towards initial value (0).
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers: [-5,5] - Perseveration bonus for repeating stage 1 choice.
    """
    learning_rate_1, learning_rate_2, decay, beta_1, beta_2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating for chosen options
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1 + learning_rate_1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        
        # Value decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != action_2[trial]:
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate learning rates for Stage 1 and 2, with perseveration for both stages.
    This model allows for habit-like repetition of both the first-stage spaceship choice 
    and the second-stage alien choice (within a specific planet).
    
    Parameters:
    learning_rate_1: [0,1] - Learning rate for updating stage 1 values.
    learning_rate_2: [0,1] - Learning rate for updating stage 2 values.
    beta_1: [0,10] - Inverse temperature for stage 1 softmax action selection.
    beta_2: [0,10] - Inverse temperature for stage 2 softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) values at stage 1.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 reward.
    pers_stage1: [-5,5] - Perseveration bonus for repeating stage 1 choice.
    pers_stage2: [-5,5] - Perseveration bonus for repeating stage 2 choice in the same state.
    """
    learning_rate_1, learning_rate_2, beta_1, beta_2, w, lam, pers_stage1, pers_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) # Tracks the last action taken in each stage 2 state

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_stage1
            
        exp_q1 = np.exp(beta_1 * (q_stage1_net + pers_bonus_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_stage2
            
        exp_q2 = np.exp(beta_2 * (q_stage2_mf[state_idx] + pers_bonus_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Action value updating
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1 + learning_rate_1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
