def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with separate inverse temperatures for stage 1 and stage 2, and choice perseveration at both stages.
    This model isolates decision noise between the two stages, as choice stochasticity can differ for the first and second step.
    
    Parameters:
    alpha: [0,1] - learning rate for value updates.
    beta1: [0,10] - inverse temperature for stage 1 softmax action selection.
    beta2: [0,10] - inverse temperature for stage 2 softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p1: [-5,5] - perseveration bonus added to the previously chosen first-stage action.
    p2: [-5,5] - perseveration bonus added to the previously chosen second-stage action (per state).
    """
    alpha, beta1, beta2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = p1
            
        q_net_pers1 = beta1 * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        pers2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers2[last_action_2[state_idx]] = p2
            
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta2 * q2 + pers2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid RL Model with reward-dependent stage 1 choice perseveration (Win-Stay/Lose-Shift) and stage 2 perseveration.
    This model examines if stickiness to the previous first-stage action is modulated by whether the last trial was rewarded.
    
    Parameters:
    alpha: [0,1] - learning rate for value updates.
    beta: [0,10] - inverse temperature for softmax action selection.
    w: [0,1] - weight of model-based vs model-free values.
    lam: [0,1] - eligibility trace parameter.
    p1_win: [-5,5] - perseveration bonus added to the previously chosen first-stage action if the previous trial was rewarded.
    p1_loss: [-5,5] - perseveration bonus added to the previously chosen first-stage action if the previous trial was unrewarded.
    p2: [-5,5] - perseveration bonus added to the previously chosen second-stage action (per state).
    """
    alpha, beta, w, lam, p1_win, p1_loss, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1 or state[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                pers1[last_action_1] = p1_win
            elif last_reward == 0:
                pers1[last_action_1] = p1_loss
            
        q_net_pers1 = beta * q_stage1_net + pers1
        exp_q1 = np.exp(q_net_pers1 - np.max(q_net_pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice

            pers2[last_action_2[state_idx]] = p2
            
        q2 = q_stage2_mf[state_idx]
        q_net_pers2 = beta * q2 + pers2
        exp_q2 = np.exp(q_net_pers2 - np.max(q_net_pers2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
