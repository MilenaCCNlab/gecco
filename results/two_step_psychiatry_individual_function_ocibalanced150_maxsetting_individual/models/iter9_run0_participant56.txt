def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with counterfactual learning for the unchosen stage 2 action on the same planet.
    Participants may infer that if one alien on a planet provides a reward, the other might not,
    leading to a counterfactual update of the unchosen alien's value.
    
    Bounds:
    alpha: [0,1] - Learning rate for chosen actions
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based vs model-free weight
    lam: [0,1] - Eligibility trace
    p1: [-5,5] - Stage 1 choice perseveration
    p2: [-5,5] - Stage 2 choice perseveration
    alpha_cf: [0,1] - Counterfactual learning rate for the unchosen action in the same state
    """
    alpha, beta, w, lam, p1, p2, alpha_cf = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_1[last_action_1] = p1
            
        logits_1 = beta * q_stage1_net + pers_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # policy for the second choice
        state_idx = state[trial]
        pers_2 = np.zeros(2)
        if last_action_2_in_state[state_idx] != -1:
            pers_2[last_action_2_in_state[state_idx]] = p2
            
        logits_2 = beta * q_stage2_mf[state_idx] + pers_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # counterfactual update for the unchosen action in the same state
        unchosen_action_2 = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_action_2]
        q_stage2_mf[state_idx, unchosen_action_2] += alpha_cf * delta_cf
        
        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model where the model-based weight (w) is dependent on the reward of the previous trial.
    This captures the behavioral pattern where participants might rely more on model-based planning 
    after a successful outcome compared to an unsuccessful one (or vice versa).
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w_win: [0,1] - Model-based weight following a rewarded trial
    w_lose: [0,1] - Model-based weight following an unrewarded trial
    lam: [0,1] - Eligibility trace
    p1: [-5,5] - Stage 1 choice perseveration
    p2: [-5,5] - Stage 2 choice perseveration
    """
    alpha, beta, w_win, w_lose, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.array([-1, -1])
    last_reward = 1 # assume win for the first trial

    for trial in range(n_trials):
        w = w_win if last_reward == 1 else w_lose
        
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        logits_2 = beta * q_stage2_mf[state_idx] + pers_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with choice perseveration, Q-value decay, and 'planet perseveration'. 
    Planet perseveration adds a bias to select the stage 1 action that commonly leads 
    to the planet visited on the previous trial, capturing spatial/state stickiness.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based vs model-free weight
    lam: [0,1] - Eligibility trace decay
    p1: [-5,5] - Stage 1 choice perseveration
    p2: [-5,5] - Stage 2 choice perseveration
    decay: [0,1] - Forgetting rate of unchosen Q-values
    p_planet: [-5,5] - Planet perseveration (bonus for the spaceship leading to the last visited planet)
    """
    alpha, beta, w, lam, p1, p2, decay, p_planet = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.array([-1, -1])
    last_state = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] += p1
            
        if last_state != -1:
            # Action 0 commonly leads to state 0, Action 1 commonly leads to state 1
            pers_bonus_1[last_state] += p_planet
            
        logits_1 = beta * q_stage1_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # policy for the second choice
        state_idx = state[trial]
        pers_bonus_2 = np.zeros(2)
        if last_action_2_in_state[state_idx] != -1:
            pers_bonus_2[last_action_2_in_state[state_idx]] = p2
            
        logits_2 = beta * q_stage2_mf[state_idx] + pers_bonus_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # eligibility trace
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen options
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
