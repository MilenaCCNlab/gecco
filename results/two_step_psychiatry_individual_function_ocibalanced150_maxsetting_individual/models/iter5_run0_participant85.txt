import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition learning.
    
    This model assumes that participants do not use a fixed transition matrix, 
    but instead dynamically learn the transition probabilities from experience 
    as they observe the states that follow their first-stage choices.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi: [-5, 5] - Choice stickiness (perseveration) parameter.
    alpha_T: [0, 1] - Learning rate for the transition matrix.
    """
    alpha, beta1, beta2, w, pi, alpha_T = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            q1_vals[last_action] += pi
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Transition matrix update
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_T * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_T * (0.0 - transition_matrix[a1, 1 - state_idx])

        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Reward-dependent stickiness and decay.
    
    This model separates the choice stickiness into two parameters depending on 
    whether the previous trial was rewarded (win-stay) or unrewarded (lose-shift). 
    It also includes a decay mechanism for unchosen option values to capture forgetting.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi_win: [-5, 5] - Choice stickiness if the last trial was rewarded.
    pi_lose: [-5, 5] - Choice stickiness if the last trial was unrewarded.
    decay: [0, 1] - Decay rate for unchosen options towards 0.5.
    """
    alpha, beta1, beta2, w, pi_win, pi_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net
        if last_action != -1:
            if last_reward == 1:
                q1_vals[last_action] += pi_win
            else:
                q1_vals[last_action] += pi_lose
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] += decay * (0.5 - q_stage2_mf[other_state, 0])
        q_stage2_mf[other_state, 1] += decay * (0.5 - q_stage2_mf[other_state, 1])
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Continuous choice trace.
    
    Instead of only perseverating on the immediately preceding choice, this model 
    maintains a continuous choice trace for the first-stage actions. This captures 
    a longer-term tendency to repeat frequently chosen actions, which decays over time.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    beta1: [0, 10] - Inverse temperature for first-stage softmax.
    beta2: [0, 10] - Inverse temperature for second-stage softmax.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pi: [-5, 5] - Weight of the choice trace on first-stage choices.
    tau_pi: [0, 1] - Decay rate of the choice trace (1 = no decay, 0 = immediate decay).
    """
    alpha, beta1, beta2, w, pi, tau_pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_vals = beta1 * q_stage1_net + pi * choice_trace
            
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_vals = beta2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Update choice trace
        choice_trace *= tau_pi
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
