def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with eligibility trace and stage 1 perseveration.
    This model adds a perseveration parameter to the baseline hybrid model, which 
    captures the participant's tendency to repeat the previous stage 1 choice 
    regardless of the reward outcome.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers: perseveration bonus for stage 1 [-5, 5]
    """
    alpha, beta, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        state_idx = state[trial]
        a1 = action_1[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseveration bonus to the previously chosen action
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = pers
            
        q_net_total = q_net + pers_bonus
        
        exp_q1 = np.exp(beta * q_net_total - np.max(beta * q_net_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        if a2 != -1:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # Action value updating
        if a2 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2
            
            q_stage1_mf[a1] += alpha * lam * delta_stage2
            
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with eligibility trace and separate learning rates.
    This model assumes that participants might update their value representations 
    at different speeds for the first-stage (spaceships) vs the second-stage (aliens).
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2: learning rate for stage 2 [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    """

        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            p_choice_2[trial] = 1.0

        # Action value updating
        if a2 != -1:

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
            
            # Eligibility trace uses alpha_1 because it's updating the stage 1 value
