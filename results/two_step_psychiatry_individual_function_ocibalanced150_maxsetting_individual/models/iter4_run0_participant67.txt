Here are 3 new cognitive models that build upon the successful mechanisms (like forgetting/decay) by introducing new cognitive hypotheses to explain the participant's choices.

### Model 1: Asymmetric Learning Rates (Pos/Neg) + Decay
This model hypothesizes that the participant learns differently from positive versus negative prediction errors, while still exhibiting forgetting of unchosen options. Combining asymmetric learning with decay often captures distinct biases in how rewards versus omissions are integrated.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Asymmetric Learning and Forgetting (Decay).
    Learning rates differ for positive and negative prediction errors.
    Unchosen Q-values decay towards 0.
    
    Bounds:
    alpha_pos: [0, 1] (Learning rate for positive PE)
    alpha_neg: [0, 1] (Learning rate for negative PE)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi: [-5, 5] (Choice perseveration)
    decay: [0, 1] (Forgetting rate for unchosen actions)
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1
            
        logits_1 = beta1 * q_net + pi * persev
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # Eligibility trace update for Choice 1
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Choice Trace (Decaying Perseveration)
Instead of just perseverating on the *immediately* preceding action, this model assumes the participant builds up a "choice trace" (a habit) over time. This trace decays gradually, causing a stronger perseverative pull towards actions that have been chosen repeatedly in recent trials.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Choice Trace.
    Instead of just the last action, perseveration is based on a decaying trace of past choices.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi: [-5, 5] (Choice trace weight)
    trace_decay: [0, 1] (Decay rate of the choice trace)
    """
    alpha, beta1, beta2, w, lam, pi, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
                    
        logits_1 = beta1 * q_net + pi * choice_trace
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for Choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update choice trace (bounded between 0 and 1)
        choice_trace *= (1 - trace_decay)
        choice_trace[action_1[trial]] += trace_decay

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Forgetting
This model proposes that the extent to which unchosen options are forgotten (decay) depends on the outcome of the current trial. For example, receiving a reward might narrow attention to the chosen option, causing faster forgetting of alternatives, whereas a lack of reward might prompt broader retention of counterfactual options.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free with Reward-Dependent Forgetting.
    Unchosen Q-values decay towards 0, but the decay rate depends on whether the trial was rewarded.
    
    Bounds:
    alpha: [0, 1] (Learning rate)
    beta1: [0, 10] (Inverse temperature for stage 1)
    beta2: [0, 10] (Inverse temperature for stage 2)
    w: [0, 1] (Model-based weight)
    lam: [0, 1] (Eligibility trace)
    pi: [-5, 5] (Choice perseveration)
    decay_win: [0, 1] (Forgetting rate for unchosen actions when rewarded)
    decay_lose: [0, 1] (Forgetting rate for unchosen actions when unrewarded)
    """
    alpha, beta1, beta2, w, lam, pi, decay_win, decay_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev = np.zeros(2)
        if last_action != -1:
            persev[last_action] = 1
            
        logits_1 = beta1 * q_net + pi * persev
        logits_1 -= np.max(logits_1) # for numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        # policy for the second choice
        state_idx = state[trial]
        logits_2 = beta2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Eligibility trace update for Choice 1
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Reward-dependent decay for unchosen
        decay = decay_win if reward[trial] > 0 else decay_lose
        
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
