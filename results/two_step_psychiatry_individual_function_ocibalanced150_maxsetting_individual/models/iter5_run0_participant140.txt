Here are 3 new cognitive models for the two-step decision-making task. They explore different cognitive mechanisms that expand upon the baseline models, including updating the internal transition model, outcome-dependent perseveration, and asymmetric learning for positive versus negative prediction errors.

### Model 1: Transition Matrix Learning
While participants are told the common and rare transition probabilities, they might not fully trust these instructions or might dynamically update their internal model of the environment based on recent experiences. This model includes an `alpha_trans` parameter that learns the transition matrix on a trial-by-trial basis.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition Matrix Learning
    
    Participants may dynamically update their internal model of the transition 
    probabilities between spaceships and planets based on their experience, rather 
    than assuming they are fixed. This model updates the transition matrix on each trial.
    
    Bounds:
    alpha: [0, 1] - Q-value learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1 choices
    beta_2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weight on model-based vs model-free values
    lam: [0, 1] - Eligibility trace for stage 1 MF update
    pers_1: [-5, 5] - Choice perseveration bonus for stage 1
    pers_2: [-5, 5] - Choice perseveration bonus for stage 2
    alpha_trans: [0, 1] - Learning rate for the transition matrix
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix to the instructed probabilities
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        # Stage 2 choice
        state_idx = state[trial]
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Transition matrix update (keeps sum to 1 for each action)
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Outcome-Dependent Perseveration (Win-Stay / Lose-Shift) with Unchosen Decay
This model modifies the perseveration mechanism by distinguishing between previously rewarded and unrewarded trials (WSLS heuristic). It retains the highly effective `unchosen_decay` mechanism for unexperienced options.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Outcome-Dependent Perseveration (WSLS) with Unchosen Decay
    
    Replaces simple perseveration with Win-Stay/Lose-Shift tendencies for Stage 1.
    If the previous trial was rewarded, a 'win-stay' bonus is applied.
    If unrewarded, a 'lose-shift' bonus (or penalty) is applied.
    It also includes decay for unchosen actions towards 0.5.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based vs model-free weight
    lam: [0, 1] - Eligibility trace
    pers_win: [-5, 5] - Stage 1 perseveration bonus if previous trial rewarded
    pers_lose: [-5, 5] - Stage 1 perseveration bonus if previous trial unrewarded
    unchosen_decay: [0, 1] - Decay rate for unchosen actions towards 0.5
    """
    alpha, beta_1, beta_2, w, lam, pers_win, pers_lose, unchosen_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus_1[last_action_1] = pers_win
            else:
                pers_bonus_1[last_action_1] = pers_lose
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Unchosen decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += unchosen_decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += unchosen_decay * (0.5 - q_stage2_mf[s, a])

        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates with Unchosen Decay
This model posits that participants update their Q-values differently for positive versus negative reward prediction errors (RPEs), which is a common finding in human reinforcement learning. It preserves the unchosen decay and stage 1 perseveration.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Learning Rates with Unchosen Decay
    
    Uses different learning rates depending on the valence of the Reward Prediction 
    Error (positive vs negative RPE). Includes unchosen decay and stage 1 perseveration.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive RPEs
    alpha_neg: [0, 1] - Learning rate for negative RPEs
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    pers_1: [-5, 5] - Choice perseveration bonus for stage 1
    unchosen_decay: [0, 1] - Decay rate for unchosen actions towards 0.5
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers_1, unchosen_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]

        state_idx = state[trial]
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Unchosen decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += unchosen_decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += unchosen_decay * (0.5 - q_stage2_mf[s, a])

        # Stage 1 Update with Asymmetric LR
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        
        # Stage 2 Update with Asymmetric LR
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        # Eligibility trace update for Stage 1
        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
