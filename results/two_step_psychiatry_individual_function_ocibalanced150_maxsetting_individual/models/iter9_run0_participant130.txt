Here are 3 new cognitive models, expressed as Python functions, that explore alternative mechanisms for learning and perseveration without reusing the exact parameter combinations previously tried. 

### Model 1: Hybrid MB/MF with Choice Traces and Q-value Decay
This model replaces the simple "last choice" stickiness with an exponentially updating **choice trace** (controlled by parameter `alpha_c`). Given the participant's tendency to stick to the same spaceship for long streaks (e.g., trials 1-79, then 80-116), a choice trace allows perseverative tendencies to build up over time and decay slowly, rather than resetting immediately upon a single switch.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Hybrid MB/MF with choice trace for perseveration and Q-value decay.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updating.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 prediction error.
    rho_1: [0,10] - Choice trace weight for stage 1 actions.
    rho_2: [0,10] - Choice trace weight for stage 2 actions.
    alpha_c: [0,1] - Choice trace update rate.
    decay: [0,1] - Decay rate for all Q-values towards 0.5 before updating.
    """
    alpha, beta, w, lam, rho_1, rho_2, alpha_c, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        v1 = beta * q_net + rho_1 * trace_1
        exp_q1 = np.exp(v1 - np.max(v1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        v2 = beta * q_stage2_mf[state_idx, :] + rho_2 * trace_2[state_idx, :]
        exp_q2 = np.exp(v2 - np.max(v2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values
        q_stage1_mf = (1 - decay) * q_stage1_mf + decay * 0.5
        q_stage2_mf = (1 - decay) * q_stage2_mf + decay * 0.5

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update choice traces
        trace_1 = (1 - alpha_c) * trace_1
        trace_1[action_1[trial]] += alpha_c
        
        trace_2 = (1 - alpha_c) * trace_2
        trace_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid MB/MF with Temporal Discounting (`gamma`) and Q-value Decay
This model introduces a temporal discount factor `gamma` for Stage 2 values. Sometimes participants discount the value of the second stage when making their first-stage decisions, which scales down the impact of Stage 2 values on both Model-Based and Model-Free Stage 1 estimations.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Hybrid MB/MF with discount factor gamma for stage 2 values and Q-value decay.
    
    Parameters:
    alpha: [0,1] - Learning rate for Q-value updating.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 prediction error.
    rho_1: [0,10] - Choice stickiness parameter for stage 1 actions.
    rho_2: [0,10] - Choice stickiness parameter for stage 2 actions, conditional on the state.
    gamma: [0,1] - Discount factor for stage 2 values.
    decay: [0,1] - Decay rate for all Q-values towards 0.5 before updating.
    """
    alpha, beta, w, lam, rho_1, rho_2, gamma, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_choice_1 = -1
    last_choice_2 = [-1, -1] 

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * gamma * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_choice_1 != -1:
            stickiness_1[last_choice_1] = 1.0
            
        v1 = beta * q_net + rho_1 * stickiness_1
        exp_q1 = np.exp(v1 - np.max(v1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_choice_2[state_idx] != -1:
            stickiness_2[last_choice_2[state_idx]] = 1.0

        v2 = beta * q_stage2_mf[state_idx, :] + rho_2 * stickiness_2
        exp_q2 = np.exp(v2 - np.max(v2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Decay Q-values
        q_stage1_mf = (1 - decay) * q_stage1_mf + decay * 0.5
        q_stage2_mf = (1 - decay) * q_stage2_mf + decay * 0.5

        # Action value updating
        delta_stage1 = gamma * q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_choice_1 = action_1[trial]
        last_choice_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid MB/MF with Asymmetric Learning Rates and Choice Traces
This model tests whether the participant learns differently from positive vs. negative prediction errors (e.g., updating beliefs faster after a reward than after an omission), while also utilizing the choice traces (`alpha_c`) to explain the long-lasting behavioral stickiness. 

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Hybrid MB/MF with separate learning rates for positive/negative RPEs and choice traces.
    
    Parameters:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting between model-based (1) and model-free (0) strategies.
    lam: [0,1] - Eligibility trace parameter for updating stage 1 with stage 2 prediction error.
    rho_1: [0,10] - Choice trace weight for stage 1 actions.
    rho_2: [0,10] - Choice trace weight for stage 2 actions.
    alpha_c: [0,1] - Choice trace update rate.
    """
    alpha_pos, alpha_neg, beta, w, lam, rho_1, rho_2, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        v1 = beta * q_net + rho_1 * trace_1
        exp_q1 = np.exp(v1 - np.max(v1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        v2 = beta * q_stage2_mf[state_idx, :] + rho_2 * trace_2[state_idx, :]
        exp_q2 = np.exp(v2 - np.max(v2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        
        # Update choice traces
        trace_1 = (1 - alpha_c) * trace_1
        trace_1[action_1[trial]] += alpha_c
        
        trace_2 = (1 - alpha_c) * trace_2
        trace_2[state_idx, action_2[trial]] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
