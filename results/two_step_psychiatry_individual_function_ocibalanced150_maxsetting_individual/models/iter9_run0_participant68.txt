Here are 3 new cognitive models that explore different theoretical mechanisms for how participants might learn and make decisions in this two-step task. They avoid the previously tried parameter combinations and introduce novel behavioral dynamics.

### Model 1: Factorial Perseveration (Reward & Transition Dependent)
This model decomposes the perseveration at Stage 1 into four distinct parameters based on the outcome of the previous trial. It directly models the classic "win-stay/lose-shift" factorial analysis often used in two-step tasks, evaluating how previous rewards (win vs. loss) and transitions (common vs. rare) independently affect the tendency to repeat the same Stage 1 choice. 

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Reward and Transition-dependent Perseveration.
    This model estimates 4 separate perseveration parameters for Stage 1 choices,
    depending on the previous trial's outcome (win/loss) and transition (common/rare).
    
    Parameters:
    alpha (float): Learning rate for both stages. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers_cw (float): Perseveration after common win. Bounds: [-5,5]
    pers_cl (float): Perseveration after common loss. Bounds: [-5,5]
    pers_rw (float): Perseveration after rare win. Bounds: [-5,5]
    pers_rl (float): Perseveration after rare loss. Bounds: [-5,5]
    """
    alpha, beta, w, lam, pers_cw, pers_cl, pers_rw, pers_rl = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0
    last_trans_common = False

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Apply factorial perseveration bonus
        if last_action1 != -1:
            if last_reward == 1 and last_trans_common:
                q_net1[last_action1] += pers_cw
            elif last_reward == 0 and last_trans_common:
                q_net1[last_action1] += pers_cl
            elif last_reward == 1 and not last_trans_common:
                q_net1[last_action1] += pers_rw
            elif last_reward == 0 and not last_trans_common:
                q_net1[last_action1] += pers_rl
                
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_net2 = q_stage2_mf[state_idx].copy()
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Store for next trial
        last_action1 = action_1[trial]
        last_reward = reward[trial]
        last_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Decaying Choice Trace
Instead of treating perseveration as a rigid 1-trial lookback, this model assumes participants maintain a decaying "choice trace" (habituation). This allows the tendency to repeat choices to build up over repeated selections and gradually decay when an option is abandoned. 

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Decaying Choice Trace (Perseveration).
    Maintains a choice trace that updates over time, allowing perseveration 
    to build up over repeated choices.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers (float): Perseveration weight. Bounds: [-5,5]
    alpha_c (float): Learning rate for the choice trace. Bounds: [0,1]
    """
    alpha1, alpha2, beta, w, lam, pers, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace1 = np.zeros(2)
    choice_trace2 = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Policy for first choice includes choice trace
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf + pers * choice_trace1
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for second choice includes choice trace
        q_net2 = q_stage2_mf[state_idx].copy() + pers * choice_trace2[state_idx]
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update choice traces
        choice_trace1 = (1 - alpha_c) * choice_trace1
        choice_trace1[action_1[trial]] += alpha_c
        
        choice_trace2 = (1 - alpha_c) * choice_trace2
        choice_trace2[state_idx, action_2[trial]] += alpha_c

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-dependent Eligibility Trace
This model introduces the theory that participants assign credit differently when they experience an expected vs. an unexpected event. By estimating separate eligibility traces (`lam_c` and `lam_r`), we can observe if rare transitions cause participants to "doubt" the causal link and inhibit the flow of model-free credit back to Stage 1.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-dependent Eligibility Trace.
    Uses different eligibility traces (lam_c and lam_r) depending on whether 
    the transition was common or rare, reflecting altered credit assignment 
    when experiencing surprising transitions.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam_c (float): Eligibility trace for common transitions. Bounds: [0,1]
    lam_r (float): Eligibility trace for rare transitions. Bounds: [0,1]
    pers (float): Perseveration for both stages. Bounds: [-5,5]
    """
    alpha1, alpha2, beta, w, lam_c, lam_r, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Select lambda based on transition type
        is_common = (action_1[trial] == state_idx)
        lam_current = lam_c if is_common else lam_r
        
        q_stage1_mf[action_1[trial]] += alpha1 * lam_current * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
