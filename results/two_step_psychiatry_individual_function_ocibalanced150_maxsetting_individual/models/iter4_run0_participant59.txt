Here are three new cognitive models based on the findings from the best model so far. Since the combination of value decay with the standard hybrid model framework worked well, these models build on that foundation by introducing new, distinct mechanisms that could further explain the participant's behavior.

### Model 1: Value Decay + Stage 1 Choice Bias
Looking at the participant data, there are long blocks where the participant repeatedly selects spaceship 0, indicating a potential baseline preference or lateral bias for that action. This model introduces a `bias` parameter to capture any innate preference for action 0 over action 1 at the first stage.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid model with value decay, perseveration, and a stage 1 choice bias.
    The bias parameter captures any baseline preference for spaceship 0 over spaceship 1.
    
    Bounds:
    alpha: [0, 1] - Learning rate for chosen updates.
    decay: [0, 1] - Decay rate for unchosen action values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseveration parameter for previous stage 1 choice.
    bias: [-5, 5] - Baseline preference (bias) for action 0 at stage 1.
    """
    alpha, decay, beta_1, beta_2, w, lam, p, bias = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add perseveration
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += p
            
        # Add bias for action 0
        q_stage1_net[0] += bias
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == state_idx and a_idx == action_2[trial]):
                    q_stage2_mf[s_idx, a_idx] *= (1 - decay)
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Value Decay + Reward-Dependent MB Control
This model tests the hypothesis that the participant's engagement with the computationally demanding model-based (MB) system fluctuates depending on recent success. It fits separate model-based weights (`w_win`, `w_lose`) depending on whether the participant received a reward on the previous trial, combined with value decay.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with value decay, perseveration, and reward-dependent MB weighting.
    The weight on model-based control (w) differs based on whether the previous trial was rewarded.
    
    Bounds:
    alpha: [0, 1] - Learning rate.
    decay: [0, 1] - Decay rate for unchosen action values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w_win: [0, 1] - MB weight after a rewarded trial.
    w_lose: [0, 1] - MB weight after an unrewarded trial.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseveration parameter for stage 1.
    """
    alpha, decay, beta_1, beta_2, w_win, w_lose, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        w = w_win if last_reward > 0 else w_lose
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += p
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == state_idx and a_idx == action_2[trial]):
                    q_stage2_mf[s_idx, a_idx] *= (1 - decay)
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Value Decay + Learned Transition Probabilities
Instead of assuming the participant fully trusts and utilizes the fixed `0.7/0.3` transition probabilities forever, this model allows the participant to update their internal model of the environment's transition matrix dynamically based on the observed transitions, combined with value decay. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with value decay and learned transition probabilities.
    Instead of fixing the transition matrix to 0.7/0.3, the model updates its 
    estimate of the transition probabilities dynamically based on observations.
    
    Bounds:
    alpha: [0, 1] - Learning rate for action values.
    decay: [0, 1] - Decay rate for unchosen action values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weight on model-based vs model-free control.
    lam: [0, 1] - Eligibility trace decay parameter.
    p: [-5, 5] - Perseveration parameter for stage 1.
    alpha_trans: [0, 1] - Learning rate for updating transition probabilities.
    """
    alpha, decay, beta_1, beta_2, w, lam, p, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize with instructed 0.7/0.3 prior, but allow it to be updated
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        if last_action_1 != -1:
            q_stage1_net[last_action_1] += p
            
        q1_scaled = beta_1 * q_stage1_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        q2_scaled = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update transition probabilities
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == state_idx and a_idx == action_2[trial]):
                    q_stage2_mf[s_idx, a_idx] *= (1 - decay)
                    
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
