Here are three new cognitive models that build upon the best-performing model so far by introducing new choice and learning mechanisms. Each model stays within the limit of 8 parameters while capturing subtle patterns in participant behavior.

### Model 1: State Stickiness Model
Participants often exhibit a tendency to return to the planet (state) they visited on the previous trial, irrespective of the reward received. This model introduces a `state_stick` parameter to capture the tendency to choose the stage 1 action that commonly leads to the previously visited stage 2 state.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Mixed MF/MB with reward-dependent stickiness and state stickiness.
    Adds a parameter for the tendency to choose the action that commonly leads to the previous state.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0, 1] - Eligibility trace (how much stage 2 reward updates stage 1 MF value).
    rep_win: [-5, 5] - Stage 1 choice stickiness if previous trial was rewarded.
    rep_lose: [-5, 5] - Stage 1 choice stickiness if previous trial was unrewarded.
    state_stick: [-5, 5] - Tendency to choose the action commonly leading to the previous stage 2 state.
    """
    alpha, beta_1, beta_2, w, lam, rep_win, rep_lose, state_stick = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_state = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness[last_action_1] += rep_win
            elif last_reward == 0:
                stickiness[last_action_1] += rep_lose
                
        if last_state != -1:
            # Action 0 commonly leads to State 0; Action 1 commonly leads to State 1
            stickiness[last_state] += state_stick
                
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Transition-Dependent Win-Stay Model
A key signature of the two-step task is how participants react to rewards following a rare transition. This model breaks `rep_win` into two components based on whether the rewarded trial involved a common or rare transition, which helps capture heuristic behaviors better than a static model-based weight alone.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Mixed MF/MB with transition-dependent win-stay.
    Separates win-stay stickiness based on whether the previous transition was common or rare.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    rep_win_common: [-5, 5] - Stage 1 stickiness if prev trial was rewarded and transition was common.
    rep_win_rare: [-5, 5] - Stage 1 stickiness if prev trial was rewarded and transition was rare.
    rep_lose: [-5, 5] - Stage 1 stickiness if prev trial was unrewarded.
    """
    alpha, beta_1, beta_2, w, lam, rep_win_common, rep_win_rare, rep_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_trans_common = False

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                if last_trans_common:
                    stickiness[last_action_1] = rep_win_common
                else:
                    stickiness[last_action_1] = rep_win_rare
            elif last_reward == 0:
                stickiness[last_action_1] = rep_lose
                
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Loss Penalty with Win/Lose Stickiness
Instead of treating 0 coins as simply a null reward, this model treats unrewarded trials as a subjective loss, which heavily penalizes the associated Q-values and forces unlearning. This is paired with the best-performing win/lose stickiness framework.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Mixed MF/MB with reward-dependent stickiness and subjective loss penalty.
    Unrewarded trials are treated as a negative reward (loss_penalty) rather than 0, 
    accelerating unlearning for unrewarded actions.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight.
    lam: [0, 1] - Eligibility trace.
    rep_win: [-5, 5] - Stage 1 stickiness if previous trial was rewarded.
    rep_lose: [-5, 5] - Stage 1 stickiness if previous trial was unrewarded.
    loss_penalty: [0, 5] - Effective negative reward applied when receiving 0 coins.
    """
    alpha, beta_1, beta_2, w, lam, rep_win, rep_lose, loss_penalty = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness[last_action_1] = rep_win
            elif last_reward == 0:
                stickiness[last_action_1] = rep_lose
                
        q1_logits = beta_1 * q_stage1_net + stickiness
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_logits = beta_2 * q_stage2_mf[state_idx]
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Use effective reward: 1 for reward, -loss_penalty for no reward
        eff_reward = 1.0 if reward[trial] == 1 else -loss_penalty

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
