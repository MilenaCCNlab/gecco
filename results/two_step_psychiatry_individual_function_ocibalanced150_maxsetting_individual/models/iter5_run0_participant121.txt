Here are three new cognitive models based on the Daw et al. (2011) framework, incorporating different variations of the active decay (forgetting) mechanism for unchosen actions. 

### Model 1: Win-Stay / Lose-Shift Stickiness + Decay
This model modifies choice stickiness by separating it into two components: `p_win` and `p_lose`. Participants might be more likely to repeat an action if it was recently rewarded (win-stay) and less likely to repeat it if it was unrewarded (lose-shift).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. (2011) model with win-stay/lose-shift stickiness and decay for unchosen actions.
    Choice stickiness at stage 1 depends on whether the previous trial was rewarded.
    
    Bounds:
    learning_rate: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p_win: [-5,5] - Choice stickiness if previous trial was rewarded
    p_lose: [-5,5] - Choice stickiness if previous trial was unrewarded
    decay: [0,1] - Decay rate for unchosen action values
    """
    learning_rate, beta, w, lambda_, p_win, p_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            if last_reward == 1:
                q_net[last_action_1] += p_win
            elif last_reward == 0:
                q_net[last_action_1] += p_lose
                
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        # Decay for unchosen actions in stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # Decay for unchosen actions in stage 1
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Separate Learning Rates for Stage 1 and Stage 2 + Decay
Participants often learn at different rates during different stages of a task. This model introduces independent learning rates (`alpha_1` and `alpha_2`) for Stage 1 and Stage 2 prediction errors, combined with the active decay mechanism.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. (2011) model with separate learning rates for stage 1 and stage 2, 
    along with decay for unchosen actions.
    
    Bounds:
    alpha_1: [0,1] - Learning rate for stage 1
    alpha_2: [0,1] - Learning rate for stage 2
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p: [0,5] - Choice stickiness
    decay: [0,1] - Decay rate for unchosen action values
    """
    alpha_1, alpha_2, beta, w, lambda_, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2
        
        # Decay for unchosen actions in stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # Decay for unchosen actions in stage 1
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Decay Towards an Informative Prior (Center Value)
Standard decay mechanisms push unchosen action values towards zero. However, in probabilistic environments, participants might regress unchosen values towards a prior mean (e.g., 0.5) to reflect environmental uncertainty, rather than assuming an unchosen action yields zero reward.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Daw et al. (2011) model where unchosen actions decay towards a specific center value 
    (e.g., prior mean) rather than strictly towards zero.
    
    Bounds:
    learning_rate: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based weight
    lambda_: [0,1] - Eligibility trace
    p: [0,5] - Choice stickiness
    decay: [0,1] - Decay rate for unchosen action values
    decay_center: [0,1] - The value towards which unchosen actions decay
    """
    learning_rate, beta, w, lambda_, p, decay, decay_center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to decay_center
    q_stage1_mf = np.full(2, decay_center)
    q_stage2_mf = np.full((2, 2), decay_center)
    
    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action_1 != -1:
            q_net[last_action_1] += p
            
        exp_q1 = np.exp(beta * q_net - np.max(beta * q_net))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] - np.max(beta * q_stage2_mf[state_idx]))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_ * delta_stage2
        
        # Decay towards center value for unchosen actions in stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + decay_center * decay

        # Decay towards center value for unchosen actions in stage 1
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] = q_stage1_mf[unchosen_a1] * (1 - decay) + decay_center * decay
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
