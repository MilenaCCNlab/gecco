import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Counterfactual Updating with Decay
    When a reward is revealed for the chosen alien, the participant might also counterfactually 
    update the unchosen alien on the same planet, assuming anti-correlated reward probabilities.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for chosen actions.
    decay_rate: [0, 1] - Decay rate for Q-values towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax.
    w: [0, 1] - Weight on MB vs MF.
    lam: [0, 1] - Eligibility trace.
    p: [-5, 5] - Stage 1 choice perseveration.
    cf_weight: [0, 1] - Weight for counterfactual updating of the unchosen alien on the visited planet.
    """
    learning_rate, decay_rate, beta, w, lam, p, cf_weight = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Counterfactual update for unchosen alien
        unchosen_a2 = 1 - action_2[trial]
        cf_delta_stage2 = (1 - reward[trial]) - q_stage2_mf[state_idx, unchosen_a2]

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2
        new_q2_unchosen = q_stage2_mf[state_idx, unchosen_a2] + learning_rate * cf_weight * cf_delta_stage2

        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen
        q_stage2_mf[state_idx, unchosen_a2] = new_q2_unchosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric Eligibility Trace Model
    Participants might credit stage 1 actions differently depending on whether the stage 2 
    outcome was a positive or negative prediction error.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values.
    decay_rate: [0, 1] - Decay rate for unchosen Q-values towards 0.5.
    beta: [0, 10] - Inverse temperature for softmax.
    w: [0, 1] - Weight on MB vs MF.
    lam_pos: [0, 1] - Eligibility trace for positive stage 2 prediction errors.
    lam_neg: [0, 1] - Eligibility trace for negative stage 2 prediction errors.
    p: [-5, 5] - Choice perseveration for stage 1.
    """
    learning_rate, decay_rate, beta, w, lam_pos, lam_neg, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        lam = lam_pos if delta_stage2 >= 0 else lam_neg

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2

        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
