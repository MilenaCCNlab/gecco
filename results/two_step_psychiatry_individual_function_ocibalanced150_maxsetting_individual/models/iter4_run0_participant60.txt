Here are 3 new cognitive models that explore different mechanisms for the two-step task. Each model is implemented as a standalone Python function and builds upon the best model so far by introducing novel dynamics (forgetting/decay, independent decision temperatures, and stage 2 perseveration).

### Model 1: Q-Value Decay (Forgetting)
This model introduces a `decay` parameter. In standard Q-learning, unchosen options retain their values indefinitely until they are chosen again. A decay parameter simulates memory decay (or forgetting), where the Q-values of unchosen actions gradually decay towards zero. This can help explain why participants might occasionally explore options they haven't chosen in a while.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) model with unchosen Q-value decay (forgetting).
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    decay: [0,1] - Forgetting rate for unchosen actions, decaying towards 0.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting of model-based vs model-free values at stage 1.
    lam: [0,1] - Eligibility trace parameter.
    perseveration: [-5,5] - Choice stickiness for stage 1.
    """
    import numpy as np
    learning_rate, decay, beta, w, lam, perseveration = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = perseveration
            
        logits_1 = beta * q_net + pers_bonus
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Unchosen decay before update
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # Chosen updates
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Independent Temperatures for MB and MF Systems
Rather than using a single inverse temperature (`beta`) and a weighting parameter (`w`), this model assigns independent inverse temperatures to the model-based and model-free values at stage 1, and an independent inverse temperature for stage 2. This assumes that exploration vs. exploitation might be governed differently within the two distinct learning systems.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Independent temperatures for MB and MF systems, plus stage 2 temperature.
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta_mb: [0,10] - Inverse temperature for model-based system.
    beta_mf: [0,10] - Inverse temperature for model-free system at stage 1.
    beta_stage2: [0,10] - Inverse temperature for stage 2 choices.
    lam: [0,1] - Eligibility trace parameter.
    perseveration: [-5,5] - Choice stickiness for stage 1.
    """
    import numpy as np
    learning_rate, beta_mb, beta_mf, beta_stage2, lam, perseveration = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus = np.zeros(2)
        if last_action != -1:
            pers_bonus[last_action] = perseveration
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_stage2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Stage 1 and Stage 2 Perseveration
Choice stickiness is typically modeled only at the first stage (spaceship choice). However, participants might also exhibit a habitual tendency to stick with the same alien they last chose when revisiting a specific planet. This model introduces a state-specific `pers_stage2` parameter to capture this secondary habituation.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) model with perseveration at both stage 1 and stage 2.
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-value updates.
    beta: [0,10] - Inverse temperature for softmax action selection.
    w: [0,1] - Weighting of model-based vs model-free values at stage 1.
    lam: [0,1] - Eligibility trace parameter.
    pers_stage1: [-5,5] - Choice stickiness for stage 1.
    pers_stage2: [-5,5] - Choice stickiness for stage 2 (state-specific).
    """
    import numpy as np
    learning_rate, beta, w, lam, pers_stage1, pers_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1]) # Tracks last stage 2 action for planet 0 and planet 1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_stage1
            
        logits_1 = beta * q_net + pers_bonus_1
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_stage2

        logits_2 = beta * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
