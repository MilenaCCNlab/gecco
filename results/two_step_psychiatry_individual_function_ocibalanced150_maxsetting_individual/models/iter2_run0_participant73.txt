def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Full Daw et al. 2011 model with both general stickiness and reward stickiness.
    This model includes a model-based/model-free mixture, an eligibility trace (lambda),
    and choice perseverance mechanisms (general stickiness and win-stay/lose-shift).
    Bounds:
    learning_rate: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature for softmax choice
    w: [0,1] - Weight of model-based vs model-free control
    lambda_trace: [0,1] - Eligibility trace parameter
    stickiness: [-5,5] - General choice perseverance (tendency to repeat previous choice)
    reward_stickiness: [-5,5] - Additional perseverance if the previous trial was rewarded
    """
    learning_rate, beta, w, lambda_trace, stickiness, reward_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stick_vector = np.zeros(2)
        if last_action != -1:
            stick_vector[last_action] = 1
            
        reward_stick_vector = np.zeros(2)
        if last_action != -1 and last_reward == 1:
            reward_stick_vector[last_action] = 1
            
        exp_q1 = np.exp(beta * q_net + stickiness * stick_vector + reward_stickiness * reward_stick_vector)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx, :])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Eligibility trace update for choice 1 from stage 2 prediction error
        q_stage1_mf[action_1[trial]] += lambda_trace * learning_rate * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF mixture with separate learning rates for positive and negative prediction errors.
    This model captures the asymmetry in learning when outcomes are better vs. worse than expected.
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    beta: [0,10] - Inverse temperature for softmax choice
    w: [0,1] - Weight of model-based vs model-free control
    """
    alpha_pos, alpha_neg, beta, w = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx, :])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF mixture with transition probability learning and separate stage temperatures.
    Participants may not perfectly trust the instructed transition probabilities (0.7/0.3) and
    can update them based on their direct experience.
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-values
    beta1: [0,10] - Inverse temperature for stage 1 choice
    beta2: [0,10] - Inverse temperature for stage 2 choice
    w: [0,1] - Weight of model-based vs model-free control
    alpha_trans: [0,1] - Learning rate for updating the transition probability matrix
    """
    learning_rate, beta1, beta2, w, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transitions as instructed, but they will be learned dynamically
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        exp_q1 = np.exp(beta1 * q_net)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx, :])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # update subjective transition matrix based on observed state transition
        a1 = action_1[trial]
        s2 = state_idx
        transition_matrix[a1, s2] += alpha_trans * (1.0 - transition_matrix[a1, s2])
        transition_matrix[a1, 1 - s2] += alpha_trans * (0.0 - transition_matrix[a1, 1 - s2])
        
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
