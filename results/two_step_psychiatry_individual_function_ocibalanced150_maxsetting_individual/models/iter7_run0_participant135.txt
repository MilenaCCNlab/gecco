import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with positive/negative prediction error learning rates, 
    and perseveration on both stage 1 (reward-dependent) and stage 2.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pers_rew: [-5, 5] - Stage 1 perseveration bonus if previous trial was rewarded.
    pers_unrew: [-5, 5] - Stage 1 perseveration bonus if previous trial was unrewarded.
    pers2: [-5, 5] - Stage 2 perseveration bonus for repeating the same stage 2 action.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, pers_rew, pers_unrew, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0
    last_action2 = np.array([-1, -1]) # Keeps track of last stage 2 action for each state

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                pers_bonus1[last_action1] = pers_rew
            else:
                pers_bonus1[last_action1] = pers_unrew
            
        logits1 = beta1 * q_net1 + pers_bonus1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        pers_bonus2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            pers_bonus2[last_action2[state_idx]] = pers2

        logits2 = beta2 * q_stage2_mf[state_idx] + pers_bonus2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with learned transition matrix, along with stage 1 and stage 2 perseveration.
    Instead of fixed 0.7/0.3 transitions, the participant updates transition probabilities dynamically.
    
    Bounds:
    alpha: [0, 1] - Learning rate for Q-values.
    alpha_trans: [0, 1] - Learning rate for transition matrix probabilities.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pers_rew: [-5, 5] - Stage 1 perseveration bonus if previous trial was rewarded.
    pers_unrew: [-5, 5] - Stage 1 perseveration bonus if previous trial was unrewarded.
    pers2: [-5, 5] - Stage 2 perseveration bonus for repeating the same stage 2 action.
    """
    alpha, alpha_trans, beta1, beta2, w, pers_rew, pers_unrew, pers2 = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix as uniform (0.5) to capture learning from scratch
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                pers_bonus1[last_action1] = pers_rew
            else:
                pers_bonus1[last_action1] = pers_unrew
            
        logits1 = beta1 * q_net1 + pers_bonus1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        pers_bonus2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            pers_bonus2[last_action2[state_idx]] = pers2

        logits2 = beta2 * q_stage2_mf[state_idx] + pers_bonus2
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Update transition matrix probabilities based on observation
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Model with separate learning rates for stage 1 and stage 2, 
    decay for unchosen actions, and reward-dependent stage 1 perseveration.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for stage 1 prediction errors.
    alpha2: [0, 1] - Learning rate for stage 2 prediction errors.
    decay: [0, 1] - Forgetting/decay rate for unchosen actions.
    beta1: [0, 10] - Inverse temperature for stage 1 choices.
    beta2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Weighting of model-based vs model-free values.
    pers_rew: [-5, 5] - Stage 1 perseveration bonus if previous trial was rewarded.
    pers_unrew: [-5, 5] - Stage 1 perseveration bonus if previous trial was unrewarded.
    """
    alpha1, alpha2, decay, beta1, beta2, w, pers_rew, pers_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_reward = 0

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                pers_bonus1[last_action1] = pers_rew
            else:
                pers_bonus1[last_action1] = pers_unrew
            
        logits1 = beta1 * q_net1 + pers_bonus1
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Decay for unchosen actions
        unchosen_stage1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_stage1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        last_action1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
