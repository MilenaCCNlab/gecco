def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Choice Kernel + Unchosen Q-value decay.
    Uses an exponentially decaying choice kernel for Stage 1 instead of simple perseveration.
    The choice kernel captures long-term stickiness to an action, which fits the participant's 
    behavior of choosing the same spaceship for long blocks. Unchosen Q-values decay towards 0.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    ck_weight: Weight of the choice kernel on stage 1 [-5, 5]
    ck_alpha: Update rate for the choice kernel [0, 1]
    lambd: Eligibility trace [0, 1]
    decay: Decay rate for unchosen options [0, 1]
    """
    alpha, beta1, beta2, w, ck_weight, ck_alpha, lambd, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    ck = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        state_idx = state[trial]
        act1 = action_1[trial]
        act2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits1 = beta1 * q_stage1_net + ck_weight * ck
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[act1]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[act2]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
        q_stage1_mf[act1] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
        q_stage2_mf[state_idx, act2] += alpha * delta_stage2

        q_stage1_mf[act1] += alpha * lambd * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - act1
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        unchosen_2 = 1 - act2
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
        
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1 - decay)
        q_stage2_mf[other_state, 1] *= (1 - decay)
        
        # Choice kernel update
        ck[act1] = ck[act1] + ck_alpha * (1.0 - ck[act1])
        ck[unchosen_1] = ck[unchosen_1] * (1.0 - ck_alpha)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition-dependent perseveration with Stage 1 action bias.
    Includes a static bias parameter for choosing spaceship 0 to account for 
    the participant's strong overall preference for action 0 (chosen 138 times).
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w: Model-based weight [0, 1]
    pi_common: Perseveration bonus if previous transition was common [-5, 5]
    pi_rare: Perseveration bonus if previous transition was rare [-5, 5]
    lambd: Eligibility trace [0, 1]
    bias1: Bias for choosing action 0 in stage 1 [-5, 5]
    """
    alpha, beta1, beta2, w, pi_common, pi_rare, lambd, bias1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_common = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_trans_common = -1
            continue

        state_idx = state[trial]
        act1 = action_1[trial]
        act2 = action_2[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        if last_trans_common == 1:
            pi = pi_common
        elif last_trans_common == 0:
            pi = pi_rare
        else:
            pi = 0.0
            
        logits1 = beta1 * (q_stage1_net + pi * rep)
        logits1[0] += bias1  # Apply bias for action 0
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[act1]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[act2]

        delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
        q_stage1_mf[act1] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
        q_stage2_mf[state_idx, act2] += alpha * delta_stage2

        q_stage1_mf[act1] += alpha * lambd * delta_stage2

        last_action = act1
        if (act1 == 0 and state_idx == 0) or (act1 == 1 and state_idx == 1):
            last_trans_common = 1
        else:
            last_trans_common = 0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent Model-Based weight with transition-dependent perseveration.
    The participant might rely more on model-based planning (w) after a rewarded trial 
    versus an unrewarded trial, capturing outcome-sensitive goal-directed behavior.
    
    Parameters:
    alpha: Learning rate [0, 1]
    beta1: Inverse temperature for stage 1 [0, 10]
    beta2: Inverse temperature for stage 2 [0, 10]
    w_win: Model-based weight after a rewarded trial [0, 1]
    w_lose: Model-based weight after an unrewarded trial [0, 1]
    pi_common: Perseveration bonus if previous transition was common [-5, 5]
    pi_rare: Perseveration bonus if previous transition was rare [-5, 5]
    lambd: Eligibility trace [0, 1]
    """
    alpha, beta1, beta2, w_win, w_lose, pi_common, pi_rare, lambd = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_trans_common = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action = -1
            last_trans_common = -1
            last_reward = -1
            continue

        state_idx = state[trial]
        act1 = action_1[trial]
        act2 = action_2[trial]

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine the model-based weight depending on previous reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = 0.5 * w_win + 0.5 * w_lose  # Default for first valid trial
            
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action != -1:
            rep[last_action] = 1.0
            
        if last_trans_common == 1:
            pi = pi_common
        elif last_trans_common == 0:
            pi = pi_rare
        else:
            pi = 0.0
            
        logits1 = beta1 * (q_stage1_net + pi * rep)
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[act1]

        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[act2]

        delta_stage1 = q_stage2_mf[state_idx, act2] - q_stage1_mf[act1]
        q_stage1_mf[act1] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, act2]
        q_stage2_mf[state_idx, act2] += alpha * delta_stage2

        q_stage1_mf[act1] += alpha * lambd * delta_stage2

        last_action = act1
        if (act1 == 0 and state_idx == 0) or (act1 == 1 and state_idx == 1):
            last_trans_common = 1
        else:
            last_trans_common = 0
            
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
