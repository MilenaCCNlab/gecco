def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with asymmetric learning rates for positive and negative prediction errors,
    separate inverse temperatures for stage 1 and stage 2, and stage 1 stickiness.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta1: [0, 10] - Inverse temperature for stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Mixing weight for model-based vs model-free values (1 = pure MB).
    lam: [0, 1] - Eligibility trace parameter.
    pi: [-5, 5] - Choice stickiness (perseveration) for stage 1.
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector = np.zeros(2)
        if last_action_1 != -1:
            stick_vector[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta1 * q_stage1_net + pi * stick_vector)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based / Model-Free Q-learning with asymmetric learning rates, stage 1 stickiness, 
    and decay for unchosen options.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive PEs.
    alpha_neg: [0, 1] - Learning rate for negative PEs.
    beta: [0, 10] - Inverse temperature.
    w: [0, 1] - Mixing weight (1 = pure MB).
    lam: [0, 1] - Eligibility trace.
    pi: [-5, 5] - Stage 1 stickiness.
    decay: [0, 1] - Decay rate for unchosen actions.
    """
    alpha_pos, alpha_neg, beta, w, lam, pi, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector = np.zeros(2)
        if last_action_1 != -1:
            stick_vector[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi * stick_vector)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        # decay unchosen action in stage 1
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2
        
        # decay unchosen actions in stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # eligibility trace update
        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
