import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for Stage 1 and Stage 2 with reward-dependent perseveration.
    This model adds a `rep_rew` parameter to capture win-stay behavior, where the participant 
    is more likely to repeat an action if it was rewarded on the previous trial, on top of 
    general perseveration.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers (float): General perseveration for both stages. Bounds: [-5,5]
    rep_rew (float): Reward-dependent perseveration (win-stay). Bounds: [-5,5]
    """
    alpha1, alpha2, beta, w, lam, pers, rep_rew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            if last_reward > 0:
                q_net1[last_action1] += rep_rew
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            if last_reward > 0:
                q_net2[last_action2[state_idx]] += rep_rew
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates for Stage 1 and Stage 2 with Q-value decay.
    This model includes a decay parameter that causes all Q-values to decay towards zero 
    at each trial, capturing forgetting of unchosen options.
    
    Parameters:
    alpha1 (float): Learning rate for Stage 1. Bounds: [0,1]
    alpha2 (float): Learning rate for Stage 2. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers (float): Perseveration for both stages. Bounds: [-5,5]
    decay (float): Decay rate for Q-values. Bounds: [0,1]
    """
    alpha1, alpha2, beta, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Q-values decay towards 0 before update
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates for positive and negative prediction errors at both stages.
    This model separates learning rates based on whether the prediction error is positive 
    or negative, independently for Stage 1 and Stage 2, allowing for differential sensitivity 
    to better-than-expected vs worse-than-expected outcomes.
    
    Parameters:
    alpha1_pos (float): Stage 1 learning rate for positive PE. Bounds: [0,1]
    alpha1_neg (float): Stage 1 learning rate for negative PE. Bounds: [0,1]
    alpha2_pos (float): Stage 2 learning rate for positive PE. Bounds: [0,1]
    alpha2_neg (float): Stage 2 learning rate for negative PE. Bounds: [0,1]
    beta (float): Inverse temperature for both stages. Bounds: [0,10]
    w (float): Model-based weight. Bounds: [0,1]
    lam (float): Eligibility trace parameter. Bounds: [0,1]
    pers (float): Perseveration for both stages. Bounds: [-5,5]
    """
    alpha1_pos, alpha1_neg, alpha2_pos, alpha2_neg, beta, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action1 = -1
    last_action2 = np.array([-1, -1])

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if last_action1 != -1:
            q_net1[last_action1] += pers
            
        exp_q1 = np.exp(beta * q_net1 - np.max(beta * q_net1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q_net2 = q_stage2_mf[state_idx].copy()
        if last_action2[state_idx] != -1:
            q_net2[last_action2[state_idx]] += pers
            
        exp_q2 = np.exp(beta * q_net2 - np.max(beta * q_net2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha1_pos if delta_stage1 > 0 else alpha1_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha2_pos if delta_stage2 > 0 else alpha2_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2

        # Eligibility trace update using Stage 1 learning rate
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_action1 = action_1[trial]
        last_action2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
