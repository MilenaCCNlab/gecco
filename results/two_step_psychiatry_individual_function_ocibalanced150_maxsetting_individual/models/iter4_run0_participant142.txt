Here are 3 new cognitive models that build upon the insights from the best models found so far. They retain the successful separation of model-based (`beta_mb`) and model-free (`beta_mf`) inverse temperatures but introduce new mechanisms to capture additional variance in participant behavior without exceeding the 8-parameter limit.

### Model 1: Decay for Unchosen Options
This model assumes that the values of unchosen actions are not static but instead decay towards zero over time. This helps explain exploratory behavior or forgetting, where participants might try options they haven't chosen in a while because their learned values have passively decayed.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate MB and MF inverse temperatures at stage 1, 
    and a decay mechanism for unchosen options at both stages.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-value updates
    decay: [0,1] - decay rate towards 0 for unchosen options
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2 choices
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - choice perseverance for stage 1
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha, decay, beta_mb, beta_mf, beta2, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_stage2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 policy
        state_idx = state[trial]
        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Value decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.0 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.0 - q_stage2_mf[s, a])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Asymmetric Learning Rates (Positive vs. Negative Prediction Errors)
This model posits that participants learn differently from better-than-expected outcomes compared to worse-than-expected outcomes. This is a well-established phenomenon in reinforcement learning that captures optimism or pessimism biases in learning.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate learning rates for positive and negative 
    prediction errors, combined with separate MB and MF inverse temperatures.
    
    Bounds:
    alpha_pos: [0,1] - learning rate for positive prediction errors
    alpha_neg: [0,1] - learning rate for negative prediction errors
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - choice perseverance for stage 1
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha_pos, alpha_neg, beta_mb, beta_mf, beta2, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_stage2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 policy
        state_idx = state[trial]
        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]
        
        # Action value updating with asymmetric learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2 * lam * delta_stage2
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Dynamic Learning of the Transition Matrix
Instead of assuming that participants completely trust and freeze the provided 0.7/0.3 transition probabilities, this model allows the participant to update their internal model of the environment's transition matrix based on the actual transitions they experience on each trial.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model where transition probabilities are dynamically learned 
    over time, combined with separate MB and MF inverse temperatures.
    
    Bounds:
    alpha_q: [0,1] - learning rate for Q-values
    alpha_trans: [0,1] - learning rate for transition probabilities
    beta_mb: [0,10] - inverse temperature for model-based Q-values
    beta_mf: [0,10] - inverse temperature for model-free Q-values
    beta2: [0,10] - inverse temperature for stage 2
    lam: [0,1] - eligibility trace parameter
    pers1: [-5,5] - choice perseverance for stage 1
    pers2: [-5,5] - choice perseverance for stage 2
    """
    alpha_q, alpha_trans, beta_mb, beta_mf, beta2, lam, pers1, pers2 = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix to the instructed probabilities
    trans_prob = np.array([[0.7, 0.3], [0.3, 0.7]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_stage2 = np.array([-1, -1])
    
    for trial in range(n_trials):
        # Stage 1 policy based on dynamically learned transitions
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = trans_prob @ max_q_stage2
        
        exponent1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if trial > 0:
            exponent1[action_1[trial-1]] += pers1
            
        exp_q1 = np.exp(exponent1 - np.max(exponent1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        # Stage 2 policy
        state_idx = state[trial]
        exponent2 = beta2 * q_stage2_mf[state_idx]
        if last_action_stage2[state_idx] != -1:
            exponent2[last_action_stage2[state_idx]] += pers2
            
        exp_q2 = np.exp(exponent2 - np.max(exponent2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_stage2[state_idx] = action_2[trial]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_q * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_q * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_q * lam * delta_stage2
        
        # Update transition probabilities based on observed transition
        trans_prob[action_1[trial], state_idx] += alpha_trans * (1.0 - trans_prob[action_1[trial], state_idx])
        trans_prob[action_1[trial], 1 - state_idx] += alpha_trans * (0.0 - trans_prob[action_1[trial], 1 - state_idx])
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
