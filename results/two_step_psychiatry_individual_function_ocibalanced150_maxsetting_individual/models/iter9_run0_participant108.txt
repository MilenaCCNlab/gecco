def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with Stage- and Valence-Dependent Learning Rates.
    
    This model allows for maximum flexibility in learning by assigning independent 
    learning rates for both stage 1 and stage 2, separated by positive and negative 
    prediction errors.
    
    Parameters:
    alpha_1_pos: learning rate for stage 1 positive prediction errors [0,1]
    alpha_1_neg: learning rate for stage 1 negative prediction errors [0,1]
    alpha_2_pos: learning rate for stage 2 positive prediction errors [0,1]
    alpha_2_neg: learning rate for stage 2 negative prediction errors [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    p: stickiness (perseveration) [-5,5]
    """
    alpha_1_pos, alpha_1_neg, alpha_2_pos, alpha_2_neg, beta, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta * q_stage1_net + p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        if a2 == -1:
            p_choice_2[trial] = 1.0
            last_action_1 = a1
            continue
            
        logits_2 = beta * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        alpha_1 = alpha_1_pos if delta_stage1 > 0 else alpha_1_neg
        alpha_2 = alpha_2_pos if delta_stage2 > 0 else alpha_2_neg
        
        q_stage1_mf[a1] += alpha_1 * delta_stage1
        q_stage2_mf[s, a2] += alpha_2 * delta_stage2
        
        alpha_1_trace = alpha_1_pos if delta_stage2 > 0 else alpha_1_neg
        q_stage1_mf[a1] += alpha_1_trace * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with Reward-Dependent Model-Based Weight.
    
    This model assumes that the degree to which a participant relies on the model-based 
    system vs. the model-free system is modulated by whether the previous trial was rewarded.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta: inverse temperature [0,10]
    w_rew: model-based weight if previous trial was rewarded [0,1]
    w_unrew: model-based weight if previous trial was unrewarded [0,1]
    lam: eligibility trace [0,1]
    p: stickiness (perseveration) [-5,5]
    decay: decay rate for unchosen options [0,1]
    """
    alpha_pos, alpha_neg, beta, w_rew, w_unrew, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        w = w_rew if last_reward > 0 else w_unrew
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta * q_stage1_net + p * stickiness
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        if a2 == -1:
            p_choice_2[trial] = 1.0
            last_action_1 = a1
            continue
            
        logits_2 = beta * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] += decay * (0.5 - q_stage2_mf[state_idx, action_idx])

        for action_idx in range(2):
            if action_idx != a1:
                q_stage1_mf[action_idx] += decay * (0.5 - q_stage1_mf[action_idx])
                
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage1_mf[a1] += alpha_1 * delta_stage1
        q_stage2_mf[s, a2] += alpha_2 * delta_stage2
        q_stage1_mf[a1] += alpha_2 * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
