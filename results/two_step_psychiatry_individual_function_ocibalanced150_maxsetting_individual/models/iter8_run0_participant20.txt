Here are 3 new cognitive models that explore alternative parameter configurations and mechanisms for the two-step task. 

### Model 1: Separate Stage Learning Rates + Win-Stay/Lose-Shift
This model proposes that participants might update their values at different rates for Stage 1 vs Stage 2 choices. It also incorporates a heuristic "Win-Stay/Lose-Shift" mechanism at Stage 1, adding a bonus to repeat the previous choice if it was rewarded, and a bonus to switch if it was unrewarded.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with separate learning rates for Stage 1 and Stage 2,
    and Win-Stay/Lose-Shift heuristic behavior at Stage 1.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for Stage 1
    alpha2: [0, 1] - Learning rate for Stage 2
    beta1: [0, 10] - Inverse temperature for Stage 1
    beta2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    stay_win: [0, 10] - Bonus for repeating the same Stage 1 choice if previously rewarded
    shift_lose: [0, 10] - Bonus for switching the Stage 1 choice if previously unrewarded
    """
    alpha1, alpha2, beta1, beta2, w, lam, stay_win, shift_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1
    prev_r = 0
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        # Handle missing trials
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if prev_a1 != -1:
            if prev_r == 1:
                logits_1[prev_a1] += stay_win
            else:
                logits_1[1 - prev_a1] += shift_lose
                
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha2 * delta_stage2
        
        prev_a1 = a1
        prev_r = r
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates + Win-Stay/Lose-Shift
This model accounts for the possibility that participants learn differently from better-than-expected outcomes compared to worse-than-expected outcomes (optimistic vs. pessimistic learning biases). It merges these asymmetric updates with the Win-Stay/Lose-Shift mechanism at Stage 1.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with asymmetric learning rates for positive and negative prediction errors,
    combined with Win-Stay/Lose-Shift behavior at Stage 1.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors
    alpha_neg: [0, 1] - Learning rate for negative prediction errors
    beta1: [0, 10] - Inverse temperature for Stage 1
    beta2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    stay_win: [0, 10] - Bonus for repeating the same Stage 1 choice if previously rewarded
    shift_lose: [0, 10] - Bonus for switching the Stage 1 choice if previously unrewarded
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, stay_win, shift_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1
    prev_r = 0
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if prev_a1 != -1:
            if prev_r == 1:
                logits_1[prev_a1] += stay_win
            else:
                logits_1[1 - prev_a1] += shift_lose
                
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        alpha_1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage1_mf[a1] += alpha_1 * delta_stage1 + alpha_2 * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha_2 * delta_stage2
        
        prev_a1 = a1
        prev_r = r
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition Learning + Unchosen Value Decay + Perseveration
Instead of assuming a fixed transition matrix of 70/30, this model learns the transition probabilities dynamically trial-by-trial. It combines this flexibility with value decay for unchosen options and a standard choice perseveration term. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-Step Task Model with dynamic transition matrix learning, value decay for unchosen actions,
    and choice perseveration at Stage 1.
    
    Bounds:
    alpha_v: [0, 1] - Learning rate for state-action values
    alpha_tr: [0, 1] - Learning rate for transition probabilities
    decay: [0, 1] - Decay rate pulling unchosen option values towards 0.5
    beta1: [0, 10] - Inverse temperature for Stage 1
    beta2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Weighting of model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter
    pers: [0, 10] - General perseveration bonus for repeating the previous Stage 1 choice
    """
    alpha_v, alpha_tr, decay, beta1, beta2, w, lam, pers = model_parameters
    n_trials = len(action_1)
    
    # Initialize transition matrix evenly
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_a1 = -1
    
    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net
        if prev_a1 != -1:
            logits_1[prev_a1] += pers
            
        max_logit_1 = np.max(logits_1)
        exp_q1 = np.exp(logits_1 - max_logit_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[s]
        max_logit_2 = np.max(logits_2)
        exp_q2 = np.exp(logits_2 - max_logit_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
        
        # Update dynamically learned transition matrix
        transition_matrix[a1, s] += alpha_tr * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1-s] += alpha_tr * (0.0 - transition_matrix[a1, 1-s])
        
        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        q_stage1_mf[a1] += alpha_v * delta_stage1 + alpha_v * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha_v * delta_stage2
        
        # Decay unchosen options towards neutral (0.5)
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1.0 - decay) * q_stage1_mf[unchosen_a1] + decay * 0.5
        
        for state_idx in range(2):
            for action_idx in range(2):
                if not (state_idx == s and action_idx == a2):
                    q_stage2_mf[state_idx, action_idx] = (1.0 - decay) * q_stage2_mf[state_idx, action_idx] + decay * 0.5
                    
        prev_a1 = a1
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
