Here are 3 new cognitive models that build on your best model by introducing different mechanisms to capture participant behavior, without duplicating previous parameter combinations. 

### Model 1: Dynamic Transition Learning
While participants are often told the transition probabilities (0.7/0.3), they might still subjectively update these probabilities based on recent experience (e.g., heavily weighting a recent rare transition). This model introduces `alpha_tr` to dynamically update the transition matrix trial-by-trial.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Learns transition probabilities dynamically, along with lambda, stickiness, and decay.
    
    Parameters:
    alpha: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature controlling choice stochasticity.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) control.
    lambda_: [0, 1] - Eligibility trace for stage 1 value update from stage 2 reward.
    stickiness: [-5, 5] - Tendency to repeat the previous stage-1 choice regardless of value.
    decay: [0, 1] - Rate at which unchosen action values decay toward zero.
    alpha_tr: [0, 1] - Learning rate for updating subjective transition probabilities.
    """
    alpha, beta, w, lambda_, stickiness, decay, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize with the objective probabilities but allow for dynamic updating
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness * stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update subjective transition probabilities based on observed transition
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0 - transition_matrix[action_1[trial], 1 - state_idx])
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2 
        
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Fully Decoupled Stage 1 and Stage 2 Sensitivities 
This model separates both the learning rate (`alpha`) and the choice stochasticity (`beta`) for stage 1 and stage 2. Participants often exhibit structurally distinct learning patterns for the spaceship choice compared to the alien choice.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate learning rates and inverse temperatures for stage 1 and stage 2,
    with lambda, stickiness, and unchosen value decay.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for stage 1 value updates.
    alpha_2: [0, 1] - Learning rate for stage 2 value updates.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight.
    lambda_: [0, 1] - Eligibility trace.
    stickiness: [-5, 5] - Perseveration on stage 1 choice.
    decay: [0, 1] - Unchosen value decay.
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lambda_, stickiness, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var = np.zeros(2)
        if trial > 0:
            stick_var[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + stickiness * stick_var)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2
        
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Decoupled Stage-Specific Perseveration and Forgetting
This configuration tests the theory that the tendency to repeat choices (stickiness) and the rate at which memories of unchosen options fade (decay) differ drastically between navigating the macro-structure (spaceships) and the micro-structure (aliens).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate stickiness and decay rates for stage 1 and stage 2.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Inverse temperature.
    w: [0, 1] - Model-based weight.
    lambda_: [0, 1] - Eligibility trace.
    stickiness_1: [-5, 5] - Perseveration on stage 1 choice.
    stickiness_2: [-5, 5] - Perseveration on stage 2 choice (state-specific).
    decay_1: [0, 1] - Unchosen value decay for stage 1.
    decay_2: [0, 1] - Unchosen value decay for stage 2.
    """
    alpha, beta, w, lambda_, stickiness_1, stickiness_2, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_var_1 = np.zeros(2)
        if trial > 0:
            stick_var_1[action_1[trial - 1]] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + stickiness_1 * stick_var_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stick_var_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stick_var_2[last_action_2[state_idx]] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + stickiness_2 * stick_var_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        last_action_2[state_idx] = action_2[trial]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay_1)
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay_2)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay_2)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay_2)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
