def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive/negative RPEs, separate temperatures 
    for stage 1 and stage 2, unchosen action decay, eligibility trace, MB/MF weighting, 
    and stage 1 stickiness.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors
    alpha_neg: [0,1] - Learning rate for negative prediction errors
    decay_rate: [0,1] - Decay parameter for unchosen actions towards 0.5
    lambda_trace: [0,1] - Eligibility trace parameter
    beta1: [0,10] - Inverse temperature for stage 1 softmax choice
    beta2: [0,10] - Inverse temperature for stage 2 softmax choice
    w: [0,1] - Weight of model-based vs model-free control
    stickiness: [-5,5] - Choice perseveration bonus for the previously chosen stage 1 action
    """
    alpha_pos, alpha_neg, decay_rate, lambda_trace, beta1, beta2, w, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if prev_action_1 != -1:
            q_net[prev_action_1] += stickiness
            
        q1_scaled = beta1 * q_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q2_scaled = beta2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += lr2 * lambda_trace * delta_stage2

        # Decay for unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_action_1])

        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] += decay_rate * (0.5 - q_stage2_mf[state_idx, unchosen_action_2])

        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay_rate * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay_rate * (0.5 - q_stage2_mf[unvisited_state, 1])
        
        prev_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Dynamic transition matrix learning. Instead of assuming fixed 0.7/0.3 transitions, 
    the model updates its internal transition matrix based on experienced state transitions. 
    Also includes decay, eligibility trace, MB/MF weighting, and stage 1 stickiness.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-values
    decay_rate: [0,1] - Decay parameter for unchosen actions towards 0.5
    lambda_trace: [0,1] - Eligibility trace parameter
    beta: [0,10] - Inverse temperature for softmax choice
    w: [0,1] - Weight of model-based vs model-free control
    stickiness: [-5,5] - Choice perseveration bonus for the previously chosen stage 1 action
    alpha_trans: [0,1] - Learning rate for updating the transition matrix probabilities
    """
    learning_rate, decay_rate, lambda_trace, beta, w, stickiness, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix assuming 0.5 probability initially
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        if prev_action_1 != -1:
            q_net[prev_action_1] += stickiness
            
        q1_scaled = beta * q_net
        exp_q1 = np.exp(q1_scaled - np.max(q1_scaled))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        q2_scaled = beta * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(q2_scaled - np.max(q2_scaled))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update transition matrix based on observed transition
        a1 = action_1[trial]
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] += alpha_trans * (0.0 - transition_matrix[a1, 1 - state_idx])

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lambda_trace * delta_stage2

        # Decay for unchosen actions
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] += decay_rate * (0.5 - q_stage1_mf[unchosen_action_1])

        unchosen_action_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_action_2] += decay_rate * (0.5 - q_stage2_mf[state_idx, unchosen_action_2])

        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, 0] += decay_rate * (0.5 - q_stage2_mf[unvisited_state, 0])
        q_stage2_mf[unvisited_state, 1] += decay_rate * (0.5 - q_stage2_mf[unvisited_state, 1])
        
        prev_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate temperatures for stage 1 and 2, combined with both general stickiness 
    and reward-dependent stickiness (win-stay). Also includes decay, eligibility trace, 
    and MB/MF weighting.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for Q-values
    decay_rate: [0,1] - Decay parameter for unchosen actions towards 0.5
    lambda_trace: [0,1] - Eligibility trace parameter
    beta1: [0,10] - Inverse temperature for stage 1 softmax choice
    beta2: [0,10] - Inverse temperature for stage 2 softmax choice
    w: [0,1] - Weight of model-based vs model-free control
    stickiness: [-5,5] - General choice perseveration bonus for the previously chosen stage 1 action
    reward_stickiness: [-5,5] - Additional perseveration bonus if the previous trial was rewarded (win-stay)
    """
    learning_rate, decay_rate, lambda_trace, beta1, beta2, w, stickiness, reward_stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    prev_action_1 = -1
    prev_reward = 0

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        # Action value updating for choice 1

        # Action value updating for choice 2
