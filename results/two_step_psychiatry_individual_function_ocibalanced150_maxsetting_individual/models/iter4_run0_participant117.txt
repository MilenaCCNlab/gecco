def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Dual-Temperature Model with Decay and Stickiness.
    Instead of a single weighting parameter (w) for model-based vs model-free values, 
    this model uses independent inverse temperatures for model-based (beta_mb) and 
    model-free (beta_mf) Stage 1 Q-values. It also retains unchosen option decay 
    and choice stickiness.
    
    Bounds:
    alpha: [0,1] - Learning rate
    decay: [0,1] - Decay rate for unchosen options
    beta_mb: [0,10] - Inverse temperature for Stage 1 model-based values
    beta_mf: [0,10] - Inverse temperature for Stage 1 model-free values
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    p1: [-5,5] - Choice stickiness for Stage 1
    p2: [-5,5] - Choice stickiness for Stage 2
    """
    alpha, decay, beta_mb, beta_mf, beta2, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if prev_a2 != -1:
            rep2[prev_a2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen options
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Transition-dependent eligibility trace.
    The eligibility trace (lambda) applied to Stage 1 depends on whether the transition 
    from Stage 1 to Stage 2 was common or rare. Participants might credit Stage 1 
    differently if the outcome was reached via a rare (surprising) transition.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam_common: [0,1] - Eligibility trace for common transitions
    lam_rare: [0,1] - Eligibility trace for rare transitions
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p1: [-5,5] - Choice stickiness for Stage 1
    p2: [-5,5] - Choice stickiness for Stage 2
    """
    alpha, beta1, beta2, lam_common, lam_rare, w, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if prev_a2 != -1:
            rep2[prev_a2] = 1
            
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # transition-dependent eligibility trace
        if action_1[trial] == state_idx:
            lam = lam_common
        else:
            lam = lam_rare

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-dependent Stage 1 Stickiness with Decay.
    Stickiness for the Stage 1 choice depends on whether the previous trial's transition 
    was common or rare. Experiencing a rare transition might disrupt choice perseveration.
    
    Bounds:
    alpha: [0,1] - Learning rate
    decay: [0,1] - Decay rate for unchosen options
    beta1: [0,10] - Inverse temperature for Stage 1 choices
    beta2: [0,10] - Inverse temperature for Stage 2 choices
    lam: [0,1] - Eligibility trace parameter
    w: [0,1] - Model-based vs model-free weight for Stage 1
    p_common: [-5,5] - Stage 1 stickiness after a common transition
    p_rare: [-5,5] - Stage 1 stickiness after a rare transition
    """
    alpha, decay, beta1, beta2, lam, w, p_common, p_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_trans_common = True

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            
        # dynamic stickiness based on previous transition
        p1 = p_common if prev_trans_common else p_rare
        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1 and 2
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # decay for unchosen options
        unchosen_a1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_a1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        prev_a1 = action_1[trial]
        # determine if transition was common (action == state)
        prev_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
