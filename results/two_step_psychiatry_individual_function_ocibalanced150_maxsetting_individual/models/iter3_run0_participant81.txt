Here are 3 new cognitive models that explore different parameter configurations and mechanisms (separate learning rates/betas per stage, asymmetric learning for positive/negative prediction errors, and outcome-dependent perseveration).

### Model 1: Separate Learning Rates and Inverse Temperatures per Stage
This model builds on the classic Daw (2011) two-step model but allows for independent learning rates (`alpha_1`, `alpha_2`) and inverse temperatures (`beta_1`, `beta_2`) for each decision stage. This isolates differences in learning speed and choice stochasticity between the first and second steps.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates and inverse temperatures for Stage 1 and Stage 2, Eligibility Trace, Perseveration.
    Bounds:
    alpha_1: Stage 1 learning rate [0, 1]
    alpha_2: Stage 2 learning rate [0, 1]
    beta_1: Stage 1 inverse temperature [0, 10]
    beta_2: Stage 2 inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p: perseveration parameter [-5, 5]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action = action_1[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue

        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_1 * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates and Separate Betas
This model evaluates if participants learn differently from positive versus negative prediction errors (e.g., updating faster when things go better than expected than when they go worse). It also implements separate choice stochasticity (`beta_1`, `beta_2`) for each stage.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric learning rates for positive and negative prediction errors, separate betas, eligibility trace, perseveration.
    Bounds:
    alpha_pos: learning rate for positive PE [0, 1]
    alpha_neg: learning rate for negative PE [0, 1]
    beta_1: Stage 1 inverse temperature [0, 10]
    beta_2: Stage 2 inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p: perseveration parameter [-5, 5]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lambda_, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action != -1:
            pers[last_action] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_stage1_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action = action_1[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue

        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1_eff = alpha_pos if delta_stage1 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1_eff * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2_eff = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2_eff * delta_stage2

        alpha_lambda_eff = alpha_pos if delta_stage2 >= 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_lambda_eff * lambda_ * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Outcome-Dependent Perseveration (Win-Stay, Lose-Shift)
This model tests whether the participant's tendency to repeat the previous Stage 1 action depends on whether the previous trial was rewarded or not (`p_win` vs. `p_lose`). This isolates heuristic "win-stay, lose-shift" behavior directly in the perseveration term.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Learning with outcome-dependent perseveration (win-stay, lose-shift) and eligibility trace.
    Bounds:
    alpha: learning rate [0, 1]
    beta: inverse temperature [0, 10]
    w: model-based weight [0, 1]
    lambda_: eligibility trace [0, 1]
    p_win: perseveration parameter after a rewarded trial [-5, 5]
    p_lose: perseveration parameter after an unrewarded trial [-5, 5]
    """
    alpha, beta, w, lambda_, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action != -1:
            if last_reward == 1:
                pers[last_action] = p_win
            elif last_reward == 0:
                pers[last_action] = p_lose
            
        exp_q1 = np.exp(beta * q_stage1_net + pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action = action_1[trial]

        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            continue

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lambda_ * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
