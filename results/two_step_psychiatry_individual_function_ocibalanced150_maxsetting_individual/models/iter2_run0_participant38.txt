Here are 3 new cognitive models that explore different well-established computational mechanisms in reinforcement learning and psychiatry, avoiding the exact parameter combinations you've already tried.

### Model 1: Asymmetric Learning Rates (Positive vs. Negative Prediction Errors)
In psychiatric populations (and the general population), individuals often learn differently from positive versus negative feedback. This model separates the learning rate into `alpha_pos` (used when prediction error > 0) and `alpha_neg` (used when prediction error < 0), while maintaining separate stage 1 and stage 2 inverse temperatures.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative prediction errors.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax choice.
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax choice.
    w: [0, 1] - Weight given to model-based Q-values for stage 1 (1 = fully MB, 0 = fully MF).
    lam: [0, 1] - Eligibility trace parameter for stage 1 MF update.
    stickiness: [-5, 5] - Choice perseveration bonus for repeating the previous stage 1 action.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            stickiness_bonus[last_action] = stickiness
            
        logits_1 = beta_1 * q_stage1_net + stickiness_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action = action_1[trial]
            continue
            
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Independent Model-Based and Model-Free Temperatures
Instead of forcing a trade-off parameter `w` between model-based (MB) and model-free (MF) values, this model uses independent inverse temperatures `beta_mb` and `beta_mf`. This allows the model to capture participants who might simply have highly stochastic choice behavior versus those who actively employ one strategy over the other.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Independent inverse temperatures for model-based and model-free Q-values at stage 1.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_mb: [0, 10] - Inverse temperature for model-based Q-values at stage 1.
    beta_mf: [0, 10] - Inverse temperature for model-free Q-values at stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax choice.
    lam: [0, 1] - Eligibility trace parameter.
    stickiness: [-5, 5] - Choice perseveration bonus.
    """
    learning_rate, beta_mb, beta_mf, beta_2, lam, stickiness = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            stickiness_bonus[last_action] = stickiness
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + stickiness_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action = action_1[trial]
            continue
            
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Stickiness (Win-Stay / Lose-Shift)
Often, participants don't just perseverate universally. Instead, they might exhibit "win-stay" behavior (repeating actions that were rewarded) and "lose-shift" behavior (switching away from actions that yielded no reward). This model introduces two distinct stickiness parameters depending on the outcome of the previous trial.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent stickiness (Win-Stay / Lose-Shift).
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta_1: [0, 10] - Inverse temperature for stage 1 softmax choice.
    beta_2: [0, 10] - Inverse temperature for stage 2 softmax choice.
    w: [0, 1] - Weight given to model-based Q-values for stage 1.
    lam: [0, 1] - Eligibility trace parameter for stage 1 MF update.
    stickiness_rew: [-5, 5] - Perseveration bonus if previous trial was rewarded.
    stickiness_unrew: [-5, 5] - Perseveration bonus if previous trial was unrewarded.
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness_rew, stickiness_unrew = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_bonus = np.zeros(2)
        if last_action != -1:
            if last_reward > 0:
                stickiness_bonus[last_action] = stickiness_rew
            else:
                stickiness_bonus[last_action] = stickiness_unrew
            
        logits_1 = beta_1 * q_stage1_net + stickiness_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] == -1:
            p_choice_2[trial] = 1.0
            last_action = action_1[trial]
            last_reward = 0
            continue
            
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Action value updating for Choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # Action value updating for Choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
