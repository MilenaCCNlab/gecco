def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with a single learning rate, reward-dependent choice stickiness (Win-Stay/Lose-Shift) 
    at stage 1, and decay (forgetting) of unchosen Q-values.
    
    Bounds:
    alpha: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    lam: [0, 1]
    w: [0, 1]
    p_win: [-5, 5]
    p_lose: [-5, 5]
    decay: [0, 1]
    """
    alpha, beta1, beta2, lam, w, p_win, p_lose, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1

        current_p = p_win if last_reward > 0 else p_lose
            
        q1 = beta1 * q_net + current_p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        # Decay unchosen Q-values
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with positive and negative learning rates (shared across stages), 
    standard choice stickiness at stage 1, and decay of unchosen Q-values.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    lam: [0, 1]
    w: [0, 1]
    p: [-5, 5]
    decay: [0, 1]
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        q1 = beta1 * q_net + p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1_current = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha2_current = alpha_pos if delta_stage2 > 0 else alpha_neg

        q_stage1_mf[action_1[trial]] += alpha1_current * delta_stage1 + alpha1_current * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_current * delta_stage2
        
        # Decay unchosen Q-values
        q_stage1_mf[1 - action_1[trial]] *= (1 - decay)
        q_stage2_mf[state_idx, 1 - action_2[trial]] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 0] *= (1 - decay)
        q_stage2_mf[1 - state_idx, 1] *= (1 - decay)
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with positive and negative learning rates (shared across stages), 
    and transition-dependent choice stickiness at stage 1.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    lam: [0, 1]
    w: [0, 1]
    p_common: [-5, 5]
    p_rare: [-5, 5]
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, p_common, p_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_trans_common = True

    for trial in range(n_trials):
        # Stage 1 choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1

        current_p = p_common if last_trans_common else p_rare
            
        q1 = beta1 * q_net + current_p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        alpha1_current = alpha_pos if delta_stage1 > 0 else alpha_neg
        alpha2_current = alpha_pos if delta_stage2 > 0 else alpha_neg

        q_stage1_mf[action_1[trial]] += alpha1_current * delta_stage1 + alpha1_current * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_current * delta_stage2
        
        last_action_1 = action_1[trial]
        # Transition is common if action 0 -> state 0 or action 1 -> state 1
        last_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
