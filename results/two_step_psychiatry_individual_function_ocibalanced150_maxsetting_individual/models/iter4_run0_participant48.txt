Here are three new cognitive models that build upon the best-performing model (which incorporates choice perseveration and unchosen value decay). They introduce additional cognitive mechanisms known to capture human behavior in two-step tasks: asymmetric learning rates for positive and negative prediction errors, state-specific perseveration at the second stage, and dynamic learning of the transition matrix. 

### Model 1: Asymmetric Learning Rates + Decay
This model separates learning from positive versus negative reward prediction errors. Humans often exhibit distinct learning rates when things go better than expected versus when they go worse than expected. 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Asymmetric Learning Rates, Choice Perseveration, and Unchosen Value Decay.
    Values of unchosen actions at both stages passively decay towards 0.5 over time.
    Updates depend on whether the prediction error is positive or negative.
    
    Bounds:
    lr_pos: [0, 1] - Learning rate for positive reward prediction errors
    lr_neg: [0, 1] - Learning rate for negative reward prediction errors
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weighting parameter for model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter for stage 1 updates
    pers: [-5, 5] - Choice perseveration (stickiness) for stage 1
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5
    """
    lr_pos, lr_neg, beta, w, lam, pers, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        # Handle missing trials
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = lr_pos if delta_stage1 > 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = lr_pos if delta_stage2 > 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2
        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2

        # Unchosen Decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Stage-Specific Perseveration + Decay
Participants don't just perseverate on their first-stage choices (spaceships); they often perseverate on their second-stage choices (aliens) as well. This model adds a state-specific perseveration parameter for the second stage.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Stage-Specific Choice Perseveration and Unchosen Value Decay.
    Includes independent stickiness parameters for Stage 1 and Stage 2 choices.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for updating Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weighting parameter for model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter for stage 1 updates
    pers_1: [-5, 5] - Choice perseveration (stickiness) for stage 1
    pers_2: [-5, 5] - Choice perseveration (stickiness) for stage 2 (state-specific)
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5
    """
    learning_rate, beta, w, lam, pers_1, pers_2, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # Tracks last action taken in each Stage 2 state

    for trial in range(n_trials):
        # Handle missing trials
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += pers_1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Choice
        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += pers_2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Unchosen Decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Transition Matrix Learning + Decay
Instead of assuming the participant inherently knows and trusts the fixed 70/30 transition probabilities, this model allows the participant to dynamically update their internal transition matrix based on their observations of rare vs. common transitions.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Transition Matrix Learning, Choice Perseveration, and Unchosen Value Decay.
    The model-based system learns the transition probabilities dynamically over time.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate for updating Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weighting parameter for model-based vs model-free values
    lam: [0, 1] - Eligibility trace parameter for stage 1 updates
    pers: [-5, 5] - Choice perseveration (stickiness) for stage 1
    decay: [0, 1] - Decay rate for unchosen Q-values towards 0.5
    lr_trans: [0, 1] - Learning rate for updating the transition matrix
    """
    learning_rate, beta, w, lam, pers, decay, lr_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action = -1

    for trial in range(n_trials):
        # Handle missing trials
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += pers
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 Choice
        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Transition matrix dynamic update
        transition_matrix[action_1[trial], state_idx] += lr_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += lr_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])

        # Value Updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Unchosen Decay
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
