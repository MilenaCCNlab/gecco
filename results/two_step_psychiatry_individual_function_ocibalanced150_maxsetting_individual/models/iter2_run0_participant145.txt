Here are 3 new cognitive models that extend the best-performing model by incorporating different psychological mechanisms. 

### Model 1: Asymmetric Learning Rates + Decay
This model adds asymmetric learning rates for positive and negative reward prediction errors to the decay mechanism. People often update their beliefs differently when they receive a better-than-expected outcome versus a worse-than-expected outcome.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MF/MB mixture with unchosen option decay and asymmetric learning rates.
    Learning rates differ for positive and negative reward prediction errors.
    
    Bounds:
    alpha_pos: [0,1] - learning rate for positive prediction errors
    alpha_neg: [0,1] - learning rate for negative prediction errors
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace
    rep: [-5,5] - stage 1 choice perseveration
    decay: [0,1] - decay rate for unchosen Q-values
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, rep, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep_term = np.zeros(2)
        if last_action != -1:
            rep_term[last_action] = 1
            
        logits1 = beta1 * q_net + rep * rep_term
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Dynamic Transition Matrix Learning
Instead of assuming the participant fully knows and fixes the 0.7 / 0.3 transition structure, this model allows the agent to dynamically learn the state transition probabilities over time.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MF/MB mixture with dynamic transition matrix learning and unchosen option decay.
    The agent actively learns the transition probabilities between stage 1 and stage 2.
    
    Bounds:
    alpha: [0,1] - learning rate for Q-values
    beta1: [0,10] - stage 1 inverse temperature
    beta2: [0,10] - stage 2 inverse temperature
    w: [0,1] - model-based weight
    lam: [0,1] - eligibility trace
    rep: [-5,5] - stage 1 choice perseveration
    decay: [0,1] - decay rate for unchosen Q-values
    alpha_tr: [0,1] - learning rate for transition matrix
    """
    alpha, beta1, beta2, w, lam, rep, decay, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]]) # Start with uniform transition priors
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep_term = np.zeros(2)
        if last_action != -1:
            rep_term[last_action] = 1
            
        logits1 = beta1 * q_net + rep * rep_term
        logits1 -= np.max(logits1)
        exp_q1 = np.exp(logits1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits2 = beta2 * q_stage2_mf[state_idx]
        logits2 -= np.max(logits2)
        exp_q2 = np.exp(logits2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        # Eligibility trace update
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        # Update transition matrix probabilities
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] = 1 - transition_matrix[action_1[trial], state_idx]
                    
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Separate Decay Rates for Stage 1 vs. Stage 2
This model posits that working memory constraints and forgetting affect the first stage (spaceships) and the second stage (aliens) at different rates.

    rep: [-5,5] - stage 1 choice perseveration
