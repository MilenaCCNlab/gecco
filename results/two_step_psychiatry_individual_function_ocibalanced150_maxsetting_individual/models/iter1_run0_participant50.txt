Here are 3 new cognitive models, each implementing a distinct mechanism to explain the participant's behavior on the two-step decision-making task. They introduce new parameterizations avoiding the combinations you've already tried.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model assumes that participants might update their values at different speeds for the first-stage (spaceships) and second-stage (aliens) choices. It uses `alpha_1` for updating the Stage 1 model-free values and `alpha_2` for updating Stage 2 values.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate learning rates for Stage 1 and Stage 2 updates.
    
    Parameters:
    alpha_1: [0, 1] - Learning rate for Stage 1 model-free updates.
    alpha_2: [0, 1] - Learning rate for Stage 2 updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Model-based weight (1 = pure model-based, 0 = pure model-free).
    eligibility: [0, 1] - Eligibility trace for updating stage 1 values from stage 2 rewards.
    stickiness: [-5, 5] - Choice stickiness (perseverance) for stage 1.
    """
    alpha_1, alpha_2, beta, w, eligibility, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        if a2 != -1:
            logits_2 = beta * q_stage2_mf[s]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            # Action value updating
            delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += alpha_1 * delta_stage1

            delta_stage2 = r - q_stage2_mf[s, a2]
            q_stage2_mf[s, a2] += alpha_2 * delta_stage2

            q_stage1_mf[a1] += eligibility * alpha_1 * delta_stage2
        else:
            p_choice_2[trial] = 1.0

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Value Forgetting (Decay) for Unchosen Options
Since the reward probabilities of aliens slowly drift over time, participants might naturally "forget" or decay the values of unchosen options toward an ambiguous baseline (0.5). This model introduces a `forgetting_rate` that decays unchosen Q-values.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Includes a forgetting rate for unchosen options.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating values.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Model-based weight (1 = pure model-based, 0 = pure model-free).
    eligibility: [0, 1] - Eligibility trace for updating stage 1 values from stage 2 rewards.
    stickiness: [-5, 5] - Choice stickiness (perseverance) for stage 1.
    forgetting_rate: [0, 1] - Decay rate for unchosen actions towards 0.5.
    """
    learning_rate, beta, w, eligibility, stickiness, forgetting_rate = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 (neutral expectation)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        if a2 != -1:
            logits_2 = beta * q_stage2_mf[s]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            # Action value updating
            delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = r - q_stage2_mf[s, a2]
            q_stage2_mf[s, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += eligibility * learning_rate * delta_stage2
            
            # Forgetting for unchosen stage 2 actions
            for state_idx in range(2):
                for action_idx in range(2):
                    if not (state_idx == s and action_idx == a2):
                        q_stage2_mf[state_idx, action_idx] = (1 - forgetting_rate) * q_stage2_mf[state_idx, action_idx] + forgetting_rate * 0.5
        else:
            p_choice_2[trial] = 1.0
            # Forgetting for all stage 2 actions if timeout occurred
            q_stage2_mf = (1 - forgetting_rate) * q_stage2_mf + forgetting_rate * 0.5

        # Forgetting for unchosen stage 1 action
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] = (1 - forgetting_rate) * q_stage1_mf[unchosen_a1] + forgetting_rate * 0.5

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Decoupled Inverse Temperatures (MB, MF, and Stage 2)
Instead of a weighting parameter `w`, this model parameterizes decision stochasticity by using separate inverse temperatures for the model-based system (`beta_mb`), the model-free system (`beta_mf`), and Stage 2 action selection (`beta_2`). This allows for differential sensitivity to the two systems rather than forcing them to trade off strictly.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Separate inverse temperatures for model-based, model-free, and stage 2.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating values.
    beta_mb: [0, 10] - Inverse temperature for model-based values at stage 1.
    beta_mf: [0, 10] - Inverse temperature for model-free values at stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2 action selection.
    eligibility: [0, 1] - Eligibility trace for updating stage 1 values from stage 2 rewards.
    stickiness: [-5, 5] - Choice stickiness (perseverance) for stage 1.
    """
    learning_rate, beta_mb, beta_mf, beta_2, eligibility, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Scaling MB and MF values by their respective temperatures
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        if a2 != -1:
            logits_2 = beta_2 * q_stage2_mf[s]
            logits_2 -= np.max(logits_2)
            exp_q2 = np.exp(logits_2)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]

            # Action value updating
            delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = r - q_stage2_mf[s, a2]
            q_stage2_mf[s, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += eligibility * learning_rate * delta_stage2
        else:
            p_choice_2[trial] = 1.0

        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
