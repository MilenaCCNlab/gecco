def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw 2011 Hybrid Model with Reward-Dependent Perseveration and decay to 0.5.
    Instead of a single stickiness parameter, this model has separate perseveration
    parameters depending on whether the previous trial was rewarded (p_win) or not (p_lose),
    capturing win-stay/lose-shift tendencies at the first stage.
    
    Bounds:
    learning_rate: [0, 1]
    decay_rate: [0, 1]
    beta: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p_win: [-5, 5]
    p_lose: [-5, 5]
    """
    learning_rate, decay_rate, beta, w, lam, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        p_weight = p_win if last_reward > 0 else p_lose
            
        exp_q1 = np.exp(beta * q_net + p_weight * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2

        # Decay towards 0.5
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates with Separate Stage Betas and Decay.
    Features separate learning rates for positive vs negative prediction errors,
    separate inverse temperatures (betas) for stage 1 and stage 2, and decay to 0.5.
    
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    decay_rate: [0, 1]
    beta_1: [0, 10]
    beta_2: [0, 10]
    w: [0, 1]
    lam: [0, 1]
    p: [-5, 5]
    """
    alpha_pos, alpha_neg, decay_rate, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta_1 * q_net + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        lr_stage1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        lr_stage2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

        new_q1_chosen = q_stage1_mf[action_1[trial]] + lr_stage1 * delta_stage1 + lr_stage1 * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + lr_stage2 * delta_stage2

        # Decay towards 0.5
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Independent MB and MF temperatures with decay to 0.5.
    Instead of a single inverse temperature and a weighting parameter w,
    this model uses separate inverse temperatures for the model-based (beta_mb)
    and model-free (beta_mf) values at stage 1, and a separate beta for stage 2.
    
    Bounds:
    learning_rate: [0, 1]
    decay_rate: [0, 1]
    beta_mb: [0, 10]
    beta_mf: [0, 10]
    beta_stage2: [0, 10]
    lam: [0, 1]
    p: [-5, 5]
    """
    learning_rate, decay_rate, beta_mb, beta_mf, beta_stage2, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = 1.0
            
        # Independent scaling of MB and MF values
        exp_q1 = np.exp(beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * pers)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_stage2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        new_q1_chosen = q_stage1_mf[action_1[trial]] + learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
        new_q2_chosen = q_stage2_mf[state_idx, action_2[trial]] + learning_rate * delta_stage2

        # Decay towards 0.5
        q_stage1_mf = q_stage1_mf + decay_rate * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay_rate * (0.5 - q_stage2_mf)

        q_stage1_mf[action_1[trial]] = new_q1_chosen
        q_stage2_mf[state_idx, action_2[trial]] = new_q2_chosen

        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
