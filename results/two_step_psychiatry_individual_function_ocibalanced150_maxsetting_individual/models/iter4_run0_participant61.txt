### Model 1: Asymmetric Learning with Outcome-Dependent Perseveration
This model tests whether participants update their values differently for positive versus negative prediction errors, and whether their tendency to stick to the same first-stage choice depends on if the previous trial was rewarded (win-stay) or unrewarded (lose-shift).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative prediction errors, 
    with outcome-dependent perseveration (win-stay/lose-shift) at stage 1.
    
    Bounds:
    alpha_pos: [0,1] (learning rate for positive PE)
    alpha_neg: [0,1] (learning rate for negative PE)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p_win: [-5,5] (stage 1 choice perseveration after rewarded trial)
    p_lose: [-5,5] (stage 1 choice perseveration after unrewarded trial)
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p_win, p_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward > 0:
                persev_1[last_action_1] = p_win
            else:
                persev_1[last_action_1] = p_lose
                
        exp_q1 = np.exp(beta_1 * q_net + persev_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        # eligibility trace
        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 2: Stage-Specific Forgetting Rates
This model investigates if unchosen options decay towards zero at different rates depending on the decision stage. A participant might hold onto the value of first-stage spaceships longer than the specific alien choices on planets.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Two-step model with separate decay rates for stage 1 and stage 2 unchosen options.
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p: [-5,5] (stage 1 choice perseveration)
    decay_1: [0,1] (decay rate of unchosen options at stage 1)
    decay_2: [0,1] (decay rate of unchosen options at stage 2)
    """
    alpha, beta_1, beta_2, w, lam, p, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            persev_1[last_action_1] = p
            
        exp_q1 = np.exp(beta_1 * q_net + persev_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Unchosen option decay
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1 - decay_1)
                
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != action_2[trial]:
                    q_stage2_mf[s, a] *= (1 - decay_2)
                    
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```


### Model 3: Transition-Dependent Perseveration
This model isolates whether experiencing a common transition versus a rare transition drives the participant to perseverate differently on the subsequent trial. Experiencing an unexpected environmental shift (rare transition) might disrupt normal "autopilot" choice repetition.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-dependent perseveration at stage 1, with unchosen Q-value decay.
    
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (stage 1 inverse temperature)
    beta_2: [0,10] (stage 2 inverse temperature)
    w: [0,1] (weighting of model-based vs model-free values)
    lam: [0,1] (eligibility trace parameter)
    p_common: [-5,5] (stage 1 perseveration after a common transition)
    p_rare: [-5,5] (stage 1 perseveration after a rare transition)
    decay: [0,1] (decay rate of unchosen options)
    """
    alpha, beta_1, beta_2, w, lam, p_common, p_rare, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_transition_common = False

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        persev_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_transition_common:
                persev_1[last_action_1] = p_common
            else:
                persev_1[last_action_1] = p_rare
                
        exp_q1 = np.exp(beta_1 * q_net + persev_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Unchosen option decay
        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] *= (1 - decay)
                
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != action_2[trial]:
                    q_stage2_mf[s, a] *= (1 - decay)
                    
        last_action_1 = action_1[trial]
        # Spaceship 0 commonly goes to planet 0; Spaceship 1 to planet 1
        last_transition_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
