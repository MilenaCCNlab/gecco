Here are 3 new cognitive models that build upon the provided template. Each model introduces a different psychological mechanism to explain the participant's choices, particularly focusing on how they make decisions and update values at the second stage (since the first stage policy is fixed by the template to be purely model-based).

### Model 1: Q-learning with Stage 2 Choice Stickiness (Perseveration)
Participants often exhibit a tendency to repeat their previous actions regardless of the reward outcome, a behavior known as choice stickiness or perseveration. This model adds a `stickiness` parameter to the stage 2 policy. If the participant previously visited the current planet, they receive a bonus to the logit of the alien they chose last time.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-learning with stage 2 choice stickiness (perseveration).
    This model adds a stickiness parameter to the stage 2 choice policy, 
    making the participant more likely to repeat the last action taken in the current state,
    independent of the Q-value.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice at stage 1 and baseline for stage 2
    stickiness: [-5,5] - choice perseveration bonus for repeating the last action at stage 2
    """
    learning_rate, beta, stickiness = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Track the last action taken at stage 2 for each state (-1 indicates no previous action)
    last_action = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits = beta * q_stage2_mf[state_idx].copy()
        if last_action[state_idx] != -1:
            logits[last_action[state_idx]] += stickiness
            
        # Subtract max for numerical stability before exponentiation
        logits -= np.max(logits)
        exp_q2 = np.exp(logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Update last action for the current state
        last_action[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Q-learning with Counterfactual (Fictitious) Updating
Participants may assume that the reward probabilities of the two aliens on a given planet are anti-correlated (e.g., if Alien W gives a coin, Alien S probably won't). This model introduces a `counterfactual_lr` that updates the unchosen alien on the visited planet using a fictitious reward (`1 - actual_reward`).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Q-learning with counterfactual updating at stage 2.
    This model assumes that participants not only update the value of the chosen alien 
    but also update the unchosen alien in the opposite direction, reflecting an assumption 
    that reward probabilities of aliens on the same planet are anti-correlated.
    
    Parameters:
    learning_rate: [0,1] - learning rate for the chosen option
    beta: [0,10] - inverse temperature for softmax choice
    counterfactual_lr: [0,1] - learning rate applied to the unchosen option using (1 - reward)
    """
    learning_rate, beta, counterfactual_lr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # action value updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # action value updating for choice 2 (chosen option)
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # Counterfactual update for the unchosen action at stage 2
        unchosen_action = 1 - action_2[trial]
        cf_reward = 1 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_action]
        q_stage2_mf[state_idx, unchosen_action] += counterfactual_lr * delta_cf

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
