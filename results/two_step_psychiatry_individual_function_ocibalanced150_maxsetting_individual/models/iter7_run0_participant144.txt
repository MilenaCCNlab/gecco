Here are 3 new cognitive models that explore different combinations of learning mechanisms, choice perseveration, and asymmetric learning. Each model is encapsulated in a standalone Python function and is parameterized to capture distinct facets of decision-making behavior.

### Model 1: Separate Learning Rates and Temperatures for Each Stage
This model extends the classical Daw et al. (2011) framework by uncoupling both the learning rate and the inverse temperature (beta) between Stage 1 and Stage 2. This allows for the possibility that a participant updates their values at different speeds and explores/exploits at different rates for the spaceship versus the alien choices.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Comprehensive Daw 2011 model with separate learning rates and betas for both stages.
    
    Parameters:
    lr_1: [0, 1] - Learning rate for Stage 1.
    lr_2: [0, 1] - Learning rate for Stage 2.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    eligibility: [0, 1] - Eligibility trace (lambda) for Stage 1 update from Stage 2 prediction error.
    stickiness_1: [-5, 5] - Bonus added to the Q-value of the previous Stage 1 choice.
    stickiness_2: [-5, 5] - Bonus added to the Q-value of the previous Stage 2 choice.
    """
    lr_1, lr_2, beta_1, beta_2, w, eligibility, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q_stage1_net.copy()
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        exp_q1 = np.exp(beta_1 * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta_2 * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += eligibility * lr_1 * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Reward-Dependent Perseveration (Win-Stay, Lose-Shift)
Instead of a simple "stickiness" parameter that blindly encourages repeating the previous action, this model checks whether the previous trial was rewarded. If it was, the participant gets a `stay_win` bonus to stick with the previous spaceship. If they weren't rewarded, they get a `shift_lose` bonus to switch to the alternative spaceship. 

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with reward-dependent perseveration (Win-Stay, Lose-Shift) at Stage 1.
    
    Parameters:
    learning_rate: [0, 1] - Shared learning rate for both stages.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    eligibility: [0, 1] - Eligibility trace (lambda).
    stay_win: [-5, 5] - Bonus to previous Stage 1 action if the previous trial was rewarded.
    shift_lose: [-5, 5] - Bonus to alternative Stage 1 action if the previous trial was unrewarded.
    stickiness_2: [-5, 5] - Choice perseveration for Stage 2.
    """
    learning_rate, beta_1, beta_2, w, eligibility, stay_win, shift_lose, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_r = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q_stage1_net.copy()
        if prev_a1 != -1:
            if prev_r == 1:
                q1_eff[prev_a1] += stay_win
            elif prev_r == 0:
                q1_eff[1 - prev_a1] += shift_lose
            
        exp_q1 = np.exp(beta_1 * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta_2 * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        q_stage1_mf[action_1[trial]] += eligibility * learning_rate * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_r = reward[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates for Stage 2 Prediction Errors
Participants often learn differently from positive and negative outcomes. This model assigns separate learning rates for positive vs. negative prediction errors at Stage 2 (the stage where rewards are actually obtained), while maintaining a standard learning rate for Stage 1. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates for positive/negative prediction errors at Stage 2.
    
    Parameters:
    lr_1: [0, 1] - Learning rate for Stage 1.
    lr_pos_2: [0, 1] - Learning rate for Stage 2 positive prediction errors.
    lr_neg_2: [0, 1] - Learning rate for Stage 2 negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for Stage 1.
    beta_2: [0, 10] - Inverse temperature for Stage 2.
    w: [0, 1] - Weighting between model-based (1) and model-free (0) Stage 1 values.
    stickiness_1: [-5, 5] - Choice perseveration for Stage 1.
    stickiness_2: [-5, 5] - Choice perseveration for Stage 2.
    """
    lr_1, lr_pos_2, lr_neg_2, beta_1, beta_2, w, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_eff = q_stage1_net.copy()
        if prev_a1 != -1:
            q1_eff[prev_a1] += stickiness_1
            
        exp_q1 = np.exp(beta_1 * q1_eff)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # Policy for the second choice
        q2_eff = q_stage2_mf[state_idx, :].copy()
        if prev_a2[state_idx] != -1:
            q2_eff[prev_a2[state_idx]] += stickiness_2
            
        exp_q2 = np.exp(beta_2 * q2_eff)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        # Asymmetric update for Stage 2
        if delta_stage2 > 0:
            q_stage2_mf[state_idx, action_2[trial]] += lr_pos_2 * delta_stage2
        else:
            q_stage2_mf[state_idx, action_2[trial]] += lr_neg_2 * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
