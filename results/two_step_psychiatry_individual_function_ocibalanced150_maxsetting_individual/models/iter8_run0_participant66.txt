Here are 3 new cognitive models that build upon the most successful mechanisms (such as separating model-based and model-free inverse temperatures) while exploring new structural variations that have not yet been evaluated.

### Model 1: Separate Stage Learning Rates
This model explores the idea that participants might update their values at different speeds for the first and second stages. It introduces separate learning rates (`alpha_1` and `alpha_2`) for stage 1 and stage 2, while retaining the separate model-based and model-free inverse temperatures and independent perseveration tendencies.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model with separate learning rates for stage 1 and stage 2, separate inverse temperatures for MB and MF at stage 1, and separate perseveration for stage 1 and stage 2.
    
    Bounds:
    alpha_1: [0, 1] - learning rate for stage 1 Q-values
    alpha_2: [0, 1] - learning rate for stage 2 Q-values
    beta_mb: [0, 10] - inverse temperature for model-based Q-values at stage 1
    beta_mf: [0, 10] - inverse temperature for model-free Q-values at stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice perseveration for stage 1
    pers_2: [-2, 2] - choice perseveration for stage 2
    """
    alpha_1, alpha_2, beta_mb, beta_mf, beta_2, lambd, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers_1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_a2 != -1:
                pers_bonus_2[last_a2] = pers_2
                
            logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_1 * lambd * delta_stage2

        last_a1 = action_1[trial]
        last_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning (Positive vs. Negative RPEs)
This model posits that learning is asymmetric; participants might weight positive reward prediction errors (better-than-expected outcomes) differently from negative reward prediction errors (worse-than-expected outcomes). It uses `alpha_pos` and `alpha_neg` to scale updates based on the sign of the prediction error.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model with separate learning rates for positive and negative RPEs, separate inverse temperatures for MB and MF at stage 1, and separate perseveration for stage 1 and stage 2.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive RPEs
    alpha_neg: [0, 1] - learning rate for negative RPEs
    beta_mb: [0, 10] - inverse temperature for model-based Q-values at stage 1
    beta_mf: [0, 10] - inverse temperature for model-free Q-values at stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice perseveration for stage 1
    pers_2: [-2, 2] - choice perseveration for stage 2
    """
    alpha_pos, alpha_neg, beta_mb, beta_mf, beta_2, lambd, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers_1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_a2 != -1:
                pers_bonus_2[last_a2] = pers_2
                
            logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
            q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_stage2 * lambd * delta_stage2

        last_a1 = action_1[trial]
        last_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Value Decay with Lapse Rate
This model integrates Q-value decay towards 0.5 (as seen in the best model) but replaces the separate perseveration parameters with a single, generalized perseveration term (`pers`) paired with an irreducible "lapse rate". A lapse rate allows the model to account for trials where participants might simply guess entirely at random, providing a robust ceiling for the likelihood of completely random choices.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model with Q-value decay, separate inverse temperatures for MB and MF at stage 1, a single perseveration parameter for both stages, and a lapse rate.
    
    Bounds:
    alpha: [0, 1] - learning rate
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0.5)
    beta_mb: [0, 10] - inverse temperature for model-based Q-values at stage 1
    beta_mf: [0, 10] - inverse temperature for model-free Q-values at stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    lambd: [0, 1] - eligibility trace
    pers: [-2, 2] - choice perseveration for both stages
    lapse: [0, 1] - random choice probability
    """
    alpha, decay, beta_mb, beta_mf, beta_2, lambd, pers, lapse = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        # apply lapse
        probs_1 = (1 - lapse) * probs_1 + lapse * 0.5
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_a2 != -1:
                pers_bonus_2[last_a2] = pers
                
            logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            
            # apply lapse
            probs_2 = (1 - lapse) * probs_2 + lapse * 0.5
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
        last_a1 = action_1[trial]
        last_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
