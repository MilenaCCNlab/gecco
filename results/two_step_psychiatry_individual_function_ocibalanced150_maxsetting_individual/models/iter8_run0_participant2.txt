def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative RPEs, with state-specific Stage 2 perseveration.
    This model assumes that participants might update their values differently depending on whether the 
    prediction error is positive (better than expected) or negative (worse than expected). 
    It also includes choice perseveration for Stage 1 and state-specific perseveration for Stage 2.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive RPEs
    alpha_neg: [0,1] - Learning rate for negative RPEs
    beta_1: [0,10] - Inverse temperature for stage 1 choices
    beta_2: [0,10] - Inverse temperature for stage 2 choices
    w: [0,1] - Model-based vs Model-free weight
    lam: [0,1] - Eligibility trace parameter
    pers_1: [-5,5] - Choice perseveration bonus for stage 1
    pers_2_state: [-5,5] - State-specific choice perseveration bonus for stage 2
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers_1, pers_2_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    last_a2_in_state = np.full(2, -1)

    for trial in range(n_trials):
        # Handle missed trials
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            continue

        # Stage 1 choice probabilities
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if prev_a1 != -1:
            pers_bonus_1[prev_a1] = pers_1
            
        logits_1 = beta_1 * q_stage1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Stage 2 choice probabilities
        pers_bonus_2 = np.zeros(2)
        if last_a2_in_state[state_idx] != -1:
            pers_bonus_2[last_a2_in_state[state_idx]] = pers_2_state

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_s1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_s2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2
        
        # Update history
        prev_a1 = action_1[trial]
        last_a2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent model-based weight (w), with state-specific Stage 2 perseveration.
    This model introduces the assumption that the reliance on the model-based system (w) is modulated 
    by the previous trial's outcome. Participants might become more model-based following a reward 
    or unrewarded trial depending on their strategy.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_1: [0,10] - Inverse temperature for stage 1 choices
    beta_2: [0,10] - Inverse temperature for stage 2 choices
    w_win: [0,1] - Model-based weight if previous trial was rewarded
    w_lose: [0,1] - Model-based weight if previous trial was unrewarded
    lam: [0,1] - Eligibility trace parameter
    pers_1: [-5,5] - Choice perseveration bonus for stage 1
    pers_2_state: [-5,5] - State-specific choice perseveration bonus for stage 2
    """
    alpha, beta_1, beta_2, w_win, w_lose, lam, pers_1, pers_2_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    last_a2_in_state = np.full(2, -1)
    prev_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            prev_a1 = -1
            prev_reward = -1
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        if prev_reward == 1:
            w_current = w_win
        elif prev_reward == 0:
            w_current = w_lose
        else:
            w_current = (w_win + w_lose) / 2.0
            
        q_stage1_net = w_current * q_stage1_mb + (1 - w_current) * q_stage1_mf

        pers_bonus_1 = np.zeros(2)
        if prev_a1 != -1:
            pers_bonus_1[prev_a1] = pers_1
            
        logits_1 = beta_1 * q_stage1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        pers_bonus_2 = np.zeros(2)
        if last_a2_in_state[state_idx] != -1:
            pers_bonus_2[last_a2_in_state[state_idx]] = pers_2_state

        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        last_a2_in_state[state_idx] = action_2[trial]
        prev_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
