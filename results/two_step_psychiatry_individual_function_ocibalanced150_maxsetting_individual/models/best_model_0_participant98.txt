def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Reward-dependent stickiness (Win-Stay / Lose-Shift) for stage 1, with unchosen Q-values decaying to a center.
    Uses a single inverse temperature for both stages.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature for both stages
    w: [0,1] - Weight of model-based vs model-free control
    lam: [0,1] - Eligibility trace parameter
    st_rew: [-5,5] - Choice stickiness for stage 1 if previous trial was rewarded
    st_unrew: [-5,5] - Choice stickiness for stage 1 if previous trial was unrewarded
    decay: [0,1] - Decay rate for unchosen actions
    center: [0,1] - The value to which unchosen actions decay
    """
    alpha, beta, w, lam, st_rew, st_unrew, decay, center = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, center)
    q_stage2_mf = np.full((2, 2), center)
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_bonus[last_action_1] = st_rew
            else:
                stickiness_bonus[last_action_1] = st_unrew
            
        q1_val = beta * q_net + stickiness_bonus
        q1_val -= np.max(q1_val) 
        exp_q1 = np.exp(q1_val)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_val = beta * q_stage2_mf[state_idx]
        q2_val -= np.max(q2_val) 
        exp_q2 = np.exp(q2_val)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] = q_stage1_mf[unchosen_1] * (1 - decay) + center * decay
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] = q_stage2_mf[s, a] * (1 - decay) + center * decay
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
