Here are 3 new cognitive models that build upon your best model by introducing novel mechanisms such as reward-dependent perseveration, asymmetric learning for positive/negative prediction errors, and memory decay for unchosen options. 

### Model 1: Reward-Dependent Perseveration (Win-Stay / Lose-Shift)
This model hypothesizes that the tendency to repeat a Stage 1 choice is heavily dependent on whether the previous trial was rewarded. Instead of a single perseveration parameter, it uses `p_win` (applied if the last trial was rewarded) and `p_lose` (applied if the last trial was unrewarded).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with reward-dependent choice perseveration for Stage 1.
    The perseveration bonus at Stage 1 depends on whether the previous trial was rewarded.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2: learning rate for stage 2 [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: weight of model-based vs model-free [0, 1]
    lam: eligibility trace [0, 1]
    p_win: choice perseveration for stage 1 if previous trial was rewarded [0, 10]
    p_lose: choice perseveration for stage 1 if previous trial was unrewarded [0, 10]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, p_win, p_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        p_bonus = 0
        if last_reward == 1:
            p_bonus = p_win
        elif last_reward == 0:
            p_bonus = p_lose
            
        logits_1 = beta_1 * q_stage1_net + p_bonus * rep_1
        logits_1 -= np.max(logits_1) # Prevent overflow
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2) # Prevent overflow
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates + Stage-Specific Perseveration
This model tests whether participants learn differently from positive versus negative prediction errors globally across both stages, while maintaining the successful separation of perseveration tendencies between stages.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with asymmetric learning rates for positive and negative 
    prediction errors, and separate choice perseveration for Stage 1 and Stage 2.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: weight of model-based vs model-free [0, 1]
    lam: eligibility trace [0, 1]
    p_1: choice perseveration for stage 1 [0, 10]
    p_2: choice perseveration for stage 2 [0, 10]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        rep_2 = np.zeros(2)
        if last_state == state_idx and last_action_2 != -1:
            rep_2[last_action_2] = 1
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + p_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha_stage1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Unchosen Option Decay (Forgetting)
This model simplifies to a single global learning rate but introduces a `decay` parameter. In this setup, values of actions that are *not* chosen slowly decay toward zero, simulating memory decay or "forgetting" over time, which commonly influences long sequences of identical choices.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid MB/MF model with a single learning rate, decay for unchosen options, 
    and separate choice perseveration for Stage 1 and Stage 2.
    
    Parameters:
    alpha: learning rate [0, 1]
    decay: decay rate for unchosen Q-values towards 0 [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: weight of model-based vs model-free [0, 1]
    lam: eligibility trace [0, 1]
    p_1: choice perseveration for stage 1 [0, 10]
    p_2: choice perseveration for stage 2 [0, 10]
    """
    alpha, decay, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1
    
    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_stage1_net + p_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        # policy for the second choice
        rep_2 = np.zeros(2)
        if last_state == state_idx and last_action_2 != -1:
            rep_2[last_action_2] = 1
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + p_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        # decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)
        
        # value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2 = action_2[trial]
        last_state = state_idx
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
