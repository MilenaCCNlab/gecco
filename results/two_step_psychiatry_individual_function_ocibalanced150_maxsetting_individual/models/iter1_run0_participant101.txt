def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) Hybrid Model with separate learning rates for Stage 1 and Stage 2.
    
    Parameters:
    lr_stage1: [0,1] - learning rate for stage 1 Q-values
    lr_stage2: [0,1] - learning rate for stage 2 Q-values
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control (1 = fully MB)
    lam: [0,1] - eligibility trace parameter
    p: [-2,2] - choice perseverance bonus (tendency to repeat previous stage 1 choice)
    """
    lr_stage1, lr_stage2, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # Action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += lr_stage1 * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += lr_stage2 * delta_stage2

            q_stage1_mf[a1] += lr_stage1 * lam * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw et al. (2011) Hybrid Model with Forgetting (decay of Q-values for unchosen options).
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    decay_rate: [0,1] - rate at which unchosen Q-values decay towards 0.5
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control (1 = fully MB)
    lam: [0,1] - eligibility trace parameter
    p: [-2,2] - choice perseverance bonus (tendency to repeat previous stage 1 choice)
    """
    learning_rate, decay_rate, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    # Initialize values at 0.5, the expected value of a random reward
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers = np.zeros(2)
        if last_action_1 != -1:
            pers[last_action_1] = p
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # Action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            
            # Decay unchosen stage 1 action value towards 0.5
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay_rate * (0.5 - q_stage1_mf[unchosen_a1])
            
            # Update chosen stage 1 action value
            q_stage1_mf[a1] += learning_rate * delta_stage1 + learning_rate * lam * delta_stage2
            
            # Decay unchosen stage 2 action values towards 0.5
            for s in range(2):
                for a in range(2):
                    if not (s == state_idx and a == a2):
                        q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])
                        
            # Update chosen stage 2 action value
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw et al. (2011) Hybrid Model with separate perseverance for Stage 1 and Stage 2.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta1: [0,10] - inverse temperature for stage 1 choices
    beta2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - mixing weight for model-based vs model-free control (1 = fully MB)
    lam: [0,1] - eligibility trace parameter
    p1: [-2,2] - choice perseverance bonus for stage 1 (tendency to repeat previous stage 1 choice)
    p2: [-2,2] - choice perseverance bonus for stage 2 (tendency to repeat previous stage 2 choice in the same state)
    """
    learning_rate, beta1, beta2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_action_1 != -1:
            pers1[last_action_1] = p1
            
        exp_q1 = np.exp(beta1 * (q_stage1_net + pers1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # policy for the second choice
        if state_idx != -1 and a2 != -1:
            pers2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers2[last_action_2[state_idx]] = p2
                
            exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] + pers2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # Action value updating
        if state_idx != -1 and a2 != -1 and a1 != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            q_stage1_mf[a1] += learning_rate * delta_stage1

            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
            q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

            q_stage1_mf[a1] += learning_rate * lam * delta_stage2
            
        if a1 != -1:
            last_action_1 = a1
        if state_idx != -1 and a2 != -1:
            last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
