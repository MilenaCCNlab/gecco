def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Mixed MF/MB with reward-dependent stage 1 stickiness and stage 2 stickiness.
    Separates choice stickiness based on whether the previous trial was rewarded 
    (win-stay) or unrewarded (lose-shift), and adds a stickiness parameter for stage 2 choices.
    
    Parameters:
    alpha: [0, 1] - Learning rate for Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0, 1] - Eligibility trace (how much stage 2 reward updates stage 1 MF value).
    rep_win: [-5, 5] - Stickiness for stage 1 if the previous trial was rewarded.
    rep_lose: [-5, 5] - Stickiness for stage 1 if the previous trial was unrewarded.
    rep_stage2: [-5, 5] - Stickiness for stage 2 choices (repeating the same action in the same state).
    """
    alpha, beta_1, beta_2, w, lam, rep_win, rep_lose, rep_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_1[last_action_1] = rep_win
            elif last_reward == 0:
                stickiness_1[last_action_1] = rep_lose
                
        q1_logits = beta_1 * q_stage1_net + stickiness_1
        q1_logits -= np.max(q1_logits)
        exp_q1 = np.exp(q1_logits)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = rep_stage2
            
        q2_logits = beta_2 * q_stage2_mf[state_idx] + stickiness_2
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Asymmetric learning rates for positive and negative reward prediction errors.
    Includes stage 1 and stage 2 stickiness parameters.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_1: [0, 10] - Inverse temperature for stage 1 choices.
    beta_2: [0, 10] - Inverse temperature for stage 2 choices.
    w: [0, 1] - Model-based weight (1 = fully model-based, 0 = fully model-free).
    lam: [0, 1] - Eligibility trace.
    rep_stage1: [-5, 5] - General stickiness for stage 1 choice.
    rep_stage2: [-5, 5] - General stickiness for stage 2 choice within a state.
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, rep_stage1, rep_stage2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        q1_logits -= np.max(q1_logits)

        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = rep_stage2
            
        q2_logits = beta_2 * q_stage2_mf[state_idx] + stickiness_2
        q2_logits -= np.max(q2_logits)
        exp_q2 = np.exp(q2_logits)
        probs_2 = exp_q2 / np.sum(exp_q2)

        # action value updating for choice 1

        q_stage1_mf[action_1[trial]] += alpha_s2 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
