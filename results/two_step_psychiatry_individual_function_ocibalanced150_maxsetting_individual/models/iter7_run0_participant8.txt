Here are 3 new cognitive models that explore alternative mechanisms, specifically focusing on dynamic transition matrix learning and counterfactual updating for unchosen options.

### Model 1: Hybrid Model with Transition Matrix Learning
This model introduces `alpha_tr`, allowing the agent to dynamically update its internal representation of the transition matrix (the probabilities of reaching each planet from each spaceship). Rather than assuming fixed transitions throughout the task, the agent updates these probabilities based on experienced transitions.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Based and Model-Free model with Transition Matrix Learning.
    The agent learns the transition probabilities between stage 1 actions and stage 2 states dynamically.
    
    Parameters:
    learning_rate: learning rate [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    rep: choice stickiness [-5,5]
    alpha_tr: learning rate for transition matrix [0,1]
    """
    learning_rate, beta, w, lam, rep, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            last_action = -1
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Model-based evaluation
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Integrate MB and MF values
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        # First choice policy
        q1_vals = beta * (q_stage1_net + rep * stickiness)
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # Second choice policy
        q2_vals = beta * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # Update Q-values
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Update transition matrix probabilities based on experienced state
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid Model with Transition Learning and Memory Decay
This model combines the best-performing decay mechanism for unchosen options with the dynamic transition matrix learning introduced in Model 1. 

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model with Transition Matrix Learning and Value Decay.
    Combines Q-value decay for unchosen options with dynamic learning of the transition matrix.
    
    Parameters:
    learning_rate: learning rate [0,1]
    decay_rate: decay rate for unchosen options [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    rep: choice stickiness [-5,5]
    alpha_tr: learning rate for transition matrix [0,1]
    """
    learning_rate, decay_rate, beta, w, lam, rep, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            last_action = -1
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta * (q_stage1_net + rep * stickiness)
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_vals = beta * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen options
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - decay_rate)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay_rate)

        # Update transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Counterfactual Updating for Unchosen Stage 2 Actions
This model introduces `kappa`, a counterfactual learning rate modifier. Because the task structure often implies anti-correlated rewards between aliens on the same planet, the agent assumes that if one alien gave a reward, the other likely wouldn't have (and vice-versa). It updates the unchosen alien on the visited planet towards `1 - reward`.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model with Decay and Counterfactual Updating.
    Updates the unchosen stage 2 action on the visited planet assuming anti-correlated rewards.
    
    Parameters:
    learning_rate: learning rate [0,1]
    decay_rate: decay rate for unchosen options [0,1]
    beta: inverse temperature [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    rep: choice stickiness [-5,5]
    kappa: counterfactual learning rate modifier [0,1]
    """
    learning_rate, decay_rate, beta, w, lam, rep, kappa = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            last_action = -1
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta * (q_stage1_net + rep * stickiness)
        q1_vals -= np.max(q1_vals)
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        q2_vals = beta * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Counterfactual update for the unchosen action in the same state
        unchosen_action_2 = 1 - action_2[trial]
        cf_reward = 1.0 - reward[trial]
        delta_cf = cf_reward - q_stage2_mf[state_idx, unchosen_action_2]
        q_stage2_mf[state_idx, unchosen_action_2] += learning_rate * kappa * delta_cf

        # Decay unchosen options (only on the unvisited planet)
        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] *= (1 - decay_rate)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]) and not (s == state_idx and a == unchosen_action_2):
                    q_stage2_mf[s, a] *= (1 - decay_rate)
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
