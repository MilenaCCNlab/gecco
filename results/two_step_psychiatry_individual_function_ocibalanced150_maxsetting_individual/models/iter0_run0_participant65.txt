Here are three cognitive models that build upon the provided template. Because the template uses a purely model-based policy for the first choice (`q_stage1_mb`), any meaningful cognitive parameters must affect the second choice's policy or the updating of `q_stage2_mf` (which then propagates back to the first choice via the transition matrix). 

Additionally, the data contains missing trials (e.g., Trial 113 where the action and state are `-1`). The models include robust checks in the `FILL IN` sections to safely handle these missing trials without modifying the required template structure.

### Model 1: Asymmetric Learning Rates
This model implements different learning rates for positive and negative prediction errors. This captures a common cognitive bias where participants update their beliefs differently for better-than-expected outcomes (positive prediction errors) versus worse-than-expected outcomes (negative prediction errors).

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-learning with asymmetric learning rates.
    This model uses different learning rates for positive and negative prediction errors,
    capturing differential sensitivity to better-than-expected vs worse-than-expected outcomes.
    
    Bounds:
    lr_pos: [0,1] (learning rate for positive prediction errors)
    lr_neg: [0,1] (learning rate for negative prediction errors)
    beta: [0,10] (inverse temperature for softmax)
    """
    lr_pos, lr_neg, beta = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_1[trial] == -1 or action_2[trial] == -1 or state_idx == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
        else:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            lr1 = lr_pos if delta_stage1 > 0 else lr_neg
            q_stage1_mf[action_1[trial]] += lr1 * delta_stage1
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            lr2 = lr_pos if delta_stage2 > 0 else lr_neg
            q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unchosen Option Decay (Forgetting)
This model introduces a decay parameter. While the chosen action's value is updated via standard Q-learning, the values of all unchosen actions (and unvisited states) decay toward zero. This accounts for memory decay over time when an option is not actively sampled.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Q-learning with decay (forgetting) for unchosen options.
    Values of unchosen actions decay towards zero, modeling memory decay.
    
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    decay: [0,1] (rate at which unchosen options decay towards 0)
    """
    learning_rate, beta, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_1[trial] == -1 or action_2[trial] == -1 or state_idx == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
        else:
            exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            unchosen_a1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_a1] *= (1 - decay)
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            unchosen_a2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_a2] *= (1 - decay)
            
            unvisited_s = 1 - state_idx
            q_stage2_mf[unvisited_s, :] *= (1 - decay)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Stage 2 Choice Stickiness (Perseveration)
Participants often exhibit a tendency to repeat their previous actions independently of the rewards received. This model adds a stickiness parameter (`pi`) that boosts the probability of selecting the same stage 2 action (alien) that was last chosen in that particular state (planet).

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Q-learning with choice stickiness (perseverance) in Stage 2.
    A stickiness parameter temporarily inflates the value of the last action 
    chosen in the current stage 2 state, making it more likely to be repeated.
    
    Bounds:
    learning_rate: [0,1]
    beta: [0,10]
    pi: [0,5] (stickiness/perseveration bonus)
    """
    learning_rate, beta, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    # Track the last action taken in each of the two stage 2 states
    last_action_stage2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_1[trial] == -1 or action_2[trial] == -1 or state_idx == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
        else:
            q2_net = np.copy(q_stage2_mf[state_idx])
            if last_action_stage2[state_idx] != -1:
                q2_net[last_action_stage2[state_idx]] += pi
                
            exp_q2 = np.exp(beta * q2_net)
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # Action value updating for choice 1
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # Action value updating for choice 2
        if action_1[trial] != -1 and action_2[trial] != -1 and state_idx != -1:
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            last_action_stage2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
