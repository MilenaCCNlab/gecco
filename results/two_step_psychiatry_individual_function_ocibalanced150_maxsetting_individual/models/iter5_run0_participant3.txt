def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid RL with Continuous Choice Traces
    
    Instead of only perseverating on the immediate last choice, this model maintains
    a continuous choice trace (habit) for both stages, updated with a specific rate alpha_c.
    This captures long streaks of the same choice, as seen in the participant's data.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam: [0, 1] - Eligibility trace parameter for stage 1 model-free update.
    p1: [-5, 5] - Perseveration weight for stage 1 choice trace.
    p2: [-5, 5] - Perseveration weight for stage 2 choice trace.
    alpha_c: [0, 1] - Learning rate / update rate for the choice traces.
    """
    learning_rate, beta, w, lam, p1, p2, alpha_c = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace_1 = np.zeros(2)
    choice_trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        logits_1 = beta * q_stage1_net + p1 * choice_trace_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        logits_2 = beta * q_stage2_mf[state_idx] + p2 * choice_trace_2[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Q-value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        # Update choice traces
        choice_trace_1 = (1 - alpha_c) * choice_trace_1
        choice_trace_1[a1] += alpha_c
        
        choice_trace_2[state_idx] = (1 - alpha_c) * choice_trace_2[state_idx]
        choice_trace_2[state_idx, a2] += alpha_c

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid RL with Stage-Specific Learning Rates and Temperatures
    
    This model allows for different learning rates and inverse temperatures 
    for Stage 1 and Stage 2, acknowledging that learning and decision noise 
    might differ between the abstract spaceship choice and the concrete alien choice.
    
    Parameters:
    lr1: [0, 1] - Learning rate for Stage 1 Q-value updates.
    lr2: [0, 1] - Learning rate for Stage 2 Q-value updates.
    beta1: [0, 10] - Inverse temperature for Stage 1 action selection.
    beta2: [0, 10] - Inverse temperature for Stage 2 action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam: [0, 1] - Eligibility trace parameter for stage 1 model-free update.
    p1: [-5, 5] - Perseveration parameter for stage 1 choice.
    p2: [-5, 5] - Perseveration parameter for stage 2 choice (specific to state).
    """
    lr1, lr2, beta1, beta2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_a1 != -1:
            pers1[last_a1] = 1.0
            
        logits_1 = beta1 * q_stage1_net + p1 * pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        pers2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            pers2[last_a2[state_idx]] = 1.0

        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Q-value updates with stage-specific learning rates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += lr1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += lr2 * delta_stage2

        q_stage1_mf[a1] += lr1 * lam * delta_stage2
        
        last_a1 = a1
        last_a2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid RL with Transition-Dependent Stage 1 Perseveration
    
    This model posits that perseveration at Stage 1 depends on whether the 
    previous trial's transition was common or rare. A rare transition might 
    break the tendency to repeat the same spaceship choice.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-value updates.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w: [0, 1] - Weighting parameter mixing model-based (1) and model-free (0) values.
    lam: [0, 1] - Eligibility trace parameter for stage 1 model-free update.
    p_common: [-5, 5] - Perseveration parameter for stage 1 if previous transition was common.
    p_rare: [-5, 5] - Perseveration parameter for stage 1 if previous transition was rare.
    p2: [-5, 5] - Perseveration parameter for stage 2 choice (specific to state).
    """
    learning_rate, beta, w, lam, p_common, p_rare, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = np.array([-1, -1])
    last_trans_common = False

    for trial in range(n_trials):
        a1 = action_1[trial]
        state_idx = state[trial]
        a2 = action_2[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        pers1 = np.zeros(2)
        if last_a1 != -1:
            if last_trans_common:
                pers1[last_a1] = p_common
            else:
                pers1[last_a1] = p_rare
            
        logits_1 = beta * q_stage1_net + pers1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        pers2 = np.zeros(2)
        if last_a2[state_idx] != -1:
            pers2[last_a2[state_idx]] = 1.0

        logits_2 = beta * q_stage2_mf[state_idx] + p2 * pers2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Q-value updates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += learning_rate * delta_stage2

        q_stage1_mf[a1] += learning_rate * lam * delta_stage2
        
        last_a1 = a1
        last_a2[state_idx] = a2
        # A transition is common if the spaceship chosen matches the planet arrived at
        last_trans_common = (a1 == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
