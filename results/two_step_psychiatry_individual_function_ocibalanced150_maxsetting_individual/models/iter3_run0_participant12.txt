def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Standard Daw et al. (2011) Two-Step Task Model.
    Features separate learning rates and inverse temperatures for stage 1 and stage 2,
    along with model-based weighting, an eligibility trace, and choice perseveration.
    
    Parameters:
    lr_stage1: [0,1] - learning rate for stage 1 Q-values
    lr_stage2: [0,1] - learning rate for stage 2 Q-values
    beta_stage1: [0,10] - inverse temperature for stage 1 choices
    beta_stage2: [0,10] - inverse temperature for stage 2 choices
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    """
    lr_stage1, lr_stage2, beta_stage1, beta_stage2, w, lam, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] = pers
            
        exp_q1_net = np.exp(beta_stage1 * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta_stage2 * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Asymmetric Eligibility Trace Model.
    This model assumes that participants might back-propagate positive and negative 
    prediction errors from stage 2 to stage 1 differently.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam_pos: [0,1] - eligibility trace for positive prediction errors at stage 2
    lam_neg: [0,1] - eligibility trace for negative prediction errors at stage 2
    pers: [-5,5] - choice perseveration bonus for repeating the previous stage 1 action
    """
    learning_rate, beta, w, lam_pos, lam_neg, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] = pers
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # asymmetric eligibility trace updating for choice 1
        if delta_stage2 > 0:
            q_stage1_mf[action_1[trial]] += learning_rate * lam_pos * delta_stage2
        else:
            q_stage1_mf[action_1[trial]] += learning_rate * lam_neg * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Win-Stay Choice Perseveration Model.
    Separates general choice perseveration from outcome-dependent perseveration (win-stay).
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight of model-based vs model-free influence (1 = pure MB, 0 = pure MF)
    lam: [0,1] - eligibility trace parameter carrying stage 2 reward to stage 1
    pers: [-5,5] - general choice perseveration bonus for stage 1
    win_stay: [-5,5] - additional perseveration bonus for the previous action if it was rewarded
    """
    learning_rate, beta, w, lam, pers, win_stay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if trial > 0:
            pers_bonus[action_1[trial-1]] += pers
            if reward[trial-1] > 0:
                pers_bonus[action_1[trial-1]] += win_stay
            
        exp_q1_net = np.exp(beta * q_stage1_net + pers_bonus)
        probs_1_net = exp_q1_net / np.sum(exp_q1_net)
        p_choice_1[trial] = probs_1_net[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        exp_q2 = np.exp(beta * q_stage2_mf[state_idx])
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
        
        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        # eligibility trace updating for choice 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
