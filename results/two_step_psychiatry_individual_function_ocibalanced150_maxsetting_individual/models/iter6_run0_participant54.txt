def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with Subjective Transition Probability.
    Participants may not believe the transition matrix is exactly 0.7/0.3, so this model
    fits a subjective transition probability for the common transition.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers_1: perseveration bonus for stage 1 [-5, 5]
    pers_2: perseveration bonus for stage 2 [-5, 5]
    trans_prob: subjective probability of common transition [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2, trans_prob = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[trans_prob, 1 - trans_prob], [1 - trans_prob, trans_prob]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # Stage 2 Choice
        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers_2
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # Value Updates
        if a1 != -1 and a2 != -1 and state_idx != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2

            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with asymmetric learning rates and subjective transition probability.
    Allows different sensitivity to positive vs negative net prediction errors, and models subjective
    beliefs about the transition structure.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers: perseveration bonus for both stages [-5, 5]
    trans_prob: subjective probability of common transition [0, 1]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, pers, trans_prob = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[trans_prob, 1 - trans_prob], [1 - trans_prob, trans_prob]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # Stage 2 Choice
        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # Value Updates
        if a1 != -1 and a2 != -1 and state_idx != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            total_delta_1 = delta_stage1 + lam * delta_stage2
            alpha_1 = alpha_pos if total_delta_1 >= 0 else alpha_neg
            alpha_2 = alpha_pos if delta_stage2 >= 0 else alpha_neg

            q_stage1_mf[a1] += alpha_1 * total_delta_1
            q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2

            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Free/Model-Based RL with separate decay rates for stage 1 and stage 2.
    Unchosen options decay towards the neutral value of 0.5, allowing flexibility in how fast
    the participant forgets values at the first versus the second stage.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 [0, 10]
    beta_2: inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    pers: perseveration bonus for both stages [-5, 5]
    decay_1: forgetting rate for unchosen options in stage 1 towards 0.5 [0, 1]
    decay_2: forgetting rate for unchosen options in stage 2 towards 0.5 [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, pers, decay_1, decay_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        a1 = action_1[trial]
        a2 = action_2[trial]
        state_idx = state[trial]

        # Stage 1 Choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers
            
        q_1_total = q_net_1 + pers_bonus_1
        exp_q1 = np.exp(beta_1 * q_1_total - np.max(beta_1 * q_1_total))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        if a1 != -1:
            p_choice_1[trial] = probs_1[a1]
        else:
            p_choice_1[trial] = 1.0

        # Stage 2 Choice
        if a2 != -1 and state_idx != -1:
            pers_bonus_2 = np.zeros(2)
            if last_action_2[state_idx] != -1:
                pers_bonus_2[last_action_2[state_idx]] = pers
                
            q_2_total = q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(beta_2 * q_2_total - np.max(beta_2 * q_2_total))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[a2]
        else:
            p_choice_2[trial] = 1.0

        # Value Updates
        if a1 != -1 and a2 != -1 and state_idx != -1:
            delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]

            q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
            q_stage2_mf[state_idx, a2] += alpha * delta_stage2

            # Decay for unchosen options
            unchosen_a1 = 1 - a1
            q_stage1_mf[unchosen_a1] += decay_1 * (0.5 - q_stage1_mf[unchosen_a1])
            
            unchosen_a2 = 1 - a2
            q_stage2_mf[state_idx, unchosen_a2] += decay_2 * (0.5 - q_stage2_mf[state_idx, unchosen_a2])

            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay_2 * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay_2 * (0.5 - q_stage2_mf[unvisited_state, 1])

            last_action_2[state_idx] = a2
            
        if a1 != -1:
            last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
