Here are 3 new cognitive models that explore alternative mechanisms for the participant's behavior, based on the observed patterns and previous feedback. 

### Model 1: State-Dependent Stage 2 Stickiness
The participant may repeat their second-stage action differently depending on whether they transition to the same planet as the previous trial or a different one. This model splits Stage 2 stickiness based on state continuity.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with State-Dependent Stage 2 Stickiness.
    Stage 2 stickiness is split based on whether the current state is the same as the previous trial's state.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p_1: [-5, 5] - Choice stickiness for stage 1.
    p_2_same: [-5, 5] - Stage 2 stickiness when the state is the same as the previous trial.
    p_2_diff: [-5, 5] - Stage 2 stickiness when the state is different from the previous trial.
    lam: [0, 1] - Eligibility trace.
    """
    alpha, beta_1, beta_2, w, p_1, p_2_same, p_2_diff, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_state = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + p_1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            if s == last_state:
                stickiness_2[last_action_2] = p_2_same
            else:
                stickiness_2[last_action_2] = p_2_diff
            
        logits_2 = beta_2 * q_stage2_mf[s] + stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2
        last_state = s

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unified Temperature with Full Reward-Modulated Stickiness
Since the best model highlighted win-stay/lose-shift dynamics, this model extends it to *both* stages. To keep the parameter count within the limit (8 parameters max), it uses a single unified temperature (`beta`) for both decision stages, freeing up a parameter to model Stage 1 reward-modulation.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Single Temperature and Reward-Modulated Stickiness for Both Stages.
    Uses a single beta parameter to allow for win-stay/lose-shift parameters at both stages without exceeding 8 parameters.
    
    Parameters:
    alpha: [0, 1] - Learning rate.
    beta: [0, 10] - Shared inverse temperature for both stages.
    w: [0, 1] - Model-based weight.
    p_1_win: [-5, 5] - Stage 1 stickiness after a rewarded trial.
    p_1_lose: [-5, 5] - Stage 1 stickiness after an unrewarded trial.
    p_2_win: [-5, 5] - Stage 2 stickiness after a rewarded trial.
    p_2_lose: [-5, 5] - Stage 2 stickiness after an unrewarded trial.
    lam: [0, 1] - Eligibility trace.
    """
    alpha, beta, w, p_1_win, p_1_lose, p_2_win, p_2_lose, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_reward == 1:
                stickiness_1[last_action_1] = p_1_win
            elif last_reward == 0:
                stickiness_1[last_action_1] = p_1_lose
            
        logits_1 = beta * q_stage1_net + stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            if last_reward == 1:
                stickiness_2[last_action_2] = p_2_win
            elif last_reward == 0:
                stickiness_2[last_action_2] = p_2_lose
            
        logits_2 = beta * q_stage2_mf[s] + stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2 = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition Matrix Learning
Rather than assuming the participant perfectly knows and trusts the 0.7/0.3 fixed transition probabilities from the start, this model assumes the participant learns the transition matrix over time dynamically through experience.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model with Transition Matrix Learning.
    Instead of assuming a fixed transition matrix, the model updates its estimate of the transition probabilities based on experience.
    
    Parameters:
    alpha: [0, 1] - Learning rate for values.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Model-based weight.
    p_1: [-5, 5] - Choice stickiness for stage 1.
    p_2: [-5, 5] - Choice stickiness for stage 2.
    lam: [0, 1] - Eligibility trace.
    alpha_tr: [0, 1] - Learning rate for the transition matrix.
    """
    alpha, beta_1, beta_2, w, p_1, p_2, lam, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize un-biased transition matrix
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        a1 = int(action_1[trial])
        s = int(state[trial])
        a2 = int(action_2[trial])
        r = reward[trial]

        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + p_1 * stickiness_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Stage 2 policy
        stickiness_2 = np.zeros(2)
        if last_action_2 != -1:
            stickiness_2[last_action_2] = 1.0
            
        logits_2 = beta_2 * q_stage2_mf[s] + p_2 * stickiness_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Update Transition Matrix (Rescorla-Wagner rule for probabilities)
        transition_matrix[a1, s] += alpha_tr * (1.0 - transition_matrix[a1, s])
        transition_matrix[a1, 1 - s] += alpha_tr * (0.0 - transition_matrix[a1, 1 - s])
        
        last_action_1 = a1
        last_action_2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
