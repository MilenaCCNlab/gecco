Here are 3 new cognitive models that build upon your best model by incorporating additional well-known mechanisms from the literature on the two-step task (e.g., Daw et al., 2011). These models explore eligibility traces, state-specific stage-2 stickiness, and stage-specific learning rates to better capture the participant's decision-making process.

### Model 1: Hybrid Model with Eligibility Trace and Stickiness
This model adds an eligibility trace (`lam`) to the previous best model. The eligibility trace allows the reward prediction error from the second stage to directly update the model-free value of the first-stage action, capturing how rewards intuitively reinforce the entire sequence of actions.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF learning with eligibility trace (lambda) and stage 1 stickiness.
    Combines model-based and model-free learning, with an eligibility trace allowing 
    stage 2 rewards to directly update stage 1 model-free values, plus choice perseveration.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    lam: [0,1] - eligibility trace parameter
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    """
    learning_rate, beta, w, lam, stickiness_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        # eligibility trace update for stage 1
        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Hybrid Model with Stage 1 and Stage 2 Stickiness
Participants often exhibit perseverative behavior at both stages. This model introduces a second stickiness parameter (`stickiness_2`) that gives a bonus to repeating the same stage 2 action *in a specific state* (i.e., choosing the same alien when returning to the same planet).

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF learning with stage 1 and state-specific stage 2 stickiness.
    Adds perseveration bonuses for both stage 1 choices and state-specific stage 2 choices,
    capturing tendencies to repeat actions at both stages.
    
    Parameters:
    learning_rate: [0,1] - learning rate for updating Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    stickiness_2: [-5,5] - choice perseveration bonus for repeating the last stage 2 action in a given state
    """
    learning_rate, beta, w, stickiness_1, stickiness_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1
    last_action_2 = np.full(2, -1) # Tracks the last stage 2 action taken in each state

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx].copy()
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Hybrid Model with Stage-Specific Learning Rates
Participants might update their expectations at different speeds for the two stages. This model replaces the single learning rate with two independent learning rates (`learning_rate_1` and `learning_rate_2`), allowing for differential flexibility between the transition-dependent first stage and the direct reward-dependent second stage.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF learning with separate learning rates for each stage and stage 1 stickiness.
    Having separate learning rates allows the model to capture differences in how quickly participants 
    update their expectations for the transition-dependent first stage versus the reward-dependent second stage.
    
    Parameters:
    learning_rate_1: [0,1] - learning rate for stage 1 Q-values
    learning_rate_2: [0,1] - learning rate for stage 2 Q-values
    beta: [0,10] - inverse temperature for softmax choice
    w: [0,1] - weight on model-based vs model-free strategy (1 = MB, 0 = MF)
    stickiness_1: [-5,5] - choice perseveration bonus for repeating the last stage 1 action
    """
    learning_rate_1, learning_rate_2, beta, w, stickiness_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    last_action_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net.copy()
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
