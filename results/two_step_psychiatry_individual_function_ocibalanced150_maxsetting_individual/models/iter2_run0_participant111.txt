def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with learned transition probabilities.
    Instead of assuming fixed 0.7/0.3 transition probabilities, the agent learns the transition 
    matrix dynamically over time via a Rescorla-Wagner updating rule.
    
    Bounds:
    alpha1: [0, 1] - learning rate for stage 1
    alpha2: [0, 1] - learning rate for stage 2
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p: [-5, 5] - choice stickiness for stage 1
    alpha_trans: [0, 1] - learning rate for the transition matrix
    """
    alpha1, alpha2, beta1, beta2, lam, w, p, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    # Initialize transition matrix at 0.5 for all transitions
    transition_matrix = np.array([[0.5, 0.5], [0.5, 0.5]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        q1 = beta1 * q_net + p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # action value updating
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        # update transition matrix
        transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with transition-dependent choice stickiness.
    Stickiness at stage 1 depends on whether the previous trial's transition was common or rare.
    Common transitions in this task are Spaceship A -> Planet X (0->0) and Spaceship U -> Planet Y (1->1).
    
    Bounds:
    alpha1: [0, 1] - learning rate for stage 1
    alpha2: [0, 1] - learning rate for stage 2
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam: [0, 1] - eligibility trace parameter
    w: [0, 1] - model-based weight
    p_common: [-5, 5] - choice stickiness if previous transition was common
    p_rare: [-5, 5] - choice stickiness if previous transition was rare
    """
    alpha1, alpha2, beta1, beta2, lam, w, p_common, p_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_transition_common = True

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        current_p = p_common if last_transition_common else p_rare
            
        q1 = beta1 * q_net + current_p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        # action value updating
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]
        last_transition_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric eligibility traces for positive and negative rewards.
    The extent to which stage 2 rewards directly update stage 1 model-free values depends on 
    whether the reward obtained was positive (1 coin) or negative (0 coins).
    
    Bounds:
    alpha1: [0, 1] - learning rate for stage 1
    alpha2: [0, 1] - learning rate for stage 2
    beta1: [0, 10] - inverse temperature for stage 1
    beta2: [0, 10] - inverse temperature for stage 2
    lam_pos: [0, 1] - eligibility trace for positive rewards
    lam_neg: [0, 1] - eligibility trace for negative rewards (0 coins)
    w: [0, 1] - model-based weight
    p: [-5, 5] - choice stickiness for stage 1
    """
    alpha1, alpha2, beta1, beta2, lam_pos, lam_neg, w, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        q1 = beta1 * q_net + p * rep
        q1_max = np.max(q1)
        exp_q1 = np.exp(q1 - q1_max)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        q2 = beta2 * q_stage2_mf[state_idx]
        q2_max = np.max(q2)
        exp_q2 = np.exp(q2 - q2_max)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        lam = lam_pos if reward[trial] > 0 else lam_neg

        # action value updating
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
