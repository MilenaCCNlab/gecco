import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Separate learning rates for stages, Win-stay/Lose-shift for Stage 1, stickiness for Stage 2.
    
    Bounds:
    alpha1: [0, 1] - Learning rate for Stage 1
    alpha2: [0, 1] - Learning rate for Stage 2
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    w: [0, 1] - Model-based weight
    p_win: [0, 10] - Stage 1 stickiness after a rewarded trial
    p_lose: [0, 10] - Stage 1 stickiness after an unrewarded trial
    p2_stay: [0, 10] - Stage 2 stickiness
    """
    alpha1, alpha2, beta1, beta2, w, p_win, p_lose, p2_stay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action1 = -1
    last_reward = 0
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                stickiness1[last_action1] = p_win
            else:
                stickiness1[last_action1] = p_lose
                
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            stickiness2[last_action2[state_idx]] = p2_stay
            
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + stickiness2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        q_stage1_mf[a1] += alpha1 * delta_stage2
        
        last_action1 = a1
        last_action2[state_idx] = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Asymmetric learning rates for positive and negative PE, separate stickiness.
    
    Bounds:
    alpha_pos: [0, 1] - Learning rate for positive PE
    alpha_neg: [0, 1] - Learning rate for negative PE
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    w: [0, 1] - Model-based weight
    p_stay1: [0, 10] - Stage 1 stickiness
    p_stay2: [0, 10] - Stage 2 stickiness
    """
    alpha_pos, alpha_neg, beta1, beta2, w, p_stay1, p_stay2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action1 = -1
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness1 = np.zeros(2)
        if last_action1 != -1:
            stickiness1[last_action1] = p_stay1
                
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            stickiness2[last_action2[state_idx]] = p_stay2
            
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + stickiness2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        q_stage1_mf[a1] += alpha2 * delta_stage2
        
        last_action1 = a1
        last_action2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model 3: Win-stay/Lose-shift for Stage 1, Stage 2 stickiness, and Q-value decay.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta1: [0, 10] - Stage 1 inverse temperature
    beta2: [0, 10] - Stage 2 inverse temperature
    w: [0, 1] - Model-based weight
    p_win: [0, 10] - Stage 1 stickiness after a rewarded trial
    p_lose: [0, 10] - Stage 1 stickiness after an unrewarded trial
    p2_stay: [0, 10] - Stage 2 stickiness
    decay: [0, 1] - Decay rate for Q-values towards 0.5
    """
    alpha, beta1, beta2, w, p_win, p_lose, p2_stay, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action1 = -1
    last_reward = 0
    last_action2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness1 = np.zeros(2)
        if last_action1 != -1:
            if last_reward == 1:
                stickiness1[last_action1] = p_win
            else:
                stickiness1[last_action1] = p_lose
                
        exp_q1 = np.exp(beta1 * q_stage1_net + stickiness1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        stickiness2 = np.zeros(2)
        if last_action2[state_idx] != -1:
            stickiness2[last_action2[state_idx]] = p2_stay
            
        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + stickiness2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
  
        a1 = action_1[trial]
        a2 = action_2[trial]
        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        # Apply decay to all Q-values before update
        q_stage1_mf = q_stage1_mf + decay * (0.5 - q_stage1_mf)
        q_stage2_mf = q_stage2_mf + decay * (0.5 - q_stage2_mf)

        # Update Q-values
        q_stage1_mf[a1] += alpha * delta_stage1
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        q_stage1_mf[a1] += alpha * delta_stage2
        
        last_action1 = a1
        last_action2[state_idx] = a2
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
