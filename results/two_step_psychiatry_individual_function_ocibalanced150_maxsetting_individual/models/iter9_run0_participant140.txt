def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Win-Stay/Lose-Shift and Preference for Last State
    
    Includes win-stay/lose-shift (outcome-dependent perseveration) at Stage 1.
    Additionally, it includes a preference bonus for the Stage 1 action that commonly 
    leads to the state visited on the previous trial, capturing a tendency to 
    return to the same environment.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w: [0, 1] - Model-based vs model-free weight
    lam: [0, 1] - Eligibility trace
    pers_win: [-5, 5] - Stage 1 perseveration bonus if previous trial rewarded
    pers_lose: [-5, 5] - Stage 1 perseveration bonus if previous trial unrewarded
    pref_last_state: [-5, 5] - Bonus to the Stage 1 action that commonly transitions to the previous state
    """
    alpha, beta_1, beta_2, w, lam, pers_win, pers_lose, pref_last_state = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_state = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_reward = -1
            last_state = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus_1[last_action_1] += pers_win
            else:
                pers_bonus_1[last_action_1] += pers_lose
                
        if last_state != -1:
            # Action 0 commonly leads to State 0, Action 1 commonly leads to State 1
            pers_bonus_1[last_state] += pref_last_state
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        last_state = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Outcome-Dependent Model-Based Weight and Perseveration
    
    The degree to which the participant relies on the model-based system (w) 
    depends on whether the previous trial was rewarded (w_win) or unrewarded (w_lose).
    Also includes outcome-dependent perseveration at Stage 1.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1
    beta_2: [0, 10] - Inverse temperature for stage 2
    w_win: [0, 1] - Model-based weight if previous trial rewarded
    w_lose: [0, 1] - Model-based weight if previous trial unrewarded
    lam: [0, 1] - Eligibility trace
    pers_win: [-5, 5] - Stage 1 perseveration bonus if previous trial rewarded
    pers_lose: [-5, 5] - Stage 1 perseveration bonus if previous trial unrewarded
    """
    alpha, beta_1, beta_2, w_win, w_lose, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_reward = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        if last_reward == 1:
            w_current = w_win
        elif last_reward == 0:
            w_current = w_lose
        else:
            w_current = (w_win + w_lose) / 2.0
            
        q1_net = w_current * q_stage1_mb + (1 - w_current) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus_1[last_action_1] = pers_win
            else:
                pers_bonus_1[last_action_1] = pers_lose
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Stage-Specific Outcome-Dependent Perseveration
    
    Features win-stay/lose-shift (outcome-dependent perseveration) for BOTH Stage 1 and Stage 2.
    Uses a single inverse temperature (beta) for parsimony to allow for 4 perseveration parameters.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta: [0, 10] - Inverse temperature for both stages
    w: [0, 1] - Model-based vs model-free weight
    lam: [0, 1] - Eligibility trace
    pers_win_1: [-5, 5] - Stage 1 perseveration bonus if previous trial rewarded
    pers_lose_1: [-5, 5] - Stage 1 perseveration bonus if previous trial unrewarded
    pers_win_2: [-5, 5] - Stage 2 perseveration bonus if previous trial rewarded
    pers_lose_2: [-5, 5] - Stage 2 perseveration bonus if previous trial unrewarded
    """
    alpha, beta, w, lam, pers_win_1, pers_lose_1, pers_win_2, pers_lose_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    last_action_2 = np.full(2, -1)
    last_reward_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_reward = -1
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1 and last_reward != -1:
            if last_reward == 1:
                pers_bonus_1[last_action_1] = pers_win_1
            else:
                pers_bonus_1[last_action_1] = pers_lose_1
            
        logits_1 = beta * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]

        # policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1 and last_reward_2[state_idx] != -1:
            if last_reward_2[state_idx] == 1:
                pers_bonus_2[last_action_2[state_idx]] = pers_win_2
            else:
                pers_bonus_2[last_action_2[state_idx]] = pers_lose_2
                
        logits_2 = beta * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        
        last_action_2[state_idx] = action_2[trial]
        last_reward_2[state_idx] = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
