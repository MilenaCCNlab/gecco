Here are 3 new cognitive models that expand on the best model so far by introducing new mechanisms, while remaining within the parameter limit.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model tests whether participants update their expectations at different speeds for the first and second stages. It replaces the general decay parameter with a distinct learning rate for Stage 2, while preserving the choice perseveration at both stages.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate Learning Rates for Stage 1 and Stage 2 with Stage 1 & 2 Perseveration
    
    This model uses different learning rates for Stage 1 and Stage 2, alongside
    choice perseveration for both stages, allowing for distinct updating speeds.
    
    Bounds:
    alpha_1: [0, 1] - Learning rate for stage 1
    alpha_2: [0, 1] - Learning rate for stage 2
    beta_1: [0, 10] - Inverse temperature for stage 1 choices
    beta_2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weight on model-based vs model-free values
    lam: [0, 1] - Eligibility trace for stage 1 MF update from stage 2 reward
    pers_1: [-5, 5] - Choice perseveration bonus for stage 1
    pers_2: [-5, 5] - Choice perseveration bonus for stage 2
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Unchosen Decay (Forgetting)
Instead of globally decaying all Q-values, this model tests the hypothesis that participants specifically "forget" the values of unchosen options, which regress toward a neutral expectation (0.5).

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Single Learning Rate with Stage 1 & 2 Perseveration and Unchosen Decay
    
    This model decays the Q-values of unchosen actions towards 0.5 (neutral expectation) 
    to account for forgetting or regression to the mean of unexperienced options.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1 choices
    beta_2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weight on model-based vs model-free values
    lam: [0, 1] - Eligibility trace for stage 1 MF update
    pers_1: [-5, 5] - Choice perseveration bonus for stage 1
    pers_2: [-5, 5] - Choice perseveration bonus for stage 2
    unchosen_decay: [0, 1] - Decay rate for unchosen actions towards 0.5
    """
    alpha, beta_1, beta_2, w, lam, pers_1, pers_2, unchosen_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = pers_1
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        pers_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            pers_bonus_2[last_action_2[state_idx]] = pers_2
            
        logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Unchosen decay towards 0.5
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] += unchosen_decay * (0.5 - q_stage1_mf[unchosen_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += unchosen_decay * (0.5 - q_stage2_mf[s, a])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Perseveration
This model tests whether the participant's tendency to repeat a Stage 1 choice depends on whether they experienced a common or uncommon transition on the previous trial, which can capture an interaction between the transition structure and perseverative habits. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Transition-Dependent Perseveration with Decay
    
    This model differentiates perseveration at Stage 1 based on whether the 
    previous trial's transition was common or uncommon. It also includes Q-value decay.
    
    Bounds:
    alpha: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for stage 1 choices
    beta_2: [0, 10] - Inverse temperature for stage 2 choices
    w: [0, 1] - Weight on model-based vs model-free values
    lam: [0, 1] - Eligibility trace for stage 1 MF update
    pers_c: [-5, 5] - Perseveration bonus after a common transition
    pers_u: [-5, 5] - Perseveration bonus after an uncommon transition
    decay: [0, 1] - Decay rate for all Q-values
    """
    alpha, beta_1, beta_2, w, lam, pers_c, pers_u, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_transition_common = False

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            if last_transition_common:
                pers_bonus_1[last_action_1] = pers_c
            else:
                pers_bonus_1[last_action_1] = pers_u
            
        logits_1 = beta_1 * q1_net + pers_bonus_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        p_choice_1[trial] = (exp_q1 / np.sum(exp_q1))[action_1[trial]]
        
        # Policy for the second choice
        state_idx = state[trial]
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        p_choice_2[trial] = (exp_q2 / np.sum(exp_q2))[action_2[trial]]

        # Decay
        q_stage1_mf = (1 - decay) * q_stage1_mf
        q_stage2_mf = (1 - decay) * q_stage2_mf

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        # In this task, action 0 -> state 0 and action 1 -> state 1 are common transitions
        last_transition_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
