Here are 3 new cognitive models that explore different mechanisms for the two-step task. Each model introduces a unique hypothesis about how participants might be making decisions, strictly adhering to the 8-parameter limit and avoiding the parameter combinations you've already tried.

### Model 1: Subjective Value of Unrewarded Outcomes
This model hypothesis tests whether participants treat a "0 coin" outcome as a neutral event (0) or as an active penalty/loss (negative value). It introduces a parameter `rho` which sets the subjective value of an unrewarded trial, capturing risk sensitivity or loss aversion.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Subjective Value of Unrewarded Outcomes.
    This model introduces a parameter 'rho' which represents the subjective value 
    of receiving 0 coins. This allows the model to capture different degrees of 
    aversion or neutral response to reward omissions.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for Stage 1
    beta2: [0,10] - Inverse temperature for Stage 2
    lam: [0,1] - Eligibility trace
    w: [0,1] - Model-based weight for Q-values
    p1: [-5,5] - Stage 1 choice stickiness
    p2: [-5,5] - Stage 2 choice stickiness
    rho: [-5,5] - Subjective value of 0 coins (unrewarded outcome)
    """
    alpha, beta1, beta2, lam, w, p1, p2, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1
    prev_s = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        if prev_a1 != -1:
            rep1[prev_a1] = 1

        logits_1 = beta1 * q_net + p1 * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if prev_s == state_idx and prev_a2 != -1:
            rep2[prev_a2] = 1

        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        r_eff = 1.0 if reward[trial] == 1 else rho

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = r_eff - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]
        prev_s = state_idx

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Exponential Choice Trace (N-back Perseveration)
Instead of only considering the immediately preceding trial for choice stickiness, this model maintains an exponentially decaying trace of past choices. This tests if participants have a longer-term choice perseveration habit irrespective of reward.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Exponential Choice Trace (N-back Perseveration).
    Instead of only considering the immediately preceding trial for choice stickiness,
    this model maintains an exponentially decaying trace of past choices.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for Stage 1
    beta2: [0,10] - Inverse temperature for Stage 2
    lam: [0,1] - Eligibility trace
    w: [0,1] - Model-based weight for Q-values
    p1: [-5,5] - Stage 1 choice stickiness weight
    p2: [-5,5] - Stage 2 choice stickiness weight
    tau_choice: [0,1] - Decay rate of the choice trace (0 = 1-back only, 1 = no decay)
    """
    alpha, beta1, beta2, lam, w, p1, p2, tau_choice = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    trace_a1 = np.zeros(2)
    trace_a2 = np.zeros((2, 2))

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_net + p1 * trace_a1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * trace_a2[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Update choice traces
        trace_a1 *= tau_choice
        trace_a1[action_1[trial]] += 1
        
        trace_a2 *= tau_choice
        trace_a2[state_idx, action_2[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Direct MB/MF Stay Biases (Factorial Logic)
This model embeds the classic behavioral stay/shift factorial logic directly into the policy. Rather than relying solely on Q-values to produce model-based behavior, it applies direct logit bonuses to repeat the Stage 1 choice based on the interaction of the previous reward and transition type.

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Direct MB/MF Stay Biases.
    In addition to Q-learning, this model includes direct choice perseveration biases 
    based on the previous trial's outcome and transition (Daw et al. 2011 stay/shift logic).
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta1: [0,10] - Inverse temperature for Stage 1
    beta2: [0,10] - Inverse temperature for Stage 2
    lam: [0,1] - Eligibility trace
    w: [0,1] - Model-based weight for Q-values
    p_stay_mb: [-5,5] - Model-based stay bias (applied if Rew+Com or Unrew+Rare)
    p_stay_mf: [-5,5] - Model-free stay bias (applied if Rew)
    p2: [-5,5] - Stage 2 choice stickiness
    """
    alpha, beta1, beta2, lam, w, p_stay_mb, p_stay_mf, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    prev_a1 = -1
    prev_a2 = -1
    prev_s = -1
    prev_r = -1
    prev_trans_common = False

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep1 = np.zeros(2)
        mb_stay = 0
        mf_stay = 0
        if prev_a1 != -1:
            rep1[prev_a1] = 1
            if prev_r == 1:
                mf_stay = 1
                if prev_trans_common:
                    mb_stay = 1
            else:
                if not prev_trans_common:
                    mb_stay = 1

        logits_1 = beta1 * q_net + (p_stay_mb * mb_stay + p_stay_mf * mf_stay) * rep1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        rep2 = np.zeros(2)
        if prev_s == state_idx and prev_a2 != -1:
            rep2[prev_a2] = 1

        logits_2 = beta2 * q_stage2_mf[state_idx] + p2 * rep2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        prev_a1 = action_1[trial]
        prev_a2 = action_2[trial]
        prev_s = state_idx
        prev_r = reward[trial]
        prev_trans_common = (action_1[trial] == state_idx)

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
