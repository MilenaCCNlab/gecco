def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Q-learning with asymmetric eligibility traces for positive and negative reward prediction errors.
    This model captures the idea that participants might update their Stage 1 choices differently 
    based on whether the Stage 2 outcome was better or worse than expected.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for updating Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    p_stage1: [-5, 5] - Perseveration bonus for repeating the last Stage 1 choice
    p_stage2: [-5, 5] - Perseveration bonus for repeating the last Stage 2 choice
    lam_pos: [0, 1] - Eligibility trace for positive reward prediction errors
    lam_neg: [0, 1] - Eligibility trace for negative reward prediction errors
    """
    learning_rate, beta, w, p_stage1, p_stage2, lam_pos, lam_neg = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_a1 = -1
    last_a2 = -1
    last_s = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # Handle missing or invalid data
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        val_1 = beta * q_net
        if last_a1 != -1:
            val_1[last_a1] += p_stage1
        
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        val_2 = beta * q_stage2_mf[s]
        if last_s == s and last_a2 != -1:
            val_2[last_a2] += p_stage2
            
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        # Action value updating for choice 2 and eligibility trace
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        
        # Apply asymmetric eligibility trace depending on RPE sign
        lam = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        last_a1 = a1
        last_s = s
        last_a2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Q-learning with outcome-dependent perseveration at both stages.
    This model separates the tendency to repeat choices based on whether the previous trial 
    was rewarded or unrewarded, applying this logic separately to both Stage 1 and Stage 2 choices.
    
    Parameters:
    learning_rate: [0, 1] - Learning rate for Q-values
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    p_stage1_rew: [-5, 5] - Stage 1 perseveration bonus after a rewarded trial
    p_stage1_unrew: [-5, 5] - Stage 1 perseveration bonus after an unrewarded trial
    p_stage2_rew: [-5, 5] - Stage 2 perseveration bonus after a rewarded trial
    p_stage2_unrew: [-5, 5] - Stage 2 perseveration bonus after an unrewarded trial
    lam: [0, 1] - Eligibility trace connecting Stage 2 prediction error to Stage 1 Q-value
    """
    learning_rate, beta, w, p_stage1_rew, p_stage1_unrew, p_stage2_rew, p_stage2_unrew, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_a1 = -1
    last_a2 = -1
    last_s = -1
    last_r = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        val_1 = beta * q_net
        if last_a1 != -1 and last_r != -1:
            if last_r == 1:
                val_1[last_a1] += p_stage1_rew
            else:
                val_1[last_a1] += p_stage1_unrew
        
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        val_2 = beta * q_stage2_mf[s]
        if last_s == s and last_a2 != -1 and last_r != -1:
            if last_r == 1:
                val_2[last_a2] += p_stage2_rew
            else:
                val_2[last_a2] += p_stage2_unrew
                
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += learning_rate * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        q_stage2_mf[s, a2] += learning_rate * delta_stage2
        q_stage1_mf[a1] += learning_rate * lam * delta_stage2

        last_a1 = a1
        last_s = s
        last_a2 = a2
        last_r = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Q-learning with choice traces and separate learning rates for positive and negative RPEs.
    This model combines continuous choice traces (which capture long-term stickiness) with asymmetric 
    learning to reflect different sensitivity to better-than-expected versus worse-than-expected outcomes.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive reward prediction errors
    alpha_neg: [0, 1] - Learning rate for negative reward prediction errors
    beta: [0, 10] - Inverse temperature for softmax action selection
    w: [0, 1] - Weight placed on model-based vs model-free Q-values
    phi_stage1: [-5, 5] - Weight of stage 1 choice trace
    phi_stage2: [-5, 5] - Weight of stage 2 choice trace
    alpha_trace: [0, 1] - Learning rate/decay for the choice trace
    lam: [0, 1] - Eligibility trace connecting Stage 2 prediction error to Stage 1 Q-value
    """
    alpha_pos, alpha_neg, beta, w, phi_stage1, phi_stage2, alpha_trace, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    trace_1 = np.zeros(2)
    trace_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        val_1 = beta * q_net + phi_stage1 * trace_1
        val_1 -= np.max(val_1)
        exp_q1 = np.exp(val_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        val_2 = beta * q_stage2_mf[s] + phi_stage2 * trace_2[s]
        val_2 -= np.max(val_2)
        exp_q2 = np.exp(val_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        lr1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += lr1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = r - q_stage2_mf[s, a2]
        lr2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[s, a2] += lr2 * delta_stage2
        q_stage1_mf[a1] += lr2 * lam * delta_stage2

        # Update choice traces
        trace_1[a1] += alpha_trace * (1.0 - trace_1[a1])
        trace_1[1 - a1] += alpha_trace * (0.0 - trace_1[1 - a1])
        
        trace_2[s, a2] += alpha_trace * (1.0 - trace_2[s, a2])
        trace_2[s, 1 - a2] += alpha_trace * (0.0 - trace_2[s, 1 - a2])
        other_state = 1 - s
        trace_2[other_state, 0] += alpha_trace * (0.0 - trace_2[other_state, 0])
        trace_2[other_state, 1] += alpha_trace * (0.0 - trace_2[other_state, 1])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
