def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with Q-value decay and reward sensitivity (rho).
    This model assumes that the values of unchosen actions decay towards zero over time, 
    and that the subjective value of the reward is scaled by a parameter rho.
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers: [-5,5]
    decay: [0,1]
    rho: [0,10]
    """
    alpha, beta_1, beta_2, w, lam, pers, decay, rho = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + pers * pers_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating
        scaled_r = r * rho
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = scaled_r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Decay for unchosen options
        for act1 in range(2):
            if act1 != a1:
                q_stage1_mf[act1] *= (1 - decay)
                
        for st2 in range(2):
            for act2 in range(2):
                if not (st2 == s and act2 == a2):
                    q_stage2_mf[st2, act2] *= (1 - decay)
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with Q-value decay and counterfactual learning for the unchosen stage-2 action.
    This model updates the chosen stage-2 action with the observed reward, and the unchosen 
    stage-2 action in the same state with the counterfactual reward (1 - reward).
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    w: [0,1]
    lam: [0,1]
    pers: [-5,5]
    decay: [0,1]
    alpha_cf: [0,1]
    """
    alpha, beta_1, beta_2, w, lam, pers, decay, alpha_cf = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            continue

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + pers * pers_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Counterfactual update for unchosen option in the same state
        unchosen_a2 = 1 - a2
        cf_r = 1 - r # Assuming anti-correlated rewards
        delta_cf = cf_r - q_stage2_mf[s, unchosen_a2]
        q_stage2_mf[s, unchosen_a2] += alpha_cf * delta_cf

        # Decay for unchosen options
        for act1 in range(2):
            if act1 != a1:
                q_stage1_mf[act1] *= (1 - decay)
                
        for st2 in range(2):
            for act2 in range(2):
                if not (st2 == s and (act2 == a2 or act2 == unchosen_a2)):
                    q_stage2_mf[st2, act2] *= (1 - decay)
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with Q-value decay and reward-dependent model-based weight (w_win, w_lose).
    This model allows the balance between model-based and model-free control to shift 
    depending on whether the previous trial was rewarded (w_win) or unrewarded (w_lose).
    
    Bounds:
    alpha: [0,1]
    beta_1: [0,10]
    beta_2: [0,10]
    lam: [0,1]
    pers: [-5,5]
    decay: [0,1]
    w_win: [0,1]
    w_lose: [0,1]
    """
    alpha, beta_1, beta_2, lam, pers, decay, w_win, w_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]
        
        if a1 == -1 or s == -1 or a2 == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            last_action_1 = -1
            last_reward = -1
            continue
            
        # Determine w based on last reward
        if last_reward == 1:
            current_w = w_win
        elif last_reward == 0:
            current_w = w_lose
        else:
            current_w = (w_win + w_lose) / 2.0

        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = current_w * q_stage1_mb + (1 - current_w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus[last_action_1] = 1.0
            
        logits_1 = beta_1 * q_stage1_net + pers * pers_bonus
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]

        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Decay for unchosen options
        for act1 in range(2):
            if act1 != a1:
                q_stage1_mf[act1] *= (1 - decay)
                
        for st2 in range(2):
            for act2 in range(2):
                if not (st2 == s and act2 == a2):
                    q_stage2_mf[st2, act2] *= (1 - decay)
        
        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
