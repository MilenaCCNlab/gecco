def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with continuous habit learning (alpha_h) for both stages, 
    replacing simple previous-trial perseveration with an exponentially weighted habit trace.
    Habit traces decay on every trial for all states and actions.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based vs model-free weight
    lam: [0,1] - Eligibility trace decay
    alpha_h: [0,1] - Habit learning rate
    p1: [-5,5] - Habit weight for stage 1
    p2: [-5,5] - Habit weight for stage 2
    """
    alpha, beta, w, lam, alpha_h, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    habit_1 = np.zeros(2)
    habit_2 = np.zeros((2, 2))

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net + p1 * habit_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        logits_2 = beta * q_stage2_mf[state_idx] + p2 * habit_2[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        # Habit updates
        habit_1 *= (1 - alpha_h)
        habit_1[action_1[trial]] += alpha_h
        
        habit_2 *= (1 - alpha_h)
        habit_2[state_idx, action_2[trial]] += alpha_h

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a reward baseline parameter 'b' to represent a subjective reference point
    for calculating reward prediction errors, along with stage 1 and stage 2 perseveration.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta: [0,10] - Inverse temperature
    w: [0,1] - Model-based vs model-free weight
    lam: [0,1] - Eligibility trace decay
    p1: [-5,5] - Choice perseveration bonus for stage 1
    p2: [-5,5] - State-specific choice perseveration bonus for stage 2
    b: [-1,1] - Reward baseline/reference point
    """
    alpha, beta, w, lam, p1, p2, b = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.array([-1, -1])

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = p1
            
        logits_1 = beta * q_stage1_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2_in_state[state_idx] != -1:
            pers_bonus_2[last_action_2_in_state[state_idx]] = p2
            
        logits_2 = beta * q_stage2_mf[state_idx] + pers_bonus_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        # Calculate prediction error using effective reward (reward - baseline)
        effective_reward = reward[trial] - b
        delta_stage2 = effective_reward - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with independent inverse temperatures for model-based and model-free 
    values at stage 1, an independent temperature for stage 2, an eligibility trace, 
    and perseveration bonuses for both stages.
    
    Bounds:
    alpha: [0,1] - Learning rate
    beta_mb: [0,10] - Inverse temperature for model-based Q-values at stage 1
    beta_mf1: [0,10] - Inverse temperature for model-free Q-values at stage 1
    beta_mf2: [0,10] - Inverse temperature for model-free Q-values at stage 2
    lam: [0,1] - Eligibility trace decay
    p1: [-5,5] - Choice perseveration bonus for stage 1
    p2: [-5,5] - State-specific choice perseveration bonus for stage 2
    """
    alpha, beta_mb, beta_mf1, beta_mf2, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2_in_state = np.array([-1, -1])

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            pers_bonus_1[last_action_1] = p1
            
        # Independent weighting of MB and MF values
        logits_1 = beta_mb * q_stage1_mb + beta_mf1 * q_stage1_mf + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Policy for the second choice
        pers_bonus_2 = np.zeros(2)
        if last_action_2_in_state[state_idx] != -1:
            pers_bonus_2[last_action_2_in_state[state_idx]] = p2
            
        logits_2 = beta_mf2 * q_stage2_mf[state_idx] + pers_bonus_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2_in_state[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
