def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates, separate inverse temperatures for MB and MF at stage 1,
    and a separate inverse temperature for stage 2, plus stage-specific stickiness.
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta_mb: [0, 10] - Inverse temperature for model-based values at stage 1.
    beta_mf: [0, 10] - Inverse temperature for model-free values at stage 1.
    beta2: [0, 10] - Inverse temperature for stage 2 actions.
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness for stage 1.
    pi2: [-5, 5] - Choice stickiness for stage 2.
    """
    alpha_pos, alpha_neg, beta_mb, beta_mf, beta2, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta2 * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with asymmetric learning rates, stage-specific stickiness, and a model-based weight (w) 
    that depends on whether the previous trial was rewarded (w_win) or unrewarded (w_lose).
    
    Parameters:
    alpha_pos: [0, 1] - Learning rate for positive prediction errors.
    alpha_neg: [0, 1] - Learning rate for negative prediction errors.
    beta: [0, 10] - Inverse temperature for softmax action selection.
    w_win: [0, 1] - Mixing weight for MB vs MF values after a rewarded trial.
    w_lose: [0, 1] - Mixing weight for MB vs MF values after an unrewarded trial.
    lam: [0, 1] - Eligibility trace parameter.
    pi1: [-5, 5] - Choice stickiness for stage 1.
    pi2: [-5, 5] - Choice stickiness for stage 2.
    """
    alpha_pos, alpha_neg, beta, w_win, w_lose, lam, pi1, pi2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = -1
    last_reward = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Determine w based on last reward
        if last_reward == 1:
            w = w_win
        elif last_reward == 0:
            w = w_lose
        else:
            w = 0.5 # default for first trial
            
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stick_vector_1 = np.zeros(2)
        if last_action_1 != -1:
            stick_vector_1[last_action_1] = 1.0
            
        exp_q1 = np.exp(beta * q_stage1_net + pi1 * stick_vector_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]
        
        last_action_1 = action_1[trial]

        # policy for the second choice
        stick_vector_2 = np.zeros(2)
        if last_action_2 != -1:
            stick_vector_2[last_action_2] = 1.0

        exp_q2 = np.exp(beta * q_stage2_mf[state_idx] + pi2 * stick_vector_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        last_action_2 = action_2[trial]

        # action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += lr_stage1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_stage2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr_stage1 * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
