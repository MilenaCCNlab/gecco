def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model with asymmetric learning rates (positive/negative RPE), decay, separate stage betas, and perseveration.
    
    Bounds:
    alpha_pos: [0, 1] - learning rate for positive RPEs
    alpha_neg: [0, 1] - learning rate for negative RPEs
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0)
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambd: [0, 1] - eligibility trace
    pers: [-2, 2] - choice perseveration (stickiness) for stage 1
    """
    alpha_pos, alpha_neg, decay, beta_1, beta_2, w, lambd, pers = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    last_a1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        pers_bonus = np.zeros(2)
        if last_a1 != -1:
            pers_bonus[last_a1] = pers
            
        logits_1 = beta_1 * q_net + pers_bonus
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
            q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
            q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha_2 * lambd * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] *= (1 - decay)
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] *= (1 - decay)
            q_stage2_mf[unvisited_state, 1] *= (1 - decay)
            
        last_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model with decay for unchosen Q-values, separate stage betas, and an exponentially decaying choice trace for perseveration.
    
    Bounds:
    alpha: [0, 1] - learning rate
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0)
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambd: [0, 1] - eligibility trace
    pers: [-2, 2] - choice trace weight for stage 1
    pers_decay: [0, 1] - decay rate for the choice trace
    """
    alpha, decay, beta_1, beta_2, w, lambd, pers, pers_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
            
        logits_1 = beta_1 * q_net + pers * choice_trace
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] *= (1 - decay)
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] *= (1 - decay)
            q_stage2_mf[unvisited_state, 1] *= (1 - decay)
            
        choice_trace *= (1 - pers_decay)
        choice_trace[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model with Q-value decay towards 0.5 (chance level), separate stage betas, and perseveration at both stages.
    
    Bounds:
    alpha: [0, 1] - learning rate
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0.5)
    beta_1: [0, 10] - inverse temperature for stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    w: [0, 1] - model-based weight
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice perseveration (stickiness) for stage 1
    pers_2: [-2, 2] - choice perseveration (stickiness) for stage 2
    """
    alpha, decay, beta_1, beta_2, w, lambd, pers_1, pers_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values at the expected value of chance level
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers_1
            
        logits_1 = beta_1 * q_net + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        if action_2[trial] != -1:
            pers_bonus_2 = np.zeros(2)
            if last_a2 != -1:
                pers_bonus_2[last_a2] = pers_2
                
            logits_2 = beta_2 * q_stage2_mf[state_idx] + pers_bonus_2
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
        last_a1 = action_1[trial]
        last_a2 = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
