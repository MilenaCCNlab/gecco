def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid MB/MF with Stage 1 Choice Kernel.
    Instead of a simple perseveration bonus for the immediately preceding trial,
    this model maintains a choice kernel for Stage 1 actions that updates over time,
    capturing longer-term choice autocorrelation.
    
    Parameters:
    alpha: [0, 1] - learning rate for Q-values
    beta_1: [0, 10] - stage 1 inverse temperature for Q-values
    beta_2: [0, 10] - stage 2 inverse temperature for Q-values
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    alpha_ck: [0, 1] - learning rate for the choice kernel
    beta_ck: [-10, 10] - inverse temperature (weight) for the choice kernel
    """
    alpha, beta_1, beta_2, w, lam, alpha_ck, beta_ck = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    ck_stage1 = np.zeros(2)

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + beta_ck * ck_stage1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        # Q-value updating
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Choice kernel updating
        ck_stage1 = (1 - alpha_ck) * ck_stage1
        ck_stage1[a1] += alpha_ck

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid MB/MF with Choice Kernels for both Stage 1 and Stage 2.
    Captures temporal choice autocorrelation (stickiness) at both decision stages
    using a shared choice kernel learning rate but separate weights.
    
    Parameters:
    alpha: [0, 1] - learning rate for Q-values
    beta_1: [0, 10] - stage 1 inverse temperature for Q-values
    beta_2: [0, 10] - stage 2 inverse temperature for Q-values
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    alpha_ck: [0, 1] - learning rate for choice kernels
    beta_ck1: [-10, 10] - weight for Stage 1 choice kernel
    beta_ck2: [-10, 10] - weight for Stage 2 choice kernel
    """
    alpha, beta_1, beta_2, w, lam, alpha_ck, beta_ck1, beta_ck2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    ck_stage1 = np.zeros(2)
    ck_stage2 = np.zeros((2, 2))

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + beta_ck1 * ck_stage1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s] + beta_ck2 * ck_stage2[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        # Q-value updating
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2
        
        # Choice kernel updating
        ck_stage1 = (1 - alpha_ck) * ck_stage1
        ck_stage1[a1] += alpha_ck
        
        # Update stage 2 choice kernel only for the visited state
        ck_stage2[s] = (1 - alpha_ck) * ck_stage2[s]
        ck_stage2[s, a2] += alpha_ck

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid MB/MF with Decay and Reward-Dependent Perseveration.
    Unchosen options decay towards 0.5. Perseveration at Stage 1 depends on 
    whether the previous trial was rewarded (p_rew) or unrewarded (p_unrew),
    distinguishing between win-stay and lose-stay tendencies.
    
    Parameters:
    alpha: [0, 1] - learning rate
    beta_1: [0, 10] - stage 1 inverse temperature
    beta_2: [0, 10] - stage 2 inverse temperature
    w: [0, 1] - model-based weight
    lam: [0, 1] - eligibility trace
    decay: [0, 1] - decay rate for unchosen options
    p_rew: [-10, 10] - perseveration bonus if previous trial was rewarded
    p_unrew: [-10, 10] - perseveration bonus if previous trial was unrewarded
    """
    alpha, beta_1, beta_2, w, lam, decay, p_rew, p_unrew = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5

    for trial in range(n_trials):
        a1 = action_1[trial]
        s = state[trial]
        a2 = action_2[trial]
        r = reward[trial]

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net

        if trial > 0:
            if reward[trial - 1] == 1:
                logits_1[action_1[trial - 1]] += p_rew
            else:
                logits_1[action_1[trial - 1]] += p_unrew
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[a1]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[s]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        delta_stage1 = q_stage2_mf[s, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[s, a2]
        
        # Q-value updating
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[s, a2] += alpha * delta_stage2

        # Decay for unchosen options
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        for s_idx in range(2):
            for a_idx in range(2):
                if not (s_idx == s and a_idx == a2):
                    q_stage2_mf[s_idx, a_idx] += decay * (0.5 - q_stage2_mf[s_idx, a_idx])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
