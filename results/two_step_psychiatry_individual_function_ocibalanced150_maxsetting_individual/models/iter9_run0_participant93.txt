Here are three new cognitive models that build upon the best model so far by introducing new mechanisms that may better capture the participant's behavior. 

### Model 1: Loss Penalty Model
Research suggests that some individuals process the omission of a reward not just as a zero, but as an active punishment. This model introduces a `loss_penalty` parameter that replaces 0 rewards with a negative value, allowing the model to penalize unrewarded actions more heavily.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates, separate inverse temperatures for MB and MF systems,
    and a loss penalty parameter for unrewarded trials.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta_mb: inverse temperature for model-based Q-values [0,10]
    beta_mf: inverse temperature for model-free Q-values [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    lam: eligibility trace [0,1]
    pi: choice stickiness [0,5]
    loss_penalty: penalty applied when reward is 0 [0,5]
    """
    import numpy as np
    alpha_pos, alpha_neg, beta_mb, beta_mf, beta_2, lam, pi, loss_penalty = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi * stickiness
        q1_vals -= np.max(q1_vals) 
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        q2_vals = beta_2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Apply loss penalty for 0 rewards
        eff_reward = reward[trial] if reward[trial] > 0 else -loss_penalty

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        delta_stage2 = eff_reward - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Choice Trace Decay Model
Instead of a simple 1-trial "stickiness" parameter, this model assumes that past choices leave a trace that decays over time. This acts as a leaky accumulator for perseveration, representing a longer-term habituation toward specific stage 1 choices.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates, separate inverse temperatures for MB and MF systems,
    and a choice trace mechanism that decays over time for stage 1 stickiness.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta_mb: inverse temperature for model-based Q-values [0,10]
    beta_mf: inverse temperature for model-free Q-values [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    lam: eligibility trace [0,1]
    pi: choice stickiness weight [0,5]
    choice_decay: decay rate of the choice trace [0,1]
    """
    import numpy as np
    alpha_pos, alpha_neg, beta_mb, beta_mf, beta_2, lam, pi, choice_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
            
        q1_vals = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi * choice_trace
        q1_vals -= np.max(q1_vals) 
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        q2_vals = beta_2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam * delta_stage2
        
        # Update choice trace
        choice_trace *= choice_decay
        choice_trace[action_1[trial]] += 1.0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Eligibility Trace Model
While the best model so far has asymmetric learning rates, it assumes the eligibility trace is uniform. This model proposes that positive prediction errors might back-propagate differently to the first stage than negative prediction errors, introducing `lam_pos` and `lam_neg`.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid model with asymmetric learning rates, separate inverse temperatures for MB and MF systems,
    and asymmetric eligibility traces (lam_pos, lam_neg) based on the sign of the stage 2 prediction error.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0,1]
    alpha_neg: learning rate for negative prediction errors [0,1]
    beta_mb: inverse temperature for model-based Q-values [0,10]
    beta_mf: inverse temperature for model-free Q-values [0,10]
    beta_2: inverse temperature for stage 2 [0,10]
    lam_pos: eligibility trace for positive stage 2 prediction errors [0,1]
    lam_neg: eligibility trace for negative stage 2 prediction errors [0,1]
    pi: choice stickiness [0,5]
    """
    import numpy as np
    alpha_pos, alpha_neg, beta_mb, beta_mf, beta_2, lam_pos, lam_neg, pi = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        if action_1[trial] == -1 or state[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        stickiness = np.zeros(2)
        if last_action != -1:
            stickiness[last_action] = 1.0
            
        q1_vals = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pi * stickiness
        q1_vals -= np.max(q1_vals) 
        exp_q1 = np.exp(q1_vals)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]

        state_idx = state[trial]
        q2_vals = beta_2 * q_stage2_mf[state_idx]
        q2_vals -= np.max(q2_vals)
        exp_q2 = np.exp(q2_vals)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha_stage1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha_stage1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha_stage2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha_stage2 * delta_stage2

        lam_current = lam_pos if delta_stage2 > 0 else lam_neg
        q_stage1_mf[action_1[trial]] += alpha_stage2 * lam_current * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
