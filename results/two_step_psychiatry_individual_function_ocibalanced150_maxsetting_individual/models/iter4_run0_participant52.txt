def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with separate learning rates for stage 1 and stage 2, value decay for unchosen options,
    separate inverse temperatures, and choice stickiness at stage 1.
    
    Bounds:
    alpha1: [0,1] - learning rate for stage 1
    alpha2: [0,1] - learning rate for stage 2
    decay: [0,1] - decay rate for unchosen options
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - model-based weight
    lam: [0,1] - eligibility trace parameter
    st: [-5,5] - stickiness parameter for previous stage 1 choice
    """
    alpha1, alpha2, decay, beta1, beta2, w, lam, st = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus[last_action_1] = st
            
        exp_q1 = np.exp(beta1 * (q_net + stickiness_bonus - np.max(q_net + stickiness_bonus)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] - np.max(q_stage2_mf[state_idx])))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] += decay * (0.5 - q_stage1_mf[unchosen_action_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Two-step model with value decay for unchosen options, separate inverse temperatures,
    and choice stickiness at BOTH stage 1 and stage 2.
    
    Bounds:
    alpha: [0,1] - learning rate
    decay: [0,1] - decay rate for unchosen options
    beta1: [0,10] - inverse temperature for stage 1
    beta2: [0,10] - inverse temperature for stage 2
    w: [0,1] - model-based weight
    lam: [0,1] - eligibility trace parameter
    st1: [-5,5] - stickiness parameter for previous stage 1 choice
    st2: [-5,5] - stickiness parameter for previous stage 2 choice (state-specific)
    """
    alpha, decay, beta1, beta2, w, lam, st1, st2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.full(2, 0.5)
    q_stage2_mf = np.full((2, 2), 0.5)
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) # Track last stage 2 action per state

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        stickiness_bonus_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_bonus_1[last_action_1] = st1
            
        exp_q1 = np.exp(beta1 * (q_net + stickiness_bonus_1 - np.max(q_net + stickiness_bonus_1)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        stickiness_bonus_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_bonus_2[last_action_2[state_idx]] = st2

        exp_q2 = np.exp(beta2 * (q_stage2_mf[state_idx] + stickiness_bonus_2 - np.max(q_stage2_mf[state_idx] + stickiness_bonus_2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        unchosen_action_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_action_1] += decay * (0.5 - q_stage1_mf[unchosen_action_1])
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
