import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw et al. (2011) with transition matrix learning.
    The participant may dynamically update their belief about the transition structure 
    between spaceships and planets based on experienced transitions.
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1 choice stochasticity.
    beta_2: [0, 10] - Inverse temperature for stage 2 choice stochasticity.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    stickiness_1: [-5, 5] - Tendency to repeat the previous first-stage choice.
    stickiness_2: [-5, 5] - Tendency to repeat the previous second-stage choice in the same state.
    alpha_tr: [0, 1] - Learning rate for updating the transition matrix.
    """
    learning_rate, beta_1, beta_2, w, lam, stickiness_1, stickiness_2, alpha_tr = model_parameters
    n_trials = len(action_1)
  
    # Initialize with instructed probabilities, but allow learning
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Transition matrix updating
        transition_matrix[action_1[trial], state_idx] += alpha_tr * (1.0 - transition_matrix[action_1[trial], state_idx])
        transition_matrix[action_1[trial], 1 - state_idx] += alpha_tr * (0.0 - transition_matrix[action_1[trial], 1 - state_idx])

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Incorporates differential stickiness for stage 1 based on previous reward, 
    plus Q-value decay for unchosen options. Explains behavior where switching 
    only occurs after unrewarded trials.
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    decay: [0, 1] - Decay rate for unchosen Q-values toward 0.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    stickiness_1_win: [-5, 5] - Tendency to repeat first-stage choice if previous trial was rewarded.
    stickiness_1_lose: [-5, 5] - Tendency to repeat first-stage choice if previous trial was unrewarded.
    """
    learning_rate, decay, beta_1, beta_2, w, lam, stickiness_1_win, stickiness_1_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if last_action_1 != -1:
            if last_reward == 1:
                logits_1[last_action_1] += stickiness_1_win
            elif last_reward == 0:
                logits_1[last_action_1] += stickiness_1_lose
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1.0 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1.0 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, :] *= (1.0 - decay)
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Includes Q-value decay and an auto-regressive choice trace for stage 1, 
    allowing stickiness to build up over consecutive identical choices, capturing 
    long streaks of perseveration.
    
    Parameters/Bounds:
    learning_rate: [0, 1] - Rate at which prediction errors update Q-values.
    decay: [0, 1] - Decay rate for unchosen Q-values.
    beta_1: [0, 10] - Inverse temperature for stage 1.
    beta_2: [0, 10] - Inverse temperature for stage 2.
    w: [0, 1] - Weighting parameter (1 = purely model-based, 0 = purely model-free).
    lam: [0, 1] - Eligibility trace parameter.
    alpha_c: [0, 1] - Learning rate for the choice trace.
    phi_1: [-5, 5] - Weight of the choice trace on stage 1 logits.
    """
    learning_rate, decay, beta_1, beta_2, w, lam, alpha_c, phi_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    choice_trace = np.zeros(2)

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1.0 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + phi_1 * choice_trace
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        # Decay unchosen Q-values
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1.0 - decay)
        
        unchosen_2 = 1 - action_2[trial]
        q_stage2_mf[state_idx, unchosen_2] *= (1.0 - decay)
        
        unvisited_state = 1 - state_idx
        q_stage2_mf[unvisited_state, :] *= (1.0 - decay)
        
        # Update choice trace
        choice_trace[action_1[trial]] += alpha_c * (1.0 - choice_trace[action_1[trial]])
        choice_trace[unchosen_1] += alpha_c * (0.0 - choice_trace[unchosen_1])

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
