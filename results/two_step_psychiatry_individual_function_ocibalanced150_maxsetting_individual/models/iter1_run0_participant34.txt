def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: MB/MF mixture model with separate stage inverse temperatures.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for both stages.
    beta_1: [0,10] - Inverse temperature for stage 1.
    beta_2: [0,10] - Inverse temperature for stage 2.
    w: [0,1] - Model-based weight.
    pers1: [-5,5] - Perseveration bonus for choice 1.
    """
    learning_rate, beta_1, beta_2, w, pers1 = model_parameters
    beta = beta_1
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta_1 * q_net_1
        if last_choice_1 != -1:
            logits_1[last_choice_1] += pers1
        logits_1 -= np.max(logits_1)
        probs_1_net = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1_net[action_1[trial]]
        
        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        if action_2[trial] != -1:
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        if action_2[trial] != -1:
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: MB/MF mixture model with perseveration on both choice 1 and choice 2.
    
    Parameters:
    learning_rate: [0,1] - Learning rate.
    beta: [0,10] - Inverse temperature for softmax.

    pers1: [-5,5] - Perseveration bonus for choice 1.

    """

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        # policy for the second choice

        logits_1 = beta * q_net_1
        if last_choice_1 != -1:
            logits_1[last_choice_1] += pers1
        logits_1 -= np.max(logits_1)
        probs_1_net = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1_net[action_1[trial]]
        
        if action_2[trial] != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            if last_choice_2[state_idx] != -1:
                logits_2[last_choice_2[state_idx]] += pers2
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]

        if action_2[trial] != -1:
            q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        if action_2[trial] != -1:
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2
            
        last_choice_1 = action_1[trial]
        if action_2[trial] != -1:
            last_choice_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: MB/MF mixture model with separate learning rates for stage 1 and stage 2, eligibility trace, and choice 1 perseveration.
    
    Parameters:
    learning_rate_1: [0,1] - Learning rate for stage 1.
    learning_rate_2: [0,1] - Learning rate for stage 2.
    beta: [0,10] - Inverse temperature for softmax.
    w: [0,1] - Model-based weight.
    pers1: [-5,5] - Perseveration bonus for choice 1.
    lam: [0,1] - Eligibility trace parameter.
    """
    learning_rate_1, learning_rate_2, beta, w, pers1, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_choice_1 = -1

    for trial in range(n_trials):

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        exp_q1 = np.exp(beta * q_stage1_mb)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        # [FILL IN THE POLICY FOR THE SECOND CHOICE]
        q_net_1 = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta * q_net_1
        if last_choice_1 != -1:
            logits_1[last_choice_1] += pers1
        logits_1 -= np.max(logits_1)
        probs_1_net = np.exp(logits_1) / np.sum(np.exp(logits_1))
        p_choice_1[trial] = probs_1_net[action_1[trial]]
        
        if action_2[trial] != -1:
            logits_2 = beta * q_stage2_mf[state_idx]
            logits_2 -= np.max(logits_2)
            probs_2 = np.exp(logits_2) / np.sum(np.exp(logits_2))
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0
  
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 1]
        if action_2[trial] != -1:
            q_stage1_mf[action_1[trial]] += learning_rate_1 * delta_stage1
            
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        # [FILL IN ACTION VALUE UPDATING FOR CHOICE 2]
        if action_2[trial] != -1:
            q_stage2_mf[state_idx, action_2[trial]] += learning_rate_2 * delta_stage2
            # Eligibility trace update for stage 1
            q_stage1_mf[action_1[trial]] += learning_rate_1 * lam * delta_stage2
            
        last_choice_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
