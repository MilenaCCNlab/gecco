Here are 3 new cognitive models that explore alternative mechanisms such as stage-specific learning rates, reward-dependent stickiness (win-stay/lose-shift), and asymmetric learning for positive versus negative prediction errors combined with unchosen option decay.

### Model 1: Stage-Specific Learning Rates
This model proposes that participants might update their values at different rates for Stage 1 (spaceships) and Stage 2 (aliens). It includes an eligibility trace, model-based weight, and choice stickiness.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Separate learning rates for Stage 1 and Stage 2, with eligibility trace, model-based weight, and choice stickiness.
    
    Bounds:
    alpha1: [0,1] - Learning rate for stage 1.
    alpha2: [0,1] - Learning rate for stage 2.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace.
    w: [0,1] - Model-based weight.
    st: [0,10] - Stickiness for previous stage 1 choice.
    """
    alpha1, alpha2, beta1, beta2, lam, w, st = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    
    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta1 * q_net
        if last_action_1 != -1:
            q1_total[last_action_1] += st
            
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1 + alpha1 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        
        last_action_1 = action_1[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Win-Stay / Lose-Shift Stickiness
This model separates the stickiness parameter into two: one for when the previous trial was rewarded (`st_win`) and one for when it was unrewarded (`st_lose`). This captures the tendency of participants to strongly repeat actions if they just won, but behave differently if they lost.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with Win-Stay/Lose-Shift stickiness for Stage 1 choices, eligibility trace, and model-based weight.
    
    Bounds:
    alpha: [0,1] - Learning rate.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace.
    w: [0,1] - Model-based weight.
    st_win: [0,10] - Stickiness for previous stage 1 choice if rewarded.
    st_lose: [0,10] - Stickiness for previous stage 1 choice if unrewarded.
    """
    alpha, beta1, beta2, lam, w, st_win, st_lose = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = -1
    
    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta1 * q_net
        if last_action_1 != -1:
            if last_reward > 0:
                q1_total[last_action_1] += st_win
            else:
                q1_total[last_action_1] += st_lose
                
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]
        
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Asymmetric Learning Rates + Value Decay
This model uses separate learning rates for positive vs. negative prediction errors (to capture optimism/pessimism biases), while retaining the eligibility trace and value decay for unchosen options. 

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with separate learning rates for positive and negative prediction errors, eligibility trace, model-based weight, and decay for unchosen options.
    
    Bounds:
    alpha_pos: [0,1] - Learning rate for positive prediction errors.
    alpha_neg: [0,1] - Learning rate for negative prediction errors.
    beta1: [0,10] - Inverse temperature for stage 1.
    beta2: [0,10] - Inverse temperature for stage 2.
    lam: [0,1] - Eligibility trace.
    w: [0,1] - Model-based weight.
    decay: [0,1] - Decay rate for unchosen option values.
    """
    alpha_pos, alpha_neg, beta1, beta2, lam, w, decay = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    for trial in range(n_trials):
        if action_1[trial] == -1 or action_2[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue
            
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        q1_total = beta1 * q_net
        q1_total -= np.max(q1_total)
        exp_q1 = np.exp(q1_total)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        
        q2_total = beta2 * q_stage2_mf[state_idx]
        q2_total -= np.max(q2_total)
        exp_q2 = np.exp(q2_total)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]
        
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        
        alpha_s1 = alpha_pos if delta_stage1 >= 0 else alpha_neg
        alpha_s2 = alpha_pos if delta_stage2 >= 0 else alpha_neg
        
        q_stage1_mf[action_1[trial]] += alpha_s1 * delta_stage1 + alpha_s2 * lam * delta_stage2
        q_stage2_mf[state_idx, action_2[trial]] += alpha_s2 * delta_stage2
        
        # Decay for unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if s != state_idx or a != action_2[trial]:
                    q_stage2_mf[s, a] *= (1 - decay)
                    
    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
