import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Separate Learning Rates for Stage 1 and Stage 2
    
    This model separates the learning rate for the first-stage and second-stage 
    decisions. This captures the possibility that participants update their 
    model-free values for the spaceships at a different rate than for the aliens.
    
    Parameters:
    alpha1: learning rate for stage 1 [0, 1]
    alpha2: learning rate for stage 2 [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p1: perseveration parameter for stage 1 [-5, 5]
    p2: perseveration parameter for stage 2 [-5, 5]
    """
    alpha1, alpha2, beta1, beta2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2

        # Track previous actions for perseveration
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Asymmetric Learning Rates (Positive/Negative PE) with Two-Stage Perseveration
    
    This model assumes participants might learn differently from better-than-expected 
    outcomes (positive prediction errors) versus worse-than-expected outcomes 
    (negative prediction errors), and includes independent perseveration at both stages.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p1: perseveration parameter for stage 1 [-5, 5]
    p2: perseveration parameter for stage 2 [-5, 5]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, lam, p1, p2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p1
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p2
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates with asymmetric learning rates based on PE sign
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        alpha1_eff = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[action_1[trial]] += alpha1_eff * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        alpha2_eff = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, action_2[trial]] += alpha2_eff * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha2_eff * lam * delta_stage2

        # Track previous actions for perseveration
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Value Decay for Unchosen Options
    
    This model introduces a decay mechanism where the Q-values of unchosen actions 
    decay towards 0 over time. This captures the possibility that participants 
    forget the values of options they have not sampled recently, along with a shared
    perseveration parameter for both stages.
    
    Parameters:
    alpha: learning rate [0, 1]
    decay: decay rate for unchosen options [0, 1]
    beta1: softmax inverse temperature for stage 1 [0, 10]
    beta2: softmax inverse temperature for stage 2 [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace [0, 1]
    p: overall perseveration parameter [-5, 5]
    """
    alpha, decay, beta1, beta2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta1 * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += p
            
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += p
            
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2

        # Decay unchosen options
        unchosen_1 = 1 - action_1[trial]
        q_stage1_mf[unchosen_1] *= (1 - decay)
        
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] *= (1 - decay)

        # Track previous actions for perseveration
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
