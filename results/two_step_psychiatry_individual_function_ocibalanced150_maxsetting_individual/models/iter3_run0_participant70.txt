Here are 3 new cognitive models that explore alternative mechanisms such as action biases, continuous choice traces (habits), and transition-dependent perseveration.

### Model 1: Action Biases for Stage 1 and Stage 2
This model tests whether the participant has an inherent bias towards choosing a specific spaceship (Stage 1) or a specific alien (Stage 2), independent of the expected values. This is particularly relevant if the participant repeatedly selects the same options (like spaceship 1) regardless of rewards.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model with action biases for Stage 1 and Stage 2.
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd: [0,1] (eligibility trace)
    pers: [0,10] (perseveration bonus)
    bias_1: [-5,5] (bias for choosing action 1 in stage 1)
    bias_2: [-5,5] (bias for choosing action 1 in stage 2)
    """
    alpha, beta_1, beta_2, w, lambd, pers, bias_1, bias_2 = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            logits_1[action_1[trial - 1]] += pers
        logits_1[1] += bias_1  # Apply Stage 1 bias to action 1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2[1] += bias_2  # Apply Stage 2 bias to action 1
        
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Continuous Choice Trace (Habit)
Unlike standard perseveration which only looks at the immediately preceding trial, this model keeps a continuous choice trace. The trace builds up if an action is chosen repeatedly and decays when it is not, acting as a dynamic habit strength.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model with a continuous choice trace (dynamic habit) for Stage 1 actions.
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd: [0,1] (eligibility trace)
    pers: [0,10] (weight of the choice trace / habit strength)
    alpha_c: [0,1] (learning/decay rate for the choice trace)
    """
    alpha, beta_1, beta_2, w, lambd, pers, alpha_c = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    choice_trace = np.zeros(2)

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net + pers * choice_trace
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Update choice trace for the next trial
        choice_trace *= (1 - alpha_c)
        choice_trace[action_1[trial]] += alpha_c

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Perseveration
This model evaluates if the participant's tendency to repeat an action depends on the state transition they just experienced. They might be more prone to repeat their choice if the spaceship went to the expected planet (common transition) vs an unexpected planet (rare transition).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model with different perseveration bonuses depending on whether the 
    previous transition was common or rare.
    Bounds:
    alpha: [0,1] (learning rate)
    beta_1: [0,10] (inverse temperature for stage 1)
    beta_2: [0,10] (inverse temperature for stage 2)
    w: [0,1] (model-based weight)
    lambd: [0,1] (eligibility trace)
    pers_c: [0,10] (perseveration bonus after a common transition)
    pers_r: [0,10] (perseveration bonus after a rare transition)
    """
    alpha, beta_1, beta_2, w, lambd, pers_c, pers_r = model_parameters
    n_trials = len(action_1)
    
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5

    for trial in range(n_trials):
        # Stage 1 policy
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta_1 * q_stage1_net
        if trial > 0:
            prev_a = action_1[trial - 1]
            prev_s = state[trial - 1]
            # Spaceship 0 commonly goes to planet 0; Spaceship 1 commonly goes to planet 1
            if prev_a == prev_s:
                logits_1[prev_a] += pers_c
            else:
                logits_1[prev_a] += pers_r
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # Stage 2 policy
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Value updates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1
        
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
