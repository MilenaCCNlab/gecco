Here are 3 new cognitive models that build upon the best model so far while introducing novel mechanisms to capture different aspects of participant behavior.

### Model 1: Subjective Transition Probability
Participants may not internalize the exact 0.7/0.3 transition probabilities. This model introduces a free parameter `trans_prob` to represent the participant's subjective belief about the probability of a common transition, which directly affects their Model-Based calculations.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with subjective transition probability.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    trans_prob: subjective probability of common transition [0.5, 1.0]
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, trans_prob = model_parameters
    n_trials = len(action_1)
  
    # Subjective transition matrix based on the trans_prob parameter
    transition_matrix = np.array([[trans_prob, 1 - trans_prob], [1 - trans_prob, trans_prob]])
    
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # Stage 1 Model-Based Q-values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 Q-values combining MB and MF
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Reward-dependent perseveration
        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose

        # Stage 1 choice probabilities
        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # Stage 2 choice probabilities
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage-Specific Learning Rates with Shared Temperature
This model tests whether participants update their expectations at different speeds for Stage 1 vs Stage 2. To prevent parameter bloat, it collapses the inverse temperature into a single shared `beta` parameter for both stages, maintaining exactly 7 parameters.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with separate learning rates for Stage 1 and Stage 2, 
    and a shared inverse temperature.
    
    Parameters:
    alpha1: learning rate for stage 1 [0,1]
    alpha2: learning rate for stage 2 [0,1]
    beta: shared inverse temperature for both stages [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    """
    alpha1, alpha2, beta, w, lam, pers_win, pers_lose = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # Stage 1 Model-Based Q-values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 Q-values combining MB and MF
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Reward-dependent perseveration
        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose

        # Stage 1 choice probabilities (using shared beta)
        exp_q1 = np.exp(beta * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # Stage 2 choice probabilities (using shared beta)
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating with separate learning rates
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha2 * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha1 * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Free Initial Q-Values (Optimistic/Pessimistic Initialization)
The way individuals behave in early trials is heavily dependent on their initial expectations. By adding a free parameter `q0`, this model can capture differing baseline expectations (e.g., optimistic initialization encouraging more exploration early on).

```python
import numpy as np

def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Hybrid Model-Based and Model-Free learning with reward-dependent perseveration 
    and a free parameter for initial Q-values.
    
    Parameters:
    alpha: learning rate [0,1]
    beta1: inverse temperature for stage 1 [0,10]
    beta2: inverse temperature for stage 2 [0,10]
    w: model-based weight [0,1]
    lam: eligibility trace [0,1]
    pers_win: perseveration bonus if previous trial was rewarded [-5,5]
    pers_lose: perseveration bonus if previous trial was unrewarded [-5,5]
    q0: initial Q-value for all actions [0,1]
    """
    alpha, beta1, beta2, w, lam, pers_win, pers_lose, q0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values with the free parameter q0
    q_stage1_mf = np.full(2, q0)
    q_stage2_mf = np.full((2, 2), q0)
    
    last_action = -1
    last_reward = -1

    for trial in range(n_trials):

        # Stage 1 Model-Based Q-values
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Net Stage 1 Q-values combining MB and MF
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Reward-dependent perseveration
        if last_action >= 0:
            if last_reward == 1:
                q_net[last_action] += pers_win
            elif last_reward == 0:
                q_net[last_action] += pers_lose

        # Stage 1 choice probabilities
        exp_q1 = np.exp(beta1 * (q_net - np.max(q_net)))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]
        last_action = action_1[trial]

        # Stage 2 choice probabilities
        q2 = q_stage2_mf[state_idx]
        exp_q2 = np.exp(beta2 * (q2 - np.max(q2)))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
        q_stage1_mf[action_1[trial]] += alpha * lam * delta_stage2
        
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
