Here are 3 new cognitive models that build upon the best-performing model by integrating separate model-based (MB) and model-free (MF) inverse temperatures with new mechanisms for learning and perseveration, while keeping the total parameter count at 8.

### Model 1: Separate Learning Rates for Stage 1 and Stage 2
This model extends the best model by allowing different learning rates for the first and second stages. Participants may update the value of the spaceships (Stage 1) at a different speed than the values of the aliens (Stage 2). 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Daw Two-Step Model with Separate MB and MF Inverse Temperatures, 
    and separate learning rates for Stage 1 and Stage 2.
    
    Bounds:
    alpha_1: [0,1] - Learning rate for stage 1.
    alpha_2: [0,1] - Learning rate for stage 2.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.
    beta_mb: [0,10] - Inverse temperature for model-based Q-values at stage 1.
    beta_mf: [0,10] - Inverse temperature for model-free Q-values at stage 1.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    lam: [0,1] - Eligibility trace parameter.
    p: [0,10] - Choice stickiness (perseveration) for the first-stage choice.
    """
    alpha_1, alpha_2, decay, beta_mb, beta_mf, beta_2, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        stickiness = np.zeros(2)
        if last_action_1 != -1:
            stickiness[last_action_1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p * stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += alpha_1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += alpha_2 * delta_stage2

        q_stage1_mf[action_1[trial]] += alpha_1 * lam * delta_stage2

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
        
        last_action_1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Two-Stage Stickiness
Participants may exhibit perseveration not only at the first stage (spaceships) but also at the second stage (aliens). This model introduces a second stickiness parameter `p_2` to capture the tendency to repeat the same alien choice when revisiting a planet.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Daw Two-Step Model with Separate MB and MF Inverse Temperatures, 
    and stickiness for both stages.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen actions.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.
    beta_mb: [0,10] - Inverse temperature for model-based Q-values at stage 1.
    beta_mf: [0,10] - Inverse temperature for model-free Q-values at stage 1.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    lam: [0,1] - Eligibility trace parameter.
    p_1: [0,10] - Choice stickiness for the first-stage choice.
    p_2: [0,10] - Choice stickiness for the second-stage choice.
    """
    learning_rate, decay, beta_mb, beta_mf, beta_2, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_action_2 = np.full(2, -1) 

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = 1.0
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + p_1 * stickiness_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = 1.0

        logits_2 = beta_2 * q_stage2_mf[state_idx, :] + p_2 * stickiness_2
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Transition-Dependent Stickiness
Often, participants' tendency to repeat a Stage 1 choice depends on whether they experienced a common or rare transition on the previous trial (e.g., they might be more likely to switch after a rare transition). This model splits the stickiness parameter into `p_common` and `p_rare`.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Daw Two-Step Model with Separate MB and MF Inverse Temperatures, 
    and transition-dependent stickiness.
    
    Bounds:
    learning_rate: [0,1] - Learning rate for chosen actions.
    decay: [0,1] - Decay rate for unchosen Q-values toward 0.5.
    beta_mb: [0,10] - Inverse temperature for model-based Q-values at stage 1.
    beta_mf: [0,10] - Inverse temperature for model-free Q-values at stage 1.
    beta_2: [0,10] - Inverse temperature for second-stage choice.
    lam: [0,1] - Eligibility trace parameter.
    p_common: [0,10] - Choice stickiness for the first-stage choice following a common transition.
    p_rare: [0,10] - Choice stickiness for the first-stage choice following a rare transition.
    """
    learning_rate, decay, beta_mb, beta_mf, beta_2, lam, p_common, p_rare = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.zeros(2) + 0.5
    q_stage2_mf = np.zeros((2, 2)) + 0.5
    
    last_action_1 = -1
    last_trans_was_common = True

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        stickiness = np.zeros(2)
        if last_action_1 != -1:
            if last_trans_was_common:
                stickiness[last_action_1] = p_common
            else:
                stickiness[last_action_1] = p_rare
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + stickiness
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx, :]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        q_stage2_mf[state_idx, action_2[trial]] += learning_rate * delta_stage2

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2

        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == action_2[trial]):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])

        for a in range(2):
            if a != action_1[trial]:
                q_stage1_mf[a] += decay * (0.5 - q_stage1_mf[a])
        
        last_action_1 = action_1[trial]
        
        if action_1[trial] == state_idx:
            last_trans_was_common = True
        else:
            last_trans_was_common = False

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
