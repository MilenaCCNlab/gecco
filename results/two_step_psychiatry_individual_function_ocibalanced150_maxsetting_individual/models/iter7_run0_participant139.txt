### Model 1: Separate Learning Rates, Betas, Decay, and Two-Stage Stickiness
This model builds on your best model by adding choice stickiness parameters for both stage 1 (`stick1`) and stage 2 (`stick2`). Participants often exhibit perseveration (or alternation) tendencies independent of the expected values, which can be captured by these parameters.

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model 1: Separate learning rates, separate betas, w, decay, stick1, and stick2.
    Bounds:
    alpha1: [0, 1]
    alpha2: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    decay: [0, 1]
    stick1: [-5, 5]
    stick2: [-5, 5]
    """
    alpha1, alpha2, beta1, beta2, w, decay, stick1, stick2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_a1 = -1
    last_a2 = -1

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_net
        if last_a1 != -1:
            logits_1[last_a1] += stick1
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        if last_a2 != -1:
            logits_2[last_a2] += stick2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
  
        unchosen_1 = 1 - a1
        unchosen_2 = 1 - a2

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha1 * delta_stage1
        q_stage1_mf[unchosen_1] *= (1 - decay)

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)

        # decay for the unvisited state
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1 - decay)
        q_stage2_mf[other_state, 1] *= (1 - decay)

        # stage 1 value updated by stage 2 PE (eligibility trace = 1)
        q_stage1_mf[a1] += alpha1 * delta_stage2
        
        last_a1 = a1
        last_a2 = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Asymmetric Learning Rates, Betas, Decay, and Eligibility Trace
This model tests whether participants learn differently from positive versus negative prediction errors (PEs). It also includes an explicit eligibility trace (`lam`), allowing the stage 2 PE to update the stage 1 value by a fractional amount rather than fully.

```python
import numpy as np

def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model 2: Asymmetric learning rates, separate betas, w, decay, and eligibility trace lam.
    Bounds:
    alpha_pos: [0, 1]
    alpha_neg: [0, 1]
    beta1: [0, 10]
    beta2: [0, 10]
    w: [0, 1]
    decay: [0, 1]
    lam: [0, 1]
    """
    alpha_pos, alpha_neg, beta1, beta2, w, decay, lam = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))

    for trial in range(n_trials):
        if action_1[trial] == -1:
            p_choice_1[trial] = 1.0
            p_choice_2[trial] = 1.0
            continue

        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        logits_1 = beta1 * q_net
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]

        # policy for the second choice
        logits_2 = beta2 * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        
        a2 = action_2[trial]
        p_choice_2[trial] = probs_2[a2]
  
        unchosen_1 = 1 - a1
        unchosen_2 = 1 - a2

        # action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha_1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha_1 * delta_stage1
        q_stage1_mf[unchosen_1] *= (1 - decay)

        # action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, a2]
        alpha_2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2
        q_stage2_mf[state_idx, unchosen_2] *= (1 - decay)

        # decay for the unvisited state
        other_state = 1 - state_idx
        q_stage2_mf[other_state, 0] *= (1 - decay)
        q_stage2_mf[other_state, 1] *= (1 - decay)

        # stage 1 value updated by stage 2 PE (with eligibility trace lam)
        q_stage1_mf[a1] += alpha_2 * lam * delta_stage2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Separate Learning Rates, Betas, Eligibility Trace, Decay, and Stage 1 Stickiness
