import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Asymmetric learning rates for positive and negative RPEs, 
    with separate choice stickiness for Stage 1 and Stage 2.
    This model captures the possibility that the participant updates values 
    differently when receiving a reward versus no reward, while also accounting 
    for their strong tendency to repeat choices at both stages.
    
    Bounds:
    lr_pos: [0, 1] - Learning rate for positive prediction errors
    lr_neg: [0, 1] - Learning rate for negative prediction errors
    beta: [0, 10] - Inverse temperature for softmax
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    rho_1: [0, 5] - Stage 1 choice stickiness
    rho_2: [0, 5] - Stage 2 choice stickiness
    """
    lr_pos, lr_neg, beta, w, lam, rho_1, rho_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        stickiness_1 = np.zeros(2)
        if last_action_1 != -1:
            stickiness_1[last_action_1] = rho_1
            
        exp_q1 = np.exp(beta * (q_stage1_net + stickiness_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        # policy for the second choice
        stickiness_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            stickiness_2[last_action_2[state_idx]] = rho_2

        exp_q2 = np.exp(beta * (q_stage2_mf[state_idx] + stickiness_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        # Action value updating for choice 1
        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr_1 = lr_pos if delta_stage1 >= 0 else lr_neg
        q_stage1_mf[action_1[trial]] += lr_1 * delta_stage1

        # Action value updating for choice 2
        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr_2 = lr_pos if delta_stage2 >= 0 else lr_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr_2 * delta_stage2

        # Eligibility trace update for choice 1 based on stage 2 reward
        q_stage1_mf[action_1[trial]] += lr_1 * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Separate inverse temperatures for Stage 1 and Stage 2, 
    with separate choice stickiness for Stage 1 and Stage 2.
    This model allows for different levels of exploration/exploitation (decision noise)
    between the first step (which requires planning/model-based integration) and the 
    second step (which is purely model-free), alongside independent stickiness.
    
    Bounds:
    learning_rate: [0, 1] - Learning rate
    beta_1: [0, 10] - Inverse temperature for Stage 1
    beta_2: [0, 10] - Inverse temperature for Stage 2
    w: [0, 1] - Model-based weight
    lam: [0, 1] - Eligibility trace
    rho_1: [0, 5] - Stage 1 choice stickiness
    rho_2: [0, 5] - Stage 2 choice stickiness
    """
    learning_rate, beta_1, beta_2, w, lam, rho_1, rho_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):

    return log_loss
