### Model 1: Exponential Choice Trace (Frequency-based Perseveration)
Instead of only remembering the immediate last choice (1-back perseveration), this model maintains an exponentially decaying trace of past choices. This allows the model to capture longer-term habits and frequency-based stickiness, where a participant might prefer a spaceship they have chosen often in the recent past.

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """Model with an exponentially decaying choice trace for stage 1 perseveration.
    Instead of just 1-back perseveration, it tracks the frequency of past choices.
    
    Bounds:
    alpha: [0, 1] - learning rate for Q-values
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0.5)
    beta_mb: [0, 10] - inverse temperature for model-based Q-values at stage 1
    beta_mf: [0, 10] - inverse temperature for model-free Q-values at stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice perseveration weight for stage 1
    trace_decay: [0, 1] - decay rate of the choice trace (0 = full memory, 1 = only last choice)
    """
    alpha, decay, beta_mb, beta_mf, beta_2, lambd, pers_1, trace_decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    choice_trace_1 = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_1 * choice_trace_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

            # Decay for unchosen options
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
        # Update choice trace
        choice_trace_1 *= (1 - trace_decay)
        choice_trace_1[action_1[trial]] += 1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Dynamic Transition Learning
Even though the task instructions state that transitions are mostly fixed, participants often do not trust this and update their internal model of the environment as they experience rare transitions. This model introduces a learning rate for the transition matrix itself.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """Model that actively learns the transition matrix over time instead of keeping it fixed.
    
    Bounds:
    alpha: [0, 1] - learning rate for Q-values
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0.5)
    beta_mb: [0, 10] - inverse temperature for model-based Q-values at stage 1
    beta_mf: [0, 10] - inverse temperature for model-free Q-values at stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice perseveration (stickiness) for stage 1
    alpha_trans: [0, 1] - learning rate for updating the transition matrix
    """
    alpha, decay, beta_mb, beta_mf, beta_2, lambd, pers_1, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers_1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus_1
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            # Update the participant's internal transition matrix based on observation
            transition_matrix[action_1[trial], state_idx] += alpha_trans * (1 - transition_matrix[action_1[trial], state_idx])
            transition_matrix[action_1[trial], 1 - state_idx] += alpha_trans * (0 - transition_matrix[action_1[trial], 1 - state_idx])

            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

            # Decay for unchosen options
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
        last_a1 = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: UCB Exploration Bonus (Recency-based Boredom/Novelty)
Instead of just perseverating blindly, humans often exhibit novelty-seeking behaviors. This model tracks the time elapsed since an action was last chosen and adds a UCB-like (Upper Confidence Bound) exploration bonus to the Stage 1 logits. A positive parameter indicates novelty-seeking, while a negative one captures recency-driven familiarity.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """Model with an exploration bonus (UCB-like) for Stage 1 choices.
    It adds a bonus based on the square root of the number of trials since an action was last chosen.
    
    Bounds:
    alpha: [0, 1] - learning rate for Q-values
    decay: [0, 1] - decay rate for unchosen Q-values (towards 0.5)
    beta_mb: [0, 10] - inverse temperature for model-based Q-values at stage 1
    beta_mf: [0, 10] - inverse temperature for model-free Q-values at stage 1
    beta_2: [0, 10] - inverse temperature for stage 2
    lambd: [0, 1] - eligibility trace
    pers_1: [-2, 2] - choice perseveration (stickiness) for stage 1
    explore_1: [-2, 2] - exploration bonus weight for stage 1 (positive = novelty seeking, negative = familiarity)
    """
    alpha, decay, beta_mb, beta_mf, beta_2, lambd, pers_1, explore_1 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)

    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_a1 = -1
    time_since_chosen_1 = np.zeros(2)

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        pers_bonus_1 = np.zeros(2)
        if last_a1 != -1:
            pers_bonus_1[last_a1] = pers_1
            
        logits_1 = beta_mb * q_stage1_mb + beta_mf * q_stage1_mf + pers_bonus_1 + explore_1 * np.sqrt(time_since_chosen_1)
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        state_idx = state[trial]

        # policy for the second choice
        if action_2[trial] != -1:
            logits_2 = beta_2 * q_stage2_mf[state_idx]
            exp_q2 = np.exp(logits_2 - np.max(logits_2))
            probs_2 = exp_q2 / np.sum(exp_q2)
            p_choice_2[trial] = probs_2[action_2[trial]]
        else:
            p_choice_2[trial] = 1.0

        if action_2[trial] != -1:
            # Action value updating
            delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
            q_stage1_mf[action_1[trial]] += alpha * delta_stage1
            
            delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
            q_stage2_mf[state_idx, action_2[trial]] += alpha * delta_stage2
            
            q_stage1_mf[action_1[trial]] += alpha * lambd * delta_stage2

            # Decay for unchosen options
            unchosen_1 = 1 - action_1[trial]
            q_stage1_mf[unchosen_1] += decay * (0.5 - q_stage1_mf[unchosen_1])
            
            unchosen_2 = 1 - action_2[trial]
            q_stage2_mf[state_idx, unchosen_2] += decay * (0.5 - q_stage2_mf[state_idx, unchosen_2])
            
            unvisited_state = 1 - state_idx
            q_stage2_mf[unvisited_state, 0] += decay * (0.5 - q_stage2_mf[unvisited_state, 0])
            q_stage2_mf[unvisited_state, 1] += decay * (0.5 - q_stage2_mf[unvisited_state, 1])
            
        last_a1 = action_1[trial]
        
        # Update exploration trace
        time_since_chosen_1 += 1
        time_since_chosen_1[action_1[trial]] = 0

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
