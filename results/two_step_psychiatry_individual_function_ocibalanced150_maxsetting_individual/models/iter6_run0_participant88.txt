Here are 3 new cognitive models based on the participant's data and the feedback on previously tried combinations. Since the participant demonstrated a massive bias towards spaceship 0 (action 0 in Stage 1) for the vast majority of the task, all three models retain the `bias_action0` parameter from the best model, but explore different mechanisms to explain the remaining variance in behavior.

### Model 1: Asymmetric Learning Rates
This model replaces the `decay_rate` parameter with separate learning rates for positive and negative reward prediction errors. People often learn differently from better-than-expected outcomes compared to worse-than-expected outcomes. 

```python
def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: 7 parameters.
    Uses separate learning rates for positive and negative prediction errors,
    a single beta for both stages, and a bias for action 0 in Stage 1.
    
    Parameters:
    learning_rate_pos: [0,1] - Learning rate for positive reward prediction errors.
    learning_rate_neg: [0,1] - Learning rate for negative reward prediction errors.
    beta: [0,10] - Inverse temperature for both Stage 1 and Stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) systems.
    stickiness: [0,10] - Choice perseveration bonus for Stage 1.
    lam: [0,1] - Eligibility trace for Stage 2 reward updating Stage 1 value.
    bias_action0: [0,10] - Intrinsic bias/bonus for choosing Action 0 in Stage 1.
    """
    learning_rate_pos, learning_rate_neg, beta, w, stickiness, lam, bias_action0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action = -1

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action != -1:
            logits_1[last_action] += stickiness
        logits_1[0] += bias_action0
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        lr1 = learning_rate_pos if delta_stage1 > 0 else learning_rate_neg
        q_stage1_mf[action_1[trial]] += lr1 * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]
        lr2 = learning_rate_pos if delta_stage2 > 0 else learning_rate_neg
        q_stage2_mf[state_idx, action_2[trial]] += lr2 * delta_stage2

        q_stage1_mf[action_1[trial]] += lr2 * lam * delta_stage2
        
        last_action = action_1[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 2: Stage 2 State-Specific Stickiness
This model builds on the previously best-performing model but introduces an extra parameter to capture perseverative behavior (stickiness) in Stage 2. It tracks the last action chosen specifically within each state (planet) and applies a perseveration bonus on subsequent visits to that planet.

```python
def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: 8 parameters.
    Builds on the best model by adding a separate state-specific stickiness parameter for Stage 2.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating action values.
    decay_rate: [0,1] - Rate at which unchosen Stage 2 actions decay toward 0.5.
    beta: [0,10] - Inverse temperature for both Stage 1 and Stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) systems.
    stickiness_stage1: [0,10] - Choice perseveration bonus for Stage 1.
    stickiness_stage2: [0,10] - Choice perseveration bonus for Stage 2 (state-specific).
    lam: [0,1] - Eligibility trace for Stage 2 reward updating Stage 1 value.
    bias_action0: [0,10] - Intrinsic bias/bonus for choosing Action 0 in Stage 1.
    """
    learning_rate, decay_rate, beta, w, stickiness_stage1, stickiness_stage2, lam, bias_action0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            logits_1[last_action_1] += stickiness_stage1
        logits_1[0] += bias_action0
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        if last_action_2[state_idx] != -1:
            logits_2[last_action_2[state_idx]] += stickiness_stage2
            
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += learning_rate * delta_stage2
                else:
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_action_2[state_idx] = action_2[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```

### Model 3: Reward-Dependent Stickiness (Win-Stay / Lose-Shift)
This model splits the standard `stickiness` parameter into two separate parameters based on whether the participant received a reward on the previous trial or not. This effectively captures win-stay vs. lose-shift dynamics on top of the model-based and model-free RL systems.

```python
def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: 8 parameters.
    Builds on the best model by splitting Stage 1 stickiness into two parameters:
    one for when the previous trial was rewarded, and one for when it was unrewarded.
    
    Parameters:
    learning_rate: [0,1] - Learning rate for updating action values.
    decay_rate: [0,1] - Rate at which unchosen Stage 2 actions decay toward 0.5.
    beta: [0,10] - Inverse temperature for both Stage 1 and Stage 2 choices.
    w: [0,1] - Weighting between model-based (1) and model-free (0) systems.
    stickiness_rewarded: [0,10] - Choice perseveration bonus for Stage 1 if previously rewarded.
    stickiness_unrewarded: [-10,10] - Choice perseveration bonus for Stage 1 if previously unrewarded.
    lam: [0,1] - Eligibility trace for Stage 2 reward updating Stage 1 value.
    bias_action0: [0,10] - Intrinsic bias/bonus for choosing Action 0 in Stage 1.
    """
    learning_rate, decay_rate, beta, w, stickiness_rewarded, stickiness_unrewarded, lam, bias_action0 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_stage1_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        logits_1 = beta * q_stage1_net
        if last_action_1 != -1:
            if last_reward == 1:
                logits_1[last_action_1] += stickiness_rewarded
            else:
                logits_1[last_action_1] += stickiness_unrewarded
        logits_1[0] += bias_action0
            
        exp_q1 = np.exp(logits_1 - np.max(logits_1))
        probs_1 = exp_q1 / np.sum(exp_q1)
        p_choice_1[trial] = probs_1[action_1[trial]]
        
        state_idx = state[trial]

        logits_2 = beta * q_stage2_mf[state_idx]
        exp_q2 = np.exp(logits_2 - np.max(logits_2))
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[action_2[trial]]

        delta_stage1 = q_stage2_mf[state_idx, action_2[trial]] - q_stage1_mf[action_1[trial]]
        q_stage1_mf[action_1[trial]] += learning_rate * delta_stage1

        delta_stage2 = reward[trial] - q_stage2_mf[state_idx, action_2[trial]]

        for s in range(2):
            for a in range(2):
                if s == state_idx and a == action_2[trial]:
                    q_stage2_mf[s, a] += learning_rate * delta_stage2
                else:
                    q_stage2_mf[s, a] += decay_rate * (0.5 - q_stage2_mf[s, a])

        q_stage1_mf[action_1[trial]] += learning_rate * lam * delta_stage2
        
        last_action_1 = action_1[trial]
        last_reward = reward[trial]

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
