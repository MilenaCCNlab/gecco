Here are three new cognitive models that explore alternative learning and decision-making mechanisms for the two-step task. These models build upon the hybrid Model-Free and Model-Based reinforcement learning framework (Daw et al., 2011) by introducing distinct variations:

1. **Model 1 (Separate Learning Rates):** Allows learning to occur at different speeds for the first and second stages.
2. **Model 2 (Value Decay / Forgetting):** Captures forgetting by decaying the value of unchosen actions towards a baseline (0.5).
3. **Model 3 (Reward-Dependent Perseveration):** Differentiates stay bias based on previous trial outcomes (Win-Stay / Lose-Shift dynamics).

```python
import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Free and Model-Based RL with separate learning rates.
    This model extends the baseline model by allowing different learning rates 
    for stage 1 and stage 2, reflecting potentially different learning dynamics 
    or attention allocated to the two stages.
    
    Parameters:
    alpha_1: learning rate for stage 1 [0, 1]
    alpha_2: learning rate for stage 2 [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p: choice perseveration (stay bias) for stage 1 [0, 5]
    """
    alpha_1, alpha_2, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):

        # Stage 1 choice probabilities
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * rep
        logits_1 -= np.max(logits_1) # For numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        # Stage 2 choice probabilities
        state_idx = state[trial]
        a2 = action_2[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Action value updating with separate learning rates
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha_1 * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha_2 * delta_stage2

        # Eligibility trace update for stage 1
        q_stage1_mf[a1] += alpha_1 * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid Model-Free and Model-Based RL with value decay (forgetting).
    This model assumes that the values of unchosen actions decay towards a baseline 
    value of 0.5 over time, which captures forgetting in dynamic environments where 
    reward probabilities drift.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p: choice perseveration (stay bias) for stage 1 [0, 5]
    decay: decay rate towards 0.5 for unchosen actions [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize Q-values to 0.5 for decay model
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):

        # Stage 1 choice probabilities
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        # Stage 2 choice probabilities
        state_idx = state[trial]
        a2 = action_2[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        # Update chosen actions
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        # Decay unchosen stage 1 action
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])

        # Decay unchosen stage 2 actions
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid Model-Free and Model-Based RL with reward-dependent perseveration.
    Instead of a single stay bias, this model applies different perseveration biases
    depending on whether the previous trial was rewarded (win-stay) or unrewarded 
    (lose-shift), capturing heuristic choice patterns directly on top of the RL process.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p_reward: choice perseveration if previous trial was rewarded [0, 5]
    p_unreward: choice perseveration if previous trial was unrewarded [-5, 5]
    """
    alpha, beta_1, beta_2, w, lam, p_reward, p_unreward = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_reward = 0

    for trial in range(n_trials):

        # Stage 1 choice probabilities
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        current_p = 0
        if last_action_1 != -1:
            rep[last_action_1] = 1
            if last_reward == 1:
                current_p = p_reward
            else:
                current_p = p_unreward
            
        logits_1 = beta_1 * q_net + current_p * rep
        logits_1 -= np.max(logits_1) # For numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        
        # Stage 2 choice probabilities
        state_idx = state[trial]
        a2 = action_2[trial]

        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Action value updating
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_reward = r

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
```
