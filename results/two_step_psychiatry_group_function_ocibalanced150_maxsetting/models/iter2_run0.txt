import numpy as np

def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Transition Learning Model
    Instead of assuming fixed transition probabilities throughout the task, 
    this model learns the transition matrix dynamically based on the observed 
    state transitions. This captures the possibility that participants update 
    their internal model of the environment's structure.
    
    Parameters:
    alpha: learning rate for action values [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p: stage 1 perseveration [0, 5]
    alpha_trans: learning rate for transition probabilities [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, p, alpha_trans = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * rep
        logits_1 -= np.max(logits_1) # For numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Update transition matrix
        transition_matrix[a1, state_idx] += alpha_trans * (1.0 - transition_matrix[a1, state_idx])
        transition_matrix[a1, 1 - state_idx] -= alpha_trans * transition_matrix[a1, 1 - state_idx]

        # Update action values
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Value Decay (Forgetting) Model
    Action values for unchosen options decay toward 0 on each trial. This captures 
    forgetting of options that are not frequently selected and encourages exploration 
    if an option hasn't been chosen in a while.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p: stage 1 perseveration [0, 5]
    decay: decay rate for unchosen options [0, 1]
    """
    alpha, beta_1, beta_2, w, lam, p, decay = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Update action values
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        # Decay unchosen options
        q_stage1_mf[1 - a1] *= (1.0 - decay)
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] *= (1.0 - decay)
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Dual Perseveration Model
    This model includes perseveration parameters for both stage 1 and stage 2 choices,
    capturing an independent tendency to repeat state-action pairs at both levels 
    of the task hierarchy.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p_1: perseveration for stage 1 [0, 5]
    p_2: perseveration for stage 2 [0, 5]
    """
    alpha, beta_1, beta_2, w, lam, p_1, p_2 = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1
    last_action_2 = np.array([-1, -1])

    for trial in range(n_trials):
        # policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2

        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf

        rep_1 = np.zeros(2)
        if last_action_1 != -1:
            rep_1[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p_1 * rep_1
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]

        state_idx = state[trial]
        a2 = action_2[trial]

        # policy for the second choice
        rep_2 = np.zeros(2)
        if last_action_2[state_idx] != -1:
            rep_2[last_action_2[state_idx]] = 1

        logits_2 = beta_2 * q_stage2_mf[state_idx] + p_2 * rep_2
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]

        r = reward[trial]

        # Update action values
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1

        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2

        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1
        last_action_2[state_idx] = a2

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
