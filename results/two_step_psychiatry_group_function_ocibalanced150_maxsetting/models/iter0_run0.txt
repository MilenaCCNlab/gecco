def cognitive_model1(action_1, state, action_2, reward, model_parameters):
    """
    Model 1: Hybrid Model-Free and Model-Based reinforcement learning model (Daw et al., 2011).
    This model integrates both model-free and model-based learning strategies. It updates
    stage 1 values using an eligibility trace from stage 2 rewards and includes a choice 
    perseveration (stay bias) parameter.
    
    Parameters:
    alpha: learning rate [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight (1 = pure MB, 0 = pure MF) [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p: choice perseveration (stay bias) for stage 1 [0, 5]
    """
    alpha, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        # Combine model-based and model-free values
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        # Add choice perseveration (stay bias)
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * rep
        logits_1 -= np.max(logits_1) # For numerical stability
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating
        r = reward[trial]
        
        # Stage 1 MF update
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        q_stage1_mf[a1] += alpha * delta_stage1
        
        # Stage 2 MF update
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Eligibility trace for Stage 1
        q_stage1_mf[a1] += alpha * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model2(action_1, state, action_2, reward, model_parameters):
    """
    Model 2: Hybrid model with asymmetric learning rates for positive and negative prediction errors.
    This model assumes participants update their expectations differently for better-than-expected 
    outcomes compared to worse-than-expected outcomes.
    
    Parameters:
    alpha_pos: learning rate for positive prediction errors [0, 1]
    alpha_neg: learning rate for negative prediction errors [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p: choice perseveration (stay bias) for stage 1 [0, 5]
    """
    alpha_pos, alpha_neg, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    q_stage1_mf = np.zeros(2)
    q_stage2_mf = np.zeros((2, 2))
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating
        r = reward[trial]
        
        # Asymmetric stage 1 update
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        alpha1 = alpha_pos if delta_stage1 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha1 * delta_stage1
        
        # Asymmetric stage 2 update
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        alpha2 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage2_mf[state_idx, a2] += alpha2 * delta_stage2
        
        # Asymmetric eligibility trace update
        alpha3 = alpha_pos if delta_stage2 > 0 else alpha_neg
        q_stage1_mf[a1] += alpha3 * lam * delta_stage2
        
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss


def cognitive_model3(action_1, state, action_2, reward, model_parameters):
    """
    Model 3: Hybrid model with value decay (forgetting) for unchosen actions.
    This model assumes that the values of actions not chosen decay over time towards 
    an initial neutral expectation (0.5), modeling memory decay in dynamic environments.
    
    Parameters:
    alpha: learning rate [0, 1]
    decay: decay rate for unchosen actions towards 0.5 [0, 1]
    beta_1: inverse temperature for stage 1 choices [0, 10]
    beta_2: inverse temperature for stage 2 choices [0, 10]
    w: model-based weight [0, 1]
    lam: eligibility trace for stage 1 updating [0, 1]
    p: choice perseveration (stay bias) for stage 1 [0, 5]
    """
    alpha, decay, beta_1, beta_2, w, lam, p = model_parameters
    n_trials = len(action_1)
  
    transition_matrix = np.array([[0.7, 0.3], [0.3, 0.7]])
    p_choice_1 = np.zeros(n_trials)
    p_choice_2 = np.zeros(n_trials)
    
    # Initialize values at 0.5 (neutral expectation)
    q_stage1_mf = np.ones(2) * 0.5
    q_stage2_mf = np.ones((2, 2)) * 0.5
    
    last_action_1 = -1

    for trial in range(n_trials):
        # Policy for the first choice
        max_q_stage2 = np.max(q_stage2_mf, axis=1)
        q_stage1_mb = transition_matrix @ max_q_stage2
        
        q_net = w * q_stage1_mb + (1 - w) * q_stage1_mf
        
        rep = np.zeros(2)
        if last_action_1 != -1:
            rep[last_action_1] = 1
            
        logits_1 = beta_1 * q_net + p * rep
        logits_1 -= np.max(logits_1)
        exp_q1 = np.exp(logits_1)
        probs_1 = exp_q1 / np.sum(exp_q1)
        
        a1 = action_1[trial]
        p_choice_1[trial] = probs_1[a1]
        state_idx = state[trial]
        a2 = action_2[trial]

        # Policy for the second choice
        logits_2 = beta_2 * q_stage2_mf[state_idx]
        logits_2 -= np.max(logits_2)
        exp_q2 = np.exp(logits_2)
        probs_2 = exp_q2 / np.sum(exp_q2)
        p_choice_2[trial] = probs_2[a2]
  
        # Action value updating
        r = reward[trial]
        
        delta_stage1 = q_stage2_mf[state_idx, a2] - q_stage1_mf[a1]
        delta_stage2 = r - q_stage2_mf[state_idx, a2]
        
        # Update chosen actions
        q_stage1_mf[a1] += alpha * delta_stage1 + alpha * lam * delta_stage2
        q_stage2_mf[state_idx, a2] += alpha * delta_stage2
        
        # Decay unchosen action in stage 1
        unchosen_a1 = 1 - a1
        q_stage1_mf[unchosen_a1] += decay * (0.5 - q_stage1_mf[unchosen_a1])
        
        # Decay unchosen actions in stage 2
        for s in range(2):
            for a in range(2):
                if not (s == state_idx and a == a2):
                    q_stage2_mf[s, a] += decay * (0.5 - q_stage2_mf[s, a])
                    
        last_action_1 = a1

    eps = 1e-10
    log_loss = -(np.sum(np.log(p_choice_1 + eps)) + np.sum(np.log(p_choice_2 + eps)))
    return log_loss
