def cognitive_model(stimulus, actions, rewards, blocks, set_sizes, model_parameters):
    """
    Composed cognitive model: backbone + Outcome-Gated WM Reliance, WM Normalization
    
    Parameters:
    learning_rate: [0, 1] - RL delta-rule learning rate.
    beta: [0, 10] - RL softmax inverse temperature.
    wm_weight: [0, 1] - WM weight in the RL/WM policy mixture.
    outcome_gated_wm_reliance_win_boost: [0, 10] - module parameter
    outcome_gated_wm_reliance_loss_suppression: [0, 10] - module parameter
    outcome_gated_wm_reliance_kappa: [0, 1] - module parameter
    """
    learning_rate, beta, wm_weight, outcome_gated_wm_reliance_win_boost, outcome_gated_wm_reliance_loss_suppression, outcome_gated_wm_reliance_kappa = model_parameters
    nA = 3
    log_loss = 0.0
    eps = 1e-10
    outcome_gated_wm_reliance_g = 0.0

    for b in np.unique(blocks):
        block_mask = blocks == b
        block_states = stimulus[block_mask]
        block_actions = actions[block_mask]
        block_rewards = rewards[block_mask]
        nS = int(set_sizes[block_mask][0])
        q = (1.0 / nA) * np.ones((nS, nA))
        w = (1.0 / nA) * np.ones((nS, nA))
        w_0 = (1.0 / nA) * np.ones((nS, nA))
        outcome_gated_wm_reliance_g = 0.0

        for trial in range(len(block_states)):
            s = int(block_states[trial])
            a = int(block_actions[trial])
            r = float(block_rewards[trial])
            if 0 <= s < nS and 0 <= a < nA:
                outcome_gated_wm_reliance_g = (1.0 - outcome_gated_wm_reliance_kappa) * outcome_gated_wm_reliance_g + outcome_gated_wm_reliance_kappa * (outcome_gated_wm_reliance_win_boost if r > 0.5 else -outcome_gated_wm_reliance_loss_suppression)

                rl_values = q[s]
                logits_rl = (beta) * rl_values
                exp_rl = np.exp(logits_rl - np.max(logits_rl))
                probs_rl = exp_rl / np.sum(exp_rl)
                wm_values = w[s]
                logits_wm = (50.0) * wm_values
                exp_wm = np.exp(logits_wm - np.max(logits_wm))
                probs_wm = exp_wm / np.sum(exp_wm)
                mix = 1.0 / (1.0 + np.exp(-(np.log(np.clip(wm_weight, 1e-12, 1.0 - 1e-12) / (1.0 - np.clip(wm_weight, 1e-12, 1.0 - 1e-12))) + outcome_gated_wm_reliance_g)))
                probs = mix * probs_wm + (1.0 - mix) * probs_rl
                log_loss -= np.log(probs[a] + eps)
                delta = r - q[s, a]
                q[s, a] += learning_rate * delta
                w[s, a] = r
                wm_normalization_row_sum = np.sum(w[s, :])
                if wm_normalization_row_sum > 0:
                    w[s, :] /= wm_normalization_row_sum
                else:
                    w[s, :] = 1.0 / nA

    return log_loss
